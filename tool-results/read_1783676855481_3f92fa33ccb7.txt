   167→   167→function formatTxnDate(iso: string): string {
   168→   168→  if (!iso) return '—'
   169→   169→  const d = new Date(iso)
   170→   170→  if (Number.isNaN(d.getTime())) return iso
   171→   171→  const dd = String(d.getDate()).padStart(2, '0')
   172→   172→  const mm = String(d.getMonth() + 1).padStart(2, '0')
   173→   173→  const yyyy = d.getFullYear()
   174→   174→  return `${dd}/${mm}/${yyyy}`
   175→   175→}
   176→   176→
   177→   177→function titleCaseMode(mode: string): string {
   178→   178→  if (!mode) return 'Bank'
   179→   179→  const lower = mode.toLowerCase()
   180→   180→  if (lower === 'upi') return 'UPI'
   181→   181→  if (lower === 'bank') return 'Net Banking'
   182→   182→  if (lower === 'card') return 'Card'
   183→   183→  if (lower === 'cheque') return 'Cheque'
   184→   184→  if (lower === 'cash') return 'Cash'
   185→   185→  return mode.charAt(0).toUpperCase() + mode.slice(1)
   186→   186→}
