   234→// ─── Portfolio Derivation Helpers ──────────────────────────────────────────────
   235→
   236→/** Compute a ClientPortfolio from an API client by deriving portfolio metrics */
   237→function derivePortfolio(
   238→  client: ApiClient,
   239→  clientFilings: Array<{ returnType: string; period: string; status: string; filedDate?: string; totalTaxableValue: number }>
   240→): ClientPortfolio {
   241→  const hs = client.healthScore;
   242→  const pendingFilings = client._aggregations?.pendingReturns ?? clientFilings.filter(f => f.status !== 'filed').length;
   243→  const taxVolume = clientFilings.reduce((sum, f) => sum + f.totalTaxableValue, 0);
   244→  const matchRate = client._aggregations?.matchPercentage ?? 0;
   245→
   246→  // Derive risk level from health score
   247→  const riskLevel: RiskLevel = hs > 80 ? 'Low' : hs >= 50 ? 'Medium' : 'High';
   248→  const riskDetail = riskLevel === 'High' ? (pendingFilings > 3 ? 'Multiple delays' : 'Filing delays')
   249→    : riskLevel === 'Medium' ? `${pendingFilings} pending filing${pendingFilings !== 1 ? 's' : ''}`
   250→    : undefined;
   251→
   252→  // Derive health breakdown from health score and data
   253→  const gstinValidity = Math.min(100, hs + 3);
   254→  const filingTimeliness = pendingFilings === 0 ? Math.min(100, hs + 5) : Math.max(10, hs - pendingFilings * 8);
   255→  const invoiceAccuracy = Math.min(100, Math.max(10, matchRate > 0 ? matchRate : hs > 80 ? 94 : hs < 50 ? 58 : 78));
   256→
   257→  // Recent filings (last 3 filings)
   258→  const recentFilings = clientFilings.slice(0, 3).map(f => {
   259→    const monthNames = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];
   260→    const [year, month] = f.period.split('-').map(Number);
   261→    const label = `${monthNames[month - 1]} ${year}`;
   262→    return {
   263→      period: label,
   264→      type: f.returnType,
   265→      status: f.status === 'filed' ? 'Filed' : f.filedDate ? 'Filed' : pendingFilings > 2 ? 'Overdue' : 'Pending',
   266→      date: f.filedDate ?? '',
   267→    };
   268→  });
   269→
   270→  // Monthly volume chart — derive from filings' taxable values
   271→  const monthlyVolumeChart: MonthlyVolume[] = [
   272→    { month: 'Dec', amount: Math.round(taxVolume * 0.85) },
   273→    { month: 'Jan', amount: Math.round(taxVolume * 0.92) },
   274→    { month: 'Feb', amount: Math.round(taxVolume * 0.88) },
   275→    { month: 'Mar', amount: Math.round(taxVolume * 1.1) },
   276→    { month: 'Apr', amount: Math.round(taxVolume * 0.95) },
   277→    { month: 'May', amount: taxVolume },
   278→  ];
   279→
   280→  return {
   281→    id: client.id,
   282→    gstin: client.gstin,
   283→    tradeName: client.tradeName,
   284→    legalName: client.legalName ?? undefined,
   285→    state: client.state ?? undefined,
   286→    stateCode: client.stateCode ?? undefined,
   287→    entityType: client.entityType,
   288→    returnPeriod: client.returnPeriod ?? undefined,
   289→    lastFilingDate: client.lastFilingDate ?? undefined,
   290→    status: client.status,
   291→    healthScore: client.healthScore,
   292→    contactEmail: client.contactEmail ?? undefined,
   293→    contactPhone: client.contactPhone ?? undefined,
   294→    createdAt: client.createdAt,
   295→    updatedAt: client.updatedAt,
   296→    _portfolio: {
   297→      monthlyTaxVolume: taxVolume,
   298→      monthlyVolumeChart,
   299→      pendingFilings,
   300→      riskLevel,
   301→      riskDetail,
   302→      healthBreakdown: { gstinValidity, filingTimeliness, invoiceAccuracy },
   303→      recentFilings,
   304→    },
   305→  };
   306→}

   308→// ─── GSTIN Form Validation ──────────────────────────────────────────────────
   309→
   310→function validateGSTINForm(gstin: string): { valid: boolean; error?: string } {
   311→  if (!gstin) return { valid: false, error: 'GSTIN is required' };
   312→  const clean = formatGSTIN(gstin);
   313→  if (clean.length !== 15) return { valid: false, error: 'GSTIN must be 15 characters' };
   314→  if (!validateGSTIN(clean)) return { valid: false, error: 'Invalid GSTIN format (e.g. 27AABCS1234F1ZH)' };
   315→  const stateCode = clean.slice(0, 2);
   316→  const validCodes = INDIAN_STATES.map(s => s.code);
   317→  if (!validCodes.includes(stateCode)) return { valid: false, error: 'Invalid state code in GSTIN' };
   318→  return { valid: true };
   319→}

   326→  return { bg: 'bg-red-50', text: 'text-red-700', bar: 'bg-red-500', stroke: '#ef4444' };
   327→}

   394→function formatVolumeShort(amount: number): string {
   395→  if (amount >= 10000000) return `₹${(amount / 10000000).toFixed(2)} Cr`;
   396→  if (amount >= 100000) return `₹${(amount / 100000).toFixed(1)}L`;
   397→  if (amount >= 1000) return `₹${(amount / 1000).toFixed(1)}K`;
   398→  return `₹${amount}`;
   399→}

   408→function HealthRing({ score, size = 60, strokeWidth = 5, showLabel = true }: { score: number; size?: number; strokeWidth?: number; showLabel?: boolean }) {
   409→  const radius = (size - strokeWidth) / 2;
   410→  const circumference = 2 * Math.PI * radius;
   411→  const offset = circumference - (score / 100) * circumference;
   412→  const color = getHealthColor(score);
   413→
   414→  return (
   415→    <div className="relative inline-flex items-center justify-center" style={{ width: size, height: size }}>
   416→      <svg width={size} height={size} className="-rotate-90">
   417→        <circle
   418→          cx={size / 2}
   419→          cy={size / 2}
   420→          r={radius}
   421→          fill="none"
   422→          stroke="#e2e8f0"
   423→          strokeWidth={strokeWidth}
   424→        />
   425→        <motion.circle
   426→          cx={size / 2}
   427→          cy={size / 2}
   428→          r={radius}
   429→          fill="none"
   430→          stroke={color.stroke}
   431→          strokeWidth={strokeWidth}
   432→          strokeLinecap="round"
   433→          strokeDasharray={circumference}
   434→          initial={{ strokeDashoffset: circumference }}
   435→          animate={{ strokeDashoffset: offset }}
   436→          transition={{ duration: 1, delay: 0.3, ease: 'easeOut' }}
   437→        />
   438→      </svg>
   439→      {showLabel && (
   440→        <div className="absolute inset-0 flex flex-col items-center justify-center">
   441→          <span className={`font-bold leading-none ${size >= 100 ? 'text-2xl' : 'text-sm'} ${color.text}`}>
   442→            {score}
   443→          </span>
   444→          {size >= 100 && (
   445→            <span className="text-[9px] text-muted-foreground mt-0.5">/ 100</span>
   446→          )}
   447→        </div>
   448→      )}
   449→    </div>
   450→  );
   451→}

   466→// ─── Filing Trend Area Chart ────────────────────────────────────────────────────
   467→
   468→function FilingTrendChart({ data }: { data: MonthlyVolume[] }) {
   469→  const width = 260;
   470→  const height = 80;
   471→  const padding = { top: 8, right: 8, bottom: 20, left: 8 };
   472→  const chartW = width - padding.left - padding.right;
   473→  const chartH = height - padding.top - padding.bottom;
   474→  const maxAmount = Math.max(...data.map(d => d.amount), 1);
   475→
   476→  const points = data.map((d, i) => ({
   477→    x: padding.left + (i / (data.length - 1)) * chartW,
   478→    y: padding.top + chartH - (d.amount / maxAmount) * chartH,
   479→    ...d,
   480→  }));
   481→
   482→  const linePath = points.map((p, i) => `${i === 0 ? 'M' : 'L'} ${p.x} ${p.y}`).join(' ');
   483→  const areaPath = `${linePath} L ${points[points.length - 1].x} ${padding.top + chartH} L ${points[0].x} ${padding.top + chartH} Z`;
   484→
   485→  const [hoverIdx, setHoverIdx] = useState<number | null>(null);
   486→
   487→  return (
   488→    <div className="relative">
   489→      <svg width="100%" viewBox={`0 0 ${width} ${height}`} className="overflow-visible">
   490→        {/* Area fill */}
   491→        <motion.path
   492→          d={areaPath}
   493→          fill="url(#areaGradient)"
   494→          initial={{ opacity: 0 }}
   495→          animate={{ opacity: 1 }}
   496→          transition={{ duration: 0.6, delay: 0.3 }}
   497→        />
   498→        {/* Line */}
   499→        <motion.path
   500→          d={linePath}
   501→          fill="none"
   502→          stroke="#10b981"
   503→          strokeWidth={2}
   504→          strokeLinecap="round"
   505→          strokeLinejoin="round"
   506→          initial={{ pathLength: 0 }}
   507→          animate={{ pathLength: 1 }}
   508→          transition={{ duration: 0.8, delay: 0.2 }}
   509→        />
   510→        {/* Dots */}
   511→        {points.map((p, i) => (
   512→          <circle
   513→            key={i}
   514→            cx={p.x}
   515→            cy={p.y}
   516→            r={hoverIdx === i ? 4 : 2.5}
   517→            fill="#10b981"
   518→            stroke="white"
   519→            strokeWidth={2}
   520→            className="transition-all duration-150 cursor-pointer"
   521→            onMouseEnter={() => setHoverIdx(i)}
   522→            onMouseLeave={() => setHoverIdx(null)}
   523→          />
   524→        ))}
   525→        {/* Month labels */}
   526→        {points.map((p, i) => (
   527→          <text
   528→            key={i}
   529→            x={p.x}
   530→            y={height - 4}
   531→            textAnchor="middle"
   532→            className="fill-muted-foreground"
   533→            fontSize="8"
   534→          >
   535→            {data[i].month}
   536→          </text>
   537→        ))}
   538→        <defs>
   539→          <linearGradient id="areaGradient" x1="0" y1="0" x2="0" y2="1">
   540→            <stop offset="0%" stopColor="#10b981" stopOpacity="0.2" />
   541→            <stop offset="100%" stopColor="#10b981" stopOpacity="0.02" />
   542→          </linearGradient>
   543→        </defs>
   544→      </svg>
   545→      {/* Hover tooltip */}
   546→      {hoverIdx !== null && (
