    35→let decisionCounter = 0;
    36→function decisionId(type: DecisionType): string {
    37→  decisionCounter += 1;
    38→  return `ceo-dec-${type}-${Date.now()}-${decisionCounter}`;
    39→}
    40→
    41→function confidenceFrom(dataPoints: number, base = 60): number {
    42→  // More data points → higher confidence (capped at 95 — never 100% certain)
    43→  return Math.min(95, Math.round(base + Math.min(35, dataPoints * 3)));
    44→}
    45→
    46→function priorityFrom(daysLate: number, impact: number): DecisionPriority {
    47→  if (daysLate > 60 || impact > 500000) return 'critical';
    48→  if (daysLate > 30 || impact > 100000) return 'high';
    49→  if (daysLate > 7 || impact > 10000) return 'medium';
    50→  return 'low';
    51→}
    52→
    53→function evidence(source: string, fact: string, value?: number): DecisionEvidence {
    54→  return { source, fact, value };
    55→}

    57→function action(
    58→  label: string,
    59→  description: string,
    60→  agent: string,
    61→  estimatedMinutes: number,
    62→  automated: boolean,
    63→  destructive = false,
    64→): DecisionAction {
    65→  return { label, description, agent, estimatedMinutes, automated, destructive };
    66→}

    72→function buildRecoverPayment(data: CEODataView): ExecutiveDecision | null {
    73→  const overdue = data.cfo.collections.latePayments.filter((l) => l.daysOverdue > 0);
    74→  if (overdue.length === 0) return null;
    75→
    76→  const top = overdue[0];
    77→  const impact = top.outstandingAmount ?? top.invoiceAmount ?? 0;
    78→  if (impact <= 0) return null;
    79→
    80→  const daysLate = top.daysOverdue;
    81→  const confidence = confidenceFrom(overdue.length, 70);
    82→  const priority = priorityFrom(daysLate, impact);
    83→  const { requirement, requiresRole, risk } = resolveApprovalRequirement('recover_payment', impact);
    84→
    85→  return {
    86→    id: decisionId('recover_payment'),
    87→    type: 'recover_payment',
    88→    title: `Recover ${formatINR(impact)} from ${top.clientName ?? 'a client'}`,
    89→    reason: `${top.clientName ?? 'A client'} has an invoice ${daysLate} days overdue (₹${Math.round(impact).toLocaleString('en-IN')}). The longer it stays unpaid, the lower the recovery probability.`,
    90→    financialImpact: impact,
    91→    financialImpactLabel: `Recover ${formatINRFull(impact)} in overdue receivables`,
    92→    confidence,
    93→    priority,
    94→    risk,
    95→    businessImpact: `Immediate cash inflow of ${formatINR(impact)}. Improves runway by ~${Math.round(impact / (data.liveState.burnRate / 30 || 1))} days at current burn.`,
    96→    rollbackPlan: 'If the client disputes, pause follow-up and escalate to a demand letter. Recovery attempts have no financial downside.',
    97→    approvalRequired: requirement,
    98→    requiresRole,
    99→    actions: [
   100→      action('Send reminder email', `Send a polite reminder to ${top.clientName} with the invoice attached.`, 'oracle', 5, true),
   101→      action('Schedule a call', `If no reply in 48 hours, schedule a recovery call with ${top.clientName}.`, 'oracle', 15, false),
   102→      action('Escalate to demand letter', `If still unpaid after 7 days, send a formal demand letter.`, 'cfo_agent', 30, false),
   103→    ],
   104→    relatedEntityType: 'invoice',
   105→    relatedEntityId: top.invoiceId ?? top.invoiceNumber,
   106→    relatedEntityLabel: top.invoiceNumber,
   107→    evidence: [
   108→      evidence('invoice', `Invoice ${top.invoiceNumber} is ${daysLate} days overdue`, impact),
   109→      evidence('collection_engine', `Collection probability: ${top.collectionProbabilityPct}%`, top.collectionProbabilityPct),
   110→      evidence('cfo_phase1', `Total overdue receivables: ${formatINR(data.cfo.collections.overdueAmount)}`, data.cfo.collections.overdueAmount),
   111→      evidence('cfo_phase1', `Total outstanding: ${formatINR(data.cfo.collections.totalOutstanding)}`, data.cfo.collections.totalOutstanding),
   112→    ],
   113→    status: 'pending',
   114→    createdAt: new Date().toISOString(),
   115→    expiresAt: new Date(Date.now() + 7 * 24 * 60 * 60 * 1000).toISOString(),
   116→  };
   117→}

   119→function buildRemindClient(data: CEODataView): ExecutiveDecision | null {
   120→  // Due-soon invoices (not yet overdue, but due within 3 days)
   121→  const dueSoon = data.raw.invoices.filter((i) => {
   122→    if (i.paymentStatus === 'paid' || !i.dueDate) return false;
   123→    const due = new Date(i.dueDate);
   124→    const diff = (due.getTime() - Date.now()) / (24 * 60 * 60 * 1000);
   125→    return diff >= 0 && diff <= 3;
   126→  });
   127→  if (dueSoon.length === 0) return null;
   128→
   129→  const inv = dueSoon[0];
   130→  const impact = inv.balanceAmount ?? inv.totalAmount ?? 0;
   131→  // Note: Invoice table uses balanceAmount (Phase 8 Invoice Cloud™). CollectionRow uses outstandingAmount.
   132→  const client = data.raw.clients.find((c) => c.id === inv.clientId);
   133→  const { requirement, requiresRole, risk } = resolveApprovalRequirement('remind_client', impact);
   134→
   135→  return {
   136→    id: decisionId('remind_client'),
   137→    type: 'remind_client',
   138→    title: `Send payment reminder to ${client?.tradeName ?? 'client'}`,
   139→    reason: `Invoice ${inv.invoiceNumber} (${formatINR(impact)}) is due in ≤3 days. A friendly reminder now maximises on-time payment.`,
   140→    financialImpact: impact,
   141→    financialImpactLabel: `Protect ${formatINRFull(impact)} from going overdue`,
   142→    confidence: confidenceFrom(dueSoon.length, 75),
   143→    priority: 'medium',
   144→    risk,
   145→    businessImpact: 'Reduces Days Sales Outstanding and prevents future overdue-chase cycles.',
   146→    rollbackPlan: 'Reminders are non-destructive. If the client has already paid, simply acknowledge.',
   147→    approvalRequired: requirement,
   148→    requiresRole,
   149→    actions: [
   150→      action('Send reminder email', `Send a polite reminder with invoice ${inv.invoiceNumber} attached.`, 'oracle', 5, true),
   151→    ],
   152→    relatedEntityType: 'invoice',
   153→    relatedEntityId: inv.id,
   154→    relatedEntityLabel: inv.invoiceNumber,
   155→    evidence: [
   156→      evidence('invoice', `Invoice ${inv.invoiceNumber} due ${inv.dueDate}`, impact),
   157→    ],
   158→    status: 'pending',
   159→    createdAt: new Date().toISOString(),
   160→    expiresAt: new Date(Date.now() + 3 * 24 * 60 * 60 * 1000).toISOString(),
   161→  };
   162→}

   368→function buildReduceExpenses(data: CEODataView): ExecutiveDecision | null {

   370→  const cats = data.cfo.expenses.byCategory
   371→    .filter((c) => !['payroll', 'gst', 'rent'].includes(c.category))
   372→    .sort((a, b) => b.amount - a.amount);
   373→  if (cats.length === 0) return null;
   374→
   375→  const top = cats[0];
   376→  const cutTarget = top.amount * 0.20; // 20% reduction
   377→  if (cutTarget < 5000) return null;
   378→
   379→  const { requirement, requiresRole, risk } = resolveApprovalRequirement('reduce_expenses', cutTarget);
   380→
   381→  return {
   382→    id: decisionId('reduce_expenses'),
   383→    type: 'reduce_expenses',
   384→    title: `Cut ${formatINR(cutTarget)}/mo from ${top.label}`,
   385→    reason: `${top.label} is your largest discretionary expense at ${formatINR(top.amount)}/mo (MoM ${top.momChangePct >= 0 ? '+' : ''}${top.momChangePct.toFixed(1)}%). A 20% trim saves ${formatINR(cutTarget)}/mo.`,
   386→    financialImpact: cutTarget * 12, // annualised

   411→function buildPauseMarketing(data: CEODataView): ExecutiveDecision | null {
   412→  const marketing = data.cfo.expenses.byCategory.find((c) => c.category === 'marketing');
   413→  if (!marketing || marketing.amount < 5000) return null;
   414→
   415→  // Pause only if runway is short or marketing > 15% of expenses
   416→  const isRunwayShort = data.liveState.runwayDays < 90 && data.liveState.runwayDays > 0;
   417→  const isMarketingTooHigh = marketing.amount / Math.max(1, data.cfo.expenses.totalThisMonth) > 0.15;
   418→  if (!isRunwayShort && !isMarketingTooHigh) return null;
   419→
   420→  const { requirement, requiresRole, risk } = resolveApprovalRequirement('pause_marketing', marketing.amount);
   421→
   422→  return {
   423→    id: decisionId('pause_marketing'),
   424→    type: 'pause_marketing',
   425→    title: `Pause marketing spend (${formatINR(marketing.amount)}/mo)`,
   426→    reason: `Marketing is ${formatINR(marketing.amount)}/mo (${((marketing.amount / Math.max(1, data.cfo.expenses.totalThisMonth)) * 100).toFixed(1)}% of expenses) and runway is ${data.liveState.runwayDays} days. Pausing protects cash.`,

   493→function buildReplyCustomer(data: CEODataView): ExecutiveDecision | null {
   494→  // Look at un-acknowledged WhatsApp / email messages
   495→  const unhandled = data.raw.syncedRecords.filter(
   496→    (r) => r.sourceType === 'whatsapp_msg' || r.sourceType === 'email',
   497→  ).filter((r) => {
   498→    if (!r.date) return false;
   499→    const ageHours = (Date.now() - new Date(r.date).getTime()) / (60 * 60 * 1000);
   500→    return ageHours > 4; // waiting > 4 hours
   501→  });
   502→  if (unhandled.length === 0) return null;
   503→
   504→  const msg = unhandled[0];
   505→  const { requirement, requiresRole, risk } = resolveApprovalRequirement('reply_customer', 0);
   506→  const ageHours = msg.date ? Math.round((Date.now() - new Date(msg.date).getTime()) / (60 * 60 * 1000)) : 0;
   507→
   508→  return {
   509→    id: decisionId('reply_customer'),
   510→    type: 'reply_customer',
   511→    title: `Reply to ${msg.sourceType === 'email' ? 'email' : 'WhatsApp message'}`,
   512→    reason: `A customer message has been waiting ${ageHours} hours. Slow responses hurt client retention.`,
   513→    financialImpact: 0,
   514→    financialImpactLabel: 'Protects client relationship and retention',
   515→    confidence: confidenceFrom(unhandled.length, 65),
   516→    priority: 'medium',
   517→    risk,
   518→    businessImpact: 'Customer satisfaction improves → reduces churn risk and supports revenue continuity.',
   519→    rollbackPlan: 'A reply can be edited or recalled. No financial downside.',
   520→    approvalRequired: requirement,
   521→    requiresRole,
   522→    actions: [
   523→      action('Draft reply', `Use Oracle to draft a context-aware reply to: ${msg.title ?? 'the message'}.`, 'oracle', 5, true),
   524→      action('Send reply', 'Send the reply via the same channel.', 'oracle', 2, true),
   525→    ],
   526→    relatedEntityType: 'communication',
   527→    relatedEntityId: msg.id,
   528→    relatedEntityLabel: msg.title ?? msg.sourceType,
   529→    evidence: [
   530→      evidence('communication', `Unhandled messages: ${unhandled.length}`, unhandled.length),
   531→      evidence('communication', `Oldest waiting: ${ageHours}h`, ageHours),
   532→    ],
   533→    status: 'pending',
   534→    createdAt: new Date().toISOString(),
   535→    expiresAt: new Date(Date.now() + 24 * 60 * 60 * 1000).toISOString(),
   536→  };
   537→}

   539→function buildOptimizeCash(data: CEODataView): ExecutiveDecision | null {
   540→  // Suggest moving idle cash from savings → OD account if there's an OD with negative balance
   541→  const accounts = data.twin.state.bankAccounts;
   542→  if (accounts.length < 2) return null;
   543→
   544→  const idleAccounts = accounts.filter((a) => a.type === 'savings' && a.balance > 200000);
   545→  const odAccounts = accounts.filter((a) => (a.type === 'od' || a.type === 'cc') && a.balance < 0);
   546→  if (idleAccounts.length === 0 || odAccounts.length === 0) return null;
   547→
   548→  const idle = idleAccounts[0];
   549→  const od = odAccounts[0];
   550→  const moveAmount = Math.min(idle.balance - 100000, Math.abs(od.balance));
   551→
   552→  const { requirement, requiresRole, risk } = resolveApprovalRequirement('optimize_cash', moveAmount);
   553→
   554→  return {
   555→    id: decisionId('optimize_cash'),
   556→    type: 'optimize_cash',
   557→    title: `Move ${formatINR(moveAmount)} from ${idle.bank} to ${od.bank}`,
   558→    reason: `${idle.bank} has ${formatINR(idle.balance)} idle (savings). ${od.bank} OD is at ${formatINR(od.balance)} (costing ~12% interest). Sweeping saves interest cost.`,
   559→    financialImpact: Math.abs(od.balance) * 0.12 / 12, // monthly interest saved
   560→    financialImpactLabel: `Save ${formatINRFull(Math.abs(od.balance) * 0.12 / 12)}/mo in OD interest`,
   561→    confidence: confidenceFrom(accounts.length, 70),
   562→    priority: 'medium',
   563→    risk,
   564→    businessImpact: 'Reduces interest cost without changing total cash position.',
   565→    rollbackPlan: 'Funds can be swept back at any time. No penalty on intra-bank transfers.',
   566→    approvalRequired: requirement,
   567→    requiresRole,
   568→    actions: [
   569→      action('Sweep funds', `Transfer ${formatINR(moveAmount)} from ${idle.bank} to ${od.bank}.`, 'oracle', 10, true, true),
   570→    ],
   571→    relatedEntityType: 'bank_account',
   572→    relatedEntityId: idle.id,
   573→    relatedEntityLabel: `${idle.bank} → ${od.bank}`,
   574→    evidence: [
   575→      evidence('banking', `${idle.bank} balance: ${formatINR(idle.balance)}`, idle.balance),
   576→      evidence('banking', `${od.bank} OD balance: ${formatINR(od.balance)}`, od.balance),
   577→    ],
   578→    status: 'pending',
   579→    createdAt: new Date().toISOString(),
   580→    expiresAt: new Date(Date.now() + 7 * 24 * 60 * 60 * 1000).toISOString(),
   581→  };
   582→}

   586→export function computeExecutiveDecisions(data: CEODataView): ExecutiveDecision[] {
   587→  const builders = [
   588→    buildRecoverPayment,
   589→    buildRemindClient,
   590→    buildFileOverdueReturn,
   591→    buildClaimITC,
   592→    buildPayGST,
   593→    buildDelayHiring,
   594→    buildSuggestLoan,
   595→    buildReduceExpenses,
   596→    buildPauseMarketing,
   597→    buildInvestigateAnomaly,
   598→    buildReplyCustomer,
   599→    buildOptimizeCash,
   600→  ];
   601→
   602→  const decisions: ExecutiveDecision[] = [];
   603→  for (const b of builders) {
   604→    try {
   605→      const d = b(data);
   606→      if (d) decisions.push(d);
   607→    } catch (err) {
   608→      console.warn(`[AI CEO] Decision builder ${b.name} failed:`, err);
   609→    }
   610→  }
   611→
   612→  // Sort by priority (critical first) then by financial impact (desc)
   613→  const priorityRank: Record<DecisionPriority, number> = {
   614→    critical: 0, high: 1, medium: 2, low: 3,
   615→  };
   616→  return decisions.sort((a, b) => {
   617→    if (priorityRank[a.priority] !== priorityRank[b.priority]) {
   618→      return priorityRank[a.priority] - priorityRank[b.priority];
   619→    }
   620→    return Math.abs(b.financialImpact) - Math.abs(a.financialImpact);
   621→  });
   622→}
