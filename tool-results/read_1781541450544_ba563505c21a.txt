   271→   271→        setSettings({
   272→   272→          firmName: s.firmName ?? DEFAULT_SETTINGS.firmName,
   273→   273→          logoUrl: s.logoUrl ?? DEFAULT_SETTINGS.logoUrl,

   276→   276→          customDomain: s.customDomain ?? DEFAULT_SETTINGS.customDomain,

   278→   278→          emailTemplate: s.emailTemplate ?? DEFAULT_SETTINGS.emailTemplate,

   452→   452→        transition={{ duration: 0.4 }}
   453→   453→        className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3"
   454→   454→      >
