   114→function getDaysRemaining(dueDateStr: string): number {
   115→  const dueDate = new Date(dueDateStr);
   116→  const today = new Date('2025-07-08');
   117→  today.setHours(0, 0, 0, 0);
   118→  dueDate.setHours(0, 0, 0, 0);
   119→  return Math.ceil((dueDate.getTime() - today.getTime()) / (1000 * 60 * 60 * 24));
   120→}
   121→
   122→function formatDaysRemaining(days: number): string {
   123→  if (days < 0) return `${Math.abs(days)}d overdue`;
   124→  if (days === 0) return 'Due today';
   125→  if (days === 1) return 'Tomorrow';
   126→  return `${days}d left`;
   127→}

   133→    .join('')

   135→    .toUpperCase();
   136→}
