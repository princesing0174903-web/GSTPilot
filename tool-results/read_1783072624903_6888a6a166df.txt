   258→   258→
   259→   259→function getInitials(name: string | null): string {
   260→   260→  if (!name) return '?'
   261→   261→  return name.split(' ').map(w => w[0]).join('').slice(0, 2).toUpperCase()
   262→   262→}
   263→   263→
   264→   264→function formatDate(dateStr: string | null): string {
   265→   265→  if (!dateStr) return '—'
   266→   266→  const d = new Date(dateStr)
   267→   267→  return d.toLocaleDateString('en-IN', { day: 'numeric', month: 'short', year: 'numeric' })
   268→   268→}
   269→   269→
   270→   270→function getDaysUntilDue(dateStr: string | null): number | null {
   271→   271→  if (!dateStr) return null
   272→   272→  const due = new Date(dateStr)
   273→   273→  const now = new Date()
   274→   274→  now.setHours(0, 0, 0, 0)
   275→   275→  due.setHours(0, 0, 0, 0)
   276→   276→  return Math.ceil((due.getTime() - now.getTime()) / (1000 * 60 * 60 * 24))
   277→   277→}
   278→   278→
   279→   279→function getDueDateColor(dateStr: string | null, status: TaskStatus): string {
   280→   280→  if (!dateStr || status === 'completed') return 'text-muted-foreground'
   281→   281→  const days = getDaysUntilDue(dateStr)
   282→   282→  if (days === null) return 'text-muted-foreground'
   283→   283→  if (days < 0) return 'text-red-600 font-medium'
   284→   284→  if (days <= 3) return 'text-amber-600 font-medium'
   285→   285→  return 'text-muted-foreground'
   286→   286→}
