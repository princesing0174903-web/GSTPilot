    33→function ValidationGauge({ percent }: { percent: number }) {
    34→  const r = 38
    35→  const c = 2 * Math.PI * r
    36→  const offset = c - (percent / 100) * c
    37→  return (
    38→    <svg width="100" height="100" className="overflow-visible">
    39→      <circle cx="50" cy="50" r={r} fill="none" stroke="currentColor" strokeWidth="8" className="text-slate-100 dark:text-slate-800" />
    40→      <circle cx="50" cy="50" r={r} fill="none" stroke="#10b981" strokeWidth="8" strokeLinecap="round" strokeDasharray={c} strokeDashoffset={offset} transform="rotate(-90 50 50)" />
    41→      <text x="50" y="46" textAnchor="middle" className="text-lg font-bold fill-emerald-600 dark:fill-emerald-400">{percent}%</text>
    42→      <text x="50" y="60" textAnchor="middle" className="text-[8px] fill-muted-foreground">Pass Rate</text>
    43→    </svg>
    44→  )
    45→}

    47→function DailyIRNChart({ data }: { data: { day: string; count: number }[] }) {
    48→  const max = Math.max(...data.map(d => d.count))
    49→  const barW = 24
    50→  const gap = 6
    51→  const h = 70
    52→  return (
    53→    <svg width={data.length * (barW + gap)} height={h + 20} className="overflow-visible">
    54→      {data.map((d, i) => {
    55→        const barH = (d.count / max) * (h - 10)
    56→        return (
    57→          <g key={d.day}>
    58→            <rect x={i * (barW + gap)} y={h - barH} width={barW} height={barH} rx={3} fill="#10b981" opacity={0.6 + (d.count / max) * 0.4} />
    59→            <text x={i * (barW + gap) + barW / 2} y={h + 14} textAnchor="middle" className="text-[9px] fill-muted-foreground">{d.day}</text>
    60→          </g>
    61→        )
    62→      })}
    63→    </svg>
    64→  )
    65→}

    98→// Maps an Invoice row from /api/invoices to the EInvoiceRow shape used by the UI.
    99→// The Invoice model has no dedicated IRN column, so we treat the invoice number
   100→// as the IRN identifier (e-invoices are invoices that have been pushed through
   101→// the IRN generation flow).
   102→function mapInvoiceToEInvoice(inv: ApiInvoice): EInvoiceRow {
   103→  const tax = (inv.cgst || 0) + (inv.sgst || 0) + (inv.igst || 0) + (inv.cess || 0)
   104→  const status: EInvoiceRow['status'] =
   105→    inv.status === 'cancelled' ? 'cancelled' :
   106→    inv.status === 'expired' ? 'expired' : 'valid'
   107→  const dateStr = inv.invoiceDate
   108→    ? new Date(inv.invoiceDate).toLocaleDateString('en-IN', { day: '2-digit', month: '2-digit', year: 'numeric' })
   109→    : '—'
   110→  return {
   111→    irn: inv.invoiceNumber || '—',
   112→    date: dateStr,
   113→    gstin: inv.buyerGstin || inv.sellerGstin || '—',
   114→    buyer: inv.buyerName || inv.client?.tradeName || '—',
   115→    amount: inv.totalAmount || 0,
   116→    tax,
   117→    status,
   118→  }
   119→}
