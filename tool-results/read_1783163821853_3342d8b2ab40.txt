    58→function getDaysRemaining(dueDateStr: string): number {
    59→  const dueDate = new Date(dueDateStr);
    60→  const today = new Date();
    61→  today.setHours(0, 0, 0, 0);
    62→  dueDate.setHours(0, 0, 0, 0);
    63→  return Math.ceil((dueDate.getTime() - today.getTime()) / (1000 * 60 * 60 * 24));
    64→}
    65→
    66→function formatDaysRemaining(days: number): string {
    67→  if (days < 0) return `${Math.abs(days)}d overdue`;
    68→  if (days === 0) return 'Due today';
    69→  if (days === 1) return 'Tomorrow';
    70→  return `${days}d left`;
    71→}
    72→
    73→function formatINR(amount: number): string {
    74→  return amount.toLocaleString('en-IN');
    75→}
    76→
    77→function formatDateIN(dateStr: string): string {
    78→  const d = new Date(dateStr);
    79→  if (Number.isNaN(d.getTime())) return '—';
    80→  const dd = String(d.getDate()).padStart(2, '0');
    81→  const mm = String(d.getMonth() + 1).padStart(2, '0');
    82→  const yyyy = d.getFullYear();
    83→  return `${dd}/${mm}/${yyyy}`;
    84→}
    85→
    86→function timeAgo(dateStr: string): string {
    87→  const now = new Date();
    88→  const date = new Date(dateStr);
    89→  const seconds = Math.floor((now.getTime() - date.getTime()) / 1000);
    90→  if (seconds < 60) return 'just now';
    91→  const minutes = Math.floor(seconds / 60);
    92→  if (minutes < 60) return `${minutes}m ago`;
    93→  const hours = Math.floor(minutes / 60);
    94→  if (hours < 24) return `${hours}h ago`;
    95→  const days = Math.floor(hours / 24);
    96→  if (days < 7) return `${days}d ago`;
    97→  return date.toLocaleDateString('en-IN', { day: 'numeric', month: 'short' });
    98→}

   113→// One-sentence AI insight derived from live metrics.
   114→function buildInsightSentence(
   115→  metrics: LiveDashboardMetrics,
   116→  pendingCollection: number,
   117→): string {
   118→  const parts: string[] = [];
   119→  if (metrics.overdueReturns > 0) {
   120→    parts.push(
   121→      `${metrics.overdueReturns} overdue return${metrics.overdueReturns > 1 ? 's' : ''}`,
   122→    );
   123→  } else if (metrics.pendingReturns > 0) {
   124→    parts.push(
   125→      `${metrics.pendingReturns} return${metrics.pendingReturns > 1 ? 's' : ''} to file`,
   126→    );
   127→  }
   128→  if (pendingCollection > 0) {
   129→    parts.push(`₹${formatINR(pendingCollection)} pending collection`);
   130→  }
   131→  if (metrics.criticalIssues > 0) {
   132→    parts.push(
   133→      `${metrics.criticalIssues} critical issue${metrics.criticalIssues > 1 ? 's' : ''}`,
   134→    );
   135→  }
   136→  if (parts.length === 0) {
   137→    return `All clear — ${metrics.filedReturns} returns filed and ${metrics.totalClients} clients in good standing.`;
   138→  }
   139→  if (parts.length === 1) return `${parts[0]}.`;
   140→  const last = parts.pop();
   141→  return `${parts.join(', ')} and ${last}.`;
   142→}

   261→function healthTier(score: number): { label: string; tone: string } {

   676→        view: 'invoices',
   677→      });

   681→        id: 'fb-extractions',
   682→        label: `Review ${metrics.extractionsPending} pending extraction${metrics.extractionsPending > 1 ? 's' : ''}`,
   683→        category: 'upload',
   684→        urgency: 5,
   685→        view: 'invoices',

   695→      { id: 'einvoice', name: 'E-Invoice', initial: 'E', connected: false },

   697→      { id: 'bank', name: 'Bank APIs', initial: 'B', connected: false },
   698→      { id: 'whatsapp', name: 'WhatsApp', initial: 'W', connected: false },
   699→      { id: 'gmail', name: 'Gmail', initial: 'M', connected: false },

   708→      name: m.userId?.split('@')[0] || 'Team member',
   709→      role: m.role || 'staff',
   710→      status: m.status || 'invited',
   711→    })),
   712→    [memberships],

   714→
   715→  // ── Upcoming filings (non-filed, sorted by urgency) ───────────────────
   716→  const upcomingFilings = useMemo(() => {

   718→      .filter((r) => r.status !== 'filed')
   719→      .sort((a, b) => {
   720→        const aOverdue = isOverdue(a.period) ? 0 : 1;
   721→        const bOverdue = isOverdue(b.period) ? 0 : 1;

   723→        return a.period.localeCompare(b.period);

   732→    // Overdue returns → file now
   733→    for (const r of returns) {

   735→      if (r.status === 'filed' || !isOverdue(r.period)) continue;
   736→      const client = clientMap.get(r.clientId);
   737→      const name = client?.tradeName ?? 'Unknown client';
   738→      recs.push({
   739→        id: `rec-overdue-${r.id}`,
   740→        icon: <AlertTriangle className="h-3.5 w-3.5 accent-text" />,
   741→        title: `File ${r.returnType} for ${name} — overdue`,
   742→        actionLabel: 'File now',
   743→        onAction: () => handleFileReturn(r.clientId, r.returnType, r.period),
   744→      });
   745→    }
   746→
   747→    // Returns due soon (≤ 7 days)
   748→    for (const r of returns) {
   749→      if (recs.length >= 5) break;
   750→      if (r.status === 'filed' || isOverdue(r.period)) continue;

   795→        actionLabel: 'Review',

   812→      clientId,
   813→      returnType: (returnType === 'GSTR-3B' ? 'GSTR-3B' : 'GSTR-1') as 'GSTR-1' | 'GSTR-3B',
   814→      period,

   820→    if (filingInProgress.has(returnId)) return;
   821→    setFilingInProgress((prev) => new Set(prev).add(returnId));
   822→    try {
   823→      const arn = await fileReturn(returnId);
   824→      toast.success(`${returnType} Filed Successfully`, {
   825→        description: `${clientName} — ARN: ${arn}`,

   828→      toast.error('Filing Failed', {
   829→        description: err instanceof Error ? err.message : 'Unknown error',
   830→      });
   831→    } finally {
   832→      setFilingInProgress((prev) => {
   833→        const next = new Set(prev);
   834→        next.delete(returnId);
   835→        return next;
   836→      });

   856→          <CardContent className="p-6 text-center">
   857→            <AlertTriangle className="h-8 w-8 text-red-500 mx-auto mb-2" />
   858→            <h3 className="font-semibold text-foreground">Failed to load dashboard</h3>
   859→            <p className="text-sm text-muted-foreground mt-1">{error}</p>
   860→            <Button
   861→              variant="outline"
   862→              className="mt-4 border-border"
   863→              onClick={() => window.location.reload()}
   864→            >
   865→              Retry
   866→            </Button>
   867→          </CardContent>
   868→        </Card>
   869→      </div>
   870→    );
   871→  }
   872→
   873→  // ── Empty state — no clients at all ───────────────────────────────────
   874→  if (metrics.totalClients === 0) {
   875→    return (
   876→      <div className="relative max-w-6xl mx-auto px-4 md:px-6 py-8 md:py-10">
   877→        <WelcomeEmptyState
   878→          onAddClient={() => setCurrentView('clients')}
   879→          onUploadDoc={() => setCurrentView('invoices')}
   880→        />
   881→      </div>
   882→    );
   883→  }

   886→  // Revenue KPI: REAL revenue from the invoice engine — Σ grandTotal of
   887→  // non-draft, non-cancelled invoices (server-calculated per invoice).
   888→  const revenueValue =
   889→    invoiceStats.totalRevenue > 0 ? `₹${formatINR(invoiceStats.totalRevenue)}` : '—';
   890→  const complianceValue = String(pendingComplianceCount);
   891→  // Cash Position KPI: REAL outstanding from the invoice engine — Σ balanceDue
   892→  // of non-draft, non-cancelled invoices.
   893→  const cashValue =
   894→    invoiceStats.totalOutstanding > 0 ? `₹${formatINR(invoiceStats.totalOutstanding)}` : '—';
   895→
   896→  const firstName = getFirstName(user?.name);
   897→
   898→  return (
   899→    <div className="relative max-w-6xl mx-auto px-4 md:px-6 py-8 md:py-10">
   900→      {/* ── Ambient radial glow at top ── */}
   901→      <div
   902→        aria-hidden
   903→        className="pointer-events-none absolute inset-x-0 top-0 h-[420px] bg-[radial-gradient(ellipse_at_top,_rgba(16,185,129,0.08),_transparent_60%)]"
   904→      />
   905→
   906→      {/* ── Content ── */}
   907→      <div className="relative space-y-8">
   908→        {/* ═══ GREETING + AI INSIGHT ═══ */}
   909→        <motion.div
   910→          initial={{ opacity: 0, y: -8 }}
   911→          animate={{ opacity: 1, y: 0 }}
   912→          transition={{ duration: 0.5, ease: 'easeOut' }}
   913→          className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4"
   914→        >
   915→          <div className="space-y-2 min-w-0">
   916→            <h1 className="text-3xl md:text-4xl font-bold text-foreground tracking-tight">
