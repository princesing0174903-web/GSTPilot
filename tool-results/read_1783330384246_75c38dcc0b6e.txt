   200→      await onSave()
   201→      setState('saved')
   202→      setTimeout(() => setState('idle'), 2000)
   203→    } catch {
   204→      setState('idle')
   205→    }
   206→  }
   207→
   208→  return (
   209→    <Button
   210→      onClick={handleClick}
   211→      disabled={state === 'saving'}
   212→      className="bg-emerald-600 hover:bg-emerald-700 text-white gap-2 min-w-[110px]"
   213→    >
   214→      <AnimatePresence mode="wait">
   215→        {state === 'idle' && (
   216→          <motion.span
   217→            key="idle"
   218→            initial={{ opacity: 0, scale: 0.8 }}
   219→            animate={{ opacity: 1, scale: 1 }}

   223→            <Save className="h-4 w-4" />

   256→// ─── Password Strength Helper ──────────────────────────────────────────
   257→function getPasswordStrength(password: string): { score: number; label: string; color: string } {
   258→  if (!password) return { score: 0, label: '', color: '' }
   259→  let score = 0
   260→  if (password.length >= 8) score++
   261→  if (password.length >= 12) score++
   262→  if (/[A-Z]/.test(password)) score++
   263→  if (/[0-9]/.test(password)) score++
   264→  if (/[^A-Za-z0-9]/.test(password)) score++
   265→
   266→  if (score <= 1) return { score: 20, label: 'Weak', color: 'bg-red-500' }
   267→  if (score <= 2) return { score: 40, label: 'Fair', color: 'bg-orange-500' }
   268→  if (score <= 3) return { score: 60, label: 'Good', color: 'bg-yellow-500' }
   269→  if (score <= 4) return { score: 80, label: 'Strong', color: 'bg-emerald-500' }
   270→  return { score: 100, label: 'Excellent', color: 'bg-emerald-600' }
   271→}
