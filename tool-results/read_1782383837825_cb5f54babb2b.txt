    49→function getDaysRemaining(dueDateStr: string): number {
    50→  const dueDate = new Date(dueDateStr);
    51→  const today = new Date();
    52→  today.setHours(0, 0, 0, 0);
    53→  dueDate.setHours(0, 0, 0, 0);
    54→  return Math.ceil((dueDate.getTime() - today.getTime()) / (1000 * 60 * 60 * 24));
    55→}
    56→
    57→function formatDaysRemaining(days: number): string {
    58→  if (days < 0) return `${Math.abs(days)}d overdue`;
    59→  if (days === 0) return 'Due today';
    60→  if (days === 1) return 'Tomorrow';
    61→  return `${days}d left`;
    62→}
    63→
    64→function formatINR(amount: number): string {
    65→  return amount.toLocaleString('en-IN');
    66→}
    67→
    68→function formatDateIN(dateStr: string): string {
    69→  const d = new Date(dateStr);
    70→  if (Number.isNaN(d.getTime())) return '—';
    71→  const dd = String(d.getDate()).padStart(2, '0');
    72→  const mm = String(d.getMonth() + 1).padStart(2, '0');
    73→  const yyyy = d.getFullYear();
    74→  return `${dd}/${mm}/${yyyy}`;
    75→}
    76→
    77→function timeAgo(dateStr: string): string {
    78→  const now = new Date();
    79→  const date = new Date(dateStr);
    80→  const seconds = Math.floor((now.getTime() - date.getTime()) / 1000);
    81→  if (seconds < 60) return 'just now';
    82→  const minutes = Math.floor(seconds / 60);
    83→  if (minutes < 60) return `${minutes}m ago`;
    84→  const hours = Math.floor(minutes / 60);
    85→  if (hours < 24) return `${hours}h ago`;
    86→  const days = Math.floor(hours / 24);
    87→  if (days < 7) return `${days}d ago`;
    88→  return date.toLocaleDateString('en-IN', { day: 'numeric', month: 'short' });
    89→}

    91→function getGreeting(): string {
    92→  const h = new Date().getHours();
    93→  if (h < 12) return 'Good Morning';
    94→  if (h < 17) return 'Good Afternoon';
    95→  return 'Good Evening';
    96→}

    98→function getFirstName(name: string | undefined | null): string {
    99→  if (!name) return 'Prince';
   100→  const first = name.trim().split(/\s+/)[0];
   101→  return first || 'Prince';

   104→// One-sentence AI insight derived from live metrics.
   105→function buildInsightSentence(
   106→  metrics: LiveDashboardMetrics,
   107→  pendingCollection: number,
   108→): string {
   109→  const parts: string[] = [];
   110→  if (metrics.overdueReturns > 0) {
   111→    parts.push(
   112→      `${metrics.overdueReturns} overdue return${metrics.overdueReturns > 1 ? 's' : ''}`,
   113→    );
   114→  } else if (metrics.pendingReturns > 0) {
   115→    parts.push(
   116→      `${metrics.pendingReturns} return${metrics.pendingReturns > 1 ? 's' : ''} to file`,
   117→    );
   118→  }
   119→  if (pendingCollection > 0) {
   120→    parts.push(`₹${formatINR(pendingCollection)} pending collection`);
   121→  }
   122→  if (metrics.criticalIssues > 0) {
   123→    parts.push(
   124→      `${metrics.criticalIssues} critical issue${metrics.criticalIssues > 1 ? 's' : ''}`,
   125→    );
   126→  }
   127→  if (parts.length === 0) {
   128→    return `All clear — ${metrics.filedReturns} returns filed and ${metrics.totalClients} clients in good standing.`;
   129→  }
   130→  if (parts.length === 1) return `${parts[0]}.`;
   131→  const last = parts.pop();
   132→  return `${parts.join(', ')} and ${last}.`;
   133→}

   140→  if (type.includes('filed')) return <CheckCircle2 className="h-3.5 w-3.5 accent-text" />;

   145→  if (type.includes('reconcil') || type.includes('mismatch'))

   345→interface Recommendation {
   346→  id: string;
   347→  icon: React.ReactNode;
   348→  title: string;
   349→  actionLabel: string;
   350→  onAction: () => void;

   352→
   353→export default function DashboardPage() {
   354→  const { setCurrentView, setSelectedClientId, setReturnPrepCtx } = useApp();
   355→  const { user } = useAuth();
   356→
   357→  // ── Firebase Firestore hooks ──────────────────────────────────────────
   358→  const { metrics, loading, error } = useLiveDashboardMetrics();
   359→  const { data: clients } = useFireClients();
   360→  const { data: returns } = useFireReturns();
   361→  const { data: invoices } = useFireInvoices();
   362→  const { data: recentActivities } = useFireRecentActivities(10);
   363→
   364→  // ── Filing state ──────────────────────────────────────────────────────
   365→  const [filingInProgress, setFilingInProgress] = useState<Set<string>>(new Set());
   366→
   367→  // ── Client lookup map ─────────────────────────────────────────────────
   368→  const clientMap = useMemo(() => {
   369→    const map = new Map<string, FirestoreClient & { id: string }>();
   370→    for (const c of clients) map.set(c.clientId, c);
   371→    return map;
   372→  }, [clients]);
   373→
   374→  // ── KPI derivations ───────────────────────────────────────────────────
   375→  const pendingComplianceCount = useMemo(
   376→    () => returns.filter((r) => r.status !== 'filed').length,
   377→    [returns],
   378→  );
   379→
   380→  const pendingInvoices = useMemo(
   381→    () => invoices.filter((i) => i.status === 'draft' || i.status === 'approved'),
   382→    [invoices],
   383→  );
   384→
   385→  const pendingCollection = useMemo(
   386→    () => pendingInvoices.reduce((sum, i) => sum + (i.totalAmount || 0), 0),
   387→    [pendingInvoices],
   388→  );
   389→
   390→  const insight = useMemo(
   391→    () => buildInsightSentence(metrics, pendingCollection),
   392→    [metrics, pendingCollection],
   393→  );
   394→
   395→  // ── Upcoming filings (non-filed, sorted by urgency) ───────────────────
   396→  const upcomingFilings = useMemo(() => {
   397→    return returns
   398→      .filter((r) => r.status !== 'filed')
   399→      .sort((a, b) => {
   400→        const aOverdue = isOverdue(a.period) ? 0 : 1;
   401→        const bOverdue = isOverdue(b.period) ? 0 : 1;
   402→        if (aOverdue !== bOverdue) return aOverdue - bOverdue;
   403→        return a.period.localeCompare(b.period);
   404→      })
   405→      .slice(0, 5);
   406→  }, [returns]);
   407→
   408→  // ── AI Recommendations ────────────────────────────────────────────────
   409→  const recommendations = useMemo<Recommendation[]>(() => {
   410→    const recs: Recommendation[] = [];
   411→
   412→    // Overdue returns → file now
   413→    for (const r of returns) {
   414→      if (recs.length >= 5) break;
   415→      if (r.status === 'filed' || !isOverdue(r.period)) continue;
   416→      const client = clientMap.get(r.clientId);
   417→      const name = client?.tradeName ?? 'Unknown client';
   418→      recs.push({
   419→        id: `rec-overdue-${r.id}`,
   420→        icon: <AlertTriangle className="h-3.5 w-3.5 accent-text" />,
   421→        title: `File ${r.returnType} for ${name} — overdue`,
   422→        actionLabel: 'File now',
   423→        onAction: () => handleFileReturn(r.clientId, r.returnType, r.period),
   424→      });
   425→    }
   426→
   427→    // Returns due soon (≤ 7 days)
   428→    for (const r of returns) {
   429→      if (recs.length >= 5) break;
   430→      if (r.status === 'filed' || isOverdue(r.period)) continue;
   431→      const dueDate = getFilingDueDate(r.returnType, r.period);
   432→      const days = getDaysRemaining(dueDate);
   433→      if (days > 7) continue;
   434→      const client = clientMap.get(r.clientId);
   435→      const name = client?.tradeName ?? 'Unknown client';
   436→      recs.push({
   437→        id: `rec-soon-${r.id}`,
   438→        icon: <Clock className="h-3.5 w-3.5 accent-text" />,
   439→        title: `File ${r.returnType} for ${name} — due in ${days} day${days === 1 ? '' : 's'}`,
   440→        actionLabel: 'Prepare',
   441→        onAction: () => handleFileReturn(r.clientId, r.returnType, r.period),
   442→      });
   443→    }
   444→
   445→    // Low health score clients
   446→    for (const c of clients) {
   447→      if (recs.length >= 5) break;
   448→      if (c.healthScore >= 60) continue;
   449→      recs.push({
   450→        id: `rec-health-${c.clientId}`,
   451→        icon: <ShieldCheck className="h-3.5 w-3.5 accent-text" />,
   452→        title: `${c.tradeName} — health score ${Math.round(c.healthScore)}% needs attention`,
   453→        actionLabel: 'Open',
   454→        onAction: () => handleOpenClient(c.clientId),
   455→      });
   456→    }
   457→
   458→    // Pending collection
   459→    if (recs.length < 5 && pendingCollection > 0) {
   460→      recs.push({
   461→        id: 'rec-collection',
   462→        icon: <IndianRupee className="h-3.5 w-3.5 accent-text" />,
   463→        title: `₹${formatINR(pendingCollection)} pending collection across ${pendingInvoices.length} invoice${pendingInvoices.length === 1 ? '' : 's'}`,
   464→        actionLabel: 'Review',
   465→        onAction: () => setCurrentView('invoices'),
   466→      });
   467→    }
   468→
   469→    // Critical issues
   470→    if (recs.length < 5 && metrics.criticalIssues > 0) {
   471→      recs.push({
   472→        id: 'rec-critical',
   473→        icon: <AlertTriangle className="h-3.5 w-3.5 accent-text" />,
   474→        title: `${metrics.criticalIssues} critical compliance issue${metrics.criticalIssues === 1 ? '' : 's'} need${metrics.criticalIssues === 1 ? 's' : ''} review`,
   475→        actionLabel: 'Review',
   476→        onAction: () => setCurrentView('reconcile'),
   477→      });
   478→    }
   479→
   480→    return recs;
   481→  }, [returns, clients, clientMap, pendingCollection, pendingInvoices.length, metrics.criticalIssues]);
   482→
   483→  // ── Handlers ──────────────────────────────────────────────────────────
   484→  const handleOpenClient = (clientId: string) => {
   485→    setSelectedClientId(clientId);
   486→    setCurrentView('client-workspace');
   487→  };
   488→
   489→  const handleFileReturn = (clientId: string, returnType: string, period: string) => {
   490→    setSelectedClientId(clientId);
   491→    setReturnPrepCtx({
   492→      clientId,
   493→      returnType: (returnType === 'GSTR-3B' ? 'GSTR-3B' : 'GSTR-1') as 'GSTR-1' | 'GSTR-3B',
   494→      period,
   495→    });
   496→    setCurrentView('return-prep');
   497→  };
   498→
   499→  const handleQuickFile = async (returnId: string, clientName: string, returnType: string) => {
   500→    if (filingInProgress.has(returnId)) return;
   501→    setFilingInProgress((prev) => new Set(prev).add(returnId));
   502→    try {
   503→      const arn = await fileReturn(returnId);
   504→      toast.success(`${returnType} Filed Successfully`, {
   505→        description: `${clientName} — ARN: ${arn}`,
   506→      });
   507→    } catch (err) {
   508→      toast.error('Filing Failed', {
   509→        description: err instanceof Error ? err.message : 'Unknown error',
   510→      });
   511→    } finally {
   512→      setFilingInProgress((prev) => {
   513→        const next = new Set(prev);
   514→        next.delete(returnId);
   515→        return next;
   516→      });
   517→    }
   518→  };
   519→
   520→  // ═══════════════════════════════════════════════════════════════════════════
   521→  // RENDER
   522→  // ═══════════════════════════════════════════════════════════════════════════
   523→
   524→  // ── Loading ───────────────────────────────────────────────────────────
   525→  if (loading) return <DashboardSkeleton />;
   526→
   527→  // ── Error ─────────────────────────────────────────────────────────────
