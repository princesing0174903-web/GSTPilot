    62→function BalanceTrendChart({ data }: { data: { day: string; balance: number }[] }) {
    63→  const max = Math.max(...data.map(d => d.balance))
    64→  const min = Math.min(...data.map(d => d.balance))
    65→  const range = max - min || 1
    66→  const w = 280
    67→  const h = 60
    68→  const pts = data.map((d, i) => `${(i / (data.length - 1)) * w},${h - ((d.balance - min) / range) * (h - 8) - 4}`)
    69→  return (
    70→    <svg width={w} height={h + 16} className="overflow-visible">
    71→      <defs>
    72→        <linearGradient id="balGrad" x1="0" y1="0" x2="0" y2="1">
    73→          <stop offset="0%" stopColor="#10b981" stopOpacity="0.2" />
    74→          <stop offset="100%" stopColor="#10b981" stopOpacity="0.02" />
    75→        </linearGradient>
    76→      </defs>
    77→      <polygon points={`0,${h} ${pts.join(' ')} ${w},${h}`} fill="url(#balGrad)" />
    78→      <polyline points={pts.join(' ')} fill="none" stroke="#10b981" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" />
    79→      {data.filter((_, i) => i % 2 === 0).map((d, i) => {
    80→        const x = (data.indexOf(d) / (data.length - 1)) * w
    81→        return <text key={i} x={x} y={h + 12} textAnchor="middle" className="text-[8px] fill-muted-foreground">{d.day}</text>
    82→      })}
    83→    </svg>
    84→  )
    85→}

    87→function ReconcileDonut({ reconciled, unreconciled }: { reconciled: number; unreconciled: number }) {
    88→  const total = reconciled + unreconciled
    89→  const r = 36
    90→  const c = 2 * Math.PI * r
    91→  const reconcilePct = (reconciled / total) * 100
    92→  const dashOffset = c - (reconcilePct / 100) * c
    93→  return (
    94→    <svg width="90" height="90" className="overflow-visible">
    95→      <circle cx="45" cy="45" r={r} fill="none" stroke="currentColor" strokeWidth="8" className="text-slate-100 dark:text-slate-800" />
    96→      <circle cx="45" cy="45" r={r} fill="none" stroke="#10b981" strokeWidth="8" strokeLinecap="round" strokeDasharray={c} strokeDashoffset={dashOffset} transform="rotate(-90 45 45)" />
    97→      <text x="45" y="42" textAnchor="middle" className="text-sm font-bold fill-emerald-600 dark:fill-emerald-400">{reconcilePct.toFixed(0)}%</text>
    98→      <text x="45" y="54" textAnchor="middle" className="text-[7px] fill-muted-foreground">Reconciled</text>
    99→    </svg>
   100→  )
   101→}

   180→function formatSyncDate(iso: string | null): string {
   181→  if (!iso) return '—'
   182→  const d = new Date(iso)
   183→  if (Number.isNaN(d.getTime())) return '—'
   184→  const dd = String(d.getDate()).padStart(2, '0')
   185→  const mm = String(d.getMonth() + 1).padStart(2, '0')
   186→  const yyyy = d.getFullYear()
   187→  const hh = String(d.getHours()).padStart(2, '0')
   188→  const min = String(d.getMinutes()).padStart(2, '0')
   189→  return `${dd}/${mm}/${yyyy} ${hh}:${min}`
   190→}
   191→
   192→function formatTxnDate(iso: string): string {
   193→  if (!iso) return '—'
   194→  const d = new Date(iso)
   195→  if (Number.isNaN(d.getTime())) return iso
   196→  const dd = String(d.getDate()).padStart(2, '0')
   197→  const mm = String(d.getMonth() + 1).padStart(2, '0')
   198→  const yyyy = d.getFullYear()
   199→  return `${dd}/${mm}/${yyyy}`
   200→}

   202→function upperMode(mode: string): string {
   203→  return (mode || 'bank').toUpperCase()
   204→}

   206→// Map Firestore BankAccount → BankAccount shape used by this page.
   207→function mapBankAccount(acc: FirestoreBankAccount & { id: string }): BankAccount {
   208→  const accountType = (acc.accountType || 'current') as string
   209→  return {
   210→    id: acc.id,
   211→    bank: acc.bankName || 'Bank Account',
   212→    account: acc.accountNumberMasked || '****',
   213→    type: accountType.charAt(0).toUpperCase() + accountType.slice(1),
   214→    balance: Number(acc.currentBalance ?? acc.availableBalance ?? 0) || 0,
   215→    lastSync: formatSyncDate((acc.lastSyncAt as string | null) ?? null),
   216→    status: (acc.status as string) || 'connected',
   217→  }
   218→}
   219→
   220→// Map Firestore BankTransaction → BankTransaction shape used by this page.
   221→function mapBankTxnToTxn(t: FirestoreBankTransaction & { id: string }): BankTransaction {
   222→  const type = t.type === 'credit' ? 'credit' : 'debit'
   223→  return {
   224→    id: t.id,
   225→    date: formatTxnDate(t.date || ''),
   226→    description: t.description || 'Bank Transaction',
   227→    amount: Math.abs(Number(t.amount) || 0),
   228→    type,
   229→    balance: t.balanceAfter == null ? null : Number(t.balanceAfter),
   230→    account: t.referenceNo ? t.referenceNo.toUpperCase() : 'BANK',
   231→    category: (t.category as string) || (type === 'credit' ? 'Revenue' : 'Purchase'),
   232→  }
   233→}
   234→
   235→// Map Firestore Payment row → BankTransaction shape used by this page.
   236→function mapPaymentToTxn(p: FirestorePayment & { id: string }): BankTransaction {
   237→  const isVendor = (p.partyType || 'customer') === 'vendor'
   238→  return {
   239→    id: p.id,
   240→    date: formatTxnDate(p.paymentDate || ''),
   241→    description: `${upperMode(p.paymentMode || 'bank')} - ${p.partyName || 'Unknown'}`,
   242→    amount: Number(p.amount) || 0,
   243→    type: isVendor ? 'debit' : 'credit',
   244→    balance: null,
   245→    account: upperMode(p.paymentMode || 'bank'),
   246→    category: isVendor ? 'Purchase' : 'Revenue',
   247→  }
   248→}
   249→
   250→// Map Firestore Payment row → ReconciliationEntry shape used by this page.
   251→function mapPaymentToRecon(p: FirestorePayment & { id: string }): ReconciliationEntry {
   252→  let status: ReconciliationEntry['status'] = 'unmatched'
   253→  if (p.reconciled) status = 'matched'
   254→  else if (p.status === 'failed') status = 'disputed'
   255→  return {
   256→    id: p.id,
   257→    date: formatTxnDate(p.paymentDate || ''),
   258→    bankTxn: p.referenceNo || `${upperMode(p.paymentMode || 'bank')}-${p.id.slice(-6)}`,
   259→    bookEntry: p.invoiceId || null,
   260→    amount: Number(p.amount) || 0,
   261→    status,
   262→    account: upperMode(p.paymentMode || 'bank'),
   263→  }
   264→}
   265→
   266→// Map Firestore Expense → BankTransaction (always a debit) shape used by this page.
   267→function mapExpenseToTxn(e: FirestoreExpense & { id: string }): BankTransaction {
   268→  const mode = (e.paymentMode || 'bank') as string
   269→  return {
   270→    id: e.id,
   271→    date: formatTxnDate(e.date || ''),
   272→    description: `${upperMode(mode)} - ${e.vendor || e.description || 'Expense'}`,
   273→    amount: Number(e.amount) || 0,
   274→    type: 'debit',
   275→    balance: null,
   276→    account: upperMode(mode),
   277→    category: (e.category as string) || 'Purchase',
   278→  }
   279→}

   281→// Map Firestore BankTransaction → banking-provider BankTransaction shape for
   282→// the /api/banking/reconcile endpoint. The engine is pure + deterministic
   283→// (amount + counterparty + reference matching) — no LLM, no fake data.
   284→function mapBankTxnForRecon(
   285→  t: FirestoreBankTransaction & { id: string },
   286→  organizationId: string,
   287→): ReconBankTransaction {
   288→  const type = t.type === 'credit' ? 'credit' : 'debit'
   289→  return {
   290→    id: t.id,
   291→    organizationId,
   292→    connectionId: t.bankAccountId || 'legacy',
   293→    accountId: t.bankAccountId || 'legacy',
   294→    date: t.date || new Date().toISOString(),
   295→    description: t.description || 'Bank Transaction',
   296→    amount: Math.abs(Number(t.amount) || 0),
   297→    type,
   298→    balance: t.balanceAfter == null ? null : Number(t.balanceAfter),
   299→    category: ((t.category as TransactionCategory | null) ?? 'other'),
   300→    counterparty: null,
   301→    referenceNumber: t.referenceNo,
   302→    invoiceId: t.reconciledWith,
   303→    reconciled: t.reconciled ? 'matched' : 'unmatched',
   304→    matchConfidence: 0,
   305→    syncedAt: new Date().toISOString(),
   306→    createdAt: new Date().toISOString(),
   307→    updatedAt: new Date().toISOString(),
   308→  }
   309→}

   311→// Map Firestore Invoice → ReconcileInvoiceRef for the reconcile engine.
   312→// FirestoreInvoice rows are GSTR-1 outward supplies (gstr1Section is one of
   313→// b2b/b2cl/b2cs/cdnr/cdnur/exp), so we map them all to invoiceType:'sales'
   314→// — the engine then matches them against credit (incoming) bank transactions.
   315→function mapInvoiceForRecon(
   316→  inv: FirestoreInvoice & { id: string },
   317→): ReconcileInvoiceRef {
   318→  const total = Number(inv.totalAmount) || 0
   319→  return {
   320→    id: inv.id,
   321→    invoiceNumber: inv.invoiceNumber || '',
   322→    clientName: inv.buyerName || '',
   323→    grandTotal: total,
   324→    balanceDue: total, // FirestoreInvoice does not track balance separately yet
   325→    invoiceType: 'sales',
   326→    issueDate: inv.invoiceDate,
   327→    referenceNumber: null,
   328→  }
   329→}

   331→// Extract a human-readable error message from a failed fetch JSON body.
   332→function extractApiError(body: unknown, fallback: string): string {
   333→  if (body && typeof body === 'object' && 'error' in body) {
   334→    const err = (body as { error: unknown }).error
   335→    if (typeof err === 'string' && err.length > 0) return err
   336→  }
   337→  return fallback
   338→}
