   130→// ─── Helper ───────────────────────────────────────────────────────────────────

   132→function formatFileSize(bytes: number): string {
   133→  if (bytes < 1024) return `${bytes} B`;
   134→  if (bytes < 1048576) return `${(bytes / 1024).toFixed(1)} KB`;
   135→  return `${(bytes / 1048576).toFixed(1)} MB`;
   136→}
   137→
   138→function getFileIcon(fileName: string) {
   139→  const ext = fileName.split('.').pop()?.toLowerCase();
   140→  if (ext === 'json') return <FileJson className="size-4 text-amber-500" />;
   141→  if (ext === 'csv' || ext === 'xlsx') return <FileSpreadsheet className="size-4 text-emerald-500" />;
   142→  return <FileText className="size-4 text-slate-500" />;
   143→}

   187→    const riskItems = invoices.filter(i => i.riskLevel === 'high' || i.riskLevel === 'critical').length;

   195→      if (statusFilter !== 'all' && inv.status !== statusFilter) return false;

   202→        const matchesBuyer = (inv.buyerName ?? '').toLowerCase().includes(q);

   204→        if (!matchesNumber && !matchesBuyer && !matchesClient) return false;

   206→      return true;

   213→      const fileId = `upload-${Date.now()}-${Math.random().toString(36).slice(2, 7)}`;
   214→      setUploadingFiles(prev => [...prev, { id: fileId, name: file.name, progress: 0 }]);
   215→

   217→        // Simulate upload progress
   218→        setUploadingFiles(prev =>
   219→          prev.map(f => f.id === fileId ? { ...f, progress: 30 } : f)

   223→        const ext = file.name.split('.').pop()?.toLowerCase() ?? '';

   226→        else if (ext === 'csv' || ext === 'xlsx') docType = 'sales_register';

   238→          fileName: file.name,

   242→          period: '2025-06',

   257→          prev.map(f => f.id === fileId ? { ...f, progress: 0 } : f)

   315→    }
   316→  };

   333→        </div>

   335→      </div>
   336→    );
   337→  }

   339→  // ── Error state ──
   340→  if (invoicesError || clientsError) {

   342→      <div className="flex flex-col items-center justify-center py-20 text-center">
   343→        <AlertCircle className="size-10 text-red-400 mb-4" />
   344→        <h3 className="text-lg font-semibold text-foreground">Failed to load data</h3>
   345→        <p className="text-sm text-muted-foreground mt-1 max-w-md">
   346→          {invoicesError || clientsError}
   347→        </p>
   348→      </div>
   349→    );
   350→  }
   351→
   352→  // ── Empty state when no invoices exist ──
   353→  if (invoices.length === 0 && uploadingFiles.length === 0) {
   354→    return (
   355→      <div className="min-h-screen bg-gradient-to-b from-slate-50/80 to-white">
   356→        <div className="space-y-6 p-4 md:p-6 lg:p-8">
   357→          <motion.div
   358→            initial={{ opacity: 0, y: -12 }}
   359→            animate={{ opacity: 1, y: 0 }}
   360→            transition={{ duration: 0.4 }}
   361→            className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between"
   362→          >
   363→            <div>
   364→              <h1 className="text-2xl font-bold tracking-tight text-slate-900 sm:text-3xl">
   365→                Invoice Workspace
   366→              </h1>
   367→              <p className="mt-1 text-sm text-muted-foreground">
   368→                Upload, extract, and validate GST invoices
   369→              </p>
   370→            </div>
   371→          </motion.div>
   372→
   373→          <EmptyState
   374→            icon={FileUp}
   375→            title="No invoices yet"
   376→            description="Upload your first document to start processing and validating GST invoices."
   377→            action={{
   378→              label: 'Upload your first document',
   379→              onClick: () => fileInputRef.current?.click(),
   380→              icon: Upload,
   381→            }}
   382→          />
   383→        </div>
   384→        <input
   385→          ref={fileInputRef}
   386→          type="file"
   387→          multiple
   388→          accept=".csv,.xlsx,.xls,.json,.pdf"
   389→          className="hidden"
   390→          onChange={handleFileSelect}
   391→        />
   392→      </div>
   393→    );
   394→  }
   395→

   397→
   398→  const summaryCards = [
   399→    {
   400→      title: 'Total Invoices',
   401→      value: formatNumber(summary.total),
   402→      icon: FileText,
   403→      color: 'text-slate-600',
   404→      bgColor: 'bg-slate-50',
   405→    },
   406→    {
   407→      title: 'Approved',
   408→      value: formatNumber(summary.approved),
   409→      icon: CheckCircle2,
   410→      color: 'text-emerald-600',
   411→      bgColor: 'bg-emerald-50',
   412→    },
   413→    {
   414→      title: 'Pending',
   415→      value: formatNumber(summary.pending),
   416→      icon: Clock,
   417→      color: 'text-amber-600',
   418→      bgColor: 'bg-amber-50',
   419→    },
   420→    {
   421→      title: 'Tax Volume',
   422→      value: formatCurrency(summary.taxVolume),
   423→      icon: TrendingUp,
   424→      color: 'text-teal-600',
   425→      bgColor: 'bg-teal-50',
   426→    },
   427→    {
   428→      title: 'Risk Items',
   429→      value: formatNumber(summary.riskItems),
   430→      icon: ShieldAlert,
   431→      color: summary.riskItems > 0 ? 'text-red-600' : 'text-slate-600',
   432→      bgColor: summary.riskItems > 0 ? 'bg-red-50' : 'bg-slate-50',

   435→
   436→  return (
   437→    <div className="min-h-screen bg-gradient-to-b from-slate-50/80 to-white">
   438→      <div className="space-y-6 p-4 md:p-6 lg:p-8">
   439→        {/* ── Header ── */}
   440→        <motion.div
   441→          initial={{ opacity: 0, y: -12 }}
   442→          animate={{ opacity: 1, y: 0 }}

   444→          className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between"

   446→          <div>

   463→        {/* ── Summary Cards ── */}

   489→        {/* ── Upload Area ── */}

   495→              <div

   519→                {/* Uploading files indicator */}
