   252→// MAIN COMPONENT
   253→// ═══════════════════════════════════════════════════════════════════════════
   254→export default function WhiteLabelPage() {
   255→  // ── State ────────────────────────────────────────────────────────────────
   256→  const [loading, setLoading] = useState(true);
   257→  const [saving, setSaving] = useState(false);
   258→  const [settings, setSettings] = useState<FirmSettingsData>(DEFAULT_SETTINGS);
   259→  const [logoPreview, setLogoPreview] = useState<string | null>(null);
   260→  const [domainStatus, setDomainStatus] = useState<DomainStatus>('not_configured');
   261→  const fileInputRef = useRef<HTMLInputElement>(null);
   262→
   263→  // ── Data Fetching ────────────────────────────────────────────────────────
   264→  const fetchSettings = useCallback(async () => {
   265→    try {
   266→      setLoading(true);
   267→      const res = await fetch('/api/firm-settings');
   268→      if (res.ok) {
   269→        const data = await res.json();
   270→        const s = data.settings || data;
   271→        setSettings({
   272→          firmName: s.firmName ?? DEFAULT_SETTINGS.firmName,
   273→          logoUrl: s.logoUrl ?? DEFAULT_SETTINGS.logoUrl,
   274→          primaryColor: s.primaryColor ?? DEFAULT_SETTINGS.primaryColor,
   275→          accentColor: s.accentColor ?? DEFAULT_SETTINGS.accentColor,
   276→          customDomain: s.customDomain ?? DEFAULT_SETTINGS.customDomain,
   277→          emailFromName: s.emailFromName ?? DEFAULT_SETTINGS.emailFromName,
   278→          emailTemplate: s.emailTemplate ?? DEFAULT_SETTINGS.emailTemplate,
   279→        });
   280→        if (s.logoUrl) setLogoPreview(s.logoUrl);
   281→        if (s.customDomain) {
   282→          setDomainStatus('active');
   283→        }
   284→      }
   285→    } catch (err) {
   286→      console.error('Failed to fetch firm settings:', err);
   287→      toast.error('Failed to load settings', {
   288→        description: 'Using default values. Please try again.',
   289→      });
   290→    } finally {
   291→      setLoading(false);
   292→    }
   293→  }, []);
   294→
   295→  useEffect(() => {
   296→    fetchSettings();
   297→  }, [fetchSettings]);
   298→
   299→  // ── Handlers ─────────────────────────────────────────────────────────────
   300→  const handleSave = async () => {
   301→    try {
   302→      setSaving(true);
   303→      const res = await fetch('/api/firm-settings', {
   304→        method: 'PUT',
   305→        headers: { 'Content-Type': 'application/json' },
   306→        body: JSON.stringify({
   307→          firmName: settings.firmName,
   308→          logoUrl: settings.logoUrl,
   309→          primaryColor: settings.primaryColor,
   310→          accentColor: settings.accentColor,
   311→          customDomain: settings.customDomain,
   312→          emailFromName: settings.emailFromName,
   313→          emailTemplate: settings.emailTemplate,
   314→        }),
   315→      });
   316→
   317→      if (res.ok) {
   318→        const data = await res.json();
   319→        const s = data.settings || data;
   320→        setSettings((prev) => ({
   321→          ...prev,
   322→          id: s.id,
   323→          firmName: s.firmName ?? prev.firmName,
   324→          logoUrl: s.logoUrl ?? prev.logoUrl,
   325→          primaryColor: s.primaryColor ?? prev.primaryColor,
   326→          accentColor: s.accentColor ?? prev.accentColor,
   327→          customDomain: s.customDomain ?? prev.customDomain,
   328→          emailFromName: s.emailFromName ?? prev.emailFromName,
   329→          emailTemplate: s.emailTemplate ?? prev.emailTemplate,
   330→        }));
   331→        toast.success('Settings saved successfully', {
   332→          description: 'Your white label branding has been updated.',
   333→          icon: <CheckCircle2 className="h-4 w-4 text-emerald-500" />,
   334→        });
   335→      } else {
   336→        throw new Error('Failed to save');
   337→      }
   338→    } catch {
   339→      toast.error('Failed to save settings', {
   340→        description: 'Please check your inputs and try again.',
   341→        icon: <AlertCircle className="h-4 w-4 text-red-500" />,
   342→      });
   343→    } finally {
   344→      setSaving(false);
   345→    }
   346→  };
   347→
   348→  const handleLogoUpload = (e: React.ChangeEvent<HTMLInputElement>) => {
   349→    const file = e.target.files?.[0];
   350→    if (!file) return;
   351→
   352→    if (!file.type.startsWith('image/')) {
   353→      toast.error('Invalid file type', {
   354→        description: 'Please upload an image file (PNG, JPG, SVG).',
   355→      });
   356→      return;
   357→    }
   358→
   359→    if (file.size > 2 * 1024 * 1024) {
   360→      toast.error('File too large', {
   361→        description: 'Logo must be under 2MB.',
   362→      });
   363→      return;
   364→    }
   365→
   366→    const reader = new FileReader();
   367→    reader.onloadend = () => {
   368→      const dataUrl = reader.result as string;
   369→      setLogoPreview(dataUrl);
   370→      setSettings((prev) => ({ ...prev, logoUrl: dataUrl }));
   371→    };
   372→    reader.readAsDataURL(file);
   373→  };
   374→
   375→  const handleRemoveLogo = () => {
   376→    setLogoPreview(null);
   377→    setSettings((prev) => ({ ...prev, logoUrl: null }));
   378→    if (fileInputRef.current) {
   379→      fileInputRef.current.value = '';
   380→    }
   381→  };
   382→
   383→  const handleDragOver = (e: React.DragEvent) => {
   384→    e.preventDefault();
   385→    e.stopPropagation();
   386→  };
   387→
   388→  const handleDrop = (e: React.DragEvent) => {
   389→    e.preventDefault();
   390→    e.stopPropagation();
   391→    const file = e.dataTransfer.files?.[0];
   392→    if (file && file.type.startsWith('image/')) {
   393→      const reader = new FileReader();
   394→      reader.onloadend = () => {
   395→        const dataUrl = reader.result as string;
   396→        setLogoPreview(dataUrl);
   397→        setSettings((prev) => ({ ...prev, logoUrl: dataUrl }));
   398→      };
   399→      reader.readAsDataURL(file);
   400→    }
   401→  };
   402→
   403→  const getDomainStatusDisplay = () => {
   404→    switch (domainStatus) {
   405→      case 'active':
   406→        return (
   407→          <Badge className="bg-emerald-50 text-emerald-700 hover:bg-emerald-100 border-emerald-200 gap-1">
   408→            <CheckCircle2 className="h-3 w-3" />
   409→            Active
   410→          </Badge>
   411→        );
   412→      case 'pending':
   413→        return (
   414→          <Badge className="bg-amber-50 text-amber-700 hover:bg-amber-100 border-amber-200 gap-1">
   415→            <AlertCircle className="h-3 w-3" />
   416→            Pending
   417→          </Badge>
   418→        );
   419→      default:
   420→        return (
   421→          <Badge className="bg-slate-50 text-slate-600 hover:bg-slate-100 border-slate-200 gap-1">
   422→            <Globe className="h-3 w-3" />
   423→            Not Configured
   424→          </Badge>
   425→        );
   426→    }
   427→  };
   428→
   429→  // ── Render ───────────────────────────────────────────────────────────────
   430→  if (loading) {
   431→    return (
   432→      <div className="p-4 md:p-6 space-y-4">
   433→        <motion.div
   434→          initial={{ opacity: 0, y: -10 }}
   435→          animate={{ opacity: 1, y: 0 }}
   436→          transition={{ duration: 0.4 }}
   437→        >
   438→          <Skeleton className="h-10 w-64 mb-2" />
   439→          <Skeleton className="h-4 w-48" />
   440→        </motion.div>
   441→        <SettingsSkeleton />
   442→      </div>
   443→    );
   444→  }
   445→
   446→  return (
   447→    <div className="p-4 md:p-6 space-y-4">
   448→      {/* ═══ PAGE HEADER ═══ */}
   449→      <motion.div
   450→        initial={{ opacity: 0, y: -10 }}
   451→        animate={{ opacity: 1, y: 0 }}
   452→        transition={{ duration: 0.4 }}
   453→        className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3"
   454→      >
   455→        <div className="flex items-center gap-3">
   456→          <div className="flex items-center justify-center h-10 w-10 rounded-xl bg-gradient-to-br from-emerald-500 to-emerald-600 shadow-md shadow-emerald-600/20">
   457→            <Palette className="h-5 w-5 text-white" />
   458→          </div>
   459→          <div>
   460→            <h1 className="text-2xl md:text-3xl font-bold tracking-tight text-foreground">
   461→              White Label Settings
   462→            </h1>
   463→            <p className="text-sm text-muted-foreground mt-0.5">
   464→              Customize your firm&apos;s branding and appearance
   465→            </p>
   466→          </div>
   467→        </div>
   468→        <Badge
   469→          variant="outline"
   470→          className="gap-1.5 px-3 py-1.5 w-fit border-emerald-200 text-emerald-700 bg-emerald-50/80 dark:border-emerald-800 dark:text-emerald-400 dark:bg-emerald-950/40 font-medium"
   471→        >
   472→          <Eye className="h-3.5 w-3.5" />
   473→          Live Preview
   474→        </Badge>
   475→      </motion.div>
   476→
   477→      {/* ═══ MAIN CONTENT: Settings + Preview ═══ */}
   478→      <div className="grid grid-cols-1 lg:grid-cols-3 gap-4">
   479→        {/* ─── LEFT: Settings Cards ─── */}
   480→        <div className="lg:col-span-2 space-y-4">
   481→          {/* Firm Branding Section */}
   482→          <AnimatedCard delay={0.05}>
   483→            <CardHeader className="pb-3">
   484→              <CardTitle className="flex items-center gap-2 text-base">
   485→                <Palette className="h-5 w-5 text-emerald-500" />
   486→                Firm Branding
   487→              </CardTitle>
   488→              <CardDescription>
   489→                Configure your firm name, logo, and brand colors
   490→              </CardDescription>
   491→            </CardHeader>
   492→            <CardContent className="space-y-5">
   493→              {/* Firm Name */}
   494→              <div className="space-y-2">
   495→                <Label htmlFor="firmName" className="text-sm font-medium">
   496→                  Firm Name
   497→                </Label>
   498→                <Input
   499→                  id="firmName"
   500→                  value={settings.firmName}
   501→                  onChange={(e) =>
   502→                    setSettings((prev) => ({ ...prev, firmName: e.target.value }))
   503→                  }
   504→                  placeholder="Enter your firm name"
   505→                  className="transition-all duration-200 focus:ring-2 focus:ring-emerald-500/20"
   506→                />
   507→                <p className="text-xs text-muted-foreground">
   508→                  This name will appear in the sidebar and all client-facing pages
   509→                </p>
   510→              </div>
   511→
   512→              <Separator />
   513→
   514→              {/* Logo Upload */}
   515→              <div className="space-y-2">
   516→                <Label className="text-sm font-medium">Firm Logo</Label>
   517→                <div
   518→                  onDragOver={handleDragOver}
   519→                  onDrop={handleDrop}
   520→                  className="relative"
   521→                >
   522→                  {logoPreview ? (
   523→                    <div className="flex items-center gap-4 p-4 rounded-xl border border-border/50 bg-muted/20">
   524→                      <div className="h-16 w-16 rounded-xl overflow-hidden border border-border/50 shadow-sm shrink-0">
   525→                        <img
   526→                          src={logoPreview}
   527→                          alt="Firm Logo"
   528→                          className="h-full w-full object-cover"
   529→                        />
   530→                      </div>
   531→                      <div className="flex-1 min-w-0">
   532→                        <p className="text-sm font-medium text-foreground truncate">
   533→                          Logo uploaded
   534→                        </p>
   535→                        <p className="text-xs text-muted-foreground">
   536→                          PNG, JPG, or SVG — Max 2MB
   537→                        </p>
   538→                      </div>
   539→                      <Button
   540→                        variant="ghost"
   541→                        size="sm"
   542→                        onClick={handleRemoveLogo}
   543→                        className="h-8 w-8 p-0 text-muted-foreground hover:text-red-500 hover:bg-red-50 dark:hover:bg-red-950/30"
   544→                      >
   545→                        <X className="h-4 w-4" />
   546→                      </Button>
   547→                    </div>
   548→                  ) : (
   549→                    <label
   550→                      htmlFor="logo-upload"
   551→                      className="flex flex-col items-center justify-center h-32 rounded-xl border-2 border-dashed border-border/50 bg-muted/10 hover:bg-emerald-50/30 hover:border-emerald-300 dark:hover:bg-emerald-950/10 dark:hover:border-emerald-800 transition-all cursor-pointer group"
   552→                    >
   553→                      <div className="flex items-center justify-center h-10 w-10 rounded-xl bg-emerald-50 dark:bg-emerald-950/30 mb-2 group-hover:bg-emerald-100 dark:group-hover:bg-emerald-900/50 transition-colors">
   554→                        <Upload className="h-5 w-5 text-emerald-500" />
   555→                      </div>
   556→                      <span className="text-sm font-medium text-muted-foreground group-hover:text-emerald-700 dark:group-hover:text-emerald-400 transition-colors">
   557→                        Drag &amp; drop or click to upload
   558→                      </span>
   559→                      <span className="text-xs text-muted-foreground/60 mt-0.5">
   560→                        PNG, JPG, or SVG — Max 2MB
   561→                      </span>
   562→                    </label>
   563→                  )}
   564→                  <input
   565→                    ref={fileInputRef}
   566→                    id="logo-upload"
   567→                    type="file"
   568→                    accept="image/*"
   569→                    onChange={handleLogoUpload}
   570→                    className="hidden"
   571→                  />
   572→                </div>
   573→              </div>
   574→
   575→              <Separator />
   576→
   577→              {/* Color Pickers */}
   578→              <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   579→                {/* Primary Color */}
   580→                <div className="space-y-2">
   581→                  <Label
   582→                    htmlFor="primaryColor"
   583→                    className="text-sm font-medium flex items-center gap-2"
   584→                  >
