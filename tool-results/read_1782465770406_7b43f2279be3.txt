    56→const inrFmt = (n: number) => Math.round(n).toLocaleString('en-IN');
    57→const inrShort = (n: number) => {
    58→  if (!isFinite(n) || isNaN(n)) return '₹0';
    59→  const abs = Math.abs(n);
    60→  if (abs >= 10000000) return `₹${(n / 10000000).toFixed(2)} Cr`;
    61→  if (abs >= 100000) return `₹${(n / 100000).toFixed(2)} L`;
    62→  if (abs >= 1000) return `₹${(n / 1000).toFixed(1)}K`;
    63→  return '₹' + Math.round(n).toLocaleString('en-IN');
    64→};
    65→const nowISO = () => new Date().toISOString();
    66→const uid = (p: string) => `${p}_${Math.random().toString(36).slice(2, 10)}`;
    67→const daysAgo = (d: number) => {
    68→  const x = new Date();
    69→  x.setDate(x.getDate() - d);
    70→  return x.toISOString();
    71→};
    72→
    73→function scoreToLevel(score: number): RiskLevel {
    74→  if (score >= 75) return 'critical';
    75→  if (score >= 55) return 'high';
    76→  if (score >= 30) return 'medium';
    77→  return 'low';
    78→}
    79→
    80→function levelRank(l: RiskLevel): number {
    81→  return { low: 0, medium: 1, high: 2, critical: 3 }[l];
    82→}
    83→
    84→function filingDueDate(returnType: string, period: string): Date | null {
    85→  const m = period.match(/^(\d{4})-(\d{2})$/);
    86→  if (!m) return null;
    87→  const year = parseInt(m[1], 10);
    88→  const month = parseInt(m[2], 10);
    89→  if (month === 12) return new Date(year + 1, 0, 11);
    90→  const day = returnType === 'GSTR-1' ? 11 : 20;
    91→  return new Date(year, month, day);
    92→}

    96→interface RawRows {
    97→  clients: Array<{ id: string; gstin: string; tradeName: string; status: string; healthScore: number }>;
    98→  invoices: Array<{
    99→    id: string; invoiceNumber: string; invoiceDate: string; totalAmount: number;

   101→    status: string; period: string | null; buyerGstin: string | null; buyerName: string | null;
   102→  }>;
   103→  filings: Array<{ id: string; returnType: string; period: string; status: string; clientId: string; totalTax: number }>;
   104→  notices: Array<{ id: string; noticeType: string; status: string; noticeDate: string | null; dueDate: string | null; clientId: string }>;
   105→}

   107→async function fetchRawRows(): Promise<RawRows> {
   108→  const [clients, invoices, filings, notices] = await Promise.all([
   109→    db.client.findMany({
   110→      select: { id: true, gstin: true, tradeName: true, status: true, healthScore: true },
   111→      take: 1000,
   112→    }),
   113→    db.invoice.findMany({
   114→      select: {
   115→        id: true, invoiceNumber: true, invoiceDate: true, totalAmount: true,
   116→        taxableValue: true, cgst: true, sgst: true, igst: true,
   117→        status: true, period: true, buyerGstin: true, buyerName: true,
   118→      },
   119→      take: 5000,
   120→    }),
   121→    db.gSTRFiling.findMany({
   122→      select: { id: true, returnType: true, period: true, status: true, clientId: true, totalTax: true },
   123→      take: 2000,
   124→    }),
   125→    db.notice.findMany({
   126→      select: { id: true, noticeType: true, status: true, noticeDate: true, dueDate: true, clientId: true },
   127→      take: 500,
   128→    }) as Promise<Array<{ id: string; noticeType: string; status: string; noticeDate: string | null; dueDate: string | null; clientId: string }>>,
   129→  ]);
   130→  return { clients, invoices, filings, notices };
   131→}

   133→export function buildKnowledgeGraph(rows: RawRows, cfo: CFOResponse): KnowledgeGraph {
   134→  const nodes: GraphNode[] = [];
   135→  const edges: { id: string; source: string; target: string; type: RelationshipType; amount?: number; label?: string }[] = [];
   136→
   137→  const pushNode = (n: GraphNode) => nodes.push(n);
   138→  const pushEdge = (source: string, target: string, type: RelationshipType, amount?: number, label?: string) => {
   139→    edges.push({ id: uid('e'), source, target, type, amount, label });
   140→  };
   141→
   142→  // 1. Business node — the firm itself (the user's business). Single root.
   143→  const businessId = 'business:firm';
   144→  const firmName = 'Your Business';
   145→  pushNode({
   146→    id: businessId,
   147→    type: 'business',
   148→    entityId: 'firm',
   149→    label: firmName,
   150→    subtitle: 'The firm at the center of the graph',
   151→    amount: cfo.dashboard.revenue.thisMonth,
   152→    meta: { healthScore: cfo.dashboard.healthScore.overall, cash: cfo.dashboard.cash.currentBalance },
   153→    x: 0, y: 0,
   154→  });
   155→
   156→  // 2. Bank account node — derived from cash position
   157→  const bankId = 'bank-account:primary';
   158→  pushNode({
   159→    id: bankId,
   160→    type: 'bank-account',
   161→    entityId: 'primary',
   162→    label: 'Primary Bank Account',
   163→    subtitle: `Balance ${inrShort(cfo.dashboard.cash.currentBalance)}`,
   164→    amount: cfo.dashboard.cash.currentBalance,
   165→    meta: { runway: cfo.dashboard.cash.runwayDays },
   166→    x: -200, y: 0,
   167→  });
   168→  pushEdge(businessId, bankId, 'OWNS');
   169→
   170→  // 3. Employees — synthesize a small team
   171→  const employees = [
   172→    { id: 'employee:priya', name: 'Priya Sharma', role: 'Senior CA', handles: 'Returns + ITC' },
   173→    { id: 'employee:arjun', name: 'Arjun Patel', role: 'Accountant', handles: 'Reconciliation' },
   174→    { id: 'employee:neha',  name: 'Neha Singh',  role: 'Collections', handles: 'Follow-ups' },
   175→  ];
   176→  employees.forEach((emp, i) => {
   177→    pushNode({
   178→      id: emp.id,
   179→      type: 'employee',
   180→      entityId: emp.id,
   181→      label: emp.name,
   182→      subtitle: emp.role,
   183→      meta: { handles: emp.handles },
   184→      x: -250 + i * 60, y: 220,
   185→    });
   186→    pushEdge(businessId, emp.id, 'OWNS');
   187→  });
   188→
   189→  // 4. Clients — from Prisma
   190→  rows.clients.forEach((c, i) => {
   191→    const angle = (i / Math.max(rows.clients.length, 1)) * 2 * Math.PI;
   192→    const r = 320;
   193→    const nodeId = `client:${c.id}`;
   194→    pushNode({
   195→      id: nodeId,
   196→      type: 'client',
   197→      entityId: c.id,
   198→      label: c.tradeName || c.gstin,
   199→      subtitle: c.gstin,
   200→      meta: { status: c.status, healthScore: c.healthScore },
   201→      x: Math.cos(angle) * r,
   202→      y: Math.sin(angle) * r,
   203→    });
   204→    pushEdge(businessId, nodeId, 'OWNS');
   205→  });
   206→
   207→  // 5. Invoices — link to clients (by buyerGstin match) and to business
   208→  rows.invoices.forEach((inv, i) => {
   209→    const angle = (i / Math.max(rows.invoices.length, 1)) * 2 * Math.PI;
   210→    const r = 480;
   211→    const invNodeId = `invoice:${inv.id}`;
   212→    pushNode({
   213→      id: invNodeId,
   214→      type: 'invoice',
   215→      entityId: inv.id,
   216→      label: inv.invoiceNumber,
   217→      subtitle: `${inrShort(inv.totalAmount)} · ${inv.status}`,
   218→      amount: inv.totalAmount,
   219→      meta: { status: inv.status, period: inv.period ?? '', date: inv.invoiceDate },
   220→      x: Math.cos(angle) * r,
   221→      y: Math.sin(angle) * r,
   222→    });
   223→    // Link to client by buyerGstin
   224→    const matchedClient = rows.clients.find((c) => c.gstin === inv.buyerGstin);
   225→    if (matchedClient) {
   226→      pushEdge(`client:${matchedClient.id}`, invNodeId, 'GENERATES', inv.totalAmount);
   227→    } else {
   228→      // Otherwise link to business directly (sales invoice)
   229→      pushEdge(businessId, invNodeId, 'GENERATES', inv.totalAmount);
   230→    }
   231→  });
   232→
   233→  // 6. GST Returns — link to clients
   234→  rows.filings.forEach((f, i) => {
   235→    const angle = (i / Math.max(rows.filings.length, 1)) * 2 * Math.PI;
   236→    const r = 580;
   237→    const nodeId = `gst-return:${f.id}`;
   238→    const due = filingDueDate(f.returnType, f.period);
   239→    const daysLeft = due ? Math.ceil((due.getTime() - Date.now()) / (1000 * 60 * 60 * 24)) : null;
   240→    pushNode({
   241→      id: nodeId,
   242→      type: 'gst-return',
   243→      entityId: f.id,
   244→      label: `${f.returnType} · ${f.period}`,
   245→      subtitle: `${f.status} · tax ${inrShort(f.totalTax)}`,
   246→      amount: f.totalTax,
   247→      meta: { status: f.status, period: f.period, returnType: f.returnType, daysLeft: daysLeft ?? 0 },
   248→      x: Math.cos(angle) * r,
   249→      y: Math.sin(angle) * r,
   250→    });
   251→    pushEdge(`client:${f.clientId}`, nodeId, 'FILES', f.totalTax);
   252→  });
   253→
   254→  // 7. Notices — link to clients
   255→  rows.notices.forEach((n, i) => {
   256→    const angle = (i / Math.max(rows.notices.length, 1)) * 2 * Math.PI;
   257→    const r = 660;
   258→    const nodeId = `notice:${n.id}`;
   259→    pushNode({
   260→      id: nodeId,
   261→      type: 'notice',
   262→      entityId: n.id,
   263→      label: n.noticeType,
   264→      subtitle: `${n.status} · due ${n.dueDate || 'n/a'}`,
   265→      meta: { status: n.status, noticeDate: n.noticeDate ?? '', dueDate: n.dueDate ?? '' },
   266→      x: Math.cos(angle) * r,
   267→      y: Math.sin(angle) * r,
   268→    });
   269→    pushEdge(`client:${n.clientId}`, nodeId, 'RESPONDS_TO');
   270→    pushEdge(businessId, nodeId, 'CONNECTED_TO');
   271→  });
   272→
   273→  // 8. Vendors — derive from invoice buyerGstin where it's a purchase (status==='purchase')
   274→  // Simplification: synthesize 3 representative vendors from GST memory + client list
   275→  const vendorNames = ['Tata Steel Supplier', 'Reliance Logistics', 'Office Supplies Co.'];
   276→  vendorNames.forEach((vname, i) => {
   277→    const nodeId = `vendor:vendor-${i + 1}`;
   278→    pushNode({
   279→      id: nodeId,
   280→      type: 'vendor',
   281→      entityId: `vendor-${i + 1}`,
   282→      label: vname,
   283→      subtitle: 'Critical supplier',
   284→      amount: 100000 * (i + 1),
   285→      meta: { dependencyScore: 60 - i * 10 },
   286→      x: -400 + i * 60, y: -240,
   287→    });
   288→    pushEdge(businessId, nodeId, 'PAYS', 100000 * (i + 1));
   289→  });
   290→
   291→  // 9. Tasks — pull from CFO priority actions
   292→  cfo.brief.priorityActions.slice(0, 5).forEach((a, i) => {
   293→    const nodeId = `task:${a.id}`;
   294→    pushNode({
   295→      id: nodeId,
   296→      type: 'task',
   297→      entityId: a.id,
   298→      label: a.title,
   299→      subtitle: a.urgency,
   300→      amount: a.amount,
   301→      meta: { urgency: a.urgency, actionType: a.actionType },
   302→      x: 300 + i * 50, y: -200,
   303→    });
   304→    pushEdge(businessId, nodeId, 'CREATED_BY');
   305→  });
   306→
   307→  // 10. Reports — synthesize from reporting routines
   308→  const reports = [
   309→    { id: 'report:daily', name: 'Daily Business Brief', type: 'daily' },
   310→    { id: 'report:monthly', name: 'Monthly Compliance Report', type: 'monthly' },
   311→    { id: 'report:cashflow', name: 'Cash Flow Forecast', type: 'forecast' },
   312→  ];
   313→  reports.forEach((r, i) => {
   314→    pushNode({
   315→      id: r.id,
   316→      type: 'report',
   317→      entityId: r.id,
   318→      label: r.name,
   319→      subtitle: r.type,
   320→      meta: { type: r.type },
   321→      x: 350 + i * 60, y: 280,
   322→    });
   323→    pushEdge(businessId, r.id, 'GENERATES');
   324→  });
   325→
   326→  // 11. Conversation — single Oracle conversation meta-node
   327→  pushNode({
   328→    id: 'conversation:oracle',
   329→    type: 'conversation',
   330→    entityId: 'oracle',
   331→    label: 'Oracle Conversations',
   332→    subtitle: `${cfo.memory.insights.length} insights remembered`,
   333→    x: -450, y: 100,
   334→  });
   335→  pushEdge(businessId, 'conversation:oracle', 'CONNECTED_TO');
   336→
   337→  // 12. Predictions — link forecast numbers as prediction nodes
   338→  const predNodes: GraphNode[] = [
   339→    { id: 'prediction:revenue30', type: 'prediction', entityId: 'revenue30', label: 'Revenue 30d', subtitle: inrShort(cfo.predictions.revenue.thirtyDay), amount: cfo.predictions.revenue.thirtyDay, meta: { confidence: cfo.predictions.revenue.confidencePct }, x: 480, y: -80 },
   340→    { id: 'prediction:cash30', type: 'prediction', entityId: 'cash30', label: 'Cash 30d', subtitle: inrShort(cfo.predictions.cashFlow.monthlyPosition), amount: cfo.predictions.cashFlow.monthlyPosition, meta: { confidence: cfo.predictions.cashFlow.confidencePct }, x: 480, y: 20 },
   341→    { id: 'prediction:gst30', type: 'prediction', entityId: 'gst30', label: 'GST Liability 30d', subtitle: inrShort(cfo.predictions.gst.upcomingLiability), amount: cfo.predictions.gst.upcomingLiability, meta: { confidence: cfo.predictions.gst.confidencePct }, x: 480, y: 120 },
   342→  ];
   343→  predNodes.forEach((n) => {
   344→    pushNode(n);
   345→    pushEdge(businessId, n.id, 'PREDICTED_BY', n.amount);
   346→  });
   347→
   348→  // Index edges by type and nodes by type for stats
   349→  const nodeCountByType = {} as Record<NodeType, number>;
   350→  const edgeCountByType = {} as Record<RelationshipType, number>;
   351→  for (const n of nodes) nodeCountByType[n.type] = (nodeCountByType[n.type] || 0) + 1;
   352→  for (const e of edges) edgeCountByType[e.type] = (edgeCountByType[e.type] || 0) + 1;
   353→
   354→  return {
   355→    nodes,
   356→    edges: edges.map((e) => ({ ...e, weight: e.amount ? Math.min(10, Math.max(1, Math.log10(e.amount + 1))) : 1 })),
   357→    nodeCountByType,
   358→    edgeCountByType,
   359→  };
   360→}

   364→export function buildRiskGraph(graph: KnowledgeGraph, cfo: CFOResponse): RiskGraph {
   365→  const riskNodes: RiskNode[] = [];
   366→
   367→  // 1. Late-paying clients — from CFO memory client behavior
   368→  cfo.memory.clientBehavior.forEach((c) => {
   369→    const score =
   370→      c.riskLabel === 'High' ? 78 :
   371→      c.riskLabel === 'Medium' ? 50 :
   372→      c.riskLabel === 'Low' ? 18 : 8;
   373→    if (score < 30) return; // skip low-risk
   374→    const clientNode = graph.nodes.find((n) => n.type === 'client' && n.label === c.clientName);
   375→    const level = scoreToLevel(score);
   376→    riskNodes.push({
   377→      id: uid('risk'),
   378→      nodeId: clientNode?.id || `client:${c.clientName}`,
   379→      entityName: c.clientName,
   380→      entityType: 'client',
   381→      category: 'late_payment',
   382→      level,
   383→      score,
   384→      reasons: [
   385→        `${c.delays} delayed payment(s) recorded`,
   386→        `Average delay ${c.averageDelayDays} days`,
   387→        `Outstanding ${inrShort(c.totalOutstanding)}`,
   388→      ],
   389→      impact: `If ${c.clientName} delays further, cash flow impact up to ${inrShort(c.totalOutstanding)}.`,
   390→      amountAtRisk: c.totalOutstanding,
   391→      recommendation:
   392→        level === 'critical' || level === 'high'
   393→          ? 'Escalate to tier-2 follow-up; consider revised payment terms.'
   394→          : 'Send automated reminders 7 days before due date.',
   395→    });
   396→  });
   397→
   398→  // 2. GST notice risk — for each open notice
   399→  graph.nodes.filter((n) => n.type === 'notice').forEach((n) => {
   400→    const score = 70;
   401→    riskNodes.push({
   402→      id: uid('risk'),
   403→      nodeId: n.id,
   404→      entityName: n.label,
   405→      entityType: 'notice',
   406→      category: 'gst_notice',
   407→      level: 'high',
   408→      score,
   409→      reasons: [
   410→        `Notice type: ${n.label}`,
   411→        'Statutory response required within timeline',
   412→        'Non-response → penalty + escalation',
   413→      ],
   414→      impact: 'Non-response may trigger penalty and freeze ITC claims.',
   415→      recommendation: 'Respond within 7 days. Engage GST counsel if escalation tier.',
   416→    });
   417→  });
   418→
   419→  // 3. Cash flow risk — from CFO risks
   420→  const cashRisk = cfo.risks.find((r) => r.category === 'cash' && r.level !== 'low');
   421→  if (cashRisk) {
   422→    const score = cashRisk.score;
   423→    riskNodes.push({
   424→      id: uid('risk'),
   425→      nodeId: 'bank-account:primary',
   426→      entityName: 'Primary Bank Account',
   427→      entityType: 'bank-account',
   428→      category: 'cash_flow',
   429→      level: scoreToLevel(score),
   430→      score,
   431→      reasons: cashRisk.reasons,
   432→      impact: `Runway ${cfo.dashboard.cash.runwayDays || '∞'} days; daily burn ${inrShort(cfo.dashboard.cash.burnRatePerDay)}.`,
   433→      amountAtRisk: cfo.dashboard.cash.currentBalance,
   434→      recommendation: cashRisk.recommendation || 'Review cash position and accelerate collections.',
   435→    });
   436→  }
   437→
   438→  // 4. Compliance risk
   439→  const complianceRisk = cfo.risks.find((r) => r.category === 'compliance' && r.level !== 'low');
   440→  if (complianceRisk) {
   441→    const score = complianceRisk.score;
   442→    riskNodes.push({
   443→      id: uid('risk'),
   444→      nodeId: 'business:firm',
   445→      entityName: 'Business (Compliance)',
   446→      entityType: 'business',
   447→      category: 'compliance',
   448→      level: scoreToLevel(score),
   449→      score,
   450→      reasons: complianceRisk.reasons,
   451→      impact: 'Late fee ₹50/day + 18% interest p.a. on overdue returns.',
   452→      amountAtRisk: cfo.dashboard.gst.liability,
   453→      recommendation: complianceRisk.recommendation || 'File overdue returns immediately to avoid further penalty.',
   454→    });
   455→  }
   456→
   457→  // 5. Revenue concentration — if one client > 35% of revenue
   458→  const totalRevenue = cfo.dashboard.revenue.thisMonth || 1;
   459→  graph.nodes
   460→    .filter((n) => n.type === 'client')
   461→    .forEach((cNode) => {
   462→      const clientInvoices = graph.edges
   463→        .filter((e) => e.source === cNode.id && e.type === 'GENERATES')
   464→        .reduce((sum, e) => sum + (e.amount || 0), 0);
   465→      const concentration = clientInvoices / totalRevenue;
   466→      if (concentration > 0.35 && clientInvoices > 0) {
   467→        const score = Math.min(85, Math.round(concentration * 100));
   468→        riskNodes.push({
   469→          id: uid('risk'),
   470→          nodeId: cNode.id,
   471→          entityName: cNode.label,
   472→          entityType: 'client',
   473→          category: 'revenue_concentration',
   474→          level: scoreToLevel(score),
   475→          score,
   476→          reasons: [
   477→            `Generates ${(concentration * 100).toFixed(0)}% of revenue`,
   478→            `Revenue exposure ${inrShort(clientInvoices)}`,
   479→            'Single-point-of-failure risk',
   480→          ],
   481→          impact: `Loss of this client → ${(concentration * 100).toFixed(0)}% revenue drop.`,
   482→          amountAtRisk: clientInvoices,
   483→          recommendation: 'Diversify client base; lock in multi-year contract.',
   484→        });
   485→      }
   486→    });
   487→
   488→  // 6. Vendor dependency — synthesized
   489→  graph.nodes.filter((n) => n.type === 'vendor').forEach((v) => {
   490→    const depScore = (v.meta?.dependencyScore as number) || 40;
   491→    if (depScore < 50) return;
   492→    riskNodes.push({
   493→      id: uid('risk'),
   494→      nodeId: v.id,
   495→      entityName: v.label,
   496→      entityType: 'vendor',
   497→      category: 'vendor_dependency',
   498→      level: scoreToLevel(depScore),
   499→      score: depScore,
   500→      reasons: [
   501→        'Single source for critical supply',
   502→        `Annual spend ${inrShort(v.amount || 0)}`,
   503→      ],
   504→      impact: `Disruption → procurement delay, potential revenue loss ${inrShort((v.amount || 0) * 0.3)}.`,
   505→      amountAtRisk: v.amount,
   506→      recommendation: 'Identify backup vendor; negotiate 60-day stock buffer.',
   507→    });
   508→  });
   509→
   510→  // 7. Fraud risk — heuristic: invoice with riskLevel 'high' in invoice book
   511→  // (Invoices with riskScore >= 70 would be flagged; here we synthesize 1 if no other critical risk)
   512→  const hasCritical = riskNodes.some((r) => r.level === 'critical');
   513→  if (!hasCritical) {
   514→    const suspiciousInvoices = graph.nodes.filter((n) => n.type === 'invoice' && (n.meta?.status === 'flagged'));
   515→    if (suspiciousInvoices.length > 0) {
   516→      suspiciousInvoices.slice(0, 2).forEach((inv) => {
   517→        riskNodes.push({
   518→          id: uid('risk'),
   519→          nodeId: inv.id,
   520→          entityName: inv.label,
   521→          entityType: 'invoice',
   522→          category: 'fraud',
   523→          level: 'high',
   524→          score: 65,
   525→          reasons: ['Round-trip pattern detected', 'Vendor mismatch in 2B'],
   526→          impact: 'Potential ITC reversal + penalty up to 100% of tax.',
   527→          amountAtRisk: inv.amount,
   528→          recommendation: 'Hold ITC claim; verify vendor GSTIN status.',
   529→        });
   530→      });
   531→    }
   532→  }
   533→
   534→  // Aggregate stats
   535→  const countByLevel = { low: 0, medium: 0, high: 0, critical: 0 } as Record<RiskLevel, number>;
   536→  const countByCategory = {} as Record<RiskCategory, number>;
   537→  riskNodes.forEach((r) => {
   538→    countByLevel[r.level]++;
   539→    countByCategory[r.category] = (countByCategory[r.category] || 0) + 1;
   540→  });
   541→  const overallLevel = riskNodes.reduce<RiskLevel>(
   542→    (max, r) => (levelRank(r.level) > levelRank(max) ? r.level : max),
   543→    'low',
   544→  );
   545→  const topRisks = [...riskNodes].sort((a, b) => b.score - a.score).slice(0, 5);
   546→
   547→  // Stamp risk on graph nodes
   548→  riskNodes.forEach((r) => {
   549→    const node = graph.nodes.find((n) => n.id === r.nodeId);
   550→    if (node) {
   551→      node.riskScore = r.score;
   552→      node.riskLevel = r.level;
   553→    }
   554→  });
   555→
   556→  return { nodes: riskNodes, countByLevel, countByCategory, overallLevel, topRisks };
   557→}

   617→export function buildDependencyGraph(graph: KnowledgeGraph, cfo: CFOResponse): DependencyGraph {

   619→  const topRevenueClients: DependencyEdge[] = graph.nodes
   620→    .filter((n) => n.type === 'client')

   623→        .filter((e) => e.source === cNode.id && e.type === 'GENERATES')

   631→      };
   632→    })
   633→    .sort((a, b) => (b.amount || 0) - (a.amount || 0))
   634→    .slice(0, 5);
   635→
   636→  // Critical vendors
   637→  const criticalVendors: DependencyEdge[] = graph.nodes
   638→    .filter((n) => n.type === 'vendor')
   639→    .map((v) => ({

   641→      toId: v.id, toName: v.label, toType: 'vendor' as NodeType,
   642→      relationship: 'PAYS' as RelationshipType,
   643→      amount: v.amount,
   644→      note: `Annual spend ${inrShort(v.amount || 0)} · dependency score ${v.meta?.dependencyScore ?? 40}/100`,
   645→    }));
   646→
   647→  // Notices affecting cash flow
   648→  const noticesAffectingCashFlow: DependencyEdge[] = graph.nodes
   649→    .filter((n) => n.type === 'notice')
   650→    .slice(0, 5)
   651→    .map((n) => ({
   652→      fromId: n.id, fromName: n.label, fromType: 'notice' as NodeType,
   653→      toId: 'bank-account:primary', toName: 'Primary Bank Account', toType: 'bank-account' as NodeType,
   654→      relationship: 'CONNECTED_TO' as RelationshipType,
   655→      note: 'Penalty + ITC freeze risk if not responded in time.',
   656→    }));
   657→
   658→  // Invoices linked to overdue returns — match by client
   659→  const overdueReturns = graph.nodes.filter((n) => n.type === 'gst-return' && (n.meta?.status === 'overdue' || ((n.meta?.daysLeft as number) || 0) < 0));
   660→  const invoicesLinkedToOverdueReturns: DependencyEdge[] = [];
   661→  overdueReturns.slice(0, 5).forEach((ret) => {
   662→    const clientIdEdge = graph.edges.find((e) => e.target === ret.id && e.type === 'FILES');
   663→    if (!clientIdEdge) return;
   664→    const clientNode = graph.nodes.find((n) => n.id === clientIdEdge.source);
   665→    if (!clientNode) return;
   666→    const invEdge = graph.edges.find((e) => e.source === clientNode.id && e.type === 'GENERATES');
   667→    if (!invEdge) return;
   668→    const invNode = graph.nodes.find((n) => n.id === invEdge.target);
   669→    if (!invNode) return;
   670→    invoicesLinkedToOverdueReturns.push({
   671→      fromId: invNode.id, fromName: invNode.label, fromType: 'invoice' as NodeType,
   672→      toId: ret.id, toName: ret.label, toType: 'gst-return' as NodeType,
   673→      relationship: 'CONNECTED_TO' as RelationshipType,
   674→      amount: invNode.amount,
   675→      note: `Invoice ${inrShort(invNode.amount || 0)} linked to overdue ${ret.label}.`,
   676→    });
   677→  });
   678→
   679→  // Employees and their clients
   680→  const employeesAndTheirClients: DependencyEdge[] = [];
   681→  const employees = graph.nodes.filter((n) => n.type === 'employee');
   682→  const clients = graph.nodes.filter((n) => n.type === 'client');
   683→  employees.forEach((emp, i) => {
   684→    // Round-robin assign clients to employees for the demo
   685→    const assigned = clients.slice(i, i + 2);

   687→      employeesAndTheirClients.push({
   688→        fromId: emp.id, fromName: emp.label, fromType: 'employee' as NodeType,
   689→        toId: cNode.id, toName: cNode.label, toType: 'client' as NodeType,
   690→        relationship: 'WORKS_WITH' as RelationshipType,
   691→        note: `${emp.label} manages ${cNode.label}'s compliance + collections.`,
   692→      });
   693→    });
   694→  });
   695→
   696→  return {
   697→    topRevenueClients,
   698→    criticalVendors,
   699→    noticesAffectingCashFlow,
   700→    invoicesLinkedToOverdueReturns,
   701→    employeesAndTheirClients,
   702→  };
   703→}

   781→  { intent: 'critical_vendor', phrases: ['critical vendor', 'key supplier', 'most important vendor'] },
   782→  { intent: 'collection_bottleneck', phrases: ['collection bottleneck', 'collection slow', 'why collection'] },

   785→export function executeQuery(text: string, state: GraphState): GraphQueryResult {
   786→  const lower = text.toLowerCase().trim();
   787→  let matched: GraphQueryIntent = 'unknown';
   788→  let bestPhrases: string[] = [];
   789→
   790→  for (const rule of QUERY_RULES) {
   791→    const hits = rule.phrases.filter((p) => lower.includes(p));
   792→    if (hits.length > 0 && hits.length > bestPhrases.length) {
   793→      matched = rule.intent;
   794→      bestPhrases = hits;
   795→    }
   796→  }
   797→
   798→  const confidence = matched === 'unknown' ? 0.15 : Math.min(0.95, 0.6 + bestPhrases.length * 0.15);
   799→  const bullets: string[] = [];
   800→  const relatedNodeIds: string[] = [];
   801→  const relatedRiskIds: string[] = [];
   802→  let answer = '';
   803→  let spokenAck = '';
   804→
   805→  switch (matched) {
   806→    case 'why_revenue_drop': {
   807→      spokenAck = "I've traced the revenue drop through your business graph.";

   814→        relatedNodeIds.push(reason.nodeId);
   815→        relatedRiskIds.push(reason.id);

   819→        bullets.push('Review invoice pipeline for delays');
   820→      }
   821→      relatedNodeIds.push('business:firm', 'prediction:revenue30');
   822→      break;
   823→    }
   824→
   825→    case 'cash_flow_down': {
   826→      spokenAck = "I've analyzed the cash flow chain through your graph.";

   920→        bullets.push(`${a.fromName} → ${a.toName}`);
   921→        relatedNodeIds.push(a.fromId, a.toId);
   922→      });
   923→      break;
   924→    }
   925→
   926→    case 'most_profitable': {
   927→      spokenAck = "I've ranked clients by revenue contribution.";
   928→      const top = state.dependencyGraph.topRevenueClients[0];
   929→      answer = top
   930→        ? `${top.fromName} is your most profitable client at ${inrShort(top.amount || 0)}.`
   931→        : 'No revenue data yet.';
   932→      state.dependencyGraph.topRevenueClients.slice(0, 5).forEach((c) => {
   933→        bullets.push(`${c.fromName}: ${inrShort(c.amount || 0)}`);
   934→        relatedNodeIds.push(c.fromId);
   935→      });
   936→      break;
   937→    }
   938→
   939→    case 'critical_vendor': {
   940→      spokenAck = "I've identified your critical vendors from the dependency graph.";
   941→      const top = state.dependencyGraph.criticalVendors[0];
   942→      answer = top
   943→        ? `${top.toName} is your most critical vendor — ${top.note}`
   944→        : 'No critical vendors flagged.';
   945→      state.dependencyGraph.criticalVendors.forEach((v) => {
   946→        bullets.push(`${v.toName}: ${v.note}`);
   947→        relatedNodeIds.push(v.toId);
   948→      });
   949→      break;
   950→    }
   951→
   952→    case 'collection_bottleneck': {
   953→      spokenAck = "I've traced collection bottlenecks through your client-bank chain.";
   954→      const chains = state.relationshipChains;
   955→      answer = chains.length
   956→        ? `${chains.length} collection chain(s) detected. Top bottleneck: ${chains[0].title} with ${inrShort(chains[0].totalImpact)} at risk.`
   957→        : 'No collection bottlenecks detected.';
   958→      chains.slice(0, 3).forEach((c) => {
   959→        bullets.push(`${c.title}: ${inrShort(c.totalImpact)} impact (${c.riskLevel})`);
   960→        c.steps.forEach((s) => relatedNodeIds.push(s.nodeId));
   961→      });
   962→      break;
   963→    }
   964→
   965→    case 'unknown':
   966→    default: {
   967→      spokenAck = "I've searched the business graph for an answer.";
   968→      answer = "I couldn't map that to a specific graph query, but I've scanned the full knowledge graph. Try asking about revenue drop, risky clients, overdue invoices, vendors, or employee-client assignments.";
   969→      bullets.push(`Graph contains ${state.knowledgeGraph.nodes.length} nodes and ${state.knowledgeGraph.edges.length} edges`);
   970→      bullets.push(`${state.riskGraph.countByLevel.high + state.riskGraph.countByLevel.critical} active high/critical risks`);
   971→      break;
   972→    }
   973→  }
   974→
   975→  return {
   976→    rawText: text,
   977→    intent: matched,
   978→    confidence,
   979→    answer,
   980→    bullets,
   981→    relatedNodeIds: Array.from(new Set(relatedNodeIds)),
   982→    relatedRiskIds: Array.from(new Set(relatedRiskIds)),
   983→    spokenAck,
   984→  };
   985→}

  1312→export function buildClientSubgraph(state: GraphState, clientId: string): ClientSubgraph | null {
  1313→  const centerNode = state.knowledgeGraph.nodes.find((n) => n.id === clientId || (n.type === 'client' && n.entityId === clientId));
  1314→  if (!centerNode) return null;
  1315→
  1316→  // BFS 2 hops from center
  1317→  const centerId = centerNode.id;
  1318→  const visibleIds = new Set<string>([centerId]);
  1319→  // First hop
  1320→  state.knowledgeGraph.edges.forEach((e) => {
  1321→    if (e.source === centerId) visibleIds.add(e.target);
  1322→    if (e.target === centerId) visibleIds.add(e.source);
  1323→  });
  1324→  // Second hop
  1325→  const firstHop = new Set(visibleIds);
  1326→  state.knowledgeGraph.edges.forEach((e) => {
  1327→    if (firstHop.has(e.source)) visibleIds.add(e.target);
  1328→    if (firstHop.has(e.target)) visibleIds.add(e.source);
  1329→  });
  1330→
  1331→  const subNodes = state.knowledgeGraph.nodes.filter((n) => visibleIds.has(n.id));
  1332→  const subEdges = state.knowledgeGraph.edges.filter((e) => visibleIds.has(e.source) && visibleIds.has(e.target));
  1333→
  1334→  const riskForClient = state.riskGraph.nodes.find((r) => r.nodeId === centerId);
  1335→  const relationshipChain = state.relationshipChains.find((c) => c.steps[0]?.nodeId === centerId);
  1336→  const insights = state.insights.filter((i) => i.relatedNodeIds.includes(centerId));
  1337→
  1338→  return {
  1339→    centerNodeId: centerId,
  1340→    graph: { nodes: subNodes, edges: subEdges, nodeCountByType: {} as any, edgeCountByType: {} as any },
  1341→    riskForClient,
  1342→    relationshipChain,
  1343→    insights,
  1344→  };
  1345→}
