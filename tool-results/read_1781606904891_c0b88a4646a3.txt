    75→const formatINR = (num: number): string => {
    76→  const str = num.toLocaleString('en-IN');
    77→  return '₹' + str;
    78→};
    79→
    80→const formatDate = (dateStr: string | null): string => {
    81→  if (!dateStr) return '—';
    82→  const d = new Date(dateStr);
    83→  const dd = String(d.getDate()).padStart(2, '0');
    84→  const mm = String(d.getMonth() + 1).padStart(2, '0');
    85→  const yyyy = d.getFullYear();
    86→  return `${dd}/${mm}/${yyyy}`;
    87→};

   694→function Sparkline({ data, color = '#10b981', width = 120, height = 36 }: {
   695→  data: number[];
   696→  color?: string;
   697→  width?: number;
   698→  height?: number;
   699→}) {
   700→  if (data.length < 2) return null;
   701→  const max = Math.max(...data);
   702→  const min = Math.min(...data);
   703→  const range = max - min || 1;
   704→  const points = data.map((v, i) => {
   705→    const x = (i / (data.length - 1)) * width;
   706→    const y = height - ((v - min) / range) * (height - 4) - 2;
   707→    return `${x},${y}`;
   708→  }).join(' ');
   709→
   710→  return (
   711→    <svg width={width} height={height} className="inline-block">
   712→      <polyline
   713→        points={points}
   714→        fill="none"
   715→        stroke={color}
   716→        strokeWidth="2"
   717→        strokeLinecap="round"
   718→        strokeLinejoin="round"
   719→      />
   720→      {data.length > 0 && (
   721→        <circle
   722→          cx={(data.length - 1) / (data.length - 1) * width}
   723→          cy={height - ((data[data.length - 1] - min) / range) * (height - 4) - 2}
   724→          r="3"
   725→          fill={color}
   726→        />
   727→      )}
   728→    </svg>
   729→  );
   730→}

   736→function BarChart({ data, labels, color = '#10b981', width = 300, height = 140 }: {
   737→  data: number[];
   738→  labels: string[];
   739→  color?: string;
   740→  width?: number;
   741→  height?: number;
   742→}) {
   743→  const max = Math.max(...data, 1);
   744→  const barWidth = (width - 40) / data.length - 8;
   745→  const chartHeight = height - 30;
   746→
   747→  return (
   748→    <svg width={width} height={height} className="inline-block">
   749→      {data.map((v, i) => {
   750→        const barH = (v / max) * (chartHeight - 10);
   751→        const x = 30 + i * ((width - 40) / data.length) + 4;
   752→        const y = chartHeight - barH;
   753→        return (
   754→          <g key={i}>
   755→            <rect
   756→              x={x}
   757→              y={y}
   758→              width={barWidth}
   759→              height={barH}
   760→              fill={color}
   761→              opacity={0.7 + (i / data.length) * 0.3}
   762→              rx="3"

   791→function GaugeChart({ value, maxValue = 100, size = 180 }: {
   792→  value: number;
   793→  maxValue?: number;
   794→  size?: number;
   795→}) {
   796→  const pct = value / maxValue;
   797→  const cx = size / 2;
   798→  const cy = size / 2 + 10;
   799→  const r = size / 2 - 20;
   800→  const startAngle = -225;
   801→  const endAngle = 45;
   802→  const totalAngle = endAngle - startAngle;
   803→  const currentAngle = startAngle + pct * totalAngle;
   804→
   805→  const polarToCartesian = (angle: number) => {
   806→    const rad = (angle * Math.PI) / 180;
   807→    return {
   808→      x: cx + r * Math.cos(rad),
   809→      y: cy + r * Math.sin(rad),
   810→    };
   811→  };
   812→
   813→  const pStart = polarToCartesian(startAngle);
   814→  const pEnd = polarToCartesian(endAngle);
   815→  const pCurrent = polarToCartesian(currentAngle);
   816→  const largeArcFlag = pct > 0.5 ? 1 : 0;
   817→
   818→  const bgColor = '#e2e8f0';
   819→  const fillColor = value >= 80 ? '#10b981' : value >= 60 ? '#f59e0b' : '#ef4444';
   820→
   821→  return (
   822→    <svg width={size} height={size / 2 + 30}>
   823→      {/* Background arc */}
   824→      <path
   825→        d={`M ${pStart.x} ${pStart.y} A ${r} ${r} 0 1 1 ${pEnd.x} ${pEnd.y}`}
   826→        fill="none"
   827→        stroke={bgColor}
   828→        strokeWidth="14"
   829→        strokeLinecap="round"
   830→      />
   831→      {/* Value arc */}
   832→      <path
   833→        d={`M ${pStart.x} ${pStart.y} A ${r} ${r} 0 ${largeArcFlag} 1 ${pCurrent.x} ${pCurrent.y}`}
   834→        fill="none"
   835→        stroke={fillColor}
   836→        strokeWidth="14"
   837→        strokeLinecap="round"
   838→        className="transition-all duration-1000"
   839→      />
   840→      {/* Value text */}
   841→      <text
   842→        x={cx}
   843→        y={cy - 5}
   844→        textAnchor="middle"
   845→        className="text-3xl font-bold"
   846→        fill={fillColor}
   847→      >
   848→        {value}
   849→      </text>
   850→      <text
   851→        x={cx}
   852→        y={cy + 15}
   853→        textAnchor="middle"
   854→        className="text-[10px] fill-slate-400"
   855→      >
   856→        Moat Strength
   857→      </text>
   858→    </svg>
   859→  );
   860→}
