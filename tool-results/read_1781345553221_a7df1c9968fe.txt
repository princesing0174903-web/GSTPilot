    71→// ─── Helpers ─────────────────────────────────────────────────────────────────
    72→function formatRelativeTime(ts: string) {
    73→  const diff = Date.now() - new Date(ts).getTime();
    74→  const m = Math.floor(diff / 60000);
    75→  if (m < 1) return 'Just now';
    76→  if (m < 60) return `${m}m ago`;
    77→  const h = Math.floor(m / 60);
    78→  if (h < 24) return `${h}h ago`;
    79→  const d = Math.floor(h / 24);
    80→  if (d < 7) return `${d}d ago`;
    81→  return new Date(ts).toLocaleDateString();
    82→}
    83→
    84→function formatFileSize(bytes: number) {
    85→  if (bytes < 1024) return `${bytes} B`;
    86→  if (bytes < 1048576) return `${(bytes / 1024).toFixed(1)} KB`;
    87→  return `${(bytes / 1048576).toFixed(1)} MB`;
    88→}

    90→function getFileIcon(ft: DocumentType) {
    91→  switch (ft) {
    92→    case 'pdf': return <FileText className="h-4 w-4 text-red-500" />;
    93→    case 'excel': case 'csv': return <FileSpreadsheet className="h-4 w-4 text-emerald-600" />;
    94→    case 'json': return <FileJson className="h-4 w-4 text-amber-500" />;
    95→    case 'image': return <Image className="h-4 w-4 text-purple-500" alt="" />;
    96→    default: return <FileIcon className="h-4 w-4 text-slate-400" />;
    97→  }
    98→}

   100→function getActivityIcon(type: string) {
   101→  if (type.includes('client')) return <Building2 className="h-3.5 w-3.5" />;
   102→  if (type.includes('document') || type.includes('upload')) return <Upload className="h-3.5 w-3.5" />;
   103→  if (type.includes('return') && type.includes('filed')) return <Send className="h-3.5 w-3.5" />;
   104→  if (type.includes('return')) return <FileText className="h-3.5 w-3.5" />;
   105→  if (type.includes('recon')) return <GitCompareArrows className="h-3.5 w-3.5" />;
   106→  if (type.includes('mismatch')) return <CheckCircle2 className="h-3.5 w-3.5" />;
   107→  return <Activity className="h-3.5 w-3.5" />;
   108→}
   109→
   110→function getActivityColor(type: string) {
   111→  if (type.includes('added') || type.includes('filed') || type.includes('resolved') || type.includes('processed')) return 'text-emerald-600 bg-emerald-50';
   112→  if (type.includes('deleted') || type.includes('failed')) return 'text-red-600 bg-red-50';
   113→  if (type.includes('updated') || type.includes('uploaded')) return 'text-blue-600 bg-blue-50';
   114→  return 'text-slate-600 bg-slate-50';
   115→}
