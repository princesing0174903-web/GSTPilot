    82→function StockByCategoryChart() {
    83→  const data = [
    84→    { label: 'Printers', value: 27, color: '#10b981' },
    85→    { label: 'Storage', value: 156, color: '#34d399' },
    86→    { label: 'Networking', value: 7, color: '#f59e0b' },
    87→    { label: 'Monitors', value: 8, color: '#6ee7b7' },
    88→    { label: 'Accessories', value: 156, color: '#a7f3d0' },
    89→    { label: 'Power', value: 12, color: '#fbbf24' },
    90→    { label: 'Memory', value: 34, color: '#94a3b8' },
    91→    { label: 'Processors', value: 4, color: '#d1d5db' },
    92→  ]
    93→  const maxVal = Math.max(...data.map(d => d.value))
    94→  const barH = 26
    95→  const gap = 6
    96→  const h = data.length * (barH + gap) + 30
    97→  const chartW = 280
    98→
    99→  return (
   100→    <svg viewBox={`0 0 ${chartW + 100} ${h}`} className="w-full h-auto">
   101→      {data.map((d, i) => {
   102→        const w = (d.value / maxVal) * chartW
   103→        return (
   104→          <g key={d.label} transform={`translate(0, ${i * (barH + gap) + 15})`}>
   105→            <text x="0" y={barH / 2 + 4} className="text-[10px] fill-slate-600" textAnchor="start">{d.label}</text>
   106→            <rect x="70" y="0" width={w} height={barH} rx="4" fill={d.color} opacity="0.85">
   107→              <animate attributeName="width" from="0" to={w} dur="0.6s" fill="freeze" />
   108→            </rect>
   109→            <text x={70 + w + 6} y={barH / 2 + 4} className="text-[9px] fill-slate-500">
   110→              {d.value} units
   111→            </text>
   112→          </g>
   113→        )
   114→      })}
   115→    </svg>
   116→  )
   117→}

   119→function WarehouseCapacityChart() {
   120→  const data = warehouses
   121→  const w = 300
   122→  const h = 140
   123→  const padL = 40
   124→  const padB = 25
   125→  const padT = 10
   126→  const chartW = w - padL - 10
   127→  const chartH = h - padB - padT
   128→  const barW = chartW / data.length * 0.6
   129→  const gapX = chartW / data.length
   130→
   131→  return (
   132→    <svg viewBox={`0 0 ${w} ${h}`} className="w-full h-auto">
   133→      {[0, 1, 2, 3, 4].map(i => {
   134→        const y = padT + (i / 4) * chartH
   135→        return <line key={i} x1={padL} y1={y} x2={w - 10} y2={y} stroke="#e2e8f0" strokeWidth="0.5" />
   136→      })}
   137→      {data.map((d, i) => {
   138→        const x = padL + i * gapX + (gapX - barW) / 2
   139→        const barH = (d.capacity / 100) * chartH
   140→        const fillColor = d.capacity > 80 ? '#ef4444' : d.capacity > 60 ? '#f59e0b' : '#10b981'
   141→        return (
   142→          <g key={d.code}>
   143→            <rect x={x} y={padT + chartH - barH} width={barW} height={barH} rx="3" fill={fillColor} opacity="0.8">
   144→              <animate attributeName="height" from="0" to={barH} dur="0.5s" fill="freeze" />
   145→              <animate attributeName="y" from={padT + chartH} to={padT + chartH - barH} dur="0.5s" fill="freeze" />
   146→            </rect>
   147→            <text x={x + barW / 2} y={padT + chartH - barH - 5} className="text-[8px] fill-slate-600" textAnchor="middle">{d.capacity}%</text>
   148→            <text x={x + barW / 2} y={h - 6} className="text-[8px] fill-slate-400" textAnchor="middle">{d.city}</text>
   149→          </g>
   150→        )
   151→      })}
   152→    </svg>
   153→  )
   154→}
