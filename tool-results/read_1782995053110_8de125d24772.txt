   226→
   227→  const pendingApprovals = useMemo(() =>
   228→    approvals.filter((a) => a.status === 'pending' || a.status === 'in_review'),
   229→  [approvals]);
   230→
   231→  const completedApprovals = useMemo(() =>
   232→    approvals.filter((a) => a.status === 'approved' || a.status === 'rejected' || a.status === 'changes_requested'),
   233→  [approvals]);
   234→
   235→  const displayList = activeTab === 'pending' ? pendingApprovals : completedApprovals;
   236→
   237→  const filteredList = useMemo(() => {
   238→    let result = [...displayList];
   239→    if (filterType !== 'all') result = result.filter((a) => a.type === filterType);
   240→    if (filterPriority !== 'all') result = result.filter((a) => a.priority === filterPriority);
   241→    return result;

   249→      const lastStep = a.steps.filter((s) => s.status === 'approved').pop();

   251→        return sum + differenceInHours(parseISO(lastStep.timestamp), parseISO(a.requestedAt));
   252→      }
   253→      return sum;
   254→    }, 0);
   255→    return Math.round(totalHours / completed.length);
   256→  }, [approvals]);
   257→
   258→  const approvalRate = useMemo(() => {
   259→    const total = approvals.length;
   260→    const approved = approvals.filter((a) => a.status === 'approved').length;

   262→  }, [approvals]);

   265→    setSelectedApproval(approval);

   269→  };

   300→        </div>
   301→      </div>
   302→
   303→      {/* Stats */}

   306→          <CardContent className="pt-6">
   307→            <div className="flex items-center justify-between">
   308→              <div>
