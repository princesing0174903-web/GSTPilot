     2→     2→// GSTPilot Oracle CFO™ — Explainable Decision Card Builder
     3→     3→//
     4→     4→// Every recommendation Oracle makes is accompanied by a structured decision
     5→     5→// card containing:
     6→     6→//   • why          — the business reasoning (1-3 sentences)
     7→     7→//   • records      — supporting records with real IDs from the DB
     8→     8→//   • confidence   — 0-100, with factor breakdown
     9→     9→//   • calculation  — line-by-line math that led to the recommendation
    10→    10→//   • risks        — what could go wrong
    11→    11→//   • alternatives — 1-2 other options with trade-offs
    12→    12→//
    13→    13→// No unexplained AI responses. Every number traces back to a record.
    14→    14→// ═══════════════════════════════════════════════════════════════════════════════
    15→    15→
    16→    16→import type { DetectedToolCall } from './intent';
    17→    17→import type { LiveBusinessData, ToolContext } from './tools';
    18→    18→
    19→    19→export interface SupportingRecord {
    20→    20→  collection: string;
    21→    21→  id: string;
    22→    22→  label: string;
    23→    23→  detail: string;
    24→    24→}
    25→    25→
    26→    26→export interface CalculationLine {
    27→    27→  label: string;
    28→    28→  value: string;
    29→    29→}
    30→    30→
    31→    31→export interface RiskItem {
    32→    32→  severity: 'low' | 'medium' | 'high';
    33→    33→  description: string;
    34→    34→  mitigation?: string;
    35→    35→}
    36→    36→
    37→    37→export interface Alternative {
    38→    38→  title: string;
    39→    39→  tradeOff: string;
    40→    40→  recommended: boolean;
    41→    41→}
    42→    42→
    43→    43→export interface ConfidenceFactor {
    44→    44→  label: string;
    45→    45→  weight: number; // 0-1
    46→    46→  score: number; // 0-1
    47→    47→}
    48→    48→
    49→    49→export interface DecisionCard {
    50→    50→  toolId: string;
    51→    51→  toolName: string;
    52→    52→  why: string;
    53→    53→  records: SupportingRecord[];
    54→    54→  confidence: number; // 0-100
    55→    55→  confidenceFactors: ConfidenceFactor[];
    56→    56→  calculation: CalculationLine[];
    57→    57→  risks: RiskItem[];
    58→    58→  alternatives: Alternative[];
    59→    59→  generatedAt: string;
    60→    60→}
    61→    61→
    62→    62→/**
    63→    63→ * Build an explainable decision card for a detected tool call.
    64→    64→ */
    65→    65→export function buildDecisionCard(
    66→    66→  detected: DetectedToolCall,
    67→    67→  liveData: LiveBusinessData,
    68→    68→  ctx: ToolContext,
    69→    69→): DecisionCard {
    70→    70→  const records: SupportingRecord[] = [];
    71→    71→  const calculation: CalculationLine[] = [];
    72→    72→  const risks: RiskItem[] = [];
    73→    73→  const alternatives: Alternative[] = [];
    74→    74→  const confidenceFactors: ConfidenceFactor[] = [];
    75→    75→  let why = '';
    76→    76→
    77→    77→  // ─── Per-tool explanation logic ─────────────────────────────────────────
    78→    78→  switch (detected.toolId) {
    79→    79→    case 'create-invoice': {
    80→    80→      const clientParam = detected.extractedParams.find((p) => p.key === 'clientId');
    81→    81→      const amountParam = detected.extractedParams.find((p) => p.key === 'totalAmount');
    82→    82→      const gstParam = detected.extractedParams.find((p) => p.key === 'gstRate');
    83→    83→      const client = liveData.clients.find((c) => c.id === clientParam?.value);
    84→    84→      const amount = Number(amountParam?.value ?? 0);
    85→    85→      const gstRate = Number(gstParam?.value ?? 18);
    86→    86→      const taxable = amount / (1 + gstRate / 100);
    87→    87→      const gstAmount = amount - taxable;
    88→    88→
    89→    89→      why = `You asked to create an invoice${client ? ` for ${client.name}` : ''}. I found the client in your live client list and computed the GST breakdown at ${gstRate}% so the invoice is filing-ready.`;
    90→    90→
    91→    91→      if (client) {
    92→    92→        records.push({
    93→    93→          collection: 'clients',
    94→    94→          id: client.id,
    95→    95→          label: client.name,
    96→    96→          detail: `Client${client.gstin ? ` · GSTIN: ${client.gstin}` : ''}${client.email ? ` · ${client.email}` : ''}`,
    97→    97→        });
    98→    98→      }
    99→    99→
   100→   100→      calculation.push(
   101→   101→        { label: 'Total Amount (incl. GST)', value: `₹${amount.toLocaleString('en-IN')}` },
   102→   102→        { label: 'GST Rate', value: `${gstRate}%` },
   103→   103→        { label: 'Taxable Value', value: `₹${Math.round(taxable).toLocaleString('en-IN')}` },
   104→   104→        { label: 'GST Amount', value: `₹${Math.round(gstAmount).toLocaleString('en-IN')}` },
   105→   105→        { label: 'CGST (50%)', value: `₹${Math.round(gstAmount / 2).toLocaleString('en-IN')}` },
   106→   106→        { label: 'SGST (50%)', value: `₹${Math.round(gstAmount / 2).toLocaleString('en-IN')}` },
   107→   107→      );
   108→   108→
   109→   109→      risks.push(
   110→   110→        { severity: 'low', description: 'Invoice is created as draft — not sent or filed.', mitigation: 'Review the draft and send when ready.' },
   111→   111→        { severity: 'medium', description: 'GST rate assumption may not match the actual supply category.', mitigation: 'Confirm the HSN/SAC code and applicable rate before filing.' },
   112→   112→      );
   113→   113→
   114→   114→      alternatives.push(
   115→   115→        { title: 'Create as Quotation instead', tradeOff: 'A quote is non-binding and won\'t affect GST liability until converted.', recommended: false },
   116→   116→        { title: 'Use a different GST rate', tradeOff: '5% or 12% may apply depending on the goods/services supplied.', recommended: false },
   117→   117→      );
   118→   118→
   119→   119→      confidenceFactors.push(
   120→   120→        { label: 'Client matched from live data', weight: 0.4, score: client ? 1 : 0 },
   121→   121→        { label: 'Amount explicitly stated', weight: 0.3, score: amountParam ? 1 : 0 },
   122→   122→        { label: 'GST rate specified', weight: 0.2, score: gstParam ? 1 : 0.5 },
   123→   123→        { label: 'Intent clarity', weight: 0.1, score: detected.confidence },
   124→   124→      );
   125→   125→      break;
   126→   126→    }
   127→   127→
   128→   128→    case 'send-reminder-email':
   129→   129→    case 'send-reminder-whatsapp': {
   130→   130→      const channel = detected.toolId.includes('whatsapp') ? 'WhatsApp' : 'email';
   131→   131→      const invoiceParam = detected.extractedParams.find((p) => p.key === 'invoiceId');
   132→   132→      const amountParam = detected.extractedParams.find((p) => p.key === 'amount');
   133→   133→      const daysParam = detected.extractedParams.find((p) => p.key === 'daysOverdue');
   134→   134→      const invoice = liveData.overdueInvoices.find((i) => i.id === invoiceParam?.value);
   135→   135→
   136→   136→      why = `This client has an overdue invoice${invoice ? ` (${invoice.invoiceNumber}, ${invoice.daysOverdue} days past due)` : ''}. A ${channel} reminder is the appropriate next step in the collection ladder before escalation.`;
   137→   137→
   138→   138→      if (invoice) {
   139→   139→        records.push({
   140→   140→          collection: 'invoices',
   141→   141→          id: invoice.id,
   142→   142→          label: invoice.invoiceNumber,
   143→   143→          detail: `${invoice.clientName} · ₹${invoice.totalAmount.toLocaleString('en-IN')} · ${invoice.daysOverdue} days overdue`,
   144→   144→        });
   145→   145→      }
   146→   146→
   147→   147→      calculation.push(
   148→   148→        { label: 'Outstanding Amount', value: `₹${Number(amountParam?.value ?? 0).toLocaleString('en-IN')}` },
   149→   149→        { label: 'Days Overdue', value: `${daysParam?.value ?? '?'} days` },
   150→   150→        { label: 'Channel', value: channel },
   151→   151→        { label: 'Escalation Level', value: invoice && invoice.daysOverdue > 14 ? 'Level 2 (formal)' : 'Level 1 (gentle)' },
   152→   152→      );
   153→   153→
   154→   154→      risks.push(
   155→   155→        { severity: 'medium', description: 'Over-aggressive reminders can damage client relationships.', mitigation: 'Keep the tone professional and offer payment plan options.' },
   156→   156→        { severity: 'low', description: 'Email may land in spam.', mitigation: 'Follow up via WhatsApp if no response in 3 days.' },
   157→   157→      );
   158→   158→
   159→   159→      alternatives.push(
   160→   160→        { title: `Send via ${channel === 'WhatsApp' ? 'Email' : 'WhatsApp'} instead`, tradeOff: 'Different channel may get better response depending on client preference.', recommended: false },
   161→   161→        { title: 'Call the client directly', tradeOff: 'More personal but time-consuming; reserve for high-value or 90+ day overdue.', recommended: invoice?.daysOverdue > 30 },
   162→   162→      );
   163→   163→
   164→   164→      confidenceFactors.push(
   165→   165→        { label: 'Overdue invoice matched', weight: 0.5, score: invoice ? 1 : 0 },
   166→   166→        { label: 'Client contact available', weight: 0.3, score: 0.8 },
   167→   167→        { label: 'Intent clarity', weight: 0.2, score: detected.confidence },
   168→   168→      );
   169→   169→      break;
   170→   170→    }
   171→   171→
   172→   172→    case 'create-payment-link': {
   173→   173→      const amountParam = detected.extractedParams.find((p) => p.key === 'amount');
   174→   174→      const invoiceParam = detected.extractedParams.find((p) => p.key === 'invoiceId');
   175→   175→      const invoice = liveData.recentInvoices.find((i) => i.id === invoiceParam?.value);
   176→   176→
   177→   177→      why = `A payment link lets the client pay instantly via UPI/card without manual reconciliation. Oracle looks up the real invoice, validates it's unpaid, detects the connected provider (Razorpay/Stripe), and creates a REAL payment link via the provider API. The link is then emailed and WhatsApp'd to the customer, with webhook monitoring for automatic status updates.`;
   178→   178→
   179→   179→      if (invoice) {
   180→   180→        records.push({
   181→   181→          collection: 'invoices',
   182→   182→          id: invoice.id,
   183→   183→          label: invoice.invoiceNumber,
   184→   184→          detail: `${invoice.clientName} · ₹${invoice.totalAmount.toLocaleString('en-IN')} · ${invoice.status}`,
   185→   185→        });
   186→   186→      }
   187→   187→
   188→   188→      calculation.push(
   189→   189→        { label: 'Payment Amount', value: `₹${Number(amountParam?.value ?? 0).toLocaleString('en-IN')}` },
   190→   190→        { label: 'Provider', value: 'Razorpay or Stripe (auto-detected from integrations)' },
   191→   191→        { label: 'Payment Methods', value: 'UPI, Card, Net Banking, Wallet, EMI' },
   192→   192→        { label: 'Link Expiry', value: '30 days' },
   193→   193→        { label: 'Delivery', value: 'Email + WhatsApp (if connected)' },
   194→   194→        { label: 'Webhook Monitoring', value: 'Paid / Failed / Expired / Refunded auto-update' },
   195→   195→      );
   196→   196→
   197→   197→      risks.push(
   198→   198→        { severity: 'medium', description: 'No payment provider connected (preview mode).', mitigation: 'Go to Settings → Integrations → Razorpay or Stripe and add API credentials to create real links.' },
   199→   199→        { severity: 'low', description: 'Link may expire before payment.', mitigation: 'Regenerate if client delays beyond 30 days.' },
   200→   200→        { severity: 'low', description: 'Gateway fees apply (≈2% + ₹3).', mitigation: 'Fees are estimated in the approval summary before creation.' },
   201→   201→      );
   202→   202→
   203→   203→      alternatives.push(
   204→   204→        { title: 'Send bank transfer details instead', tradeOff: 'No gateway fees but slower collection and manual reconciliation.', recommended: false },
   205→   205→        { title: 'Use the production Payment Link card', tradeOff: 'Type "Create a payment link for Invoice XXX" — Oracle opens a dedicated approval card with real provider integration.', recommended: true },
   206→   206→      );
   207→   207→
   208→   208→      confidenceFactors.push(
   209→   209→        { label: 'Invoice matched', weight: 0.4, score: invoice ? 1 : 0 },
   210→   210→        { label: 'Amount specified', weight: 0.4, score: amountParam ? 1 : 0 },
   211→   211→        { label: 'Intent clarity', weight: 0.2, score: detected.confidence },
   212→   212→      );
   213→   213→      break;
   214→   214→    }
   215→   215→
   216→   216→    case 'generate-collection-report': {
   217→   217→      const totalOverdue = liveData.overdueInvoices.reduce((s, i) => s + i.totalAmount, 0);
   218→   218→      why = `You currently have ${liveData.overdueInvoices.length} overdue invoices worth ₹${totalOverdue.toLocaleString('en-IN')}. A structured collection report with aging buckets will help prioritize recovery efforts.`;
   219→   219→
   220→   220→      liveData.overdueInvoices.slice(0, 5).forEach((inv) => {
   221→   221→        records.push({
   222→   222→          collection: 'invoices',
   223→   223→          id: inv.id,
   224→   224→          label: inv.invoiceNumber,
   225→   225→          detail: `${inv.clientName} · ₹${inv.totalAmount.toLocaleString('en-IN')} · ${inv.daysOverdue}d overdue`,
   226→   226→        });
   227→   227→      });
   228→   228→
   229→   229→      const aging = {
   230→   230→        '0-30': liveData.overdueInvoices.filter((i) => i.daysOverdue <= 30).reduce((s, i) => s + i.totalAmount, 0),
   231→   231→        '31-60': liveData.overdueInvoices.filter((i) => i.daysOverdue > 30 && i.daysOverdue <= 60).reduce((s, i) => s + i.totalAmount, 0),
   232→   232→        '61-90': liveData.overdueInvoices.filter((i) => i.daysOverdue > 60 && i.daysOverdue <= 90).reduce((s, i) => s + i.totalAmount, 0),
   233→   233→        '90+': liveData.overdueInvoices.filter((i) => i.daysOverdue > 90).reduce((s, i) => s + i.totalAmount, 0),
   234→   234→      };
   235→   235→
   236→   236→      calculation.push(
   237→   237→        { label: 'Total Overdue', value: `₹${totalOverdue.toLocaleString('en-IN')}` },
   238→   238→        { label: 'Overdue Invoice Count', value: String(liveData.overdueInvoices.length) },
   239→   239→        { label: '0-30 days', value: `₹${aging['0-30'].toLocaleString('en-IN')}` },
   240→   240→        { label: '31-60 days', value: `₹${aging['31-60'].toLocaleString('en-IN')}` },
   241→   241→        { label: '61-90 days', value: `₹${aging['61-90'].toLocaleString('en-IN')}` },
   242→   242→        { label: '90+ days', value: `₹${aging['90+'].toLocaleString('en-IN')}` },
   243→   243→      );
   244→   244→
   245→   245→      risks.push(
   246→   246→        { severity: 'medium', description: 'Aging buckets may miss partial payments.', mitigation: 'Reconcile with bank transactions before acting.' },
   247→   247→      );
   248→   248→
   249→   249→      alternatives.push(
   250→   250→        { title: 'Client-wise report instead', tradeOff: 'Better for relationship management; worse for prioritization.', recommended: false },
   251→   251→      );
   252→   252→
   253→   253→      confidenceFactors.push(
   254→   254→        { label: 'Live overdue data available', weight: 0.6, score: liveData.overdueInvoices.length > 0 ? 1 : 0.3 },
   255→   255→        { label: 'Intent clarity', weight: 0.4, score: detected.confidence },
   256→   256→      );
   257→   257→      break;
   258→   258→    }
   259→   259→
   260→   260→    case 'prepare-gst-return': {
   261→   261→      const totalSales = liveData.recentInvoices.reduce((s, i) => s + i.totalAmount, 0);
   262→   262→      const outputLiability = liveData.recentInvoices.reduce((s, i) => s + (i.totalAmount - i.totalAmount / 1.18), 0);
   263→   263→      why = `Your recent sales invoices show ₹${Math.round(totalSales).toLocaleString('en-IN')} in total sales with an estimated output GST liability of ₹${Math.round(outputLiability).toLocaleString('en-IN')}. Drafting a return prepares the data for your review without filing.`;
   264→   264→
   265→   265→      liveData.recentInvoices.slice(0, 5).forEach((inv) => {
   266→   266→        records.push({
   267→   267→          collection: 'invoices',
   268→   268→          id: inv.id,
   269→   269→          label: inv.invoiceNumber,
   270→   270→          detail: `${inv.clientName} · ₹${inv.totalAmount.toLocaleString('en-IN')} · ${inv.status}`,
   271→   271→        });
   272→   272→      });
   273→   273→
   274→   274→      calculation.push(
   275→   275→        { label: 'Total Sales (from live invoices)', value: `₹${Math.round(totalSales).toLocaleString('en-IN')}` },
   276→   276→        { label: 'Invoice Count', value: String(liveData.recentInvoices.length) },
   277→   277→        { label: 'Output GST Liability (est.)', value: `₹${Math.round(outputLiability).toLocaleString('en-IN')}` },
   278→   278→        { label: 'Input Tax Credit', value: '₹0 (to be computed from GSTR-2B)' },
   279→   279→        { label: 'Net Payable (est.)', value: `₹${Math.round(outputLiability).toLocaleString('en-IN')}` },
   280→   280→      );
   281→   281→
   282→   282→      risks.push(
   283→   283→        { severity: 'high', description: 'This is a DRAFT — do not file without verifying ITC from GSTR-2B.', mitigation: 'Reconcile purchase ITC before filing on the GST portal.' },
   284→   284→        { severity: 'medium', description: 'GST rate assumption (18%) may not apply to all invoices.', mitigation: 'Verify HSN/SAC codes and recompute liability per slab.' },
   285→   285→      );
   286→   286→
   287→   287→      alternatives.push(
   288→   288→        { title: 'Wait for GSTR-2B reconciliation first', tradeOff: 'More accurate ITC but delays filing preparation.', recommended: true },
   289→   289→      );
   290→   290→
   291→   291→      confidenceFactors.push(
   292→   292→        { label: 'Live invoice data', weight: 0.5, score: liveData.recentInvoices.length > 0 ? 1 : 0.3 },
   293→   293→        { label: 'Period specified', weight: 0.3, score: detected.extractedParams.some((p) => p.key === 'period') ? 1 : 0.3 },
   294→   294→        { label: 'Intent clarity', weight: 0.2, score: detected.confidence },
   295→   295→      );
   296→   296→      break;
   297→   297→    }
   298→   298→
   299→   299→    case 'create-task': {
   300→   300→      const titleParam = detected.extractedParams.find((p) => p.key === 'title');
   301→   301→      why = `Creating a tracked task ensures this action doesn't get forgotten. Tasks appear in the priority queue and can be assigned, scheduled, and monitored.`;
   302→   302→
   303→   303→      calculation.push(
   304→   304→        { label: 'Task Title', value: String(titleParam?.value ?? 'Untitled task') },
   305→   305→        { label: 'Default Priority', value: 'Medium' },
   306→   306→        { label: 'Assignee', value: ctx.userEmail },
   307→   307→      );
   308→   308→
   309→   309→      risks.push({ severity: 'low', description: 'Task may be ignored if not prioritized.', mitigation: 'Set a due date and check the priority queue daily.' });
   310→   310→
   311→   311→      confidenceFactors.push(
   312→   312→        { label: 'Task title extracted', weight: 0.6, score: titleParam ? 1 : 0.3 },
   313→   313→        { label: 'Intent clarity', weight: 0.4, score: detected.confidence },
   314→   314→      );
   315→   315→      break;
   316→   316→    }
   317→   317→
   318→   318→    case 'mark-invoice-paid': {
   319→   319→      const invoiceParam = detected.extractedParams.find((p) => p.key === 'invoiceId');
   320→   320→      const amountParam = detected.extractedParams.find((p) => p.key === 'amount');
   321→   321→      const invoice = liveData.recentInvoices.find((i) => i.id === invoiceParam?.value);
   322→   322→
   323→   323→      why = `Recording this payment closes the receivable, updates your cash position, and ensures accurate financial reporting. The invoice status, a payment record, and an activity log are all updated atomically.`;
   324→   324→
   325→   325→      if (invoice) {
   326→   326→        records.push({
   327→   327→          collection: 'invoices',
   328→   328→          id: invoice.id,
   329→   329→          label: invoice.invoiceNumber,
   330→   330→          detail: `${invoice.clientName} · ₹${invoice.totalAmount.toLocaleString('en-IN')}`,
   331→   331→        });
   332→   332→      }
   333→   333→
   334→   334→      calculation.push(
   335→   335→        { label: 'Invoice', value: String(invoiceParam?.value ?? '?') },
   336→   336→        { label: 'Amount Paid', value: `₹${Number(amountParam?.value ?? 0).toLocaleString('en-IN')}` },
   337→   337→        { label: 'New Invoice Status', value: 'Paid' },
   338→   338→      );
   339→   339→
   340→   340→      risks.push(
   341→   341→        { severity: 'high', description: 'Marking paid before funds clear can cause reconciliation errors.', mitigation: 'Confirm the payment has settled in your bank account first.' },
   342→   342→      );
   343→   343→
   344→   344→      confidenceFactors.push(
   345→   345→        { label: 'Invoice matched', weight: 0.5, score: invoice ? 1 : 0.3 },
   346→   346→        { label: 'Amount verified', weight: 0.3, score: amountParam ? 1 : 0.3 },
   347→   347→        { label: 'Intent clarity', weight: 0.2, score: detected.confidence },
   348→   348→      );
   349→   349→      break;
   350→   350→    }
   351→   351→
   352→   352→    case 'generate-gst-report': {
   353→   353→      const totalSales = liveData.recentInvoices.reduce((s, i) => s + i.totalAmount, 0);
   354→   354→      const periodParam = detected.extractedParams.find((p) => p.key === 'period');
   355→   355→      const reportTypeParam = detected.extractedParams.find((p) => p.key === 'reportType');
   356→   356→      const periodLabel =
   357→   357→        periodParam?.value === 'last' ? 'last month' :
   358→   358→        periodParam?.value === 'quarter' ? 'this quarter' :
   359→   359→        periodParam?.value === 'fy' ? 'this financial year' :
   360→   360→        'this month';
   361→   361→      const reportTypeLabel =
   362→   362→        reportTypeParam?.value === 'gstr-1' ? 'GSTR-1 (outward supplies)' :
   363→   363→        reportTypeParam?.value === 'gstr-3b' ? 'GSTR-3B (summary return)' :
   364→   364→        reportTypeParam?.value === 'sales-tax' ? 'sales tax breakdown' :
   365→   365→        reportTypeParam?.value === 'purchase-tax' ? 'purchase tax / ITC breakdown' :
   366→   366→        reportTypeParam?.value === 'gst-liability' ? 'GST liability focus' :
   367→   367→        'executive GST summary';
   368→   368→
   369→   369→      why = `Generating a ${reportTypeLabel} report for ${periodLabel} pulls every live invoice in the period, validates GSTIN formats / duplicates / dates / reverse-charge flags, then computes per-slab CGST/SGST/IGST, ITC available + utilized, and net payable. Every number references a real invoice ID — no estimates, no simulations. The report is saved to the reports collection (version history) and downloadable as PDF / Excel / CSV.`;
   370→   370→
   371→   371→      // Surface the recent sales invoices that will be in the report
   372→   372→      liveData.recentInvoices.slice(0, 5).forEach((inv) => {
   373→   373→        records.push({
   374→   374→          collection: 'invoices',
