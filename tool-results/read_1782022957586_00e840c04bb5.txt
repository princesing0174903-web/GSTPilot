    38→function inrShort(n: number): string {
    39→  if (!isFinite(n) || isNaN(n)) return '₹0';
    40→  const abs = Math.abs(n);
    41→  if (abs >= 10000000) return `₹${(n / 10000000).toFixed(2)} Cr`;
    42→  if (abs >= 100000) return `₹${(n / 100000).toFixed(2)} L`;
    43→  if (abs >= 1000) return `₹${(n / 1000).toFixed(1)}K`;
    44→  return '₹' + Math.round(n).toLocaleString('en-IN');
    45→}

   492→/** Map our role names to the model's role names. */
   493→function toModelMessages(
   494→  messages: OracleChatRequest['messages'],
   495→): { role: 'assistant' | 'user'; content: string }[] {
   496→  return messages.map((m) => ({
   497→    role: (m.role === 'oracle' ? 'assistant' : 'user') as 'assistant' | 'user',
   498→    content: m.content,
   499→  }));
   500→}

   502→/** Server-side language hint derived from the latest user message. */
   503→function inferLanguageHint(messages: OracleChatRequest['messages']): OracleLanguageId | undefined {
   504→  for (let i = messages.length - 1; i >= 0; i--) {
   505→    if (messages[i].role === 'user' && messages[i].content?.trim()) {
   506→      const s = messages[i].content.slice(0, 500);
   507→      if (/[\u0900-\u097F]/.test(s)) return 'hindi';
   508→      if (/[\u0600-\u06FF]/.test(s)) return 'urdu';
   509→      if (/[\u0A00-\u0A7F]/.test(s)) return 'punjabi';
   510→      if (/[\u0A80-\u0AFF]/.test(s)) return 'gujarati';
   511→      if (/[\u0B80-\u0BFF]/.test(s)) return 'tamil';
   512→      if (/[\u0C00-\u0C7F]/.test(s)) return 'telugu';
   513→      if (/[\u0980-\u09FF]/.test(s)) return 'bengali';
   514→      return 'english';
   515→    }
   516→  }
   517→  return undefined;
   518→}

   522→export async function POST(request: Request) {
   523→  let body: OracleChatRequest;
   524→  try {
   525→    body = (await request.json()) as OracleChatRequest;
   526→  } catch {
   527→    return new Response(JSON.stringify({ error: 'Invalid JSON body' }), {
   528→      status: 400,
   529→      headers: { 'Content-Type': 'application/json' },
   530→    });
   531→  }
   532→
   533→  const messages = body.messages ?? [];
   534→  if (!Array.isArray(messages) || messages.length === 0) {
   535→    return new Response(JSON.stringify({ error: 'messages[] is required' }), {
   536→      status: 400,
   537→      headers: { 'Content-Type': 'application/json' },
   538→    });
   539→  }
   540→
   541→  const systemPrompt = await buildSystemPrompt(body);
   542→  const languageHint = inferLanguageHint(messages);
   543→
   544→  const modelMessages: { role: 'assistant' | 'user' | 'system'; content: string }[] = [
   545→    { role: 'assistant', content: systemPrompt },
   546→    ...toModelMessages(messages),
   547→  ];
   548→
   549→  // ── Acquire the upstream stream from the SDK ───────────────────────────────
   550→  let upstream: ReadableStream<Uint8Array> | null = null;
   551→  try {
   552→    const zai = await ZAI.create();
   553→    const result = await zai.chat.completions.create({
   554→      messages: modelMessages,
   555→      stream: true,
   556→      thinking: { type: 'disabled' },
   557→    });
   558→    if (result && typeof (result as ReadableStream<Uint8Array>).getReader === 'function') {
   559→      upstream = result as ReadableStream<Uint8Array>;
   560→    } else {
   561→      // Non-streaming fallback: emit the full text as one chunk then close.
   562→      const text =
   563→        (result as { choices?: { message?: { content?: string } }[] })?.choices?.[0]?.message
   564→          ?.content ?? '';
   565→      upstream = new ReadableStream<Uint8Array>({
   566→        start(controller) {
   567→          if (text) controller.enqueue(sseChunk({ token: text }));
   568→          controller.close();
   569→        },
   570→      });
   571→    }
   572→  } catch (err) {
   573→    const message = err instanceof Error ? err.message : 'Unknown error';
   574→    const stream = new ReadableStream<Uint8Array>({
   575→      start(controller) {
   576→        controller.enqueue(
   577→          sseChunk({
   578→            token: `I'm here, but I hit a temporary issue reaching my reasoning service (${message}). Please try again in a moment — your conversation is safe.`,
   579→          }),
   580→        );
   581→        controller.enqueue(sseChunk({ done: true }));
   582→        controller.close();
   583→      },
   584→    });
   585→    return new Response(stream, { status: 200, headers: sseHeaders() });
   586→  }
   587→
   588→  // ── Transform the upstream SSE stream into our token stream ────────────────
   589→  const decoder = new TextDecoder();
   590→  const transformed = new ReadableStream<Uint8Array>({
   591→    async start(controller) {
   592→      // Emit a tiny first nudge so the UI shows the pulsing cursor within the
   593→      // first frame — real tokens follow immediately.
   594→      if (languageHint) controller.enqueue(sseChunk({ language: languageHint }));
   595→
   596→      const reader = upstream!.getReader();
   597→      let buffer = '';
   598→      let emittedAny = false;
   599→      try {
   600→        while (true) {
   601→          const { done, value } = await reader.read();
   602→          if (done) break;
   603→          buffer += decoder.decode(value, { stream: true });
   604→
   605→          const lines = buffer.split('\n');
   606→          buffer = lines.pop() ?? '';
   607→
   608→          for (const rawLine of lines) {
   609→            const line = rawLine.trim();
   610→            if (!line || !line.startsWith('data:')) continue;
   611→            const data = line.slice(5).trim();
   612→            if (data === '[DONE]' || !data) continue;
   613→            try {
   614→              const json = JSON.parse(data);
   615→              const token: string =
   616→                json?.choices?.[0]?.delta?.content ??
   617→                json?.choices?.[0]?.message?.content ??
   618→                '';
   619→              if (token) {
   620→                emittedAny = true;
   621→                controller.enqueue(sseChunk({ token }));
   622→              }
   623→            } catch {
   624→              // Partial JSON across a chunk boundary — resolves on next read.
   625→            }
   626→          }
   627→        }
   628→        // Flush any trailing buffered line.
   629→        const tail = buffer.trim();
   630→        if (tail.startsWith('data:')) {
   631→          const data = tail.slice(5).trim();
   632→          if (data && data !== '[DONE]') {
   633→            try {
   634→              const json = JSON.parse(data);
   635→              const token: string =
   636→                json?.choices?.[0]?.delta?.content ??
   637→                json?.choices?.[0]?.message?.content ??
   638→                '';
   639→              if (token) {
   640→                emittedAny = true;
   641→                controller.enqueue(sseChunk({ token }));
   642→              }
   643→            } catch {
   644→              /* ignore */
   645→            }
   646→          }
   647→        }
   648→        if (!emittedAny) {
   649→          controller.enqueue(
   650→            sseChunk({
   651→              token:
   652→                "I'm here. Based on current GST rules and the information available, I'd be glad to help — could you share a bit more about what you're looking to do?",
   653→            }),
   654→          );
   655→        }
   656→        controller.enqueue(sseChunk({ done: true }));
   657→        controller.close();
   658→      } catch (err) {
   659→        const message = err instanceof Error ? err.message : 'Stream interrupted';
   660→        if (!emittedAny) {
   661→          controller.enqueue(
   662→            sseChunk({
   663→              token: `My response was interrupted (${message}). Please try sending that again.`,
   664→            }),
   665→          );
   666→        }
   667→        controller.enqueue(sseChunk({ done: true }));
   668→        controller.close();
   669→      }
   670→    },
   671→    cancel() {
   672→      upstream?.cancel?.().catch(() => undefined);
   673→    },
   674→  });
   675→
   676→  return new Response(transformed, { status: 200, headers: sseHeaders() });
   677→}
