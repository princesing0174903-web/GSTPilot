   208→// ─── Animated Number Hook ──────────────────────────────────────────────────
   209→function useAnimatedNumber(target: number, duration: number = 1200) {
   210→  const [current, setCurrent] = useState(0);
   211→  const ref = useRef<number | null>(null);
   212→  const startTime = useRef<number | null>(null);
   213→
   214→  useEffect(() => {
   215→    startTime.current = null;
   216→    const startValue = current;
   217→
   218→    function step(timestamp: number) {
   219→      if (!startTime.current) startTime.current = timestamp;
   220→      const elapsed = timestamp - startTime.current;
   221→      const progress = Math.min(elapsed / duration, 1);
   222→      const eased = 1 - Math.pow(1 - progress, 3);
   223→      setCurrent(Math.round(startValue + (target - startValue) * eased));
   224→      if (progress < 1) {
   225→        ref.current = requestAnimationFrame(step);
   226→      }
   227→    }
   228→
   229→    ref.current = requestAnimationFrame(step);
   230→    return () => {
   231→      if (ref.current) cancelAnimationFrame(ref.current);
   232→    };
   233→  }, [target, duration]);
   234→
   235→  return current;
   236→}
