   188→  const handlePointerDown = (e: React.PointerEvent) => {
   189→    setIsDragging(true)

   410→function useTypewriter(text: string, enabled: boolean, speed = 12) {
   411→  const [displayed, setDisplayed] = useState('')
   412→  const [done, setDone] = useState(false)
   413→
   414→  useEffect(() => {
   415→    if (!enabled) {
   416→      setDisplayed(text)
   417→      setDone(true)
   418→      return
   419→    }
   420→    setDisplayed('')
   421→    setDone(false)
   422→    let i = 0
   423→    const interval = setInterval(() => {
   424→      i++
   425→      setDisplayed(text.slice(0, i))
   426→      if (i >= text.length) {
   427→        clearInterval(interval)
   428→        setDone(true)
   429→      }
   430→    }, speed)
   431→    return () => clearInterval(interval)
   432→  }, [text, enabled, speed])
   433→
   434→  return { displayed, done }
   435→}

   437→// --- Markdown-ish renderer (lightweight) -----------------------------------
   438→
   439→function renderMarkdown(text: string) {
   440→  // Split by lines, render basic formatting
   441→  const lines = text.split('\n')
   442→  const elements: React.ReactNode[] = []
   443→  lines.forEach((line, idx) => {
   444→    if (!line.trim()) {
   445→      elements.push(<div key={idx} className="h-2" />)
   446→      return
   447→    }
   448→    // Headings (## or #)
   449→    if (line.startsWith('## ')) {
   450→      elements.push(
   451→        <div key={idx} className="font-bold text-sm mt-2 mb-1 text-slate-900 dark:text-slate-100">
   452→          {line.replace(/^##\s+/, '')}
   453→        </div>
   454→      )
   455→      return
   456→    }
   457→    if (line.startsWith('# ')) {
   458→      elements.push(
   459→        <div key={idx} className="font-bold text-base mt-2 mb-1 text-slate-900 dark:text-slate-100">
   460→          {line.replace(/^#\s+/, '')}
   461→        </div>
   462→      )
   463→      return
   464→    }
   465→    // Bullet points
   466→    if (line.startsWith('• ') || line.startsWith('- ')) {
   467→      const content = line.replace(/^[•-]\s+/, '')
   468→      elements.push(
   469→        <div key={idx} className="flex gap-2 text-sm leading-relaxed">
   470→          <span className="text-emerald-500 mt-0.5">•</span>
   471→          <span className="flex-1">{renderInline(content)}</span>
   472→        </div>
   473→      )
   474→      return
   475→    }
   476→    // Numbered list
   477→    const numMatch = line.match(/^(\d+)\.\s+(.*)/)
   478→    if (numMatch) {
   479→      elements.push(
   480→        <div key={idx} className="flex gap-2 text-sm leading-relaxed">
   481→          <span className="font-bold text-emerald-600 dark:text-emerald-400 min-w-[1.2rem]">{numMatch[1]}.</span>
   482→          <span className="flex-1">{renderInline(numMatch[2])}</span>
   483→        </div>
   484→      )
   485→      return
   486→    }
   487→    // Regular paragraph
   488→    elements.push(
   489→      <div key={idx} className="text-sm leading-relaxed">
   490→        {renderInline(line)}
   491→      </div>
   492→    )
   493→  })
   494→  return elements
   495→}
   496→
   497→function renderInline(text: string): React.ReactNode {
   498→  // Handle **bold** and `code`
   499→  const parts: React.ReactNode[] = []
   500→  const regex = /(\*\*[^*]+\*\*|`[^`]+`)/g
   501→  let lastIndex = 0
   502→  let match
   503→  let key = 0
   504→  while ((match = regex.exec(text)) !== null) {
   505→    if (match.index > lastIndex) {
   506→      parts.push(text.slice(lastIndex, match.index))
   507→    }
   508→    const token = match[0]
   509→    if (token.startsWith('**')) {
   510→      parts.push(
   511→        <strong key={key++} className="font-bold text-slate-900 dark:text-slate-100">
   512→          {token.slice(2, -2)}
   513→        </strong>
   514→      )
   515→    } else if (token.startsWith('`')) {
   516→      parts.push(
   517→        <code key={key++} className="px-1 py-0.5 rounded bg-emerald-50 dark:bg-emerald-950/50 text-emerald-700 dark:text-emerald-300 text-xs font-mono">
   518→          {token.slice(1, -1)}
   519→        </code>
   520→      )
   521→    }
   522→    lastIndex = match.index + token.length
   523→  }
   524→  if (lastIndex < text.length) {
   525→    parts.push(text.slice(lastIndex))
   526→  }
   527→  return parts.length > 0 ? parts : text
   528→}

  1163→  }
