    57→    57→// Relative time formatter — defensively handles ISO string / Date / Firestore Timestamp.
    58→    58→function timeAgo(createdAt: unknown): string {
    59→    59→  if (!createdAt) return 'just now';
    60→    60→  let date: Date;
    61→    61→  if (createdAt instanceof Date) {
    62→    62→    date = createdAt;
    63→    63→  } else if (typeof createdAt === 'string') {
    64→    64→    date = new Date(createdAt);
    65→    65→  } else if (
    66→    66→    typeof createdAt === 'object' &&
    67→    67→    createdAt !== null &&
    68→    68→    'toDate' in createdAt &&
    69→    69→    typeof (createdAt as { toDate: () => Date }).toDate === 'function'
    70→    70→  ) {
    71→    71→    date = (createdAt as { toDate: () => Date }).toDate();
    72→    72→  } else {
    73→    73→    return 'just now';
    74→    74→  }
    75→    75→  if (isNaN(date.getTime())) return 'just now';
    76→    76→  const seconds = Math.floor((Date.now() - date.getTime()) / 1000);
    77→    77→  if (seconds < 60) return 'just now';
    78→    78→  const minutes = Math.floor(seconds / 60);
    79→    79→  if (minutes < 60) return `${minutes}m ago`;
    80→    80→  const hours = Math.floor(minutes / 60);
    81→    81→  if (hours < 24) return `${hours}h ago`;
    82→    82→  const days = Math.floor(hours / 24);
    83→    83→  if (days < 7) return `${days}d ago`;
    84→    84→  const weeks = Math.floor(days / 7);
    85→    85→  if (weeks < 5) return `${weeks}w ago`;
    86→    86→  const months = Math.floor(days / 30);
    87→    87→  if (months < 12) return `${months}mo ago`;
    88→    88→  const years = Math.floor(days / 365);
    89→    89→  return `${years}y ago`;
    90→    90→}

    92→    92→// ─── Business Score computation ───────────────────────────────────────────────
    93→    93→// Start at 100. Subtract weighted penalties for risk signals. Clamp 0–100.

    95→    95→
    96→    96→function computeBusinessScore(opts: {
    97→    97→  totalClients: number;
    98→    98→  criticalIssues: number;
    99→    99→  warnings: number;
   100→   100→  pendingReturns: number;
   101→   101→  overdueReturns: number;
   102→   102→  averageHealthScore: number;
   103→   103→  matchPercentage: number;
   104→   104→}): number | null {
   105→   105→  const { totalClients, criticalIssues, warnings, pendingReturns, overdueReturns, averageHealthScore, matchPercentage } = opts;
   106→   106→  if (totalClients === 0 && pendingReturns === 0 && overdueReturns === 0) return null;
   107→   107→
   108→   108→  let score = 100;
   109→   109→  score -= criticalIssues * 6;
   110→   110→  score -= warnings * 2;
   111→   111→  score -= overdueReturns * 8;
   112→   112→  score -= pendingReturns * 2;
   113→   113→  if (averageHealthScore > 0) {
   114→   114→    score -= Math.max(0, 80 - averageHealthScore) * 0.6;
   115→   115→  }
   116→   116→  score -= Math.max(0, 100 - matchPercentage) * 0.3;
   117→   117→  return Math.max(0, Math.min(100, Math.round(score)));
   118→   118→}

   256→   256→function ScoreGauge({ score }: { score: number | null }) {
   257→   257→  const size = 220;
   258→   258→  const stroke = 14;
   259→   259→  const r = (size - stroke) / 2;
   260→   260→  const c = 2 * Math.PI * r;
   261→   261→  const pct = score === null ? 0 : score / 100;
   262→   262→  const offset = c * (1 - pct);
   263→   263→
   264→   264→  return (
   265→   265→    <div className="relative" style={{ width: size, height: size }}>
   266→   266→      <svg width={size} height={size} className="-rotate-90">
   267→   267→        <defs>
   268→   268→          <linearGradient id="missionScoreGradient" x1="0%" y1="0%" x2="100%" y2="100%">
   269→   269→            <stop offset="0%" stopColor="#10b981" />
   270→   270→            <stop offset="50%" stopColor="#06b6d4" />
   271→   271→            <stop offset="100%" stopColor="#3b82f6" />
   272→   272→          </linearGradient>
   273→   273→        </defs>
   274→   274→        <circle
   275→   275→          cx={size / 2}
   276→   276→          cy={size / 2}
   277→   277→          r={r}
   278→   278→          fill="none"
   279→   279→          stroke="rgba(255,255,255,0.06)"
   280→   280→          strokeWidth={stroke}

   284→   284→            cx={size / 2}
   285→   285→            cy={size / 2}
   286→   286→            r={r}
   287→   287→            fill="none"
   288→   288→            stroke="url(#missionScoreGradient)"
   289→   289→            strokeWidth={stroke}
   290→   290→            strokeLinecap="round"
   291→   291→            strokeDasharray={c}
   292→   292→            initial={{ strokeDashoffset: c }}

   307→   307→          >

   312→   312→          / 100
   313→   313→        </span>
   314→   314→      </div>
   315→   315→    </div>
   316→   316→  );
   317→   317→}
