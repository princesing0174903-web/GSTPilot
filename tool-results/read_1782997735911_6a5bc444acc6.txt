   589→function EmailVerificationBanner() {

   640→function OnboardingScreen() {
   641→  const { user, markOnboardingComplete } = useAuth()
   642→  const { setCurrentScreen, setCurrentView } = useApp()
   643→  const [saveError, setSaveError] = React.useState<string | null>(null)
   644→
   645→  // ── Fire-and-forget: save onboarding data to Firestore in the background ──
   646→  const saveOnboardingToFirestore = async (
   647→    data: import('@/components/onboarding/OnboardingFlow').OnboardingData,
   648→    firmId: string
   649→  ) => {
   650→    try {
   651→      const { doc, setDoc, serverTimestamp } = await import('firebase/firestore')
   652→      const { db } = await import('@/lib/firebase')
   653→
   654→      if (!user) return
   655→
   656→      // Update user document
   657→      await setDoc(doc(db, 'users', user.id), {
   658→        uid: user.id,
   659→        fullName: data.fullName,
   660→        email: data.email,
   661→        phone: data.phone,
   662→        ageGroup: data.ageGroup,
   663→        profession: data.profession,
   664→        experience: data.experience,
   665→        firmId: firmId || null,
   666→        firmName: data.firmName,
   667→        onboardingCompleted: true,
   668→        clientCount: data.clientCount,
   669→        monthlyReturns: data.monthlyReturns,
   670→        gstServices: data.gstServices,
   671→        painPoints: data.painPoints,
   672→        referralSource: data.referralSource,
   673→        trialReasons: data.trialReasons,
   674→        wantsUpdates: data.wantsUpdates,
   675→        updatedAt: serverTimestamp(),
   676→      }, { merge: true })
   677→      console.log('[Onboarding] ✅ User document saved to Firestore')
   678→
   679→      // Save onboarding record
   680→      await setDoc(doc(db, 'onboarding', user.id), {
   681→        ...data,
   682→        firmId: firmId || null,
   683→        completedAt: serverTimestamp(),
   684→      })
   685→      console.log('[Onboarding] ✅ Onboarding record saved to Firestore')
   686→    } catch (error) {
   687→      console.warn('[Onboarding] ⚠️ Background Firestore save failed:', error)
   688→    }
   689→  }
   690→
   691→  const handleOnboardingComplete = (
   692→    data: import('@/components/onboarding/OnboardingFlow').OnboardingData,
   693→    destination?: import('@/components/onboarding/OnboardingFlow').OnboardingDestination
   694→  ) => {
   695→    setSaveError(null)
   696→
   697→    if (!user) {
   698→      setSaveError('No authenticated user found. Please sign in again.')
   699→      return
   700→    }
   701→
   702→    // ── INSTANT: Mark onboarding complete locally & navigate ──
   703→    // This happens synchronously — user sees dashboard immediately
   704→    console.log('[Onboarding] 🚀 Navigating to', destination || 'dashboard', '(optimistic)')
   705→    markOnboardingComplete(undefined, data.firmName)
   706→    setCurrentView(destination || 'dashboard')
   707→    setCurrentScreen('app')
   708→
   709→    // ── BACKGROUND: Create firm doc + save everything to Firestore ──
   710→    // These happen async — user is already in the dashboard
   711→    console.log('[Onboarding] 📝 Starting background Firestore writes...')
   712→    ;(async () => {
   713→      try {
   714→        const { collection, addDoc, serverTimestamp } = await import('firebase/firestore')
   715→        const { db } = await import('@/lib/firebase')
   716→
   717→        // Create firm document
   718→        let firmId = ''
   719→        try {
   720→          const firmRef = await addDoc(collection(db, 'firms'), {
   721→            ownerId: user.id,
   722→            firmName: data.firmName,
   723→            gstin: data.gstin || null,
   724→            state: data.state,
   725→            stateCode: data.stateCode,
   726→            organizationType: data.organizationType,
   727→            icaiMembershipNo: data.icaiMembershipNo || null,
   728→            officeAddress: data.officeAddress || null,
   729→            createdAt: serverTimestamp(),
   730→          })
   731→          firmId = firmRef.id
   732→          console.log('[Onboarding] ✅ Firm created:', firmId)
   733→
   734→          // Update the local user with firmId
   735→          markOnboardingComplete(firmId, data.firmName)
   736→        } catch (firmError) {
   737→          console.warn('[Onboarding] ⚠️ Firm creation failed:', firmError)
   738→        }
   739→
   740→        // Save remaining data
   741→        await saveOnboardingToFirestore(data, firmId)
   742→      } catch (error) {
   743→        console.warn('[Onboarding] ⚠️ Background save error:', error)
   744→      }
   745→    })()
   746→  }
   747→
   748→  const handleSkip = () => {
   749→    setSaveError(null)
   750→
   751→    if (!user) {
   752→      setSaveError('No authenticated user found. Please sign in again.')
   753→      return
   754→    }
   755→
   756→    // ── INSTANT: Navigate immediately ──
   757→    console.log('[Onboarding] 🚀 Skipping onboarding (optimistic)')
   758→    markOnboardingComplete()
   759→    setCurrentView('dashboard')
   760→    setCurrentScreen('app')
   761→
   762→    // ── BACKGROUND: Mark as completed in Firestore ──
   763→    ;(async () => {
   764→      try {
   765→        const { doc, setDoc, serverTimestamp } = await import('firebase/firestore')
   766→        const { db } = await import('@/lib/firebase')
   767→        await setDoc(doc(db, 'users', user.id), {
   768→          onboardingCompleted: true,
   769→          updatedAt: serverTimestamp(),
   770→        }, { merge: true })
   771→        console.log('[Onboarding] ✅ Skip saved to Firestore')
   772→      } catch (error) {
   773→        console.warn('[Onboarding] ⚠️ Background skip save failed:', error)
   774→      }
   775→    })()
   776→  }
   777→
   778→  return (
   779→    <OnboardingFlow
   780→      onComplete={handleOnboardingComplete}
   781→      onSkip={handleSkip}
   782→      userEmail={user?.email}
   783→      userName={user?.name}
   784→      error={saveError}
   785→      onDismissError={() => setSaveError(null)}
   786→    />
   787→  )
   788→}

   790→function AppRouter() {
   791→  const { currentScreen, setCurrentScreen } = useApp()
   792→  const { isAuthenticated, isInitializing, needsOnboarding, needsEmailVerification } = useAuth()
   793→
   794→  useEffect(() => {
   795→    if (isInitializing) return
   796→    // Only auto-redirect to app if authenticated AND onboarding is complete
   797→    // Don't redirect if user still needs onboarding — the OnboardingScreen handles its own navigation
   798→    if (isAuthenticated && !needsOnboarding && currentScreen !== 'app') {
   799→      setCurrentScreen('app')
   800→    }
   801→  }, [isAuthenticated, isInitializing, currentScreen, setCurrentScreen, needsOnboarding])
   802→
   803→  useEffect(() => {
   804→    if (isInitializing) return
   805→    if (!isAuthenticated && currentScreen === 'app') {
   806→      setCurrentScreen('landing')
   807→    }
   808→  }, [isAuthenticated, isInitializing, currentScreen, setCurrentScreen])
   809→
   810→  const handleGetStarted = () => setCurrentScreen('login')
   811→  const handleBookDemo = () => setCurrentScreen('login')
   812→  const handleBackToLanding = () => setCurrentScreen('landing')
   813→
   814→  if (isInitializing && !isAuthenticated) {
   815→    return (
   816→      <div className="min-h-screen flex items-center justify-center bg-black">
   817→        <div className="flex flex-col items-center gap-4">
   818→          <div className="flex h-11 w-11 items-center justify-center rounded-2xl glass-surface motion-pulse">
   819→            <Zap className="h-5 w-5 accent-text" />
   820→          </div>
   821→          <span className="text-sm text-white/55 font-medium">Loading GSTPilot…</span>
   822→        </div>
   823→      </div>
   824→    )
   825→  }
   826→
   827→  // Show onboarding if user hasn't completed it
   828→  if (isAuthenticated && needsOnboarding) {
   829→    return <OnboardingScreen />
   830→  }
   831→
   832→  if (currentScreen === 'app' && isAuthenticated) {
   833→    return (
   834→      <div className="flex min-h-screen flex-col">
   835→        {needsEmailVerification && <EmailVerificationBanner />}
   836→        <DashboardContent />
   837→      </div>
   838→    )
   839→  }
   840→
   841→  if (currentScreen === 'login') {
   842→    return <LoginPage onBack={handleBackToLanding} onGetStarted={handleGetStarted} />
   843→  }
   844→
   845→  return <LandingPage onGetStarted={handleGetStarted} onBookDemo={handleBookDemo} />
   846→}

   848→export default function Home() {
