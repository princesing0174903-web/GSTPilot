   172→  // ─── Live business data cache (30s TTL) ────────────────────────────────────
   173→  // Fetches /api/oracle/context once per 30s window so we don't hit it on every
   174→  // single message. Invalidated immediately on `connections-updated` events so a
   175→  // freshly connected GSTN/Bank shows up in the very next prompt.
   176→  const liveDataCacheRef = useRef<{ data: OracleLiveData | undefined; ts: number } | null>(null);
   177→  const fetchLiveData = useCallback(async (): Promise<OracleLiveData | undefined> => {
   178→    const now = Date.now();
   179→    if (liveDataCacheRef.current && now - liveDataCacheRef.current.ts < 30000) {
   180→      return liveDataCacheRef.current.data;
   181→    }
   182→    try {
   183→      const res = await fetch('/api/oracle/context');
   184→      if (res.ok) {
   185→        const data = (await res.json()) as OracleLiveData;
   186→        liveDataCacheRef.current = { data, ts: now };
   187→        return data;
   188→      }
   189→    } catch {
   190→      /* ignore — chat still works without context */
   191→    }
   192→    return undefined;
   193→  }, []);
   194→
   195→  useEffect(() => {
   196→    const handler = () => {
   197→      liveDataCacheRef.current = null;
   198→    };

   200→    return () => window.removeEventListener('connections-updated', handler);
   201→  }, []);
   202→
   203→  // ─── Load + migrate store on open, with AUTO NEW CHAT ──────────────────────
   204→  // Oracle starts a fresh conversation (never deleting history) when ANY of:
   205→  //   • no active conversation exists
   206→  //   • active conversation's lastActiveAt is older than 30 minutes (inactivity)
   207→  //   • active conversation's updatedAt is older than 2 hours (stale)
   208→  // The "Ask Oracle" launcher (initialPrompt) additionally forces a fresh chat
   209→  // if the current one already has messages — handled in the initialPrompt effect.
   210→  useEffect(() => {

   214→    // ── Migrate v2 store (no lastActiveAt/pinned) → v3 shape ────────────────

   220→          if (parsedV2 && Array.isArray(parsedV2.conversations) && parsedV2.conversations.length > 0) {
   221→            const migrated: ConversationStore = {
   222→              conversations: parsedV2.conversations.map((c) => ({
   223→                ...c,
   224→                lastActiveAt: c.updatedAt ?? c.createdAt ?? new Date().toISOString(),
   225→                pinned: false,
   226→              })),
   227→              activeId: parsedV2.activeId,
   228→            };
   229→            loaded.conversations = migrated.conversations;
   230→            loaded.activeId = migrated.activeId;
   231→          }
   232→        }
   233→      } catch {
   234→        /* ignore */
   235→      }
   236→    }
   237→
   238→    // ── Migrate v1 single-conversation store (array of messages) ────────────
   239→    if (loaded.conversations.length === 0) {
   240→      try {
   241→        const rawV1 = localStorage.getItem(LEGACY_KEY_V1);
   242→        if (rawV1) {
   243→          const parsed = JSON.parse(rawV1) as OracleMessage[];
   244→          if (Array.isArray(parsed) && parsed.length > 0) {
   245→            const now = new Date().toISOString();
   246→            const conv: OracleConversation = {
   247→              id: cryptoId(),
   248→              title: generateTitle(parsed[0]?.content ?? 'New chat'),
   249→              messages: parsed.map((m) => (m.streaming ? { ...m, streaming: false } : m)),
   250→              createdAt: parsed[0]?.createdAt ?? now,
   251→              updatedAt: now,
   252→              lastActiveAt: now,
   253→              pinned: false,
   254→            };
   255→            const next: ConversationStore = { conversations: [conv], activeId: conv.id };
   256→            setStore(next);
   257→            setActiveId(conv.id);
   258→            setMessages(conv.messages);
   259→            saveStore(next);
   260→            return;
   261→          }
   262→        }
   263→      } catch {
   264→        /* ignore */
   265→      }
   266→    }
   267→
   268→    // ── Nothing to load — start fresh ────────────────────────────────────────
   269→    if (loaded.conversations.length === 0) {
   270→      const conv = newConversation();
   271→      const next: ConversationStore = { conversations: [conv], activeId: conv.id };
   272→      setStore(next);
   273→      setActiveId(conv.id);
   274→      setMessages([]);
   275→      saveStore(next);
   276→      return;
   277→    }
   278→
   279→    // ── AUTO NEW CHAT: check inactivity / staleness of the active conv ──────
   280→    setStore(loaded);
   281→    const active =
   282→      loaded.conversations.find((c) => c.id === loaded.activeId) ?? loaded.conversations[0];
   283→    const now = Date.now();
   284→    const lastActive = active ? new Date(active.lastActiveAt ?? active.updatedAt).getTime() : 0;
   285→    const updated = active ? new Date(active.updatedAt).getTime() : 0;
   286→    const isInactive = active && active.messages.length > 0 && (now - lastActive) > INACTIVITY_NEW_CHAT_MS;
   287→    const isStale = active && (now - updated) > STALE_CONVERSATION_MS;
   288→
   289→    if (!active || isInactive || isStale) {
   290→      // Start a fresh conversation; keep all history in the sidebar.
   291→      const conv = newConversation();
   292→      const next: ConversationStore = {
   293→        conversations: [conv, ...loaded.conversations],
   294→        activeId: conv.id,
   295→      };
   296→      setStore(next);
   297→      setActiveId(conv.id);
   298→      setMessages([]);
   299→      saveStore(next);
   300→      return;
   301→    }
   302→
   303→    // ── Resume the active conversation ──────────────────────────────────────
   304→    setActiveId(active.id);
   305→    setMessages(active.messages);
   306→  }, [open]);
   307→
   308→  // ─── Persist active conversation whenever messages change ───────────────────

   311→    if (!open || !activeId) return;

   313→      if (!prev.conversations.length) return prev;
   314→      const exists = prev.conversations.some((c) => c.id === activeId);
   315→      let conversations: OracleConversation[];
   316→      let nextActiveId = prev.activeId;
   317→      const nowIso = new Date().toISOString();
   318→      if (exists) {

   320→          c.id === activeId
   321→            ? {
   322→                ...c,
   323→                messages,
   324→                title: messages.length > 0 ? (generateTitle(firstUserText(messages)) || c.title) : c.title,

   339→          lastActiveAt: nowIso,
   340→          pinned: false,
   341→        };

   344→      }
   345→      const next = { conversations, activeId: nextActiveId };
   346→      saveStore(next);
   347→      return next;
   348→    });

   503→      const copy: OracleConversation = {

   520→  // ─── Share (copy markdown to clipboard) ─────────────────────────────────────
   521→  const handleShareConversation = useCallback(async (id: string) => {
