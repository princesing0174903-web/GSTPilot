   163→export const convertCurrency = (amount: number, from: string, to: string): number => {
   164→  const fromCur = CURRENCIES.find(c => c.code === from);
   165→  const toCur = CURRENCIES.find(c => c.code === to);
   166→  if (!fromCur || !toCur) return amount;
   167→  const usd = amount * fromCur.rateToUSD;
   168→  return usd / toCur.rateToUSD;

   170→
   171→export const formatCurrency = (amount: number, code: string): string => {
   172→  const cur = CURRENCIES.find(c => c.code === code);
   173→  if (!cur) return `${amount.toFixed(2)}`;
   174→  const decimals = code === 'JPY' || code === 'INR' ? 0 : 2;
   175→  return `${cur.symbol}${amount.toLocaleString('en-US', { minimumFractionDigits: decimals, maximumFractionDigits: decimals })}`;
   176→};

   222→export interface TaxCalculation {
   223→  taxType: string;
   224→  rate: number;
   225→  baseAmount: number;
   226→  taxAmount: number;
   227→  totalAmount: number;
   228→}
   229→
   230→export const calculateTax = (
   231→  amount: number,
   232→  countryCode: CountryCode,
   233→  taxType: 'indirect' | 'corporate' | 'payroll' | 'import' | 'export' = 'indirect'
   234→): TaxCalculation => {
   235→  const country = getCountry(countryCode);
   236→  let rate = 0;
   237→  let taxName = '';
   238→  switch (taxType) {
   239→    case 'indirect': rate = country.taxRate; taxName = country.primaryTaxName; break;
   240→    case 'corporate': rate = country.corporateTaxRate; taxName = 'Corporate Tax'; break;
   241→    case 'payroll': rate = country.payrollTaxRate; taxName = 'Payroll Tax'; break;
   242→    case 'import': rate = country.importDuty; taxName = 'Import Duty'; break;
   243→    case 'export': rate = country.exportDuty; taxName = 'Export Duty'; break;
   244→  }
   245→  return {
   246→    taxType: taxName,
   247→    rate,
   248→    baseAmount: amount,
   249→    taxAmount: (amount * rate) / 100,
   250→    totalAmount: amount + (amount * rate) / 100,
   251→  };
   252→};
