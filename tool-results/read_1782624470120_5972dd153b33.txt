    30→// ─── Status detector — derives an employee's live status from data ───────────
    31→
    32→function detectStatus(
    33→  role: EmployeeRole,
    34→  data: WorkforceDataView,
    35→  alertCount: number,
    36→  pendingDecisions: number,
    37→  hasLiveData: boolean,
    38→): EmployeeStatus {
    39→  if (!hasLiveData) return 'idle';
    40→  if (pendingDecisions > 0) return 'deciding';
    41→  if (alertCount > 0) return 'reviewing';
    42→  // Check if this employee is involved in an active collaboration chain
    43→  const collaborationActive = data.twin.timeline.todayCount > 0;
    44→  if (collaborationActive && (role === 'coo' || role === 'operations_manager' || role === 'ceo')) {
    45→    return 'collaborating';
    46→  }
    47→  return 'active';
    48→}

    52→function clamp(n: number, lo = 0, hi = 100): number {
    53→  return Math.max(lo, Math.min(hi, n));
    54→}

    56→function financeHealth(data: WorkforceDataView): number {
    57→  const f = data.finance;
    58→  const cashScore = f.runwayDays > 180 ? 100 : f.runwayDays > 90 ? 70 : f.runwayDays > 30 ? 40 : 10;
    59→  const marginScore = f.marginPct > 20 ? 100 : f.marginPct > 10 ? 70 : f.marginPct > 0 ? 40 : 10;
    60→  const collectionScore = f.overdueAmount === 0 ? 100
    61→    : f.receivables > 0 ? clamp(100 - (f.overdueAmount / f.receivables) * 100) : 50;
    62→  const gstScore = f.gst === 0 ? 100 : 70;
    63→  return Math.round((cashScore + marginScore + collectionScore + gstScore) / 4);
    64→}
    65→
    66→function operationsHealth(data: WorkforceDataView): number {
    67→  const o = data.operations;
    68→  const taskScore = o.openTasks === 0 ? 100 : clamp(100 - o.openTasks * 5);
    69→  const efficiencyScore = o.efficiencyPct;
    70→  const deliveryScore = o.deliveryPending === 0 ? 100 : clamp(100 - o.deliveryPending * 10);
    71→  return Math.round((taskScore + efficiencyScore + deliveryScore) / 3);
    72→}

    74→function salesHealth(data: WorkforceDataView): number {
    75→  const s = data.sales;
    76→  const clientScore = s.clientCount > 10 ? 100 : s.clientCount > 5 ? 70 : s.clientCount > 0 ? 40 : 10;
    77→  const pipelineScore = s.pipelineValue > 500000 ? 100 : s.pipelineValue > 100000 ? 70 : s.pipelineValue > 0 ? 40 : 10;
    78→  const conversionScore = s.conversionRate > 50 ? 100 : s.conversionRate > 20 ? 70 : 40;
    79→  return Math.round((clientScore + pipelineScore + conversionScore) / 3);
    80→}

   337→function financeRecs(data: WorkforceDataView): EmployeeRecommendation[] {
   338→  const f = data.finance;
   339→  const recs: EmployeeRecommendation[] = [];
   340→  if (f.overdueAmount > 0) {
   341→    recs.push({
   342→      title: `Recover ${formatINRShort(f.overdueAmount)} in overdue receivables`,
   343→      rationale: `${f.outstandingInvoices} invoices are overdue, tying up working capital.`,
   344→      impact: 'Improves cash position and reduces bad debt risk',
   345→      priority: f.overdueAmount > 100000 ? 'critical' : 'high',
   346→      estimatedValue: f.overdueAmount,
   347→      suggestedAction: 'Send automated payment reminders via Sales AI',
   348→    });
   349→  }
   350→  if (f.runwayDays < 90 && f.runwayDays > 0) {
   351→    recs.push({
   352→      title: `Cash runway is ${f.runwayDays} days — accelerate collections or arrange credit`,
   353→      rationale: 'At current burn rate, cash will be exhausted within 3 months.',
   354→      impact: 'Prevents liquidity crisis',
   355→      priority: f.runwayDays < 30 ? 'critical' : 'high',
   356→      estimatedValue: f.burnRate * 3,
   357→      suggestedAction: 'Coordinate with CFO AI on credit line or expense reduction',
   358→    });
   359→  }
   360→  if (f.gst > 0) {
   361→    recs.push({
   362→      title: `File GST return — ${formatINRShort(f.gst)} payable`,
   363→      rationale: 'GST liability is pending. Late filing attracts penalties.',
   364→      impact: 'Avoids penalties and compliance risk',
   365→      priority: 'high',
   366→      estimatedValue: f.gst * 0.18, // penalty estimate
   367→      suggestedAction: 'Hand off to Compliance Manager AI',
   368→    });
   369→  }
   370→  if (f.marginPct < 10 && f.revenue > 0) {
   371→    recs.push({
   372→      title: 'Margin below 10% — review pricing and expenses',
   373→      rationale: `Current net margin is ${f.marginPct.toFixed(1)}%.`,
   374→      impact: 'Improves profitability',
   375→      priority: 'medium',
   376→      suggestedAction: 'Coordinate with COO AI on cost optimization',
   377→    });
   378→  }
   379→  return recs;
   380→}

   744→function departmentTaskCount(role: EmployeeRole, data: WorkforceDataView): number {
   745→  const recs = departmentRecs(role, data);
   746→  const ops = data.operations;
   747→  // Each recommendation generates a task; plus department-specific baseline
   748→  const dept = getRoleDefinition(role).department;
   749→  let baseline = 0;
   750→  if (dept === 'operations') baseline = ops.openTasks;
   751→  if (dept === 'compliance') baseline = data.compliance.pendingFilings + data.compliance.notices;
   752→  if (dept === 'support') baseline = data.support.openTickets;
   753→  if (dept === 'finance') baseline = data.finance.outstandingInvoices;
   754→  if (dept === 'procurement') baseline = data.procurement.pendingBills;
   755→  return recs.length + baseline;
   756→}

   762→// ─── Formatter helper (avoid circular import — inline minimal version) ───────
   763→
   764→function formatINRShort(n: number): string {
   765→  if (!isFinite(n) || isNaN(n)) return '₹0';
   766→  const abs = Math.abs(n);
   767→  if (abs >= 10000000) return `₹${(n / 10000000).toFixed(2)} Cr`;
   768→  if (abs >= 100000) return `₹${(n / 100000).toFixed(2)} L`;
   769→  if (abs >= 1000) return `₹${(n / 1000).toFixed(1)}K`;
   770→  return '₹' + Math.round(n).toLocaleString('en-IN');
   771→}

   773→// ─── Build a single AI employee ───────────────────────────────────────────────
   774→
   775→function buildEmployee(
   776→  def: RoleDefinition,
   777→  data: WorkforceDataView,
   778→): AIEmployee {
   779→  const kpis = departmentKPIs(def.role, data);
   780→  const recommendations = departmentRecs(def.role, data).slice(0, 5);
   781→  const alertCount = departmentAlertCount(def.role, data);
   782→  const taskCount = departmentTaskCount(def.role, data);
   783→  const decisionCount = departmentDecisionCount(def.role, data);
   784→  const healthScore = departmentHealth(def.role, data);
   785→  const hasLiveData = data.hasLiveData;
   786→
   787→  // Performance, memory, skills computed by their respective engines
   788→  const performance = computeEmployeePerformance(def.role, data, healthScore, taskCount);
   789→  const recentMemory = computeEmployeeMemory(def.role, data).slice(0, 5);
   790→  const skills = computeEmployeeSkills(def.role, data).slice(0, 4);
   791→
   792→  const status = detectStatus(def.role, data, alertCount, decisionCount, hasLiveData);
   793→
   794→  return {
   795→    role: def.role,
   796→    name: def.name,
   797→    title: def.title,
   798→    department: def.department,
   799→    tier: def.tier,
   800→    reportsTo: def.reportsTo,
   801→    directReports: def.directReports,
   802→    status,
   803→    healthScore,
   804→    avatarColor: def.avatarColor,
   805→    icon: def.icon,
   806→    responsibilities: def.responsibilities,
   807→    monitors: def.monitors,
   808→    kpis,
   809→    openTaskCount: taskCount,
   810→    activeAlertCount: alertCount,
   811→    pendingDecisionCount: decisionCount,
   812→    recommendations,
   813→    performance,
   814→    recentMemory,
   815→    skills,
   816→    lastActiveAt: new Date().toISOString(),
   817→    dataSources: def.dataSources,
   818→    hasLiveData,
   819→  };
   820→}
