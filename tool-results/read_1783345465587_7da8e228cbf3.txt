    76→/** Convert each country's local-currency revenue to USD for fair comparison. */
    77→function revenueUSD(c: Country): number {
    78→  return convertCurrency(c.revenue, c.currency, 'USD');
    79→}
    80→
    81→function fmtCompactUSD(n: number): string {
    82→  const abs = Math.abs(n);
    83→  if (abs >= 1_000_000_000) return `$${(n / 1_000_000_000).toFixed(2)}B`;
    84→  if (abs >= 1_000_000) return `$${(n / 1_000_000).toFixed(2)}M`;
    85→  if (abs >= 1_000) return `$${(n / 1_000).toFixed(1)}K`;
    86→  return `$${Math.round(n).toLocaleString('en-US')}`;
    87→}
    88→
    89→function scoreColor(score: number): string {
    90→  if (score >= 90) return 'text-emerald-400';
    91→  if (score >= 80) return 'text-teal-400';
    92→  if (score >= 70) return 'text-amber-400';
    93→  return 'text-rose-400';
    94→}

   233→// ─── Exchange-rate mini chart (pure CSS) ───────────────────────────────────────
   234→
   235→function FxMiniChart({
   236→  title, from, to, data, color,
   237→}: {
   238→  title: string;
   239→  from: string;
   240→  to: string;
   241→  data: { month: string; value: number }[];
   242→  color: string;
   243→}) {
   244→  const max = Math.max(...data.map((d) => d.value));
   245→  const min = Math.min(...data.map((d) => d.value));
   246→  const range = max - min || 1;
   247→  const last = data[data.length - 1].value;
   248→  const first = data[0].value;
   249→  const delta = last - first;
   250→  const deltaPct = (delta / first) * 100;
   251→  const up = delta >= 0;
   252→
   253→  return (
   254→    <div className="rounded-xl border border-white/[0.06] bg-white/[0.02] p-4">
   255→      <div className="flex items-center justify-between">
   256→        <div>
   257→          <p className="text-sm font-semibold text-white">{title}</p>
   258→          <p className="text-[11px] text-muted-foreground">6-month trend</p>
   259→        </div>
   260→        <div className="text-right">
   261→          <p className={cn('text-base font-semibold', color)}>{last.toFixed(2)}</p>
   262→          <p className={cn('text-[11px] flex items-center gap-0.5 justify-end', up ? 'text-emerald-400' : 'text-rose-400')}>
   263→            {up ? <ArrowUpRight className="h-3 w-3" /> : <ArrowDownRight className="h-3 w-3" />}
   264→            {up ? '+' : ''}{deltaPct.toFixed(2)}%
   265→          </p>
   266→        </div>
   267→      </div>
   268→      <div className="mt-4 flex items-end gap-1.5 h-24">
   269→        {data.map((d, i) => {
   270→          const h = 18 + ((d.value - min) / range) * 80;
   271→          return (
   272→            <motion.div
   273→              key={d.month}
   274→              initial={{ height: 0 }}
   275→              animate={{ height: `${h}%` }}
   276→              transition={{ duration: 0.5, delay: i * 0.05 }}
   277→              className="flex-1 flex flex-col items-center gap-1"
   278→            >
   279→              <div className={cn('w-full rounded-t-sm', color.replace('text-', 'bg-'))} style={{ height: '100%' }} />
   280→              <span className="text-[9px] text-muted-foreground">{d.month}</span>
   281→            </motion.div>
   282→          );
   283→        })}
   284→      </div>
   285→      <div className="mt-2 flex items-center justify-between text-[10px] text-muted-foreground">
   286→        <span>1 {from} =</span>
   287→        <span className="font-mono">{last.toFixed(4)} {to}</span>
   288→      </div>
   289→    </div>
   290→  );
   291→}
