   208→   208→function formatRelativeTime(timestamp: string): string {
   209→   209→  const now = new Date();
   210→   210→  const date = new Date(timestamp);
   211→   211→  const diffMs = now.getTime() - date.getTime();
   212→   212→  const diffMins = Math.floor(diffMs / 60000);
   213→   213→  const diffHours = Math.floor(diffMins / 60);
   214→   214→  const diffDays = Math.floor(diffHours / 24);
   215→   215→
   216→   216→  if (diffMins < 60) return `${diffMins}m ago`;
   217→   217→  if (diffHours < 24) return `${diffHours}h ago`;
   218→   218→  if (diffDays < 7) return `${diffDays}d ago`;
   219→   219→  return date.toLocaleDateString('en-IN', { day: 'numeric', month: 'short' });
   220→   220→}

   244→   244→/** Map an API document to a DocumentEntry for the UI */
   245→   245→function mapApiDocument(doc: Record<string, unknown>): DocumentEntry {
   246→   246→  const statusStr = (doc.processingStep as string) ?? (doc.status as string) ?? 'uploaded';
   247→   247→  const statusMap: Record<string, DocumentEntry['status']> = {
   248→   248→    extracted: 'Processed',
   249→   249→    processing: 'Processing',
   250→   250→    processed: 'Processed',
   251→   251→    uploaded: 'Uploaded',
   252→   252→    failed: 'Error',
   253→   253→    error: 'Error',
   254→   254→  };
   255→   255→  const sizeBytes = (doc.size as number) ?? 0;
   256→   256→  const sizeStr = sizeBytes > 1048576
   257→   257→    ? `${(sizeBytes / 1048576).toFixed(1)} MB`
   258→   258→    : sizeBytes > 1024
   259→   259→      ? `${(sizeBytes / 1024).toFixed(0)} KB`
   260→   260→      : sizeBytes > 0
   261→   261→        ? `${sizeBytes} B`
   262→   262→        : '1.0 MB';
   263→   263→
   264→   264→  return {
   265→   265→    id: (doc.id as string) ?? '',
   266→   266→    name: (doc.name as string) ?? (doc.filename as string) ?? 'Document',
   267→   267→    type: (doc.fileType as string) ?? (doc.folder as string) ?? 'Document',
   268→   268→    uploadDate: ((doc.createdAt as string) ?? new Date().toISOString()).split('T')[0],
   269→   269→    status: statusMap[statusStr.toLowerCase()] ?? 'Uploaded',
   270→   270→    invoicesExtracted: (doc.invoiceCount as number) ?? 0,
   271→   271→    totalRows: (doc.rowCount as number) ?? 0,
   272→   272→    accuracy: (doc.accuracy as number) ?? 0,
   273→   273→    size: sizeStr,
   274→   274→  };
   275→   275→}

   278→   278→function mapApiReconRun(run: ReconciliationRun): ReconRun {

   285→   285→    missingInvoices: run.unmatched,
   286→   286→    taxDifference: run.gstDifference,
   287→   287→    runDate: run.createdAt.split('T')[0],
   288→   288→    status: run.status === 'completed' ? 'Completed' as const : run.status === 'running' ? 'Running' as const : 'Failed' as const,
   289→   289→  };

   292→   292→/** Map an API audit log to an ActivityEvent for the UI */
   293→   293→function mapAuditLogToEvent(log: AuditLogEntry): ActivityEvent {

   295→   295→  let type: ActivityEvent['type'] = 'user_action';

   297→   297→  else if (action.includes('file') || action.includes('submit')) type = 'filing_submitted';
   298→   298→  else if (action.includes('reconcil')) type = 'recon_run';
   299→   299→  else if (action.includes('valid') || action.includes('generat')) type = 'ai_action';
   300→   300→  else if (action.includes('return') || action.includes('creat')) type = 'return_created';
   301→   301→
   302→   302→  return {
   303→   303→    id: log.id,
   304→   304→    type,
   305→   305→    description: log.details ?? log.action,
   306→   306→    timestamp: log.timestamp,
   307→   307→  };

   353→   353→function HealthRing({ score, size = 72, strokeWidth = 5 }: { score: number; size?: number; strokeWidth?: number }) {
   354→   354→  const radius = (size - strokeWidth) / 2;
   355→   355→  const circumference = 2 * Math.PI * radius;
   356→   356→  const offset = circumference - (score / 100) * circumference;
   357→   357→  const color = getHealthColor(score);
   358→   358→
   359→   359→  return (
   360→   360→    <div className="relative inline-flex items-center justify-center" style={{ width: size, height: size }}>
   361→   361→      <svg width={size} height={size} className="-rotate-90">
   362→   362→        <circle cx={size / 2} cy={size / 2} r={radius} fill="none" stroke="#e2e8f0" strokeWidth={strokeWidth} />
   363→   363→        <motion.circle cx={size / 2} cy={size / 2} r={radius} fill="none" stroke={color.stroke} strokeWidth={strokeWidth} strokeLinecap="round" strokeDasharray={circumference} initial={{ strokeDashoffset: circumference }} animate={{ strokeDashoffset: offset }} transition={{ duration: 1, delay: 0.3, ease: 'easeOut' }} />
   364→   364→      </svg>
   365→   365→      <div className="absolute inset-0 flex flex-col items-center justify-center">
   366→   366→        <span className={`font-bold text-lg ${color.text}`}>{score}</span>
   367→   367→        <span className="text-[8px] text-muted-foreground">/ 100</span>
   368→   368→      </div>
   369→   369→    </div>
   370→   370→  );
