   133→// ─── Helper ───────────────────────────────────────────────────────────────────
   134→
   135→function formatFileSize(bytes: number): string {
   136→  if (bytes < 1024) return `${bytes} B`;
   137→  if (bytes < 1048576) return `${(bytes / 1024).toFixed(1)} KB`;
   138→  return `${(bytes / 1048576).toFixed(1)} MB`;
   139→}
   140→
   141→function getFileIcon(fileName: string) {
   142→  const ext = fileName.split('.').pop()?.toLowerCase();
   143→  if (ext === 'json') return <FileJson className="size-4 text-amber-500" />;
   144→  if (ext === 'csv' || ext === 'xlsx') return <FileSpreadsheet className="size-4 text-emerald-500" />;
   145→  return <FileText className="size-4 text-slate-500" />;
   146→}

   187→

   204→

   230→  const handleUpload = useCallback(async (files: File[]) => {
   231→    const valid: File[] = [];

   233→      const err = validateFile(file);

   239→    }

   243→      const uploaded = await uploadMany(valid, { category: 'invoices' });

   245→      if (uploaded.length > 0) {
   246→        toast.success(`${uploaded.length} file${uploaded.length !== 1 ? 's' : ''} uploaded to Firebase Storage`);

   272→    const files = Array.from(e.dataTransfer.files);
   273→    if (files.length > 0) handleUpload(files);

   391→
   392→  // ─── Summary Cards ────────────────────────────────────────────────────────────
   393→
   394→  const summaryCards = [

   396→      title: 'Total Invoices',

   421→      bgColor: 'bg-teal-50',
   422→    },
   423→    {

   431→
   432→  return (
   433→    <div className="min-h-screen bg-gradient-to-b from-slate-50/80 to-white">

   435→        {/* ── Header ── */}
   436→        <motion.div
   437→          initial={{ opacity: 0, y: -12 }}
   438→          animate={{ opacity: 1, y: 0 }}

   441→        >
   442→          <div>
   443→            <h1 className="text-2xl font-bold tracking-tight text-slate-900 sm:text-3xl">

   447→              {invoices.length} invoice{invoices.length !== 1 ? 's' : ''} &middot; {documents.length} document{documents.length !== 1 ? 's' : ''}

   549→        <input
