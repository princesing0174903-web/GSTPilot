     5→function rand(min: number, max: number) {
     6→  return Math.floor(Math.random() * (max - min + 1)) + min
     7→}
     8→
     9→function pick<T>(arr: T[]): T {
    10→  return arr[Math.floor(Math.random() * arr.length)]
    11→}

    13→function randomDate(baseYear: number, month: number, dayRange: [number, number]): string {
    14→  const day = rand(dayRange[0], dayRange[1])
    15→  const m = String(month).padStart(2, '0')
    16→  const d = String(day).padStart(2, '0')
    17→  return `${baseYear}-${m}-${d}`
    18→}

   483→      if (matchStatus === 'perfect_match') {

   486→
   487→      // Parse mismatches to determine recommendation

   491→          const fields = mismatchArr.map((m) => m.field)
   492→
   493→          if (fields.includes('buyerGstin') || fields.includes('sellerGstin') || fields.includes('matchedGstin')) {
   494→            return 'correct_gstin'
   495→          }
   496→          if (fields.includes('taxableValue') || fields.includes('cgst') || fields.includes('sgst') || fields.includes('igst')) {
   497→            return 'adjust_gst_amount'
   498→          }
   499→          if (fields.includes('invoiceNumber')) {
   500→            return 'correct_invoice_number'
   501→          }
   502→        } catch {
   503→          // fallback below
   504→        }
   505→      }
   506→
   507→      // Default recommendations by matchStatus
   508→      switch (matchStatus) {
   509→        case 'mismatch':
   510→          return pick(['adjust_gst_amount', 'correct_gstin', 'review_vendor_data'])
   511→        case 'partial_match':
   512→          return pick(['correct_gstin', 'correct_invoice_number', 'review_vendor_data'])
   513→        case 'missing_in_books':
   514→          return pick(['mark_as_duplicate', 'review_manually'])
   515→        case 'missing_in_gstr':
   516→          return 'review_manually'
   517→        default:
   518→          return 'review_manually'
   519→      }
   520→    }
