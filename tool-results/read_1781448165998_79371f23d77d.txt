   203→// ─── Portfolio Derivation Helpers ──────────────────────────────────────────────
   204→
   205→/** Compute a ClientPortfolio from a store client by deriving portfolio metrics from store data */
   206→function derivePortfolio(client: SampleClient, store: ReturnType<typeof useGSTStore>): ClientPortfolio {
   207→  const filings = store.getFilingsForClient(client.id);
   208→  const issues = store.getIssuesForClient(client.id);
   209→  const invoices = store.getInvoicesForClient(client.id);
   210→  const reconSummary = store.getReconSummary(client.id);
   211→
   212→  const hs = client.healthScore;
   213→  const pendingFilings = filings.filter(f => f.status !== 'filed').length;
   214→  const taxVolume = filings.reduce((sum, f) => sum + f.totalTaxableValue, 0);
   215→  const unresolvedIssues = issues.filter(i => !i.resolved).length;
   216→
   217→  // Derive risk level from health score and issues
   218→  const riskLevel: RiskLevel = hs > 80 ? 'Low' : hs >= 50 ? 'Medium' : 'High';
   219→  const riskDetail = riskLevel === 'High' ? (pendingFilings > 3 ? 'Multiple delays' : 'Filing delays')
   220→    : riskLevel === 'Medium' ? `${unresolvedIssues} issue${unresolvedIssues !== 1 ? 's' : ''}`
   221→    : undefined;
   222→
   223→  // Derive match rate from recon summary
   224→  let matchRate = 78;
   225→  if (reconSummary.length > 0) {
   226→    const total = reconSummary.reduce((sum, c) => sum + c.count, 0);
   227→    const perfectMatch = reconSummary.find(c => c.label === 'Perfect Match');
   228→    matchRate = perfectMatch && total > 0 ? Math.round((perfectMatch.count / total) * 100) : 0;
   229→  } else {
   230→    matchRate = hs > 80 ? 94 : hs < 50 ? 58 : 78;
   231→  }
   232→
   233→  // Derive health breakdown from health score components
   234→  const gstinValidity = Math.min(100, hs + 3);
   235→  const filingTimeliness = pendingFilings === 0 ? Math.min(100, hs + 5) : Math.max(10, hs - pendingFilings * 8);
   236→  const invoiceAccuracy = Math.min(100, Math.max(10, matchRate));
   237→
   238→  // Recent filings (last 3 filings from store)
   239→  const recentFilings = filings.slice(0, 3).map(f => {
   240→    const periodLabel = f.period.replace('-', ' ');
   241→    const monthNames = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];
   242→    const [year, month] = f.period.split('-').map(Number);
   243→    const label = `${monthNames[month - 1]} ${year}`;
   244→    return {
   245→      period: label,
   246→      type: f.returnType,
   247→      status: f.status === 'filed' ? 'Filed' : f.filedDate ? 'Filed' : pendingFilings > 2 ? 'Overdue' : 'Pending',
   248→      date: f.filedDate ?? '',
   249→    };
   250→  });
   251→
   252→  // Monthly volume chart — derive from filings' taxable values
   253→  const monthlyVolumeChart: MonthlyVolume[] = [
   254→    { month: 'Dec', amount: Math.round(taxVolume * 0.85) },
   255→    { month: 'Jan', amount: Math.round(taxVolume * 0.92) },
   256→    { month: 'Feb', amount: Math.round(taxVolume * 0.88) },
   257→    { month: 'Mar', amount: Math.round(taxVolume * 1.1) },
   258→    { month: 'Apr', amount: Math.round(taxVolume * 0.95) },
   259→    { month: 'May', amount: taxVolume },
   260→  ];
   261→
   262→  return {
   263→    ...client,
   264→    _portfolio: {
   265→      monthlyTaxVolume: taxVolume,
   266→      monthlyVolumeChart,
   267→      pendingFilings,
   268→      riskLevel,
   269→      riskDetail,
   270→      healthBreakdown: { gstinValidity, filingTimeliness, invoiceAccuracy },
   271→      recentFilings,
   272→    },
   273→  };
   274→}

   276→// ─── GSTIN Form Validation ──────────────────────────────────────────────────
   277→
   278→function validateGSTINForm(gstin: string): { valid: boolean; error?: string } {
   279→  if (!gstin) return { valid: false, error: 'GSTIN is required' };
   280→  const clean = formatGSTIN(gstin);
   281→  if (clean.length !== 15) return { valid: false, error: 'GSTIN must be 15 characters' };
   282→  if (!validateGSTIN(clean)) return { valid: false, error: 'Invalid GSTIN format (e.g. 27AABCS1234F1ZH)' };
   283→  const stateCode = clean.slice(0, 2);
   284→  const validCodes = INDIAN_STATES.map(s => s.code);
   285→  if (!validCodes.includes(stateCode)) return { valid: false, error: 'Invalid state code in GSTIN' };
   286→  return { valid: true };
   287→}

   291→function getHealthColor(score: number): { bg: string; text: string; bar: string; stroke: string } {

   294→  return { bg: 'bg-red-50', text: 'text-red-700', bar: 'bg-red-500', stroke: '#ef4444' };
   295→}

   362→function formatVolumeShort(amount: number): string {
   363→  if (amount >= 10000000) return `₹${(amount / 10000000).toFixed(2)} Cr`;
   364→  if (amount >= 100000) return `₹${(amount / 100000).toFixed(1)}L`;
   365→  if (amount >= 1000) return `₹${(amount / 1000).toFixed(1)}K`;
   366→  return `₹${amount}`;
   367→}

   370→  if (amount >= 10000000) return `₹${(amount / 10000000).toFixed(2)} Cr`;
   371→  return `₹${(amount / 100000).toFixed(2)}L`;
   372→}

   376→function HealthRing({ score, size = 60, strokeWidth = 5, showLabel = true }: { score: number; size?: number; strokeWidth?: number; showLabel?: boolean }) {
   377→  const radius = (size - strokeWidth) / 2;
   378→  const circumference = 2 * Math.PI * radius;
   379→  const offset = circumference - (score / 100) * circumference;
   380→  const color = getHealthColor(score);
   381→
   382→  return (
   383→    <div className="relative inline-flex items-center justify-center" style={{ width: size, height: size }}>
   384→      <svg width={size} height={size} className="-rotate-90">
   385→        <circle
   386→          cx={size / 2}
   387→          cy={size / 2}
   388→          r={radius}
   389→          fill="none"
   390→          stroke="#e2e8f0"
   391→          strokeWidth={strokeWidth}
   392→        />
   393→        <motion.circle
   394→          cx={size / 2}
   395→          cy={size / 2}
   396→          r={radius}
   397→          fill="none"
   398→          stroke={color.stroke}
   399→          strokeWidth={strokeWidth}
   400→          strokeLinecap="round"
   401→          strokeDasharray={circumference}
   402→          initial={{ strokeDashoffset: circumference }}
   403→          animate={{ strokeDashoffset: offset }}
   404→          transition={{ duration: 1, delay: 0.3, ease: 'easeOut' }}
   405→        />
   406→      </svg>
   407→      {showLabel && (
   408→        <div className="absolute inset-0 flex flex-col items-center justify-center">
   409→          <span className={`font-bold leading-none ${size >= 100 ? 'text-2xl' : 'text-sm'} ${color.text}`}>

   411→          </span>
   412→          {size >= 100 && (
   413→            <span className="text-[9px] text-muted-foreground mt-0.5">/ 100</span>

   415→        </div>
   416→      )}
   417→    </div>
   418→  );
   419→}

   434→// ─── Filing Trend Area Chart ────────────────────────────────────────────────────
   435→
   436→function FilingTrendChart({ data }: { data: MonthlyVolume[] }) {
   437→  const width = 260;
   438→  const height = 80;
   439→  const padding = { top: 8, right: 8, bottom: 20, left: 8 };
   440→  const chartW = width - padding.left - padding.right;
   441→  const chartH = height - padding.top - padding.bottom;
   442→  const maxAmount = Math.max(...data.map(d => d.amount), 1);
   443→
   444→  const points = data.map((d, i) => ({
   445→    x: padding.left + (i / (data.length - 1)) * chartW,
   446→    y: padding.top + chartH - (d.amount / maxAmount) * chartH,
   447→    ...d,
   448→  }));
   449→
   450→  const linePath = points.map((p, i) => `${i === 0 ? 'M' : 'L'} ${p.x} ${p.y}`).join(' ');
   451→  const areaPath = `${linePath} L ${points[points.length - 1].x} ${padding.top + chartH} L ${points[0].x} ${padding.top + chartH} Z`;
   452→
   453→  const [hoverIdx, setHoverIdx] = useState<number | null>(null);
   454→
   455→  return (
   456→    <div className="relative">
   457→      <svg width="100%" viewBox={`0 0 ${width} ${height}`} className="overflow-visible">
   458→        {/* Area fill */}
   459→        <motion.path
   460→          d={areaPath}
   461→          fill="url(#areaGradient)"
   462→          initial={{ opacity: 0 }}
   463→          animate={{ opacity: 1 }}
   464→          transition={{ duration: 0.6, delay: 0.3 }}
   465→        />
   466→        {/* Line */}
   467→        <motion.path
   468→          d={linePath}
   469→          fill="none"
   470→          stroke="#10b981"
   471→          strokeWidth={2}
   472→          strokeLinecap="round"
   473→          strokeLinejoin="round"
   474→          initial={{ pathLength: 0 }}
   475→          animate={{ pathLength: 1 }}
   476→          transition={{ duration: 0.8, delay: 0.2 }}
   477→        />
   478→        {/* Dots */}
   479→        {points.map((p, i) => (
   480→          <circle
   481→            key={i}
   482→            cx={p.x}
   483→            cy={p.y}
   484→            r={hoverIdx === i ? 4 : 2.5}
   485→            fill="#10b981"
   486→            stroke="white"
   487→            strokeWidth={2}
   488→            className="transition-all duration-150 cursor-pointer"
   489→            onMouseEnter={() => setHoverIdx(i)}
   490→            onMouseLeave={() => setHoverIdx(null)}
   491→          />
   492→        ))}
   493→        {/* Month labels */}
   494→        {points.map((p, i) => (
   495→          <text
   496→            key={i}
   497→            x={p.x}
   498→            y={height - 4}
   499→            textAnchor="middle"
   500→            className="fill-muted-foreground"
   501→            fontSize="8"
   502→          >
   503→            {data[i].month}
   504→          </text>
   505→        ))}
   506→        <defs>
   507→          <linearGradient id="areaGradient" x1="0" y1="0" x2="0" y2="1">
   508→            <stop offset="0%" stopColor="#10b981" stopOpacity="0.2" />
   509→            <stop offset="100%" stopColor="#10b981" stopOpacity="0.02" />
   510→          </linearGradient>
   511→        </defs>
   512→      </svg>
   513→      {/* Hover tooltip */}
   514→      {hoverIdx !== null && (
   515→        <div
   516→          className="absolute -top-8 px-2 py-1 rounded bg-slate-800 text-white text-[10px] font-medium pointer-events-none whitespace-nowrap z-10"
   517→          style={{ left: `${((points[hoverIdx].x) / width) * 100}%`, transform: 'translateX(-50%)' }}
   518→        >
   519→          {formatCurrency(data[hoverIdx].amount)}
   520→        </div>
   521→      )}
   522→    </div>
   523→  );
   524→}

   528→function MiniBarChart({ data, height = 28, barColor }: { data: MonthlyVolume[]; height?: number; barColor?: string }) {

   530→

   534→        const barHeight = Math.max(3, (d.amount / maxAmount) * height);
