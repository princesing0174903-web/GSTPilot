   216→   216→function MatchRateRing({
   217→   217→  matchPercent,
   218→   218→  partialPercent,
   219→   219→  mismatchPercent,
   220→   220→  size = 160,
   221→   221→  strokeWidth = 14,
   222→   222→}: {
   223→   223→  matchPercent: number;
   224→   224→  partialPercent: number;
   225→   225→  mismatchPercent: number;
   226→   226→  size?: number;
   227→   227→  strokeWidth?: number;
   228→   228→}) {
   229→   229→  const radius = (size - strokeWidth) / 2;
   230→   230→  const circumference = 2 * Math.PI * radius;
   231→   231→  const gapAngle = 2; // degrees gap between segments
   232→   232→
   233→   233→  const totalGap = gapAngle * 3 * (circumference / 360);
   234→   234→  const availableCircumference = circumference - totalGap;
   235→   235→  const gapLen = totalGap / 3;
   236→   236→
   237→   237→  const matchLen = (matchPercent / 100) * availableCircumference;
   238→   238→  const partialLen = (partialPercent / 100) * availableCircumference;
   239→   239→  const mismatchLen = (mismatchPercent / 100) * availableCircumference;
   240→   240→
   241→   241→  const matchOffset = 0;
   242→   242→  const partialOffset = -(matchLen + gapLen);
   243→   243→  const mismatchOffset = -(matchLen + gapLen + partialLen + gapLen);
   244→   244→
   245→   245→  return (
   246→   246→    <div className="relative inline-flex items-center justify-center" style={{ width: size, height: size }}>
   247→   247→      <svg width={size} height={size} className="-rotate-90">
   248→   248→        {/* Background track */}
   249→   249→        <circle
   250→   250→          cx={size / 2}
   251→   251→          cy={size / 2}
   252→   252→          r={radius}
   253→   253→          fill="none"
   254→   254→          stroke="#f1f5f9"
   255→   255→          strokeWidth={strokeWidth}
   256→   256→        />
   257→   257→        {/* Glow filter */}
   258→   258→        <defs>
   259→   259→          <filter id="ringGlow" x="-20%" y="-20%" width="140%" height="140%">
   260→   260→            <feGaussianBlur stdDeviation="2.5" result="blur" />
   261→   261→            <feMerge>
   262→   262→              <feMergeNode in="blur" />
   263→   263→              <feMergeNode in="SourceGraphic" />
   264→   264→            </feMerge>
   265→   265→          </filter>
   266→   266→        </defs>
   267→   267→        {/* Green segment — matched */}
   268→   268→        {matchPercent > 0 && (
   269→   269→          <motion.circle
   270→   270→            cx={size / 2}
   271→   271→            cy={size / 2}
   272→   272→            r={radius}
   273→   273→            fill="none"
   274→   274→            stroke="#10b981"
   275→   275→            strokeWidth={strokeWidth}
   276→   276→            strokeLinecap="round"
   277→   277→            strokeDasharray={`${matchLen} ${circumference - matchLen}`}
   278→   278→            strokeDashoffset={matchOffset}
   279→   279→            initial={{ strokeDasharray: `0 ${circumference}` }}
   280→   280→            animate={{ strokeDasharray: `${matchLen} ${circumference - matchLen}` }}
   281→   281→            transition={{ duration: 1.4, ease: [0.25, 0.46, 0.45, 0.94] }}
   282→   282→            filter="url(#ringGlow)"
   283→   283→          />
   284→   284→        )}
   285→   285→        {/* Amber segment — partial */}
   286→   286→        {partialPercent > 0 && (
   287→   287→          <motion.circle
   288→   288→            cx={size / 2}
   289→   289→            cy={size / 2}
   290→   290→            r={radius}
   291→   291→            fill="none"
   292→   292→            stroke="#f59e0b"
   293→   293→            strokeWidth={strokeWidth}
   294→   294→            strokeLinecap="round"
   295→   295→            strokeDasharray={`${partialLen} ${circumference - partialLen}`}
   296→   296→            strokeDashoffset={partialOffset}
   297→   297→            initial={{ strokeDasharray: `0 ${circumference}`, strokeDashoffset: 0 }}
   298→   298→            animate={{ strokeDasharray: `${partialLen} ${circumference - partialLen}`, strokeDashoffset: partialOffset }}
   299→   299→            transition={{ duration: 1.4, ease: [0.25, 0.46, 0.45, 0.94], delay: 0.3 }}
   300→   300→            filter="url(#ringGlow)"
   301→   301→          />
   302→   302→        )}
   303→   303→        {/* Red segment — mismatch/missing */}
   304→   304→        {mismatchPercent > 0 && (
   305→   305→          <motion.circle
   306→   306→            cx={size / 2}
   307→   307→            cy={size / 2}
   308→   308→            r={radius}
   309→   309→            fill="none"
   310→   310→            stroke="#ef4444"
   311→   311→            strokeWidth={strokeWidth}
   312→   312→            strokeLinecap="round"
   313→   313→            strokeDasharray={`${mismatchLen} ${circumference - mismatchLen}`}
   314→   314→            strokeDashoffset={mismatchOffset}
   315→   315→            initial={{ strokeDasharray: `0 ${circumference}`, strokeDashoffset: 0 }}
   316→   316→            animate={{ strokeDasharray: `${mismatchLen} ${circumference - mismatchLen}`, strokeDashoffset: mismatchOffset }}
   317→   317→            transition={{ duration: 1.4, ease: [0.25, 0.46, 0.45, 0.94], delay: 0.6 }}
   318→   318→            filter="url(#ringGlow)"
   319→   319→          />
   320→   320→        )}
   321→   321→      </svg>
   322→   322→      {/* Center content */}
   323→   323→      <div className="absolute inset-0 flex flex-col items-center justify-center">
   324→   324→        <motion.span
   325→   325→          className="text-4xl font-bold tracking-tight text-foreground"
   326→   326→          initial={{ opacity: 0, scale: 0.5 }}
   327→   327→          animate={{ opacity: 1, scale: 1 }}
   328→   328→          transition={{ delay: 0.8, duration: 0.5, type: 'spring' }}

   330→   330→          {matchPercent}%
   331→   331→        </motion.span>
   332→   332→        <span className="text-[11px] font-medium text-muted-foreground mt-0.5">Match Rate</span>
   333→   333→      </div>
   334→   334→    </div>
   335→   335→  );
   336→   336→}

   390→   390→// ─── Map API ReconciliationResult to local MismatchRecord ─────────────────
   391→   391→function mapReconResultToMismatch(
   392→   392→  result: ReconciliationResult,
   393→   393→  clientName: string
   394→   394→): MismatchRecord {
   395→   395→  // Parse mismatches JSON string
   396→   396→  let parsedMismatches: MismatchDetail[] = [];
   397→   397→  try {
   398→   398→    if (result.mismatches) {
   399→   399→      parsedMismatches = JSON.parse(result.mismatches);
   400→   400→    }
   401→   401→  } catch {
   402→   402→    parsedMismatches = [];
   403→   403→  }
   404→   404→
   405→   405→  // Derive category from matchStatus
   406→   406→  const mismatchCategory =
   407→   407→    result.matchStatus === 'missing_in_gstr' ? 'missing_in_gstr' :
   408→   408→    result.matchStatus === 'missing_in_books' ? 'missing_in_books' :
   409→   409→    result.matchStatus === 'duplicate' ? 'duplicate' :
   410→   410→    parsedMismatches.some(m => m.field === 'vendor_gstin' || m.field === 'sellerGstin') ? 'gstin_mismatch' :
   411→   411→    'tax_difference';
   412→   412→
   413→   413→  const mismatchType =
   414→   414→    result.matchStatus === 'perfect_match' ? 'Perfect Match' :
   415→   415→    result.matchStatus === 'partial_match' ? 'Partial Match' :
   416→   416→    result.matchStatus === 'missing_in_gstr' ? 'Missing in Portal' :
   417→   417→    result.matchStatus === 'missing_in_books' ? 'Missing in Books' :
   418→   418→    result.matchStatus === 'duplicate' ? 'Duplicate' :
   419→   419→    'Mismatch';
   420→   420→
   421→   421→  // Compute tax difference from mismatch details
   422→   422→  const taxDiff = parsedMismatches.reduce((sum, m) => sum + (m.difference || 0), 0);
   423→   423→
   424→   424→  // Extract diff fields for highlighting
   425→   425→  const diffFields = parsedMismatches.map(m => m.field);
   426→   426→
   427→   427→  // Build booksData from invoice if available, else from mismatch details
   428→   428→  const booksTaxableValue = parsedMismatches.find(m => m.field === 'taxableValue')?.expected as number || 0;
   429→   429→  const portalTaxableValue = parsedMismatches.find(m => m.field === 'taxableValue')?.actual as number || 0;
   430→   430→
   431→   431→  return {
   432→   432→    id: result.id,
   433→   433→    invoiceNumber: result.invoice?.invoiceNumber ?? result.sourceGstin ?? 'Unknown',
   434→   434→    clientName,
   435→   435→    mismatchCategory,
   436→   436→    mismatchType,
   437→   437→    taxDifference: Math.abs(taxDiff),
   438→   438→    confidenceScore: result.confidenceScore ?? 90,
   439→   439→    aiExplanation: result.aiExplanation ?? generateMismatchExplanation(diffFields),
   440→   440→    aiRecommendation: result.aiRecommendation ?? 'Review manually',
   441→   441→    booksData: {
   442→   442→      invoiceNumber: result.invoice?.invoiceNumber ?? '—',
   443→   443→      invoiceDate: result.invoice?.invoiceDate ?? '—',
   444→   444→      sellerGstin: result.sourceGstin ?? result.invoice?.sellerGstin ?? '—',
   445→   445→      buyerGstin: result.invoice?.buyerGstin ?? '—',
   446→   446→      taxableValue: booksTaxableValue || (result.invoice?.taxableValue ?? 0),
   447→   447→      cgst: result.invoice?.cgst ?? 0,
   448→   448→      sgst: result.invoice?.sgst ?? 0,
   449→   449→      igst: result.invoice?.igst ?? 0,
   450→   450→      totalAmount: result.invoice?.totalAmount ?? 0,
   451→   451→    },
   452→   452→    gstr2bData: result.matchStatus === 'missing_in_gstr' ? null : {
   453→   453→      invoiceNumber: result.matchedGstin ?? result.invoice?.invoiceNumber ?? '—',
   454→   454→      invoiceDate: result.invoice?.invoiceDate ?? '—',
   455→   455→      sellerGstin: result.matchedGstin ?? result.sourceGstin ?? '—',
   456→   456→      buyerGstin: '—',
   457→   457→      taxableValue: portalTaxableValue,
   458→   458→      cgst: 0,
   459→   459→      sgst: 0,
   460→   460→      igst: 0,
   461→   461→      totalAmount: portalTaxableValue,
   462→   462→    },
   463→   463→    diffFields,
   464→   464→    resolved: result.resolved,
   465→   465→    workflowStatus: result.workflowStatus,
   466→   466→  };
   467→   467→}
