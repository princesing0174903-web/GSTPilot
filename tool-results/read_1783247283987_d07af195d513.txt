     4→// GSTPilot — Firestore Security Rules (Multi-Tenant Isolation)
     5→//
     6→// GOVERNANCE PRINCIPLE (PART 9):
     7→//   Every tenant-scoped document carries an `organizationId` field. A user
     8→//   may only read / write documents whose `organizationId` matches an
     9→//   organization they are an active member of.
    10→//
    11→//   Membership is resolved by reading `organization_members/{orgId}_{uid}`.
    12→//   This guarantees no user can ever see another organization's data — even
    13→//   if client-side code has a bug.
    14→//
    15→// COLLECTIONS:
    16→//   users                    — user profiles (self-owned)
    17→//   organizations            — tenant roots (members only)
    18→//   organization_members     — membership join (members only)
    19→//   onboarding               — onboarding questionnaires (self-owned)
    20→//   ── Tenant-scoped (all require matching organizationId) ──
    21→//   clients, invoices, payments, expenses, bank_accounts, bank_transactions,
    22→//   gst_profiles, gst_returns, notices, reports, tasks, ai_memory,
    23→//   notifications, documents, returns, reconciliations, activities,
    24→//   aiRecommendations, leads, deals, meetings, predictions, priorityQueue
    25→// ═══════════════════════════════════════════════════════════════════════════════

    30→    // ─── Helper Functions ─────────────────────────────────────────────────
    31→
    32→    function isSignedIn() {
    33→      return request.auth != null;
    34→    }
    35→
    36→    function uid() {
    37→      return request.auth.uid;
    38→    }
    39→
    40→    // The membership document id for a given org + user.
    41→    function memberDocId(orgId) {
    42→      return orgId + '_' + uid();
    43→    }
    44→
    45→    // True if the user is an active member of the given org.
    46→    function isOrgMember(orgId) {
    47→      return isSignedIn()
    48→        && exists(/databases/$(database)/documents/organization_members/$(memberDocId(orgId)))
    49→        && get(/databases/$(database)/documents/organization_members/$(memberDocId(orgId)))
    50→             .data.status == 'active';
    51→    }
    52→
    53→    // The user's role within the given org (or empty string).
    54→    function roleIn(orgId) {
    55→      return get(/databases/$(database)/documents/organization_members/$(memberDocId(orgId)))
    56→        .data.role;
    57→    }
    58→
    59→    function isOwnerOrAdmin(orgId) {
    60→      return isOrgMember(orgId)
    61→        && roleIn(orgId) in ['owner', 'admin'];
    62→    }
    63→
    64→    function canMutate(orgId) {
    65→      return isOrgMember(orgId)
    66→        && roleIn(orgId) in ['owner', 'admin', 'accountant', 'employee'];
    67→    }
    68→
    69→    // True if the stored resource's organizationId matches an org the user
    70→    // belongs to. Used for reads / updates / deletes.
    71→    function resourceBelongsToUserOrg() {
    72→      return isOrgMember(resource.data.organizationId);
    73→    }
    74→
    75→    // True if the incoming write's organizationId matches an org the user
    76→    // belongs to. Used on create / update.
    77→    function writeScopedToUserOrg() {
    78→      return isOrgMember(request.resource.data.organizationId);
    79→    }
    80→
    81→    // True on update if the organizationId is unchanged (prevents moving a
    82→    // document across tenants).
    83→    function orgIdUnchanged() {
    84→      return resource.data.organizationId == request.resource.data.organizationId;
    85→    }
    86→
    87→    // ─── User Profiles (self-owned) ───────────────────────────────────────
    88→    match /users/{userId} {
    89→      allow read:   if isSignedIn() && uid() == userId;
    90→      allow create: if isSignedIn() && uid() == userId;
    91→      allow update: if isSignedIn() && uid() == userId;
    92→      allow delete: if false; // account closure is server-side only
    93→    }

    96→    match /organizations/{orgId} {
    97→      allow read:   if isOrgMember(orgId);

    99→      allow update: if isOwnerOrAdmin(orgId);
   100→      allow delete: if isOrgMember(orgId) && roleIn(orgId) == 'owner';
   101→    }

   106→      allow read: if isSignedIn()
   107→        && (uid() == resource.data.userId
   108→            || isOrgMember(resource.data.organizationId));
   109→
   110→      // Owners/admins can create new member rows (invites).
   111→      allow create: if isSignedIn()
   112→        && isOwnerOrAdmin(request.resource.data.organizationId);
   113→
   114→      // Role / status changes are owner/admin only. A member may NOT change
   115→      // their own role (prevents privilege escalation).
   116→      allow update: if isSignedIn()
   117→        && isOwnerOrAdmin(resource.data.organizationId)
   118→        && uid() != resource.data.userId;
   119→
   120→      // Removal is owner/admin only. A member may remove themselves (leave).
   121→      allow delete: if isSignedIn()
   122→        && (isOwnerOrAdmin(resource.data.organizationId)
   123→            || uid() == resource.data.userId);
   124→    }
