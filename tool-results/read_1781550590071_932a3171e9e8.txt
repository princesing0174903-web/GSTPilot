   386→function formatTime(date: Date): string {
   387→  return date.toLocaleTimeString('en-IN', { hour: '2-digit', minute: '2-digit', second: '2-digit' })
   388→}
   389→
   390→function formatDate(date: Date): string {
   391→  return date.toLocaleDateString('en-IN', { day: '2-digit', month: 'short', year: 'numeric' })
   392→}
   393→
   394→function formatTimeAgo(date: Date): string {
   395→  const diff = Date.now() - date.getTime()
   396→  const mins = Math.floor(diff / 60000)
   397→  if (mins < 1) return 'Just now'
   398→  if (mins < 60) return `${mins}m ago`
   399→  const hrs = Math.floor(mins / 60)
   400→  if (hrs < 24) return `${hrs}h ago`
   401→  return `${Math.floor(hrs / 24)}d ago`
   402→}
   403→
   404→function formatDuration(seconds: number): string {
   405→  if (seconds < 60) return `${seconds}s`
   406→  const m = Math.floor(seconds / 60)
   407→  const s = seconds % 60
   408→  return `${m}m ${s}s`
   409→}
