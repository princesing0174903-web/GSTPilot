    33→function safeParseJSON<T>(raw: string | null | undefined, fallback: T): T {
    34→  if (!raw) return fallback;
    35→  try {
    36→    return JSON.parse(raw) as T;
    37→  } catch {
    38→    return fallback;
    39→  }

    42→function fmtINR(amount: number): string {
    43→  if (!Number.isFinite(amount)) return '₹0';
    44→  const abs = Math.abs(amount);
    45→  if (abs >= 1_00_00_000) {
    46→    return `₹${(amount / 1_00_00_000).toFixed(2)} Cr`;
    47→  }
    48→  if (abs >= 1_00_000) {
    49→    return `₹${(amount / 1_00_000).toFixed(2)} L`;
    50→  }
    51→  if (abs >= 1_000) {
    52→    return `₹${(amount / 1_000).toFixed(1)}K`;
    53→  }
    54→  return `₹${Math.round(amount)}`;
    55→}

    61→function clamp(value: number, min: number, max: number): number {
    62→  return Math.max(min, Math.min(max, value));

    65→/**
    66→ * Compute a 0-100 impact score from a ₹ amount.
    67→ *   ₹10L+ → 90+
    68→ *   ₹1L+  → 70+
    69→ *   ₹10K+ → 50+
    70→ *   ₹1K+  → 30+
    71→ *   else  → 10
    72→ */
    73→function impactFromAmount(amount: number): number {
    74→  const abs = Math.abs(amount);
    75→  if (abs >= 10_00_000) return clamp(90 + Math.floor(abs / 50_00_000) * 2, 90, 100);
    76→  if (abs >= 1_00_000) return clamp(70 + Math.floor((abs - 1_00_000) / 50_000), 70, 89);
    77→  if (abs >= 10_000) return clamp(50 + Math.floor((abs - 10_000) / 5_000), 50, 69);
    78→  if (abs >= 1_000) return clamp(30 + Math.floor((abs - 1_000) / 500), 30, 49);
    79→  return 10;
    80→}

    82→/** Compute a confidence score (60-95) from how many non-empty sources contributed. */
    83→function confidenceFromSources(sourceCount: number, maxValue = 95): number {
    84→  const base = 60;
    85→  const bonus = Math.min(maxValue - base, sourceCount * 8);
    86→  return clamp(base + bonus, 60, maxValue);
    87→}

   495→function buildBusinessInsight(
   496→  date: string,
   497→  raw: RawCollectedData,
   498→  ctx: BusinessContext,
   499→  cfo: CFOData | null,
   500→  ceo: CEOData | null,
   501→): InsightDraft {
   502→  const dataSources: AIModuleId[] = ['oracle'];
   503→  if (cfo) dataSources.push('cfo');
   504→  if (ceo) dataSources.push('ceo');
   505→
   506→  const reasoningBullets = (raw.recentReasoning ?? [])
   507→    .slice(0, 3)
   508→    .map((r, i) => `${i + 1}. ${r.request.slice(0, 120)} (confidence ${r.confidence}%)`)
   509→    .join('\n');
   510→
   511→  const topRisks = ctx.graph.topRisks
   512→    .slice(0, 2)
   513→    .map((r) => `• ${r.label} (score ${r.score})`)
   514→    .join('\n');
   515→
   516→  const body = [
   517→    `Date: ${date}`,
   518→    '',
   519→    'Top recent Oracle reasonings (by confidence):',
   520→    reasoningBullets || '• No recent reasoning records found.',
   521→    '',
   522→    topRisks ? `Top business-graph risks:\n${topRisks}` : null,
   523→    ctx.twin.healthScore > 0
   524→      ? `Digital Twin health: ${ctx.twin.healthScore}/100 — forecast ${ctx.twin.forecastDirection}.`
   525→      : null,
   526→    ceo?.alerts && ceo.alerts.length > 0
   527→      ? `CEO alerts active: ${ceo.alerts.length}`
   528→      : null,
   529→  ]
   530→    .filter(Boolean)
   531→    .join('\n');
   532→
   533→  const actionItems: string[] = [];
   534→  for (const r of (raw.recentReasoning ?? []).slice(0, 2)) {
   535→    actionItems.push(`Review reasoning "${r.request.slice(0, 60)}…" — act on its recommendation`);
   536→  }
   537→  if (ctx.twin.healthScore > 0 && ctx.twin.healthScore < 60) {
   538→    actionItems.push('Investigate Digital Twin health degradation');
   539→  }
   540→  if (ceo?.alerts && ceo.alerts.length > 0) {
   541→    actionItems.push(`Address ${ceo.alerts.length} open CEO alert(s)`);
   542→  }
   543→  if (actionItems.length < 2) {
   544→    actionItems.push('Schedule weekly Oracle review with executive team');
   545→  }
   546→  while (actionItems.length > 4) actionItems.pop();
   547→
   548→  return {
   549→    title: `Top 3 business insights for ${date}`,
   550→    body,
   551→    confidence: confidenceFromSources(dataSources.length),
   552→    impactScore: clamp(
   553→      Math.max(60, ctx.twin.healthScore || 60, (raw.recentReasoning ?? [])[0]?.confidence ?? 60),
   554→      0,
   555→      100,
   556→    ),
   557→    actionItems: actionItems.slice(0, 4),
   558→    dataSources,
   559→  };
   560→}

   562→function buildRiskSummary(
   563→  date: string,
   564→  raw: RawCollectedData,
   565→  connectivity: ConnectivityData | null,
   566→  autonomous: AutonomousData | null,
   567→  cfo: CFOData | null,
   568→): InsightDraft {
   569→  const dataSources: AIModuleId[] = ['oracle'];
   570→  if (connectivity) dataSources.push('connectivity');
   571→  if (autonomous) dataSources.push('autonomous');
   572→  if (cfo) dataSources.push('cfo');
   573→
   574→  const failingConnectors = [
   575→    ...((raw.failingConnectors ?? []).map((c) => `• ${c.displayName} (${c.provider}) — ${c.status}`)),
   576→    ...(Array.isArray(connectivity?.failingConnectors)
   577→      ? (connectivity?.failingConnectors ?? []).map((c: { displayName?: string; provider?: string; reason?: string }) => `• ${c.displayName ?? 'Unknown'} (${c.provider ?? '?'}) — ${c.reason ?? 'failing'}`)
   578→      : typeof connectivity?.failingConnectors === 'number' && connectivity.failingConnectors > 0
   579→        ? [`• ${connectivity.failingConnectors} failing connector(s) (see Connectivity Fabric™)`]
   580→        : []),
   581→  ].slice(0, 5);
   582→

   852→function buildRevenueForecast(
   853→  date: string,
   854→  raw: RawCollectedData,
   855→  ctx: BusinessContext,
   856→  cfo: CFOData | null,
   857→): InsightDraft {
   858→  const dataSources: AIModuleId[] = ['oracle'];
   859→  if (cfo) dataSources.push('cfo');
   860→  dataSources.push('marketing');
   861→
   862→  // Sum last 3 months of invoices (best-effort from invoiceDate strings).
   863→  const now = new Date(date);
   864→  const threeMonthsAgo = new Date(now.getFullYear(), now.getMonth() - 3, 1);
   865→  const last3MonthsInvoices = (raw.recentInvoices ?? []).filter((inv) => {
   866→    try {
   867→      const d = new Date(inv.invoiceDate);
   868→      return d >= threeMonthsAgo && d <= now;
   869→    } catch {
   870→      return false;
   871→    }
   872→  });
   873→  const last3MonthsRevenue = last3MonthsInvoices.reduce((s, i) => s + i.totalAmount, 0);
   874→  const monthlyAvg = last3MonthsRevenue / 3;
   875→  const ninetyDayForecast = Math.round(monthlyAvg * 3);
   876→
   877→  const cfoForecast = cfo?.predictions?.revenueForecast?.ninetyDay;
   878→  const blendedForecast = cfoForecast
   879→    ? Math.round((ninetyDayForecast + cfoForecast) / 2)
   880→    : ninetyDayForecast;
   881→
   882→  const body = [
   883→    `Date: ${date} (90-day horizon)`,
   884→    '',
   885→    `Last 3 months invoice revenue: ${fmtINR(last3MonthsRevenue)} (${last3MonthsInvoices.length} invoices)`,
   886→    `Monthly average: ${fmtINR(monthlyAvg)}`,
   887→    `Projected next 90 days (trend): ${fmtINR(ninetyDayForecast)}`,
   888→    cfoForecast ? `CFO model 90-day forecast: ${fmtINR(cfoForecast)}` : null,
   889→    `Blended forecast: ${fmtINR(blendedForecast)}`,
   890→    ctx.finance.monthlyRevenue > 0
   891→      ? `Context engine monthly revenue: ${fmtINR(ctx.finance.monthlyRevenue)}`
   892→      : null,
   893→  ]
   894→    .filter(Boolean)
   895→    .join('\n');
   896→
   897→  const actionItems: string[] = [
   898→    `Plan capacity for projected ${fmtINR(monthlyAvg)} monthly revenue`,
   899→    'Accelerate collection on receivables to realise forecasted revenue',
   900→  ];
   901→  if (blendedForecast > last3MonthsRevenue) {
   902→    actionItems.push('Prepare for growth — pre-position inventory and staffing');
   903→  } else if (blendedForecast < last3MonthsRevenue) {
   904→    actionItems.push('Hedge against revenue decline — review pricing and pipeline');
   905→  }
   906→  if (cfoForecast) {
   907→    actionItems.push('Reconcile trend projection against CFO model assumptions');
   908→  }
   909→  while (actionItems.length > 4) actionItems.pop();
   910→
   911→  return {
   912→    title: 'Revenue forecast for next 90 days',
   913→    body,
   914→    confidence: confidenceFromSources(dataSources.length, 90),
   915→    impactScore: impactFromAmount(blendedForecast),
   916→    actionItems: actionItems.slice(0, 4),
   917→    dataSources,
   918→  };
   919→}

   921→function buildCashForecast(
   922→  date: string,
   923→  ctx: BusinessContext,
   924→  cfo: CFOData | null,
   925→): InsightDraft {
   926→  const dataSources: AIModuleId[] = ['oracle'];
   927→  if (cfo) dataSources.push('cfo');
   928→
   929→  const bankBalance =
   930→    cfo?.dashboard?.bankBalance ??
   931→    ctx.finance.cashBalance ??
   932→    cfo?.dashboard?.cash?.currentBalance ??
   933→    0;
   934→  const receivables =
   935→    ctx.finance.receivables || cfo?.dashboard?.receivables?.pendingCollections || 0;
   936→  const payables =
   937→    ctx.finance.payables || cfo?.dashboard?.payables?.vendorDues || 0;
   938→  const monthlyBurn =
   939→    ctx.finance.monthlyExpenses ||
   940→    cfo?.dashboard?.expenses ||
   941→    (cfo?.dashboard?.cash?.burnRatePerDay ? cfo.dashboard.cash.burnRatePerDay * 30 : 0);
   942→  const threeMonthBurn = monthlyBurn * 3;
   943→  const ninetyDayCash = bankBalance + receivables - payables - threeMonthBurn;
   944→  const runwayDays =
   945→    cfo?.dashboard?.cash?.runwayDays ??
   946→    (monthlyBurn > 0 ? Math.floor(bankBalance / (monthlyBurn / 30)) : 0);
   947→
   948→  const cfoCashForecast = cfo?.predictions?.cashForecast?.ninetyDay;
   949→  const blendedCashForecast = cfoCashForecast
   950→    ? Math.round((ninetyDayCash + cfoCashForecast) / 2)
   951→    : ninetyDayCash;
   952→
   953→  const body = [
   954→    `Date: ${date} (90-day horizon)`,
   955→    '',
   956→    `Bank balance: ${fmtINR(bankBalance)}`,
   957→    `Receivables (incoming): ${fmtINR(receivables)}`,
   958→    `Payables (outgoing): ${fmtINR(payables)}`,
   959→    `Projected 3-month burn: ${fmtINR(threeMonthBurn)}`,
   960→    `Projected cash in 90 days: ${fmtINR(ninetyDayCash)}`,
   961→    cfoCashForecast ? `CFO model 90-day cash forecast: ${fmtINR(cfoCashForecast)}` : null,
   962→    `Blended forecast: ${fmtINR(blendedCashForecast)}`,
   963→    runwayDays > 0 ? `Runway: ~${runwayDays} days at current burn` : null,
   964→  ]
   965→    .filter(Boolean)
   966→    .join('\n');
   967→
   968→  const actionItems: string[] = [
   969→    'Prioritise collection of overdue receivables to strengthen cash position',
   970→  ];
   971→  if (ninetyDayCash < 0) {
   972→    actionItems.push('URGENT: 90-day cash deficit projected — arrange credit line or defer discretionary spend');
   973→  } else if (runwayDays > 0 && runwayDays < 60) {
   974→    actionItems.push(`Runway below 60 days (${runwayDays}d) — accelerate collection and freeze hiring`);
   975→  } else {
   976→    actionItems.push('Maintain cash discipline — schedule monthly cash review');
   977→  }
   978→  if (payables > receivables) {
   979→    actionItems.push('Negotiate extended vendor terms — payables exceed receivables');
   980→  }
   981→  if (cfoCashForecast) {
   982→    actionItems.push('Reconcile trend projection against CFO cash model');
   983→  }
   984→  while (actionItems.length > 4) actionItems.pop();
   985→
   986→  return {
   987→    title: 'Cash forecast for next 90 days',
   988→    body,
   989→    confidence: confidenceFromSources(dataSources.length, 90),
   990→    impactScore: impactFromAmount(Math.abs(ninetyDayCash)),
   991→    actionItems: actionItems.slice(0, 4),
   992→    dataSources,
   993→  };
   994→}

   996→function buildExecutiveSummary(
   997→  date: string,
   998→  drafts: Record<InsightCategory, InsightDraft>,
   999→): InsightDraft {

  1001→    new Set<AIModuleId>(

  1004→  );

  1011→  const revenue = drafts.revenue_forecast;
  1012→  const cash = drafts.cash_forecast;

  1017→    risk.actionItems[0] ? `Risk posture: ${risk.actionItems[0]}.` : null,

  1019→    cost.actionItems[0] ? `Cost: ${cost.actionItems[0]}.` : null,
  1020→    compliance.actionItems[0] ? `Compliance: ${compliance.actionItems[0]}.` : null,
  1021→    revenue.actionItems[0] ? `Revenue: ${revenue.actionItems[0]}.` : null,
  1022→    cash.actionItems[0] ? `Cash: ${cash.actionItems[0]}.` : null,
  1023→  ]
  1024→    .filter(Boolean)
  1025→    .join(' ');
  1026→
  1027→  const actionItems: string[] = [];
  1028→  // Pick the single highest-priority action from each insight.
  1029→  for (const cat of [
  1030→    'business_insight',
  1031→    'risk_summary',
  1032→    'compliance_warning',
  1033→    'cash_forecast',
  1034→  ] as InsightCategory[]) {

  1036→    if (a) actionItems.push(a);
  1037→  }
  1038→  while (actionItems.length > 4) actionItems.pop();
  1039→
  1040→  const avgConfidence = Math.round(
  1041→    Object.values(drafts).reduce((s, d) => s + d.confidence, 0) /
  1042→      Object.keys(drafts).length,
  1043→  );

  1046→  return {
  1047→    title: `Executive summary for ${date}`,
  1048→    body: paragraph,
  1049→    confidence: clamp(avgConfidence, 60, 95),
  1050→    impactScore: maxImpact,
  1051→    actionItems: actionItems.slice(0, 4),
  1052→    dataSources: sources.length > 0 ? sources : (['oracle'] as AIModuleId[]),
  1053→  };
  1054→}

  1116→export interface SynthesizeOptions {
  1117→  firmId?: string;
  1118→  date?: Date;

  1121→/** Generate the 8 daily insight records, persist them, and write memory entries. */
  1122→export async function synthesizeInsights(
  1123→  options?: SynthesizeOptions,
  1124→): Promise<InsightRecord[]> {
  1125→  const firmId = options?.firmId || FIRM_ID;
  1126→  const synthesisDate = options?.date ?? new Date();
  1127→  const dateStr = dateKey(synthesisDate);
  1128→
  1129→  // 1. Gather business context (try/catch — empty defaults on failure).
  1130→  let ctx: BusinessContext = emptyContext(firmId);
  1131→  const contextMod = await loadContext();
  1132→  if (contextMod) {
  1133→    try {
  1134→      ctx = await contextMod.gatherBusinessContext({ firmId });
  1135→    } catch (e) {
  1136→      console.warn('[Oracle Insights] gatherBusinessContext failed; using empty context:', e);
  1137→    }
  1138→  }
  1139→
  1140→  // 2. Gather real Prisma data.
  1141→  const raw = await collectRawData();
  1142→
  1143→  // 3. Call sibling AI orchestrators (best-effort, parallel).
  1144→  const [cfo, ceo, autonomous, connectivity] = await Promise.all([
  1145→    loadCFO(),
  1146→    loadCEO(),
  1147→    loadAutonomous(),
  1148→    loadConnectivity(),
  1149→  ]);
  1150→  // Factory data is used for completeness — currently not feeding into a
  1151→  // dedicated insight category but kept available for future expansion.
  1152→  void loadFactory();
  1153→
  1154→  // 4. Build all 8 drafts (each wrapped to surface which builder fails).
  1155→  const drafts: Partial<Record<InsightCategory, InsightDraft>> = {};
  1156→  const builders: Array<[InsightCategory, () => InsightDraft]> = [
  1157→    ['business_insight', () => buildBusinessInsight(dateStr, raw, ctx, cfo, ceo)],
  1158→    ['risk_summary', () => buildRiskSummary(dateStr, raw, connectivity, autonomous, cfo)],
  1159→    ['growth_opportunity', () => buildGrowthOpportunity(dateStr, raw, ctx, cfo)],
  1160→    ['cost_saving', () => buildCostSaving(dateStr, raw, ctx, cfo)],
  1161→    ['compliance_warning', () => buildComplianceWarning(dateStr, raw, cfo)],
  1162→    ['revenue_forecast', () => buildRevenueForecast(dateStr, raw, ctx, cfo)],
  1163→    ['cash_forecast', () => buildCashForecast(dateStr, ctx, cfo)],
  1164→  ];
  1165→  for (const [cat, fn] of builders) {
  1166→    try {
  1167→      drafts[cat] = fn();
  1168→    } catch (e) {
  1169→      console.warn(`[Oracle Insights] builder ${cat} failed:`, e);
  1170→      drafts[cat] = {
  1171→        title: `${cat} for ${dateStr}`,
  1172→        body: `Insight synthesis for ${cat} encountered an error: ${e instanceof Error ? e.message : 'unknown'}. Partial data used.`,
  1173→        confidence: 50,
  1174→        impactScore: 30,
  1175→        actionItems: ['Review data inputs for this insight category'],
  1176→        dataSources: ['oracle'],
  1177→      };
  1178→    }
  1179→  }
  1180→  try {
  1181→    drafts.executive_summary = buildExecutiveSummary(
  1182→      dateStr,
  1183→      drafts as Record<InsightCategory, InsightDraft>,
  1184→    );
  1185→  } catch (e) {
  1186→    console.warn('[Oracle Insights] executive summary builder failed:', e);
  1187→    drafts.executive_summary = {
  1188→      title: `Executive summary for ${dateStr}`,
  1189→      body: `Oracle synthesised briefings across ${Object.keys(drafts).length} categories. Executive summary encountered an error: ${e instanceof Error ? e.message : 'unknown'}.`,
  1190→      confidence: 60,
  1191→      impactScore: 50,
  1192→      actionItems: ['Review individual insight categories'],
  1193→      dataSources: ['oracle'],
  1194→    };
  1195→  }
  1196→
  1197→  // 5. Persist + write memory entries.
  1198→  const memoryMod = await loadMemory();
  1199→  const results: InsightRecord[] = [];
  1200→  for (const category of ALL_CATEGORIES) {
  1201→    const draft = drafts[category];
  1202→    if (!draft) continue;
  1203→    try {
  1204→      const record = await persistInsight(firmId, synthesisDate, category, draft);
  1205→      results.push(record);
  1206→
  1207→      if (memoryMod) {
  1208→        try {
  1209→          await memoryMod.writeMemory({
  1210→            firmId,
  1211→            category: 'insight',
  1212→            source: 'oracle',
  1213→            entityType: 'insight',
  1214→            entityId: record.id,
  1215→            title: draft.title,
  1216→            summary: draft.body.slice(0, 280),
  1217→            payload: {
  1218→              insightId: record.id,
  1219→              category,
  1220→              confidence: draft.confidence,
  1221→              impactScore: draft.impactScore,
  1222→              actionItems: draft.actionItems,
  1223→              dataSources: draft.dataSources,
  1224→            },
  1225→            tags: ['insight', category, `impact:${draft.impactScore}`],
  1226→            importance: Math.max(50, draft.impactScore),
  1227→          });
  1228→        } catch (e) {
  1229→          console.warn(
  1230→            `[Oracle Insights] memory write failed for ${category}:`,
  1231→            e,
  1232→          );
  1233→        }
  1234→      }
  1235→    } catch (e) {
  1236→      console.warn(
  1237→        `[Oracle Insights] persist failed for ${category}:`,
  1238→        e,
  1239→      );
  1240→    }
  1241→  }
  1242→
  1243→  return results;
  1244→}

  1248→export interface InsightStats {
  1249→  total: number;
  1250→  byCategory: Record<string, number>;
  1251→  acknowledged: number;
  1252→  actedOn: number;
  1253→  topImpact: InsightRecord | null;
  1254→}

  1256→/** Aggregate stats for today's insights. */
  1257→export async function getInsightStats(): Promise<InsightStats> {
  1258→  const start = new Date();
  1259→  start.setHours(0, 0, 0, 0);
  1260→  const end = new Date();
  1261→  end.setHours(23, 59, 59, 999);
  1262→
  1263→  try {
  1264→    const where = {
  1265→      synthesisDate: { gte: start, lte: end },
  1266→    };
  1267→    const [total, acknowledged, actedOn, byCategoryRows, topRow] = await Promise.all([
  1268→      db.oracleInsight.count({ where }),
  1269→      db.oracleInsight.count({ where: { ...where, acknowledged: true } }),
  1270→      db.oracleInsight.count({ where: { ...where, actedOn: true } }),
  1271→      db.oracleInsight.groupBy({
  1272→        by: ['category'],
  1273→        where,
  1274→        _count: { _all: true },
  1275→      }),
  1276→      db.oracleInsight.findFirst({
  1277→        where,
  1278→        orderBy: { impactScore: 'desc' },
  1279→      }),
  1280→    ]);
  1281→
  1282→    const byCategory: Record<string, number> = {};
  1283→    for (const r of byCategoryRows) byCategory[r.category] = r._count._all;
  1284→
  1285→    return {
  1286→      total,
  1287→      byCategory,
  1288→      acknowledged,
  1289→      actedOn,
  1290→      topImpact: topRow ? serializeInsight(topRow as never) : null,
  1291→    };
  1292→  } catch (e) {
  1293→    console.warn('[Oracle Insights] getInsightStats failed:', e);
  1294→    return { total: 0, byCategory: {}, acknowledged: 0, actedOn: 0, topImpact: null };
  1295→  }
  1296→}

  1300→/** List insights, optionally filtered by date + category. Defaults to today. */
  1301→export async function listInsights(
  1302→  date?: Date,
  1303→  category?: InsightCategory,
  1304→  limit = 50,
  1305→): Promise<InsightRecord[]> {
  1306→  const targetDate = date ?? new Date();
  1307→  const start = new Date(targetDate);
  1308→  start.setHours(0, 0, 0, 0);
  1309→  const end = new Date(targetDate);
  1310→  end.setHours(23, 59, 59, 999);
  1311→
  1312→  const where: Record<string, unknown> = {
  1313→    synthesisDate: { gte: start, lte: end },
  1314→  };
  1315→  if (category) where.category = category;
  1316→
  1317→  try {
  1318→    const rows = await db.oracleInsight.findMany({
  1319→      where,
  1320→      orderBy: [{ impactScore: 'desc' }, { createdAt: 'desc' }],
  1321→      take: Math.min(limit, 500),
  1322→    });
  1323→    return rows.map((r) => serializeInsight(r as never));
  1324→  } catch (e) {
  1325→    console.warn('[Oracle Insights] listInsights failed:', e);
  1326→    return [];
  1327→  }
  1328→}
