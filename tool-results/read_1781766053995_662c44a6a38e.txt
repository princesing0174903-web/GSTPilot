    65→
    66→export async function POST(req: NextRequest) {
    67→  let question = ''
    68→  let context = ''
    69→  try {
    70→    const body = await req.json()
    71→    question = (body.question || '').trim()
    72→    context = (body.context || '').trim()
    73→    const conversationHistory: Array<{ role: string; content: string }> =
    74→      body.conversationHistory || []
    75→
    76→    if (!question) {
    77→      return NextResponse.json({ error: 'Question is required' }, { status: 400 })
    78→    }
    79→
    80→    // 1) Deterministic, fast intent + actions + thinking steps
    81→    const intent = detectIntent(question)
    82→    const thinkingSteps = generateThinkingSteps(intent)
    83→    const detectedActions = detectActions(question, intent)
    84→
    85→    // 2) Determine whether real data is available in context.
    86→    //    The frontend builds context as a structured string starting with
    87→    //    "LIVE DATA SNAPSHOT" and a list of "- key: value" lines.
    88→    const hasLiveData = contextHasLiveData(context)
    89→
    90→    // 3) Build sources from ACTUAL context content (no fabrication)
    91→    const sources = hasLiveData ? buildSources(context) : EMPTY_CONTEXT_SOURCES
    92→
    93→    // 4) Generate the answer via LLM (or fallback)
    94→    let answer: string
    95→    if (!hasLiveData) {
    96→      // Skip LLM call entirely when no live data — guarantees no fabrication
    97→      answer = EMPTY_CONTEXT_ANSWER
    98→    } else {
    99→      try {
   100→        answer = await generateLLMAnswer(question, context, conversationHistory)
   101→      } catch (llmError) {
   102→        console.error(
   103→          '[Oracle] LLM error:',
   104→          llmError instanceof Error ? llmError.message : llmError,
   105→        )
   106→        answer = generateContextualFallback(question, context, intent)
   107→      }
   108→      if (!answer || !answer.trim()) {
   109→        answer = generateContextualFallback(question, context, intent)
   110→      }
   111→    }
   112→
   113→    // 5) Extract structured insights from the LLM answer (or derive from context)
   114→    const insights = hasLiveData
   115→      ? extractInsights(answer, context, intent)
   116→      : [
   117→          {
   118→            text: 'Connect your business data to unlock AI insights.',
   119→            tone: 'neutral' as InsightTone,
   120→          },
   121→        ]
   122→
   123→    // 6) Suggested follow-ups based on intent (and data availability)
   124→    const suggestedPrompts = hasLiveData
   125→      ? suggestFollowUps(intent, question)
   126→      : [
   127→          'Show pending returns',
   128→          'Which clients are risky?',
   129→          'What revenue will I make next month?',
   130→        ]
   131→
   132→    const responseBody: OracleResponse = {
   133→      answer,
   134→      insights,
   135→      sources,
   136→      actions: detectedActions,
   137→      suggestedPrompts,
   138→      intent,
   139→      thinkingSteps,
   140→    }
   141→
   142→    return NextResponse.json(responseBody)
   143→  } catch (error: unknown) {
   144→    console.error(
   145→      '[Oracle] Fatal:',
   146→      error instanceof Error ? error.message : error,
   147→    )
   148→    // Always return the structured shape — frontend never has to guard
   149→    const fallbackResponse: OracleResponse = {
   150→      answer: EMPTY_CONTEXT_ANSWER,
   151→      insights: [
   152→        {
   153→          text: 'I had trouble reaching my reasoning engine. Please try again.',
   154→          tone: 'warning',
   155→        },
   156→      ],
   157→      sources: EMPTY_CONTEXT_SOURCES,
   158→      actions: [],
   159→      suggestedPrompts: [
   160→        'Show pending returns',
   161→        'Which clients are risky?',
   162→        'What revenue will I make next month?',
   163→      ],
   164→      intent: 'fallback',
   165→      thinkingSteps: [
   166→        { label: 'Thinking...', duration: 400 },
   167→        { label: 'Generating answer...', duration: 500 },
   168→      ],
   169→    }
   170→    return NextResponse.json(fallbackResponse)
   171→  }
   172→}

   176→async function generateLLMAnswer(
   177→  question: string,
   178→  context: string,
   179→  conversationHistory: Array<{ role: string; content: string }>,
   180→): Promise<string> {
   181→  const ZAI = (await import('z-ai-web-dev-sdk')).default
   182→  const zai = await ZAI.create()
   183→
   184→  const systemPrompt = `You are GSTPilot Intelligence™ — the AI Brain of the user's Indian CA firm / business.
   185→
   186→You have LIVE access to the firm's Firestore data (clients, invoices, returns, payments, notifications, activities). The current data snapshot is provided below as "Live Business Context".
   187→
   188→## Your job
   189→1. Answer the user's question accurately using ONLY the live data below.
   190→2. Reference SPECIFIC NUMBERS, CLIENT NAMES, AMOUNTS, and DATES from the context.
   191→3. Use Indian number formatting (₹1,23,456 — lakhs/crores, no millions) and DD/MM/YYYY dates.
   192→4. Structure your answer as:
   193→   • A concise 2-3 sentence direct answer FIRST (no header, no preamble).
   194→   • Then 2-4 bullet points (use "• " prefix) with the most important insights,
   195→     numbers, or breakdowns supporting your answer.
   196→   • Then an optional "Recommended Actions:" line if there is a clear next step.
   197→5. Match the user's tone — professional but warm, like a senior CA partner.
   198→6. If the user asks about running automation ("run my firm", "run my business"),
   199→   describe what would happen step-by-step in 5-7 numbered steps.
   200→7. NEVER invent data. If the context does not contain what you need, say so
   201→   explicitly: "I don't have enough live data right now" and suggest what to
   202→   connect or check.
   203→8. Do NOT use markdown headers (#), do NOT use emojis, do NOT use tables.
   204→9. Keep total response under 200 words.
   205→
   206→Live Business Context:
   207→${context}
   208→
   209→Conversation history (most recent last):
   210→${conversationHistory
   211→  .slice(-4)
   212→  .map((m) => `${m.role}: ${m.content}`)
   213→  .join('\n') || '[No prior conversation]'}`
   214→
   215→  const response = await zai.chat.completions.create({
   216→    messages: [
   217→      { role: 'system', content: systemPrompt },
   218→      { role: 'user', content: question },
   219→    ],
   220→  })
   221→
   222→  return response.choices[0]?.message?.content || ''
   223→}

   227→function detectIntent(question: string): string {
   228→  const q = question.toLowerCase()
   229→  if (
   230→    /\b(run|execute|start|launch|automate)\b.*\b(firm|business|company|pipeline|workflow)\b/.test(
   231→      q,
   232→    ) ||
   233→    q.includes('run my firm') ||
   234→    q.includes('run my business') ||
   235→    q.includes('run my company')
   236→  )
   237→    return 'run_automation'
   238→  if (
   239→    q.includes('overload') ||
   240→    q.includes('overloaded') ||
   241→    q.includes('workload') ||
   242→    q.includes('capacity') ||
   243→    q.includes('bandwidth')
   244→  )
   245→    return 'workload'
   246→  if (
   247→    q.includes('revenue') ||
   248→    q.includes('income') ||
   249→    q.includes('earn') ||
   250→    q.includes('next month') ||
   251→    q.includes('forecast') ||
   252→    q.includes('predict')
   253→  )
   254→    return 'revenue_forecast'
   255→  if (
   256→    q.includes('collection') ||
   257→    q.includes('drop') ||
   258→    q.includes('decline') ||
   259→    q.includes('fell') ||
   260→    q.includes('why did')
   261→  )
   262→    return 'collection_analysis'
   263→  if (q.includes('risk') || q.includes('risky') || q.includes('health') || q.includes('at risk'))
   264→    return 'risk_analysis'
   265→  if (
   266→    q.includes('pending') ||
   267→    q.includes('overdue') ||
   268→    q.includes('filing') ||
   269→    q.includes('gstr') ||
   270→    q.includes('return')
   271→  )
   272→    return 'pending_returns'
   273→  if (
   274→    q.includes('invoice') ||
   275→    q.includes('payment') ||
   276→    q.includes('outstanding') ||
   277→    q.includes('collect')
   278→  )
   279→    return 'invoice_collection'
   280→  if (q.includes('cash') || q.includes('flow') || q.includes('liquidity')) return 'cash_flow'
   281→  if (q.includes('compliance') || q.includes('score') || q.includes('deadline')) return 'compliance'
   282→  if (q.includes('client') || q.includes('customer')) return 'client_overview'
   283→  if (q.includes('today') || q.includes('priority') || q.includes('should i') || q.includes('to do'))
   284→    return 'daily_priority'
   285→  if (q.includes('compare') || q.includes('last month') || q.includes('vs') || q.includes('trend'))
   286→    return 'comparison'
   287→  if (q.includes('task') || q.includes('reminder') || q.includes('notify')) return 'task_management'
   288→  if (q.includes('report') || q.includes('summary') || q.includes('statement')) return 'report'
   289→  return 'general'
   290→}

   522→// ─── Sources Builder (parses LIVE context string for real counts) ─────────────
   523→
   524→/**
   525→ * The frontend composes a context string starting with "LIVE DATA SNAPSHOT"
   526→ * followed by "- key: value" lines. We parse those lines to surface the
   527→ * real sources + record counts the LLM consulted. NO fabrication — if a
   528→ * section is missing or zero, the source is omitted.
   529→ */
   530→function buildSources(context: string): OracleSource[] {
   531→  const sources: OracleSource[] = []
   532→
   533→  const clients = parseCount(context, /-?\s*Total clients:\s*(\d+)/i)
   534→  if (clients > 0) {
   535→    sources.push({ name: 'Clients', count: clients, icon: 'users' })
   536→  }
   537→
   538→  const invoices = parseCount(context, /-?\s*Active invoices:\s*(\d+)/i)
   539→  if (invoices > 0) {
   540→    sources.push({ name: 'Invoices', count: invoices, icon: 'file-text' })
   541→  }
   542→
   543→  const filed = parseCount(context, /-?\s*Filed returns:\s*(\d+)/i)
   544→  // Frontend emits "- Pending returns: 3 GSTR-1, 2 GSTR-3B"
   545→  // The count comes BEFORE the return-type label, so capture the leading digit.
   546→  const gstr1Pending = parseCount(context, /-?\s*Pending returns:\s*(\d+)\s*GSTR-1/i)
   547→  const gstr3bPending = parseCount(context, /GSTR-1,?\s*(\d+)\s*GSTR-3B/i)
   548→  const totalReturns = filed + gstr1Pending + gstr3bPending
   549→  if (totalReturns > 0) {
   550→    sources.push({ name: 'GST Returns', count: totalReturns, icon: 'receipt' })
   551→  }
   552→
   553→  const unread = parseCount(context, /-?\s*Unread notifications:\s*(\d+)/i)
   554→  if (unread > 0) {
   555→    sources.push({ name: 'Notifications', count: unread, icon: 'bell' })
   556→  }
   557→
   558→  const activities = parseCount(context, /-?\s*Recent activities:\s*(\d+)/i)
   559→  if (activities > 0) {
   560→    sources.push({ name: 'Activities', count: activities, icon: 'activity' })
   561→  }
   562→
   563→  // If we somehow couldn't parse anything but the context clearly has live data,
   564→  // return a single generic source so the UI has something to show.
   565→  if (sources.length === 0 && contextHasLiveData(context)) {
   566→    sources.push({ name: 'Business Data', count: 0, icon: 'database' })
   567→  }
   568→
   569→  return sources
   570→}
   571→
   572→function parseCount(context: string, pattern: RegExp): number {
   573→  const match = context.match(pattern)
   574→  if (!match || !match[1]) return 0
   575→  const n = parseInt(match[1], 10)
   576→  return Number.isFinite(n) ? n : 0
   577→}
   578→
   579→function contextHasLiveData(context: string): boolean {
   580→  if (!context || context.length < 60) return false
   581→  // The frontend always emits this header when live data is available
   582→  if (!/LIVE DATA SNAPSHOT/i.test(context)) return false
   583→  // If the snapshot explicitly says data unavailable, treat as empty
   584→  if (/Data currently unavailable/i.test(context)) return false
   585→  // Must have at least one "- key: value" data line beyond the header
   586→  const dataLines = context
   587→    .split('\n')
   588→    .filter((l) => /^\s*-\s+\w/.test(l))
   589→  return dataLines.length >= 2
   590→}

   592→// ─── Insight Extractor (parses LLM answer for 2-4 bullets) ────────────────────
   593→
   594→/**
   595→ * Extracts 2-4 insight bullets from the LLM answer. Falls back to
   596→ * context-derived insights when the answer has no parseable bullets.
   597→ */
   598→function extractInsights(
   599→  answer: string,
   600→  context: string,
   601→  intent: string,
   602→): OracleInsight[] {
   603→  // Parse bullet/numbered lines from the answer body
   604→  const rawLines = answer
   605→    .split('\n')
   606→    .map((l) => l.trim())
   607→    .filter(Boolean)
   608→
   609→  const bullets: string[] = []
   610→  for (const line of rawLines) {
   611→    let text: string | null = null
   612→    if (line.startsWith('• ') || line.startsWith('•')) {
   613→      text = line.replace(/^•\s*/, '').trim()
   614→    } else if (line.startsWith('- ')) {
   615→      text = line.replace(/^-\s*/, '').trim()
   616→    } else if (/^\d+\.\s/.test(line)) {
   617→      text = line.replace(/^\d+\.\s*/, '').trim()
   618→    }
   619→    if (text && text.length >= 12 && text.length <= 240) {
   620→      // Strip markdown bold/italic markers for cleaner insight text
   621→      const cleaned = text.replace(/\*\*/g, '').replace(/__/g, '').trim()
   622→      if (cleaned.length >= 12) bullets.push(cleaned)
   623→    }
   624→    if (bullets.length >= 4) break
   625→  }
   626→
   627→  if (bullets.length >= 2) {
   628→    return bullets.slice(0, 4).map((text) => ({ text, tone: detectTone(text) }))
   629→  }
   630→
   631→  // Fallback: derive insights from real context numbers
   632→  const derived = deriveInsightsFromContext(context, intent)
   633→  if (derived.length >= 1) {
   634→    // If we got 1 bullet from the answer, combine with 1 derived for 2 total
   635→    const combined: OracleInsight[] = []
   636→    if (bullets.length === 1) {
   637→      combined.push({ text: bullets[0], tone: detectTone(bullets[0]) })
   638→    }
   639→    for (const d of derived) {
   640→      if (combined.length >= 2) break
   641→      if (!combined.some((c) => c.text === d.text)) combined.push(d)
   642→    }
   643→    if (combined.length >= 1) return combined.slice(0, 4)
   644→  }
   645→
   646→  // Last-resort fallback: synthesize a single neutral insight from the answer
   647→  const firstSentence = answer
   648→    .split(/(?<=[.!?])\s+/)[0]
   649→    ?.replace(/[*#]/g, '')
   650→    .trim()
   651→  if (firstSentence && firstSentence.length >= 12) {
   652→    return [{ text: firstSentence.slice(0, 240), tone: 'neutral' }]
   653→  }
   654→  return [{ text: 'Insights will appear here once live data is connected.', tone: 'neutral' }]
   655→}
   656→
   657→function detectTone(text: string): InsightTone {
   658→  const t = text.toLowerCase()
   659→  // Warning signals — risk/decline/overdue keywords
   660→  if (
   661→    /\b(overdue|drop|decline|declined|fell|missed|urgent|risk|risky|late|below|fail|failed|exposed|shortfall|delay|delayed|warning|alert|critical|breach|non-compliant|mismatch|gap)\b/.test(
   662→      t,
   663→    )
   664→  ) {
   665→    return 'warning'
   666→  }
   667→  // Positive signals — growth/improvement keywords
   668→  if (
   669→    /\b(growth|grew|increase|increased|up|healthy|profit|saved|completed|filed|recovered|improved|improvement|surplus|above|strong|stable|on track|on-time)\b/.test(
   670→      t,
   671→    )
   672→  ) {
   673→    return 'positive'
   674→  }
   675→  return 'neutral'
   676→}

   678→function deriveInsightsFromContext(context: string, intent: string): OracleInsight[] {
   679→  const insights: OracleInsight[] = []
   680→
   681→  const clients = parseCount(context, /-?\s*Total clients:\s*(\d+)/i)
   682→  const activeClients = parseCount(context, /-?\s*Active clients:\s*(\d+)/i)
   683→  const invoices = parseCount(context, /-?\s*Active invoices:\s*(\d+)/i)
   684→  const overdueInvoices = parseCount(context, /-?\s*Overdue invoices:\s*(\d+)/i)
   685→  const filed = parseCount(context, /-?\s*Filed returns:\s*(\d+)/i)
   686→  const gstr1Pending = parseCount(context, /-?\s*Pending returns:\s*(\d+)\s*GSTR-1/i)
   687→
   688→  // High-risk clients line (text after the colon)
   689→  const riskMatch = context.match(/-?\s*High-risk clients:\s*(.+)/i)
   690→  const riskLine = riskMatch?.[1]?.trim()
   691→  const highRiskCount = riskLine
   692→    ? (riskLine.match(/,/g)?.length ?? 0) + (riskLine.length > 0 ? 1 : 0)
   693→    : 0
   694→
   695→  // Top client line
   696→  const topMatch = context.match(/-?\s*Top 5 clients by revenue:\s*(.+)/i)
   697→  const topLine = topMatch?.[1]?.trim()
   698→
   699→  if (intent === 'risk_analysis' && highRiskCount > 0 && riskLine) {
   700→    insights.push({
   701→      text: `${highRiskCount} client${highRiskCount > 1 ? 's' : ''} flagged high-risk — ${riskLine.slice(0, 140)}.`,
   702→      tone: 'warning',
   703→    })
   704→  }
   705→
   706→  if (intent === 'pending_returns' && (gstr1Pending > 0 || filed > 0)) {
   707→    insights.push({
   708→      text: `${gstr1Pending} GSTR-1 ${gstr1Pending === 1 ? 'return' : 'returns'} pending, ${filed} already filed.`,
   709→      tone: gstr1Pending > 0 ? 'warning' : 'positive',
   710→    })
   711→  }
   712→
   713→  if (
   714→    (intent === 'invoice_collection' || intent === 'collection_analysis') &&
   715→    (overdueInvoices > 0 || invoices > 0)
   716→  ) {
   717→    insights.push({
   718→      text: `${overdueInvoices} overdue invoice${overdueInvoices === 1 ? '' : 's'} out of ${invoices} active.`,
   719→      tone: overdueInvoices > 0 ? 'warning' : 'positive',
   720→    })
   721→  }
   722→
   723→  if (intent === 'client_overview' && clients > 0) {
   724→    insights.push({
   725→      text: `${clients} total client${clients === 1 ? '' : 's'} (${activeClients} active).`,
   726→      tone: 'neutral',
   727→    })
   728→  }
   729→
   730→  if (intent === 'revenue_forecast' && topLine) {
   731→    insights.push({
   732→      text: `Top revenue clients: ${topLine.slice(0, 140)}.`,
   733→      tone: 'positive',
   734→    })
   735→  }
   736→
   737→  if (intent === 'compliance' && filed > 0) {
   738→    insights.push({
   739→      text: `${filed} return${filed === 1 ? '' : 's'} filed so far this period.`,
   740→      tone: 'positive',
   741→    })
   742→  }
   743→
   744→  return insights
   745→}
