    52→function inrShort(n: number): string {
    53→  if (!isFinite(n) || isNaN(n)) return '₹0';
    54→  const abs = Math.abs(n);
    55→  if (abs >= 10000000) return `₹${(n / 10000000).toFixed(2)} Cr`;
    56→  if (abs >= 100000) return `₹${(n / 100000).toFixed(2)} L`;
    57→  if (abs >= 1000) return `₹${(n / 1000).toFixed(1)}K`;
    58→  return '₹' + Math.round(n).toLocaleString('en-IN');
    59→}

   700→    'Cache-Control': 'no-cache, no-transform',
   701→    'Connection': 'keep-alive',
   702→    'X-Accel-Buffering': 'no',
   703→  };
   704→}
   705→
   706→/** Map our role names to the model's role names. */
   707→function toModelMessages(
   708→  messages: OracleChatRequest['messages'],
   709→): { role: 'assistant' | 'user'; content: string }[] {
   710→  return messages.map((m) => ({
   711→    role: (m.role === 'oracle' ? 'assistant' : 'user') as 'assistant' | 'user',
   712→    content: m.content,
   713→  }));
   714→}
   715→
   716→/** Server-side language hint derived from the latest user message. */
   717→function inferLanguageHint(messages: OracleChatRequest['messages']): OracleLanguageId | undefined {
   718→  for (let i = messages.length - 1; i >= 0; i--) {
   719→    if (messages[i].role === 'user' && messages[i].content?.trim()) {
   720→      const s = messages[i].content.slice(0, 500);
   721→      if (/[\u0900-\u097F]/.test(s)) return 'hindi';
   722→      if (/[\u0600-\u06FF]/.test(s)) return 'urdu';
   723→      if (/[\u0A00-\u0A7F]/.test(s)) return 'punjabi';
   724→      if (/[\u0A80-\u0AFF]/.test(s)) return 'gujarati';
   725→      if (/[\u0B80-\u0BFF]/.test(s)) return 'tamil';
   726→      if (/[\u0C00-\u0C7F]/.test(s)) return 'telugu';
   727→      if (/[\u0980-\u09FF]/.test(s)) return 'bengali';
   728→      return 'english';
   729→    }
   730→  }
   731→  return undefined;
   732→}

   734→// ─── POST handler ─────────────────────────────────────────────────────────────
   735→
   736→export async function POST(request: Request) {
   737→  let body: OracleChatRequest;
   738→  try {
   739→    body = (await request.json()) as OracleChatRequest;
   740→  } catch {
   741→    return new Response(JSON.stringify({ error: 'Invalid JSON body' }), {
   742→      status: 400,
   743→      headers: { 'Content-Type': 'application/json' },
   744→    });
   745→  }
   746→
   747→  const messages = body.messages ?? [];
   748→  if (!Array.isArray(messages) || messages.length === 0) {
   749→    return new Response(JSON.stringify({ error: 'messages[] is required' }), {
   750→      status: 400,
   751→      headers: { 'Content-Type': 'application/json' },
   752→    });
   753→  }
   754→
   755→  const systemPrompt = await buildSystemPrompt(body);
   756→  const languageHint = inferLanguageHint(messages);
   757→
   758→  const modelMessages: { role: 'assistant' | 'user' | 'system'; content: string }[] = [
   759→    { role: 'assistant', content: systemPrompt },
   760→    ...toModelMessages(messages),
   761→  ];
   762→
   763→  // ── Acquire the upstream stream from the SDK ───────────────────────────────
   764→  let upstream: ReadableStream<Uint8Array> | null = null;
   765→  try {
   766→    const zai = await ZAI.create();
   767→    const result = await zai.chat.completions.create({
   768→      messages: modelMessages,
   769→      stream: true,
   770→      thinking: { type: 'disabled' },
   771→    });
   772→    if (result && typeof (result as ReadableStream<Uint8Array>).getReader === 'function') {
   773→      upstream = result as ReadableStream<Uint8Array>;
   774→    } else {
   775→      // Non-streaming fallback: emit the full text as one chunk then close.
   776→      const text =
   777→        (result as { choices?: { message?: { content?: string } }[] })?.choices?.[0]?.message
   778→          ?.content ?? '';
   779→      upstream = new ReadableStream<Uint8Array>({
   780→        start(controller) {
   781→          if (text) controller.enqueue(sseChunk({ token: text }));
   782→          controller.close();
   783→        },
   784→      });
   785→    }
   786→  } catch (err) {
   787→    const message = err instanceof Error ? err.message : 'Unknown error';
   788→    const stream = new ReadableStream<Uint8Array>({
   789→      start(controller) {
   790→        controller.enqueue(
   791→          sseChunk({
   792→            token: `I'm here, but I hit a temporary issue reaching my reasoning service (${message}). Please try again in a moment — your conversation is safe.`,
   793→          }),
   794→        );
   795→        controller.enqueue(sseChunk({ done: true }));
   796→        controller.close();
   797→      },
   798→    });
   799→    return new Response(stream, { status: 200, headers: sseHeaders() });
   800→  }
   801→
   802→  // ── Transform the upstream SSE stream into our token stream ────────────────
   803→  const decoder = new TextDecoder();
   804→  const transformed = new ReadableStream<Uint8Array>({
   805→    async start(controller) {
   806→      // Emit a tiny first nudge so the UI shows the pulsing cursor within the
   807→      // first frame — real tokens follow immediately.
   808→      if (languageHint) controller.enqueue(sseChunk({ language: languageHint }));
   809→
   810→      const reader = upstream!.getReader();
   811→      let buffer = '';
   812→      let emittedAny = false;
   813→      try {
   814→        while (true) {
   815→          const { done, value } = await reader.read();
   816→          if (done) break;
   817→          buffer += decoder.decode(value, { stream: true });
   818→
   819→          const lines = buffer.split('\n');
   820→          buffer = lines.pop() ?? '';
   821→
   822→          for (const rawLine of lines) {
   823→            const line = rawLine.trim();
   824→            if (!line || !line.startsWith('data:')) continue;
   825→            const data = line.slice(5).trim();
   826→            if (data === '[DONE]' || !data) continue;
   827→            try {
   828→              const json = JSON.parse(data);
   829→              const token: string =
   830→                json?.choices?.[0]?.delta?.content ??
   831→                json?.choices?.[0]?.message?.content ??
   832→                '';
   833→              if (token) {
   834→                emittedAny = true;
   835→                controller.enqueue(sseChunk({ token }));
   836→              }
   837→            } catch {
   838→              // Partial JSON across a chunk boundary — resolves on next read.
   839→            }
   840→          }
   841→        }
   842→        // Flush any trailing buffered line.
   843→        const tail = buffer.trim();
   844→        if (tail.startsWith('data:')) {
   845→          const data = tail.slice(5).trim();
   846→          if (data && data !== '[DONE]') {
   847→            try {
   848→              const json = JSON.parse(data);
   849→              const token: string =
   850→                json?.choices?.[0]?.delta?.content ??
   851→                json?.choices?.[0]?.message?.content ??
   852→                '';
   853→              if (token) {
   854→                emittedAny = true;
   855→                controller.enqueue(sseChunk({ token }));
   856→              }
   857→            } catch {
   858→              /* ignore */
   859→            }
   860→          }
   861→        }
   862→        if (!emittedAny) {
   863→          controller.enqueue(
   864→            sseChunk({
   865→              token:
   866→                "I'm here. Based on current GST rules and the information available, I'd be glad to help — could you share a bit more about what you're looking to do?",
   867→            }),
   868→          );
   869→        }
   870→        controller.enqueue(sseChunk({ done: true }));
   871→        controller.close();
   872→      } catch (err) {
   873→        const message = err instanceof Error ? err.message : 'Stream interrupted';
   874→        if (!emittedAny) {
   875→          controller.enqueue(
   876→            sseChunk({
   877→              token: `My response was interrupted (${message}). Please try sending that again.`,
   878→            }),
   879→          );
   880→        }
   881→        controller.enqueue(sseChunk({ done: true }));
   882→        controller.close();
   883→      }
   884→    },
   885→    cancel() {
   886→      upstream?.cancel?.().catch(() => undefined);
   887→    },
   888→  });
   889→
   890→  return new Response(transformed, { status: 200, headers: sseHeaders() });
   891→}
