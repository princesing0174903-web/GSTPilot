     1→// ═══════════════════════════════════════════════════════════════════════════════
     2→// GSTPilot Billing, Subscriptions & Payments™ — Type Definitions
     3→//
     4→// The single source of truth for the billing / subscription / payment data model.
     5→// Every field maps 1:1 to a Firestore collection. All types are PURE (no Firebase
     6→// imports) so they are safe to import from both client and server code.
     7→//
     8→// Provider pattern:
     9→//   • IPaymentProvider (see provider.ts) — the contract every payment backend implements
    10→//   • MockRazorpayProvider  — deterministic simulated UPI/card/netbanking (default, India)
    11→//   • MockStripeProvider    — deterministic simulated card payments (default, international)
    12→//   • FutureRazorpayProvider — Razorpay production (placeholder, throws NotImplementedError)
    13→//   • FutureStripeProvider   — Stripe production (placeholder, throws NotImplementedError)
    14→//   • Switch to production later by changing ONE env var in registry.ts
    15→//
    16→// Multi-tenant: every document carries `organizationId` (except the two global
    17→// collections `subscription_plans` and `coupons` which are readable by every
    18→// authenticated user). Every org-scoped query filters on `organizationId`.
    19→//
    20→// Security: payment session IDs / customer IDs / provider tokens / receipt data
    21→// are encrypted with AES-256-GCM (server-only key) before being stored in
    22→// Firestore. The client reads the encrypted blob and passes it back to the
    23→// server during payment operations — the client can NEVER decrypt it.
    24→// ═══════════════════════════════════════════════════════════════════════════════
    25→

    27→
    28→/**
    29→ * The set of supported payment provider backends. Today the two 'mock' variants
    30→ * are live; the 'future' variants throw NotImplementedError until implemented.

    32→export type PaymentProviderName =
    33→  | 'mock-razorpay'   // MockRazorpayProvider — UPI / card / netbanking (India-first)

    35→  | 'razorpay'        // FutureRazorpayProvider — production Razorpay
    36→  | 'stripe';         // FutureStripeProvider   — production Stripe

    40→/**
    41→ * A subscription tier — one row per plan. Stored in Firestore
    42→ * `subscription_plans/{planId}` (e.g. `free`, `starter`, `professional`,
    43→ * `business`, `enterprise`).
    44→ *
    45→ * GLOBAL collection: every authenticated user can read; only server-side
    46→ * (Admin SDK / seed script) can write.
    47→ */
    48→export interface SubscriptionPlan {
    49→  /** Doc id = plan slug (e.g. 'free', 'starter'). */
    50→  id: string;
    51→  /** Display name (e.g. 'Free', 'Starter', 'Professional'). */
    52→  name: string;
    53→  /** One-line marketing tagline. */
    54→  tagline: string;
    55→  /** Monthly price in paise (₹0 = free). Use `monthlyPrice / 100` for display. */
    56→  monthlyPrice: number;
    57→  /** Yearly price in paise (≈ 10 × monthly — 2 months free). */
    58→  yearlyPrice: number;
    59→  /** Currency code (ISO 4217). Always 'INR' for now. */
    60→  currency: string;
    61→  /** Human-readable feature list (for the pricing card). */
    62→  features: string[];
    63→  /** Hard usage limits — enforced by the Usage Meter. */
    64→  limits: SubscriptionPlanLimits;
    65→  /** Highlighted as "Most Popular" in the pricing UI (Professional). */
    66→  isPopular: boolean;
    67→  /** Trial period in days (14 for paid plans, 0 for Free). */
    68→  trialDays: number;
    69→  /** Sort order on the pricing page (0 = first). */
    70→  sortOrder: number;
    71→  /** ISO timestamp the plan was created / last updated. */
    72→  createdAt: string;
    73→  updatedAt: string;
    74→}

    92→  apiCallsPerDay: number;

   134→  currentPeriodEnd: string;

   136→  currentPeriodStart: string;

   240→/**
   241→ * The billing invoice document — one per billing period per subscription.
   242→ * Stored in Firestore `billing_invoices/{invoiceId}`.
   243→ *
   244→ * NOTE: This is a BILLING invoice (subscription payment), distinct from the
   245→ * user's sales invoices in the `invoices` collection. We use the prefix
   246→ * `billing_invoices` to avoid name collision.
   247→ */
   248→export interface BillingInvoice {
   249→  id: string;
   250→  /** Tenant scope — NEVER null. */
   251→  organizationId: string;
   252→  /** Document type — see InvoiceDocumentType. */
   253→  type: InvoiceDocumentType;
   254→  /** Invoice number (e.g. `GSTP-INV-2024-0001`, `GSTP-RCPT-2024-0001`). */
   255→  number: string;
   256→  /** ISO timestamp the invoice was issued. */
   257→  issueDate: string;
   258→  /** ISO timestamp the invoice is due (null for receipts). */
   259→  dueDate: string | null;
   260→  /** Subscription this invoice belongs to (null for one-off charges). */
   261→  subscriptionId: string | null;
   262→  /** Original invoice this document references (for credit/debit notes). */
   263→  originalInvoiceId: string | null;
   264→  /** Bill-from details (GSTPilot). */
   265→  billFrom: BillingParty;
   266→  /** Bill-to details (the customer org). */
   267→  billTo: BillingParty;
   268→  /** Line items. */
   269→  lineItems: BillingLineItem[];
   270→  /** Subtotal in paise (Σ lineItems.amount). */
   271→  subtotal: number;
   272→  /** Discount in paise (from applied coupon, if any). */
   273→  discount: number;
   274→  /** Applied coupon id (if any). */
   275→  couponId: string | null;
   276→  /** Applied coupon code (for display). */
   277→  couponCode: string | null;
   278→  /** Tax rate percentage (e.g. 18 for 18% GST). */
   279→  taxRate: number;
   280→  /** Tax amount in paise (GST). */
   281→  taxAmount: number;
   282→  /** Total in paise (subtotal − discount + taxAmount). */
   283→  total: number;
   284→  /** Amount paid in paise. */
   285→  paidAmount: number;
   286→  /** Balance due in paise (total − paidAmount). */
   287→  balanceDue: number;
   288→  /** Currency code (ISO 4217). */
   289→  currency: string;
   290→  /** Current status. */
   291→  status: InvoiceDocumentStatus;
   292→  /** Notes shown on the invoice. */
   293→  notes: string | null;
   294→  /** Terms & conditions. */
   295→  termsAndConditions: string | null;
   296→  /** URL to the PDF (null until generated). */
   297→  pdfUrl: string | null;
   298→  /** Billing period this invoice covers (YYYY-MM). */
   299→  billingPeriod: string | null;
   300→  /** Arbitrary metadata. */
   301→  metadata: Record<string, unknown>;
   302→  createdAt: string;
   303→  updatedAt: string;
   304→}

   411→  /** Amount attempted in paise. */
   412→  amount: number;

   414→  currency: string;

   416→  status: PaymentAttemptStatus;
   417→  /** Payment method attempted. */
   418→  method: PaymentMethod | null;
   419→  /** ISO timestamp the attempt was initiated. */
   420→  initiatedAt: string;

   423→  /** Failure reason (if status='failed'). */
   424→  failureReason: string | null;
   425→  /** Provider used. */
   426→  provider: PaymentProviderName;
   427→  /** User agent that initiated the attempt (for fraud detection). */
   428→  userAgent: string | null;
   429→  /** IP address that initiated the attempt (for fraud detection). */

   431→  createdAt: string;
