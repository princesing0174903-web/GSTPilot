    32→function QuarterlyBarChart({ data }: { data: { quarter: string; deducted: number; deposited: number }[] }) {
    33→  const max = Math.max(...data.map(d => Math.max(d.deducted, d.deposited)))
    34→  const barW = 32
    35→  const gap = 20
    36→  const h = 100
    37→  const totalW = data.length * (barW * 2 + gap + 8)
    38→  return (
    39→    <svg width={totalW} height={h + 24} className="overflow-visible">
    40→      {data.map((d, i) => {
    41→        const x = i * (barW * 2 + gap + 8)
    42→        const hD = (d.deducted / max) * (h - 10)
    43→        const hP = (d.deposited / max) * (h - 10)
    44→        return (
    45→          <g key={d.quarter}>
    46→            <rect x={x} y={h - hD} width={barW} height={hD} rx={4} fill="#10b981" opacity={0.7} />
    47→            <rect x={x + barW + 4} y={h - hP} width={barW} height={hP} rx={4} fill="#64748b" opacity={0.5} />
    48→            <text x={x + barW + 2} y={h + 16} textAnchor="middle" className="text-[10px] fill-muted-foreground">{d.quarter}</text>
    49→          </g>
    50→        )
    51→      })}
    52→      <g transform={`translate(${totalW - 120}, 0)`}>
    53→        <rect x="0" y="0" width="10" height="10" rx="2" fill="#10b981" opacity={0.7} />
    54→        <text x="14" y="9" className="text-[9px] fill-muted-foreground">Deducted</text>
    55→        <rect x="70" y="0" width="10" height="10" rx="2" fill="#64748b" opacity={0.5} />
    56→        <text x="84" y="9" className="text-[9px] fill-muted-foreground">Deposited</text>
    57→      </g>
    58→    </svg>
    59→  )
    60→}
