    35→// ─── Animated Counter ────────────────────────────────────
    36→function AnimatedCounter({ target, suffix = '', prefix = '' }: { target: number; suffix?: string; prefix?: string }) {
    37→  const [count, setCount] = useState(0);
    38→  const ref = useRef<HTMLSpanElement>(null);
    39→  const isInView = useInView(ref, { once: true, margin: '-50px' });
    40→
    41→  useEffect(() => {
    42→    if (!isInView) return;
    43→    const duration = 2000;
    44→    const steps = 60;
    45→    const increment = target / steps;
    46→    let current = 0;
    47→    const timer = setInterval(() => {
    48→      current += increment;
    49→      if (current >= target) {
    50→        setCount(target);
    51→        clearInterval(timer);
    52→      } else {
    53→        setCount(Math.floor(current));
    54→      }
    55→    }, duration / steps);
    56→    return () => clearInterval(timer);
    57→  }, [isInView, target]);
    58→
    59→  return (
    60→    <span ref={ref}>
    61→      {prefix}{count.toLocaleString('en-IN')}{suffix}
    62→    </span>
    63→  );
    64→}

    67→function Section({ children, className = '', id }: { children: React.ReactNode; className?: string; id?: string }) {

    72→  useEffect(() => {
    73→    if (isInView) {
    74→      controls.start('visible');
    75→    }
    76→  }, [isInView, controls]);
    77→
    78→  return (
    79→    <motion.section
    80→      id={id}
    81→      ref={ref}
    82→      initial="hidden"
    83→      animate={controls}
    84→      variants={{
    85→        hidden: {},
    86→        visible: { transition: { staggerChildren: 0.1 } },
    87→      }}
    88→      className={className}
    89→    >
    90→      {children}
