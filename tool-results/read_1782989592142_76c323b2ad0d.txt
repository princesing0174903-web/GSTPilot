    13→// ─── Safe JSON parse ──────────────────────────────────────────────────────────
    14→function parseJSON<T>(raw: unknown, fallback: T): T {
    15→  if (typeof raw !== 'string' || raw.length === 0) return fallback;
    16→  try {
    17→    return JSON.parse(raw) as T;
    18→  } catch {
    19→    return fallback;
    20→  }
    21→}

    23→// ─── Helpers ──────────────────────────────────────────────────────────────────
    24→function computeRiskLevel(trust: number): RiskLevel {
    25→  if (trust >= 85) return 'low';
    26→  if (trust >= 65) return 'medium';
    27→  if (trust >= 40) return 'high';
    28→  return 'critical';
    29→}
    30→
    31→function computeTrustScore(compliance: number, paymentReliability: number, rating: number): number {
    32→  // rating is 0-5, scale to 0-100
    33→  const ratingPct = (rating / 5) * 100;
    34→  const composite = compliance * 0.35 + paymentReliability * 0.35 + ratingPct * 0.3;
    35→  return Math.round(Math.min(100, Math.max(0, composite)));
    36→}

   289→export async function ensureNetworkSeeded(): Promise<void> {
   290→  if (SEED_LOCK.value) return;
   291→  SEED_LOCK.value = true;
   292→  try {
   293→    // Make sure platform orgs exist first (they anchor to the real Firm)
   294→    await ensurePlatformOrganizationsSeeded();
   295→
   296→    const existingCount = await db.networkNode.count();
   297→    if (existingCount > 0) return; // idempotent
   298→
   299→    // ── 1. Create NetworkNodes for every REAL PlatformOrganization ──
   300→    const platformOrgs = await db.platformOrganization.findMany({
   301→      orderBy: { createdAt: 'asc' },
   302→    });
   303→
   304→    // The first platform org is the host firm itself — treat as 'organization' type.
   305→    // All other platform orgs are modeled as customers (they're the firm's customer base).
   306→    for (const org of platformOrgs) {
   307→      const isHost = org.organizationType === 'parent';
   308→      const nodeType: NodeType = isHost ? 'organization' : 'customer';
   309→      const complianceScore = isHost ? 92 : Math.min(95, 70 + Math.floor((org.healthScore || 80) * 0.25));
   310→      const paymentReliabilityScore = isHost ? 88 : Math.min(95, 65 + Math.floor((org.healthScore || 80) * 0.3));
   311→      const rating = isHost ? 4.5 : Math.min(5, 3.5 + (org.healthScore || 80) / 100);
   312→      const trustScore = computeTrustScore(complianceScore, paymentReliabilityScore, rating);
   313→
   314→      await db.networkNode.create({
   315→        data: {
   316→          organizationId: org.id,
   317→          nodeType,
   318→          legalName: org.legalName || org.name,
   319→          tradeName: org.name,
   320→          gstin: null,
   321→          email: null,
   322→          website: org.domain || null,
   323→          country: org.country,
   324→          state: null,
   325→          city: null,
   326→          industry: org.industry,
   327→          sector: 'services',
   328→          employeeCount: org.employeeCount,
   329→          annualRevenue: org.monthlyRevenue * 12,
   330→          yearsInBusiness: 5,
   331→          trustScore,
   332→          complianceScore,
   333→          paymentReliabilityScore,
   334→          supplierRating: rating,
   335→          customerRating: rating,
   336→          aiConfidenceScore: isHost ? 85 : 70,
   337→          riskLevel: computeRiskLevel(trustScore),
   338→          verified: isHost,
   339→          verificationDate: isHost ? new Date() : null,
   340→          verificationSource: isHost ? 'self' : null,
   341→          totalConnections: 0,
   342→          totalTransactions: 0,

   443→      orderBy: { createdAt: 'asc' },
   444→    });
   445→    const hostNode = orgNodes[0];
   446→    if (hostNode) {
   447→      const customerNodes = await db.networkNode.findMany({
   448→        where: { nodeType: 'customer' },
   449→      });
   450→      for (const cn of customerNodes) {
   451→        await db.networkEdge.create({
   452→          data: {
   453→            fromNodeId: hostNode.id,
   454→            toNodeId: cn.id,
   455→            relationshipType: 'customer',
   456→            status: 'active',
   457→            strength: 50 + (cn.trustScore % 40),
   458→            totalTransactionValue: 0,
   459→            transactionCount: 0,
   460→            startedAt: new Date(),
   461→          },
   462→        }).catch(() => { /* unique constraint guard */ });
   463→      }
   464→
   465→      // ── 5. Build edges from host org to suppliers, banks, government, accountants, auditors, partners ──
   466→      const externalTypes: Array<{ type: string; rel: string }> = [
   467→        { type: 'supplier', rel: 'supplier' },
   468→        { type: 'bank', rel: 'bank' },
   469→        { type: 'government', rel: 'government' },
   470→        { type: 'investor', rel: 'investor' },
   471→        { type: 'accountant', rel: 'accountant' },
   472→        { type: 'auditor', rel: 'auditor' },
   473→        { type: 'logistics', rel: 'logistics' },
   474→        { type: 'partner', rel: 'partner' },
   475→      ];
   476→      for (const { type, rel } of externalTypes) {
   477→        const extNodes = await db.networkNode.findMany({ where: { nodeType: type } });
   478→        for (const en of extNodes) {
   479→          await db.networkEdge.create({
   480→            data: {
   481→              fromNodeId: hostNode.id,
   482→              toNodeId: en.id,
   483→              relationshipType: rel,
   484→              status: 'active',
   485→              strength: 60 + (en.trustScore % 35),
   486→              totalTransactionValue: 0,
   487→              transactionCount: 0,
   488→              startedAt: new Date(),
   489→            },
   490→          }).catch(() => { /* unique constraint guard */ });
   491→        }
   492→      }
   493→    }
   494→
   495→    // ── 6. Derive REAL transactions from actual invoices ──
   496→    // For each real Invoice, create a NetworkTransaction from the host org to the matching client node.
   497→    if (hostNode) {
   498→      const realInvoices = await db.invoice.findMany({
   499→        take: 200,
   500→        orderBy: { createdAt: 'desc' },
   501→        include: { client: { select: { id: true, tradeName: true, legalName: true } } },
   502→      });
   503→      for (const inv of realInvoices) {
   504→        // Match by buyerName OR by the related client's tradeName/legalName
   505→        const matchName = inv.buyerName || inv.client?.tradeName || inv.client?.legalName;
   506→        if (!matchName) continue;
   507→        const clientNode = await db.networkNode.findFirst({
   508→          where: {
   509→            nodeType: 'customer',
   510→            OR: [
   511→              { legalName: matchName },
   512→              { tradeName: matchName },
   513→            ],
   514→          },
   515→          select: { id: true },
   516→        });
   517→        if (clientNode) {
   518→          const invNumber = inv.invoiceNumber || `INV-${inv.id.slice(-6)}`;
   519→          const amount = Number(inv.totalAmount || inv.taxableValue || 0);
   520→          await db.networkTransaction.create({
   521→            data: {
   522→              transactionNumber: `NTXN-${invNumber}`,
   523→              fromNodeId: hostNode.id,
   524→              toNodeId: clientNode.id,
   525→              type: 'invoice',
   526→              reference: invNumber,
   527→              amount,
   528→              currency: 'INR',
   529→              status: inv.status === 'paid' ? 'completed' : 'pending',
   530→              metadata: JSON.stringify({ source: 'real_invoice', invoiceId: inv.id }),
   531→            },
   532→          }).catch(() => { /* unique-constraint guard */ });
   533→        }
   534→      }
   535→    }
   536→
   537→    // ── 7. Update node totals based on real transactions ──
   538→    if (hostNode) {
   539→      const txnAgg = await db.networkTransaction.groupBy({
   540→        by: ['fromNodeId'],
   541→        _count: { id: true },
   542→        _sum: { amount: true },
   543→      });
   544→      for (const agg of txnAgg) {
   545→        await db.networkNode.update({
   546→          where: { id: agg.fromNodeId },
   547→          data: {
   548→            totalTransactions: agg._count.id,
   549→            totalTransactionValue: agg._sum.amount || 0,
   550→          },
   551→        });
   552→      }
   553→    }
   554→  } finally {
   555→    SEED_LOCK.value = false;
   556→  }
   557→}

   561→export async function listNodes(filter?: {
   562→  nodeType?: NodeType;
   563→  limit?: number;
   564→  verifiedOnly?: boolean;
   565→  search?: string;
   566→}): Promise<NetworkNode[]> {
   567→  const where: Record<string, unknown> = {};
   568→  if (filter?.nodeType) where.nodeType = filter.nodeType;
   569→  if (filter?.verifiedOnly) where.verified = true;
   570→  if (filter?.search) {
   571→    where.OR = [
   572→      { legalName: { contains: filter.search } },
   573→      { tradeName: { contains: filter.search } },
   574→      { gstin: { contains: filter.search } },
   575→      { city: { contains: filter.search } },
   576→      { industry: { contains: filter.search } },
   577→    ];
   578→  }
   579→  const nodes = await db.networkNode.findMany({
   580→    where,
   581→    orderBy: { createdAt: 'desc' },
   582→    take: filter?.limit ?? 100,
   583→  });
   584→  return nodes.map(mapNode);
   585→}

   595→export async function resolveOrgId(explicit?: string): Promise<{ orgId: string; hostNodeName: string }> {
   596→  if (explicit) {
   597→    const org = await db.platformOrganization.findUnique({ where: { id: explicit } });
   598→    if (org) return { orgId: org.id, hostNodeName: org.name };
   599→  }
   600→  // Fall back to the anchor host org
   601→  const host = await db.platformOrganization.findFirst({ orderBy: { createdAt: 'asc' } });
   602→  return { orgId: host?.id ?? 'anchor', hostNodeName: host?.name ?? 'GSTPilot Network' };
   603→}
