   216→function getCountdown(dueDate: string | null): { text: string; color: string; urgent: boolean } {
   217→  if (!dueDate) return { text: 'No due date', color: 'text-muted-foreground', urgent: false };
   218→  const now = new Date();
   219→  const due = new Date(dueDate);
   220→  const diff = due.getTime() - now.getTime();
   221→  const days = Math.ceil(diff / (1000 * 60 * 60 * 24));
   222→
   223→  if (days < 0) return { text: `${Math.abs(days)}d overdue`, color: 'text-red-600 dark:text-red-400', urgent: true };
   224→  if (days === 0) return { text: 'Due today', color: 'text-red-600 dark:text-red-400', urgent: true };
   225→  if (days <= 3) return { text: `${days}d left`, color: 'text-amber-600 dark:text-amber-400', urgent: true };
   226→  if (days <= 7) return { text: `${days}d left`, color: 'text-amber-600 dark:text-amber-400', urgent: false };
   227→  return { text: `${days}d left`, color: 'text-emerald-600 dark:text-emerald-400', urgent: false };
   228→}
