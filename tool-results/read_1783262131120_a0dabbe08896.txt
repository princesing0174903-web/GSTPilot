    63→    63→/**
    64→    64→ * Convert a Firestore snapshot ({ id, data() }) into a typed object with the
    65→    65→ * document id prepended, recursively converting Firestore Timestamps to ISO
    66→    66→ * strings. Mirrors the conversion logic in `withId` (firestore-schema.ts) but
    67→    67→ * drops the `T extends Record<string, unknown>` constraint so it accepts
    68→    68→ * strongly-typed interfaces (which do not have an implicit index signature).
    69→    69→ */
    70→    70→function docToData<T>(d: { id: string; data: () => Record<string, unknown> }): T & { id: string } {
    71→    71→  const data = d.data();
    72→    72→  const converted: Record<string, unknown> = {};
    73→    73→  for (const [key, value] of Object.entries(data)) {
    74→    74→    if (value && typeof value === 'object' && 'toDate' in value && typeof (value as { toDate: () => Date }).toDate === 'function') {
    75→    75→      converted[key] = (value as { toDate: () => Date }).toDate().toISOString();
    76→    76→    } else {
    77→    77→      converted[key] = value;
    78→    78→    }
    79→    79→  }
    80→    80→  return { id: d.id, ...converted } as T & { id: string };
    81→    81→}

   904→   904→/** Subscribe to a Firestore collection with firm scoping */
   905→   905→export function subscribeToCollection<T>(
   906→   906→  collectionName: CollectionName,
   907→   907→  callback: (data: Array<T & { id: string }>) => void,
   908→   908→  ...constraints: QueryConstraint[]
   909→   909→): Unsubscribe {
   910→   910→  const firmId = currentFirmId();
   911→   911→
   912→   912→  const q = firmId
   913→   913→    ? query(collection(db, collectionName), where('firmId', '==', firmId), ...constraints)
   914→   914→    : query(collection(db, collectionName), ...constraints);
   915→   915→
   916→   916→  return onSnapshot(q, (snap) => {
   917→   917→    const data = snap.docs.map(d => {
   918→   918→      const docData = d.data();
   919→   919→      const converted: Record<string, unknown> = { id: d.id };
   920→   920→      for (const [key, value] of Object.entries(docData)) {
   921→   921→        if (value && typeof value === 'object' && 'toDate' in value && typeof (value as { toDate: () => Date }).toDate === 'function') {
   922→   922→          converted[key] = (value as { toDate: () => Date }).toDate().toISOString();
   923→   923→        } else {
   924→   924→          converted[key] = value;
   925→   925→        }
   926→   926→      }
   927→   927→      return converted as T & { id: string };
   928→   928→    });
   929→   929→    callback(data);
   930→   930→  }, (error) => {
   931→   931→    console.warn(`[Firestore] Subscription error for ${collectionName}:`, error);
   932→   932→    callback([]);
   933→   933→  });
   934→   934→}

   936→   936→/** Subscribe to a single Firestore document */
   937→   937→export function subscribeToDoc<T>(
   938→   938→  collectionName: CollectionName,
   939→   939→  docId: string,
   940→   940→  callback: (data: (T & { id: string }) | null) => void,
   941→   941→): Unsubscribe {
   942→   942→  return onSnapshot(doc(db, collectionName, docId), (snap) => {
   943→   943→    if (!snap.exists()) {
   944→   944→      callback(null);
   945→   945→      return;
   946→   946→    }
   947→   947→    const docData = snap.data();
   948→   948→    const converted: Record<string, unknown> = { id: snap.id };
   949→   949→    for (const [key, value] of Object.entries(docData)) {
   950→   950→      if (value && typeof value === 'object' && 'toDate' in value && typeof (value as { toDate: () => Date }).toDate === 'function') {
   951→   951→        converted[key] = (value as { toDate: () => Date }).toDate().toISOString();
   952→   952→      } else {
   953→   953→        converted[key] = value;
   954→   954→      }
   955→   955→    }
   956→   956→    callback(converted as T & { id: string });
   957→   957→  }, (error) => {
   958→   958→    console.warn(`[Firestore] Doc subscription error for ${collectionName}/${docId}:`, error);
   959→   959→    callback(null);
   960→   960→  });
   961→   961→}

   963→   963→/** Compute live dashboard metrics from Firestore data */
   964→   964→export function computeDashboardMetrics(
   965→   965→  clients: Array<FirestoreClient & { id: string }>,
   966→   966→  invoices: Array<FirestoreInvoice & { id: string }>,
   967→   967→  returns: Array<FirestoreReturn & { id: string }>,
   968→   968→  documents: Array<FirestoreDocument & { id: string }>,
   969→   969→  activities: Array<FirestoreActivity & { id: string }>,
   970→   970→): LiveDashboardMetrics {
   971→   971→  const activeClients = clients.filter(c => c.status === 'active').length;
   972→   972→  const filedReturns = returns.filter(r => r.status === 'filed').length;
   973→   973→  const pendingReturns = returns.filter(r => r.status !== 'filed').length;
   974→   974→  const readyToFile = returns.filter(r => ['validated', 'reviewed', 'generated'].includes(r.status)).length;
   975→   975→  const overdueReturns = returns.filter(r => {
   976→   976→    if (r.status === 'filed') return false;
   977→   977→    const periodDate = parsePeriod(r.period);
   978→   978→    return periodDate && periodDate < new Date();
   979→   979→  }).length;
   980→   980→
   981→   981→  const totalTaxVolume = invoices.reduce((sum, inv) => sum + (inv.totalAmount || 0), 0);
   982→   982→  const totalTax = invoices.reduce((sum, inv) => sum + (inv.cgst || 0) + (inv.sgst || 0) + (inv.igst || 0) + (inv.cess || 0), 0);
   983→   983→
   984→   984→  const criticalIssues = invoices.filter(inv => inv.riskLevel === 'critical').length +
   985→   985→    returns.filter(r => r.criticalErrors > 0).length;
   986→   986→  const warnings = invoices.filter(inv => inv.riskLevel === 'high' || inv.riskLevel === 'medium').length +
   987→   987→    returns.filter(r => r.warnings > 0).length;
   988→   988→
   989→   989→  const averageHealthScore = clients.length > 0
   990→   990→    ? Math.round(clients.reduce((sum, c) => sum + c.healthScore, 0) / clients.length)
   991→   991→    : 0;
   992→   992→
   993→   993→  const matchedInvoices = invoices.filter(inv => inv.matchStatus === 'perfect_match').length;
   994→   994→  const matchPercentage = invoices.length > 0 ? Math.round((matchedInvoices / invoices.length) * 100) : 100;
   995→   995→
   996→   996→  const riskyInvoices = invoices.filter(inv => ['high', 'critical'].includes(inv.riskLevel)).length;
   997→   997→  const riskPercentage = invoices.length > 0 ? Math.round((riskyInvoices / invoices.length) * 100) : 0;
   998→   998→
   999→   999→  const documentsProcessed = documents.filter(d => d.extractionStatus === 'completed').length;
  1000→  1000→  const extractionsPending = documents.filter(d => d.extractionStatus === 'pending' || d.extractionStatus === 'in_progress').length;
  1001→  1001→
  1002→  1002→  const recentActivities = [...activities]
  1003→  1003→    .sort((a, b) => {
  1004→  1004→      const aTime = typeof a.createdAt === 'string' ? new Date(a.createdAt).getTime() : 0;
  1005→  1005→      const bTime = typeof b.createdAt === 'string' ? new Date(b.createdAt).getTime() : 0;
  1006→  1006→      return bTime - aTime;
  1007→  1007→    })
  1008→  1008→    .slice(0, 10);
  1009→  1009→
  1010→  1010→  const upcomingFilings = returns
  1011→  1011→    .filter(r => r.status !== 'filed')
  1012→  1012→    .sort((a, b) => {
  1013→  1013→      const aTime = typeof a.createdAt === 'string' ? new Date(a.createdAt).getTime() : 0;
  1014→  1014→      const bTime = typeof b.createdAt === 'string' ? new Date(b.createdAt).getTime() : 0;
  1015→  1015→      return aTime - bTime;
  1016→  1016→    })
  1017→  1017→    .slice(0, 5);
  1018→  1018→
  1019→  1019→  return {
  1020→  1020→    totalClients: clients.length,
  1021→  1021→    activeClients,
  1022→  1022→    totalInvoices: invoices.length,
  1023→  1023→    totalTaxVolume,
  1024→  1024→    filedReturns,
  1025→  1025→    pendingReturns,
  1026→  1026→    overdueReturns,
  1027→  1027→    readyToFile,
  1028→  1028→    criticalIssues,
  1029→  1029→    warnings,
  1030→  1030→    averageHealthScore,
  1031→  1031→    matchPercentage,
  1032→  1032→    riskPercentage,
  1033→  1033→    documentsProcessed,
  1034→  1034→    extractionsPending,
  1035→  1035→    recentActivities,
  1036→  1036→    upcomingFilings,
  1037→  1037→  };
  1038→  1038→}

  1040→  1040→function parsePeriod(period: string | null): Date | null {
  1041→  1041→  if (!period) return null;
  1042→  1042→  try {
  1043→  1043→    const [mm, yyyy] = period.split('-').map(Number);
  1044→  1044→    if (!mm || !yyyy) return null;
  1045→  1045→    return new Date(yyyy, mm, 20); // Due date is typically 20th of next month
  1046→  1046→  } catch {
  1047→  1047→    return null;
  1048→  1048→  }
  1049→  1049→}
