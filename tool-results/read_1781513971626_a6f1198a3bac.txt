   212→// ──────────────────────────────────────────────
   213→// Animated SVG Ring for Match Rate
   214→// ──────────────────────────────────────────────
   215→function MatchRateRing({
   216→  matchPercent,
   217→  partialPercent,
   218→  mismatchPercent,
   219→  size = 160,
   220→  strokeWidth = 14,
   221→}: {
   222→  matchPercent: number;
   223→  partialPercent: number;
   224→  mismatchPercent: number;
   225→  size?: number;
   226→  strokeWidth?: number;
   227→}) {
   228→  const radius = (size - strokeWidth) / 2;
   229→  const circumference = 2 * Math.PI * radius;
   230→  const gapAngle = 2; // degrees gap between segments
   231→
   232→  const totalGap = gapAngle * 3 * (circumference / 360);
   233→  const availableCircumference = circumference - totalGap;
   234→  const gapLen = totalGap / 3;
   235→
   236→  const matchLen = (matchPercent / 100) * availableCircumference;
   237→  const partialLen = (partialPercent / 100) * availableCircumference;
   238→  const mismatchLen = (mismatchPercent / 100) * availableCircumference;
   239→
   240→  const matchOffset = 0;
   241→  const partialOffset = -(matchLen + gapLen);
   242→  const mismatchOffset = -(matchLen + gapLen + partialLen + gapLen);
   243→
   244→  return (
   245→    <div className="relative inline-flex items-center justify-center" style={{ width: size, height: size }}>
   246→      <svg width={size} height={size} className="-rotate-90">
   247→        {/* Background track */}
   248→        <circle
   249→          cx={size / 2}
   250→          cy={size / 2}
   251→          r={radius}
   252→          fill="none"
   253→          stroke="#f1f5f9"
   254→          strokeWidth={strokeWidth}
   255→        />
   256→        {/* Glow filter */}
   257→        <defs>
   258→          <filter id="ringGlow" x="-20%" y="-20%" width="140%" height="140%">
   259→            <feGaussianBlur stdDeviation="2.5" result="blur" />
   260→            <feMerge>
   261→              <feMergeNode in="blur" />
   262→              <feMergeNode in="SourceGraphic" />
   263→            </feMerge>
   264→          </filter>
   265→        </defs>
   266→        {/* Green segment — matched */}
   267→        {matchPercent > 0 && (
   268→          <motion.circle
   269→            cx={size / 2}
   270→            cy={size / 2}
   271→            r={radius}
   272→            fill="none"
   273→            stroke="#10b981"
   274→            strokeWidth={strokeWidth}
   275→            strokeLinecap="round"
   276→            strokeDasharray={`${matchLen} ${circumference - matchLen}`}
   277→            strokeDashoffset={matchOffset}
   278→            initial={{ strokeDasharray: `0 ${circumference}` }}
   279→            animate={{ strokeDasharray: `${matchLen} ${circumference - matchLen}` }}
   280→            transition={{ duration: 1.4, ease: [0.25, 0.46, 0.45, 0.94] }}
   281→            filter="url(#ringGlow)"
   282→          />
   283→        )}
   284→        {/* Amber segment — partial */}
   285→        {partialPercent > 0 && (
   286→          <motion.circle
   287→            cx={size / 2}
   288→            cy={size / 2}
   289→            r={radius}
   290→            fill="none"
   291→            stroke="#f59e0b"
   292→            strokeWidth={strokeWidth}
   293→            strokeLinecap="round"
   294→            strokeDasharray={`${partialLen} ${circumference - partialLen}`}
   295→            strokeDashoffset={partialOffset}
   296→            initial={{ strokeDasharray: `0 ${circumference}`, strokeDashoffset: 0 }}
   297→            animate={{ strokeDasharray: `${partialLen} ${circumference - partialLen}`, strokeDashoffset: partialOffset }}
   298→            transition={{ duration: 1.4, ease: [0.25, 0.46, 0.45, 0.94], delay: 0.3 }}
   299→            filter="url(#ringGlow)"
   300→          />
   301→        )}
   302→        {/* Red segment — mismatch/missing */}
   303→        {mismatchPercent > 0 && (
   304→          <motion.circle
   305→            cx={size / 2}
   306→            cy={size / 2}
   307→            r={radius}
   308→            fill="none"
   309→            stroke="#ef4444"
   310→            strokeWidth={strokeWidth}
   311→            strokeLinecap="round"
   312→            strokeDasharray={`${mismatchLen} ${circumference - mismatchLen}`}
   313→            strokeDashoffset={mismatchOffset}
   314→            initial={{ strokeDasharray: `0 ${circumference}`, strokeDashoffset: 0 }}
   315→            animate={{ strokeDasharray: `${mismatchLen} ${circumference - mismatchLen}`, strokeDashoffset: mismatchOffset }}
   316→            transition={{ duration: 1.4, ease: [0.25, 0.46, 0.45, 0.94], delay: 0.6 }}
   317→            filter="url(#ringGlow)"
   318→          />
   319→        )}
   320→      </svg>
   321→      {/* Center content */}
   322→      <div className="absolute inset-0 flex flex-col items-center justify-center">
   323→        <motion.span
   324→          className="text-4xl font-bold tracking-tight text-foreground"
   325→          initial={{ opacity: 0, scale: 0.5 }}
   326→          animate={{ opacity: 1, scale: 1 }}
   327→          transition={{ delay: 0.8, duration: 0.5, type: 'spring' }}
   328→        >
   329→          {matchPercent}%
   330→        </motion.span>
   331→        <span className="text-[11px] font-medium text-muted-foreground mt-0.5">Match Rate</span>
   332→      </div>
   333→    </div>
   334→  );
   335→}

   389→// ─── Map API ReconciliationResult to local MismatchRecord ─────────────────
   390→function mapReconResultToMismatch(
   391→  result: ReconciliationResult,
   392→  clientName: string
   393→): MismatchRecord {
   394→  // Parse mismatches JSON string
   395→  let parsedMismatches: MismatchDetail[] = [];
   396→  try {
   397→    if (result.mismatches) {
   398→      parsedMismatches = JSON.parse(result.mismatches);
   399→    }
   400→  } catch {
   401→    parsedMismatches = [];
   402→  }
   403→
   404→  // Derive category from matchStatus
   405→  const mismatchCategory =
   406→    result.matchStatus === 'missing_in_gstr' ? 'missing_in_gstr' :
   407→    result.matchStatus === 'missing_in_books' ? 'missing_in_books' :
   408→    result.matchStatus === 'duplicate' ? 'duplicate' :
   409→    parsedMismatches.some(m => m.field === 'vendor_gstin' || m.field === 'sellerGstin') ? 'gstin_mismatch' :
   410→    'tax_difference';
   411→
   412→  const mismatchType =
   413→    result.matchStatus === 'perfect_match' ? 'Perfect Match' :
   414→    result.matchStatus === 'partial_match' ? 'Partial Match' :
   415→    result.matchStatus === 'missing_in_gstr' ? 'Missing in Portal' :
   416→    result.matchStatus === 'missing_in_books' ? 'Missing in Books' :
   417→    result.matchStatus === 'duplicate' ? 'Duplicate' :
   418→    'Mismatch';
   419→
   420→  // Compute tax difference from mismatch details
   421→  const taxDiff = parsedMismatches.reduce((sum, m) => sum + (m.difference || 0), 0);
   422→
   423→  // Extract diff fields for highlighting
   424→  const diffFields = parsedMismatches.map(m => m.field);
   425→
   426→  // Build booksData from invoice if available, else from mismatch details
   427→  const booksTaxableValue = parsedMismatches.find(m => m.field === 'taxableValue')?.expected as number || 0;
   428→  const portalTaxableValue = parsedMismatches.find(m => m.field === 'taxableValue')?.actual as number || 0;
   429→
   430→  return {
   431→    id: result.id,
   432→    invoiceNumber: result.invoice?.invoiceNumber ?? result.sourceGstin ?? 'Unknown',
   433→    clientName,
   434→    mismatchCategory,
   435→    mismatchType,
   436→    taxDifference: Math.abs(taxDiff),
   437→    confidenceScore: result.confidenceScore ?? 90,
   438→    aiExplanation: result.aiExplanation ?? generateMismatchExplanation(diffFields),
   439→    aiRecommendation: result.aiRecommendation ?? 'Review manually',
   440→    booksData: {
   441→      invoiceNumber: result.invoice?.invoiceNumber ?? '—',
   442→      invoiceDate: result.invoice?.invoiceDate ?? '—',
   443→      sellerGstin: result.sourceGstin ?? result.invoice?.sellerGstin ?? '—',
   444→      buyerGstin: result.invoice?.buyerGstin ?? '—',
   445→      taxableValue: booksTaxableValue || (result.invoice?.taxableValue ?? 0),
   446→      cgst: result.invoice?.cgst ?? 0,
   447→      sgst: result.invoice?.sgst ?? 0,
   448→      igst: result.invoice?.igst ?? 0,
   449→      totalAmount: result.invoice?.totalAmount ?? 0,
   450→    },
   451→    gstr2bData: result.matchStatus === 'missing_in_gstr' ? null : {
   452→      invoiceNumber: result.matchedGstin ?? result.invoice?.invoiceNumber ?? '—',
   453→      invoiceDate: result.invoice?.invoiceDate ?? '—',
   454→      sellerGstin: result.matchedGstin ?? result.sourceGstin ?? '—',
   455→      buyerGstin: '—',
   456→      taxableValue: portalTaxableValue,
   457→      cgst: 0,
   458→      sgst: 0,
   459→      igst: 0,
   460→      totalAmount: portalTaxableValue,
   461→    },
   462→    diffFields,
   463→    resolved: result.resolved,
   464→    workflowStatus: result.workflowStatus,
   465→  };
   466→}
