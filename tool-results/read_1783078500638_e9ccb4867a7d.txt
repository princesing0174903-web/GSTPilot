    63→function getHealthColor(score: number) {

    65→  if (score >= 50) return { bg: 'bg-amber-50', text: 'text-amber-700', stroke: '#f59e0b' };
    66→  return { bg: 'bg-red-50', text: 'text-red-700', stroke: '#ef4444' };

   112→function formatRelativeTime(timestamp: string): string {
   113→  if (!timestamp) return '';
   114→  const now = new Date();
   115→  const date = new Date(timestamp);
   116→  const diffMs = now.getTime() - date.getTime();
   117→  const diffMins = Math.floor(Math.abs(diffMs) / 60000);
   118→  const diffHours = Math.floor(diffMins / 60);
   119→  const diffDays = Math.floor(diffHours / 24);
   120→  if (diffMins < 60) return `${diffMins}m ago`;
   121→  if (diffHours < 24) return `${diffHours}h ago`;
   122→  if (diffDays < 7) return `${diffDays}d ago`;
   123→  return date.toLocaleDateString('en-IN', { day: 'numeric', month: 'short' });
   124→}

   139→function HealthRing({ score, size = 72, strokeWidth = 5 }: { score: number; size?: number; strokeWidth?: number }) {
   140→  const radius = (size - strokeWidth) / 2;
   141→  const circumference = 2 * Math.PI * radius;
   142→  const offset = circumference - (score / 100) * circumference;
   143→  const color = getHealthColor(score);
   144→
   145→  return (
   146→    <div className="relative inline-flex items-center justify-center" style={{ width: size, height: size }}>
   147→      <svg width={size} height={size} className="-rotate-90">
   148→        <circle cx={size / 2} cy={size / 2} r={radius} fill="none" stroke="#e2e8f0" strokeWidth={strokeWidth} />
   149→        <motion.circle
   150→          cx={size / 2} cy={size / 2} r={radius} fill="none"
   151→          stroke={color.stroke} strokeWidth={strokeWidth} strokeLinecap="round"
   152→          strokeDasharray={circumference}
   153→          initial={{ strokeDashoffset: circumference }}
   154→          animate={{ strokeDashoffset: offset }}
   155→          transition={{ duration: 1, delay: 0.3, ease: 'easeOut' }}
   156→        />
   157→      </svg>
   158→      <div className="absolute inset-0 flex flex-col items-center justify-center">
   159→        <span className={`font-bold text-lg ${color.text}`}>{score}</span>
   160→        <span className="text-[8px] text-muted-foreground">/ 100</span>
   161→      </div>
   162→    </div>
   163→  );
   164→}
