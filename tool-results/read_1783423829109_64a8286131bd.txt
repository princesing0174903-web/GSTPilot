     4→// React hooks with onSnapshot listeners for live data + optimistic UI.

     6→// ARCHITECTURE (Production-grade multi-tenant):
     7→//   • Every query is scoped by `organizationId` from OrgContext — NEVER by the
     8→//     legacy `firmId`. This matches the Firestore security rules, which require
     9→//     every tenant-scoped document to carry `organizationId` and only allow
    10→//     reads where `isOrgMember(resource.data.organizationId)` is true.

    12→//     hooks return empty data WITHOUT subscribing — so the UI renders premium
    13→//     empty states instead of throwing permission errors.
    14→//   • Permission errors (permission-denied / unauthenticated) are treated as
    15→//     "no data available" — the hook clears its error and returns an empty
    16→//     array. This is the graceful-degradation contract: the app NEVER shows a
    17→//     "Missing or insufficient permissions" wall. Genuine errors (network,
    18→//     index missing) still surface as `error`.

    46→// ─── Permission-error detection (graceful degradation) ───────────────────────
    47→
    48→/**
    49→ * True if the given Firestore error is a permission / auth failure that we
    50→ * should treat as "no data available" rather than a hard error. This is the
    51→ * core of the never-show-a-permission-wall contract.
    52→ */
    53→function isPermissionError(err: unknown): boolean {
    54→  if (!err) return false;
    55→  const e = err as { code?: string; message?: string };
    56→  const code = (e.code || '').toLowerCase();
    57→  const message = (e.message || '').toLowerCase();
    58→  if (
    59→    code === 'permission-denied' ||
    60→    code === 'unauthenticated' ||
    61→    code === 'auth/operation-not-allowed' ||
    62→    code === 'auth/user-not-found'
    63→  ) {
    64→    return true;
    65→  }
    66→  if (
    67→    message.includes('missing or insufficient permissions') ||
    68→    message.includes('permission-denied') ||
    69→    message.includes('insufficient permissions') ||
    70→    message.includes('not authorized') ||
    71→    message.includes('unauthenticated')
    72→  ) {
    73→    return true;
    74→  }
    75→  return false;
    76→}

    78→// ─── Timestamp Converter ─────────────────────────────────────────────────────
    79→
    80→function convertDoc<T extends Record<string, unknown>>(snapData: Record<string, unknown>): T {
    81→  const converted: Record<string, unknown> = {};
    82→  for (const [key, value] of Object.entries(snapData)) {
    83→    if (value && typeof value === 'object' && 'toDate' in value && typeof (value as { toDate: () => Date }).toDate === 'function') {
    84→      converted[key] = (value as { toDate: () => Date }).toDate().toISOString();
    85→    } else if (value && typeof value === 'object' && !Array.isArray(value) && !(value instanceof Date)) {
    86→      // Recursively convert nested objects
    87→      converted[key] = convertDoc(value as Record<string, unknown>);
    88→    } else {
    89→      converted[key] = value;
    90→    }
    91→  }
    92→  return converted as T;
    93→}

    95→// ─── Generic Collection Hook ─────────────────────────────────────────────────
    96→//
    97→// The hook resolves the current `organizationId` from OrgContext. If the org
    98→// isn't resolved yet (loading) OR we're in preview mode (no org), the hook
    99→// returns empty data without subscribing. This guarantees no permission errors
   100→// ever reach the UI in preview/offline mode.
   101→
   102→function useFirestoreCollection<T>(
   103→  collectionName: CollectionName,
   104→  constraints: QueryConstraint[] = [],
   105→  deps: unknown[] = [],
   106→): { data: Array<T & { id: string }>; loading: boolean; error: string | null } {
   107→  const { user } = useAuth();
   108→  const { organization, isPreviewMode, loading: orgLoading } = useOrg();
   109→  const organizationId = organization?.id ?? null;
   110→
   111→  const [data, setData] = useState<Array<T & { id: string }>>([]);
   112→  const [loading, setLoading] = useState(true);
   113→  const [error, setError] = useState<string | null>(null);
   114→
   115→  useEffect(() => {
   116→    // No user → nothing to subscribe to (empty data).
   117→    if (!user) {
   118→      setData([]);
   119→      setLoading(false);
   120→      setError(null);
   121→      return;
   122→    }
   123→
   124→    // Org still resolving — stay in loading state, but don't subscribe yet.
   125→    if (orgLoading && !organizationId) {
   126→      return;
   127→    }
   128→
   129→    // Preview mode (no real org) OR no organization at all → empty data,
   130→    // no subscription. This is the preview-mode contract.
   131→    if (!organizationId || isPreviewMode || organizationId === 'preview-org') {
   132→      setData([]);
   133→      setLoading(false);
   134→      setError(null);
   135→      return;
   136→    }
   137→
   138→    let unsub: Unsubscribe | null = null;
   139→
   140→    // Scope EVERY query by organizationId. This is required by the Firestore
   141→    // security rules — unbounded collection reads are denied.
   142→    const baseConstraints: QueryConstraint[] = [
   143→      where('organizationId', '==', organizationId),
   144→      ...constraints,
   145→    ];
   146→
   147→    const q = query(collection(db, collectionName), ...baseConstraints);
   148→
   149→    unsub = onSnapshot(q, (snap) => {
   150→      const items = snap.docs.map(d => ({
   151→        id: d.id,
   152→        ...convertDoc<T>(d.data() as Record<string, unknown>),
   153→      }));
   154→      setData(items);
   155→      setLoading(false);
   156→      setError(null);
   157→    }, (err) => {
   158→      console.warn(`[Firestore] ${collectionName} subscription error:`, err);
   159→      // Permission errors → graceful degradation (empty data, no error wall).
   160→      if (isPermissionError(err)) {
   161→        setData([]);
   162→        setError(null);
   163→        setLoading(false);
   164→        return;
   165→      }
   166→      // Genuine errors (network, missing index) still surface.
   167→      setError(err.message);
   168→      setLoading(false);
   169→    });
   170→
   171→    return () => {
   172→      if (unsub) unsub();
   173→    };
   174→  }, [user, organizationId, isPreviewMode, orgLoading, ...deps]);
   175→
   176→  // When user is null, return empty data (not loading)
   177→  if (!user && loading) {
   178→    return { data: [], loading: false, error: null };
   179→  }
   180→
   181→  return { data, loading, error };
   182→}

   184→// ─── Generic Document Hook ───────────────────────────────────────────────────
   185→
   186→function useFirestoreDoc<T>(
   187→  collectionName: CollectionName,
   188→  docId: string | null,
   189→): { data: (T & { id: string }) | null; loading: boolean; error: string | null } {
   190→  const { user } = useAuth();
   191→  const { organization, isPreviewMode } = useOrg();
   192→  const organizationId = organization?.id ?? null;
   193→
   194→  const [data, setData] = useState<(T & { id: string }) | null>(null);
   195→  const [loading, setLoading] = useState(!!docId);
   196→  const [error, setError] = useState<string | null>(null);
   197→
   198→  useEffect(() => {
   199→    if (!docId) {
   200→      setData(null);
   201→      setLoading(false);
   202→      return;
   203→    }
   204→
   205→    // Preview mode (no real org) → no doc, no subscription.
   206→    if (!user || !organizationId || isPreviewMode || organizationId === 'preview-org') {
   207→      setData(null);
   208→      setLoading(false);
   209→      setError(null);
   210→      return;
   211→    }
   212→
   213→    const unsub = onSnapshot(doc(db, collectionName, docId), (snap) => {
   214→      if (!snap.exists()) {
   215→        setData(null);
   216→        setLoading(false);
   217→        return;
   218→      }
   219→      setData({
   220→        id: snap.id,
   221→        ...convertDoc<T>(snap.data() as Record<string, unknown>),
   222→      });
   223→      setLoading(false);
   224→      setError(null);
   225→    }, (err) => {
   226→      console.warn(`[Firestore] ${collectionName}/${docId} doc error:`, err);
   227→      if (isPermissionError(err)) {
   228→        setData(null);
   229→        setError(null);
   230→        setLoading(false);
   231→        return;
   232→      }
   233→      setError(err.message);
   234→      setLoading(false);
   235→    });
   236→
   237→    return () => {
   238→      if (unsub) unsub();
   239→    };
   240→  }, [collectionName, docId, user, organizationId, isPreviewMode]);
   241→
   242→  return { data, loading, error };
   243→}

   460→// ─── Executive Scores (computed from live data) ─────────────────────────────
   461→
   462→export function useFirmExecutiveScores(): { scores: FirmExecutiveScores; loading: boolean } {
   463→  const { data: clients } = useFireClients();
   464→  const { data: returns } = useFireReturns();
   465→  const { data: invoices } = useFireInvoices();
   466→  const { data: reconciliations } = useFireReconciliations();
   467→  const { data: activities } = useFireRecentActivities(50);
   468→
   469→  const loading = clients.length === 0 && returns.length === 0;
   470→
   471→  const scores = useMemo<FirmExecutiveScores>(() => {
   472→    if (loading) {
   473→      return { firmHealth: 0, revenue: 0, compliance: 0, teamEfficiency: 0, clientSatisfaction: 0, cashFlow: 0 };
   474→    }
   475→
   476→    // Firm Health: weighted avg of client health scores
   477→    const activeClients = clients.filter(c => c.status === 'active');
   478→    const avgHealth = activeClients.length > 0
   479→      ? activeClients.reduce((s, c) => s + (c.healthScore || 0), 0) / activeClients.length
   480→      : 50;
   481→
   482→    // Revenue Score: based on tax volume
   483→    const totalTax = invoices.reduce((s, i) => s + (i.totalTax || 0) + (i.cgst || 0) + (i.sgst || 0) + (i.igst || 0), 0);
   484→    const revenueScore = Math.min(100, Math.round((totalTax / 1000000) * 100));
   485→
   486→    // Compliance Score: filed vs total returns
   487→    const filedReturns = returns.filter(r => r.status === 'filed').length;
   488→    const complianceScore = returns.length > 0 ? Math.round((filedReturns / returns.length) * 100) : 50;
   489→
   490→    // Team Efficiency: based on completed activities
   491→    const completedActivities = activities.filter(a => a.type.includes('filed') || a.type.includes('completed') || a.type.includes('resolved')).length;
   492→    const teamEfficiency = activities.length > 0 ? Math.min(100, Math.round((completedActivities / Math.max(activities.length, 1)) * 100)) : 50;
   493→
   494→    // Client Satisfaction: inverse of overdue + health
   495→    const overdueClients = activeClients.filter(c => (c.pendingReturnCount || 0) > 0).length;
   496→    const clientSatisfaction = activeClients.length > 0 ? Math.round(((activeClients.length - overdueClients) / activeClients.length) * 100) : 50;
   497→
   498→    // Cash Flow: based on reconciliation match rate
   499→    const matchedRecons = reconciliations.filter(r => r.matched > 0);
   500→    const totalMatched = matchedRecons.reduce((s, r) => s + r.matched, 0);
   501→    const totalRecords = matchedRecons.reduce((s, r) => s + r.totalRecords, 0);
   502→    const cashFlow = totalRecords > 0 ? Math.round((totalMatched / totalRecords) * 100) : 50;
   503→
   504→    return {
   505→      firmHealth: Math.round(avgHealth) || 50,
   506→      revenue: revenueScore || 50,
   507→      compliance: complianceScore || 50,
   508→      teamEfficiency: teamEfficiency || 50,
   509→      clientSatisfaction: clientSatisfaction || 50,
   510→      cashFlow: cashFlow || 50,
   511→    };
   512→  }, [clients, returns, invoices, reconciliations, activities, loading]);
   513→
   514→  return { scores, loading };
   515→}
