    59→    59→function formatDate(d: string | unknown): string {
    60→    60→  if (!d) return '—'
    61→    61→  try {
    62→    62→    const date = typeof d === 'string' ? new Date(d) : new Date()
    63→    63→    return date.toLocaleDateString('en-IN', { day: 'numeric', month: 'short', year: 'numeric' })
    64→    64→  } catch { return '—' }
    65→    65→}
    66→    66→
    67→    67→function formatTime(d: string | unknown): string {
    68→    68→  if (!d) return ''
    69→    69→  try {
    70→    70→    const date = typeof d === 'string' ? new Date(d) : new Date()
    71→    71→    return date.toLocaleTimeString('en-IN', { hour: '2-digit', minute: '2-digit' })
    72→    72→  } catch { return '' }
    73→    73→}
    74→    74→
    75→    75→function daysUntil(d: string | unknown): number {
    76→    76→  if (!d) return 999
    77→    77→  try {
    78→    78→    const date = typeof d === 'string' ? new Date(d) : new Date()
    79→    79→    return Math.ceil((date.getTime() - Date.now()) / (1000 * 60 * 60 * 24))
    80→    80→  } catch { return 999 }
    81→    81→}
