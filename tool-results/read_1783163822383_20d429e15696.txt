   120→  if (fileType === 'json') return 'json';

   126→function parseFileSizeBytes(fileSize: string): number {
   127→  // Accept formats like "12.3 KB", "1.5 MB", or "PDF" (non-numeric → 0)
   128→  const match = fileSize.match(/([\d.]+)\s*(KB|MB|GB)?/i);
   129→  if (!match) return 0;
   130→  const num = parseFloat(match[1]);
   131→  const unit = (match[2] ?? '').toUpperCase();
   132→  if (unit === 'KB') return Math.round(num * 1024);
   133→  if (unit === 'MB') return Math.round(num * 1024 * 1024);
   134→  if (unit === 'GB') return Math.round(num * 1024 * 1024 * 1024);
   135→  return Math.round(num);
   136→}

   143→function loadHistory(): RecentExport[] {
   144→  if (typeof window === 'undefined') return [];
   145→  try {
   146→    const raw = window.localStorage.getItem(HISTORY_KEY);
   147→    if (!raw) return [];
   148→    const parsed = JSON.parse(raw);
   149→    if (!Array.isArray(parsed)) return [];
   150→    return parsed.slice(0, HISTORY_LIMIT);
   151→  } catch {
   152→    return [];
   153→  }
   154→}

   162→  }

   178→function escapeHtml(s: string): string {
   179→  return String(s)
   180→    .replace(/&/g, '&amp;')
   181→    .replace(/</g, '&lt;')
   182→    .replace(/>/g, '&gt;')
   183→    .replace(/"/g, '&quot;')
   184→    .replace(/'/g, '&#39;');
   185→}

   280→function openPrintWindow(html: string): boolean {
   281→  if (typeof window === 'undefined') return false;
   282→  const printWindow = window.open('', '_blank', 'width=900,height=720');
   283→  if (!printWindow) {
   284→    // Pop-up blocked — fall back to writing into a hidden iframe
   285→    const iframe = document.createElement('iframe');
   286→    iframe.style.position = 'fixed';
   287→    iframe.style.right = '0';
   288→    iframe.style.bottom = '0';
   289→    iframe.style.width = '0';
   290→    iframe.style.height = '0';
   291→    iframe.style.border = '0';
   292→    document.body.appendChild(iframe);
   293→    const doc = iframe.contentWindow?.document;
   294→    if (!doc) {
   295→      document.body.removeChild(iframe);
   296→      return false;
   297→    }
   298→    doc.open();
   299→    doc.write(html);
   300→    doc.close();
   301→    iframe.contentWindow?.focus();
   302→    setTimeout(() => {
   303→      try { iframe.contentWindow?.print(); } catch { /* ignore */ }
   304→      setTimeout(() => document.body.removeChild(iframe), 1000);
   305→    }, 400);
   306→    return true;
   307→  }
   308→  printWindow.document.open();
   309→  printWindow.document.write(html);
   310→  printWindow.document.close();
   311→  printWindow.focus();
   312→  return true;
   313→}
