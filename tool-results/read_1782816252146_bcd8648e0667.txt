   632→function OnboardingScreen() {
   633→  const { user, markOnboardingComplete } = useAuth()
   634→  const { setCurrentScreen, setCurrentView } = useApp()
   635→  const [saveError, setSaveError] = React.useState<string | null>(null)
   636→
   637→  // ── Fire-and-forget: save onboarding data to Firestore in the background ──
   638→  const saveOnboardingToFirestore = async (
   639→    data: import('@/components/onboarding/OnboardingFlow').OnboardingData,
   640→    firmId: string
   641→  ) => {
   642→    try {
   643→      const { doc, setDoc, serverTimestamp } = await import('firebase/firestore')
   644→      const { db } = await import('@/lib/firebase')
   645→
   646→      if (!user) return
   647→
   648→      // Update user document
   649→      await setDoc(doc(db, 'users', user.id), {
   650→        uid: user.id,
   651→        fullName: data.fullName,
   652→        email: data.email,
   653→        phone: data.phone,
   654→        ageGroup: data.ageGroup,
   655→        profession: data.profession,
   656→        experience: data.experience,
   657→        firmId: firmId || null,
   658→        firmName: data.firmName,
   659→        onboardingCompleted: true,
   660→        clientCount: data.clientCount,
   661→        monthlyReturns: data.monthlyReturns,
   662→        gstServices: data.gstServices,
   663→        painPoints: data.painPoints,
   664→        referralSource: data.referralSource,
   665→        trialReasons: data.trialReasons,
   666→        wantsUpdates: data.wantsUpdates,
   667→        updatedAt: serverTimestamp(),
   668→      }, { merge: true })
   669→      console.log('[Onboarding] ✅ User document saved to Firestore')
   670→
   671→      // Save onboarding record
   672→      await setDoc(doc(db, 'onboarding', user.id), {
   673→        ...data,
   674→        firmId: firmId || null,
   675→        completedAt: serverTimestamp(),
   676→      })
   677→      console.log('[Onboarding] ✅ Onboarding record saved to Firestore')
   678→    } catch (error) {
   679→      console.warn('[Onboarding] ⚠️ Background Firestore save failed:', error)
   680→    }
   681→  }
   682→
   683→  const handleOnboardingComplete = (
   684→    data: import('@/components/onboarding/OnboardingFlow').OnboardingData,
   685→    destination?: import('@/components/onboarding/OnboardingFlow').OnboardingDestination
   686→  ) => {
   687→    setSaveError(null)
   688→
   689→    if (!user) {
   690→      setSaveError('No authenticated user found. Please sign in again.')
   691→      return
   692→    }
   693→
   694→    // ── INSTANT: Mark onboarding complete locally & navigate ──
   695→    // This happens synchronously — user sees dashboard immediately
   696→    console.log('[Onboarding] 🚀 Navigating to', destination || 'dashboard', '(optimistic)')
   697→    markOnboardingComplete(undefined, data.firmName)
   698→    setCurrentView(destination || 'dashboard')
   699→    setCurrentScreen('app')
   700→
   701→    // ── BACKGROUND: Create firm doc + save everything to Firestore ──
   702→    // These happen async — user is already in the dashboard
   703→    console.log('[Onboarding] 📝 Starting background Firestore writes...')
   704→    ;(async () => {
   705→      try {
   706→        const { collection, addDoc, serverTimestamp } = await import('firebase/firestore')
   707→        const { db } = await import('@/lib/firebase')
   708→
   709→        // Create firm document
   710→        let firmId = ''
   711→        try {
   712→          const firmRef = await addDoc(collection(db, 'firms'), {
   713→            ownerId: user.id,
   714→            firmName: data.firmName,
   715→            gstin: data.gstin || null,
   716→            state: data.state,
   717→            stateCode: data.stateCode,
   718→            organizationType: data.organizationType,
   719→            icaiMembershipNo: data.icaiMembershipNo || null,
   720→            officeAddress: data.officeAddress || null,
   721→            createdAt: serverTimestamp(),
   722→          })
   723→          firmId = firmRef.id
   724→          console.log('[Onboarding] ✅ Firm created:', firmId)
   725→
   726→          // Update the local user with firmId
   727→          markOnboardingComplete(firmId, data.firmName)
   728→        } catch (firmError) {
   729→          console.warn('[Onboarding] ⚠️ Firm creation failed:', firmError)
   730→        }
   731→
   732→        // Save remaining data
   733→        await saveOnboardingToFirestore(data, firmId)
   734→      } catch (error) {
   735→        console.warn('[Onboarding] ⚠️ Background save error:', error)
   736→      }
   737→    })()
   738→  }
   739→
   740→  const handleSkip = () => {
   741→    setSaveError(null)
   742→
   743→    if (!user) {
   744→      setSaveError('No authenticated user found. Please sign in again.')
   745→      return
   746→    }
   747→
   748→    // ── INSTANT: Navigate immediately ──
   749→    console.log('[Onboarding] 🚀 Skipping onboarding (optimistic)')
   750→    markOnboardingComplete()
   751→    setCurrentView('dashboard')
   752→    setCurrentScreen('app')
   753→
   754→    // ── BACKGROUND: Mark as completed in Firestore ──
   755→    ;(async () => {
   756→      try {
   757→        const { doc, setDoc, serverTimestamp } = await import('firebase/firestore')
   758→        const { db } = await import('@/lib/firebase')
   759→        await setDoc(doc(db, 'users', user.id), {
   760→          onboardingCompleted: true,
   761→          updatedAt: serverTimestamp(),
   762→        }, { merge: true })
   763→        console.log('[Onboarding] ✅ Skip saved to Firestore')
   764→      } catch (error) {
   765→        console.warn('[Onboarding] ⚠️ Background skip save failed:', error)
   766→      }
   767→    })()
   768→  }
   769→
   770→  return (
   771→    <OnboardingFlow
   772→      onComplete={handleOnboardingComplete}
   773→      onSkip={handleSkip}
   774→      userEmail={user?.email}
   775→      userName={user?.name}
   776→      error={saveError}
   777→      onDismissError={() => setSaveError(null)}
   778→    />
   779→  )
   780→}

   782→function AppRouter() {
   783→  const { currentScreen, setCurrentScreen } = useApp()
   784→  const { isAuthenticated, isInitializing, needsOnboarding, needsEmailVerification } = useAuth()
   785→
   786→  useEffect(() => {
   787→    if (isInitializing) return
   788→    // Only auto-redirect to app if authenticated AND onboarding is complete
   789→    // Don't redirect if user still needs onboarding — the OnboardingScreen handles its own navigation
   790→    if (isAuthenticated && !needsOnboarding && currentScreen !== 'app') {
   791→      setCurrentScreen('app')
   792→    }
   793→  }, [isAuthenticated, isInitializing, currentScreen, setCurrentScreen, needsOnboarding])
   794→
   795→  useEffect(() => {
   796→    if (isInitializing) return
   797→    if (!isAuthenticated && currentScreen === 'app') {
   798→      setCurrentScreen('landing')
   799→    }
   800→  }, [isAuthenticated, isInitializing, currentScreen, setCurrentScreen])
   801→
   802→  const handleGetStarted = () => setCurrentScreen('login')
   803→  const handleBookDemo = () => setCurrentScreen('login')
   804→  const handleBackToLanding = () => setCurrentScreen('landing')
   805→
   806→  if (isInitializing && !isAuthenticated) {
   807→    return (
   808→      <div className="min-h-screen flex items-center justify-center bg-black">
   809→        <div className="flex flex-col items-center gap-4">
   810→          <div className="flex h-11 w-11 items-center justify-center rounded-2xl glass-surface motion-pulse">
   811→            <Zap className="h-5 w-5 accent-text" />
   812→          </div>
   813→          <span className="text-sm text-white/55 font-medium">Loading GSTPilot…</span>
   814→        </div>
   815→      </div>
   816→    )
   817→  }
   818→
   819→  // Show onboarding if user hasn't completed it
   820→  if (isAuthenticated && needsOnboarding) {
   821→    return <OnboardingScreen />
   822→  }
   823→
   824→  if (currentScreen === 'app' && isAuthenticated) {
   825→    return (
   826→      <div className="flex min-h-screen flex-col">
   827→        {needsEmailVerification && <EmailVerificationBanner />}
   828→        <DashboardContent />
   829→      </div>
   830→    )
   831→  }
   832→
   833→  if (currentScreen === 'login') {
   834→    return <LoginPage onBack={handleBackToLanding} onGetStarted={handleGetStarted} />
   835→  }
   836→
   837→  return <LandingPage onGetStarted={handleGetStarted} onBookDemo={handleBookDemo} />
   838→}
