   134→function getStatusConfig(status: string, processingStep: string) {
   135→  // Map processingStep to the status for display
   136→  if (processingStep === 'uploading') return STATUS_CONFIG.uploading;
   137→  if (processingStep === 'extracting') return STATUS_CONFIG.extracting;
   138→  if (processingStep === 'validating') return STATUS_CONFIG.validating;
   139→  if (processingStep === 'completed' || status === 'completed') return STATUS_CONFIG.completed;
   140→  if (processingStep === 'failed' || status === 'failed') return STATUS_CONFIG.failed;
   141→  // Fallback based on status
   142→  if (STATUS_CONFIG[status]) return STATUS_CONFIG[status];
   143→  return STATUS_CONFIG.uploaded;

   147→function formatFileSize(bytes: number): string {
   148→  if (bytes === 0) return '0 B';
   149→  const k = 1024;
   150→  const sizes = ['B', 'KB', 'MB', 'GB'];
   151→  const i = Math.floor(Math.log(bytes) / Math.log(k));
   152→  return parseFloat((bytes / Math.pow(k, i)).toFixed(1)) + ' ' + sizes[i];
   153→}

   181→function formatDate(dateStr: string): string {
   182→  try {
   183→    return new Date(dateStr).toLocaleDateString('en-IN', {
   184→      day: '2-digit',
   185→      month: 'short',
   186→      year: 'numeric',
   187→    });
   188→  } catch {

   204→// Period helper — generates last 12 months
   205→function getPeriodOptions(): string[] {
   206→  const options: string[] = [];
   207→  const now = new Date();
   208→  for (let i = 0; i < 12; i++) {
   209→    const d = new Date(now.getFullYear(), now.getMonth() - i, 1);
   210→    const month = d.toLocaleString('en-IN', { month: 'short' });
   211→    options.push(`${month} ${d.getFullYear()}`);
   212→  }
   213→  return options;
   214→}

   216→// ─── Animated Number Hook ──────────────────────────────────────────────────
   217→function useAnimatedNumber(target: number, duration: number = 1200) {
   218→  const [current, setCurrent] = useState(0);
   219→  const ref = useRef<number | null>(null);
   220→  const startTime = useRef<number | null>(null);
   221→
   222→  useEffect(() => {
   223→    startTime.current = null;
   224→    const startValue = current;
   225→
   226→    function step(timestamp: number) {
   227→      if (!startTime.current) startTime.current = timestamp;
   228→      const elapsed = timestamp - startTime.current;
   229→      const progress = Math.min(elapsed / duration, 1);
   230→      const eased = 1 - Math.pow(1 - progress, 3);
   231→      setCurrent(Math.round(startValue + (target - startValue) * eased));
   232→      if (progress < 1) {
   233→        ref.current = requestAnimationFrame(step);
   234→      }
   235→    }
   236→
   237→    ref.current = requestAnimationFrame(step);
   238→    return () => {
   239→      if (ref.current) cancelAnimationFrame(ref.current);
   240→    };
   241→  }, [target, duration]);
   242→
   243→  return current;
   244→}
