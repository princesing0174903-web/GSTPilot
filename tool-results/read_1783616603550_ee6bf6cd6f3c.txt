   118→   118→// ═══════════════════════════════════════════════════════════════════════════════
   119→   119→
   120→   120→function loadRecent(): RecentItem[] {
   121→   121→  try {
   122→   122→    const raw = localStorage.getItem(RECENT_STORAGE_KEY);
   123→   123→    return raw ? JSON.parse(raw) : [];
   124→   124→  } catch {
   125→   125→    return [];
   126→   126→  }
   127→   127→}
   128→   128→
   129→   129→function saveRecent(items: RecentItem[]) {
   130→   130→  try {
   131→   131→    localStorage.setItem(RECENT_STORAGE_KEY, JSON.stringify(items.slice(0, MAX_RECENT)));
   132→   132→  } catch {
   133→   133→    // ignore quota errors
   134→   134→  }
   135→   135→}
   136→   136→
   137→   137→function loadFavorites(): FavoriteItem[] {
   138→   138→  try {
   139→   139→    const raw = localStorage.getItem(FAVORITES_STORAGE_KEY);
   140→   140→    return raw ? JSON.parse(raw) : [];
   141→   141→  } catch {
   142→   142→    return [];
   143→   143→  }
   144→   144→}
   145→   145→
   146→   146→function saveFavorites(items: FavoriteItem[]) {
   147→   147→  try {
   148→   148→    localStorage.setItem(FAVORITES_STORAGE_KEY, JSON.stringify(items));
   149→   149→  } catch {
   150→   150→    // ignore quota errors
   151→   151→  }
   152→   152→}
