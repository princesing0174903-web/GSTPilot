    49→    49→function getDaysRemaining(dueDateStr: string): number {
    50→    50→  const dueDate = new Date(dueDateStr);
    51→    51→  const today = new Date();
    52→    52→  today.setHours(0, 0, 0, 0);
    53→    53→  dueDate.setHours(0, 0, 0, 0);
    54→    54→  return Math.ceil((dueDate.getTime() - today.getTime()) / (1000 * 60 * 60 * 24));
    55→    55→}
    56→    56→
    57→    57→function formatDaysRemaining(days: number): string {
    58→    58→  if (days < 0) return `${Math.abs(days)}d overdue`;
    59→    59→  if (days === 0) return 'Due today';
    60→    60→  if (days === 1) return 'Tomorrow';
    61→    61→  return `${days}d left`;
    62→    62→}
    63→    63→
    64→    64→function formatINR(amount: number): string {
    65→    65→  return amount.toLocaleString('en-IN');
    66→    66→}
    67→    67→
    68→    68→function formatDateIN(dateStr: string): string {
    69→    69→  const d = new Date(dateStr);
    70→    70→  if (Number.isNaN(d.getTime())) return '—';
    71→    71→  const dd = String(d.getDate()).padStart(2, '0');
    72→    72→  const mm = String(d.getMonth() + 1).padStart(2, '0');
    73→    73→  const yyyy = d.getFullYear();
    74→    74→  return `${dd}/${mm}/${yyyy}`;
    75→    75→}
    76→    76→
    77→    77→function timeAgo(dateStr: string): string {
    78→    78→  const now = new Date();
    79→    79→  const date = new Date(dateStr);
    80→    80→  const seconds = Math.floor((now.getTime() - date.getTime()) / 1000);
    81→    81→  if (seconds < 60) return 'just now';
    82→    82→  const minutes = Math.floor(seconds / 60);
    83→    83→  if (minutes < 60) return `${minutes}m ago`;
    84→    84→  const hours = Math.floor(minutes / 60);
    85→    85→  if (hours < 24) return `${hours}h ago`;
    86→    86→  const days = Math.floor(hours / 24);
    87→    87→  if (days < 7) return `${days}d ago`;
    88→    88→  return date.toLocaleDateString('en-IN', { day: 'numeric', month: 'short' });
    89→    89→}

   104→   104→// One-sentence AI insight derived from live metrics.
   105→   105→function buildInsightSentence(
   106→   106→  metrics: LiveDashboardMetrics,
   107→   107→  pendingCollection: number,
   108→   108→): string {
   109→   109→  const parts: string[] = [];
   110→   110→  if (metrics.overdueReturns > 0) {
   111→   111→    parts.push(
   112→   112→      `${metrics.overdueReturns} overdue return${metrics.overdueReturns > 1 ? 's' : ''}`,
   113→   113→    );
   114→   114→  } else if (metrics.pendingReturns > 0) {
   115→   115→    parts.push(
   116→   116→      `${metrics.pendingReturns} return${metrics.pendingReturns > 1 ? 's' : ''} to file`,
   117→   117→    );
   118→   118→  }
   119→   119→  if (pendingCollection > 0) {
   120→   120→    parts.push(`₹${formatINR(pendingCollection)} pending collection`);
   121→   121→  }
   122→   122→  if (metrics.criticalIssues > 0) {
   123→   123→    parts.push(
   124→   124→      `${metrics.criticalIssues} critical issue${metrics.criticalIssues > 1 ? 's' : ''}`,
   125→   125→    );
   126→   126→  }
   127→   127→  if (parts.length === 0) {
   128→   128→    return `All clear — ${metrics.filedReturns} returns filed and ${metrics.totalClients} clients in good standing.`;
   129→   129→  }
   130→   130→  if (parts.length === 1) return `${parts[0]}.`;
   131→   131→  const last = parts.pop();
   132→   132→  return `${parts.join(', ')} and ${last}.`;
   133→   133→}
