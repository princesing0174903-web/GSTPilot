    38→  const d = new Date();
    39→  const ds = `${d.getFullYear()}${String(d.getMonth() + 1).padStart(2, '0')}${String(d.getDate()).padStart(2, '0')}`;

    47→function currentFirmId(): string | null {

   484→/** Auto-create draft returns for current period */
   485→async function autoCreateDraftReturns(firmId: string, clientId: string): Promise<void> {
   486→  const now = new Date();
   487→  const period = `${String(now.getMonth() + 1).padStart(2, '0')}-${now.getFullYear()}`;
   488→  const fy = now.getMonth() >= 3 ? `${now.getFullYear()}-${String(now.getFullYear() + 1).slice(2)}` : `${now.getFullYear() - 1}-${String(now.getFullYear()).slice(2)}`;
   489→
   490→  for (const returnType of ['GSTR-1', 'GSTR-3B'] as const) {
   491→    // Check if return already exists
   492→    const q = query(
   493→      collection(db, COLLECTIONS.RETURNS),
   494→      where('clientId', '==', clientId),
   495→      where('returnType', '==', returnType),
   496→      where('period', '==', period),
   497→      limit(1)
   498→    );
   499→    const existing = await getDocs(q);
   500→    if (existing.empty) {
   501→      await createReturn({
   502→        clientId,
   503→        returnType,
   504→        period,
   505→        financialYear: fy,
   506→        status: 'draft',
   507→        filedDate: null,
   508→        acknowledgmentNumber: null,
   509→        totalInvoices: 0,
   510→        readyForFiling: 0,
   511→        issuesFound: 0,
   512→        criticalErrors: 0,
   513→        warnings: 0,
   514→        totalTaxableValue: 0,
   515→        totalTax: 0,
   516→        jsonPayload: null,
   517→        assignedTo: null,
   518→        reviewedBy: null,
   519→      });
   520→    }
   521→  }
   522→}

   524→/** Update draft return when an invoice is added */
   525→async function updateDraftReturnForInvoice(firmId: string, clientId: string, returnType: string, period: string, taxableValue: number, taxAmount: number): Promise<void> {
   526→  try {
   527→    const q = query(
   528→      collection(db, COLLECTIONS.RETURNS),
   529→      where('clientId', '==', clientId),
   530→      where('returnType', '==', returnType),
   531→      where('period', '==', period),
   532→      limit(1)
   533→    );
   534→    const snap = await getDocs(q);
   535→    if (!snap.empty) {
   536→      const returnDoc = snap.docs[0];
   537→      const data = returnDoc.data() as FirestoreReturn;
   538→      await updateDoc(returnDoc.ref, {
   539→        totalInvoices: (data.totalInvoices || 0) + 1,
   540→        totalTaxableValue: (data.totalTaxableValue || 0) + taxableValue,
   541→        totalTax: (data.totalTax || 0) + taxAmount,
   542→        updatedAt: serverTimestamp(),
   543→      });
   544→    }
   545→  } catch (e) {
   546→    console.warn('[Workflow] Failed to update draft return:', e);
   547→  }
   548→}

   556→export async function createReconciliation(data: {
   557→  clientId: string;
   558→  period: string;
   559→  sources: string;
   560→}): Promise<string> {
   561→  const firmId = currentFirmId();
   562→  const reconId = generateId();
   563→  const reconRef = doc(db, COLLECTIONS.RECONCILIATIONS, reconId);
   564→
   565→  // Simulate reconciliation results
   566→  const totalRecords = Math.floor(Math.random() * 50) + 10;
   567→  const matched = Math.floor(totalRecords * (0.6 + Math.random() * 0.25));
   568→  const partialMatches = Math.floor((totalRecords - matched) * 0.4);
   569→  const unmatched = totalRecords - matched - partialMatches;
   570→  const highRisk = Math.floor(unmatched * 0.3);
   571→
   572→  const mismatches: ReconMismatch[] = [];
   573→  for (let i = 0; i < unmatched + partialMatches; i++) {
   574→    const booksAmount = Math.floor(Math.random() * 100000) + 5000;
   575→    const diff = Math.floor(booksAmount * (Math.random() * 0.15));
   576→    mismatches.push({
   577→      invoiceNumber: `INV-${String(i + 1).padStart(4, '0')}`,
   578→      invoiceDate: new Date().toISOString().split('T')[0],
   579→      sourceGstin: '27AAACR5055K1ZB',
   580→      matchedGstin: i < partialMatches ? '27AAACR5055K1ZB' : null,
   581→      matchStatus: i < partialMatches ? 'partial_match' as MatchStatus : 'missing_in_books' as MatchStatus,
   582→      matchScore: i < partialMatches ? 70 + Math.floor(Math.random() * 20) : 0,
   583→      booksAmount,
   584→      portalAmount: booksAmount + (Math.random() > 0.5 ? diff : -diff),
   585→      difference: diff,
   586→      reason: i < partialMatches ? 'Amount mismatch' : 'Invoice not found in GSTR-2B',
   587→      resolved: false,
   588→      resolvedBy: null,
   589→      resolvedAt: null,
   590→    });
   591→  }
   592→
   593→  const gstDifference = mismatches.reduce((sum, m) => sum + m.difference, 0);
   594→
   595→  const reconData: FirestoreReconciliation = {
   596→    reconId,
   597→    firmId: firmId || '',
   598→    clientId: data.clientId,
   599→    period: data.period,
   600→    sources: data.sources,
   601→    status: 'completed',
   602→    totalRecords,
   603→    matched,
   604→    unmatched,
   605→    partialMatches,
   606→    highRisk,
   607→    gstDifference,
   608→    mismatches,
   609→    runBy: currentUserId(),
   610→    createdAt: serverTimestamp(),
   611→    updatedAt: serverTimestamp(),
   612→  };
   613→
   614→  await setDoc(reconRef, reconData);
   615→
   616→  addActivity({
   617→    type: 'reconciliation_run',
   618→    title: 'Reconciliation completed',
   619→    description: `${data.sources}: ${matched}/${totalRecords} matched, ${unmatched} unmatched`,
   620→    clientId: data.clientId,
   621→    entityType: COLLECTIONS.RECONCILIATIONS,
   622→    entityId: reconId,
   623→    metadata: { matched, unmatched, gstDifference },
   624→  });
   625→
   626→  // Generate AI recommendations for mismatches
   627→  if (highRisk > 0) {
   628→    generateAIRecommendations(firmId || '', data.clientId, reconId, mismatches);
   629→  }
   630→
   631→  addNotification({
   632→    type: 'mismatch_found',
   633→    priority: highRisk > 3 ? 'urgent' : 'high',
   634→    title: 'Reconciliation mismatches found',
   635→    message: `${unmatched} unmatched and ${partialMatches} partial matches found. ITC difference: ₹${gstDifference.toLocaleString('en-IN')}`,
   636→    entityType: COLLECTIONS.RECONCILIATIONS,
   637→    entityId: reconId,
   638→  });
   639→
   640→  return reconId;
   641→}

   880→/** Subscribe to a Firestore collection with firm scoping */
   881→export function subscribeToCollection<T>(
   882→  collectionName: CollectionName,
   883→  callback: (data: Array<T & { id: string }>) => void,
   884→  ...constraints: QueryConstraint[]
   885→): Unsubscribe {
   886→  const firmId = currentFirmId();
   887→
   888→  const q = firmId
   889→    ? query(collection(db, collectionName), where('firmId', '==', firmId), ...constraints)
   890→    : query(collection(db, collectionName), ...constraints);
   891→
   892→  return onSnapshot(q, (snap) => {
   893→    const data = snap.docs.map(d => {
   894→      const docData = d.data();
   895→      const converted: Record<string, unknown> = { id: d.id };
   896→      for (const [key, value] of Object.entries(docData)) {
   897→        if (value && typeof value === 'object' && 'toDate' in value && typeof (value as { toDate: () => Date }).toDate === 'function') {
   898→          converted[key] = (value as { toDate: () => Date }).toDate().toISOString();
   899→        } else {
   900→          converted[key] = value;
   901→        }
   902→      }
   903→      return converted as T & { id: string };
   904→    });
   905→    callback(data);
   906→  }, (error) => {
   907→    console.warn(`[Firestore] Subscription error for ${collectionName}:`, error);
   908→    callback([]);
   909→  });
   910→}

   912→/** Subscribe to a single Firestore document */
   913→export function subscribeToDoc<T>(
   914→  collectionName: CollectionName,
   915→  docId: string,
   916→  callback: (data: (T & { id: string }) | null) => void,
   917→): Unsubscribe {
   918→  return onSnapshot(doc(db, collectionName, docId), (snap) => {
   919→    if (!snap.exists()) {
   920→      callback(null);
   921→      return;
   922→    }
   923→    const docData = snap.data();
   924→    const converted: Record<string, unknown> = { id: snap.id };
   925→    for (const [key, value] of Object.entries(docData)) {
   926→      if (value && typeof value === 'object' && 'toDate' in value && typeof (value as { toDate: () => Date }).toDate === 'function') {
   927→        converted[key] = (value as { toDate: () => Date }).toDate().toISOString();
   928→      } else {
   929→        converted[key] = value;
   930→      }
   931→    }
   932→    callback(converted as T & { id: string });
   933→  }, (error) => {
   934→    console.warn(`[Firestore] Doc subscription error for ${collectionName}/${docId}:`, error);
   935→    callback(null);
   936→  });
   937→}

   939→/** Compute live dashboard metrics from Firestore data */
   940→export function computeDashboardMetrics(
   941→  clients: Array<FirestoreClient & { id: string }>,
   942→  invoices: Array<FirestoreInvoice & { id: string }>,
   943→  returns: Array<FirestoreReturn & { id: string }>,
   944→  documents: Array<FirestoreDocument & { id: string }>,
   945→  activities: Array<FirestoreActivity & { id: string }>,
   946→): LiveDashboardMetrics {
   947→  const activeClients = clients.filter(c => c.status === 'active').length;
   948→  const filedReturns = returns.filter(r => r.status === 'filed').length;
   949→  const pendingReturns = returns.filter(r => r.status !== 'filed').length;
   950→  const readyToFile = returns.filter(r => ['validated', 'reviewed', 'generated'].includes(r.status)).length;
   951→  const overdueReturns = returns.filter(r => {
   952→    if (r.status === 'filed') return false;
   953→    const periodDate = parsePeriod(r.period);
   954→    return periodDate && periodDate < new Date();
   955→  }).length;
   956→
   957→  const totalTaxVolume = invoices.reduce((sum, inv) => sum + (inv.totalAmount || 0), 0);
   958→  const totalTax = invoices.reduce((sum, inv) => sum + (inv.cgst || 0) + (inv.sgst || 0) + (inv.igst || 0) + (inv.cess || 0), 0);
   959→
   960→  const criticalIssues = invoices.filter(inv => inv.riskLevel === 'critical').length +
   961→    returns.filter(r => r.criticalErrors > 0).length;
   962→  const warnings = invoices.filter(inv => inv.riskLevel === 'high' || inv.riskLevel === 'medium').length +
   963→    returns.filter(r => r.warnings > 0).length;
   964→
   965→  const averageHealthScore = clients.length > 0
   966→    ? Math.round(clients.reduce((sum, c) => sum + c.healthScore, 0) / clients.length)
   967→    : 0;
   968→
   969→  const matchedInvoices = invoices.filter(inv => inv.matchStatus === 'perfect_match').length;
   970→  const matchPercentage = invoices.length > 0 ? Math.round((matchedInvoices / invoices.length) * 100) : 100;
   971→
   972→  const riskyInvoices = invoices.filter(inv => ['high', 'critical'].includes(inv.riskLevel)).length;
   973→  const riskPercentage = invoices.length > 0 ? Math.round((riskyInvoices / invoices.length) * 100) : 0;
   974→
   975→  const documentsProcessed = documents.filter(d => d.extractionStatus === 'completed').length;
   976→  const extractionsPending = documents.filter(d => d.extractionStatus === 'pending' || d.extractionStatus === 'in_progress').length;
   977→
   978→  const recentActivities = [...activities]
   979→    .sort((a, b) => {
   980→      const aTime = typeof a.createdAt === 'string' ? new Date(a.createdAt).getTime() : 0;
   981→      const bTime = typeof b.createdAt === 'string' ? new Date(b.createdAt).getTime() : 0;
   982→      return bTime - aTime;
   983→    })
   984→    .slice(0, 10);
   985→
   986→  const upcomingFilings = returns
   987→    .filter(r => r.status !== 'filed')
   988→    .sort((a, b) => {
   989→      const aTime = typeof a.createdAt === 'string' ? new Date(a.createdAt).getTime() : 0;
   990→      const bTime = typeof b.createdAt === 'string' ? new Date(b.createdAt).getTime() : 0;
   991→      return aTime - bTime;
   992→    })
   993→    .slice(0, 5);
   994→
   995→  return {
   996→    totalClients: clients.length,
   997→    activeClients,
   998→    totalInvoices: invoices.length,
   999→    totalTaxVolume,
  1000→    filedReturns,
  1001→    pendingReturns,
  1002→    overdueReturns,
  1003→    readyToFile,
  1004→    criticalIssues,
  1005→    warnings,
  1006→    averageHealthScore,
  1007→    matchPercentage,
  1008→    riskPercentage,
  1009→    documentsProcessed,
  1010→    extractionsPending,
  1011→    recentActivities,
  1012→    upcomingFilings,
  1013→  };
  1014→}

  1016→function parsePeriod(period: string | null): Date | null {
  1017→  if (!period) return null;
  1018→  try {
  1019→    const [mm, yyyy] = period.split('-').map(Number);
  1020→    if (!mm || !yyyy) return null;
  1021→    return new Date(yyyy, mm, 20); // Due date is typically 20th of next month
  1022→  } catch {
  1023→    return null;
  1024→  }
  1025→}
