    69→    69→function getDaysRemaining(dueDateStr: string): number {
    70→    70→  const dueDate = new Date(dueDateStr);
    71→    71→  const today = new Date();
    72→    72→  today.setHours(0, 0, 0, 0);
    73→    73→  dueDate.setHours(0, 0, 0, 0);
    74→    74→  return Math.ceil((dueDate.getTime() - today.getTime()) / (1000 * 60 * 60 * 24));
    75→    75→}
    76→    76→
    77→    77→function formatDaysRemaining(days: number): string {
    78→    78→  if (days < 0) return `${Math.abs(days)}d overdue`;
    79→    79→  if (days === 0) return 'Due today';
    80→    80→  if (days === 1) return 'Tomorrow';
    81→    81→  return `${days}d left`;
    82→    82→}
    83→    83→
    84→    84→function timeAgo(dateStr: string): string {
    85→    85→  if (!dateStr) return '';
    86→    86→  const now = new Date();
    87→    87→  const date = new Date(dateStr);
    88→    88→  const seconds = Math.floor((now.getTime() - date.getTime()) / 1000);
    89→    89→  if (seconds < 60) return 'just now';
    90→    90→  const minutes = Math.floor(seconds / 60);
    91→    91→  if (minutes < 60) return `${minutes}m ago`;
    92→    92→  const hours = Math.floor(minutes / 60);
    93→    93→  if (hours < 24) return `${hours}h ago`;
    94→    94→  const days = Math.floor(hours / 24);
    95→    95→  if (days < 7) return `${days}d ago`;
    96→    96→  return date.toLocaleDateString('en-IN', { day: 'numeric', month: 'short' });
    97→    97→}
