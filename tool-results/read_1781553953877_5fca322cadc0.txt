    46→// ─── Utility ────────────────────────────────────────────────────────────────────
    47→
    48→function timeAgo(dateStr: string | null | undefined): string {
    49→  if (!dateStr) return 'Never'
    50→  const d = new Date(dateStr)
    51→  const now = new Date()
    52→  const diffMs = now.getTime() - d.getTime()
    53→  const diffMins = Math.floor(diffMs / 60000)
    54→  if (diffMins < 1) return 'Just now'
    55→  if (diffMins < 60) return `${diffMins}m ago`
    56→  const diffHrs = Math.floor(diffMins / 60)
    57→  if (diffHrs < 24) return `${diffHrs}h ago`
    58→  const diffDays = Math.floor(diffHrs / 24)
    59→  if (diffDays < 7) return `${diffDays}d ago`
    60→  return d.toLocaleDateString('en-IN', { day: 'numeric', month: 'short' })
    61→}
    62→
    63→function formatDate(dateStr: string | null | undefined): string {
    64→  if (!dateStr) return '—'
    65→  return new Date(dateStr).toLocaleDateString('en-IN', { day: '2-digit', month: 'short', year: 'numeric' })
    66→}
