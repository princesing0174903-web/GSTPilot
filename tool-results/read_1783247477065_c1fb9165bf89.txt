   124→// ─── Circular Performance Score ────────────────────────────────────────────
   125→function PerformanceCircle({
   126→  score,
   127→  size = 48,
   128→  strokeWidth = 5,
   129→}: {
   130→  score: number;
   131→  size?: number;
   132→  strokeWidth?: number;
   133→}) {
   134→  const radius = (size - strokeWidth) / 2;
   135→  const circumference = 2 * Math.PI * radius;
   136→  const offset = circumference - (score / 100) * circumference;
   137→  const center = size / 2;
   138→
   139→  const getColor = (s: number) => {
   140→    if (s >= 90) return COLORS.emerald;
   141→    if (s >= 80) return COLORS.teal;
   142→    if (s >= 70) return COLORS.amber;
   143→    return COLORS.red;
   144→  };
   145→
   146→  return (
   147→    <div className="relative" style={{ width: size, height: size }}>
   148→      <svg width={size} height={size} className="transform -rotate-90">
   149→        <circle
   150→          cx={center}
   151→          cy={center}
   152→          r={radius}
   153→          fill="none"
   154→          stroke="currentColor"
   155→          strokeWidth={strokeWidth}
   156→          className="text-muted/20"
   157→        />
   158→        <circle
   159→          cx={center}
   160→          cy={center}
   161→          r={radius}
   162→          fill="none"
   163→          stroke={getColor(score)}
   164→          strokeWidth={strokeWidth}
   165→          strokeDasharray={circumference}
   166→          strokeDashoffset={offset}
   167→          strokeLinecap="round"
   168→          className="transition-all duration-700 ease-out"
   169→          style={{ filter: `drop-shadow(0 0 3px ${getColor(score)}40)` }}
   170→        />
   171→      </svg>
   172→      <div className="absolute inset-0 flex items-center justify-center">
   173→        <span className="text-[11px] font-bold text-foreground">{Math.round(score)}</span>
   174→      </div>
   175→    </div>
   176→  );
   177→}
