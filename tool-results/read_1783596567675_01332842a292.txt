     3→//
     4→// Every tool here performs a REAL business operation against live data.
     5→// No simulations. No placeholders. If a tool says "create invoice", an invoice
     6→// is written to the invoices collection. If it says "send reminder", a real
     7→// notification + activity record is created.

    10→//   • id, name, description, category, icon
    11→//   • inputSchema           — typed fields for the UI form
    12→//   • approvalRequired      — critical actions never auto-execute
    13→//   • detect                — NL intent matcher (regex + keyword scoring)
    14→//   • extractParams         — pull params from live business data (clients, invoices)
    15→//   • dryRun                — safe preview, no writes
    16→//   • execute               — REAL write to Firestore via adminDb()
    17→//   • rollback              — undo on failure (best-effort)
    18→//   • retry                 — { maxAttempts, backoffMs }
    19→//   • auditMeta             — what to record in the audit trail

   126→async function fetchCollection(
   127→  collectionName: string,
   128→  organizationId: string,
   129→  limitCount = 50,
   130→): Promise<Array<Record<string, unknown>>> {
   131→  try {
   132→    const q = query(
   133→      collection(db, collectionName),
   134→      where('organizationId', '==', organizationId),
   135→      limit(limitCount),
   136→    );
   137→    const snap = await getDocs(q);
   138→    return snap.docs.map((d) => ({ id: d.id, ...d.data() }));
   139→  } catch {
   140→    return [];
   141→  }
   142→}

   144→// ─── Live Business Data Loader ──────────────────────────────────────────────
   145→
   146→export async function loadLiveBusinessData(organizationId: string): Promise<LiveBusinessData> {
   147→  const [clientsRaw, invoicesRaw, returnsRaw, gstRaw] = await Promise.all([
   148→    fetchCollection(COLLECTIONS.CLIENTS, organizationId, 100),
   149→    fetchCollection(COLLECTIONS.INVOICES, organizationId, 100),
   150→    fetchCollection(COLLECTIONS.GST_RETURNS, organizationId, 50),
   151→    fetchCollection(COLLECTIONS.GST_PROFILES, organizationId, 5),
   152→  ]);
   153→
   154→  const clients: LiveBusinessData['clients'] = clientsRaw.map((c) => ({
   155→    id: String(c.id ?? c.clientId ?? ''),
   156→    name: String(c.name ?? c.clientName ?? c.displayName ?? 'Unknown'),
   157→    gstin: (c.gstin as string) ?? null,
   158→    email: (c.email as string) ?? null,
   159→    phone: (c.phone as string) ?? null,
   160→  }));
   161→
   162→  const recentInvoices: LiveBusinessData['recentInvoices'] = invoicesRaw
   163→    .map((i) => ({
   164→      id: String(i.id ?? i.invoiceId ?? ''),
   165→      invoiceNumber: String(i.invoiceNumber ?? ''),
   166→      clientName: String(i.buyerName ?? i.clientName ?? ''),
   167→      totalAmount: Number(i.totalAmount ?? i.grandTotal ?? 0),
   168→      status: String(i.status ?? 'draft'),
   169→    }))
   170→    .slice(0, 20);
   171→
   172→  const now = Date.now();
   173→  const overdueInvoices: LiveBusinessData['overdueInvoices'] = invoicesRaw
   174→    .filter((i) => {
   175→      const status = String(i.status ?? '').toLowerCase();
   176→      const due = i.dueDate ? new Date(i.dueDate as string).getTime() : 0;
   177→      return status !== 'paid' && status !== 'cancelled' && due > 0 && due < now;
   178→    })
   179→    .map((i) => {
   180→      const due = new Date(i.dueDate as string).getTime();
   181→      return {
   182→        id: String(i.id ?? i.invoiceId ?? ''),
   183→        invoiceNumber: String(i.invoiceNumber ?? ''),
   184→        clientName: String(i.buyerName ?? i.clientName ?? ''),
   185→        clientId: String(i.clientId ?? ''),
   186→        totalAmount: Number(i.totalAmount ?? i.grandTotal ?? 0),
   187→        daysOverdue: Math.floor((now - due) / (24 * 60 * 60 * 1000)),
   188→      };
   189→    })
   190→    .sort((a, b) => b.daysOverdue - a.daysOverdue)
   191→    .slice(0, 20);
   192→
   193→  const pendingReturns: LiveBusinessData['pendingReturns'] = returnsRaw
   194→    .filter((r) => {
   195→      const status = String(r.status ?? '').toLowerCase();
   196→      return status === 'pending' || status === 'draft' || status === 'ready';
   197→    })
   198→    .map((r) => ({
   199→      id: String(r.id ?? r.returnId ?? ''),
   200→      returnType: String(r.returnType ?? r.formType ?? 'GSTR-3B'),
   201→      period: String(r.period ?? r.taxPeriod ?? ''),
   202→      dueDate: String(r.dueDate ?? ''),
   203→      status: String(r.status ?? 'pending'),
   204→    }));
   205→
   206→  const gstProfile =
   207→    gstRaw.length > 0
   208→      ? {
   209→          gstin: (gstRaw[0].gstin as string) ?? null,
   210→          legalName: (gstRaw[0].legalName as string) ?? (gstRaw[0].tradeName as string) ?? null,
   211→        }
   212→      : null;
   213→
   214→  return { clients, recentInvoices, overdueInvoices, pendingReturns, gstProfile };
   215→}

   217→// ─── Helper: retry with exponential backoff ─────────────────────────────────
   218→
   219→async function withRetry<T>(
   220→  fn: () => Promise<T>,
   221→  policy: { maxAttempts: number; backoffMs: number },
   222→): Promise<T> {
   223→  let lastError: unknown;
   224→  for (let attempt = 1; attempt <= policy.maxAttempts; attempt++) {
   225→    try {
   226→      return await fn();
   227→    } catch (err) {
   228→      lastError = err;
   229→      if (attempt < policy.maxAttempts) {
   230→        await new Promise((r) => setTimeout(r, policy.backoffMs * attempt));
   231→      }
   232→    }
   233→  }
   234→  throw lastError;
   235→}

   239→function genId(prefix: string): string {
   240→  return `${prefix}_${Date.now()}_${Math.random().toString(36).slice(2, 8)}`;
   241→}

   243→// ─── Helper: interpret Firestore errors into user-friendly messages ─────────
   244→// The tool executors catch errors and return a ToolResult. This helper
   245→// translates raw Firebase error codes (permission-denied, unavailable, etc.)
   246→// into actionable messages so the UI never shows a raw stack trace.
   247→
   248→function interpretError(err: unknown, actionLabel: string): { message: string; error: string } {
   249→  const raw = err instanceof Error ? err.message : String(err);
   250→  const lower = raw.toLowerCase();
   251→  if (lower.includes('permission-denied') || lower.includes('missing or insufficient permissions')) {
   252→    return {
   253→      message: `${actionLabel} was validated successfully, but could not be saved because you are in preview mode. Sign in to persist this action to your live business data.`,
   254→      error: 'PREVIEW_MODE: Firestore security rules require authentication. The action logic ran correctly — sign in to execute for real.',
   255→    };
   256→  }
   257→  if (lower.includes('unavailable') || lower.includes('offline')) {
   258→    return {
   259→      message: `${actionLabel} could not complete because the database is temporarily unavailable. Please retry in a moment.`,
   260→      error: raw,
   261→    };
   262→  }
   263→  return {
   264→    message: `${actionLabel} failed after retries. The error has been logged. Please try again or contact support if it persists.`,
   265→    error: raw,
   266→  };
   267→}
