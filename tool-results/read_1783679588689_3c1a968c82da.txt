   245→   245→function AnalysisBody({ text, isStreaming }: { text: string; isStreaming: boolean }) {
   246→   246→  // Split into paragraphs by \n\n, then within each paragraph detect bullet lines.
   247→   247→  const paragraphs = text.split('\n\n').filter((p) => p.trim().length > 0);
   248→   248→
   249→   249→  return (
   250→   250→    <div
   251→   251→      className={cn(
   252→   252→        'space-y-3 text-sm leading-relaxed text-foreground/90',
   253→   253→        isStreaming && 'typing-cursor',
   254→   254→      )}
   255→   255→    >
   256→   256→      {paragraphs.map((para, pi) => {
   257→   257→        const lines = para.split('\n');
   258→   258→        const bulletLines: string[] = [];
   259→   259→        const proseLines: string[] = [];
   260→   260→        let currentMode: 'prose' | 'bullet' | null = null;
   261→   261→
   262→   262→        for (const line of lines) {
   263→   263→          const trimmed = line.trim();
   264→   264→          const isBullet =
   265→   265→            trimmed.startsWith('• ') ||
   266→   266→            trimmed.startsWith('- ') ||
   267→   267→            trimmed.startsWith('* ');
   268→   268→          if (isBullet) {
   269→   269→            if (currentMode === 'prose' && proseLines.length > 0) {
   270→   270→              // Flush prose before bullets.
   271→   271→            }
   272→   272→            bulletLines.push(trimmed.replace(/^[•\-*]\s+/, ''));
   273→   273→            currentMode = 'bullet';
   274→   274→          } else if (trimmed.length > 0) {
   275→   275→            proseLines.push(line);
   276→   276→            currentMode = 'prose';
   277→   277→          }
   278→   278→        }
   279→   279→
   280→   280→        return (
   281→   281→          <div key={pi} className="space-y-1.5">
   282→   282→            {proseLines.length > 0 && (
   283→   283→              <p>{proseLines.join(' ')}</p>
   284→   284→            )}
   285→   285→            {bulletLines.length > 0 && (
   286→   286→              <ul className="ml-1 space-y-1">
   287→   287→                {bulletLines.map((b, bi) => (
   288→   288→                  <motion.li
   289→   289→                    key={bi}
   290→   290→                    initial={{ opacity: 0, x: -4 }}
   291→   291→                    animate={{ opacity: 1, x: 0 }}
   292→   292→                    transition={{ duration: 0.25, delay: bi * 0.04 }}
   293→   293→                    className="flex items-start gap-2 text-foreground/85"
   294→   294→                  >
   295→   295→                    <span className="mt-1.5 h-1 w-1 shrink-0 rounded-full bg-accent-start" />
   296→   296→                    <span>{b}</span>
   297→   297→                  </motion.li>
   298→   298→                ))}
   299→   299→              </ul>
   300→   300→            )}
   301→   301→          </div>
   302→   302→        );
   303→   303→      })}
   304→   304→    </div>
   305→   305→  );
   306→   306→}

   327→   327→
   328→   328→export function OracleMessageView({
   329→   329→  message,
   330→   330→  onFollowUp,
   331→   331→  onAction,
   332→   332→  onNextBestStep,
   333→   333→  onRunAction,
   334→   334→  actionsDisabled,
   335→   335→  onRegenerate,
   336→   336→  onSpeak,
   337→   337→  isSpeaking,
   338→   338→}: OracleMessageProps) {
   339→   339→  if (message.role === 'user') {
   340→   340→    return (
   341→   341→      <UserMessage
   342→   342→        content={message.content}
   343→   343→        mode={message.mode}
   344→   344→        language={message.language}
   345→   345→        emotion={message.emotion}
   346→   346→      />
   347→   347→    );
   348→   348→  }
   349→   349→
   350→   350→  const phase = message.phase ?? 'done';
   351→   351→  const isPreStream =
   352→   352→    phase === 'thinking' ||
   353→   353→    phase === 'reading' ||
   354→   354→    phase === 'analyzing' ||
   355→   355→    phase === 'preparing';
   356→   356→  const isError = phase === 'error';
   357→   357→  const parts = message.parts;
   358→   358→  const mode = message.mode ? ORACLE_MODES[message.mode] : undefined;
   359→   359→  const isStreaming = phase === 'streaming';
   360→   360→  const isDone = phase === 'done';
   361→   361→
   362→   362→  // Build a plain-text version for copy / export / speak.
   363→   363→  const plainText = parts
   364→   364→    ? [
   365→   365→        parts.keyInsight ? `KEY INSIGHT\n${parts.keyInsight}` : '',
   366→   366→        parts.analysis ? `\nANALYSIS\n${parts.analysis}` : '',
   367→   367→        parts.recommendedActions.length > 0
   368→   368→          ? `\nRECOMMENDED ACTIONS\n${parts.recommendedActions.map((a, i) => `${i + 1}. ${a}`).join('\n')}`
   369→   369→          : '',
   370→   370→        parts.potentialRisks.length > 0
   371→   371→          ? `\nPOTENTIAL RISKS\n${parts.potentialRisks.map((r) => `• ${r}`).join('\n')}`
   372→   372→          : '',
   373→   373→        parts.nextBestStep ? `\nNEXT BEST STEP\n${parts.nextBestStep}` : '',
   374→   374→      ]
   375→   375→        .filter(Boolean)
   376→   376→        .join('\n')
   377→   377→    : message.content;
   378→   378→
   379→   379→  const handleCopy = () => {
   380→   380→    if (!plainText) return;
   381→   381→    navigator.clipboard?.writeText(plainText).catch(() => {});
   382→   382→  };
   383→   383→
   384→   384→  const handleExport = () => {
   385→   385→    if (!plainText) return;
   386→   386→    const blob = new Blob([plainText], { type: 'text/plain;charset=utf-8' });
   387→   387→    const url = URL.createObjectURL(blob);
   388→   388→    const a = document.createElement('a');
   389→   389→    a.href = url;
   390→   390→    a.download = `oracle-response-${new Date(message.createdAt).getTime()}.txt`;
   391→   391→    document.body.appendChild(a);
   392→   392→    a.click();
   393→   393→    document.body.removeChild(a);
   394→   394→    URL.revokeObjectURL(url);
   395→   395→  };
   396→   396→
   397→   397→  return (
   398→   398→    <div className="flex gap-3">
   399→   399→      {/* Oracle avatar — dynamic facial expressions */}
   400→   400→      <div className="shrink-0">
   401→   401→        <OracleAvatar
   402→   402→          state={deriveAvatarState(phase, message.emotion)}
   403→   403→          size={40}
   404→   404→        />
   405→   405→      </div>
   406→   406→
   407→   407→      {/* Body */}
   408→   408→      <div className="min-w-0 flex-1 space-y-4">
   409→   409→        {/* Mode badge + status row */}
   410→   410→        <div className="flex items-center gap-2">
   411→   411→          {mode && <ModeBadge mode={mode.id} />}
   412→   412→          {isStreaming && (
   413→   413→            <span className="flex items-center gap-1.5 text-[10px] font-medium text-muted-foreground">
   414→   414→              <span className="relative flex h-1.5 w-1.5">
   415→   415→                <span className="absolute inline-flex h-full w-full animate-ping rounded-full bg-emerald-400 opacity-75" />
   416→   416→                <span className="relative inline-flex h-1.5 w-1.5 rounded-full bg-emerald-400" />
   417→   417→              </span>
   418→   418→              Oracle is responding
   419→   419→            </span>
   420→   420→          )}
   421→   421→        </div>
   422→   422→
   423→   423→        {/* Error state */}
   424→   424→        {isError && (
   425→   425→          <div className="rounded-2xl border border-red-500/30 bg-red-500/5 p-4 text-sm text-red-600 dark:text-red-400">
   426→   426→            <p className="font-medium">Oracle hit a snag.</p>
   427→   427→            <p className="mt-1 text-xs opacity-80">{message.error ?? 'Unknown error'}</p>
   428→   428→          </div>
   429→   429→        )}
   430→   430→
   431→   431→        {/* Pre-stream phase indicator */}
   432→   432→        {isPreStream && <PhaseIndicator phase={phase} />}
   433→   433→
   434→   434→        {/* Structured parts — rendered progressively during streaming */}
   435→   435→        {!isError && !isPreStream && parts && (
   436→   436→          <div className="space-y-5">
   437→   437→            {/* 1. Key Insight — the headline */}
   438→   438→            {parts.keyInsight && (
   439→   439→              <Section icon={Lightbulb} title="Key Insight" delay={0}>
   440→   440→                <div
   441→   441→                  className={cn(
   442→   442→                    'rounded-2xl border-l-[3px] border-[color-mix(in_srgb,var(--accent-start)_70%,transparent)] bg-gradient-to-r from-[color-mix(in_srgb,var(--accent-start)_8%,transparent)] to-transparent px-4 py-3 text-[15px] font-semibold leading-snug text-foreground',
   443→   443→                    isStreaming && 'typing-cursor',
   444→   444→                  )}
   445→   445→                >
   446→   446→                  {parts.keyInsight}
   447→   447→                </div>
   448→   448→              </Section>
   449→   449→            )}
   450→   450→
   451→   451→            {/* 2. Analysis — the strategic reasoning */}
   452→   452→            {parts.analysis && (
   453→   453→              <Section icon={Compass} title="Analysis" delay={0.05}>
   454→   454→                <AnalysisBody text={parts.analysis} isStreaming={isStreaming} />
   455→   455→              </Section>
   456→   456→            )}

   551→   551→                      key={i}

   554→   554→                      transition={{ duration: 0.25, delay: i * 0.05 }}
   555→   555→                      whileHover={{ x: 2 }}
   556→   556→                      onClick={() => onFollowUp?.(q)}
   557→   557→                      disabled={!isDone}
   558→   558→                      className="group flex items-center gap-2 rounded-xl border border-border bg-card/[0.4] px-3.5 py-2 text-left text-sm text-foreground/90 transition-colors hover:border-[color-mix(in_srgb,var(--accent-start)_40%,transparent)] hover:bg-card/[0.7] disabled:cursor-not-allowed disabled:opacity-50"
   559→   559→                    >
   560→   560→                      <Sparkles className="h-3 w-3 accent-text" />
   561→   561→                      {q}
   562→   562→                    </motion.button>
   563→   563→                  ))}
   564→   564→                </div>
   565→   565→              </motion.div>
   566→   566→            )}
   567→   567→
   568→   568→            {/* 6. Sources */}
   569→   569→            {parts.sources.length > 0 && (
   570→   570→              <Section icon={Quote} title="Sources" delay={0.25}>
   571→   571→                <div className="flex flex-wrap gap-2">
   572→   572→                  {parts.sources.map((s, i) => (
   573→   573→                    <SourceCard key={i} source={s} index={i} />
   574→   574→                  ))}
   575→   575→                </div>
   576→   576→              </Section>
   577→   577→            )}
   578→   578→
   579→   579→            {/* 7. Follow-up Questions */}
   580→   580→            {parts.followUps.length > 0 && (
   581→   581→              <Section icon={Sparkles} title="Follow-up Questions" delay={0.3}>
   582→   582→                <div className="flex flex-col gap-1.5">
   583→   583→                  {parts.followUps.map((q, i) => (
   584→   584→                    <motion.button
   585→   585→                      key={i}
   586→   586→                      initial={{ opacity: 0, y: 4 }}

   593→   593→                    >

   602→   602→            {/* 8. Phase 3 — Action Cards (3 large glass buttons, single-click execution) */}
   603→   603→            {parts.actionCards && parts.actionCards.length > 0 && (
   604→   604→              <motion.div
   605→   605→                initial={{ opacity: 0, y: 8 }}

   609→   609→                <OracleActionCards
   610→   610→                  cards={parts.actionCards}

   618→   618→
   619→   619→        {/* Done checkmark + action row */}
   620→   620→        {isDone && parts && (
   621→   621→          <motion.div
   622→   622→            initial={{ opacity: 0 }}
   623→   623→            animate={{ opacity: 1 }}
   624→   624→            transition={{ delay: 0.2 }}
   625→   625→            className="flex flex-wrap items-center gap-1.5"
   626→   626→          >
   627→   627→            <span className="flex items-center gap-1.5 text-[10px] text-muted-foreground">
   628→   628→              <CheckCircle2 className="h-3 w-3 text-emerald-500" />

   632→   632→            {/* Action row — copy / speak / regenerate / export */}
   633→   633→            <div className="ml-auto flex items-center gap-0.5">
   634→   634→              <MsgActionButton
   635→   635→                icon={Copy}

   638→   638→              />

   641→   641→                  icon={Volume2}

   666→   666→}
