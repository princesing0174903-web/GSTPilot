    65→function getDaysRemaining(dueDateStr: string): number {
    66→  const dueDate = new Date(dueDateStr);
    67→  const today = new Date();
    68→  today.setHours(0, 0, 0, 0);
    69→  dueDate.setHours(0, 0, 0, 0);
    70→  return Math.ceil((dueDate.getTime() - today.getTime()) / (1000 * 60 * 60 * 24));
    71→}
    72→
    73→function formatDaysRemaining(days: number): string {
    74→  if (days < 0) return `${Math.abs(days)}d overdue`;
    75→  if (days === 0) return 'Due today';
    76→  if (days === 1) return 'Tomorrow';
    77→  return `${days}d left`;
    78→}
    79→
    80→function formatINR(amount: number): string {
    81→  return amount.toLocaleString('en-IN');
    82→}
    83→
    84→function formatDateIN(dateStr: string): string {
    85→  const d = new Date(dateStr);
    86→  if (Number.isNaN(d.getTime())) return '—';
    87→  const dd = String(d.getDate()).padStart(2, '0');
    88→  const mm = String(d.getMonth() + 1).padStart(2, '0');
    89→  const yyyy = d.getFullYear();
    90→  return `${dd}/${mm}/${yyyy}`;
    91→}
    92→
    93→function timeAgo(dateStr: string): string {
    94→  const now = new Date();
    95→  const date = new Date(dateStr);
    96→  const seconds = Math.floor((now.getTime() - date.getTime()) / 1000);
    97→  if (seconds < 60) return 'just now';
    98→  const minutes = Math.floor(seconds / 60);
    99→  if (minutes < 60) return `${minutes}m ago`;
   100→  const hours = Math.floor(minutes / 60);
   101→  if (hours < 24) return `${hours}h ago`;
   102→  const days = Math.floor(hours / 24);
   103→  if (days < 7) return `${days}d ago`;
   104→  return date.toLocaleDateString('en-IN', { day: 'numeric', month: 'short' });
   105→}

   107→function getGreeting(): string {
   108→  const h = new Date().getHours();
   109→  if (h < 12) return 'Good Morning';
   110→  if (h < 17) return 'Good Afternoon';
   111→  return 'Good Evening';

   120→// One-sentence AI insight derived from live metrics.
   121→function buildInsightSentence(
   122→  metrics: LiveDashboardMetrics,
   123→  pendingCollection: number,
   124→): string {
   125→  const parts: string[] = [];
   126→  if (metrics.overdueReturns > 0) {
   127→    parts.push(
   128→      `${metrics.overdueReturns} overdue return${metrics.overdueReturns > 1 ? 's' : ''}`,
   129→    );
   130→  } else if (metrics.pendingReturns > 0) {
   131→    parts.push(
   132→      `${metrics.pendingReturns} return${metrics.pendingReturns > 1 ? 's' : ''} to file`,
   133→    );
   134→  }
   135→  if (pendingCollection > 0) {
   136→    parts.push(`₹${formatINR(pendingCollection)} pending collection`);
   137→  }
   138→  if (metrics.criticalIssues > 0) {
   139→    parts.push(
   140→      `${metrics.criticalIssues} critical issue${metrics.criticalIssues > 1 ? 's' : ''}`,
   141→    );
   142→  }
   143→  if (parts.length === 0) {
   144→    return `All clear — ${metrics.filedReturns} returns filed and ${metrics.totalClients} clients in good standing.`;
   145→  }
   146→  if (parts.length === 1) return `${parts[0]}.`;
   147→  const last = parts.pop();
   148→  return `${parts.join(', ')} and ${last}.`;
   149→}

   161→  if (type.includes('reconcil') || type.includes('mismatch'))

   164→    return <FileText className="h-3.5 w-3.5 accent-text" />;

   268→function healthTier(score: number): { label: string; tone: string } {
   269→  if (score >= 85) return { label: 'Excellent', tone: 'text-emerald-400' };
   270→  if (score >= 70) return { label: 'Healthy', tone: 'text-emerald-400' };
   271→  if (score >= 50) return { label: 'At Risk', tone: 'text-amber-400' };
   272→  return { label: 'Critical', tone: 'text-amber-400' };
   273→}
