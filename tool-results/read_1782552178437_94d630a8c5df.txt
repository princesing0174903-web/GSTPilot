    42→function formatINR(amount: number): string {
    43→  if (!isFinite(amount) || isNaN(amount)) return '₹0';
    44→  const abs = Math.abs(amount);
    45→  if (abs >= 10000000) return `₹${(amount / 10000000).toFixed(2)} Cr`;
    46→  if (abs >= 100000) return `₹${(amount / 100000).toFixed(2)} L`;
    47→  if (abs >= 1000) return `₹${(amount / 1000).toFixed(1)}K`;
    48→  return '₹' + Math.round(amount).toLocaleString('en-IN');
    49→}
    50→
    51→function formatINRFull(amount: number): string {
    52→  if (!isFinite(amount) || isNaN(amount)) return '₹0';
    53→  return '₹' + Math.round(amount).toLocaleString('en-IN');
    54→}

    61→function timeAgo(iso: string): string {
    62→  const diff = Date.now() - new Date(iso).getTime();
    63→  const s = Math.floor(diff / 1000);
    64→  if (s < 60) return 'just now';
    65→  const m = Math.floor(s / 60);
    66→  if (m < 60) return `${m}m ago`;
    67→  const h = Math.floor(m / 60);
    68→  if (h < 24) return `${h}h ago`;
    69→  return `${Math.floor(h / 24)}d ago`;
    70→}

    99→function useAnimatedNumber(target: number, duration = 1200) {
   100→  const [value, setValue] = useState(0);
   101→  useEffect(() => {
   102→    let raf = 0;
   103→    const start = performance.now();
   104→    const initial = 0;
   105→    const tick = (t: number) => {
   106→      const progress = Math.min((t - start) / duration, 1);
   107→      const eased = 1 - Math.pow(1 - progress, 3);
   108→      setValue(initial + (target - initial) * eased);
   109→      if (progress < 1) raf = requestAnimationFrame(tick);
   110→    };
   111→    raf = requestAnimationFrame(tick);
   112→    return () => cancelAnimationFrame(raf);
   113→  }, [target, duration]);
   114→  return value;
   115→}

   119→function Sparkline({ data, color = '#10b981', width = 100, height = 32 }: { data: number[]; color?: string; width?: number; height?: number }) {
   120→  if (data.length < 2) return <div style={{ width, height }} />;
   121→  const min = Math.min(...data);
   122→  const max = Math.max(...data);
   123→  const range = max - min || 1;
   124→  const pad = 2;
   125→  const points = data.map((v, i) => {
   126→    const x = pad + (i / (data.length - 1)) * (width - pad * 2);
   127→    const y = height - pad - ((v - min) / range) * (height - pad * 2);
   128→    return `${x},${y}`;
   129→  }).join(' ');
   130→  const area = `${pad},${height - pad} ${points} ${width - pad},${height - pad}`;
   131→  const id = `spark-${color.replace('#', '')}-${Math.round(Math.random() * 1e6)}`;
   132→  return (
   133→    <svg width={width} height={height} className="shrink-0">
   134→      <defs>
   135→        <linearGradient id={id} x1="0" y1="0" x2="0" y2="1">
   136→          <stop offset="0%" stopColor={color} stopOpacity={0.3} />
   137→          <stop offset="100%" stopColor={color} stopOpacity={0} />
   138→        </linearGradient>
   139→      </defs>
   140→      <polygon points={area} fill={`url(#${id})`} />
   141→      <polyline points={points} fill="none" stroke={color} strokeWidth={1.5} strokeLinecap="round" strokeLinejoin="round" />
   142→    </svg>
   143→  );
   144→}

   393→function HealthGauge({ score, size = 160 }: { score: number; size?: number }) {
   394→  const stroke = 12;
   395→  const r = (size - stroke) / 2;
   396→  const c = 2 * Math.PI * r;
   397→  const offset = c - (score / 100) * c;
   398→  const tier = score >= 80 ? 'Excellent' : score >= 65 ? 'Healthy' : score >= 50 ? 'Needs Attention' : 'At Risk';
   399→  const color = score >= 80 ? '#10b981' : score >= 65 ? '#06b6d4' : score >= 50 ? '#f59e0b' : '#ef4444';
   400→  return (
   401→    <div className="relative" style={{ width: size, height: size }}>
   402→      <svg width={size} height={size} className="-rotate-90">
   403→        <defs>
   404→          <linearGradient id="cfoHealthGrad" x1="0" y1="0" x2="1" y2="1">
   405→            <stop offset="0%" stopColor={color} />
   406→            <stop offset="100%" stopColor={color} stopOpacity={0.6} />
   407→          </linearGradient>
   408→        </defs>
   409→        <circle cx={size / 2} cy={size / 2} r={r} fill="none" stroke="rgba(255,255,255,0.05)" strokeWidth={stroke} />
   410→        <motion.circle
   411→          cx={size / 2}
   412→          cy={size / 2}
   413→          r={r}
   414→          fill="none"
   415→          stroke="url(#cfoHealthGrad)"
   416→          strokeWidth={stroke}
   417→          strokeLinecap="round"
   418→          strokeDasharray={c}
   419→          initial={{ strokeDashoffset: c }}
   420→          animate={{ strokeDashoffset: offset }}
   421→          transition={{ duration: 1.2, ease: 'easeOut' }}
   422→        />
   423→      </svg>
   424→      <div className="absolute inset-0 flex flex-col items-center justify-center">
   425→        <motion.span
   426→          initial={{ opacity: 0, scale: 0.7 }}
   427→          animate={{ opacity: 1, scale: 1 }}
   428→          transition={{ delay: 0.4, duration: 0.4 }}
   429→          className="text-4xl font-bold"
   430→          style={{ color }}
   431→        >
   432→          {score}
   433→        </motion.span>
   434→        <span className="text-[10px] uppercase tracking-wider text-muted-foreground">/ 100</span>
   435→        <span className="mt-1 text-xs font-medium" style={{ color }}>{tier}</span>
   436→      </div>
   437→    </div>
   438→  );
   439→}
