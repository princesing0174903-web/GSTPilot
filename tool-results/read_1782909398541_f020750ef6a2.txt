    47→const inr = (n: number): string => {
    48→  if (!Number.isFinite(n)) return '₹0';
    49→  const abs = Math.abs(n);
    50→  if (abs >= 1_00_00_000) return `₹${(n / 1_00_00_000).toFixed(2)}Cr`;
    51→  if (abs >= 1_00_000) return `₹${(n / 1_00_000).toFixed(2)}L`;
    52→  if (abs >= 1_000) return `₹${(n / 1_000).toFixed(1)}K`;
    53→  return `₹${n.toFixed(0)}`;
    54→};
    55→
    56→const fmtMs = (ms: number): string => {
    57→  if (!Number.isFinite(ms) || ms <= 0) return '0ms';
    58→  if (ms < 1_000) return `${Math.round(ms)}ms`;
    59→  if (ms < 60_000) return `${(ms / 1_000).toFixed(1)}s`;
    60→  return `${(ms / 60_000).toFixed(1)}m`;
    61→};
    62→
    63→const fmtPct = (n: number, decimals = 1): string => `${n.toFixed(decimals)}%`;
    64→
    65→const timeAgo = (iso: string | null): string => {
    66→  if (!iso) return '—';
    67→  const t = Date.parse(iso);
    68→  if (!Number.isFinite(t)) return '—';
    69→  const diff = Date.now() - t;
    70→  if (diff < 0) return 'just now';
    71→  const s = Math.floor(diff / 1_000);
    72→  if (s < 60) return `${s}s ago`;
    73→  const m = Math.floor(s / 60);
    74→  if (m < 60) return `${m}m ago`;
    75→  const h = Math.floor(m / 60);
    76→  if (h < 24) return `${h}h ago`;
    77→  const d = Math.floor(h / 24);
    78→  return `${d}d ago`;
    79→};
    80→
    81→const fmtTime = (iso: string | null): string => {
    82→  if (!iso) return '—';
    83→  const t = Date.parse(iso);
    84→  if (!Number.isFinite(t)) return '—';
    85→  return new Date(t).toLocaleString('en-IN', {
    86→    day: '2-digit', month: 'short', hour: '2-digit', minute: '2-digit', second: '2-digit',
    87→  });
    88→};

   590→function categoriseEvent(entry: TimelineEntry): { label: string; icon: LucideIcon; color: string } {
   591→  const cat = entry.type;
   592→  if (/fail|error|cancel/i.test(cat)) return { label: 'Failure', icon: AlertTriangle, color: 'text-rose-600' };
   593→  if (/complet|success/i.test(cat)) return { label: 'Completion', icon: CheckCircle2, color: 'text-emerald-600' };
   594→  if (/start|run|execut/i.test(cat)) return { label: 'Execution', icon: Play, color: 'text-cyan-600' };
   595→  if (/queue|schedul/i.test(cat)) return { label: 'Queue', icon: Layers, color: 'text-slate-600' };
   596→  if (/approv/i.test(cat)) return { label: 'Approval', icon: Shield, color: 'text-amber-600' };
   597→  if (/retry/i.test(cat)) return { label: 'Retry', icon: RotateCcw, color: 'text-orange-600' };
   598→  if (/oracle|ai_/i.test(cat)) return { label: 'AI Reasoning', icon: Sparkles, color: 'text-violet-600' };
   599→  return { label: 'Event', icon: Activity, color: 'text-muted-foreground' };
   600→}
