    63→  id: string;
    64→  label: string;
    65→  type: 'command' | 'client' | 'invoice' | 'return' | 'document' | 'activity';
    66→  timestamp: number;

    70→  id: string;
    71→  label: string;

    78→
    79→const RECENT_STORAGE_KEY = 'gstpilot-recent-commands';
    80→const FAVORITES_STORAGE_KEY = 'gstpilot-favorite-commands';
    81→const MAX_RECENT = 5;
    82→
    83→// ═══════════════════════════════════════════════════════════════════════════════
    84→// LOCAL STORAGE HELPERS
    85→// ═══════════════════════════════════════════════════════════════════════════════
    86→
    87→function loadRecent(): RecentItem[] {
    88→  try {
    89→    const raw = localStorage.getItem(RECENT_STORAGE_KEY);
    90→    return raw ? JSON.parse(raw) : [];
    91→  } catch {
    92→    return [];
    93→  }
    94→}
    95→
    96→function saveRecent(items: RecentItem[]) {
    97→  try {
    98→    localStorage.setItem(RECENT_STORAGE_KEY, JSON.stringify(items.slice(0, MAX_RECENT)));
    99→  } catch {
   100→    // ignore quota errors
   101→  }
   102→}
   103→
   104→function loadFavorites(): FavoriteItem[] {
   105→  try {
   106→    const raw = localStorage.getItem(FAVORITES_STORAGE_KEY);
   107→    return raw ? JSON.parse(raw) : [];
   108→  } catch {
   109→    return [];
   110→  }
   111→}
   112→
   113→function saveFavorites(items: FavoriteItem[]) {
   114→  try {
   115→    localStorage.setItem(FAVORITES_STORAGE_KEY, JSON.stringify(items));
   116→  } catch {
   117→    // ignore quota errors
   118→  }
   119→}

   609→                        onSelect={() => handleSelect(cmd.action)}
   610→                      />
   611→                    ))}
   612→                  </div>
   613→                )}
   614→
   615→                {/* ─── Search Results: Clients ────────────────────────────── */}
   616→                {isSearching && searchResults.clients.length > 0 && (

   625→                      </span>
   626→                    </div>

   628→                      <CommandItemRow
   629→                        key={c.id}
   630→                        icon={Building2}
   631→                        label={c.tradeName || c.legalName || 'Unknown'}
   632→                        description={`${c.gstin}${c.state ? ` · ${c.state}` : ''}`}
   633→                        onSelect={() =>
   634→                          handleClientSelect(c.id, c.tradeName || c.legalName || 'Unknown')
   635→                        }
   636→                      />
   637→                    ))}
   638→                  </div>
   639→                )}
   640→
   641→                {/* ─── Search Results: Invoices ───────────────────────────── */}
   642→                {isSearching && searchResults.invoices.length > 0 && (
   643→                  <div className="p-2">
   644→                    <div className="flex items-center gap-1.5 px-2 py-1.5">

   646→                      <span className="text-[10px] font-semibold text-white/40 uppercase tracking-[0.12em]">

   649→                      <span className="text-[10px] text-white/40 ml-auto">
   650→                        {searchResults.invoices.length} found
   651→                      </span>
   652→                    </div>
   653→                    {searchResults.invoices.map((inv) => (
   654→                      <CommandItemRow
   655→                        key={inv.id}
   656→                        icon={FileSpreadsheet}
   657→                        label={inv.invoiceNumber || 'Unknown'}
   658→                        description={`${inv.invoiceType} · ₹${(inv.totalAmount || 0).toLocaleString('en-IN')}`}

   660→                      />
   661→                    ))}
   662→                  </div>
   663→                )}
   664→
   665→                {/* ─── Search Results: Returns ────────────────────────────── */}
   666→                {isSearching && searchResults.returns.length > 0 && (
   667→                  <div className="p-2">
   668→                    <div className="flex items-center gap-1.5 px-2 py-1.5">
   669→                      <FileText className="h-3 w-3 text-blue-600" />
   670→                      <span className="text-[10px] font-semibold text-white/40 uppercase tracking-[0.12em]">
   671→                        Returns
   672→                      </span>
   673→                      <span className="text-[10px] text-white/40 ml-auto">
   674→                        {searchResults.returns.length} found
   675→                      </span>
   676→                    </div>
   677→                    {searchResults.returns.map((r) => (
   678→                      <CommandItemRow
   679→                        key={r.id}
   680→                        icon={FileText}
   681→                        label={`${r.returnType} · ${r.period}`}
   682→                        description={`${r.status}`}

   684→                          handleReturnSelect(
   685→                            `${r.returnType} · ${r.period}`,

   691→                  </div>
   692→                )}
   693→
   694→                {/* ─── Search Results: Documents ──────────────────────────── */}
   695→                {isSearching && searchResults.documents.length > 0 && (
   696→                  <div className="p-2">

   700→                        Documents
   701→                      </span>
   702→                      <span className="text-[10px] text-white/40 ml-auto">
   703→                        {searchResults.documents.length} found
   704→                      </span>
   705→                    </div>
   706→                    {searchResults.documents.map((d) => (
   707→                      <CommandItemRow
   708→                        key={d.id}
   709→                        icon={FolderOpen}
   710→                        label={d.fileName || 'Unknown document'}
   711→                        description={`${d.documentType} · ${d.status}`}
   712→                        onSelect={() => handleDocumentSelect(d.fileName || d.id)}
   713→                      />

   716→                )}
   717→
   718→                {/* ─── Search Results: Activities ─────────────────────────── */}
   719→                {isSearching && searchResults.activities.length > 0 && (

   724→                        Activities
   725→                      </span>
   726→                      <span className="text-[10px] text-white/40 ml-auto">

   728→                      </span>
   729→                    </div>
   730→                    {searchResults.activities.map((a) => (
   731→                      <CommandItemRow
   732→                        key={a.id}
   733→                        icon={Activity}
   734→                        label={a.title || 'Activity'}
   735→                        description={a.description?.slice(0, 60) || ''}

   743→                        }}
   744→                      />

   747→                )}
   748→              </div>
   749→
   750→              {/* Footer with keyboard hints */}
   751→              <div className="border-t border-white/[0.06] px-4 py-2.5 flex items-center gap-4 text-xs text-white/45">
   752→                <span className="flex items-center gap-1">
   753→                  <kbd className="badge-premium font-mono text-[10px]">↑↓</kbd>
   754→                  Navigate
   755→                </span>
   756→                <span className="flex items-center gap-1">
   757→                  <kbd className="badge-premium font-mono text-[10px]">↵</kbd>
   758→                  Select
   759→                </span>
   760→                <span className="flex items-center gap-1">

   763→                </span>

   766→                </span>
   767→              </div>
   768→            </div>
   769→          </motion.div>

   771→      )}
   772→    </AnimatePresence>
   773→  );
   774→}
