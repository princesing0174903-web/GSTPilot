   654→  1253→function groupConversations(convos: OracleConversation[]): ConversationGroups {
   655→  1254→  const now = new Date();
   656→  1255→  const startOfToday = new Date(now.getFullYear(), now.getMonth(), now.getDate()).getTime();
   657→  1256→  const startOfYesterday = startOfToday - 86_400_000;
   658→  1257→  const startOf7Days = startOfToday - 7 * 86_400_000;
   659→  1258→
   660→  1259→  const today: OracleConversation[] = [];
   661→  1260→  const yesterday: OracleConversation[] = [];
   662→  1261→  const prev7: OracleConversation[] = [];
   663→  1262→  const older: OracleConversation[] = [];
   664→  1263→
   665→  1264→  const sorted = [...convos].sort(
   666→  1265→    (a, b) => new Date(b.updatedAt).getTime() - new Date(a.updatedAt).getTime(),
   667→  1266→  );
   668→  1267→
   669→  1268→  for (const c of sorted) {
   670→  1269→    const t = new Date(c.updatedAt).getTime();
   671→  1270→    if (isNaN(t)) {
   672→  1271→      older.push(c);
   673→  1272→    } else if (t >= startOfToday) {
   674→  1273→      today.push(c);
   675→  1274→    } else if (t >= startOfYesterday) {
   676→  1275→      yesterday.push(c);
   677→  1276→    } else if (t >= startOf7Days) {
   678→  1277→      prev7.push(c);
   679→  1278→    } else {
   680→  1279→      older.push(c);
   681→  1280→    }
   682→  1281→  }
   683→  1282→
   684→  1283→  return { today, yesterday, prev7, older };
   685→  1284→}

   698→  1297→  if (/(reverse charge|rcm)/.test(lower)) pool.push('RCM किन पर लागू है?', 'How to report RCM in GSTR-3B?');
   699→  1298→  if (/(refund)/.test(lower)) pool.push('Refund process क्या है?', 'Refund timeline कितनी है?');
   700→  1299→
   701→  1300→  pool.push('GST kya hota hai?', 'Explain ITC rules', 'How can Oracle help me daily?');
   702→  1301→
   703→  1302→  const seen = new Set<string>();
   704→  1303→  const picks: string[] = [];
   705→  1304→  for (const p of pool) {
