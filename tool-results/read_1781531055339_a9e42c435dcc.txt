   285→// ─── Format Timestamp ──────────────────────────────────────────────────────
   286→function formatTimestamp(dateStr: string | null): string {
   287→  if (!dateStr) return 'Never';
   288→  try {
   289→    const date = new Date(dateStr);
   290→    const now = new Date();
   291→    const diffMs = now.getTime() - date.getTime();
   292→    const diffMins = Math.floor(diffMs / 60000);
   293→    const diffHours = Math.floor(diffMs / 3600000);
   294→    const diffDays = Math.floor(diffMs / 86400000);
   295→
   296→    if (diffMins < 1) return 'Just now';
   297→    if (diffMins < 60) return `${diffMins}m ago`;
   298→    if (diffHours < 24) return `${diffHours}h ago`;
   299→    if (diffDays < 7) return `${diffDays}d ago`;
   300→    return date.toLocaleDateString('en-IN', {
   301→      day: 'numeric',
   302→      month: 'short',
   303→      year: date.getFullYear() !== now.getFullYear() ? 'numeric' : undefined,
   304→    });
   305→  } catch {
   306→    return 'Invalid date';
   307→  }
   308→}
