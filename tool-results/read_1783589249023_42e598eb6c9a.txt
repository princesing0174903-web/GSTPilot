     3→//
     4→// Every AI decision must contain:
     5→//   • business reasoning    • financial reasoning    • risk reasoning
     6→//   • compliance reasoning  • operational reasoning  • legal reasoning
     7→//   • historical evidence   • supporting data        • confidence (0-100)
     8→//   • 2-3 alternatives with pros/cons/ROI
     9→//   • expected ROI           • rollback strategy     • final answer
    10→//
    11→// Pipeline:
    12→//   1. gatherBusinessContext(req.firmId)     — Context Engine™
    13→//   2. chooseModel({purpose:'reasoning'})    — Multi-Model AI Router™
    14→//   3. renderPrompt('oracle_orchestrate')    — Enterprise Prompt Engine™
    15→//   4. optional z-ai-web-dev-sdk LLM call (skip if req.callLLM===false or on failure)
    16→//   5. parse 6 reasoning dimensions; fall back to deterministic structured reasoning
    17→//   6. generate 2-3 alternatives with heuristic templates
    18→//   7. compute confidence from data completeness + historical evidence + LLM success
    19→//   8. persist to db.oracleReasoning + log model call + write memory record
    20→//   9. return ReasoningResult
    21→//
    22→// Never throws — always returns a valid ReasoningResult (with deterministic
    23→// fallback reasoning built from the gathered context).

    43→function safeParseJSON<T>(raw: string | null, fallback: T): T {
    44→  if (!raw) return fallback;
    45→  try {
    46→    return JSON.parse(raw) as T;

    48→    return fallback;
    49→  }

    58→interface LLMOutcome {

    66→
    67→async function callLLM(
    68→  systemPrompt: string,
    69→  userPrompt: string,
    70→): Promise<LLMOutcome> {
    71→  const start = Date.now();
    72→  try {
    73→    // Dynamic import so this module can load even if the SDK isn't ready.
    74→    const ZAIModule = await import('z-ai-web-dev-sdk');
    75→    const ZAI = (ZAIModule as any).default ?? ZAIModule;
    76→    const zai = await ZAI.create();
    77→    const completion = await zai.chat.completions.create({
    78→      messages: [
    79→        { role: 'system', content: systemPrompt },
    80→        { role: 'user', content: userPrompt },
    81→      ],
    82→      temperature: 0.4,
    83→      max_tokens: 2000,
    84→    });
    85→    const answer: string = completion?.choices?.[0]?.message?.content ?? '';
    86→    return {
    87→      answer,
    88→      success: answer.length > 0,
    89→      promptTokens: Math.ceil(systemPrompt.length / 4) + Math.ceil(userPrompt.length / 4),
    90→      outputTokens: Math.ceil(answer.length / 4),
    91→      latencyMs: Date.now() - start,
    92→    };
    93→  } catch (err: any) {
    94→    return {
    95→      answer: '',
    96→      success: false,
    97→      promptTokens: 0,
    98→      outputTokens: 0,
    99→      latencyMs: Date.now() - start,
   100→      error: err?.message ?? String(err),
   101→    };
   102→  }
   103→}

   126→function assessContext(ctx: Awaited<ReturnType<typeof gatherBusinessContext>>): ContextSlice {
   127→  let modulesWithData = 0;
   128→  if (ctx.graph.nodeCount > 0) modulesWithData++;
   129→  if (ctx.twin.healthScore > 0 || ctx.twin.cashRunwayDays > 0) modulesWithData++;
   130→  if (ctx.finance.cashBalance > 0 || ctx.finance.monthlyRevenue > 0) modulesWithData++;
   131→  if (ctx.connectedSystems.total > 0) modulesWithData++;
   132→  if (ctx.recentInvoices.length > 0) modulesWithData++;
   133→  if (ctx.pendingApprovals.length > 0) modulesWithData++;
   134→  if (ctx.goals.length > 0) modulesWithData++;
   135→  if (ctx.previousConversations.length > 0) modulesWithData++;
   136→  if (ctx.knowledge.entityCount > 0) modulesWithData++;
   137→
   138→  return {
   139→    hasCash: ctx.finance.cashBalance > 0,
   140→    hasRevenue: ctx.finance.monthlyRevenue > 0,
   141→    cashBalance: ctx.finance.cashBalance,
   142→    monthlyRevenue: ctx.finance.monthlyRevenue,
   143→    monthlyExpenses: ctx.finance.monthlyExpenses,
   144→    receivables: ctx.finance.receivables,
   145→    payables: ctx.finance.payables,
   146→    gstCollected: ctx.finance.gstCollected,
   147→    gstPaid: ctx.finance.gstPaid,
   148→    healthScore: ctx.twin.healthScore,
   149→    runwayDays: ctx.twin.cashRunwayDays,
   150→    anomalies: ctx.twin.anomalies,
   151→    pendingApprovals: ctx.pendingApprovals.length,
   152→    goalsCount: ctx.goals.length,
   153→    strategiesCount: ctx.strategies.length,
   154→    modulesWithData,
   155→  };
   156→}

   216→// ─── LLM output parsing ─────────────────────────────────────────────────────
   217→
   218→interface ParsedReasoning {
   219→  business: string;
   220→  financial: string;
   221→  risk: string;
   222→  compliance: string;
   223→  operational: string;
   224→  legal: string;
   225→  finalAnswer: string;
   226→}
   227→
   228→/**
   229→ * Parse the 6 reasoning dimensions out of the LLM answer.
   230→ * Tries (a) JSON, (b) markdown section headers, (c) plain keyword prefixes.
   231→ * On any failure, returns null and the caller falls back to deterministic.
   232→ */
   233→function parseReasoningFromAnswer(answer: string): ParsedReasoning | null {
   234→  if (!answer || answer.trim().length === 0) return null;
   235→
   236→  // (a) Try JSON first
   237→  const jsonStart = answer.indexOf('{');
   238→  const jsonEnd = answer.lastIndexOf('}');
   239→  if (jsonStart >= 0 && jsonEnd > jsonStart) {
   240→    try {
   241→      const obj = JSON.parse(answer.slice(jsonStart, jsonEnd + 1));
   242→      const fields = ['businessReasoning', 'financialReasoning', 'riskReasoning', 'complianceReasoning', 'operationalReasoning', 'legalReasoning', 'finalAnswer'];
   243→      if (fields.every((f) => typeof (obj as any)[f] === 'string' && (obj as any)[f].length > 0)) {
   244→        return {
   245→          business: obj.businessReasoning,
   246→          financial: obj.financialReasoning,
   247→          risk: obj.riskReasoning,
   248→          compliance: obj.complianceReasoning,
   249→          operational: obj.operationalReasoning,
   250→          legal: obj.legalReasoning,
   251→          finalAnswer: obj.finalAnswer,
   252→        };
   253→      }
   254→    } catch {
   255→      /* fall through */
   256→    }
   257→  }
   258→
   259→  // (b) Try markdown headers
   260→  const headerRe = /(?:^|\n)#{1,4}\s*([A-Za-z][A-Za-z &/\-]{2,40})\s*\n([\s\S]*?)(?=\n#{1,4}\s|$)/g;
   261→  const sections: Record<string, string> = {};
   262→  let m: RegExpExecArray | null;
   263→  while ((m = headerRe.exec(answer)) !== null) {
   264→    const label = m[1].toLowerCase().trim();
   265→    const body = m[2].trim();
   266→    if (label.includes('business')) sections.business = body;
   267→    else if (label.includes('financial')) sections.financial = body;
   268→    else if (label.includes('risk') && !label.includes('compliance')) sections.risk = body;
   269→    else if (label.includes('compliance')) sections.compliance = body;
   270→    else if (label.includes('operation')) sections.operational = body;
   271→    else if (label.includes('legal')) sections.legal = body;
   272→    else if (label.includes('final') || label.includes('recommendation') || label.includes('answer')) sections.finalAnswer = body;
   273→  }
   274→
   275→  if (sections.business && sections.financial && sections.risk) {
   276→    return {
   277→      business: sections.business,
   278→      financial: sections.financial,
   279→      risk: sections.risk,
   280→      compliance: sections.compliance || '',
   281→      operational: sections.operational || '',
   282→      legal: sections.legal || '',
   283→      finalAnswer: sections.finalAnswer || answer.slice(0, 500),
   284→    };
   285→  }
   286→
   287→  // (c) Try keyword-prefixed lines
   288→  const lines = answer.split('\n');
   289→  const grab = (kw: string): string => {
   290→    const ln = lines.find((l) => l.toLowerCase().includes(kw));
   291→    return ln ? ln.replace(/^[\s>*\-\d.]+/, '').trim() : '';
   292→  };
   293→  const business = grab('business reasoning');
   294→  const financial = grab('financial reasoning');
   295→  const risk = grab('risk reasoning');
   296→  const compliance = grab('compliance reasoning');
   297→  const operational = grab('operational reasoning');
   298→  const legal = grab('legal reasoning');
   299→  if (business && financial && risk) {
   300→    return {
   301→      business,
   302→      financial,
   303→      risk,
   304→      compliance: compliance || '',
   305→      operational: operational || '',
   306→      legal: legal || '',
   307→      finalAnswer: answer.slice(0, 800),
   308→    };
   309→  }
   310→
   311→  return null;
   312→}

   378→// ─── Confidence computation ─────────────────────────────────────────────────
   379→
   380→function computeConfidence(
   381→  slice: ContextSlice,
   382→  historicalAvailable: boolean,
   383→  llmSuccess: boolean,
   384→): number {
   385→  // Base: data completeness (0-50 pts, 9 modules max)
   386→  const dataCompleteness = Math.min(50, Math.round((slice.modulesWithData / 9) * 50));
   387→
   388→  // Historical evidence (0-25 pts)
   389→  const historical = historicalAvailable ? 25 : 5;
   390→
   391→  // LLM success (0-25 pts)
   392→  const llm = llmSuccess ? 25 : 10;
   393→
   394→  // Penalty for anomalies (max -10)
   395→  const anomalyPenalty = Math.min(10, slice.anomalies * 2);
   396→
   397→  const total = Math.max(0, Math.min(100, dataCompleteness + historical + llm - anomalyPenalty));
   398→  return total;
   399→}

   401→// ─── Main entry: reason() ───────────────────────────────────────────────────
   402→
   403→export async function reason(req: ReasoningRequest): Promise<ReasoningResult> {
   404→  const firmId = req.firmId || FIRM_ID;
   405→
   406→  // Step 1 — gather context (never throws)
   407→  const ctx = await gatherBusinessContext(firmId).catch(() => null);
   408→  const safeCtx = ctx ?? (await gatherBusinessContext(firmId));
   409→  const slice = assessContext(safeCtx);
   410→
   411→  // Step 2 — choose model
   412→  const tier: ModelTier = req.preferredTier || 'standard';
   413→  const modelChoice = chooseModel({
   414→    purpose: 'reasoning',
   415→    tier,
   416→    inputTokensEstimate: safeCtx.estimatedTokens + Math.ceil(req.request.length / 4),
   417→  });
   418→
   419→  // Step 3 — render prompt
   420→  const rendered = renderPrompt('oracle_orchestrate', {
   421→    context: formatContextForPrompt(safeCtx),
   422→    question: req.request,
   423→  });
   424→  const systemPrompt = rendered?.system ?? 'You are Oracle™ — the unified AI brain of GSTPilot Infinity™.';
   425→  const userPrompt = rendered?.user ?? `CONTEXT:\n${formatContextForPrompt(safeCtx)}\n\nQUESTION:\n${req.request}`;
   426→
   427→  // Step 4 — optional LLM call
   428→  const shouldCallLLM = req.callLLM !== false;
   429→  let llmOutcome: LLMOutcome | null = null;
   430→  if (shouldCallLLM) {
   431→    llmOutcome = await callLLM(systemPrompt, userPrompt);
   432→  }
   433→
   434→  // Step 5 — build the 6 reasoning dimensions
   435→  const fallback = buildDeterministicReasoning(req, safeCtx, slice);
   436→  let business = fallback.business;
   437→  let financial = fallback.financial;
   438→  let risk = fallback.risk;
   439→  let compliance = fallback.compliance;
   440→  let operational = fallback.operational;
   441→  let legal = fallback.legal;
   442→  let historical = fallback.historical;
   443→  let finalAnswer = fallback.finalAnswer;
   444→
   445→  if (llmOutcome?.success) {
   446→    const parsed = parseReasoningFromAnswer(llmOutcome.answer);
   447→    if (parsed) {
   448→      // Use parsed LLM output, but never accept empty strings — fall back to deterministic.
   449→      business = parsed.business || fallback.business;
   450→      financial = parsed.financial || fallback.financial;
   451→      risk = parsed.risk || fallback.risk;
   452→      compliance = parsed.compliance || fallback.compliance;
   453→      operational = parsed.operational || fallback.operational;
   454→      legal = parsed.legal || fallback.legal;
   455→      finalAnswer = parsed.finalAnswer || fallback.finalAnswer;
   456→      // If LLM didn't surface historical evidence, keep our deterministic version.
   457→      if (!historical || historical.length < 20) historical = fallback.historical;
   458→    } else {
   459→      // Parsing failed — put the full answer in business reasoning (per spec) but keep the rest deterministic.
   460→      business = `${llmOutcome.answer.slice(0, 1200)}\n\n[Structured parse failed — showing raw LLM output. Deterministic reasoning used for other dimensions.]`;
   461→    }
   462→  }
   463→
   464→  // Step 6 — alternatives
   465→  const alternatives = generateAlternatives(req, slice);
   466→
   467→  // Step 7 — confidence
   468→  const confidence = computeConfidence(slice, safeCtx.previousConversations.length > 0, llmOutcome?.success ?? false);
   469→
   470→  // Step 8 — supporting data (real context snapshot)
   471→  const supportingData: Record<string, unknown> = {
   472→    firmId,
   473→    contextGatheredAt: safeCtx.gatheredAt,
   474→    estimatedContextTokens: safeCtx.estimatedTokens,
   475→    finance: safeCtx.finance,
   476→    twin: safeCtx.twin,
   477→    graph: {
   478→      nodeCount: safeCtx.graph.nodeCount,
   479→      edgeCount: safeCtx.graph.edgeCount,
   480→      topRisks: safeCtx.graph.topRisks,
   481→    },
   482→    connectedSystems: safeCtx.connectedSystems,
   483→    pendingApprovals: safeCtx.pendingApprovals.length,
   484→    activeGoals: safeCtx.goals.length,
   485→    activeStrategies: safeCtx.strategies.length,
   486→    recentInvoicesCount: safeCtx.recentInvoices.length,
   487→    modulesWithData: slice.modulesWithData,
   488→    modelChoice: {
   489→      provider: modelChoice.provider,
   490→      model: modelChoice.model,
   491→      tier: modelChoice.tier,
   492→      rationale: modelChoice.rationale,
   493→      costPer1kUsd: modelChoice.costPer1kUsd,
   494→      estimatedLatencyMs: modelChoice.estimatedLatencyMs,
   495→    },
   496→    llmCalled: shouldCallLLM,
   497→    llmSuccess: llmOutcome?.success ?? false,
   498→    llmLatencyMs: llmOutcome?.latencyMs ?? 0,
   499→  };
   500→
   501→  const executivesConsulted: AIModuleId[] = req.executives && req.executives.length > 0
   502→    ? req.executives
   503→    : ['oracle', 'ceo', 'cfo', 'cro'];
   504→
   505→  // Step 9 — persist to db.oracleReasoning
   506→  let reasoningId = '';
   507→  let createdAt = new Date().toISOString();
   508→  try {
   509→    const created = await (db as any).oracleReasoning.create({
   510→      data: {
   511→        firmId,
   512→        userId: req.userId ?? null,
   513→        request: req.request,
   514→        requestType: req.requestType,
   515→        businessReasoning: business,
   516→        financialReasoning: financial,
   517→        riskReasoning: risk,
   518→        complianceReasoning: compliance,
   519→        operationalReasoning: operational,
   520→        legalReasoning: legal,
   521→        historicalEvidence: historical,
   522→        supportingData: JSON.stringify(supportingData),
   523→        confidence,
   524→        alternatives: JSON.stringify(alternatives),
   525→        expectedRoi: fallback.expectedRoi,
   526→        rollbackStrategy: fallback.rollback,
   527→        finalAnswer,
   528→        modelUsed: modelChoice.provider,
   529→        modelTier: modelChoice.tier,
   530→        executivesConsulted: JSON.stringify(executivesConsulted),
   531→        approved: false,
   532→        rejected: false,
   533→        outcome: 'pending',
   534→        outcomeNote: null,
   535→      },
   536→    });
   537→    reasoningId = created.id;
   538→    createdAt = created.createdAt instanceof Date ? created.createdAt.toISOString() : String(created.createdAt);
   539→  } catch (err) {
   540→    console.warn('[Reasoning Engine] failed to persist OracleReasoning:', err);
   541→  }
   542→
   543→  // Step 10 — log the model call (for audit + cost tracking)
   544→  try {
   545→    await logModelCall({
   546→      firmId,
   547→      userId: req.userId ?? null,
   548→      provider: modelChoice.provider as AIProvider,
   549→      model: modelChoice.model,
   550→      tier: modelChoice.tier,
   551→      purpose: 'reasoning',
   552→      promptTokens: llmOutcome?.promptTokens ?? safeCtx.estimatedTokens,
   553→      outputTokens: llmOutcome?.outputTokens ?? Math.ceil(finalAnswer.length / 4),
   554→      latencyMs: llmOutcome?.latencyMs ?? 0,
   555→      success: shouldCallLLM ? (llmOutcome?.success ?? false) : true,
   556→      errorMessage: llmOutcome?.error ?? (shouldCallLLM ? null : 'LLM call skipped (callLLM=false)'),
   557→      reasoningId: reasoningId || null,
   558→    });
   559→  } catch (err) {
   560→    console.warn('[Reasoning Engine] failed to log model call:', err);
   561→  }
   562→
   563→  // Step 11 — write memory record
   564→  try {
   565→    await writeMemory({
   566→      firmId,
   567→      userId: req.userId ?? null,
   568→      category: 'reasoning',
   569→      entityType: 'reasoning',
   570→      entityId: reasoningId || null,
   571→      title: `Oracle reasoning: ${req.request.slice(0, 80)}`,
   572→      summary: finalAnswer.slice(0, 500),
   573→      payload: {
   574→        requestType: req.requestType,
   575→        confidence,
   576→        modelUsed: modelChoice.provider,
   577→        modelTier: modelChoice.tier,
   578→        llmCalled: shouldCallLLM,
   579→        llmSuccess: llmOutcome?.success ?? false,
   580→        executivesConsulted,
   581→        expectedRoi: fallback.expectedRoi,
   582→      },
   583→      tags: ['oracle', 'reasoning', req.requestType],
   584→      importance: confidence >= 70 ? 80 : 60,
   585→      source: 'oracle',
   586→    });
   587→  } catch (err) {
   588→    console.warn('[Reasoning Engine] failed to write memory:', err);
   589→  }
   590→
   591→  // Step 12 — assemble & return
   592→  const result: ReasoningResult = {
   593→    id: reasoningId,
   594→    firmId,
   595→    userId: req.userId ?? null,
   596→    request: req.request,
   597→    requestType: req.requestType,
   598→    businessReasoning: business,
   599→    financialReasoning: financial,
   600→    riskReasoning: risk,
   601→    complianceReasoning: compliance,
   602→    operationalReasoning: operational,
   603→    legalReasoning: legal,
   604→    historicalEvidence: historical,
   605→    supportingData,
   606→    confidence,
   607→    alternatives,
   608→    expectedRoi: fallback.expectedRoi,
   609→    rollbackStrategy: fallback.rollback,
   610→    finalAnswer,
   611→    modelUsed: modelChoice.provider,
   612→    modelTier: modelChoice.tier,
   613→    executivesConsulted,
   614→    approved: false,
   615→    rejected: false,
   616→    outcome: 'pending',
   617→    outcomeNote: null,
   618→    createdAt,
   619→  };
   620→
   621→  return result;
   622→}

   701→export interface ReasoningStats {
   702→  total: number;
   703→  approved: number;
   704→  rejected: number;
   705→  pending: number;
   706→  byType: Record<string, number>;
   707→  avgConfidence: number;
   708→}
   709→
   710→export async function getReasoningStats(): Promise<ReasoningStats> {
   711→  const firmId = FIRM_ID;
   712→  const since = new Date(Date.now() - 24 * 60 * 60 * 1000);
   713→  const where = { firmId, createdAt: { gte: since } };
   714→
   715→  try {
   716→    const [total, approved, rejected, pending, byTypeRows, confAgg] = await Promise.all([
   717→      (db as any).oracleReasoning.count({ where }),
   718→      (db as any).oracleReasoning.count({ where: { ...where, approved: true } }),
   719→      (db as any).oracleReasoning.count({ where: { ...where, rejected: true } }),
   720→      (db as any).oracleReasoning.count({
   721→        where: { ...where, approved: false, rejected: false },
   722→      }),
   723→      (db as any).oracleReasoning.groupBy({
   724→        by: ['requestType'],
   725→        where,
   726→        _count: { _all: true },
   727→      }),
   728→      (db as any).oracleReasoning.aggregate({
   729→        where,
   730→        _avg: { confidence: true },
   731→      }),
   732→    ]);
   733→
   734→    const byType: Record<string, number> = {};
   735→    for (const r of byTypeRows ?? []) byType[r.requestType] = r._count._all;
   736→
   737→    return {
   738→      total: total ?? 0,
   739→      approved: approved ?? 0,
   740→      rejected: rejected ?? 0,
   741→      pending: pending ?? 0,
   742→      byType,
   743→      avgConfidence: Math.round(confAgg?._avg?.confidence ?? 0),
   744→    };
   745→  } catch (err) {
   746→    console.warn('[Reasoning Engine] getReasoningStats failed:', err);
   747→    return {
   748→      total: 0,
   749→      approved: 0,
   750→      rejected: 0,
   751→      pending: 0,
   752→      byType: {},
   753→      avgConfidence: 0,
   754→    };
   755→  }
   756→}
