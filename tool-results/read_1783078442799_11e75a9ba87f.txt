   190→function formatFileSize(bytes: number): string {
   191→  if (bytes < 1024) return `${bytes} B`;
   192→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`;
   193→  return `${(bytes / (1024 * 1024)).toFixed(1)} MB`;
   194→}

   198→    return <FileSpreadsheet className="size-5 text-emerald-600" />;
   199→  if (type.includes('image')) return <FileImage className="size-5 text-teal-600" />;

   204→function getOverdueFilings(filings: GSTRFiling[]): Set<string> {
   205→  const now = new Date();
   206→  const overdueIds = new Set<string>();
   207→  filings.forEach((f) => {
   208→    if (f.status !== 'filed' && f.status !== 'reopened') {
   209→      const [year, month] = f.period.split('-').map(Number);
   210→      if (year && month) {
   211→        const dueDate = new Date(year, month, 11);
   212→        if (now > dueDate) overdueIds.add(f.id);
   213→      }
   214→    }
   215→  });
   216→  return overdueIds;
   217→}
