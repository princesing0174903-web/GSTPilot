    47→    47→function formatDate(d: string | unknown): string {
    48→    48→  if (!d) return '—'
    49→    49→  try {
    50→    50→    const date = typeof d === 'string' ? new Date(d) : new Date()
    51→    51→    return date.toLocaleDateString('en-IN', { day: 'numeric', month: 'short', year: 'numeric' })
    52→    52→  } catch { return '—' }
    53→    53→}
    54→    54→
    55→    55→function formatTime(d: string | unknown): string {
    56→    56→  if (!d) return ''
    57→    57→  try {
    58→    58→    const date = typeof d === 'string' ? new Date(d) : new Date()
    59→    59→    return date.toLocaleTimeString('en-IN', { hour: '2-digit', minute: '2-digit' })
    60→    60→  } catch { return '' }
    61→    61→}

    63→    63→function daysUntil(d: string | unknown): number {

    65→    65→  try {
    66→    66→    const date = typeof d === 'string' ? new Date(d) : new Date()
    67→    67→    return Math.ceil((date.getTime() - Date.now()) / (1000 * 60 * 60 * 24))
    68→    68→  } catch { return 999 }
    69→    69→}

   512→   512→function DealsView({ deals, leads, loading }: { deals: Array<FirestoreDeal & { id: string }>; leads: Array<FirestoreLead & { id: string }>; loading: boolean }) {
   513→   513→  const [sortField, setSortField] = useState<'value' | 'probability' | 'expectedCloseDate'>('value')
   514→   514→  const [sortDir, setSortDir] = useState<'asc' | 'desc'>('desc')
   515→   515→  const [showNewDeal, setShowNewDeal] = useState(false)

   520→   520→      const dir = sortDir === 'asc' ? 1 : -1
   521→   521→      if (sortField === 'value' || sortField === 'probability') {
   522→   522→        return ((a[sortField] as number) - (b[sortField] as number)) * dir
   523→   523→      }
   524→   524→      const aDate = typeof a.expectedCloseDate === 'string' ? new Date(a.expectedCloseDate).getTime() : 0
   525→   525→      const bDate = typeof b.expectedCloseDate === 'string' ? new Date(b.expectedCloseDate).getTime() : 0
   526→   526→      return (aDate - bDate) * dir
   527→   527→    })
   528→   528→  }, [deals, sortField, sortDir])
   529→   529→
   530→   530→  const handleSort = (field: typeof sortField) => {
   531→   531→    if (sortField === field) setSortDir(d => d === 'asc' ? 'desc' : 'asc')
   532→   532→    else { setSortField(field); setSortDir('desc') }
   533→   533→  }

   541→   541→        stage: newDeal.stage,
   542→   542→        probability: Number(newDeal.probability) || 30,

   577→   577→              </div>
