   215→function formatRelativeTime(timestamp: string): string {
   216→  const now = new Date();
   217→  const date = new Date(timestamp);
   218→  const diffMs = now.getTime() - date.getTime();
   219→  const diffMins = Math.floor(diffMs / 60000);
   220→  const diffHours = Math.floor(diffMins / 60);
   221→  const diffDays = Math.floor(diffHours / 24);
   222→
   223→  if (diffMins < 60) return `${diffMins}m ago`;
   224→  if (diffHours < 24) return `${diffHours}h ago`;
   225→  if (diffDays < 7) return `${diffDays}d ago`;
   226→  return date.toLocaleDateString('en-IN', { day: 'numeric', month: 'short' });
   227→}

   231→/** Map a store upload to a DocumentEntry for the UI */
   232→function mapUploadToDocument(upload: SampleUpload): DocumentEntry {
   233→  const statusMap: Record<SampleUpload['status'], DocumentEntry['status']> = {
   234→    extracted: 'Processed',
   235→    processing: 'Processing',
   236→    failed: 'Error',
   237→  };
   238→  const sizeBytes = (upload.rowCount ?? 0) * 128; // Approximate size
   239→  const sizeStr = sizeBytes > 1048576 ? `${(sizeBytes / 1048576).toFixed(1)} MB` : sizeBytes > 1024 ? `${(sizeBytes / 1024).toFixed(0)} KB` : `${sizeBytes} B`;
   240→  // Derive document type from filename
   241→  const ext = upload.filename.split('.').pop()?.toLowerCase() ?? '';
   242→  const typeMap: Record<string, string> = {
   243→    xlsx: 'Sales Register',
   244→    xls: 'Sales Register',
   245→    pdf: 'Purchase Register',
   246→    json: 'GST Portal Data',
   247→    csv: 'Data Import',
   248→  };
   249→  return {
   250→    id: upload.id,
   251→    name: upload.filename,
   252→    type: typeMap[ext] ?? 'Document',
   253→    uploadDate: upload.uploadTime.split('T')[0],
   254→    status: statusMap[upload.status],
   255→    invoicesExtracted: upload.invoiceCount ?? 0,
   256→    totalRows: upload.rowCount ?? 0,
   257→    accuracy: upload.accuracy ?? 0,
   258→    size: sizeStr || '1.0 MB',
   259→  };

   294→/** Map store filing status to UI ReturnEntry status */
   295→function mapFilingStatus(filing: SampleFiling): ReturnEntry['status'] {
   296→  if (filing.status === 'filed') return 'Filed';
   297→
   298→  const readyStatuses: SampleFiling['status'][] = ['validated', 'reviewed', 'generated'];
   299→  if (readyStatuses.includes(filing.status)) {
   300→    // Even "ready" filings can be overdue
   301→    return isFilingOverdue(filing.period) ? 'Overdue' : 'Ready to File';
   302→  }
   303→
   304→  // draft, prepared, reopened
   305→  return isFilingOverdue(filing.period) ? 'Overdue' : 'Draft';
   306→}

   308→/** Derive recon runs from store summary data */
   309→function deriveReconRuns(summary: SampleReconCategory[], healthScore: number): ReconRun[] {
   310→  if (summary.length === 0) {
   311→    // Fallback based on health score
   312→    const isLowRisk = healthScore > 80;
   313→    const isHighRisk = healthScore < 50;
   314→    return [
   315→      { id: 'rc1', period: 'May 2025', matchRate: isLowRisk ? 92 : isHighRisk ? 58 : 78, mismatches: isLowRisk ? 3 : isHighRisk ? 22 : 9, missingInvoices: isLowRisk ? 1 : isHighRisk ? 11 : 4, taxDifference: isLowRisk ? 12000 : isHighRisk ? 280000 : 62000, runDate: '2025-06-28', status: 'Completed' },
   316→      { id: 'rc2', period: 'Apr 2025', matchRate: isLowRisk ? 94 : isHighRisk ? 55 : 80, mismatches: isLowRisk ? 2 : isHighRisk ? 28 : 8, missingInvoices: isLowRisk ? 0 : isHighRisk ? 14 : 3, taxDifference: isLowRisk ? 5000 : isHighRisk ? 345000 : 48000, runDate: '2025-05-29', status: 'Completed' },
   317→    ];
   318→  }
   319→
   320→  const total = summary.reduce((sum, c) => sum + c.count, 0);
   321→  const perfectMatch = summary.find(c => c.label === 'Perfect Match');
   322→  const matchRate = perfectMatch && total > 0 ? Math.round((perfectMatch.count / total) * 100) : 0;
   323→  const mismatches = summary.filter(c => ['Mismatch', 'Partial Match'].includes(c.label)).reduce((sum, c) => sum + c.count, 0);
   324→  const missingInvoices = summary.filter(c => c.label.includes('Missing')).reduce((sum, c) => sum + c.count, 0);
   325→  const taxDifference = summary.filter(c => c.label !== 'Perfect Match').reduce((sum, c) => sum + c.amount, 0);
   326→
   327→  return [
   328→    { id: 'rc1', period: 'May 2025', matchRate, mismatches, missingInvoices, taxDifference, runDate: '2025-06-28', status: 'Completed' as const },
   329→    { id: 'rc2', period: 'Apr 2025', matchRate: Math.min(100, matchRate + 2), mismatches: Math.max(0, mismatches - 1), missingInvoices: Math.max(0, missingInvoices - 1), taxDifference: Math.round(taxDifference * 0.7), runDate: '2025-05-29', status: 'Completed' as const },
   330→  ];
   331→}

   333→/** Compute overall match rate from recon summary */
   334→function computeMatchRate(summary: SampleReconCategory[], healthScore: number): number {
   335→  if (summary.length === 0) {
   336→    if (healthScore > 80) return 94;
   337→    if (healthScore < 50) return 58;
   338→    return 78;
   339→  }
   340→  const total = summary.reduce((sum, c) => sum + c.count, 0);
   341→  const perfectMatch = summary.find(c => c.label === 'Perfect Match');
   342→  return perfectMatch && total > 0 ? Math.round((perfectMatch.count / total) * 100) : 0;
   343→}

   380→function HealthRing({ score, size = 72, strokeWidth = 5 }: { score: number; size?: number; strokeWidth?: number }) {
   381→  const radius = (size - strokeWidth) / 2;
   382→  const circumference = 2 * Math.PI * radius;
   383→  const offset = circumference - (score / 100) * circumference;
   384→  const color = getHealthColor(score);
   385→
   386→  return (
   387→    <div className="relative inline-flex items-center justify-center" style={{ width: size, height: size }}>
   388→      <svg width={size} height={size} className="-rotate-90">
   389→        <circle cx={size / 2} cy={size / 2} r={radius} fill="none" stroke="#e2e8f0" strokeWidth={strokeWidth} />
   390→        <motion.circle cx={size / 2} cy={size / 2} r={radius} fill="none" stroke={color.stroke} strokeWidth={strokeWidth} strokeLinecap="round" strokeDasharray={circumference} initial={{ strokeDashoffset: circumference }} animate={{ strokeDashoffset: offset }} transition={{ duration: 1, delay: 0.3, ease: 'easeOut' }} />
   391→      </svg>
   392→      <div className="absolute inset-0 flex flex-col items-center justify-center">
   393→        <span className={`font-bold text-lg ${color.text}`}>{score}</span>
   394→        <span className="text-[8px] text-muted-foreground">/ 100</span>
   395→      </div>
   396→    </div>
   397→  );
   398→}

   418→export default function ClientWorkspacePage() {
   419→  const { selectedClientId, setCurrentView, setSelectedClientId, setReturnPrepCtx } = useApp();
   420→  const store = useGSTStore();
   421→  const [returnFilter, setReturnFilter] = useState('all');
   422→
   423→  // ─── Derive client from store ────────────────────────────────────────────
   424→  // Try direct ID match first, then fall back to finding by index from clients array
   425→  const client = useMemo(() => {
   426→    if (!selectedClientId) return undefined;
   427→    // Direct lookup
   428→    const direct = store.getClient(selectedClientId);
   429→    if (direct) return direct;
   430→    // Fallback: try to find by matching against known client IDs
   431→    // The registry page may use different IDs (cl_001 vs client-1)
   432→    // Try extracting a numeric index and mapping to store format
   433→    const numMatch = selectedClientId.match(/(\d+)/);
   434→    if (numMatch) {
   435→      const idx = parseInt(numMatch[1], 10) - 1;
   436→      if (idx >= 0 && idx < store.clients.length) {
   437→        return store.clients[idx];
   438→      }
   439→    }
   440→    // Last resort: return first client
   441→    return store.clients[0];
   442→  }, [selectedClientId, store]);
   443→
   444→  // ─── Derive all workspace data reactively from the store ─────────────────
   445→  const workspace = useMemo<ClientWorkspaceData | null>(() => {
   446→    if (!client) return null;
   447→    const clientId = client.id;
   448→    const hs = client.healthScore;
   449→
   450→    // ── Filings → Returns ──
   451→    const filings = store.getFilingsForClient(clientId);
   452→    const returns: ReturnEntry[] = filings.map(f => ({
   453→      id: f.id,
   454→      type: f.returnType,
   455→      period: periodToLabel(f.period),
   456→      filingDate: f.filedDate ?? '',
   457→      arn: f.acknowledgmentNumber ?? '',
   458→      status: mapFilingStatus(f),
   459→      taxAmount: f.totalTax,
   460→    }));
   461→
   462→    // ── Issues → Pending Actions ──
   463→    const issues = store.getIssuesForClient(clientId);
   464→    const issueActions: PendingAction[] = issues
   465→      .filter(i => !i.resolved)
   466→      .map(i => {
   467→        const actionType = issueCategoryToActionType(i.category);
   468→        const priority: PendingAction['priority'] =
   469→          i.severity === 'critical' ? 'high' : i.severity === 'warning' ? 'medium' : 'low';
   470→        const targetView: AppView =
