   190→function LanguageCard({
   191→  lang, active, onSelect, index,
   192→}: {
   193→  lang: Language;
   194→  active: boolean;
   195→  onSelect: () => void;
   196→  index: number;

   198→  const accentBar =
   199→    lang.coverage >= 90 ? 'bg-emerald-500' :

   202→

   204→    <motion.button
   205→      type="button"
   206→      onClick={onSelect}
   207→      initial={{ opacity: 0, y: 14 }}
   208→      animate={{ opacity: 1, y: 0 }}
   209→      transition={{ duration: 0.35, delay: index * 0.04 }}
   210→      whileHover={{ y: -2 }}
   211→      className={`text-left rounded-xl border p-4 backdrop-blur-sm transition-all ${
   212→        active
   213→          ? 'border-emerald-500/40 bg-emerald-500/[0.06] ring-1 ring-emerald-500/30'
   214→          : 'border-white/[0.06] bg-white/[0.02] hover:border-white/[0.14]'
   215→      }`}
   216→    >
   217→      <div className="flex items-start justify-between gap-2">
   218→        <div className="flex items-center gap-2 min-w-0">
   219→          <span className="text-2xl leading-none">{lang.flag}</span>
   220→          <div className="min-w-0">
   221→            <div className="text-base font-semibold text-zinc-50 leading-tight truncate" dir={lang.rtl ? 'rtl' : 'ltr'}>
   222→              {lang.nativeName}
   223→            </div>
   224→            <div className="text-[11px] text-zinc-400">{lang.name} · <span className="font-mono text-zinc-500">{lang.code}</span></div>
   225→          </div>
   226→        </div>
   227→        {active && (
   228→          <Badge variant="outline" className="text-[9px] border-emerald-500/30 bg-emerald-500/10 text-emerald-300 shrink-0">

   230→          </Badge>

   232→      </div>
   233→
   234→      <div className="mt-3 flex items-center gap-1.5 flex-wrap">
   235→        {lang.rtl && (
   236→          <Badge variant="outline" className="text-[9px] border-violet-500/30 bg-violet-500/10 text-violet-300">
