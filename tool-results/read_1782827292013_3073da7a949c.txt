   265→function timeAgo(iso: string): string {
   266→  try {
   267→    const diff = Date.now() - new Date(iso).getTime();
   268→    const s = Math.floor(diff / 1000);
   269→    if (s < 0) return 'just now';
   270→    if (s < 60) return `${s}s ago`;
   271→    const m = Math.floor(s / 60);
   272→    if (m < 60) return `${m}m ago`;
   273→    const h = Math.floor(m / 60);
   274→    if (h < 24) return `${h}h ago`;
   275→    return `${Math.floor(h / 24)}d ago`;
   276→  } catch {
   277→    return '—';
   278→  }
   279→}
   280→
   281→function fmtDateTime(iso: string): string {
   282→  try {
   283→    return new Date(iso).toLocaleString(undefined, {
   284→      month: 'short', day: 'numeric', hour: '2-digit', minute: '2-digit',
   285→    });
   286→  } catch {
   287→    return iso;
   288→  }
   289→}
   290→
   291→function fmtINR(n: number): string {
   292→  if (!Number.isFinite(n)) return '₹0';
   293→  const abs = Math.abs(n);
   294→  if (abs >= 1e7) return `₹${(n / 1e7).toFixed(2)} Cr`;
   295→  if (abs >= 1e5) return `₹${(n / 1e5).toFixed(2)} L`;
   296→  if (abs >= 1e3) return `₹${(n / 1e3).toFixed(1)} K`;
   297→  return `₹${Math.round(n)}`;
   298→}
   299→
   300→function fmtNum(n: number): string {
   301→  if (!Number.isFinite(n)) return '0';
   302→  if (n >= 1e6) return `${(n / 1e6).toFixed(1)}M`;
   303→  if (n >= 1e3) return `${(n / 1e3).toFixed(1)}K`;
   304→  return String(n);
   305→}
   306→
   307→function fmtUsd(n: number): string {
   308→  if (!Number.isFinite(n)) return '$0';
   309→  if (n < 0.01) return `$${n.toFixed(6)}`;
   310→  if (n < 1) return `$${n.toFixed(4)}`;
   311→  return `$${n.toFixed(2)}`;
   312→}

   367→async function fetchJson<T>(url: string, init?: RequestInit): Promise<T> {
   368→  const res = await fetch(url, {
   369→    headers: { 'Content-Type': 'application/json' },
   370→    ...init,
   371→  });
   372→  const data = await res.json().catch(() => ({ error: 'Invalid JSON' }));
   373→  if (!res.ok) {
   374→    throw new Error((data as { error?: string }).error || `Request failed (${res.status})`);
   375→  }
   376→  return data as T;
   377→}
