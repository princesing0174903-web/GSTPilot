    78→// ─── Permission-error detection (graceful degradation) ──────────────────
    79→// Mirrors the helper in src/hooks/use-firestore.ts so the Settings page can
    80→// detect Firestore permission-denied errors and surface a friendly message
    81→// instead of the raw "Missing or insufficient permissions" wall.
    82→function isPermissionError(err: unknown): boolean {
    83→  if (!err) return false
    84→  const e = err as { code?: string; message?: string; name?: string }
    85→  const code = (e.code || '').toLowerCase()
    86→  const message = (e.message || '').toLowerCase()
    87→  if (
    88→    code === 'permission-denied' ||
    89→    code === 'unauthenticated' ||
    90→    code === 'auth/operation-not-allowed' ||
    91→    code === 'auth/user-not-found'
    92→  ) {
    93→    return true
    94→  }
    95→  if (
    96→    message.includes('missing or insufficient permissions') ||
    97→    message.includes('permission-denied') ||
    98→    message.includes('insufficient permissions') ||
    99→    message.includes('not authorized') ||
   100→    message.includes('unauthenticated')
   101→  ) {
   102→    return true
   103→  }
   104→  return false
   105→}

   234→  const [state, setState] = useState<'idle' | 'saving' | 'saved'>('idle')
   235→
   236→  const handleClick = async () => {
   237→    setState('saving')
   238→    try {
   239→      await onSave()
   240→      setState('saved')
   241→      setTimeout(() => setState('idle'), 2000)
   242→    } catch {
   243→      setState('idle')
   244→    }
   245→  }
   246→
   247→  return (
   248→    <Button
   249→      onClick={handleClick}
   250→      disabled={state === 'saving'}
   251→      className="bg-emerald-600 hover:bg-emerald-700 text-white gap-2 min-w-[110px]"
   252→    >
   253→      <AnimatePresence mode="wait">
   254→        {state === 'idle' && (
   255→          <motion.span
   256→            key="idle"
   257→            initial={{ opacity: 0, scale: 0.8 }}
   258→            animate={{ opacity: 1, scale: 1 }}
   259→            exit={{ opacity: 0, scale: 0.8 }}
   260→            className="flex items-center gap-2"
   261→          >
   262→            <Save className="h-4 w-4" />
   263→            Save

   295→// ─── Password Strength Helper ──────────────────────────────────────────
   296→function getPasswordStrength(password: string): { score: number; label: string; color: string } {
   297→  if (!password) return { score: 0, label: '', color: '' }
   298→  let score = 0
   299→  if (password.length >= 8) score++
   300→  if (password.length >= 12) score++
   301→  if (/[A-Z]/.test(password)) score++
   302→  if (/[0-9]/.test(password)) score++
   303→  if (/[^A-Za-z0-9]/.test(password)) score++
   304→
   305→  if (score <= 1) return { score: 20, label: 'Weak', color: 'bg-red-500' }
   306→  if (score <= 2) return { score: 40, label: 'Fair', color: 'bg-orange-500' }
   307→  if (score <= 3) return { score: 60, label: 'Good', color: 'bg-yellow-500' }
   308→  if (score <= 4) return { score: 80, label: 'Strong', color: 'bg-emerald-500' }
   309→  return { score: 100, label: 'Excellent', color: 'bg-emerald-600' }
   310→}
