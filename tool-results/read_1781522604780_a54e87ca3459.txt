   235→   235→// ─── Portfolio Derivation Helpers ──────────────────────────────────────────────
   236→   236→
   237→   237→/** Compute a ClientPortfolio from an API client by deriving portfolio metrics */
   238→   238→function derivePortfolio(
   239→   239→  client: ApiClient,
   240→   240→  clientFilings: Array<{ returnType: string; period: string; status: string; filedDate?: string; totalTaxableValue: number }>
   241→   241→): ClientPortfolio {
   242→   242→  const hs = client.healthScore;
   243→   243→  const pendingFilings = client._aggregations?.pendingReturns ?? clientFilings.filter(f => f.status !== 'filed').length;
   244→   244→  const taxVolume = clientFilings.reduce((sum, f) => sum + f.totalTaxableValue, 0);
   245→   245→  const matchRate = client._aggregations?.matchPercentage ?? 0;
   246→   246→
   247→   247→  // Derive risk level from health score
   248→   248→  const riskLevel: RiskLevel = hs > 80 ? 'Low' : hs >= 50 ? 'Medium' : 'High';
   249→   249→  const riskDetail = riskLevel === 'High' ? (pendingFilings > 3 ? 'Multiple delays' : 'Filing delays')
   250→   250→    : riskLevel === 'Medium' ? `${pendingFilings} pending filing${pendingFilings !== 1 ? 's' : ''}`
   251→   251→    : undefined;
   252→   252→
   253→   253→  // Derive health breakdown from health score and data
   254→   254→  const gstinValidity = Math.min(100, hs + 3);
   255→   255→  const filingTimeliness = pendingFilings === 0 ? Math.min(100, hs + 5) : Math.max(10, hs - pendingFilings * 8);
   256→   256→  const invoiceAccuracy = Math.min(100, Math.max(10, matchRate > 0 ? matchRate : hs > 80 ? 94 : hs < 50 ? 58 : 78));
   257→   257→
   258→   258→  // Recent filings (last 3 filings)
   259→   259→  const recentFilings = clientFilings.slice(0, 3).map(f => {
   260→   260→    const monthNames = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];
   261→   261→    const [year, month] = f.period.split('-').map(Number);
   262→   262→    const label = `${monthNames[month - 1]} ${year}`;
   263→   263→    return {
   264→   264→      period: label,
   265→   265→      type: f.returnType,
   266→   266→      status: f.status === 'filed' ? 'Filed' : f.filedDate ? 'Filed' : pendingFilings > 2 ? 'Overdue' : 'Pending',
   267→   267→      date: f.filedDate ?? '',
   268→   268→    };
   269→   269→  });
   270→   270→
   271→   271→  // Monthly volume chart — derive from filings' taxable values
   272→   272→  const monthlyVolumeChart: MonthlyVolume[] = [
   273→   273→    { month: 'Dec', amount: Math.round(taxVolume * 0.85) },
   274→   274→    { month: 'Jan', amount: Math.round(taxVolume * 0.92) },
   275→   275→    { month: 'Feb', amount: Math.round(taxVolume * 0.88) },
   276→   276→    { month: 'Mar', amount: Math.round(taxVolume * 1.1) },
   277→   277→    { month: 'Apr', amount: Math.round(taxVolume * 0.95) },
   278→   278→    { month: 'May', amount: taxVolume },
   279→   279→  ];
   280→   280→
   281→   281→  return {
   282→   282→    id: client.id,
   283→   283→    gstin: client.gstin,
   284→   284→    tradeName: client.tradeName,
   285→   285→    legalName: client.legalName ?? undefined,
   286→   286→    state: client.state ?? undefined,
   287→   287→    stateCode: client.stateCode ?? undefined,
   288→   288→    entityType: client.entityType,
   289→   289→    returnPeriod: client.returnPeriod ?? undefined,
   290→   290→    lastFilingDate: client.lastFilingDate ?? undefined,
   291→   291→    status: client.status,
   292→   292→    healthScore: client.healthScore,
   293→   293→    contactEmail: client.contactEmail ?? undefined,
   294→   294→    contactPhone: client.contactPhone ?? undefined,
   295→   295→    createdAt: client.createdAt,
   296→   296→    updatedAt: client.updatedAt,
   297→   297→    _portfolio: {
   298→   298→      monthlyTaxVolume: taxVolume,
   299→   299→      monthlyVolumeChart,
   300→   300→      pendingFilings,
   301→   301→      riskLevel,
   302→   302→      riskDetail,
   303→   303→      healthBreakdown: { gstinValidity, filingTimeliness, invoiceAccuracy },
   304→   304→      recentFilings,
   305→   305→    },
   306→   306→  };
   307→   307→}

   309→   309→// ─── GSTIN Form Validation ──────────────────────────────────────────────────
   310→   310→
   311→   311→function validateGSTINForm(gstin: string): { valid: boolean; error?: string } {
   312→   312→  if (!gstin) return { valid: false, error: 'GSTIN is required' };
   313→   313→  const clean = formatGSTIN(gstin);
   314→   314→  if (clean.length !== 15) return { valid: false, error: 'GSTIN must be 15 characters' };
   315→   315→  if (!validateGSTIN(clean)) return { valid: false, error: 'Invalid GSTIN format (e.g. 27AABCS1234F1ZH)' };
   316→   316→  const stateCode = clean.slice(0, 2);
   317→   317→  const validCodes = INDIAN_STATES.map(s => s.code);
   318→   318→  if (!validCodes.includes(stateCode)) return { valid: false, error: 'Invalid state code in GSTIN' };
   319→   319→  return { valid: true };
   320→   320→}

   409→   409→function HealthRing({ score, size = 60, strokeWidth = 5, showLabel = true }: { score: number; size?: number; strokeWidth?: number; showLabel?: boolean }) {
   410→   410→  const radius = (size - strokeWidth) / 2;
   411→   411→  const circumference = 2 * Math.PI * radius;
   412→   412→  const offset = circumference - (score / 100) * circumference;
   413→   413→  const color = getHealthColor(score);
   414→   414→
   415→   415→  return (
   416→   416→    <div className="relative inline-flex items-center justify-center" style={{ width: size, height: size }}>
   417→   417→      <svg width={size} height={size} className="-rotate-90">
   418→   418→        <circle
   419→   419→          cx={size / 2}
   420→   420→          cy={size / 2}
   421→   421→          r={radius}
   422→   422→          fill="none"
   423→   423→          stroke="#e2e8f0"
   424→   424→          strokeWidth={strokeWidth}
   425→   425→        />
   426→   426→        <motion.circle
   427→   427→          cx={size / 2}
   428→   428→          cy={size / 2}
   429→   429→          r={radius}
   430→   430→          fill="none"
   431→   431→          stroke={color.stroke}
   432→   432→          strokeWidth={strokeWidth}
   433→   433→          strokeLinecap="round"
   434→   434→          strokeDasharray={circumference}
   435→   435→          initial={{ strokeDashoffset: circumference }}
   436→   436→          animate={{ strokeDashoffset: offset }}
   437→   437→          transition={{ duration: 1, delay: 0.3, ease: 'easeOut' }}
   438→   438→        />
   439→   439→      </svg>
   440→   440→      {showLabel && (
   441→   441→        <div className="absolute inset-0 flex flex-col items-center justify-center">
   442→   442→          <span className={`font-bold leading-none ${size >= 100 ? 'text-2xl' : 'text-sm'} ${color.text}`}>
   443→   443→            {score}
   444→   444→          </span>
   445→   445→          {size >= 100 && (
   446→   446→            <span className="text-[9px] text-muted-foreground mt-0.5">/ 100</span>
   447→   447→          )}
   448→   448→        </div>
   449→   449→      )}
   450→   450→    </div>
   451→   451→  );
   452→   452→}

   469→   469→function FilingTrendChart({ data }: { data: MonthlyVolume[] }) {
   470→   470→  const width = 260;
   471→   471→  const height = 80;
   472→   472→  const padding = { top: 8, right: 8, bottom: 20, left: 8 };
   473→   473→  const chartW = width - padding.left - padding.right;
   474→   474→  const chartH = height - padding.top - padding.bottom;
   475→   475→  const maxAmount = Math.max(...data.map(d => d.amount), 1);
   476→   476→
   477→   477→  const points = data.map((d, i) => ({
   478→   478→    x: padding.left + (i / (data.length - 1)) * chartW,
   479→   479→    y: padding.top + chartH - (d.amount / maxAmount) * chartH,
   480→   480→    ...d,
   481→   481→  }));
   482→   482→
   483→   483→  const linePath = points.map((p, i) => `${i === 0 ? 'M' : 'L'} ${p.x} ${p.y}`).join(' ');
   484→   484→  const areaPath = `${linePath} L ${points[points.length - 1].x} ${padding.top + chartH} L ${points[0].x} ${padding.top + chartH} Z`;
   485→   485→
   486→   486→  const [hoverIdx, setHoverIdx] = useState<number | null>(null);
   487→   487→
   488→   488→  return (
   489→   489→    <div className="relative">
   490→   490→      <svg width="100%" viewBox={`0 0 ${width} ${height}`} className="overflow-visible">
   491→   491→        {/* Area fill */}
   492→   492→        <motion.path
   493→   493→          d={areaPath}
   494→   494→          fill="url(#areaGradient)"
   495→   495→          initial={{ opacity: 0 }}
   496→   496→          animate={{ opacity: 1 }}
   497→   497→          transition={{ duration: 0.6, delay: 0.3 }}
   498→   498→        />
   499→   499→        {/* Line */}
   500→   500→        <motion.path
   501→   501→          d={linePath}
   502→   502→          fill="none"
   503→   503→          stroke="#10b981"
   504→   504→          strokeWidth={2}
   505→   505→          strokeLinecap="round"
   506→   506→          strokeLinejoin="round"
   507→   507→          initial={{ pathLength: 0 }}
   508→   508→          animate={{ pathLength: 1 }}
   509→   509→          transition={{ duration: 0.8, delay: 0.2 }}
   510→   510→        />
   511→   511→        {/* Dots */}
   512→   512→        {points.map((p, i) => (
   513→   513→          <circle
   514→   514→            key={i}
   515→   515→            cx={p.x}
   516→   516→            cy={p.y}
   517→   517→            r={hoverIdx === i ? 4 : 2.5}
   518→   518→            fill="#10b981"
   519→   519→            stroke="white"
   520→   520→            strokeWidth={2}
   521→   521→            className="transition-all duration-150 cursor-pointer"
   522→   522→            onMouseEnter={() => setHoverIdx(i)}
   523→   523→            onMouseLeave={() => setHoverIdx(null)}
   524→   524→          />
   525→   525→        ))}
   526→   526→        {/* Month labels */}
   527→   527→        {points.map((p, i) => (
   528→   528→          <text
   529→   529→            key={i}
   530→   530→            x={p.x}
   531→   531→            y={height - 4}
   532→   532→            textAnchor="middle"
   533→   533→            className="fill-muted-foreground"
   534→   534→            fontSize="8"
   535→   535→          >
   536→   536→            {data[i].month}
   537→   537→          </text>
   538→   538→        ))}
   539→   539→        <defs>
   540→   540→          <linearGradient id="areaGradient" x1="0" y1="0" x2="0" y2="1">
   541→   541→            <stop offset="0%" stopColor="#10b981" stopOpacity="0.2" />
   542→   542→            <stop offset="100%" stopColor="#10b981" stopOpacity="0.02" />
   543→   543→          </linearGradient>
   544→   544→        </defs>
   545→   545→      </svg>
   546→   546→      {/* Hover tooltip */}
   547→   547→      {hoverIdx !== null && (
   548→   548→        <div
   549→   549→          className="absolute -top-8 px-2 py-1 rounded bg-slate-800 text-white text-[10px] font-medium pointer-events-none whitespace-nowrap z-10"
   550→   550→          style={{ left: `${((points[hoverIdx].x) / width) * 100}%`, transform: 'translateX(-50%)' }}
   551→   551→        >
   552→   552→          {formatCurrency(data[hoverIdx].amount)}
   553→   553→        </div>
   554→   554→      )}
   555→   555→    </div>
   556→   556→  );
   557→   557→}
