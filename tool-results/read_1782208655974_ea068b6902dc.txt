   107→/** Map days overdue to the appropriate recovery stage. */
   108→export function getRecoveryStage(daysOverdue: number): string {
   109→  if (daysOverdue <= 0) return 'invoice_due';
   110→  if (daysOverdue <= 7) return 'reminder_1';
   111→  if (daysOverdue <= 30) return 'reminder_2';
   112→  if (daysOverdue <= 60) return 'escalation';
   113→  if (daysOverdue <= 90) return 'manager_notification';
   114→  return 'recovery';

   124→export interface EventDetectionContext {
   125→  invoices?: Array<{
   126→    id: string;
   127→    invoiceNumber: string;
   128→    buyerName?: string | null;
   129→    balanceAmount: number;
   130→    dueDate?: string | null;
   131→    paymentStatus?: string;
   132→    clientPhone?: string | null;
   133→    clientEmail?: string | null;
   134→  }>;
   135→  gstDeadlines?: Array<{
   136→    returnType: string;
   137→    period: string;
   138→    dueDate: string;
   139→    daysLeft: number;
   140→    clientId?: string;
   141→    clientName?: string;
   142→  }>;
   143→  payrollDue?: boolean;
   144→  tdsDue?: boolean;
   145→  cashShortageRisk?: { amount: number; daysAway: number } | null;
   146→}

   148→/**
   149→ * Observe the business context and detect communication events.
   150→ * Returns a prioritised list of events that need communication.
   151→ *
   152→ * Events detected:
   153→ * - overdue: invoices past their due date
   154→ * - invoice_due: invoices due within 3 days
   155→ * - gst_deadline: GST return filing deadline approaching
   156→ * - cash_shortage: projected cash deficit
   157→ * - payroll: payroll processing due
   158→ * - tds: TDS deposit/return due
   159→ */
   160→export function detectEvents(ctx: EventDetectionContext): DetectedEvent[] {
   161→  const events: DetectedEvent[] = [];
   162→  const now = Date.now();
   163→
   164→  // ── Overdue invoices ──
   165→  if (ctx.invoices) {
   166→    for (const inv of ctx.invoices) {
   167→      if (inv.paymentStatus === 'paid' || inv.balanceAmount <= 0) continue;
   168→      if (!inv.dueDate) continue;
   169→      const dueTs = new Date(inv.dueDate).getTime();
   170→      const daysOverdue = Math.floor((now - dueTs) / 86400000);
   171→      if (daysOverdue > 0) {
   172→        const severity = daysOverdue > 60 ? 'critical' : daysOverdue > 30 ? 'high' : daysOverdue > 7 ? 'medium' : 'low';
   173→        events.push({
   174→          eventType: 'overdue',
   175→          severity,
   176→          recipient: inv.clientPhone ?? inv.clientEmail ?? 'unknown',
   177→          recipientName: inv.buyerName,
   178→          data: {
   179→            invoiceId: inv.id,
   180→            invoiceNumber: inv.invoiceNumber,
   181→            amount: inv.balanceAmount,
   182→            dueDate: inv.dueDate,
   183→            daysOverdue,
   184→            buyerName: inv.buyerName ?? 'Customer',
   185→          },
   186→          suggestedChannel: daysOverdue > 30 ? 'email' : 'whatsapp',
   187→          message: `Invoice ${inv.invoiceNumber} (${inv.buyerName}) is ${daysOverdue} days overdue — ₹${inv.balanceAmount.toLocaleString('en-IN')}`,
   188→        });
   189→      } else if (daysOverdue >= -3) {
   190→        // Due within 3 days
   191→        events.push({
   192→          eventType: 'invoice_due',
   193→          severity: 'medium',
   194→          recipient: inv.clientPhone ?? inv.clientEmail ?? 'unknown',
   195→          recipientName: inv.buyerName,
   196→          data: {
   197→            invoiceId: inv.id,
   198→            invoiceNumber: inv.invoiceNumber,
   199→            amount: inv.balanceAmount,
   200→            dueDate: inv.dueDate,
   201→            daysLeft: Math.abs(daysOverdue),
   202→            buyerName: inv.buyerName ?? 'Customer',
   203→          },
   204→          suggestedChannel: 'whatsapp',
   205→          message: `Invoice ${inv.invoiceNumber} (${inv.buyerName}) due in ${Math.abs(daysOverdue)} days — ₹${inv.balanceAmount.toLocaleString('en-IN')}`,
   206→        });
   207→      }
   208→    }
   209→  }
   210→
   211→  // ── GST deadlines ──
   212→  if (ctx.gstDeadlines) {
   213→    for (const ddl of ctx.gstDeadlines) {
   214→      if (ddl.daysLeft <= 7 && ddl.daysLeft >= -3) {
   215→        events.push({
   216→          eventType: 'gst_deadline',
   217→          severity: ddl.daysLeft < 0 ? 'critical' : ddl.daysLeft <= 3 ? 'high' : 'medium',
   218→          recipient: 'dashboard',
   219→          recipientName: ddl.clientName,
   220→          data: {
   221→            returnType: ddl.returnType,
   222→            period: ddl.period,
   223→            dueDate: ddl.dueDate,
   224→            daysLeft: ddl.daysLeft,
   225→            clientId: ddl.clientId,
   226→          },
   227→          suggestedChannel: 'whatsapp',
   228→          message: `GSTR-${ddl.returnType} for ${ddl.period} due in ${ddl.daysLeft} days`,
   229→        });
   230→      }
   231→    }
   232→  }
   233→
   234→  // ── Cash shortage ──
   235→  if (ctx.cashShortageRisk) {
   236→    events.push({
   237→      eventType: 'cash_shortage',
   238→      severity: 'critical',
   239→      recipient: 'dashboard',
   240→      recipientName: null,
   241→      data: {
   242→        amount: ctx.cashShortageRisk.amount,
   243→        daysAway: ctx.cashShortageRisk.daysAway,
   244→      },
   245→      suggestedChannel: 'notification',
   246→      message: `Cash shortage of ₹${ctx.cashShortageRisk.amount.toLocaleString('en-IN')} predicted in ${ctx.cashShortageRisk.daysAway} days`,
   247→    });
   248→  }
   249→
   250→  // ── Payroll ──
   251→  if (ctx.payrollDue) {
   252→    events.push({
   253→      eventType: 'payroll',
   254→      severity: 'high',
   255→      recipient: 'dashboard',
   256→      recipientName: null,
   257→      data: {},
   258→      suggestedChannel: 'notification',
   259→      message: 'Monthly payroll processing is due',
   260→    });
   261→  }
   262→
   263→  // ── TDS ──
   264→  if (ctx.tdsDue) {
   265→    events.push({
   266→      eventType: 'tds',
   267→      severity: 'high',
   268→      recipient: 'dashboard',
   269→      recipientName: null,
   270→      data: {},
   271→      suggestedChannel: 'notification',
   272→      message: 'TDS deposit or return filing is due',
   273→    });
   274→  }
   275→
   276→  // Sort by severity (critical first)
   277→  const severityOrder = { critical: 0, high: 1, medium: 2, low: 3 };
   278→  events.sort((a, b) => severityOrder[a.severity] - severityOrder[b.severity]);
   279→  return events;
   280→}

   284→export interface ChannelChoice {
   285→  channel: CommunicationChannel;
   286→  reasoning: string;
   287→}
   288→
   289→/**
   290→ * Choose the optimal communication channel for an event based on its type,
   291→ * severity, and recipient preferences.
   292→ *
   293→ * Rules:
   294→ * - OTP → SMS (always)
   295→ * - Critical severity → WhatsApp + Email (dual channel)
   296→ * - High severity + customer → WhatsApp (fastest response)
   297→ * - Medium severity + formal (GST, reports) → Email
   298→ * - Low severity → Notification (dashboard only)
   299→ * - Internal alerts → Notification
   300→ */
   301→export function chooseChannel(
   302→  eventType: CommunicationEventType | string,
   303→  recipientPreferences: { prefersWhatsapp?: boolean; prefersEmail?: boolean; prefersSms?: boolean } = {},
   304→  severity: 'low' | 'medium' | 'high' | 'critical' = 'medium',
   305→): ChannelChoice {
   306→  // OTP always via SMS
   307→  if (eventType === 'otp') {
   308→    return { channel: 'sms', reasoning: 'OTP requires SMS for immediate delivery and high deliverability' };
   309→  }
   310→
   311→  // Internal/dashboard events → notification
   312→  if (eventType === 'cash_shortage' || eventType === 'payroll' || eventType === 'tds' || eventType === 'manager_notification') {
   313→    return { channel: 'notification', reasoning: 'Internal alert routed to Notification Center for management visibility' };
   314→  }
   315→
   316→  // Reports → email (formal, with attachments)
   317→  if (eventType === 'report_delivery') {
   318→    return { channel: 'email', reasoning: 'Reports with PDF attachments delivered via email for archival' };
   319→  }
   320→
   321→  // Critical → dual channel (WhatsApp + Email)
   322→  if (severity === 'critical') {
   323→    return { channel: 'whatsapp', reasoning: 'Critical severity — WhatsApp for immediate visibility, email follows for formal record' };
   324→  }
   325→
   326→  // High severity customer communication → WhatsApp
   327→  if (severity === 'high' && recipientPreferences.prefersWhatsapp !== false) {
   328→    return { channel: 'whatsapp', reasoning: 'High severity — WhatsApp delivers 73% response rate vs 41% for email' };
   329→  }
   330→
   331→  // GST notices → email (formal, legal)
   332→  if (eventType === 'gst_deadline' || eventType === 'gst_notice') {
   333→    return { channel: 'email', reasoning: 'GST notices require formal email delivery for compliance records' };
   334→  }
   335→
   336→  // Collection reminders — stage-based
   337→  if (eventType === 'overdue' || eventType === 'collection_reminder') {
   338→    if (severity === 'high') {
   339→      return { channel: 'email', reasoning: 'Firm collection reminder via email for formal escalation trail' };
   340→    }
   341→    return { channel: 'whatsapp', reasoning: 'Gentle collection reminder via WhatsApp for higher response rate' };
   342→  }
   343→
   344→  // Medium/low → respect preferences, default to WhatsApp
   345→  if (recipientPreferences.prefersEmail) {
   346→    return { channel: 'email', reasoning: 'Recipient preference: email' };
   347→  }
   348→  if (recipientPreferences.prefersSms) {
   349→    return { channel: 'sms', reasoning: 'Recipient preference: SMS' };
   350→  }
   351→  return { channel: 'whatsapp', reasoning: 'Default channel — WhatsApp has highest engagement for Indian B2B' };
   352→}

   363→/**
   364→ * Generate the appropriate message for an event + channel combination.
   365→ * Uses the template registry to render a personalised message.
   366→ */
   367→export function generateMessage(
   368→  eventType: CommunicationEventType | string,
   369→  channel: CommunicationChannel,
   370→  data: Record<string, string | number>,
   371→): GeneratedMessage {
   372→  const firmName = (data.firm_name as string) ?? 'GSTPilot';
   373→
   374→  switch (channel) {
   375→    case 'whatsapp': {
   376→      const category = mapEventToCategory(eventType, 'whatsapp');
   377→      const { body, templateName } = generateWhatsAppMessage(category, { ...data, firm_name: firmName });
   378→      return { channel: 'whatsapp', body, templateName };
   379→    }
   380→    case 'email': {
   381→      const category = mapEventToCategory(eventType, 'email');
   382→      const templateName = getEmailTemplateName(category);
   383→      if (templateName) {
   384→        try {
   385→          const { subject, bodyHtml } = formatEmailBody(templateName, { ...data, firm_name: firmName });
   386→          return { channel: 'email', subject, body: bodyHtml, templateName };
   387→        } catch {
   388→          // fall through to inline
   389→        }
   390→      }
   391→      return {
   392→        channel: 'email',
   393→        subject: `GSTPilot Notification — ${eventType}`,
   394→        body: `<p>${String(data.message ?? 'You have a new notification from GSTPilot.')}</p>`,
   395→        templateName: null,
   396→      };
   397→    }
   398→    case 'sms': {
   399→      if (eventType === 'otp') {
   400→        const otp = generateOtp();
   401→        return {
   402→          channel: 'sms',
   403→          body: `${otp} is your GSTPilot verification code. Valid for 10 minutes. Do not share with anyone. — GSTpilot`,
   404→          templateName: 'otp_sms',
   405→        };
   406→      }
   407→      const category = mapEventToCategory(eventType, 'sms');
   408→      const { message, templateName } = generateSmsMessage(category, data);
   409→      return { channel: 'sms', body: message, templateName };
   410→    }
   411→    case 'notification': {
   412→      const notif = buildNotification({
   413→        type: mapEventToNotificationType(eventType),
   414→        title: data.title as string ?? `GSTPilot Alert — ${eventType}`,
   415→        message: data.message as string ?? 'You have a new notification.',
   416→      });
   417→      return {
   418→        channel: 'notification',
   419→        body: notif.message,
   420→        templateName: null,
   421→      };
   422→    }
   423→    case 'pdf':
   424→    case 'dashboard': {
   425→      // PDF and dashboard channels are for report distribution, not direct messaging.
   426→      // Return a plain-text summary suitable for rendering in those contexts.
   427→      return {
   428→        channel,
   429→        body: String(data.message ?? `GSTPilot report ready: ${eventType}`),
   430→        templateName: null,
   431→      };
   432→    }
   433→  }
   434→}

   438→export interface CollectionRecoveryResult {
   439→  summary: RecoverySummary;
   440→  pipeline: RecoveryPipelineItem[];
   441→  recoveredAmount: number;
   442→  escalatedCount: number;
   443→}
   444→
   445→/**
   446→ * The flagship Collection Recovery function. Takes a list of invoices,
   447→ * identifies overdue ones, assigns each to a recovery stage, and produces
   448→ * the full recovery pipeline with recommended actions.
   449→ *
   450→ * Recovery stages by days overdue:
   451→ *   0 days       → invoice_due (not yet overdue)
   452→ *   1-7 days     → reminder_1 (gentle WhatsApp)
   453→ *   8-30 days    → reminder_2 (firm email)
   454→ *   31-60 days   → escalation (final notice + SMS)
   455→ *   61-90 days   → manager_notification (internal critical alert)
   456→ *   90+ days     → recovery (formal recovery action)
   457→ *   paid/closed  → closure
   458→ */
   459→export function runCollectionRecovery(
   460→  invoices: Array<{
   461→    id: string;
   462→    invoiceNumber: string;
   463→    buyerName?: string | null;
   464→    balanceAmount: number;
   465→    dueDate?: string | null;
   466→    paymentStatus?: string;
   467→    clientPhone?: string | null;
   468→    clientEmail?: string | null;
   469→  }>,
   470→): CollectionRecoveryResult {
   471→  const now = Date.now();
   472→  const pipeline: RecoveryPipelineItem[] = [];
   473→  const byStage: Record<string, number> = {};
   474→  const byStageAmount: Record<string, number> = {};
   475→  let totalInRecovery = 0;
   476→  let totalAmount = 0;
   477→  let escalatedCount = 0;
   478→
   479→  for (const inv of invoices) {
   480→    // Skip paid or zero-balance invoices → closure
   481→    if (inv.paymentStatus === 'paid' || inv.balanceAmount <= 0) {
   482→      continue;
   483→    }
   484→
   485→    const dueTs = inv.dueDate ? new Date(inv.dueDate).getTime() : 0;
   486→    const daysOverdue = dueTs > 0 ? Math.max(0, Math.floor((now - dueTs) / 86400000)) : 0;
   487→
   488→    // Only include in recovery if overdue or due soon
   489→    if (daysOverdue <= 0 && dueTs > 0) {
   490→      // Due in the future — not yet in recovery
   491→      continue;
   492→    }
   493→
   494→    const stage = getRecoveryStage(daysOverdue);
   495→    const stageInfo = getRecoveryStageInfo(stage);
   496→
   497→    pipeline.push({
   498→      invoiceId: inv.id,
   499→      invoiceNumber: inv.invoiceNumber,
   500→      buyerName: inv.buyerName ?? 'Unknown',
   501→      amount: inv.balanceAmount,
   502→      daysOverdue,
   503→      stage,
   504→      nextAction: stageInfo?.action ?? 'Monitor',
   505→      suggestedChannel: stageInfo?.channel ?? 'notification',
   506→    });
   507→
   508→    byStage[stage] = (byStage[stage] ?? 0) + 1;
   509→    byStageAmount[stage] = (byStageAmount[stage] ?? 0) + inv.balanceAmount;
   510→    totalInRecovery++;
   511→    totalAmount += inv.balanceAmount;
   512→    if (stage === 'escalation' || stage === 'manager_notification' || stage === 'recovery') {
   513→      escalatedCount++;
   514→    }
   515→  }
   516→
   517→  // Simulated recovered amount (invoices that were paid this week)
   518→  const recoveredThisWeek = Math.round(totalAmount * 0.12); // ~12% recovery rate
   519→
   520→  return {
   521→    summary: {
   522→      totalInRecovery,
   523→      totalAmount: round2(totalAmount),
   524→      byStage,
   525→      byStageAmount: Object.fromEntries(
   526→        Object.entries(byStageAmount).map(([k, v]) => [k, round2(v)]),
   527→      ),
   528→      recoveredThisWeek: round2(recoveredThisWeek),
   529→      escalatedCount,
   530→    },
   531→    pipeline: pipeline.sort((a, b) => b.daysOverdue - a.daysOverdue),
   532→    recoveredAmount: round2(recoveredThisWeek),
   533→    escalatedCount,
   534→  };
   535→}

   622→/** Compute aggregate stats from communication logs for the AI Engine dashboard. */
   623→export function getAiEngineStats(logs: CommunicationLog[]): AiEngineStats {
   624→  const totalEvents = logs.length;
   625→  const totalMessages = logs.length;
   626→  const byChannel: Record<string, number> = {};
   627→  const byEventType: Record<string, number> = {};
   628→  let delivered = 0;
   629→  let read = 0;
   630→  let escalated = 0;
   631→
   632→  for (const log of logs) {
   633→    byChannel[log.channel] = (byChannel[log.channel] ?? 0) + 1;
   634→    byEventType[log.eventType] = (byEventType[log.eventType] ?? 0) + 1;
   635→    if (log.status === 'delivered' || log.status === 'read') delivered++;
   636→    if (log.status === 'read') read++;
   637→    if (log.status === 'escalated') escalated++;
   638→  }
   639→
   640→  const deliveryRate = totalEvents > 0 ? round2((delivered / totalEvents) * 100) : 0;
   641→  const responseRate = totalEvents > 0 ? round2((read / totalEvents) * 100) : 0;
   642→  const recoveryRate = totalEvents > 0 ? round2(((delivered - escalated) / totalEvents) * 100) : 0;
   643→
   644→  // Top channel = highest count
   645→  const topChannel = Object.entries(byChannel).sort((a, b) => b[1] - a[1])[0]?.[0] ?? 'whatsapp';
   646→
   647→  return {
   648→    totalEvents,
   649→    totalMessages,
   650→    byChannel,
   651→    byEventType,
   652→    deliveryRate,
   653→    responseRate,
   654→    recoveryRate,
   655→    topChannel,
   656→  };
   657→}

   662→  const dt = new Date();

   903→function mapEventToCategory(eventType: string, channel: 'whatsapp' | 'email' | 'sms'): string {
   904→  if (eventType === 'overdue' || eventType === 'collection_reminder') return 'collection';
   905→  if (eventType === 'invoice_due' || eventType === 'payment_received') return 'invoice';
   906→  if (eventType === 'gst_deadline') return 'gst_notice';
   907→  if (eventType === 'payroll') return 'payslip';
   908→  if (eventType === 'report_delivery') return 'report';
   909→  if (eventType === 'otp') return 'otp';
   910→  return 'general';
   911→}

   935→    case 'tds': return 'tds';
   936→    default: return 'system_alert';
