   157→function getKanbanColumn(status: FilingStatus): KanbanColumn {
   158→  switch (status) {
   159→    case 'draft':
   160→    case 'prepared':
   161→      return 'draft';
   162→    case 'validated':
   163→    case 'reviewed':
   164→    case 'generated':
   165→      return 'ready';
   166→    case 'filed':
   167→      return 'filed';
   168→    case 'reopened':
   169→      return 'attention';
   170→    default:
   171→      return 'draft';
   172→  }
   173→}
   174→
   175→function getTimelineStepIndex(status: FilingStatus): number {
   176→  switch (status) {
   177→    case 'draft':
   178→    case 'prepared':
   179→      return 0;
   180→    case 'validated':
   181→    case 'reviewed':
   182→      return 1;
   183→    case 'generated':
   184→      return 2;
   185→    case 'filed':
   186→      return 3;
   187→    case 'reopened':
   188→      return -1;
   189→    default:
   190→      return 0;
   191→  }
   192→}
   193→
   194→function getAttentionSummary(filing: GSTRFiling): string {
   195→  if (filing.issuesFound <= 0) return 'No issues';
   196→  const parts: string[] = [];
   197→  if (filing.criticalErrors > 0) parts.push(`${filing.criticalErrors} critical error${filing.criticalErrors > 1 ? 's' : ''}`);
   198→  if (filing.warnings > 0) parts.push(`${filing.warnings} warning${filing.warnings > 1 ? 's' : ''}`);
   199→  if (filing.issuesFound > 0 && parts.length === 0) parts.push(`${filing.issuesFound} issue${filing.issuesFound > 1 ? 's' : ''}`);
   200→  return parts.join(' · ');
   201→}

   210→// ─── Section Breakdown Computation ──────────────────────────────────────────────
   211→
   212→function computeSectionsForFiling(
   213→  returnType: string,
   214→  invoices: { taxableValue: number; cgst: number; sgst: number; igst: number; buyerGstin?: string | null; status: string }[]
   215→): SectionBreakdown[] {
   216→  if (returnType === 'GSTR-1') {
   217→    const b2b = invoices.filter(inv => inv.buyerGstin);
   218→    const b2cLarge = invoices.filter(inv => !inv.buyerGstin && inv.taxableValue >= 250000);
   219→    const b2cSmall = invoices.filter(inv => !inv.buyerGstin && inv.taxableValue < 250000);
   220→
   221→    return [
   222→      {
   223→        section: 'B2B',
   224→        invoiceCount: b2b.length,
   225→        taxableValue: b2b.reduce((s, i) => s + i.taxableValue, 0),
   226→        taxAmount: b2b.reduce((s, i) => s + i.cgst + i.sgst + i.igst, 0),
   227→      },
   228→      {
   229→        section: 'B2C Large',
   230→        invoiceCount: b2cLarge.length,
   231→        taxableValue: b2cLarge.reduce((s, i) => s + i.taxableValue, 0),
   232→        taxAmount: b2cLarge.reduce((s, i) => s + i.cgst + i.sgst + i.igst, 0),
   233→      },
   234→      {
   235→        section: 'B2C Small',
   236→        invoiceCount: b2cSmall.length,
   237→        taxableValue: b2cSmall.reduce((s, i) => s + i.taxableValue, 0),
   238→        taxAmount: b2cSmall.reduce((s, i) => s + i.cgst + i.sgst + i.igst, 0),
   239→      },
   240→      {
   241→        section: 'CDNR',
   242→        invoiceCount: 0,
   243→        taxableValue: 0,
   244→        taxAmount: 0,
   245→      },
   246→    ];
   247→  }
   248→
   249→  // GSTR-3B
   250→  const totalTax = invoices.reduce((s, i) => s + i.cgst + i.sgst + i.igst, 0);
   251→  const totalTaxable = invoices.reduce((s, i) => s + i.taxableValue, 0);
   252→  return [
   253→    {
   254→      section: 'Outward Supplies',
   255→      invoiceCount: invoices.length,
   256→      taxableValue: totalTaxable,
   257→      taxAmount: totalTax,
   258→    },
   259→    {
   260→      section: 'Inward Supplies (RC)',
   261→      invoiceCount: 0,
   262→      taxableValue: 0,
   263→      taxAmount: 0,
   264→    },
   265→    {
   266→      section: 'ITC Claims',
   267→      invoiceCount: 0,
   268→      taxableValue: 0,
   269→      taxAmount: Math.round(-totalTax * 0.6),
   270→    },
   271→    {
   272→      section: 'Tax Paid',
   273→      invoiceCount: 0,
   274→      taxableValue: 0,
   275→      taxAmount: Math.round(totalTax * 0.75),
   276→    },
   277→  ];
   278→}
