     1→// ═══════════════════════════════════════════════════════════════════════════════
     2→// GSTPilot Billing, Subscriptions & Payments™ — Client-Safe Firestore Service
     3→//
     4→// The single entry point for all billing Firestore operations on the CLIENT side.
     5→// Mirrors the banking-provider / erp-provider service pattern:
     6→//   • Real-time subscriptions via onSnapshot (org-scoped)
     7→//   • CRUD writes via the Firebase client SDK (rules enforce org isolation)
     8→//   • Multi-tenant — every function filters on `organizationId`
     9→//
    10→// This module is CLIENT-SAFE — it only imports from `firebase/firestore` and
    11→// `@/lib/firebase` (the client SDK). It NEVER imports the provider, crypto, or
    12→// any server-only code.
    13→//
    14→// The flow for each billing operation:
    15→//   1. Client hook calls the appropriate API route (/api/billing/*) for the
    16→//      provider work (subscribe, upgrade, pay, refund, etc.). The API route
    17→//      returns plain data + the encrypted customer id / provider tokens.
    18→//   2. The orchestrator (server-side) writes the result to Firestore directly.
    19→//   3. Real-time onSnapshot subscriptions surface the change to every
    20→//      connected client instantly.
    21→//
    22→// Firestore timeout guard: when the Firebase project has the Firestore API
    23→// disabled, the SDK retries PERMISSION_DENIED for ~120s before rejecting.
    24→// `withTimeout` races every op against a 6s deadline so the UI never hangs.
    25→// ═══════════════════════════════════════════════════════════════════════════════

    94→// ─── Firestore timeout guard ─────────────────────────────────────────────────
    95→
    96→const FIRESTORE_TIMEOUT_MS = 6000;
    97→
    98→function withTimeout<T>(promise: Promise<T>, label: string): Promise<T> {
    99→  promise.catch(() => {
   100→    /* timed-out op — ignore late rejection */
   101→  });
   102→  return Promise.race([
   103→    promise,
   104→    new Promise<T>((_, reject) =>
   105→      setTimeout(
   106→        () =>
   107→          reject(
   108→            new BillingError(
   109→              `${label} timed out — Firestore may be unreachable.`,
   110→              { code: 'FIRESTORE_TIMEOUT', statusCode: 503, retryable: true },
   111→            ),
   112→          ),
   113→        FIRESTORE_TIMEOUT_MS,
   114→      ),
   115→    ),
   116→  ]);
   117→}

   166→/** Subscribe to the org's active subscription (one per org). */
   167→export function subscribeToSubscription(
   168→  organizationId: string,
   169→  onData: (sub: Subscription | null) => void,
   170→  opts?: { onError?: (err: Error) => void },
   171→): Unsubscribe {
   172→  assertOrg(organizationId);
   173→  const q = query(
   174→    collection(db, BILLING_COLLECTIONS.SUBSCRIPTIONS),
   175→    where('organizationId', '==', organizationId),
   176→    limitFn(1),
   177→  );
   178→  return onSnapshot(
   179→    q,
   180→    (snap) => {
   181→      if (snap.empty) {
   182→        onData(null);
   183→        return;
   184→      }
   185→      const d = snap.docs[0];
   186→      onData(toSubscription(d.id, d.data() as Record<string, unknown>));
   187→    },
   188→    (err) => opts?.onError?.(err as Error),
   189→  );
   190→}

   192→export async function getSubscription(organizationId: string): Promise<Subscription | null> {
   193→  assertOrg(organizationId);
   194→  const q = query(
   195→    collection(db, BILLING_COLLECTIONS.SUBSCRIPTIONS),
   196→    where('organizationId', '==', organizationId),
   197→    limitFn(1),
   198→  );
   199→  const snap = await withTimeout(getDocs(q), 'billing.getSubscription');
   200→  if (snap.empty) return null;
   201→  const d = snap.docs[0];
   202→  return toSubscription(d.id, d.data() as Record<string, unknown>);
   203→}

   474→export function subscribeToCoupons(

   478→): Unsubscribe {
   479→  assertOrg(organizationId);
   480→  // Both global (organizationId == null) and org-private coupons.
   481→  const q = query(
   482→    collection(db, BILLING_COLLECTIONS.COUPONS),
   483→    where('isActive', '==', true),
   484→    orderBy('createdAt', 'desc'),
   485→  );
   486→  return onSnapshot(
   487→    q,
   488→    (snap) => {
   489→      const all = snap.docs.map((d) => toCoupon(d.id, d.data() as Record<string, unknown>));
   490→      // Filter: global coupons OR this org's coupons.
   491→      onData(
   492→        all.filter(
   493→          (c) => c.organizationId === null || c.organizationId === organizationId,
   494→        ),
   495→      );
   496→    },
   497→    (err) => opts?.onError?.(err as Error),
   498→  );

   528→  const q = query(collection(db, BILLING_COLLECTIONS.USAGE_RECORDS), ...constraints);
   529→  return onSnapshot(
   530→    q,
   531→    (snap) =>
   532→      onData(snap.docs.map((d) => toUsageRecord(d.id, d.data() as Record<string, unknown>))),
   533→    (err) => opts?.onError?.(err as Error),
   534→  );
   535→}

   538→ * Get the current period's usage records (org-scoped, current billing period).
   539→ * Pure read — no subscription.
   540→ */
   541→export async function getUsageForPeriod(
   542→  organizationId: string,
   543→  periodStart: string,
   544→  periodEnd: string,
   545→): Promise<UsageRecord[]> {

   547→  const q = query(

   555→}

   557→/**
   558→ * Write a usage record directly (used by usage metering on the client for
   559→ * low-stakes events like oracle_requests). High-stakes events go through the

   562→export async function recordUsage(

   589→// ─── Billing Summary (pure function, dashboard-facing) ──────────────────────
   590→
   591→/**
   592→ * Aggregate the org's billing state into a single `BillingSummary` for the
   593→ * dashboard. Pure function — takes the already-subscribed collections and
   594→ * returns the computed summary. The `useBilling()` hook calls this on every
   595→ * real-time update so the dashboard always reflects current data.
   596→ */
   597→export function computeBillingSummary(
   598→  subscription: Subscription | null,
   599→  invoices: BillingInvoice[],
   600→  payments: Payment[],
   601→  usageRecords: UsageRecord[],
   602→): BillingSummary {
   603→  const now = new Date();
   604→  const plan = subscription ? getPlan(subscription.planId) : null;
   605→
   606→  // MRR / ARR
   607→  const mrr = subscription?.mrr ?? 0;
   608→  const arr = subscription?.arr ?? 0;
   609→
   610→  // Renewal date
   611→  const nextRenewalDate = subscription?.currentPeriodEnd ?? null;
   612→  const daysToRenewal = nextRenewalDate
   613→    ? Math.max(
   614→        0,
   615→        Math.round(
   616→          (new Date(nextRenewalDate).getTime() - now.getTime()) / (24 * 60 * 60 * 1000),
   617→        ),
   618→      )
   619→    : null;
   620→
   621→  // Trial days
   622→  const trialDaysLeft = subscription?.trialEndDate
   623→    ? Math.max(
   624→        0,
   625→        Math.round(
   626→          (new Date(subscription.trialEndDate).getTime() - now.getTime()) /
   627→            (24 * 60 * 60 * 1000),
   628→        ),
   629→      )
   630→    : null;
   631→
   632→  // Outstanding amount — sum of unpaid invoices
   633→  const outstandingAmount = invoices
   634→    .filter((i) => i.status === 'sent' || i.status === 'overdue')
   635→    .reduce((sum, i) => sum + i.amountDue, 0);
   636→
   637→  // Failed payments in last 30 days
   638→  const failedPayments = payments.filter(
   639→    (p) =>
   640→      p.status === 'failed' &&
   641→      new Date(p.createdAt).getTime() > now.getTime() - 30 * 24 * 60 * 60 * 1000,
   642→  ).length;
   643→
   644→  // Usage this period — aggregate by metric
   645→  const usageThisPeriod: Record<UsageMetricType, number> = {
   646→    oracle_requests: 0,
   647→    storage_bytes: 0,
   648→    ai_tokens: 0,
   649→    invoices_generated: 0,
   650→    returns_filed: 0,
   651→    api_calls: 0,
   652→    automation_runs: 0,
   653→  };
   654→  const periodStart = subscription?.currentPeriodStart ?? null;
   655→  const periodEnd = subscription?.currentPeriodEnd ?? null;
   656→  for (const r of usageRecords) {
   657→    if (periodStart && periodEnd) {
   658→      // Only count records in the current billing period.
   659→      if (r.periodStart < periodStart || r.periodEnd > periodEnd) continue;
   660→    }
   661→    if (r.metric in usageThisPeriod) {
   662→      usageThisPeriod[r.metric] += r.quantity;
   663→    }
   664→  }
   665→
   666→  // Plan limits
   667→  const usageLimits: Record<UsageMetricType, number> | null = plan
   668→    ? {
   669→        oracle_requests: plan.limits.oracleRequestsMonthly,
   670→        storage_bytes: plan.limits.storageBytes,
   671→        ai_tokens: plan.limits.aiCreditsMonthly,
   672→        invoices_generated: plan.limits.invoicesMonthly,
   673→        returns_filed: plan.limits.returnsMonthly,
   674→        api_calls: plan.limits.apiCallsMonthly,
   675→        automation_runs: plan.limits.automationRunsMonthly,
   676→      }
   677→    : null;
   678→
   679→  // Usage percentage (0..100; Infinity if limit is Infinity)
   680→  const usagePercent: Record<UsageMetricType, number> = {
   681→    oracle_requests: 0,
   682→    storage_bytes: 0,
   683→    ai_tokens: 0,
   684→    invoices_generated: 0,
   685→    returns_filed: 0,
   686→    api_calls: 0,
   687→    automation_runs: 0,
   688→  };
   689→  if (usageLimits) {
   690→    (Object.keys(usageThisPeriod) as UsageMetricType[]).forEach((metric) => {
   691→      const used = usageThisPeriod[metric];
   692→      const limit = usageLimits[metric];
   693→      usagePercent[metric] =
   694→        limit === Infinity ? 0 : Math.min(100, Math.round((used / limit) * 100));
   695→    });
   696→  }
   697→
   698→  // Recent invoices (newest 10)
   699→  const recentInvoices = [...invoices]
   700→    .sort((a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime())
   701→    .slice(0, 10);
   702→
   703→  // Recent payments (newest 10)
   704→  const recentPayments = [...payments]
   705→    .sort((a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime())
   706→    .slice(0, 10);
   707→
   708→  // Connected providers
   709→  const connectedProviders: PaymentProviderName[] = subscription
   710→    ? [subscription.paymentProvider]
   711→    : [];
   712→
   713→  return {
   714→    currentPlan: subscription?.planId ?? null,
   715→    planName: subscription?.planName ?? plan?.name ?? null,
   716→    subscriptionStatus: subscription?.status ?? null,
   717→    mrr,
   718→    arr,
   719→    nextRenewalDate,
   720→    daysToRenewal,
   721→    trialDaysLeft,
   722→    outstandingAmount,
   723→    failedPayments,
   724→    usageThisPeriod,
   725→    usageLimits,
   726→    usagePercent,
   727→    recentInvoices,
   728→    recentPayments,
   729→    connectedProviders,
   730→  };
   731→}
