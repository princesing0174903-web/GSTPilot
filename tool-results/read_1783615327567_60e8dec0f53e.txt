    59→// Relative time formatter — defensively handles ISO string / Date / Firestore Timestamp.
    60→function timeAgo(createdAt: unknown): string {
    61→  if (!createdAt) return 'just now';
    62→  let date: Date;
    63→  if (createdAt instanceof Date) {
    64→    date = createdAt;
    65→  } else if (typeof createdAt === 'string') {
    66→    date = new Date(createdAt);
    67→  } else if (
    68→    typeof createdAt === 'object' &&
    69→    createdAt !== null &&
    70→    'toDate' in createdAt &&
    71→    typeof (createdAt as { toDate: () => Date }).toDate === 'function'
    72→  ) {
    73→    date = (createdAt as { toDate: () => Date }).toDate();
    74→  } else {
    75→    return 'just now';
    76→  }
    77→  if (isNaN(date.getTime())) return 'just now';
    78→  const seconds = Math.floor((Date.now() - date.getTime()) / 1000);
    79→  if (seconds < 60) return 'just now';
    80→  const minutes = Math.floor(seconds / 60);
    81→  if (minutes < 60) return `${minutes}m ago`;
    82→  const hours = Math.floor(minutes / 60);
    83→  if (hours < 24) return `${hours}h ago`;
    84→  const days = Math.floor(hours / 24);
    85→  if (days < 7) return `${days}d ago`;
    86→  const weeks = Math.floor(days / 7);
    87→  if (weeks < 5) return `${weeks}w ago`;
    88→  const months = Math.floor(days / 30);
    89→  if (months < 12) return `${months}mo ago`;
    90→  const years = Math.floor(days / 365);
    91→  return `${years}y ago`;
    92→}

    94→// ─── Business Score computation ───────────────────────────────────────────────
    95→// Start at 100. Subtract weighted penalties for risk signals. Clamp 0–100.
    96→// If there is no data at all, return null (UI shows premium empty state).

    98→function computeBusinessScore(opts: {
    99→  totalClients: number;
   100→  criticalIssues: number;
   101→  warnings: number;
   102→  pendingReturns: number;
   103→  overdueReturns: number;
   104→  averageHealthScore: number;
   105→  matchPercentage: number;
   106→}): number | null {
   107→  const { totalClients, criticalIssues, warnings, pendingReturns, overdueReturns, averageHealthScore, matchPercentage } = opts;
   108→  if (totalClients === 0 && pendingReturns === 0 && overdueReturns === 0) return null;
   109→
   110→  let score = 100;
   111→  score -= criticalIssues * 6;
   112→  score -= warnings * 2;
   113→  score -= overdueReturns * 8;
   114→  score -= pendingReturns * 2;
   115→  if (averageHealthScore > 0) {
   116→    score -= Math.max(0, 80 - averageHealthScore) * 0.6;
   117→  }
   118→  score -= Math.max(0, 100 - matchPercentage) * 0.3;
   119→  return Math.max(0, Math.min(100, Math.round(score)));
   120→}

   122→// V16 palette — only Emerald / Cyan / Blue. No amber, no red.
   123→function scoreTier(score: number | null): { label: string; color: string } {
   124→  if (score === null) return { label: '—', color: 'text-muted-foreground' };
   125→  if (score >= 85) return { label: 'Excellent', color: 'text-emerald-400' };
   126→  if (score >= 70) return { label: 'Healthy', color: 'text-cyan-400' };
   127→  if (score >= 50) return { label: 'At Risk', color: 'text-blue-400' };
   128→  return { label: 'Critical', color: 'text-blue-300' };
   129→}

   145→function buildInsights(opts: {
   146→  metrics: {
   147→    totalClients: number;
   148→    pendingReturns: number;
   149→    overdueReturns: number;
   150→    criticalIssues: number;
   151→    warnings: number;
   152→    totalTaxVolume: number;
   153→    averageHealthScore: number;
   154→    matchPercentage: number;
   155→    upcomingFilings: FirestoreReturn[];
   156→  };
   157→  clients: Array<FirestoreClient & { id: string }>;
   158→  /** True when the firm has real Firestore data. Gates the "all clear" insight
   159→   *  so we never say "Everything looks good today" when there's no data at all. */
   160→  hasData: boolean;
   161→}): AIInsight[] {
   162→  const { metrics, clients, hasData } = opts;
   163→  const insights: AIInsight[] = [];
   164→
   165→  // 1. Overdue returns — most urgent
   166→  if (metrics.overdueReturns > 0) {
   167→    insights.push({
   168→      id: 'overdue-returns',
   169→      icon: AlertTriangle,
   170→      tone: 'risk',
   171→      what: `${metrics.overdueReturns} GST return${metrics.overdueReturns > 1 ? 's are' : ' is'} overdue`,
   172→      why: 'Overdue returns trigger notices and late fees. File immediately to avoid penalties.',
   173→      actionLabel: 'File Now',
   174→      actionView: 'returns',
   175→    });
   176→  }
   177→
   178→  // 2. Pending returns due soon
   179→  const nextFiling = metrics.upcomingFilings[0];
   180→  if (nextFiling && metrics.pendingReturns > 0 && metrics.overdueReturns === 0) {
   181→    insights.push({
   182→      id: 'gst-due',
   183→      icon: Clock,
   184→      tone: 'info',
   185→      what: `GST filing due — ${nextFiling.returnType} for ${nextFiling.period}`,
   186→      why: `${metrics.pendingReturns} return${metrics.pendingReturns > 1 ? 's' : ''} pending. Filing on time keeps compliance score high.`,
   187→      actionLabel: 'File GST',
   188→      actionView: 'returns',
   189→    });
   190→  }
   191→
   192→  // 3. Critical issues / risky clients
   193→  const riskyClients = clients.filter(c => (c.healthScore ?? 100) < 60);
   194→  if (metrics.criticalIssues > 0 || riskyClients.length > 0) {
   195→    insights.push({
   196→      id: 'churn-risk',
   197→      icon: TrendingDown,
   198→      tone: 'risk',
   199→      what: riskyClients.length > 0
   200→        ? `${riskyClients.length} client${riskyClients.length > 1 ? 's may' : ' may'} churn`
   201→        : `${metrics.criticalIssues} critical issue${metrics.criticalIssues > 1 ? 's' : ''} need attention`,
   202→      why: riskyClients.length > 0
   203→        ? 'Low health scores signal dissatisfaction. Reach out before they leave.'
   204→        : 'Critical issues compound into bigger problems if left unaddressed.',
   205→      actionLabel: riskyClients.length > 0 ? 'View Clients' : 'Review Issues',
   206→      actionView: riskyClients.length > 0 ? 'clients' : 'ai-operating-room',
   207→    });
   208→  }
   209→
   210→  // 4. Cash position / collections
   211→  if (metrics.totalTaxVolume > 0) {
   212→    insights.push({
   213→      id: 'collections',
   214→      icon: Wallet,
   215→      tone: metrics.matchPercentage < 90 ? 'risk' : 'info',
   216→      what: metrics.matchPercentage < 90
   217→        ? 'Collections may drop — reconciliation gaps detected'
   218→        : 'Cash position is healthy',
   219→      why: metrics.matchPercentage < 90
   220→        ? `${metrics.matchPercentage.toFixed(0)}% of invoices reconciled. Unmatched invoices mean uncollected revenue.`
   221→        : `${metrics.matchPercentage.toFixed(0)}% invoice match rate. Keep chasing pending collections to maintain runway.`,
   222→      actionLabel: 'Fix Collections',
   223→      actionView: 'reconcile',
   224→    });
   225→  }
   226→
   227→  // 5. Compliance score insight
   228→  if (metrics.averageHealthScore > 0) {
   229→    const tier = metrics.averageHealthScore >= 80 ? 'success' : 'info';
   230→    insights.push({
   231→      id: 'compliance',
   232→      icon: ShieldAlert,
   233→      tone: tier,
   234→      what: `Compliance health at ${metrics.averageHealthScore.toFixed(0)}%`,
   235→      why: metrics.averageHealthScore >= 80
   236→        ? 'Strong compliance posture. Maintain by filing on time and resolving warnings.'
   237→        : 'Below optimal. Address pending returns and warnings to lift your score.',
   238→      actionLabel: 'View Compliance',
   239→      actionView: 'returns',
   240→    });
   241→  }
   242→
   243→  // 6. Always-on "all clear" nudge — only when we actually have data.
   244→  // Never say "Everything looks good today" when there is no data at all;
   245→  // in that case the widget shows its own honest empty state instead.
   246→  if (hasData && insights.length === 0) {
   247→    insights.push({
   248→      id: 'all-clear',
   249→      icon: Sparkles,
   250→      tone: 'success',
   251→      what: 'Everything looks good today',
   252→      why: 'No urgent issues. Hand control to Oracle and let it run the routine work.',
   253→      actionLabel: 'Run My Business',
   254→      actionView: 'run-my-business',
   255→    });
   256→  }
   257→
   258→  return insights.slice(0, 6);
   259→}

   263→function ScoreGauge({ score }: { score: number | null }) {
   264→  const size = 220;
   265→  const stroke = 14;
   266→  const r = (size - stroke) / 2;
   267→  const c = 2 * Math.PI * r;
   268→  const pct = score === null ? 0 : score / 100;
   269→  const offset = c * (1 - pct);
   270→
   271→  return (
   272→    <div className="relative" style={{ width: size, height: size }}>
   273→      <svg width={size} height={size} className="-rotate-90">
   274→        <defs>
   275→          <linearGradient id="missionScoreGradient" x1="0%" y1="0%" x2="100%" y2="100%">
   276→            <stop offset="0%" stopColor="#10b981" />
   277→            <stop offset="50%" stopColor="#06b6d4" />
   278→            <stop offset="100%" stopColor="#3b82f6" />
   279→          </linearGradient>
   280→        </defs>
   281→        <circle
   282→          cx={size / 2}
   283→          cy={size / 2}
   284→          r={r}
   285→          fill="none"
   286→          stroke="rgba(255,255,255,0.06)"
   287→          strokeWidth={stroke}
   288→        />
   289→        {score !== null && (
   290→          <motion.circle
   291→            cx={size / 2}
   292→            cy={size / 2}
   293→            r={r}
   294→            fill="none"
   295→            stroke="url(#missionScoreGradient)"
   296→            strokeWidth={stroke}
   297→            strokeLinecap="round"
   298→            strokeDasharray={c}
   299→            initial={{ strokeDashoffset: c }}
   300→            animate={{ strokeDashoffset: offset }}
   301→            transition={{ duration: 1.2, ease: 'easeOut' as const, delay: 0.2 }}
   302→          />
   303→        )}
   304→      </svg>
   305→      <div className="absolute inset-0 flex flex-col items-center justify-center">
   306→        {score === null ? (
   307→          <span className="text-4xl font-bold text-muted-foreground">—</span>
   308→        ) : (
   309→          <motion.span
   310→            initial={{ opacity: 0, scale: 0.8 }}
   311→            animate={{ opacity: 1, scale: 1 }}
   312→            transition={{ duration: 0.5, delay: 0.6 }}
   313→            className={`text-6xl font-bold tracking-tight ${scoreTier(score).color}`}
   314→          >
   315→            {score}
   316→          </motion.span>
   317→        )}
   318→        <span className="text-xs text-muted-foreground tracking-wider uppercase mt-1">
   319→          / 100
   320→        </span>
   321→      </div>
   322→    </div>
   323→  );
   324→}
