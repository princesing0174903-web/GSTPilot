    57→    57→function inrShort(n: number): string {
    58→    58→  if (!isFinite(n) || isNaN(n)) return '₹0';
    59→    59→  const abs = Math.abs(n);
    60→    60→  if (abs >= 10000000) return `₹${(n / 10000000).toFixed(2)} Cr`;
    61→    61→  if (abs >= 100000) return `₹${(n / 100000).toFixed(2)} L`;
    62→    62→  if (abs >= 1000) return `₹${(n / 1000).toFixed(1)}K`;
    63→    63→  return '₹' + Math.round(n).toLocaleString('en-IN');
    64→    64→}
