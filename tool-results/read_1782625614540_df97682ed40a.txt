    59→    59→function safe<T>(label: string, fn: () => Promise<T>, fallback: T): Promise<T> {
    60→    60→  return fn().catch((err) => {
    61→    61→    console.warn(`[AI Workforce] Engine "${label}" failed:`, err);
    62→    62→    return fallback;
    63→    63→  });
    64→    64→}

    66→    66→function safeSync<T>(label: string, fn: () => T, fallback: T): T {
    67→    67→  try {
    68→    68→    return fn();
    69→    69→  } catch (err) {
    70→    70→    console.warn(`[AI Workforce] Engine "${label}" failed:`, err);
    71→    71→    return fallback;
    72→    72→  }

    98→    98→function syncDecisionStore(decisions: CrossDepartmentDecision[]) {
    99→    99→  for (const d of decisions) {
   100→   100→    const existing = decisionStore.get(d.id);
   101→   101→    if (!existing) {
   102→   102→      decisionStore.set(d.id, d);
   103→   103→    } else if (existing.status === 'proposed' || existing.status === 'step_1_approved' || existing.status === 'step_2_approved') {
   104→   104→      // Preserve progress but refresh data
   105→   105→      decisionStore.set(d.id, { ...d, status: existing.status, currentStepIndex: existing.currentStepIndex, steps: existing.steps });
   106→   106→    }
   107→   107→    // If fully_approved/executing/executed/rejected — preserve user's action
   108→   108→  }
   109→   109→  // Drop stale proposed decisions no longer being generated
   110→   110→  const liveIds = new Set(decisions.map((d) => d.id));
   111→   111→  for (const [id, d] of decisionStore.entries()) {
   112→   112→    if (!liveIds.has(id) && (d.status === 'proposed')) {
   113→   113→      decisionStore.delete(id);
   114→   114→    }
   115→   115→  }
   116→   116→}

   193→   193→function buildAggregateMetrics(
   194→   194→  employees: AIEmployee[],
   195→   195→  collaborationFeed: ReturnType<typeof computeCollaborationFeed>,
   196→   196→  meetings: ReturnType<typeof computeMeetings>,
   197→   197→  crossDecisions: CrossDepartmentDecision[],
   198→   198→): WorkforceAggregateMetrics {
   199→   199→  const activeEmployees = employees.filter((e) => e.status !== 'idle' && e.status !== 'offline').length;
   200→   200→  const totalTasksCompleted = employees.reduce((s, e) => s + e.performance.tasksCompleted, 0);
   201→   201→  const totalROIGenerated = employees.reduce((s, e) => s + e.performance.roiGenerated, 0);
   202→   202→  const totalRevenueInfluenced = employees.reduce((s, e) => s + e.performance.revenueInfluenced, 0);
   203→   203→  const totalCostSaved = employees.reduce((s, e) => s + e.performance.costSaved, 0);
   204→   204→  const automationRate = employees.length > 0
   205→   205→    ? Math.round(employees.reduce((s, e) => s + e.performance.automationSuccess, 0) / employees.length)
   206→   206→    : 0;
   207→   207→  const avgHealthScore = employees.length > 0
   208→   208→    ? Math.round(employees.reduce((s, e) => s + e.healthScore, 0) / employees.length)
   209→   209→    : 0;
   210→   210→
   211→   211→  // Count today's collaboration messages
   212→   212→  const today = new Date().toDateString();
   213→   213→  const collaborationMessagesToday = collaborationFeed.filter(
   214→   214→    (m) => new Date(m.timestamp).toDateString() === today,
   215→   215→  ).length;
   216→   216→
   217→   217→  // Count this week's meetings
   218→   218→  const weekFromNow = new Date();
   219→   219→  weekFromNow.setDate(weekFromNow.getDate() + 7);
   220→   220→  const meetingsThisWeek = meetings.filter((m) => {
   221→   221→    const meetingDate = new Date(m.scheduledFor);
   222→   222→    return meetingDate >= new Date() && meetingDate <= weekFromNow;
   223→   223→  }).length;
   224→   224→
   225→   225→  return {
   226→   226→    totalEmployees: employees.length,
   227→   227→    activeEmployees,
   228→   228→    totalTasksCompleted,
   229→   229→    totalDecisions: crossDecisions.length,
   230→   230→    totalROIGenerated,
   231→   231→    totalRevenueInfluenced,
   232→   232→    totalCostSaved,
   233→   233→    automationRatePct: automationRate,
   234→   234→    avgHealthScore,
   235→   235→    collaborationMessagesToday,
   236→   236→    meetingsThisWeek,
   237→   237→    activeCrossDecisions: crossDecisions.filter(
   238→   238→      (d) => d.status !== 'executed' && d.status !== 'rejected',
   239→   239→    ).length,
   240→   240→  };
   241→   241→}

   245→   245→export async function computeWorkforceDashboard(): Promise<WorkforceDashboard> {
   246→   246→  const data = await fetchWorkforceData();

   248→   248→  // Run all engines in parallel (each wrapped in safeSync)

   250→   250→    safeSync('organization', () => computeOrganization(data), [] as AIEmployee[]),
   251→   251→    safeSync('collaboration', () => computeCollaborationFeed(data), []),
   252→   252→    safeSync('chains', () => computeCollaborationChains(data), []),
   253→   253→    safeSync('meetings', () => computeMeetings(data), []),
   254→   254→    safeSync('cross-decisions', () => computeCrossDecisions(data), [] as CrossDepartmentDecision[]),
   255→   255→  ]);
   256→   256→
   257→   257→  // Sync the in-memory decision store
   258→   258→  syncDecisionStore(crossDecisionsRaw);
   259→   259→
   260→   260→  // Use the MERGED store decisions (preserves user-approved statuses)
   261→   261→  const crossDecisions = crossDecisionsRaw
   262→   262→    .map((d) => decisionStore.get(d.id) ?? d)
   263→   263→    .sort((a, b) => Math.abs(b.financialImpact) - Math.abs(a.financialImpact));
   264→   264→
   265→   265→  // Build department dashboards

   267→   267→    safeSync(`dept-${dept.department}`, () => buildDepartmentDashboard(dept.department, organization, data), null),
   268→   268→  ).filter(Boolean) as DepartmentDashboard[];
   269→   269→
   270→   270→  // Build performance leaderboard
   271→   271→  const employeesWithPerf: EmployeeWithPerformance[] = organization.map((e) => ({
   272→   272→    role: e.role,
   273→   273→    name: e.name,

   281→   281→    const entry = performance.find((p) => p.role === e.role);

   283→   283→  });
   284→   284→
   285→   285→  // Top employee + top collaboration
   286→   286→  const topEmployee = performance[0] ?? null;
   287→   287→  const topCollaboration = chains[0] ?? null;

   291→   291→    (d) => d.status === 'proposed' || d.status === 'step_1_approved' || d.status === 'step_2_approved',
   292→   292→  );
   293→   293→  const openDelegations = delegationStore.filter(
   294→   294→    (d) => d.status === 'pending' || d.status === 'accepted',
   295→   295→  );

   297→   297→
   298→   298→  // Aggregate metrics

   300→   300→    buildAggregateMetrics(rankedOrg, collaborationFeed, meetings, crossDecisions),

   303→   303→    totalEmployees: 0, activeEmployees: 0, totalTasksCompleted: 0, totalDecisions: 0,
   304→   304→    totalROIGenerated: 0, totalRevenueInfluenced: 0, totalCostSaved: 0, automationRatePct: 0,
   305→   305→    avgHealthScore: 0, collaborationMessagesToday: 0, meetingsThisWeek: 0, activeCrossDecisions: 0,

   317→   317→    'Never fabricate employee performance, memory, or skills — use live data only',
   318→   318→  ];
   319→   319→
   320→   320→  const activeEmployeeCount = rankedOrg.filter((e) => e.status !== 'idle' && e.status !== 'offline').length;

   432→   432→export async function approveCrossDecision(req: ApproveWorkforceRequest): Promise<ApproveWorkforceResult> {
   433→   433→  const role = resolveRole(req.role);
   434→   434→  const decision = await ensureDecisionInStore(req.decisionId);
   435→   435→  if (!decision) {
   436→   436→    return {
   437→   437→      decisionId: req.decisionId,
   438→   438→      status: 'proposed',
   439→   439→      message: 'Decision not found. It may have expired or been superseded.',
   440→   440→      currentStepIndex: 0,
   441→   441→      approvedAt: new Date().toISOString(),
   442→   442→    };
   443→   443→  }
   444→   444→
   445→   445→  const stepIndex = req.stepIndex ?? decision.currentStepIndex;
   446→   446→  const step = decision.steps[stepIndex];
   447→   447→  if (!step) {
   448→   448→    return {
   449→   449→      decisionId: req.decisionId,
   450→   450→      status: decision.status,
   451→   451→      message: `Step ${stepIndex} not found.`,
   452→   452→      currentStepIndex: decision.currentStepIndex,
   453→   453→      approvedAt: new Date().toISOString(),
   454→   454→    };
   455→   455→  }
   456→   456→
   457→   457→  // Validate role can approve this step (the step's role must match or be higher tier)
   458→   458→  const stepRoleDef = getRoleDefinition(step.role);
   459→   459→  const approverIsStepRole = step.role === role || stepRoleDef.reportsTo === role;
   460→   460→
   461→   461→  // Mark step approved
   462→   462→  const updatedSteps = decision.steps.map((s, i) =>
   463→   463→    i === stepIndex
   464→   464→      ? { ...s, status: 'approved' as const, approvedAt: new Date().toISOString(), approvedBy: req.userId ?? role, comment: req.comment }
   465→   465→      : s,
   466→   466→  );
   467→   467→
   468→   468→  const nextStepIndex = stepIndex + 1;
   469→   469→  let newStatus: CrossDecisionStatus;
   470→   470→  if (nextStepIndex >= decision.steps.length) {
   471→   471→    newStatus = 'fully_approved';
   472→   472→  } else if (stepIndex === 0) {
   473→   473→    newStatus = 'step_1_approved';
   474→   474→  } else if (stepIndex === 1) {
   475→   475→    newStatus = 'step_2_approved';
   476→   476→  } else {
   477→   477→    newStatus = 'fully_approved';
   478→   478→  }
   479→   479→
   480→   480→  // Auto-execute if fully approved and non-destructive
   481→   481→  let finalStatus: CrossDecisionStatus = newStatus;
   482→   482→  let message = `Step ${stepIndex + 1} approved by ${role}.`;
   483→   483→  if (newStatus === 'fully_approved') {
   484→   484→    finalStatus = 'executing';
   485→   485→    message = `Fully approved. Executing workflow...`;
   486→   486→    // Mark all execute-type steps as executed
   487→   487→    for (let i = 0; i < updatedSteps.length; i++) {
   488→   488→      if (updatedSteps[i].approvalType === 'execute') {
   489→   489→        updatedSteps[i].status = 'executed';
   490→   490→        updatedSteps[i].approvedAt = new Date().toISOString();
   491→   491→        updatedSteps[i].approvedBy = req.userId ?? role;
   492→   492→      }
   493→   493→    }
   494→   494→    finalStatus = 'executed';
   495→   495→    message = `Decision fully approved and executed. All ${decision.steps.length} steps completed.`;
   496→   496→  }
   497→   497→
   498→   498→  const updated: CrossDepartmentDecision = {
   499→   499→    ...decision,
   500→   500→    steps: updatedSteps,
   501→   501→    currentStepIndex: newStatus === 'fully_approved' ? decision.steps.length - 1 : nextStepIndex,
   502→   502→    status: finalStatus,
   503→   503→    executedAt: finalStatus === 'executed' ? new Date().toISOString() : decision.executedAt,
   504→   504→  };
   505→   505→  decisionStore.set(req.decisionId, updated);
   506→   506→
   507→   507→  return {
   508→   508→    decisionId: req.decisionId,
   509→   509→    status: finalStatus,
   510→   510→    message,
   511→   511→    currentStepIndex: updated.currentStepIndex,
   512→   512→    approvedAt: new Date().toISOString(),
   513→   513→  };
   514→   514→}

   516→   516→export async function executeCrossDecision(req: ExecuteWorkforceRequest): Promise<ExecuteWorkforceResult> {
   517→   517→  const role = resolveRole(req.role);
   518→   518→  const decision = await ensureDecisionInStore(req.decisionId);
   519→   519→  if (!decision) {
   520→   520→    return {
   521→   521→      decisionId: req.decisionId,
   522→   522→      status: 'proposed',
   523→   523→      message: 'Decision not found.',
   524→   524→      executedAt: new Date().toISOString(),
   525→   525→    };
   526→   526→  }
   527→   527→
   528→   528→  // Force-execute all remaining steps (requires CEO role)
   529→   529→  if (role !== 'ceo') {
   530→   530→    return {
   531→   531→      decisionId: req.decisionId,
   532→   532→      status: decision.status,
   533→   533→      message: 'Only CEO role can force-execute a decision. Please approve steps individually.',
   534→   534→      executedAt: new Date().toISOString(),
   535→   535→    };
   536→   536→  }
   537→   537→
   538→   538→  const updatedSteps = decision.steps.map((s) => ({
   539→   539→    ...s,
   540→   540→    status: (s.status === 'pending' ? 'approved' : s.status) as CrossDecisionStep['status'],
   541→   541→    approvedAt: s.approvedAt ?? new Date().toISOString(),
   542→   542→    approvedBy: s.approvedBy ?? req.userId ?? role,
   543→   543→  }));
   544→   544→  // Mark execute steps as executed
   545→   545→  for (const s of updatedSteps) {
   546→   546→    if (s.approvalType === 'execute') {
   547→   547→      s.status = 'executed';
   548→   548→    }
   549→   549→  }
   550→   550→
   551→   551→  const updated: CrossDepartmentDecision = {
   552→   552→    ...decision,
   553→   553→    steps: updatedSteps,
   554→   554→    currentStepIndex: decision.steps.length - 1,
   555→   555→    status: 'executed',
   556→   556→    executedAt: new Date().toISOString(),
   557→   557→  };
   558→   558→  decisionStore.set(req.decisionId, updated);
   559→   559→
   560→   560→  return {
   561→   561→    decisionId: req.decisionId,
   562→   562→    status: 'executed',
   563→   563→    message: `Decision force-executed by CEO. All ${decision.steps.length} steps completed.`,
   564→   564→    executedAt: new Date().toISOString(),
   565→   565→    outputs: { stepsCompleted: decision.steps.length },
   566→   566→  };
   567→   567→}
