   577→function EmailVerificationBanner() {

   628→function OnboardingScreen() {
   629→  const { user, markOnboardingComplete } = useAuth()
   630→  const { setCurrentScreen, setCurrentView } = useApp()
   631→  const [saveError, setSaveError] = React.useState<string | null>(null)
   632→
   633→  // ── Fire-and-forget: save onboarding data to Firestore in the background ──
   634→  const saveOnboardingToFirestore = async (
   635→    data: import('@/components/onboarding/OnboardingFlow').OnboardingData,
   636→    firmId: string
   637→  ) => {
   638→    try {
   639→      const { doc, setDoc, serverTimestamp } = await import('firebase/firestore')
   640→      const { db } = await import('@/lib/firebase')
   641→
   642→      if (!user) return
   643→
   644→      // Update user document
   645→      await setDoc(doc(db, 'users', user.id), {
   646→        uid: user.id,
   647→        fullName: data.fullName,
   648→        email: data.email,
   649→        phone: data.phone,
   650→        ageGroup: data.ageGroup,
   651→        profession: data.profession,
   652→        experience: data.experience,
   653→        firmId: firmId || null,
   654→        firmName: data.firmName,
   655→        onboardingCompleted: true,
   656→        clientCount: data.clientCount,
   657→        monthlyReturns: data.monthlyReturns,
   658→        gstServices: data.gstServices,
   659→        painPoints: data.painPoints,
   660→        referralSource: data.referralSource,
   661→        trialReasons: data.trialReasons,
   662→        wantsUpdates: data.wantsUpdates,
   663→        updatedAt: serverTimestamp(),
   664→      }, { merge: true })
   665→      console.log('[Onboarding] ✅ User document saved to Firestore')
   666→
   667→      // Save onboarding record
   668→      await setDoc(doc(db, 'onboarding', user.id), {
   669→        ...data,
   670→        firmId: firmId || null,
   671→        completedAt: serverTimestamp(),
   672→      })
   673→      console.log('[Onboarding] ✅ Onboarding record saved to Firestore')
   674→    } catch (error) {
   675→      console.warn('[Onboarding] ⚠️ Background Firestore save failed:', error)
   676→    }
   677→  }
   678→
   679→  const handleOnboardingComplete = (
   680→    data: import('@/components/onboarding/OnboardingFlow').OnboardingData,
   681→    destination?: import('@/components/onboarding/OnboardingFlow').OnboardingDestination
   682→  ) => {
   683→    setSaveError(null)
   684→
   685→    if (!user) {
   686→      setSaveError('No authenticated user found. Please sign in again.')
   687→      return
   688→    }
   689→
   690→    // ── INSTANT: Mark onboarding complete locally & navigate ──
   691→    // This happens synchronously — user sees dashboard immediately
   692→    console.log('[Onboarding] 🚀 Navigating to', destination || 'dashboard', '(optimistic)')
   693→    markOnboardingComplete(undefined, data.firmName)
   694→    setCurrentView(destination || 'dashboard')
   695→    setCurrentScreen('app')
   696→
   697→    // ── BACKGROUND: Create firm doc + save everything to Firestore ──
   698→    // These happen async — user is already in the dashboard
   699→    console.log('[Onboarding] 📝 Starting background Firestore writes...')
   700→    ;(async () => {
   701→      try {
   702→        const { collection, addDoc, serverTimestamp } = await import('firebase/firestore')
   703→        const { db } = await import('@/lib/firebase')
   704→
   705→        // Create firm document
   706→        let firmId = ''
   707→        try {
   708→          const firmRef = await addDoc(collection(db, 'firms'), {
   709→            ownerId: user.id,
   710→            firmName: data.firmName,
   711→            gstin: data.gstin || null,
   712→            state: data.state,
   713→            stateCode: data.stateCode,
   714→            organizationType: data.organizationType,
   715→            icaiMembershipNo: data.icaiMembershipNo || null,
   716→            officeAddress: data.officeAddress || null,
   717→            createdAt: serverTimestamp(),
   718→          })
   719→          firmId = firmRef.id
   720→          console.log('[Onboarding] ✅ Firm created:', firmId)
   721→
   722→          // Update the local user with firmId
   723→          markOnboardingComplete(firmId, data.firmName)
   724→        } catch (firmError) {
   725→          console.warn('[Onboarding] ⚠️ Firm creation failed:', firmError)
   726→        }
   727→
   728→        // Save remaining data
   729→        await saveOnboardingToFirestore(data, firmId)
   730→      } catch (error) {
   731→        console.warn('[Onboarding] ⚠️ Background save error:', error)
   732→      }
   733→    })()
   734→  }
   735→
   736→  const handleSkip = () => {
   737→    setSaveError(null)
   738→
   739→    if (!user) {
   740→      setSaveError('No authenticated user found. Please sign in again.')
   741→      return
   742→    }
   743→
   744→    // ── INSTANT: Navigate immediately ──
   745→    console.log('[Onboarding] 🚀 Skipping onboarding (optimistic)')
   746→    markOnboardingComplete()
   747→    setCurrentView('dashboard')
   748→    setCurrentScreen('app')
   749→
   750→    // ── BACKGROUND: Mark as completed in Firestore ──
   751→    ;(async () => {
   752→      try {
   753→        const { doc, setDoc, serverTimestamp } = await import('firebase/firestore')
   754→        const { db } = await import('@/lib/firebase')
   755→        await setDoc(doc(db, 'users', user.id), {
   756→          onboardingCompleted: true,
   757→          updatedAt: serverTimestamp(),
   758→        }, { merge: true })
   759→        console.log('[Onboarding] ✅ Skip saved to Firestore')
   760→      } catch (error) {
   761→        console.warn('[Onboarding] ⚠️ Background skip save failed:', error)
   762→      }
   763→    })()
   764→  }
   765→
   766→  return (
   767→    <OnboardingFlow
   768→      onComplete={handleOnboardingComplete}
   769→      onSkip={handleSkip}
   770→      userEmail={user?.email}
   771→      userName={user?.name}
   772→      error={saveError}
   773→      onDismissError={() => setSaveError(null)}
   774→    />
   775→  )
   776→}

   778→function AppRouter() {
   779→  const { currentScreen, setCurrentScreen } = useApp()
   780→  const { isAuthenticated, isInitializing, needsOnboarding, needsEmailVerification } = useAuth()
   781→
   782→  useEffect(() => {
   783→    if (isInitializing) return
   784→    // Only auto-redirect to app if authenticated AND onboarding is complete
   785→    // Don't redirect if user still needs onboarding — the OnboardingScreen handles its own navigation
   786→    if (isAuthenticated && !needsOnboarding && currentScreen !== 'app') {
   787→      setCurrentScreen('app')
   788→    }
   789→  }, [isAuthenticated, isInitializing, currentScreen, setCurrentScreen, needsOnboarding])
   790→
   791→  useEffect(() => {
   792→    if (isInitializing) return
   793→    if (!isAuthenticated && currentScreen === 'app') {
   794→      setCurrentScreen('landing')
   795→    }
   796→  }, [isAuthenticated, isInitializing, currentScreen, setCurrentScreen])
   797→
   798→  const handleGetStarted = () => setCurrentScreen('login')
   799→  const handleBookDemo = () => setCurrentScreen('login')
   800→  const handleBackToLanding = () => setCurrentScreen('landing')
   801→
   802→  if (isInitializing && !isAuthenticated) {
   803→    return (
   804→      <div className="min-h-screen flex items-center justify-center bg-black">
   805→        <div className="flex flex-col items-center gap-4">
   806→          <div className="flex h-11 w-11 items-center justify-center rounded-2xl glass-surface motion-pulse">
   807→            <Zap className="h-5 w-5 accent-text" />
   808→          </div>
   809→          <span className="text-sm text-white/55 font-medium">Loading GSTPilot…</span>
   810→        </div>
   811→      </div>
   812→    )
   813→  }
   814→
   815→  // Show onboarding if user hasn't completed it
   816→  if (isAuthenticated && needsOnboarding) {
   817→    return <OnboardingScreen />
   818→  }
   819→
   820→  if (currentScreen === 'app' && isAuthenticated) {
   821→    return (
   822→      <div className="flex min-h-screen flex-col">
   823→        {needsEmailVerification && <EmailVerificationBanner />}
   824→        <DashboardContent />
   825→      </div>
   826→    )
   827→  }
   828→
   829→  if (currentScreen === 'login') {
   830→    return <LoginPage onBack={handleBackToLanding} onGetStarted={handleGetStarted} />
   831→  }
   832→
   833→  return <LandingPage onGetStarted={handleGetStarted} onBookDemo={handleBookDemo} />
   834→}
