   108→// Relative time formatter — defensively handles ISO string / Date / Firestore Timestamp.
   109→function timeAgo(createdAt: unknown): string {
   110→  if (!createdAt) return 'just now';
   111→  let date: Date;
   112→  if (createdAt instanceof Date) {
   113→    date = createdAt;
   114→  } else if (typeof createdAt === 'string') {
   115→    date = new Date(createdAt);
   116→  } else if (
   117→    typeof createdAt === 'object' &&
   118→    createdAt !== null &&
   119→    'toDate' in createdAt &&
   120→    typeof (createdAt as { toDate: () => Date }).toDate === 'function'
   121→  ) {
   122→    date = (createdAt as { toDate: () => Date }).toDate();
   123→  } else {
   124→    return 'just now';
   125→  }
   126→  if (isNaN(date.getTime())) return 'just now';
   127→  const seconds = Math.floor((Date.now() - date.getTime()) / 1000);
   128→  if (seconds < 60) return 'just now';
   129→  const minutes = Math.floor(seconds / 60);
   130→  if (minutes < 60) return `${minutes}m ago`;
   131→  const hours = Math.floor(minutes / 60);
   132→  if (hours < 24) return `${hours}h ago`;
   133→  const days = Math.floor(hours / 24);
   134→  if (days < 7) return `${days}d ago`;
   135→  const weeks = Math.floor(days / 7);
   136→  if (weeks < 5) return `${weeks}w ago`;
   137→  const months = Math.floor(days / 30);
   138→  if (months < 12) return `${months}mo ago`;
   139→  const years = Math.floor(days / 365);
   140→  return `${years}y ago`;
   141→}

   143→// ─── Business Score computation ───────────────────────────────────────────────
   144→// Start at 100. Subtract weighted penalties for risk signals. Clamp 0–100.
   145→// If there is no data at all, return null (UI shows premium empty state).
   146→
   147→function computeBusinessScore(opts: {
   148→  totalClients: number;
   149→  criticalIssues: number;
   150→  warnings: number;
   151→  pendingReturns: number;
   152→  overdueReturns: number;
   153→  averageHealthScore: number;
   154→  matchPercentage: number;
   155→}): number | null {
   156→  const { totalClients, criticalIssues, warnings, pendingReturns, overdueReturns, averageHealthScore, matchPercentage } = opts;
   157→  if (totalClients === 0 && pendingReturns === 0 && overdueReturns === 0) return null;
   158→
   159→  let score = 100;
   160→  score -= criticalIssues * 6;
   161→  score -= warnings * 2;
   162→  score -= overdueReturns * 8;
   163→  score -= pendingReturns * 2;
   164→  if (averageHealthScore > 0) {
   165→    score -= Math.max(0, 80 - averageHealthScore) * 0.6;
   166→  }
   167→  score -= Math.max(0, 100 - matchPercentage) * 0.3;
   168→  return Math.max(0, Math.min(100, Math.round(score)));
   169→}

   171→// V16 palette — only Emerald / Cyan / Blue. No amber, no red.
   172→function scoreTier(score: number | null): { label: string; color: string } {
   173→  if (score === null) return { label: '—', color: 'text-muted-foreground' };
   174→  if (score >= 85) return { label: 'Excellent', color: 'text-emerald-400' };
   175→  if (score >= 70) return { label: 'Healthy', color: 'text-cyan-400' };
   176→  if (score >= 50) return { label: 'At Risk', color: 'text-blue-400' };
   177→  return { label: 'Critical', color: 'text-blue-300' };
   178→}

   194→function buildInsights(opts: {
   195→  metrics: {
   196→    totalClients: number;
   197→    pendingReturns: number;
   198→    overdueReturns: number;
   199→    criticalIssues: number;
   200→    warnings: number;
   201→    totalTaxVolume: number;
   202→    averageHealthScore: number;
   203→    matchPercentage: number;
   204→    upcomingFilings: FirestoreReturn[];
   205→  };
   206→  clients: Array<FirestoreClient & { id: string }>;
   207→}): AIInsight[] {
   208→  const { metrics, clients } = opts;
   209→  const insights: AIInsight[] = [];
   210→
   211→  // 1. Overdue returns — most urgent
   212→  if (metrics.overdueReturns > 0) {
   213→    insights.push({
   214→      id: 'overdue-returns',
   215→      icon: AlertTriangle,
   216→      tone: 'risk',
   217→      what: `${metrics.overdueReturns} GST return${metrics.overdueReturns > 1 ? 's are' : ' is'} overdue`,
   218→      why: 'Overdue returns trigger notices and late fees. File immediately to avoid penalties.',
   219→      actionLabel: 'File Now',
   220→      actionView: 'returns',
   221→    });
   222→  }
   223→
   224→  // 2. Pending returns due soon
   225→  const nextFiling = metrics.upcomingFilings[0];
   226→  if (nextFiling && metrics.pendingReturns > 0 && metrics.overdueReturns === 0) {
   227→    insights.push({
   228→      id: 'gst-due',
   229→      icon: Clock,
   230→      tone: 'info',
   231→      what: `GST filing due — ${nextFiling.returnType} for ${nextFiling.period}`,
   232→      why: `${metrics.pendingReturns} return${metrics.pendingReturns > 1 ? 's' : ''} pending. Filing on time keeps compliance score high.`,
   233→      actionLabel: 'File GST',
   234→      actionView: 'returns',
   235→    });
   236→  }
   237→
   238→  // 3. Critical issues / risky clients
   239→  const riskyClients = clients.filter(c => (c.healthScore ?? 100) < 60);
   240→  if (metrics.criticalIssues > 0 || riskyClients.length > 0) {
   241→    insights.push({
   242→      id: 'churn-risk',
   243→      icon: TrendingDown,
   244→      tone: 'risk',
   245→      what: riskyClients.length > 0
   246→        ? `${riskyClients.length} client${riskyClients.length > 1 ? 's may' : ' may'} churn`
   247→        : `${metrics.criticalIssues} critical issue${metrics.criticalIssues > 1 ? 's' : ''} need attention`,
   248→      why: riskyClients.length > 0
   249→        ? 'Low health scores signal dissatisfaction. Reach out before they leave.'
   250→        : 'Critical issues compound into bigger problems if left unaddressed.',
   251→      actionLabel: riskyClients.length > 0 ? 'View Clients' : 'Review Issues',
   252→      actionView: riskyClients.length > 0 ? 'clients' : 'ai-operating-room',
   253→    });
   254→  }
   255→
   256→  // 4. Cash position / collections
   257→  if (metrics.totalTaxVolume > 0) {
   258→    insights.push({
   259→      id: 'collections',
   260→      icon: Wallet,
   261→      tone: metrics.matchPercentage < 90 ? 'risk' : 'info',
   262→      what: metrics.matchPercentage < 90
   263→        ? 'Collections may drop — reconciliation gaps detected'
   264→        : 'Cash position is healthy',
   265→      why: metrics.matchPercentage < 90
   266→        ? `${metrics.matchPercentage.toFixed(0)}% of invoices reconciled. Unmatched invoices mean uncollected revenue.`
   267→        : `${metrics.matchPercentage.toFixed(0)}% invoice match rate. Keep chasing pending collections to maintain runway.`,
   268→      actionLabel: 'Fix Collections',
   269→      actionView: 'reconcile',
   270→    });
   271→  }
   272→
   273→  // 5. Compliance score insight
   274→  if (metrics.averageHealthScore > 0) {
   275→    const tier = metrics.averageHealthScore >= 80 ? 'success' : 'info';
   276→    insights.push({
   277→      id: 'compliance',
   278→      icon: ShieldAlert,
   279→      tone: tier,
   280→      what: `Compliance health at ${metrics.averageHealthScore.toFixed(0)}%`,
   281→      why: metrics.averageHealthScore >= 80
   282→        ? 'Strong compliance posture. Maintain by filing on time and resolving warnings.'
   283→        : 'Below optimal. Address pending returns and warnings to lift your score.',
   284→      actionLabel: 'View Compliance',
   285→      actionView: 'returns',
   286→    });
   287→  }
   288→
   289→  // 6. Always-on "all clear" nudge if nothing urgent
   290→  if (insights.length === 0) {
   291→    insights.push({
   292→      id: 'all-clear',
   293→      icon: Sparkles,
   294→      tone: 'success',
   295→      what: 'Everything looks good today',
   296→      why: 'No urgent issues. Hand control to Oracle and let it run the routine work.',
   297→      actionLabel: 'Run My Business',
   298→      actionView: 'run-my-business',
   299→    });
   300→  }
   301→
   302→  return insights.slice(0, 6);
   303→}

   307→function ScoreGauge({ score }: { score: number | null }) {
   308→  const size = 220;
   309→  const stroke = 14;
   310→  const r = (size - stroke) / 2;
   311→  const c = 2 * Math.PI * r;
   312→  const pct = score === null ? 0 : score / 100;
   313→  const offset = c * (1 - pct);
   314→
   315→  return (
   316→    <div className="relative" style={{ width: size, height: size }}>
   317→      <svg width={size} height={size} className="-rotate-90">
   318→        <defs>
   319→          <linearGradient id="missionScoreGradient" x1="0%" y1="0%" x2="100%" y2="100%">
   320→            <stop offset="0%" stopColor="#10b981" />
   321→            <stop offset="50%" stopColor="#06b6d4" />
   322→            <stop offset="100%" stopColor="#3b82f6" />
   323→          </linearGradient>
   324→        </defs>
   325→        <circle
   326→          cx={size / 2}
   327→          cy={size / 2}
   328→          r={r}
   329→          fill="none"
   330→          stroke="rgba(255,255,255,0.06)"
   331→          strokeWidth={stroke}
   332→        />
   333→        {score !== null && (
   334→          <motion.circle
   335→            cx={size / 2}
   336→            cy={size / 2}
   337→            r={r}
   338→            fill="none"
   339→            stroke="url(#missionScoreGradient)"
   340→            strokeWidth={stroke}
   341→            strokeLinecap="round"
   342→            strokeDasharray={c}
   343→            initial={{ strokeDashoffset: c }}
   344→            animate={{ strokeDashoffset: offset }}
   345→            transition={{ duration: 1.2, ease: 'easeOut', delay: 0.2 }}
   346→          />
   347→        )}
   348→      </svg>
   349→      <div className="absolute inset-0 flex flex-col items-center justify-center">
   350→        {score === null ? (
   351→          <span className="text-4xl font-bold text-muted-foreground">—</span>
   352→        ) : (
   353→          <motion.span
   354→            initial={{ opacity: 0, scale: 0.8 }}
   355→            animate={{ opacity: 1, scale: 1 }}
   356→            transition={{ duration: 0.5, delay: 0.6 }}
   357→            className={`text-6xl font-bold tracking-tight ${scoreTier(score).color}`}
   358→          >
   359→            {score}
   360→          </motion.span>
   361→        )}
   362→        <span className="text-xs text-muted-foreground tracking-wider uppercase mt-1">
   363→          / 100
   364→        </span>
   365→      </div>
   366→    </div>
   367→  );
   368→}
