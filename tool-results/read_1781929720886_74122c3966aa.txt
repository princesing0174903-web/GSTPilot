   160→// ─── Helpers ───────────────────────────────────────────────────────────────────
   161→
   162→function formatINR(n: number): string {
   163→  const sign = n < 0 ? '-' : '';
   164→  const abs = Math.abs(n);
   165→  return (
   166→    sign +
   167→    new Intl.NumberFormat('en-IN', {
   168→      maximumFractionDigits: 0,
   169→    }).format(abs)
   170→  );
   171→}
   172→
   173→function formatMonth(m: string): string {
   174→  const months = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];
   175→  const [y, mo] = m.split('-').map(Number);
   176→  if (!y || !mo) return m;
   177→  return `${months[mo - 1]} ${y}`;
   178→}
   179→
   180→function complianceBadgeClass(status: 'compliant' | 'at_risk' | 'non_compliant'): string {
   181→  if (status === 'compliant') return 'bg-emerald-500/15 text-emerald-300 border-emerald-500/30';
   182→  if (status === 'at_risk')    return 'bg-amber-500/15 text-amber-300 border-amber-500/30';
   183→  return 'bg-red-500/15 text-red-300 border-red-500/30';
   184→}

   244→    setLoadingConnections(true);
   245→    try {
   246→      const res = await fetch('/api/connections', { cache: 'no-store' });
   247→      if (!res.ok) return;
   248→      const json = (await res.json()) as { connections: ConnectionSummary[] };
   249→      setConnections(json.connections ?? []);
   250→    } catch {
   251→      // swallow — non-fatal
   252→    } finally {
   253→      setLoadingConnections(false);
   254→    }
   255→  }, []);
   256→
   257→  // ─── Broadcast connections-updated ──────────────────────────────────────────
   258→  const broadcastUpdate = useCallback(() => {

   261→
   262→  // ─── Connect GSTN ───────────────────────────────────────────────────────────
   263→  const handleConnectGstn = useCallback(async () => {
   264→    setGstnError(null);
   265→    const value = gstin.toUpperCase().trim();
   266→    if (!value) {
   267→      setGstnError('Enter your 15-character GSTIN to continue.');
   268→      return;
   269→    }
   270→    if (!GSTIN_REGEX.test(value)) {
   271→      setGstnError('Invalid GSTIN format. Example: 27ABCDE1234F1Z5');
   272→      return;
   273→    }
   274→    setGstnLoading(true);
   275→    setGstnResult(null);
   276→    try {
   277→      const res = await fetch('/api/connections', {
   278→        method: 'POST',
   279→        headers: { 'Content-Type': 'application/json' },
   280→        body: JSON.stringify({ type: 'gstn', gstin: value }),
   281→      });
   282→      const json = (await res.json()) as { ok?: true; error?: string; connection?: unknown; dataset?: unknown };
   283→      if (!res.ok || !json.ok) {
   284→        setGstnError(json.error ?? 'Verification failed. Please try again.');
   285→        return;
   286→      }
   287→      setGstnResult(json as unknown as GstnConnectResult);
   288→      broadcastUpdate();
   289→      void loadConnections();
   290→    } catch {
   291→      setGstnError('Network error. Check your connection and try again.');
   292→    } finally {
   293→      setGstnLoading(false);
   294→    }
   295→  }, [gstin, broadcastUpdate, loadConnections]);
   296→
   297→  // ─── Connect Bank ───────────────────────────────────────────────────────────

   299→    setBankError(null);
   300→    if (!bankProvider) {
   301→      setBankError('Select your bank first.');
   302→      return;
   303→    }
   304→    const ref = accountRef.replace(/\s+/g, '');
   305→    if (ref.length < 4) {
   306→      setBankError('Enter a valid account number (at least 4 digits).');
   307→      return;
   308→    }
   309→    setBankLoading(true);
   310→    setBankResult(null);
   311→    try {
   312→      const res = await fetch('/api/connections', {
   313→        method: 'POST',
   314→        headers: { 'Content-Type': 'application/json' },
   315→        body: JSON.stringify({ type: 'bank', provider: bankProvider, accountRef: ref }),
   316→      });
   317→      const json = (await res.json()) as { ok?: true; error?: string; connection?: unknown; dataset?: unknown };
   318→      if (!res.ok || !json.ok) {
   319→        setBankError(json.error ?? 'Connection failed. Please try again.');
   320→        return;
   321→      }
   322→      setBankResult(json as unknown as BankConnectResult);
   323→      broadcastUpdate();
   324→      void loadConnections();
   325→    } catch {
   326→      setBankError('Network error. Check your connection and try again.');
   327→    } finally {
   328→      setBankLoading(false);
   329→    }
   330→  }, [bankProvider, accountRef, broadcastUpdate, loadConnections]);
   331→
   332→  // ─── Disconnect a connection ────────────────────────────────────────────────
   333→  const handleDisconnect = useCallback(
   334→    async (id: string) => {
   335→      setDisconnectingId(id);
   336→      try {

   338→          method: 'DELETE',
   339→        });

   341→        // Optimistic: remove from list
   342→        setConnections((prev) => prev.filter((c) => c.id !== id));

   345→        if (bankResult?.connection.id === id) setBankResult(null);
   346→        broadcastUpdate();
   347→        // Refresh from server to stay in sync
   348→        void loadConnections();
   349→      } finally {
   350→        setDisconnectingId(null);
   351→      }
   352→    },
   353→    [gstnResult, bankResult, broadcastUpdate, loadConnections],
   354→  );
   355→
   356→  // ─── Reset form when dialog closed ──────────────────────────────────────────
   357→  useEffect(() => {
   358→    if (open) return;
   359→    // Don't wipe results immediately — wait for the close animation to finish.
   360→    const t = window.setTimeout(() => {
   361→      setGstin('');
   362→      setGstnError(null);
   363→      setGstnResult(null);
   364→      setBankProvider(null);
   365→      setAccountRef('');
   366→      setBankError(null);
   367→      setBankResult(null);

   381→          exit={{ opacity: 0 }}
   382→          transition={{ duration: 0.18 }}
   383→          className="fixed inset-0 z-[150] flex items-center justify-center p-4 sm:p-6"

   414→                <h2
