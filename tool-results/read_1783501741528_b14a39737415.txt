    55→function formatINR(n: number): string {
    56→  return new Intl.NumberFormat('en-IN', { style: 'currency', currency: 'INR', maximumFractionDigits: 0 }).format(n)
    57→}
    58→
    59→function formatDate(d: string | unknown): string {
    60→  if (!d) return '—'
    61→  try {
    62→    const date = typeof d === 'string' ? new Date(d) : new Date()
    63→    return date.toLocaleDateString('en-IN', { day: 'numeric', month: 'short', year: 'numeric' })
    64→  } catch { return '—' }
    65→}
    66→
    67→function formatTime(d: string | unknown): string {
    68→  if (!d) return ''
    69→  try {
    70→    const date = typeof d === 'string' ? new Date(d) : new Date()
    71→    return date.toLocaleTimeString('en-IN', { hour: '2-digit', minute: '2-digit' })
    72→  } catch { return '' }
    73→}
    74→
    75→function daysUntil(d: string | unknown): number {
    76→  if (!d) return 999
    77→  try {
    78→    const date = typeof d === 'string' ? new Date(d) : new Date()
    79→    return Math.ceil((date.getTime() - Date.now()) / (1000 * 60 * 60 * 24))
    80→  } catch { return 999 }
    81→}

    83→function scoreColor(score: number): string {
    84→  if (score >= 70) return 'text-emerald-600 bg-emerald-50 border-emerald-200'
    85→  if (score >= 40) return 'text-amber-600 bg-amber-50 border-amber-200'
    86→  return 'text-red-600 bg-red-50 border-red-200'
    87→}
    88→
    89→function scoreBgGradient(score: number): string {
    90→  if (score >= 70) return 'from-emerald-500 to-emerald-600'
    91→  if (score >= 40) return 'from-amber-500 to-amber-600'
    92→  return 'from-red-500 to-red-600'
    93→}

   359→          </span>

   363→      {lead.status !== 'converted' && lead.status !== 'lost' && (
   364→        <div className="mt-2 pt-2 border-t border-border/40 flex items-center gap-1.5 opacity-0 group-hover:opacity-100 transition-opacity">

   366→            <Button
   367→              variant="ghost"
   368→              size="sm"
   369→              className="h-6 text-[10px] px-2 text-emerald-600 hover:text-emerald-700 hover:bg-emerald-50"

   371→            >
   372→              <UserPlus className="h-3 w-3 mr-1" /> Convert
   373→            </Button>
   374→          )}
   375→          {PIPELINE_COLUMNS.findIndex(c => c.status === lead.status) < PIPELINE_COLUMNS.length - 2 && (
   376→            <Button
   377→              variant="ghost"
   378→              size="sm"
   379→              className="h-6 text-[10px] px-2 text-blue-600 hover:text-blue-700 hover:bg-blue-50"
   380→              onClick={() => {

   386→              }}

   388→              Next <ChevronRight className="h-3 w-3 ml-0.5" />
   389→            </Button>
   390→          )}
   391→        </div>
   392→      )}
   393→    </motion.div>
   394→  )
   395→}

   832→            </div>

  1267→function RevenueForecastView({ leads, deals, loading }: { leads: Array<FirestoreLead & { id: string }>; deals: Array<FirestoreDeal & { id: string }>; loading: boolean }) {
  1268→  const stageValues = useMemo(() => {
  1269→    const stages: Record<string, { label: string; value: number; color: string }> = {
  1270→      new: { label: 'New', value: 0, color: '#94a3b8' },

  1276→    leads.filter(l => l.status !== 'converted' && l.status !== 'lost').forEach(lead => {
  1277→      if (stages[lead.status]) stages[lead.status].value += lead.estimatedValue
  1278→    })
  1279→    return Object.values(stages)

  1281→
  1282→  const maxBarValue = useMemo(() => Math.max(...stageValues.map(s => s.value), 1), [stageValues])
  1283→
  1284→  const weightedPipeline = useMemo(() => {
  1285→    return deals.filter(d => d.stage !== 'closed_lost').reduce((sum, d) => sum + d.value * d.probability / 100, 0)
  1286→  }, [deals])
  1287→
  1288→  const totalPipeline = useMemo(() => {
  1289→    return deals.filter(d => d.stage !== 'closed_lost').reduce((sum, d) => sum + d.value, 0)
  1290→  }, [deals])
  1291→
  1292→  const monthlyTrend = useMemo(() => {
  1293→    // Compute from real deals — no Math.random fabrication.
  1294→    // Returns empty array when no deals exist (honest empty state).
  1295→    if (!deals || deals.length === 0) return []
  1296→    const months = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun']
  1297→    return months.map(m => {
