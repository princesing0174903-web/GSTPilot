    54→function BalanceTrendChart({ data }: { data: { day: string; balance: number }[] }) {
    55→  const max = Math.max(...data.map(d => d.balance))
    56→  const min = Math.min(...data.map(d => d.balance))
    57→  const range = max - min || 1
    58→  const w = 280
    59→  const h = 60
    60→  const pts = data.map((d, i) => `${(i / (data.length - 1)) * w},${h - ((d.balance - min) / range) * (h - 8) - 4}`)
    61→  return (
    62→    <svg width={w} height={h + 16} className="overflow-visible">
    63→      <defs>
    64→        <linearGradient id="balGrad" x1="0" y1="0" x2="0" y2="1">
    65→          <stop offset="0%" stopColor="#10b981" stopOpacity="0.2" />
    66→          <stop offset="100%" stopColor="#10b981" stopOpacity="0.02" />
    67→        </linearGradient>
    68→      </defs>
    69→      <polygon points={`0,${h} ${pts.join(' ')} ${w},${h}`} fill="url(#balGrad)" />
    70→      <polyline points={pts.join(' ')} fill="none" stroke="#10b981" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" />
    71→      {data.filter((_, i) => i % 2 === 0).map((d, i) => {
    72→        const x = (data.indexOf(d) / (data.length - 1)) * w
    73→        return <text key={i} x={x} y={h + 12} textAnchor="middle" className="text-[8px] fill-muted-foreground">{d.day}</text>
    74→      })}
    75→    </svg>
    76→  )
    77→}

    79→function ReconcileDonut({ reconciled, unreconciled }: { reconciled: number; unreconciled: number }) {
    80→  const total = reconciled + unreconciled
    81→  const r = 36
    82→  const c = 2 * Math.PI * r
    83→  const reconcilePct = (reconciled / total) * 100
    84→  const dashOffset = c - (reconcilePct / 100) * c
    85→  return (
    86→    <svg width="90" height="90" className="overflow-visible">
    87→      <circle cx="45" cy="45" r={r} fill="none" stroke="currentColor" strokeWidth="8" className="text-slate-100 dark:text-slate-800" />
    88→      <circle cx="45" cy="45" r={r} fill="none" stroke="#10b981" strokeWidth="8" strokeLinecap="round" strokeDasharray={c} strokeDashoffset={dashOffset} transform="rotate(-90 45 45)" />
    89→      <text x="45" y="42" textAnchor="middle" className="text-sm font-bold fill-emerald-600 dark:fill-emerald-400">{reconcilePct.toFixed(0)}%</text>
    90→      <text x="45" y="54" textAnchor="middle" className="text-[7px] fill-muted-foreground">Reconciled</text>
    91→    </svg>
    92→  )
    93→}

   172→function formatSyncDate(iso: string | null): string {
   173→  if (!iso) return '—'
   174→  const d = new Date(iso)
   175→  if (Number.isNaN(d.getTime())) return '—'
   176→  const dd = String(d.getDate()).padStart(2, '0')
   177→  const mm = String(d.getMonth() + 1).padStart(2, '0')
   178→  const yyyy = d.getFullYear()
   179→  const hh = String(d.getHours()).padStart(2, '0')
   180→  const min = String(d.getMinutes()).padStart(2, '0')
   181→  return `${dd}/${mm}/${yyyy} ${hh}:${min}`
   182→}
   183→
   184→function formatTxnDate(iso: string): string {
   185→  if (!iso) return '—'
   186→  const d = new Date(iso)
   187→  if (Number.isNaN(d.getTime())) return iso
   188→  const dd = String(d.getDate()).padStart(2, '0')
   189→  const mm = String(d.getMonth() + 1).padStart(2, '0')
   190→  const yyyy = d.getFullYear()
   191→  return `${dd}/${mm}/${yyyy}`
   192→}

   194→function upperMode(mode: string): string {
   195→  return (mode || 'bank').toUpperCase()
   196→}
   197→
   198→// Map Firestore BankAccount → BankAccount shape used by this page.
   199→function mapBankAccount(acc: FirestoreBankAccount & { id: string }): BankAccount {
   200→  const accountType = (acc.accountType || 'current') as string
   201→  return {
   202→    id: acc.id,
   203→    bank: acc.bankName || 'Bank Account',
   204→    account: acc.accountNumberMasked || '****',
   205→    type: accountType.charAt(0).toUpperCase() + accountType.slice(1),
   206→    balance: Number(acc.currentBalance ?? acc.availableBalance ?? 0) || 0,
   207→    lastSync: formatSyncDate((acc.lastSyncAt as string | null) ?? null),
   208→    status: (acc.status as string) || 'connected',
   209→  }
   210→}
   211→
   212→// Map Firestore BankTransaction → BankTransaction shape used by this page.
   213→function mapBankTxnToTxn(t: FirestoreBankTransaction & { id: string }): BankTransaction {
   214→  const type = t.type === 'credit' ? 'credit' : 'debit'
   215→  return {
   216→    id: t.id,
   217→    date: formatTxnDate(t.date || ''),
   218→    description: t.description || 'Bank Transaction',
   219→    amount: Math.abs(Number(t.amount) || 0),
   220→    type,
   221→    balance: t.balanceAfter == null ? null : Number(t.balanceAfter),
   222→    account: t.referenceNo ? t.referenceNo.toUpperCase() : 'BANK',
   223→    category: (t.category as string) || (type === 'credit' ? 'Revenue' : 'Purchase'),
   224→  }
   225→}
   226→
   227→// Map Firestore Payment row → BankTransaction shape used by this page.
   228→function mapPaymentToTxn(p: FirestorePayment & { id: string }): BankTransaction {
   229→  const isVendor = (p.partyType || 'customer') === 'vendor'
   230→  return {
   231→    id: p.id,
   232→    date: formatTxnDate(p.paymentDate || ''),
   233→    description: `${upperMode(p.paymentMode || 'bank')} - ${p.partyName || 'Unknown'}`,
   234→    amount: Number(p.amount) || 0,
   235→    type: isVendor ? 'debit' : 'credit',
   236→    balance: null,
   237→    account: upperMode(p.paymentMode || 'bank'),
   238→    category: isVendor ? 'Purchase' : 'Revenue',
   239→  }
   240→}
   241→
   242→// Map Firestore Payment row → ReconciliationEntry shape used by this page.
   243→function mapPaymentToRecon(p: FirestorePayment & { id: string }): ReconciliationEntry {
   244→  let status: ReconciliationEntry['status'] = 'unmatched'
   245→  if (p.reconciled) status = 'matched'
   246→  else if (p.status === 'failed') status = 'disputed'
   247→  return {
   248→    id: p.id,
   249→    date: formatTxnDate(p.paymentDate || ''),
   250→    bankTxn: p.referenceNo || `${upperMode(p.paymentMode || 'bank')}-${p.id.slice(-6)}`,
   251→    bookEntry: p.invoiceId || null,
   252→    amount: Number(p.amount) || 0,
   253→    status,
   254→    account: upperMode(p.paymentMode || 'bank'),
   255→  }
   256→}
   257→
   258→// Map Firestore Expense → BankTransaction (always a debit) shape used by this page.
   259→function mapExpenseToTxn(e: FirestoreExpense & { id: string }): BankTransaction {
   260→  const mode = (e.paymentMode || 'bank') as string
   261→  return {
   262→    id: e.id,
   263→    date: formatTxnDate(e.date || ''),
   264→    description: `${upperMode(mode)} - ${e.vendor || e.description || 'Expense'}`,
   265→    amount: Number(e.amount) || 0,
   266→    type: 'debit',
   267→    balance: null,
   268→    account: upperMode(mode),
   269→    category: (e.category as string) || 'Purchase',
   270→  }
   271→}
