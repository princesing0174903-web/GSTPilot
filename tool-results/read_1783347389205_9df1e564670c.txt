   461→function AnimatedCounter({ value, unit, accent }: { value: number; unit: string; accent: string }) {
   462→  const [display, setDisplay] = useState(value);
   463→  useEffect(() => {
   464→    // Deterministic small oscillation (no Math.random)
   465→    let tick = 0;
   466→    const id = setInterval(() => {
   467→      tick += 1;
   468→      // Pseudorandom using tick, but stays deterministic per render
   469→      const wobble = ((tick * 7) % 11) - 5; // -5..+5
   470→      setDisplay(Math.max(0, value + wobble));
   471→    }, 2000);
   472→    return () => clearInterval(id);
   473→  }, [value]);
   474→  const decimals = value < 1 ? 2 : value < 100 ? 0 : 0;
   475→  return (
   476→    <span className={cn('font-mono font-bold tabular-nums', accent)}>
   477→      {display.toLocaleString('en-US', { minimumFractionDigits: decimals, maximumFractionDigits: decimals })}
   478→      <span className="text-[10px] text-muted-foreground ml-1">{unit}</span>
   479→    </span>
   480→  );
   481→}
