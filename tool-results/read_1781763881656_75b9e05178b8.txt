    65→// Relative time formatter — defensively handles ISO string / Date / Firestore Timestamp.
    66→function timeAgo(createdAt: unknown): string {
    67→  if (!createdAt) return 'just now';
    68→  let date: Date;
    69→  if (createdAt instanceof Date) {
    70→    date = createdAt;
    71→  } else if (typeof createdAt === 'string') {
    72→    date = new Date(createdAt);
    73→  } else if (
    74→    typeof createdAt === 'object' &&
    75→    createdAt !== null &&
    76→    'toDate' in createdAt &&
    77→    typeof (createdAt as { toDate: () => Date }).toDate === 'function'
    78→  ) {
    79→    date = (createdAt as { toDate: () => Date }).toDate();
    80→  } else {
    81→    return 'just now';
    82→  }
    83→  if (isNaN(date.getTime())) return 'just now';
    84→  const seconds = Math.floor((Date.now() - date.getTime()) / 1000);
    85→  if (seconds < 60) return 'just now';
    86→  const minutes = Math.floor(seconds / 60);
    87→  if (minutes < 60) return `${minutes}m ago`;
    88→  const hours = Math.floor(minutes / 60);
    89→  if (hours < 24) return `${hours}h ago`;
    90→  const days = Math.floor(hours / 24);
    91→  if (days < 7) return `${days}d ago`;
    92→  const weeks = Math.floor(days / 7);
    93→  if (weeks < 5) return `${weeks}w ago`;
    94→  const months = Math.floor(days / 30);
    95→  if (months < 12) return `${months}mo ago`;
    96→  const years = Math.floor(days / 365);
    97→  return `${years}y ago`;
    98→}

   100→// ─── Business Score computation ───────────────────────────────────────────────
   101→// Start at 100. Subtract weighted penalties for risk signals. Clamp 0–100.
   102→// If there is no data at all, return null (UI shows premium empty state).
   103→
   104→function computeBusinessScore(opts: {
   105→  totalClients: number;
   106→  criticalIssues: number;
   107→  warnings: number;
   108→  pendingReturns: number;
   109→  overdueReturns: number;
   110→  averageHealthScore: number;
   111→  matchPercentage: number;
   112→}): number | null {
   113→  const { totalClients, criticalIssues, warnings, pendingReturns, overdueReturns, averageHealthScore, matchPercentage } = opts;
   114→  if (totalClients === 0 && pendingReturns === 0 && overdueReturns === 0) return null;
   115→
   116→  let score = 100;
   117→  score -= criticalIssues * 6;
   118→  score -= warnings * 2;
   119→  score -= overdueReturns * 8;
   120→  score -= pendingReturns * 2;
   121→  if (averageHealthScore > 0) {
   122→    score -= Math.max(0, 80 - averageHealthScore) * 0.6;
   123→  }
   124→  score -= Math.max(0, 100 - matchPercentage) * 0.3;
   125→  return Math.max(0, Math.min(100, Math.round(score)));
   126→}

   128→// V17 palette — only Emerald / Cyan / Blue.
   129→function scoreTier(score: number | null): { label: string; color: string } {
   130→  if (score === null) return { label: '—', color: 'text-muted-foreground' };
   131→  if (score >= 85) return { label: 'Excellent', color: 'text-emerald-400' };
   132→  if (score >= 70) return { label: 'Healthy', color: 'text-cyan-400' };
   133→  if (score >= 50) return { label: 'At Risk', color: 'text-blue-400' };
   134→  return { label: 'Critical', color: 'text-blue-300' };
   135→}

   149→function buildInsights(opts: {
   150→  metrics: {
   151→    totalClients: number;
   152→    pendingReturns: number;
   153→    overdueReturns: number;
   154→    criticalIssues: number;
   155→    warnings: number;
   156→    totalTaxVolume: number;
   157→    averageHealthScore: number;
   158→    matchPercentage: number;
   159→    upcomingFilings: FirestoreReturn[];
   160→  };
   161→  clients: Array<FirestoreClient & { id: string }>;
   162→}): AIInsight[] {
   163→  const { metrics, clients } = opts;
   164→  const insights: AIInsight[] = [];
   165→
   166→  if (metrics.overdueReturns > 0) {
   167→    insights.push({
   168→      id: 'overdue-returns',
   169→      icon: AlertTriangle,
   170→      tone: 'risk',
   171→      what: `${metrics.overdueReturns} GST return${metrics.overdueReturns > 1 ? 's are' : ' is'} overdue`,
   172→      why: 'Overdue returns trigger notices and late fees. File immediately to avoid penalties.',
   173→      actionLabel: 'File Now',
   174→      actionView: 'returns',
   175→    });
   176→  }
   177→
   178→  const nextFiling = metrics.upcomingFilings[0];
   179→  if (nextFiling && metrics.pendingReturns > 0 && metrics.overdueReturns === 0) {
   180→    insights.push({
   181→      id: 'gst-due',
   182→      icon: Clock,
   183→      tone: 'info',
   184→      what: `GST filing due — ${nextFiling.returnType} for ${nextFiling.period}`,
   185→      why: `${metrics.pendingReturns} return${metrics.pendingReturns > 1 ? 's' : ''} pending. Filing on time keeps compliance score high.`,
   186→      actionLabel: 'File GST',
   187→      actionView: 'returns',
   188→    });
   189→  }
   190→
   191→  const riskyClients = clients.filter(c => (c.healthScore ?? 100) < 60);
   192→  if (metrics.criticalIssues > 0 || riskyClients.length > 0) {
   193→    insights.push({
   194→      id: 'churn-risk',
   195→      icon: TrendingDown,
   196→      tone: 'risk',
   197→      what: riskyClients.length > 0
   198→        ? `${riskyClients.length} client${riskyClients.length > 1 ? 's may' : ' may'} churn`
   199→        : `${metrics.criticalIssues} critical issue${metrics.criticalIssues > 1 ? 's' : ''} need attention`,
   200→      why: riskyClients.length > 0
   201→        ? 'Low health scores signal dissatisfaction. Reach out before they leave.'
   202→        : 'Critical issues compound into bigger problems if left unaddressed.',
   203→      actionLabel: riskyClients.length > 0 ? 'View Clients' : 'Review Issues',
   204→      actionView: riskyClients.length > 0 ? 'clients' : 'ai-operating-room',
   205→    });
   206→  }
   207→
   208→  if (metrics.totalTaxVolume > 0) {
   209→    insights.push({
   210→      id: 'collections',
   211→      icon: Wallet,
   212→      tone: metrics.matchPercentage < 90 ? 'risk' : 'info',
   213→      what: metrics.matchPercentage < 90
   214→        ? 'Collections may drop — reconciliation gaps detected'
   215→        : 'Cash position is healthy',
   216→      why: metrics.matchPercentage < 90
   217→        ? `${metrics.matchPercentage.toFixed(0)}% of invoices reconciled. Unmatched invoices mean uncollected revenue.`
   218→        : `${metrics.matchPercentage.toFixed(0)}% invoice match rate. Keep chasing pending collections to maintain runway.`,
   219→      actionLabel: 'Fix Collections',
   220→      actionView: 'reconcile',
   221→    });
   222→  }
   223→
   224→  if (metrics.averageHealthScore > 0) {
   225→    const tier = metrics.averageHealthScore >= 80 ? 'success' : 'info';
   226→    insights.push({
   227→      id: 'compliance',
   228→      icon: ShieldAlert,
   229→      tone: tier,
   230→      what: `Compliance health at ${metrics.averageHealthScore.toFixed(0)}%`,
   231→      why: metrics.averageHealthScore >= 80
   232→        ? 'Strong compliance posture. Maintain by filing on time and resolving warnings.'
   233→        : 'Below optimal. Address pending returns and warnings to lift your score.',
   234→      actionLabel: 'View Compliance',
   235→      actionView: 'returns',
   236→    });
   237→  }
   238→
   239→  if (insights.length === 0) {
   240→    insights.push({
   241→      id: 'all-clear',
   242→      icon: Sparkles,
   243→      tone: 'success',
   244→      what: 'Everything looks good today',
   245→      why: 'No urgent issues. Hand control to Oracle and let it run the routine work.',
   246→      actionLabel: 'Run My Business',
   247→      actionView: 'run-my-business',
   248→    });
   249→  }
   250→
   251→  return insights.slice(0, 6);
   252→}

   256→function ScoreGauge({ score, size = 180 }: { score: number; size?: number }) {
   257→  const stroke = 12;
   258→  const r = (size - stroke) / 2;
   259→  const c = 2 * Math.PI * r;
   260→  const pct = score / 100;
   261→  const offset = c * (1 - pct);
   262→
   263→  return (
   264→    <div className="relative" style={{ width: size, height: size }}>
   265→      <svg width={size} height={size} className="-rotate-90">
   266→        <defs>
   267→          <linearGradient id="missionScoreGradient" x1="0%" y1="0%" x2="100%" y2="100%">
   268→            <stop offset="0%" stopColor="#10b981" />
   269→            <stop offset="50%" stopColor="#06b6d4" />
   270→            <stop offset="100%" stopColor="#3b82f6" />
   271→          </linearGradient>
   272→        </defs>
   273→        <circle
   274→          cx={size / 2}
   275→          cy={size / 2}
   276→          r={r}
   277→          fill="none"
   278→          stroke="rgba(255,255,255,0.06)"
   279→          strokeWidth={stroke}
   280→        />
   281→        <motion.circle
   282→          cx={size / 2}
   283→          cy={size / 2}
   284→          r={r}
   285→          fill="none"
   286→          stroke="url(#missionScoreGradient)"
   287→          strokeWidth={stroke}
   288→          strokeLinecap="round"
   289→          strokeDasharray={c}
   290→          initial={{ strokeDashoffset: c }}
   291→          animate={{ strokeDashoffset: offset }}
   292→          transition={{ duration: 1.2, ease: 'easeOut', delay: 0.2 }}
   293→        />
   294→      </svg>
   295→      <div className="absolute inset-0 flex flex-col items-center justify-center">
   296→        <motion.span
   297→          initial={{ opacity: 0, scale: 0.8 }}
   298→          animate={{ opacity: 1, scale: 1 }}
   299→          transition={{ duration: 0.5, delay: 0.6 }}
   300→          className={`text-5xl font-bold tracking-tight ${scoreTier(score).color}`}
   301→        >
   302→          {score}
   303→        </motion.span>
   304→        <span className="text-[10px] text-muted-foreground tracking-wider uppercase mt-1">
   305→          / 100
   306→        </span>
   307→      </div>
   308→    </div>
   309→  );
   310→}
