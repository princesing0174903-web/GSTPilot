   208→function formatRelativeTime(timestamp: string): string {
   209→  const now = new Date();
   210→  const date = new Date(timestamp);
   211→  const diffMs = now.getTime() - date.getTime();
   212→  const diffMins = Math.floor(diffMs / 60000);
   213→  const diffHours = Math.floor(diffMins / 60);
   214→  const diffDays = Math.floor(diffHours / 24);
   215→
   216→  if (diffMins < 60) return `${diffMins}m ago`;
   217→  if (diffHours < 24) return `${diffHours}h ago`;
   218→  if (diffDays < 7) return `${diffDays}d ago`;
   219→  return date.toLocaleDateString('en-IN', { day: 'numeric', month: 'short' });
   220→}

   234→/** Map API filing status to UI ReturnEntry status */
   235→function mapFilingStatus(status: FilingStatus, period: string): ReturnEntry['status'] {
   236→  if (status === 'filed') return 'Filed';
   237→  const readyStatuses: FilingStatus[] = ['validated', 'reviewed', 'generated'];
   238→  if (readyStatuses.includes(status)) {
   239→    return isFilingOverdue(period) ? 'Overdue' : 'Ready to File';
   240→  }
   241→  return isFilingOverdue(period) ? 'Overdue' : 'Draft';
   242→}

   244→/** Map an API document to a DocumentEntry for the UI */
   245→function mapApiDocument(doc: Record<string, unknown>): DocumentEntry {
   246→  const statusStr = (doc.processingStep as string) ?? (doc.status as string) ?? 'uploaded';
   247→  const statusMap: Record<string, DocumentEntry['status']> = {
   248→    extracted: 'Processed',
   249→    processing: 'Processing',
   250→    processed: 'Processed',
   251→    uploaded: 'Uploaded',
   252→    failed: 'Error',
   253→    error: 'Error',
   254→  };
   255→  const sizeBytes = (doc.size as number) ?? 0;
   256→  const sizeStr = sizeBytes > 1048576
   257→    ? `${(sizeBytes / 1048576).toFixed(1)} MB`
   258→    : sizeBytes > 1024
   259→      ? `${(sizeBytes / 1024).toFixed(0)} KB`
   260→      : sizeBytes > 0
   261→        ? `${sizeBytes} B`
   262→        : '1.0 MB';
   263→
   264→  return {
   265→    id: (doc.id as string) ?? '',
   266→    name: (doc.name as string) ?? (doc.filename as string) ?? 'Document',
   267→    type: (doc.fileType as string) ?? (doc.folder as string) ?? 'Document',
   268→    uploadDate: ((doc.createdAt as string) ?? new Date().toISOString()).split('T')[0],
   269→    status: statusMap[statusStr.toLowerCase()] ?? 'Uploaded',
   270→    invoicesExtracted: (doc.invoiceCount as number) ?? 0,
   271→    totalRows: (doc.rowCount as number) ?? 0,
   272→    accuracy: (doc.accuracy as number) ?? 0,
   273→    size: sizeStr,
   274→  };
   275→}

   277→/** Map an API reconciliation run to a UI ReconRun */
   278→function mapApiReconRun(run: ReconciliationRun): ReconRun {
   279→  const matchRate = run.totalRecords > 0 ? Math.round((run.matched / run.totalRecords) * 100) : 0;
   280→  return {
   281→    id: run.id,
   282→    period: periodToLabel(run.period),
   283→    matchRate,
   284→    mismatches: run.unmatched + run.partialMatches,
   285→    missingInvoices: run.unmatched,
   286→    taxDifference: run.gstDifference,
   287→    runDate: run.createdAt.split('T')[0],
   288→    status: run.status === 'completed' ? 'Completed' as const : run.status === 'running' ? 'Running' as const : 'Failed' as const,
   289→  };
   290→}

   292→/** Map an API audit log to an ActivityEvent for the UI */
   293→function mapAuditLogToEvent(log: AuditLogEntry): ActivityEvent {
   294→  const action = log.action?.toLowerCase() ?? '';
   295→  let type: ActivityEvent['type'] = 'user_action';
   296→  if (action.includes('upload') || action.includes('import')) type = 'upload';
   297→  else if (action.includes('file') || action.includes('submit')) type = 'filing_submitted';
   298→  else if (action.includes('reconcil')) type = 'recon_run';
   299→  else if (action.includes('valid') || action.includes('generat')) type = 'ai_action';
   300→  else if (action.includes('return') || action.includes('creat')) type = 'return_created';
   301→
   302→  return {
   303→    id: log.id,
   304→    type,
   305→    description: log.details ?? log.action,
   306→    timestamp: log.timestamp,
   307→  };
   308→}

   310→/** Map issue category to pending action type */
   311→function issueCategoryToActionType(category: string): PendingAction['type'] {
   312→  const lower = category.toLowerCase();
   313→  if (lower.includes('gstin') || lower.includes('invalid')) return 'gstin_error';
   314→  if (lower.includes('missing') || lower.includes('mandatory') || lower.includes('hsn')) return 'missing_documents';
   315→  if (lower.includes('duplicate') || lower.includes('calculation') || lower.includes('tax')) return 'recon_mismatch';
   316→  return 'awaiting_review';
   317→}

   319→/** Map API issue to AIInsight */
   320→function mapIssueToInsight(issue: Issue): AIInsight {
   321→  const severity: AIInsight['severity'] =
   322→    issue.severity === 'critical' ? 'critical' : issue.severity === 'warning' ? 'warning' : 'info';
   323→  const category = issue.category ?? 'General';
   324→
   325→  let actionView: AppView = 'dashboard';
   326→  let suggestedAction = 'Review';
   327→  if (category.toLowerCase().includes('gstin') || category.toLowerCase().includes('tax')) {
   328→    actionView = 'reconcile';
   329→    suggestedAction = 'Review Details';
   330→  } else if (category.toLowerCase().includes('missing') || category.toLowerCase().includes('document')) {
   331→    actionView = 'invoices';
   332→    suggestedAction = 'Upload Documents';
   333→  } else if (category.toLowerCase().includes('filing') || category.toLowerCase().includes('deadline')) {
   334→    actionView = 'returns';
   335→    suggestedAction = 'File Return';
   336→  }
   337→
   338→  return {
   339→    id: issue.id,
   340→    severity,
   341→    title: issue.title ?? category,
   342→    description: issue.description ?? '',
   343→    suggestedAction,
   344→    actionView,
   345→    category,
   346→  };
   347→}

   353→function HealthRing({ score, size = 72, strokeWidth = 5 }: { score: number; size?: number; strokeWidth?: number }) {
   354→  const radius = (size - strokeWidth) / 2;
   355→  const circumference = 2 * Math.PI * radius;
   356→  const offset = circumference - (score / 100) * circumference;
   357→  const color = getHealthColor(score);
   358→
   359→  return (
   360→    <div className="relative inline-flex items-center justify-center" style={{ width: size, height: size }}>
   361→      <svg width={size} height={size} className="-rotate-90">
   362→        <circle cx={size / 2} cy={size / 2} r={radius} fill="none" stroke="#e2e8f0" strokeWidth={strokeWidth} />
   363→        <motion.circle cx={size / 2} cy={size / 2} r={radius} fill="none" stroke={color.stroke} strokeWidth={strokeWidth} strokeLinecap="round" strokeDasharray={circumference} initial={{ strokeDashoffset: circumference }} animate={{ strokeDashoffset: offset }} transition={{ duration: 1, delay: 0.3, ease: 'easeOut' }} />
   364→      </svg>
   365→      <div className="absolute inset-0 flex flex-col items-center justify-center">
   366→        <span className={`font-bold text-lg ${color.text}`}>{score}</span>
   367→        <span className="text-[8px] text-muted-foreground">/ 100</span>
   368→      </div>
   369→    </div>
   370→  );
   371→}
