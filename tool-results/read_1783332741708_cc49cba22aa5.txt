   198→// ─── Save Button with Animated States ──────────────────────────────────
   199→function SaveButton({ onSave }: { onSave: () => Promise<void> }) {
   200→  const [state, setState] = useState<'idle' | 'saving' | 'saved'>('idle')
   201→
   202→  const handleClick = async () => {
   203→    setState('saving')
   204→    try {
   205→      await onSave()
   206→      setState('saved')
   207→      setTimeout(() => setState('idle'), 2000)
   208→    } catch {
   209→      setState('idle')
   210→    }
   211→  }
   212→
   213→  return (
   214→    <Button
   215→      onClick={handleClick}
   216→      disabled={state === 'saving'}
   217→      className="bg-emerald-600 hover:bg-emerald-700 text-white gap-2 min-w-[110px]"
   218→    >
   219→      <AnimatePresence mode="wait">
   220→        {state === 'idle' && (
   221→          <motion.span
   222→            key="idle"
   223→            initial={{ opacity: 0, scale: 0.8 }}
   224→            animate={{ opacity: 1, scale: 1 }}
   225→            exit={{ opacity: 0, scale: 0.8 }}
   226→            className="flex items-center gap-2"
   227→          >
   228→            <Save className="h-4 w-4" />
   229→            Save
   230→          </motion.span>
   231→        )}
   232→        {state === 'saving' && (
   233→          <motion.span
   234→            key="saving"
   235→            initial={{ opacity: 0, scale: 0.8 }}
   236→            animate={{ opacity: 1, scale: 1 }}
   237→            exit={{ opacity: 0, scale: 0.8 }}
   238→            className="flex items-center gap-2"
   239→          >
   240→            <Loader2 className="h-4 w-4 animate-spin" />
   241→            Saving...
   242→          </motion.span>

   245→          <motion.span
   246→            key="saved"

   261→// ─── Password Strength Helper ──────────────────────────────────────────
   262→function getPasswordStrength(password: string): { score: number; label: string; color: string } {
   263→  if (!password) return { score: 0, label: '', color: '' }
   264→  let score = 0
   265→  if (password.length >= 8) score++
   266→  if (password.length >= 12) score++
   267→  if (/[A-Z]/.test(password)) score++
   268→  if (/[0-9]/.test(password)) score++
   269→  if (/[^A-Za-z0-9]/.test(password)) score++
   270→
   271→  if (score <= 1) return { score: 20, label: 'Weak', color: 'bg-red-500' }
   272→  if (score <= 2) return { score: 40, label: 'Fair', color: 'bg-orange-500' }
   273→  if (score <= 3) return { score: 60, label: 'Good', color: 'bg-yellow-500' }
   274→  if (score <= 4) return { score: 80, label: 'Strong', color: 'bg-emerald-500' }
   275→  return { score: 100, label: 'Excellent', color: 'bg-emerald-600' }
   276→}
