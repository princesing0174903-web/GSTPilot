   104→export function OracleWorkspace({
   105→  open, onClose, onNavigate, userName, firmName, gstin, initialPrompt,
   106→}: OracleWorkspaceProps) {
   107→  const [store, setStore] = useState<ConversationStore>({ conversations: [], activeId: null });
   108→  // Portal guard: only render into document.body after mount to avoid SSR
   109→  // hydration mismatch (server has no document.body).
   110→  const [mounted, setMounted] = useState(false);
   111→  useEffect(() => setMounted(true), []);
   112→  const [messages, setMessages] = useState<OracleMessage[]>([]);
   113→  const [activeId, setActiveId] = useState<string | null>(null);
   114→  const [input, setInput] = useState('');
   115→  const [isStreaming, setIsStreaming] = useState(false);
   116→  const [lastEmotion, setLastEmotion] = useState<OracleMessage['emotion']>('helpful');
   117→  const [activeLanguage, setActiveLanguage] = useState<OracleMessage['language']>('english');
   118→  // Mobile: history drawer open state
   119→  const [historyOpen, setHistoryOpen] = useState(false);
   120→
   121→  const scrollRef = useRef<HTMLDivElement>(null);
   122→  const inputRef = useRef<HTMLTextAreaElement>(null);
   123→  const abortRef = useRef<AbortController | null>(null);
   124→  const userPinnedUpRef = useRef(false);
   125→  const streamingIdRef = useRef<string | null>(null);
   126→  // Tracks the last initialPrompt we auto-sent, so we never fire it twice.
   127→  const initialPromptSentRef = useRef<string | null>(null);
   128→
   129→  // ─── Live business data cache (30s TTL) ────────────────────────────────────
   130→  // Fetches /api/oracle/context once per 30s window so we don't hit it on every
   131→  // single message. Invalidated immediately on `connections-updated` events so a
   132→  // freshly connected GSTN/Bank shows up in the very next prompt.
   133→  const liveDataCacheRef = useRef<{ data: OracleLiveData | undefined; ts: number } | null>(null);
   134→  const fetchLiveData = useCallback(async (): Promise<OracleLiveData | undefined> => {
   135→    const now = Date.now();
   136→    if (liveDataCacheRef.current && now - liveDataCacheRef.current.ts < 30000) {
   137→      return liveDataCacheRef.current.data;
   138→    }
   139→    try {
   140→      const res = await fetch('/api/oracle/context');
   141→      if (res.ok) {
   142→        const data = (await res.json()) as OracleLiveData;
   143→        liveDataCacheRef.current = { data, ts: now };
   144→        return data;
   145→      }
   146→    } catch {
   147→      /* ignore — chat still works without context */
   148→    }
   149→    return undefined;
   150→  }, []);
   151→
   152→  useEffect(() => {
   153→    const handler = () => {
   154→      liveDataCacheRef.current = null;
   155→    };
   156→    window.addEventListener('connections-updated', handler);
   157→    return () => window.removeEventListener('connections-updated', handler);
   158→  }, []);
   159→
   160→  // ─── Load + migrate store on open ──────────────────────────────────────────
   161→  useEffect(() => {
   162→    if (!open) return;
   163→    const loaded = loadStore();
   164→    if (loaded.conversations.length === 0) {
   165→      // Migrate legacy single-conversation store if present.
   166→      try {
   167→        const raw = localStorage.getItem(LEGACY_KEY);
   168→        if (raw) {
   169→          const parsed = JSON.parse(raw) as OracleMessage[];
   170→          if (Array.isArray(parsed) && parsed.length > 0) {
   171→            const now = new Date().toISOString();
   172→            const conv: OracleConversation = {
   173→              id: cryptoId(),
   174→              title: deriveTitle(parsed),
   175→              messages: parsed.map((m) => (m.streaming ? { ...m, streaming: false } : m)),
   176→              createdAt: parsed[0]?.createdAt ?? now,
   177→              updatedAt: now,
   178→            };
   179→            const next: ConversationStore = { conversations: [conv], activeId: conv.id };
   180→            setStore(next);
   181→            setActiveId(conv.id);
   182→            setMessages(conv.messages);
   183→            saveStore(next);
   184→            return;
   185→          }
   186→        }
   187→      } catch {
   188→        /* ignore */
   189→      }
   190→      // Nothing to migrate — start fresh with one empty conversation.
   191→      const conv = newConversation();
   192→      const next: ConversationStore = { conversations: [conv], activeId: conv.id };
   193→      setStore(next);
   194→      setActiveId(conv.id);
   195→      setMessages([]);
   196→      saveStore(next);
   197→      return;
   198→    }
   199→    setStore(loaded);
   200→    const active = loaded.conversations.find((c) => c.id === loaded.activeId) ?? loaded.conversations[0];
   201→    if (active) {
   202→      setActiveId(active.id);
   203→      setMessages(active.messages);
   204→    }
   205→  }, [open]);
   206→
   207→  // ─── Persist active conversation whenever messages change ───────────────────
   208→  useEffect(() => {
   209→    if (!open || !activeId) return;
   210→    setStore((prev) => {
   211→      if (!prev.conversations.length) return prev;
   212→      const exists = prev.conversations.some((c) => c.id === activeId);
   213→      let conversations: OracleConversation[];
   214→      let nextActiveId = prev.activeId;
   215→      if (exists) {
   216→        conversations = prev.conversations.map((c) =>
   217→          c.id === activeId
   218→            ? {
   219→                ...c,
   220→                messages,
   221→                title: deriveTitle(messages) || c.title,
   222→                updatedAt: new Date().toISOString(),
   223→              }
   224→            : c,
   225→        );
   226→      } else {
   227→        // Active conversation doesn't exist yet — create it on the fly.
   228→        const conv: OracleConversation = {
   229→          id: activeId,
   230→          title: deriveTitle(messages) || 'New chat',
   231→          messages,
   232→          createdAt: new Date().toISOString(),
   233→          updatedAt: new Date().toISOString(),
   234→        };
   235→        conversations = [conv, ...prev.conversations];
   236→        nextActiveId = activeId;
   237→      }
   238→      const next = { conversations, activeId: nextActiveId };
   239→      saveStore(next);
   240→      return next;
   241→    });
   242→  }, [messages, open, activeId]);
   243→
   244→  // ─── Body scroll lock while open ───────────────────────────────────────────
   245→  useEffect(() => {
   246→    if (!open) return;
   247→    const prev = document.body.style.overflow;
   248→    document.body.style.overflow = 'hidden';
   249→    return () => {
   250→      document.body.style.overflow = prev;
   251→    };
   252→  }, [open]);
   253→
   254→  // ─── Escape to close ───────────────────────────────────────────────────────
   255→  useEffect(() => {
   256→    if (!open) return;
   257→    const onKey = (e: globalThis.KeyboardEvent) => {
   258→      if (e.key === 'Escape' && !isStreaming) onClose();
   259→    };
   260→    window.addEventListener('keydown', onKey);
   261→    return () => window.removeEventListener('keydown', onKey);
   262→  }, [open, isStreaming, onClose]);
   263→
   264→  // ─── Smart auto-scroll ─────────────────────────────────────────────────────
   265→  const handleScroll = useCallback(() => {
   266→    const el = scrollRef.current;
   267→    if (!el) return;
   268→    const distanceFromBottom = el.scrollHeight - el.scrollTop - el.clientHeight;
   269→    userPinnedUpRef.current = distanceFromBottom > 120;
   270→  }, []);
   271→
   272→  const scrollToBottom = useCallback((smooth = false) => {
   273→    const el = scrollRef.current;
   274→    if (!el) return;
   275→    el.scrollTo({ top: el.scrollHeight, behavior: smooth ? 'smooth' : 'auto' });
   276→  }, []);
   277→
   278→  useEffect(() => {
   279→    if (!userPinnedUpRef.current) {
   280→      requestAnimationFrame(() => scrollToBottom(false));
   281→    }
   282→  }, [messages, scrollToBottom]);
   283→
   284→  // ─── Conversation actions ──────────────────────────────────────────────────
   285→  const handleNewChat = useCallback(() => {
   286→    if (isStreaming) return;
   287→    const conv = newConversation();
   288→    setStore((prev) => {
   289→      const next = { conversations: [conv, ...prev.conversations], activeId: conv.id };
   290→      saveStore(next);
   291→      return next;
   292→    });
   293→    setActiveId(conv.id);
   294→    setMessages([]);
   295→    setInput('');
   296→    userPinnedUpRef.current = false;
   297→    setHistoryOpen(false);
   298→    requestAnimationFrame(() => inputRef.current?.focus());
   299→  }, [isStreaming]);
   300→
   301→  const handleSwitchConversation = useCallback(
   302→    (id: string) => {
   303→      if (isStreaming) return;
   304→      const conv = store.conversations.find((c) => c.id === id);
   305→      if (!conv) return;
   306→      setActiveId(id);
   307→      setMessages(conv.messages);
   308→      setStore((prev) => {
   309→        const next = { ...prev, activeId: id };
   310→        saveStore(next);
   311→        return next;
   312→      });
   313→      setHistoryOpen(false);
   314→      userPinnedUpRef.current = false;
   315→      requestAnimationFrame(() => scrollToBottom(false));
   316→    },
   317→    [isStreaming, store.conversations, scrollToBottom],
   318→  );
   319→
   320→  const handleDeleteConversation = useCallback(
   321→    (id: string) => {
   322→      if (isStreaming) return;
   323→      setStore((prev) => {
   324→        const filtered = prev.conversations.filter((c) => c.id !== id);
   325→        let nextActive = prev.activeId;
   326→        let nextMessages = messages;
   327→        if (prev.activeId === id) {
   328→          if (filtered.length > 0) {
   329→            nextActive = filtered[0].id;
   330→            nextMessages = filtered[0].messages;
   331→          } else {
   332→            const conv = newConversation();
   333→            filtered.unshift(conv);
   334→            nextActive = conv.id;
   335→            nextMessages = [];
   336→          }
   337→          setActiveId(nextActive);
   338→          setMessages(nextMessages);
   339→        }
   340→        const next = { conversations: filtered, activeId: nextActive };
   341→        saveStore(next);
   342→        return next;
   343→      });
   344→    },
   345→    [isStreaming, messages],
   346→  );
   347→
   348→  // ─── Send flow (preserved from previous implementation) ─────────────────────
   349→  const sendMessage = useCallback(
   350→    async (raw: string) => {
   351→      const text = raw.trim();
   352→      if (!text || isStreaming) return;
   353→
   354→      // ── Brand-question short-circuit ───────────────────────────────────────
   355→      const brand = detectBrandQuestion(text);
   356→      if (brand.matched && brand.answer) {
   357→        const userMsg: OracleMessage = {
   358→          id: cryptoId(),
   359→          role: 'user',
   360→          content: text,
   361→          language: detectLanguage(text),
   362→          createdAt: new Date().toISOString(),
   363→        };
   364→        const oracleMsg: OracleMessage = {
   365→          id: cryptoId(),
   366→          role: 'oracle',
   367→          content: brand.answer,
   368→          language: detectLanguage(text),
   369→          emotion: 'success',
   370→          createdAt: new Date().toISOString(),
   371→          streaming: false,
   372→          followUps: ['Who founded GSTPilot?', 'What can Oracle do?', 'GST kya hota hai?'],
   373→        };
   374→        setMessages((prev) => [...prev, userMsg, oracleMsg]);
   375→        setLastEmotion('success');
   376→        setActiveLanguage(detectLanguage(text));
   377→        setInput('');
   378→        if (inputRef.current) inputRef.current.style.height = 'auto';
   379→        userPinnedUpRef.current = false;
   380→        requestAnimationFrame(() => scrollToBottom(true));
   381→        return;
   382→      }
   383→
   384→      const userMsg: OracleMessage = {
   385→        id: cryptoId(),
   386→        role: 'user',
   387→        content: text,
   388→        language: detectLanguage(text),
   389→        createdAt: new Date().toISOString(),
   390→      };
   391→      const oracleId = cryptoId();
   392→      const oraclePlaceholder: OracleMessage = {
   393→        id: oracleId,
   394→        role: 'oracle',
   395→        content: '',
   396→        language: userMsg.language,
   397→        createdAt: new Date().toISOString(),
   398→        streaming: true,
   399→      };
   400→      setMessages((prev) => [...prev, userMsg, oraclePlaceholder]);
   401→      streamingIdRef.current = oracleId;
   402→      setIsStreaming(true);
   403→      setActiveLanguage(userMsg.language);
   404→      setInput('');
   405→      userPinnedUpRef.current = false;
   406→      if (inputRef.current) inputRef.current.style.height = 'auto';
   407→
   408→      const history: OracleChatRequest['messages'] = [
   409→        ...messages
   410→          .filter((m) => m.content.trim().length > 0)
   411→          .map((m) => ({ role: m.role, content: m.content })),
   412→        { role: 'user', content: text },
   413→      ];
   414→
   415→      // ── Fetch live business data (cached 30s, invalidated on connect) ──────
   416→      const liveData = await fetchLiveData();
   417→
   418→      const payload: OracleChatRequest = {
   419→        messages: history,
   420→        memory: {
   421→          userName,
   422→          firmName,
   423→          gstin,
   424→          preferredLanguage: activeLanguage,
   425→          recentTopics: messages
   426→            .filter((m) => m.role === 'user')
   427→            .slice(-4)
   428→            .map((m) => m.content.slice(0, 60)),
   429→        },
   430→        context: { liveData },
   431→      };
   432→
   433→      const controller = new AbortController();
   434→      abortRef.current = controller;
   435→
   436→      try {
   437→        const res = await fetch('/api/oracle/chat', {
   438→          method: 'POST',
   439→          headers: { 'Content-Type': 'application/json' },
   440→          body: JSON.stringify(payload),
   441→          signal: controller.signal,
   442→        });
   443→        if (!res.ok || !res.body) {
   444→          throw new Error(`Request failed (${res.status})`);
   445→        }
   446→
   447→        const reader = res.body.getReader();
   448→        const decoder = new TextDecoder();
   449→        let buffer = '';
   450→        let acc = '';
   451→
   452→        while (true) {
   453→          const { done, value } = await reader.read();
   454→          if (done) break;
   455→          buffer += decoder.decode(value, { stream: true });
   456→
   457→          const lines = buffer.split('\n');
   458→          buffer = lines.pop() ?? '';
   459→
   460→          for (const rawLine of lines) {
   461→            const line = rawLine.trim();
   462→            if (!line || !line.startsWith('data:')) continue;
   463→            const data = line.slice(5).trim();
   464→            if (!data || data === '[DONE]') continue;
   465→            let chunk: OracleStreamChunk;
   466→            try {
   467→              chunk = JSON.parse(data) as OracleStreamChunk;
   468→            } catch {
   469→              continue;
   470→            }
   471→            if (chunk.language) setActiveLanguage(chunk.language);
   472→            if (chunk.token) {
   473→              acc += chunk.token;
   474→              setMessages((prev) =>
   475→                prev.map((m) => (m.id === oracleId ? { ...m, content: acc } : m)),
   476→              );
   477→            }
   478→            if (chunk.done) {
   479→              setMessages((prev) =>
   480→                prev.map((m) =>
   481→                  m.id === oracleId
   482→                    ? {
   483→                        ...m,
   484→                        content: acc,
   485→                        streaming: false,
   486→                        emotion: detectEmotion(acc),
   487→                        followUps: buildFollowUps(acc),
   488→                      }
   489→                    : m,
   490→                ),
   491→              );
   492→              setLastEmotion(detectEmotion(acc));
   493→            }
   494→            if (chunk.error) {
   495→              setMessages((prev) =>
   496→                prev.map((m) =>
   497→                  m.id === oracleId
   498→                    ? {
   499→                        ...m,
   500→                        content:
   501→                          acc ||
   502→                          "I ran into a temporary issue reaching my reasoning service. Please try that again — your conversation is safe.",
   503→                        streaming: false,
   504→                        emotion: 'warning',
   505→                      }
   506→                    : m,
   507→                ),
   508→              );
   509→              setLastEmotion('warning');
   510→            }
   511→          }
   512→        }
   513→
   514→        setMessages((prev) =>
   515→          prev.map((m) =>
   516→            m.id === oracleId && m.streaming
   517→              ? {
   518→                  ...m,
   519→                  streaming: false,
   520→                  emotion: detectEmotion(m.content),
   521→                  followUps: buildFollowUps(m.content),
   522→                }
   523→              : m,
   524→          ),
   525→        );
   526→      } catch (err) {
   527→        const aborted = err instanceof DOMException && err.name === 'AbortError';
   528→        setMessages((prev) =>
   529→          prev.map((m) =>
   530→            m.id === oracleId
   531→              ? {
   532→                  ...m,
   533→                  streaming: false,
   534→                  content:
   535→                    m.content ||
   536→                    (aborted
   537→                      ? 'Stopped.'
   538→                      : "I had trouble reaching my reasoning service. Please try again in a moment."),
   539→                  emotion: aborted ? m.emotion : 'warning',
   540→                }
   541→              : m,
   542→          ),
   543→        );
   544→        if (!aborted) setLastEmotion('warning');
   545→      } finally {
   546→        setIsStreaming(false);
   547→        streamingIdRef.current = null;
   548→        abortRef.current = null;
   549→        requestAnimationFrame(() => inputRef.current?.focus());
   550→      }
   551→    },
   552→    [isStreaming, messages, userName, firmName, gstin, activeLanguage, scrollToBottom],
   553→  );
   554→
   555→  // ─── Stop streaming ────────────────────────────────────────────────────────
   556→  const handleStop = useCallback(() => {
   557→    abortRef.current?.abort();
   558→  }, []);
   559→
   560→  // ─── Input handling ────────────────────────────────────────────────────────
   561→  const handleInputChange = useCallback((e: React.ChangeEvent<HTMLTextAreaElement>) => {
   562→    setInput(e.target.value);
   563→    const el = e.target;
   564→    el.style.height = 'auto';
   565→    el.style.height = `${Math.min(el.scrollHeight, 160)}px`;
   566→  }, []);
   567→
   568→  const handleKeyDown = useCallback(
   569→    (e: KeyboardEvent<HTMLTextAreaElement>) => {
   570→      if (e.key === 'Enter' && !e.shiftKey) {
   571→        e.preventDefault();
   572→        sendMessage(input);
   573→      }
   574→    },
   575→    [input, sendMessage],
   576→  );
   577→
   578→  // ─── Focus input on open ───────────────────────────────────────────────────
   579→  useEffect(() => {
   580→    if (open) {
   581→      requestAnimationFrame(() => inputRef.current?.focus());
   582→    }

  1246→interface ConversationGroups {
  1247→  today: OracleConversation[];
  1248→  yesterday: OracleConversation[];

  1250→  older: OracleConversation[];

  1253→function groupConversations(convos: OracleConversation[]): ConversationGroups {
  1254→  const now = new Date();
  1255→  const startOfToday = new Date(now.getFullYear(), now.getMonth(), now.getDate()).getTime();
  1256→  const startOfYesterday = startOfToday - 86_400_000;
  1257→  const startOf7Days = startOfToday - 7 * 86_400_000;
  1258→
  1259→  const today: OracleConversation[] = [];
  1260→  const yesterday: OracleConversation[] = [];
  1261→  const prev7: OracleConversation[] = [];
  1262→  const older: OracleConversation[] = [];
  1263→
  1264→  const sorted = [...convos].sort(
  1265→    (a, b) => new Date(b.updatedAt).getTime() - new Date(a.updatedAt).getTime(),
  1266→  );
  1267→
  1268→  for (const c of sorted) {
  1269→    const t = new Date(c.updatedAt).getTime();
  1270→    if (isNaN(t)) {
  1271→      older.push(c);
  1272→    } else if (t >= startOfToday) {
  1273→      today.push(c);
  1274→    } else if (t >= startOfYesterday) {
  1275→      yesterday.push(c);
  1276→    } else if (t >= startOf7Days) {
  1277→      prev7.push(c);
  1278→    } else {
  1279→      older.push(c);
  1280→    }
  1281→  }
  1282→
  1283→  return { today, yesterday, prev7, older };
  1284→}

  1286→/** Generate contextual follow-up chips from the response content. */
  1287→function buildFollowUps(content: string): string[] | undefined {
  1288→  if (!content || content.length < 30) return undefined;
  1289→  const lower = content.toLowerCase();
  1290→
  1291→  const pool: string[] = [];
  1292→  if (/(gstr-3b|3b)/.test(lower)) pool.push('GSTR-3B की last date क्या है?', 'Late fee कितनी लगेगी?');
  1293→  if (/(gstr-1|gstr 1)/.test(lower)) pool.push('GSTR-1 कैसे file करें?', 'B2B और B2C में अंतर?');
  1294→  if (/(itc|input tax credit)/.test(lower)) pool.push('ITC claim कैसे करें?', 'Blocked ITC के rules?');
  1295→  if (/(late fee|penalty|overdue)/.test(lower)) pool.push('Late fee waiver मिल सकती है?', 'How to avoid this next time?');
  1296→  if (/(cash flow|collections)/.test(lower)) pool.push('Receivables कैसे recover करें?', 'Working capital optimize करें');
  1297→  if (/(reverse charge|rcm)/.test(lower)) pool.push('RCM किन पर लागू है?', 'How to report RCM in GSTR-3B?');
  1298→  if (/(refund)/.test(lower)) pool.push('Refund process क्या है?', 'Refund timeline कितनी है?');
  1299→
  1300→  pool.push('GST kya hota hai?', 'Explain ITC rules', 'How can Oracle help me daily?');
  1301→
  1302→  const seen = new Set<string>();
  1303→  const picks: string[] = [];
  1304→  for (const p of pool) {
  1305→    if (seen.has(p)) continue;
  1306→    seen.add(p);
  1307→    picks.push(p);
  1308→    if (picks.length >= 3) break;
  1309→  }
  1310→  return picks;
  1311→}
