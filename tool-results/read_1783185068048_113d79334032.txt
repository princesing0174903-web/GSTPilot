    43→function formatINR(amount: number): string {
    44→  if (!isFinite(amount) || isNaN(amount)) return '₹0';
    45→  const abs = Math.abs(amount);
    46→  if (abs >= 10000000) return `₹${(amount / 10000000).toFixed(2)} Cr`;
    47→  if (abs >= 100000) return `₹${(amount / 100000).toFixed(2)} L`;
    48→  if (abs >= 1000) return `₹${(amount / 1000).toFixed(1)}K`;
    49→  return '₹' + Math.round(amount).toLocaleString('en-IN');
    50→}
    51→
    52→function formatINRFull(amount: number): string {
    53→  if (!isFinite(amount) || isNaN(amount)) return '₹0';
    54→  return '₹' + Math.round(amount).toLocaleString('en-IN');
    55→}

    62→function timeAgo(iso: string): string {
    63→  const diff = Date.now() - new Date(iso).getTime();
    64→  const s = Math.floor(diff / 1000);
    65→  if (s < 60) return 'just now';
    66→  const m = Math.floor(s / 60);
    67→  if (m < 60) return `${m}m ago`;
    68→  const h = Math.floor(m / 60);
    69→  if (h < 24) return `${h}h ago`;
    70→  return `${Math.floor(h / 24)}d ago`;
    71→}

   100→function useAnimatedNumber(target: number, duration = 1200) {
   101→  const [value, setValue] = useState(0);
   102→  useEffect(() => {
   103→    let raf = 0;
   104→    const start = performance.now();
   105→    const initial = 0;
   106→    const tick = (t: number) => {
   107→      const progress = Math.min((t - start) / duration, 1);
   108→      const eased = 1 - Math.pow(1 - progress, 3);
   109→      setValue(initial + (target - initial) * eased);
   110→      if (progress < 1) raf = requestAnimationFrame(tick);
   111→    };
   112→    raf = requestAnimationFrame(tick);
   113→    return () => cancelAnimationFrame(raf);
   114→  }, [target, duration]);
   115→  return value;
   116→}

   120→function Sparkline({ data, color = '#10b981', width = 100, height = 32 }: { data: number[]; color?: string; width?: number; height?: number }) {
   121→  if (data.length < 2) return <div style={{ width, height }} />;
   122→  const min = Math.min(...data);
   123→  const max = Math.max(...data);
   124→  const range = max - min || 1;
   125→  const pad = 2;
   126→  const points = data.map((v, i) => {
   127→    const x = pad + (i / (data.length - 1)) * (width - pad * 2);
   128→    const y = height - pad - ((v - min) / range) * (height - pad * 2);
   129→    return `${x},${y}`;
   130→  }).join(' ');
   131→  const area = `${pad},${height - pad} ${points} ${width - pad},${height - pad}`;
   132→  const id = `spark-${color.replace('#', '')}-${Math.round(Math.random() * 1e6)}`;
   133→  return (
   134→    <svg width={width} height={height} className="shrink-0">
   135→      <defs>
   136→        <linearGradient id={id} x1="0" y1="0" x2="0" y2="1">
   137→          <stop offset="0%" stopColor={color} stopOpacity={0.3} />
   138→          <stop offset="100%" stopColor={color} stopOpacity={0} />
   139→        </linearGradient>
   140→      </defs>
   141→      <polygon points={area} fill={`url(#${id})`} />
   142→      <polyline points={points} fill="none" stroke={color} strokeWidth={1.5} strokeLinecap="round" strokeLinejoin="round" />
   143→    </svg>
   144→  );
   145→}

   394→function HealthGauge({ score, size = 160 }: { score: number; size?: number }) {
   395→  const stroke = 12;
   396→  const r = (size - stroke) / 2;
   397→  const c = 2 * Math.PI * r;
   398→  const offset = c - (score / 100) * c;
   399→  const tier = score >= 80 ? 'Excellent' : score >= 65 ? 'Healthy' : score >= 50 ? 'Needs Attention' : 'At Risk';
   400→  const color = score >= 80 ? '#10b981' : score >= 65 ? '#06b6d4' : score >= 50 ? '#f59e0b' : '#ef4444';
   401→  return (
   402→    <div className="relative" style={{ width: size, height: size }}>
   403→      <svg width={size} height={size} className="-rotate-90">
   404→        <defs>
   405→          <linearGradient id="cfoHealthGrad" x1="0" y1="0" x2="1" y2="1">
   406→            <stop offset="0%" stopColor={color} />
   407→            <stop offset="100%" stopColor={color} stopOpacity={0.6} />
   408→          </linearGradient>
   409→        </defs>
   410→        <circle cx={size / 2} cy={size / 2} r={r} fill="none" stroke="rgba(255,255,255,0.05)" strokeWidth={stroke} />
   411→        <motion.circle
   412→          cx={size / 2}
   413→          cy={size / 2}
   414→          r={r}
   415→          fill="none"
   416→          stroke="url(#cfoHealthGrad)"
   417→          strokeWidth={stroke}
   418→          strokeLinecap="round"
   419→          strokeDasharray={c}
   420→          initial={{ strokeDashoffset: c }}
   421→          animate={{ strokeDashoffset: offset }}
   422→          transition={{ duration: 1.2, ease: 'easeOut' }}
   423→        />
   424→      </svg>
   425→      <div className="absolute inset-0 flex flex-col items-center justify-center">
   426→        <motion.span
   427→          initial={{ opacity: 0, scale: 0.7 }}
   428→          animate={{ opacity: 1, scale: 1 }}
   429→          transition={{ delay: 0.4, duration: 0.4 }}
   430→          className="text-4xl font-bold"
   431→          style={{ color }}
   432→        >
   433→          {score}
   434→        </motion.span>
   435→        <span className="text-[10px] uppercase tracking-wider text-muted-foreground">/ 100</span>
   436→        <span className="mt-1 text-xs font-medium" style={{ color }}>{tier}</span>
   437→      </div>
   438→    </div>
   439→  );
   440→}
