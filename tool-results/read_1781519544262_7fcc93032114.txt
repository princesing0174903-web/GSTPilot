   203→
   204→export function OnboardingFlow({
   205→  onComplete,
   206→  onSkip,
   207→  userEmail = '',
   208→  userName = '',
   209→}: OnboardingFlowProps) {
   210→  const [currentStep, setCurrentStep] = useState(0); // 0=welcome, 1-4=steps, 5=final
   211→  const [direction, setDirection] = useState(1);
   212→
   213→  const [formData, setFormData] = useState<OnboardingData>({
   214→    fullName: userName,
   215→    email: userEmail,
   216→    phone: '',
   217→    ageGroup: '',
   218→    profession: '',
   219→    experience: '',
   220→    firmName: '',
   221→    organizationType: '',
   222→    gstin: '',
   223→    state: '',
   224→    stateCode: '',
   225→    icaiMembershipNo: '',
   226→    officeAddress: '',
   227→    clientCount: '',
   228→    monthlyReturns: '',
   229→    gstServices: [],
   230→    painPoints: [],
   231→    referralSource: '',
   232→    trialReasons: [],
   233→    wantsUpdates: true,
   234→  });
   235→
   236→  const [errors, setErrors] = useState<Record<string, string>>({});
   237→
   238→  // ─── Helpers ──────────────────────────────────────────────────────────────
   239→
   240→  const updateField = useCallback(
   241→    <K extends keyof OnboardingData>(field: K, value: OnboardingData[K]) => {
   242→      setFormData((prev) => ({ ...prev, [field]: value }));
   243→      // Clear error on change
   244→      setErrors((prev) => {
   245→        const next = { ...prev };
   246→        delete next[field];
   247→        return next;
   248→      });
   249→    },
   250→    []
   251→  );
   252→
   253→  const toggleArrayItem = useCallback(
   254→    (field: 'gstServices' | 'painPoints' | 'trialReasons', item: string) => {
   255→      setFormData((prev) => {
   256→        const arr = prev[field] as string[];
   257→        const next = arr.includes(item)
   258→          ? arr.filter((i) => i !== item)
   259→          : [...arr, item];
   260→        return { ...prev, [field]: next };
   261→      });
   262→    },
   263→    []
   264→  );
   265→
   266→  const goNext = useCallback(() => {
   267→    setDirection(1);
   268→    setCurrentStep((prev) => Math.min(prev + 1, 5));
   269→  }, []);
   270→
   271→  const goBack = useCallback(() => {
   272→    setDirection(-1);
   273→    setCurrentStep((prev) => Math.max(prev - 1, 0));
   274→  }, []);
   275→
   276→  // ─── Validation ───────────────────────────────────────────────────────────
   277→
   278→  const validateStep = useCallback(
   279→    (step: number): boolean => {
   280→      const newErrors: Record<string, string> = {};
   281→
   282→      if (step === 1) {
   283→        if (!formData.fullName.trim()) newErrors.fullName = 'Full name is required';
   284→        if (!formData.email.trim()) newErrors.email = 'Email is required';
   285→        else if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(formData.email))
   286→          newErrors.email = 'Enter a valid email';
   287→        if (!formData.phone.trim()) newErrors.phone = 'Mobile number is required';
   288→        else if (!/^\d{10}$/.test(formData.phone.replace(/\s/g, '')))
   289→          newErrors.phone = 'Enter a valid 10-digit mobile number';
   290→      }
   291→
   292→      if (step === 2) {
   293→        if (!formData.firmName.trim()) newErrors.firmName = 'Firm name is required';
   294→      }
   295→
   296→      setErrors(newErrors);
   297→      return Object.keys(newErrors).length === 0;
   298→    },
   299→    [formData]
   300→  );
   301→
   302→  const handleNext = useCallback(() => {
   303→    if (validateStep(currentStep)) {
   304→      goNext();
   305→    }
   306→  }, [currentStep, validateStep, goNext]);
   307→
   308→  // ─── GSTIN Validation ─────────────────────────────────────────────────────
   309→
   310→  const getGstinStatus = useCallback(() => {
   311→    const val = formData.gstin.trim();
   312→    if (!val) return null;
   313→    return GSTIN_REGEX.test(val) ? 'valid' : 'invalid';
   314→  }, [formData.gstin]);
   315→
   316→  // ─── State Code Lookup ────────────────────────────────────────────────────
   317→
   318→  const handleStateChange = useCallback(
   319→    (stateName: string) => {
   320→      const stateObj = INDIAN_STATES.find((s) => s.name === stateName);
   321→      setFormData((prev) => ({
   322→        ...prev,
   323→        state: stateName,
   324→        stateCode: stateObj?.code ?? '',
   325→      }));
   326→      setErrors((prev) => {
   327→        const next = { ...prev };
   328→        delete next.state;
   329→        return next;
   330→      });
   331→    },
   332→    []
   333→  );
   334→
   335→  // ─── Progress ─────────────────────────────────────────────────────────────
   336→
   337→  const totalSteps = 5;
   338→  const progressPercent =
   339→    currentStep === 0 ? 0 : (currentStep / totalSteps) * 100;
   340→
   341→  // ─── Step Icons ───────────────────────────────────────────────────────────
   342→
   343→  const stepIcons = [Zap, User, Building2, Briefcase, Target];
   344→
   345→  // ─── Render Step Content ──────────────────────────────────────────────────
   346→
   347→  const renderStep = () => {
   348→    // Step 0 — Welcome
   349→    if (currentStep === 0) {
   350→      return (
   351→        <motion.div
   352→          key="step-0"
   353→          custom={direction}
   354→          variants={slideVariants}
   355→          initial="enter"
   356→          animate="center"
   357→          exit="exit"
   358→          transition={stepTransition}
   359→          className="flex flex-col items-center justify-center text-center py-8"
   360→        >
   361→          <motion.div
   362→            initial={{ scale: 0.5, opacity: 0 }}
   363→            animate={{ scale: 1, opacity: 1 }}
   364→            transition={{ type: 'spring', stiffness: 200, damping: 15, delay: 0.1 }}
   365→            className="mb-8"
   366→          >
   367→            <div className="w-20 h-20 rounded-2xl bg-gradient-to-br from-emerald-400 to-teal-600 flex items-center justify-center shadow-lg shadow-emerald-200">
   368→              <Zap className="size-10 text-white" />
   369→            </div>
   370→          </motion.div>
   371→          <motion.h1
   372→            initial={{ y: 20, opacity: 0 }}
   373→            animate={{ y: 0, opacity: 1 }}
   374→            transition={{ delay: 0.2 }}
   375→            className="text-2xl font-bold tracking-tight text-foreground"
   376→          >
   377→            Welcome to GSTPilot
   378→          </motion.h1>
   379→          <motion.p
   380→            initial={{ y: 20, opacity: 0 }}
   381→            animate={{ y: 0, opacity: 1 }}
   382→            transition={{ delay: 0.3 }}
   383→            className="text-muted-foreground mt-2 max-w-sm"
   384→          >
   385→            Let&apos;s set up your workspace in less than 2 minutes.
   386→          </motion.p>
   387→          <motion.div
   388→            initial={{ y: 20, opacity: 0 }}
   389→            animate={{ y: 0, opacity: 1 }}
   390→            transition={{ delay: 0.4 }}
   391→            className="mt-8"
   392→          >
   393→            <Button
   394→              onClick={goNext}
   395→              size="lg"
   396→              className="bg-emerald-600 hover:bg-emerald-700 text-white gap-2 px-8"
   397→            >
   398→              Get Started
   399→              <ArrowRight className="size-4" />
   400→            </Button>
   401→          </motion.div>
   402→        </motion.div>
   403→      );
   404→    }
   405→
   406→    // Step 1 — About You
   407→    if (currentStep === 1) {
   408→      return (
   409→        <motion.div
   410→          key="step-1"
   411→          custom={direction}
   412→          variants={slideVariants}
   413→          initial="enter"
   414→          animate="center"
   415→          exit="exit"
   416→          transition={stepTransition}
   417→          className="py-4"
   418→        >
   419→          <div className="flex items-center gap-3 mb-6">
   420→            <div className="w-10 h-10 rounded-xl bg-emerald-100 flex items-center justify-center">
   421→              <User className="size-5 text-emerald-600" />
   422→            </div>
   423→            <div>
   424→              <h2 className="text-2xl font-bold">About You</h2>
   425→              <p className="text-sm text-muted-foreground">
   426→                Tell us about yourself so we can personalize your experience
   427→              </p>
   428→            </div>
   429→          </div>
   430→
   431→          <div className="space-y-5">
   432→            {/* Full Name */}
   433→            <div className="space-y-2">
   434→              <Label htmlFor="fullName">
   435→                Full Name <span className="text-destructive">*</span>
   436→              </Label>
   437→              <Input
   438→                id="fullName"
   439→                placeholder="Rajesh Kumar Sharma"
   440→                value={formData.fullName}
   441→                onChange={(e) => updateField('fullName', e.target.value)}
   442→                aria-invalid={!!errors.fullName}
   443→                className={errors.fullName ? 'border-destructive' : ''}
   444→              />
   445→              {errors.fullName && (
   446→                <p className="text-xs text-destructive">{errors.fullName}</p>
   447→              )}
   448→            </div>
   449→
   450→            {/* Work Email */}
   451→            <div className="space-y-2">
   452→              <Label htmlFor="email">
   453→                Work Email <span className="text-destructive">*</span>
   454→              </Label>
   455→              <Input
   456→                id="email"
   457→                type="email"
   458→                placeholder="rajesh@yourfirm.com"
   459→                value={formData.email}
   460→                onChange={(e) => updateField('email', e.target.value)}
   461→                aria-invalid={!!errors.email}
   462→                className={errors.email ? 'border-destructive' : ''}
   463→              />
   464→              {errors.email && (
   465→                <p className="text-xs text-destructive">{errors.email}</p>
   466→              )}
   467→            </div>
   468→
   469→            {/* Mobile Number */}
   470→            <div className="space-y-2">
   471→              <Label htmlFor="phone">
   472→                Mobile Number <span className="text-destructive">*</span>
   473→              </Label>
   474→              <div className="flex">
   475→                <div className="flex items-center rounded-l-md border border-r-0 bg-muted px-3 text-sm text-muted-foreground h-9">
   476→                  +91
   477→                </div>
   478→                <Input
   479→                  id="phone"
   480→                  type="tel"
   481→                  placeholder="9876543210"
   482→                  value={formData.phone}
   483→                  onChange={(e) =>
   484→                    updateField('phone', e.target.value.replace(/[^\d]/g, '').slice(0, 10))
   485→                  }
   486→                  aria-invalid={!!errors.phone}
   487→                  className={`rounded-l-none ${errors.phone ? 'border-destructive' : ''}`}
   488→                />
   489→              </div>
   490→              {errors.phone && (
   491→                <p className="text-xs text-destructive">{errors.phone}</p>
   492→              )}
   493→            </div>
   494→
   495→            {/* Age Group */}
   496→            <div className="space-y-2">
   497→              <Label>Age Group</Label>
   498→              <Select
   499→                value={formData.ageGroup}
   500→                onValueChange={(v) => updateField('ageGroup', v)}
   501→              >
   502→                <SelectTrigger className="w-full">
   503→                  <SelectValue placeholder="Select age group" />
   504→                </SelectTrigger>
   505→                <SelectContent>
   506→                  {AGE_GROUPS.map((g) => (
   507→                    <SelectItem key={g} value={g}>
   508→                      {g}
   509→                    </SelectItem>
   510→                  ))}
   511→                </SelectContent>
   512→              </Select>
   513→            </div>
   514→
   515→            {/* Profession */}
   516→            <div className="space-y-2">
   517→              <Label>Profession</Label>
   518→              <Select
   519→                value={formData.profession}
   520→                onValueChange={(v) => updateField('profession', v)}
   521→              >
   522→                <SelectTrigger className="w-full">
   523→                  <SelectValue placeholder="Select your profession" />
   524→                </SelectTrigger>
   525→                <SelectContent>
   526→                  {PROFESSIONS.map((p) => (
   527→                    <SelectItem key={p} value={p}>
   528→                      {p}
   529→                    </SelectItem>
   530→                  ))}
   531→                </SelectContent>
   532→              </Select>
   533→            </div>
   534→
   535→            {/* Years of Experience */}
   536→            <div className="space-y-2">
   537→              <Label>Years of Experience</Label>
   538→              <Select
   539→                value={formData.experience}
   540→                onValueChange={(v) => updateField('experience', v)}
   541→              >
   542→                <SelectTrigger className="w-full">
   543→                  <SelectValue placeholder="Select experience" />
   544→                </SelectTrigger>
   545→                <SelectContent>
   546→                  {EXPERIENCE_LEVELS.map((e) => (
   547→                    <SelectItem key={e} value={e}>
   548→                      {e}
   549→                    </SelectItem>
   550→                  ))}
   551→                </SelectContent>
   552→              </Select>
   553→            </div>
   554→          </div>
   555→        </motion.div>
   556→      );
   557→    }
   558→
   559→    // Step 2 — Firm Information
   560→    if (currentStep === 2) {
   561→      const gstinStatus = getGstinStatus();
   562→      return (
   563→        <motion.div
   564→          key="step-2"
   565→          custom={direction}
   566→          variants={slideVariants}
   567→          initial="enter"
   568→          animate="center"
   569→          exit="exit"
   570→          transition={stepTransition}
   571→          className="py-4"
   572→        >
   573→          <div className="flex items-center gap-3 mb-6">
   574→            <div className="w-10 h-10 rounded-xl bg-emerald-100 flex items-center justify-center">
   575→              <Building2 className="size-5 text-emerald-600" />
   576→            </div>
   577→            <div>
   578→              <h2 className="text-2xl font-bold">Firm Information</h2>
   579→              <p className="text-sm text-muted-foreground">
   580→                Details about your CA firm or practice
   581→              </p>
   582→            </div>
   583→          </div>
   584→
   585→          <div className="space-y-5">
   586→            {/* Firm Name */}
   587→            <div className="space-y-2">
   588→              <Label htmlFor="firmName">
   589→                Firm Name <span className="text-destructive">*</span>
   590→              </Label>
   591→              <Input
   592→                id="firmName"
   593→                placeholder="Sharma & Associates"
   594→                value={formData.firmName}
   595→                onChange={(e) => updateField('firmName', e.target.value)}
   596→                aria-invalid={!!errors.firmName}
   597→                className={errors.firmName ? 'border-destructive' : ''}
   598→              />
   599→              {errors.firmName && (
   600→                <p className="text-xs text-destructive">{errors.firmName}</p>
   601→              )}
   602→            </div>
   603→
   604→            {/* Type of Organization */}
   605→            <div className="space-y-2">
   606→              <Label>Type of Organization</Label>
   607→              <Select
   608→                value={formData.organizationType}
   609→                onValueChange={(v) => updateField('organizationType', v)}
   610→              >
   611→                <SelectTrigger className="w-full">
   612→                  <SelectValue placeholder="Select organization type" />
   613→                </SelectTrigger>
   614→                <SelectContent>
   615→                  {ORGANIZATION_TYPES.map((t) => (
   616→                    <SelectItem key={t} value={t}>
   617→                      {t}
   618→                    </SelectItem>
   619→                  ))}
   620→                </SelectContent>
   621→              </Select>
   622→            </div>
   623→
   624→            {/* GSTIN */}
   625→            <div className="space-y-2">
   626→              <Label htmlFor="gstin">GSTIN</Label>
   627→              <div className="relative">
   628→                <Input
   629→                  id="gstin"
   630→                  placeholder="22AAAAA0000A1Z5"
   631→                  value={formData.gstin}
   632→                  onChange={(e) =>
   633→                    updateField(
   634→                      'gstin',
   635→                      e.target.value.toUpperCase().slice(0, 15)
   636→                    )
   637→                  }
   638→                  className={`pr-10 ${
   639→                    gstinStatus === 'invalid'

   642→                      ? 'border-emerald-400'
