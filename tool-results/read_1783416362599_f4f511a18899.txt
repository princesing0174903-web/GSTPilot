    37→// ─── Helpers ───────────────────────────────────────────────────────────────────
    38→export const fmt = (n: number): string => {
    39→  if (Math.abs(n) >= 1_000_000_000) return `$${(n / 1_000_000_000).toFixed(2)}B`;
    40→  if (Math.abs(n) >= 1_000_000)     return `$${(n / 1_000_000).toFixed(2)}M`;
    41→  if (Math.abs(n) >= 1_000)         return `$${(n / 1_000).toFixed(1)}K`;
    42→  return `$${n.toFixed(0)}`;
    43→};
    44→export const fmtN = (n: number): string => {
    45→  if (Math.abs(n) >= 1_000_000_000) return `${(n / 1_000_000_000).toFixed(2)}B`;
    46→  if (Math.abs(n) >= 1_000_000)     return `${(n / 1_000_000).toFixed(2)}M`;
    47→  if (Math.abs(n) >= 1_000)         return `${(n / 1_000).toFixed(1)}K`;
    48→  return `${n.toFixed(0)}`;
    49→};
    50→export const fmtPct = (n: number): string => `${n > 0 ? '+' : ''}${n.toFixed(1)}%`;
