    89→function fmtCompactUSD(n: number): string {
    90→  const abs = Math.abs(n);
    91→  if (abs >= 1_000_000) return `$${(n / 1_000_000).toFixed(2)}M`;
    92→  if (abs >= 1_000) return `$${(n / 1_000).toFixed(1)}K`;
    93→  return `$${Math.round(n).toLocaleString('en-US')}`;
    94→}
    95→
    96→function fmtCurrencyValue(amount: number, code: string): string {
    97→  const sym = code === 'USD' ? '$' : code === 'EUR' ? '€' : code === 'GBP' ? '£'
    98→    : code === 'AED' ? 'AED ' : code === 'JPY' ? '¥' : code === 'AUD' ? 'A$'
    99→    : code === 'SGD' ? 'S$' : code === 'INR' ? '₹' : `${code} `;
   100→  return `${sym}${amount.toLocaleString('en-US', { maximumFractionDigits: 0 })}`;
   101→}

   209→function PaymentLifecycleTracker({ payment }: { payment: CrossBorderPayment }) {
   210→  const stages = PAYMENT_LIFECYCLE[payment.id];
   211→  // Fallback: synthesize generic lifecycle if no specific data
   212→  const fallback: PaymentStage[] = payment.status === 'completed'
   213→    ? [
   214→        { stage: 'Initiated', timestamp: payment.date, location: getCountry(payment.fromCountry).name, status: 'completed', detail: `Payment initiated via ${payment.method}` },
   215→        { stage: 'Compliance Check', timestamp: payment.date, location: 'Regulatory', status: 'completed', detail: 'Compliance verified' },
   216→        { stage: 'FX Conversion', timestamp: payment.date, location: payment.method, status: 'completed', detail: `Converted @ ${payment.fxRate.toFixed(2)}` },
   217→        { stage: 'Beneficiary Credit', timestamp: payment.date, location: getCountry(payment.toCountry).name, status: 'completed', detail: 'Credited to beneficiary' },
   218→      ]
   219→    : [
   220→        { stage: 'Initiated', timestamp: payment.date, location: getCountry(payment.fromCountry).name, status: 'completed', detail: `Payment initiated via ${payment.method}` },
   221→        { stage: 'Compliance Check', timestamp: '—', location: 'Regulatory', status: 'pending', detail: 'Awaiting compliance verification' },
   222→        { stage: 'FX Conversion', timestamp: '—', location: payment.method, status: 'pending', detail: 'Pending FX conversion' },
   223→        { stage: 'Beneficiary Credit', timestamp: '—', location: getCountry(payment.toCountry).name, status: 'pending', detail: 'Pending credit' },
   224→      ];
   225→  const stagesToRender = stages ?? fallback;
   226→  const completed = stagesToRender.filter((s) => s.status === 'completed').length;
   227→  const pct = (completed / stagesToRender.length) * 100;
   228→
   229→  return (
   230→    <div>
   231→      <div className="flex items-center justify-between mb-3">
   232→        <div className="flex items-center gap-2">
   233→          <Radio className="h-3.5 w-3.5 text-emerald-400" />
   234→          <p className="text-xs font-semibold text-white">Payment Journey</p>
   235→        </div>
   236→        <Badge variant="outline" className="border-emerald-500/30 bg-emerald-500/10 text-[10px] text-emerald-300">
   237→          {completed}/{stagesToRender.length} stages
   238→        </Badge>
   239→      </div>
   240→      {/* Progress bar */}
   241→      <div className="mb-4 h-1 w-full overflow-hidden rounded-full bg-white/[0.04]">
   242→        <motion.div
   243→          initial={{ width: 0 }}
   244→          animate={{ width: `${pct}%` }}
   245→          transition={{ duration: 0.5 }}
   246→          className="h-full rounded-full bg-emerald-500/70"
   247→        />
   248→      </div>
   249→      {/* Timeline */}
   250→      <div className="relative space-y-4 pl-2">
   251→        {stagesToRender.map((s, i) => (
   252→          <motion.div
   253→            key={`${s.stage}-${i}`}
   254→            initial={{ opacity: 0, x: -8 }}
   255→            animate={{ opacity: 1, x: 0 }}
   256→            transition={{ duration: 0.3, delay: i * 0.06 }}
   257→            className="relative flex gap-3"
   258→          >
   259→            {/* Vertical line */}
   260→            {i < stagesToRender.length - 1 && (
   261→              <div className="absolute left-[14px] top-7 bottom-[-16px] w-px bg-white/[0.08]" />
   262→            )}
   263→            <StageIcon status={s.status} />
   264→            <div className="flex-1 min-w-0 pb-1">
   265→              <div className="flex items-center justify-between gap-2">
   266→                <p className={cn(
   267→                  'text-xs font-semibold',
   268→                  s.status === 'completed' && 'text-white',
   269→                  s.status === 'pending' && 'text-amber-300',
   270→                  s.status === 'failed' && 'text-rose-300',
   271→                )}>
   272→                  {s.stage}
   273→                </p>

   280→            </div>
   281→          </motion.div>
   282→        ))}
   283→      </div>
   284→    </div>
   285→  );
