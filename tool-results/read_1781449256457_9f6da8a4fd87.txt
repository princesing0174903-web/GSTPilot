    28→// ─── API Fetch Helper ─────────────────────────────────────────────────────────
    29→
    30→async function apiFetch<T>(url: string, options?: RequestInit): Promise<T> {
    31→  const res = await fetch(url, {
    32→    headers: { 'Content-Type': 'application/json', ...options?.headers },
    33→    ...options,
    34→  });
    35→  if (!res.ok) {
    36→    const error = await res.json().catch(() => ({ error: 'Request failed' }));
    37→    throw new Error(error.error || `HTTP ${res.status}`);
    38→  }
    39→  return res.json();
    40→}

   566→  filters?: {

   569→    runId?: string;
   570→    workflowStatus?: string;
   571→    riskLevel?: string;

   576→    UseQueryOptions<ReconResultsResponse, Error>,

   590→    queryKey: queryKeys.reconciliation.results(filterRecord),

   593→      const qs = params.toString();

   595→        `/api/reconciliation${qs ? `?${qs}` : ''}`
