    73→// ─── Utility: Indian Number Formatting ──────────────────────────────────────────
    74→
    75→function formatINR(amount: number): string {
    76→  const str = amount.toFixed(0)
    77→  const parts = str.split('.')
    78→  let intPart = parts[0]
    79→  const decPart = parts[1]
    80→  const isNegative = intPart.startsWith('-')
    81→  if (isNegative) intPart = intPart.slice(1)
    82→
    83→  if (intPart.length > 3) {
    84→    const lastThree = intPart.slice(-3)
    85→    const rest = intPart.slice(0, -3)
    86→    intPart = rest.replace(/\B(?=(\d{2})+(?!\d))/g, ',') + ',' + lastThree
    87→  }
    88→
    89→  const formatted = isNegative ? '-' + intPart : intPart
    90→  return '₹' + formatted + (decPart ? '.' + decPart : '')
    91→}
    92→
    93→function formatDate(date: Date): string {
    94→  const d = new Date(date)
    95→  const day = String(d.getDate()).padStart(2, '0')
    96→  const month = String(d.getMonth() + 1).padStart(2, '0')
    97→  const year = d.getFullYear()
    98→  return `${day}/${month}/${year}`
    99→}

   101→// ─── Message Formatting ─────────────────────────────────────────────────────────
   102→
   103→function formatMessageContent(content: string) {
   104→  const lines = content.split('\n')
   105→  return lines.map((line, i) => {
   106→    // Process bold markers
   107→    let formatted = line.replace(/\*\*(.*?)\*\*/g, '<strong>$1</strong>')
   108→
   109→    // Process bullet points
   110→    if (formatted.startsWith('•') || formatted.startsWith('- ') || formatted.startsWith('* ')) {
   111→      const cleaned = formatted.replace(/^[•\-\*]\s*/, '')
   112→      return (
   113→        <div key={i} className="ml-3 flex items-start gap-1.5">
   114→          <span className="text-emerald-500 mt-0.5 shrink-0">•</span>
   115→          <span dangerouslySetInnerHTML={{ __html: cleaned }} />
   116→        </div>
   117→      )
   118→    }
   119→
   120→    // Numbered list
   121→    if (/^\d+\./.test(formatted)) {
   122→      return (
   123→        <div key={i} className="ml-2" dangerouslySetInnerHTML={{ __html: formatted }} />
   124→      )
   125→    }
   126→
   127→    // Empty line
   128→    if (formatted.trim() === '') {
   129→      return <div key={i} className="h-2" />
   130→    }
   131→
   132→    return (
   133→      <React.Fragment key={i}>
   134→        <span dangerouslySetInnerHTML={{ __html: formatted }} />
   135→        {i < lines.length - 1 && <br />}
   136→      </React.Fragment>
   137→    )
   138→  })
   139→}
