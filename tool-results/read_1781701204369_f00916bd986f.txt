    52→    52→function timeAgo(createdAt: unknown): string {
    53→    53→  if (!createdAt) return 'just now';
    54→    54→  let date: Date;
    55→    55→  if (createdAt instanceof Date) {
    56→    56→    date = createdAt;
    57→    57→  } else if (typeof createdAt === 'string') {
    58→    58→    date = new Date(createdAt);
    59→    59→  } else if (
    60→    60→    typeof createdAt === 'object' &&
    61→    61→    createdAt !== null &&
    62→    62→    'toDate' in createdAt &&
    63→    63→    typeof (createdAt as { toDate: () => Date }).toDate === 'function'
    64→    64→  ) {
    65→    65→    date = (createdAt as { toDate: () => Date }).toDate();
    66→    66→  } else {
    67→    67→    return 'just now';
    68→    68→  }
    69→    69→  if (isNaN(date.getTime())) return 'just now';
    70→    70→  const seconds = Math.floor((Date.now() - date.getTime()) / 1000);
    71→    71→  if (seconds < 60) return 'just now';
    72→    72→  const minutes = Math.floor(seconds / 60);
    73→    73→  if (minutes < 60) return `${minutes}m ago`;
    74→    74→  const hours = Math.floor(minutes / 60);
    75→    75→  if (hours < 24) return `${hours}h ago`;
    76→    76→  const days = Math.floor(hours / 24);
    77→    77→  if (days < 7) return `${days}d ago`;
    78→    78→  const weeks = Math.floor(days / 7);
    79→    79→  if (weeks < 5) return `${weeks}w ago`;
    80→    80→  const months = Math.floor(days / 30);
    81→    81→  if (months < 12) return `${months}mo ago`;
    82→    82→  const years = Math.floor(days / 365);
    83→    83→  return `${years}y ago`;
    84→    84→}

    86→    86→// ─── Business Score computation ───────────────────────────────────────────────
    87→    87→// Start at 100. Subtract weighted penalties for risk signals. Clamp 0–100.
    88→    88→// If there is no data at all, return null (UI shows welcoming state).
    89→    89→
    90→    90→function computeBusinessScore(opts: {
    91→    91→  totalClients: number;
    92→    92→  criticalIssues: number;
    93→    93→  warnings: number;
    94→    94→  pendingReturns: number;
    95→    95→  overdueReturns: number;
    96→    96→  averageHealthScore: number;
    97→    97→  matchPercentage: number;
    98→    98→}): number | null {
    99→    99→  const { totalClients, criticalIssues, warnings, pendingReturns, overdueReturns, averageHealthScore, matchPercentage } = opts;
   100→   100→  if (totalClients === 0 && pendingReturns === 0 && overdueReturns === 0) return null;
   101→   101→
   102→   102→  let score = 100;
   103→   103→  score -= criticalIssues * 6;      // critical issues are serious
   104→   104→  score -= warnings * 2;            // warnings are minor
   105→   105→  score -= overdueReturns * 8;      // overdue returns hurt most
   106→   106→  score -= pendingReturns * 2;      // pending returns are a mild drag
   107→   107→  // Health score gap: if avg health < 80, drag down
   108→   108→  if (averageHealthScore > 0) {
   109→   109→    score -= Math.max(0, 80 - averageHealthScore) * 0.6;
   110→   110→  }
   111→   111→  // Reconciliation gap: if match % < 100, mild drag
   112→   112→  score -= Math.max(0, 100 - matchPercentage) * 0.3;
   113→   113→  return Math.max(0, Math.min(100, Math.round(score)));
   114→   114→}

   116→   116→function scoreTier(score: number | null): { label: string; color: string } {
   117→   117→  if (score === null) return { label: '—', color: 'text-muted-foreground' };
   118→   118→  if (score >= 85) return { label: 'Excellent', color: 'text-emerald-400' };
   119→   119→  if (score >= 70) return { label: 'Healthy', color: 'text-cyan-400' };
   120→   120→  if (score >= 50) return { label: 'At Risk', color: 'text-amber-400' };
   121→   121→  return { label: 'Critical', color: 'text-red-400' };
   122→   122→}

   251→   251→function ScoreGauge({ score }: { score: number | null }) {
   252→   252→  const size = 220;
   253→   253→  const stroke = 14;
   254→   254→  const r = (size - stroke) / 2;
   255→   255→  const c = 2 * Math.PI * r;
   256→   256→  const pct = score === null ? 0 : score / 100;
   257→   257→  const offset = c * (1 - pct);
   258→   258→
   259→   259→  return (
   260→   260→    <div className="relative" style={{ width: size, height: size }}>
   261→   261→      <svg width={size} height={size} className="-rotate-90">
   262→   262→        <defs>
   263→   263→          <linearGradient id="missionScoreGradient" x1="0%" y1="0%" x2="100%" y2="100%">
   264→   264→            <stop offset="0%" stopColor="#10b981" />
   265→   265→            <stop offset="50%" stopColor="#06b6d4" />
   266→   266→            <stop offset="100%" stopColor="#3b82f6" />
   267→   267→          </linearGradient>
   268→   268→        </defs>
   269→   269→        {/* Track */}
   270→   270→        <circle
   271→   271→          cx={size / 2}
   272→   272→          cy={size / 2}
   273→   273→          r={r}
   274→   274→          fill="none"
   275→   275→          stroke="rgba(255,255,255,0.06)"
   276→   276→          strokeWidth={stroke}
   277→   277→        />
   278→   278→        {/* Progress arc */}
   279→   279→        {score !== null && (
   280→   280→          <motion.circle
   281→   281→            cx={size / 2}
   282→   282→            cy={size / 2}
   283→   283→            r={r}
   284→   284→            fill="none"
   285→   285→            stroke="url(#missionScoreGradient)"
   286→   286→            strokeWidth={stroke}
   287→   287→            strokeLinecap="round"
   288→   288→            strokeDasharray={c}
   289→   289→            initial={{ strokeDashoffset: c }}
   290→   290→            animate={{ strokeDashoffset: offset }}
   291→   291→            transition={{ duration: 1.2, ease: 'easeOut', delay: 0.2 }}
   292→   292→          />
   293→   293→        )}
   294→   294→      </svg>
   295→   295→      <div className="absolute inset-0 flex flex-col items-center justify-center">
   296→   296→        {score === null ? (
   297→   297→          <span className="text-4xl font-bold text-muted-foreground">—</span>
   298→   298→        ) : (
   299→   299→          <motion.span
   300→   300→            initial={{ opacity: 0, scale: 0.8 }}
   301→   301→            animate={{ opacity: 1, scale: 1 }}
   302→   302→            transition={{ duration: 0.5, delay: 0.6 }}
   303→   303→            className={`text-6xl font-bold tracking-tight ${scoreTier(score).color}`}
   304→   304→          >
   305→   305→            {score}
   306→   306→          </motion.span>
   307→   307→        )}
   308→   308→        <span className="text-xs text-muted-foreground tracking-wider uppercase mt-1">
   309→   309→          / 100
   310→   310→        </span>
   311→   311→      </div>
   312→   312→    </div>
   313→   313→  );
   314→   314→}
