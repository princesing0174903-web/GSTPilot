    76→// ─── Animated Number Hook ─────────────────────────────────────────────────
    77→function useAnimatedNumber(target: number, duration: number = 1200) {
    78→  const [current, setCurrent] = useState(0);
    79→  const ref = useRef<number | null>(null);
    80→  const startTime = useRef<number | null>(null);
    81→
    82→  useEffect(() => {
    83→    startTime.current = null;
    84→    const startValue = current;
    85→
    86→    function step(timestamp: number) {
    87→      if (!startTime.current) startTime.current = timestamp;
    88→      const elapsed = timestamp - startTime.current;
    89→      const progress = Math.min(elapsed / duration, 1);
    90→      const eased = 1 - Math.pow(1 - progress, 3);
    91→      setCurrent(Math.round(startValue + (target - startValue) * eased));
    92→      if (progress < 1) {
    93→        ref.current = requestAnimationFrame(step);
    94→      }
    95→    }
    96→
    97→    ref.current = requestAnimationFrame(step);
    98→    return () => {
    99→      if (ref.current) cancelAnimationFrame(ref.current);
   100→    };
   101→  }, [target, duration]);
   102→
   103→  return current;
   104→}

   106→// ─── Mini Sparkline Component ─────────────────────────────────────────────
   107→function MiniSparkline({
   108→  data,
   109→  color,
   110→  width = 80,
   111→  height = 32,
   112→}: {
   113→  data: number[];
   114→  color: string;
   115→  width?: number;
   116→  height?: number;
   117→}) {
   118→  if (data.length < 2) return null;
   119→  const min = Math.min(...data);
   120→  const max = Math.max(...data);
   121→  const range = max - min || 1;
   122→  const padding = 2;
   123→
   124→  const points = data
   125→    .map((v, i) => {
   126→      const x = padding + (i / (data.length - 1)) * (width - padding * 2);
   127→      const y = height - padding - ((v - min) / range) * (height - padding * 2);
   128→      return `${x},${y}`;
   129→    })
   130→    .join(' ');
   131→
   132→  const areaPoints = `${padding},${height - padding} ${points} ${width - padding},${height - padding}`;
   133→
   134→  return (
   135→    <svg width={width} height={height} className="shrink-0">
   136→      <defs>
   137→        <linearGradient id={`sparkGrad-${color.replace('#', '')}`} x1="0" y1="0" x2="0" y2="1">
   138→          <stop offset="0%" stopColor={color} stopOpacity={0.3} />
   139→          <stop offset="100%" stopColor={color} stopOpacity={0} />
   140→        </linearGradient>
   141→      </defs>
   142→      <polygon
   143→        points={areaPoints}
   144→        fill={`url(#sparkGrad-${color.replace('#', '')})`}
   145→      />
   146→      <polyline
   147→        points={points}
   148→        fill="none"
   149→        stroke={color}
   150→        strokeWidth={1.5}
   151→        strokeLinecap="round"
   152→        strokeLinejoin="round"
   153→      />
   154→    </svg>
   155→  );
   156→}

   363→// ─── AI Insights Data (derived from predictions) ──────────────────────────
   364→function generateInsights(predictions: Record<string, CategoryPrediction>) {
   365→  const insights: {
   366→    icon: React.ReactNode;
   367→    text: string;
   368→    confidence: number;
   369→    color: string;
   370→  }[] = [];
   371→
   372→  const revenue = predictions['revenue'];
   373→  const gstLiability = predictions['gst_liability'];
   374→  const churn = predictions['churn'];
   375→  const filingLoad = predictions['filing_load'];
   376→  const teamLoad = predictions['team_load'];
   377→  const collections = predictions['collections'];
   378→
   379→  if (revenue) {
   380→    insights.push({
   381→      icon: <TrendingUp className="h-4 w-4" />,
   382→      text:
   383→        revenue.trend === 'up'
   384→          ? `Revenue is trending upward at ${formatCurrency(revenue.predictedValue)}. Maintain current client engagement strategies.`
   385→          : `Revenue is projected at ${formatCurrency(revenue.predictedValue)}. Consider proactive outreach to boost collections.`,
   386→      confidence: revenue.confidence,
   387→      color: revenue.trend === 'up' ? 'emerald' : 'amber',
   388→    });
   389→  }
   390→
   391→  if (gstLiability) {
   392→    insights.push({
   393→      icon: <IndianRupee className="h-4 w-4" />,
   394→      text: `GST liability forecast at ${formatCurrency(gstLiability.predictedValue)}. Ensure adequate cash reserves by the 20th of filing month.`,
   395→      confidence: gstLiability.confidence,
   396→      color: gstLiability.trend === 'up' ? 'amber' : 'emerald',
   397→    });
   398→  }
   399→
   400→  if (churn && churn.predictedValue > 5) {
   401→    insights.push({
   402→      icon: <AlertTriangle className="h-4 w-4" />,
   403→      text: `Client churn risk at ${churn.predictedValue.toFixed(1)}%. Identify at-risk clients and schedule retention reviews.`,
   404→      confidence: churn.confidence,
   405→      color: 'red',
   406→    });
   407→  }
   408→
   409→  if (filingLoad && filingLoad.predictedValue > 10) {
   410→    insights.push({
   411→      icon: <FileCheck className="h-4 w-4" />,
   412→      text: `Filing load of ${formatNumber(filingLoad.predictedValue)} returns expected. Optimize team allocation to avoid bottlenecks.`,
   413→      confidence: filingLoad.confidence,
   414→      color: 'amber',
   415→    });
   416→  }
   417→
   418→  if (teamLoad && teamLoad.predictedValue > 75) {
   419→    insights.push({
   420→      icon: <Users className="h-4 w-4" />,
   421→      text: `Team utilization projected at ${teamLoad.predictedValue.toFixed(0)}%. Consider temporary resources to prevent burnout.`,
   422→      confidence: teamLoad.confidence,
   423→      color: 'amber',
   424→    });
   425→  }
   426→
   427→  if (insights.length === 0) {
   428→    insights.push({
   429→      icon: <Lightbulb className="h-4 w-4" />,
   430→      text: 'All metrics are within normal range. Continue monitoring for emerging trends.',
   431→      confidence: 0.8,
   432→      color: 'emerald',
   433→    });
   434→  }
   435→
   436→  return insights;
   437→}
