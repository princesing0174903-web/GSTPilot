    35→/** Indian-style number formatting for ₹ amounts (lakhs/crores grouping). */
    36→function inr(amount: number | undefined | null): string {
    37→  if (amount === undefined || amount === null || Number.isNaN(amount)) return '0';
    38→  return Math.round(amount).toLocaleString('en-IN');
    39→}
    40→
    41→/** Format a percentage change with explicit + / − sign. */
    42→function signedPct(pct: number | undefined): string {
    43→  if (pct === undefined || Number.isNaN(pct)) return '0%';
    44→  const rounded = Math.round(pct);
    45→  const sign = rounded > 0 ? '+' : '';
    46→  return `${sign}${rounded}%`;
    47→}

    51→function buildLiveDataBlock(data: OracleLiveData): string {
    52→  const today = new Date().toLocaleDateString('en-IN', {
    53→    day: 'numeric', month: 'short', year: 'numeric',
    54→  });
    55→
    56→  const lines: string[] = [];
    57→  lines.push(`## LIVE BUSINESS DATA (AUTHORITATIVE — reference these numbers)`);
    58→  lines.push(`Today: ${today}`);
    59→  if (data.tradeName || data.gstin) {
    60→    lines.push(`Company: ${data.tradeName ?? '—'} (GSTIN: ${data.gstin ?? '—'})`);
    61→  }
    62→
    63→  // ── GST & Compliance (only if GSTN is connected) ──
    64→  if (data.hasGstn && data.compliance) {
    65→    const c = data.compliance;
    66→    lines.push('');
    67→    lines.push('### GST & Compliance');
    68→    lines.push(`- Compliance score: ${c.score}/100 (status: ${c.status})`);
    69→    lines.push(`- Pending returns: ${c.pendingReturns}`);
    70→    lines.push(`- Overdue returns: ${c.overdueReturns}`);
    71→    if (c.lastFilingDate) lines.push(`- Last filing: ${c.lastFilingDate}`);
    72→    if (c.nextDueDate) lines.push(`- Next due date: ${c.nextDueDate}`);
    73→    lines.push(`- ITC available to claim: ₹${inr(c.itcAvailable)}`);
    74→    lines.push(`- Active GST notices: ${c.activeNotices}`);
    75→  }
    76→
    77→  // ── Banking & Cash (only if Bank is connected) ──
    78→  if (data.hasBank && data.bank) {
    79→    const b = data.bank;
    80→    lines.push('');
    81→    lines.push('### Banking & Cash');
    82→    if (b.provider) {
    83→      lines.push(`- Bank: ${b.provider}`);
    84→    }
    85→    lines.push(`- Cash available today: ₹${inr(b.cashAvailable)}`);
    86→    lines.push(`- This month collections: ₹${inr(b.monthlyCollections)} (change: ${signedPct(b.collectionChangePct)} vs last month)`);
    87→    lines.push(`- This month expenses: ₹${inr(b.monthlyExpenses)} (change: ${signedPct(b.expenseChangePct)} vs last month)`);
    88→    lines.push(`- Total credits (90d): ₹${inr(b.totalCredits)}`);
    89→    lines.push(`- Total debits (90d): ₹${inr(b.totalDebits)}`);
    90→  }
    91→
    92→  // ── Business Health Score (only if health is computed) ──
    93→  if (data.health) {
    94→    const h = data.health;
    95→    lines.push('');
    96→    lines.push(`### Business Health Score: ${h.overall}/100`);
    97→    lines.push(
    98→      `- Compliance: ${h.compliance} · Cash Flow: ${h.cashFlow} · Collection: ${h.collection} · Growth: ${h.growth} · Profitability: ${h.profitability} · Risk: ${h.risk}`,
    99→    );
   100→    if (h.signals && h.signals.length > 0) {
   101→      lines.push('- Key signals:');
   102→      for (const s of h.signals.slice(0, 3)) {
   103→        lines.push(`  - ${s.text}`);
   104→      }
   105→    }
   106→  }
   107→
   108→  // ── Risky Clients (only if any exist) ──
   109→  if (data.riskyClients && data.riskyClients.length > 0) {
   110→    lines.push('');
   111→    lines.push('### Risky Clients');
   112→    for (const r of data.riskyClients.slice(0, 5)) {
   113→      lines.push(`- ${r.name}: ${r.reason} (₹${inr(r.amount)})`);
   114→    }
   115→  }
   116→
   117→  // ── Recent transactions (last few, for grounding) ──
   118→  if (data.recentTransactions && data.recentTransactions.length > 0) {
   119→    lines.push('');
   120→    lines.push('### Recent Bank Transactions (latest 5)');
   121→    for (const t of data.recentTransactions.slice(0, 5)) {
   122→      const sign = t.type === 'credit' ? '+' : '-';
   123→      lines.push(`- ${t.txnDate.slice(0, 10)} · ${t.description} · ${sign}₹${inr(Math.abs(t.amount))}${t.counterparty ? ` (${t.counterparty})` : ''}`);
   124→    }
   125→  }
   126→
   127→  lines.push('');
   128→  lines.push('Use these numbers when the user asks about their business status, cash flow, collections, compliance, returns, or what to do. Treat them as authoritative and current.');
   129→  return lines.join('\n');
   130→}

   366→/** Server-side language hint derived from the latest user message. */
   367→function inferLanguageHint(messages: OracleChatRequest['messages']): OracleLanguageId | undefined {
   368→  for (let i = messages.length - 1; i >= 0; i--) {
   369→    if (messages[i].role === 'user' && messages[i].content?.trim()) {
   370→      const s = messages[i].content.slice(0, 500);
   371→      if (/[\u0900-\u097F]/.test(s)) return 'hindi';
   372→      if (/[\u0600-\u06FF]/.test(s)) return 'urdu';
   373→      if (/[\u0A00-\u0A7F]/.test(s)) return 'punjabi';
   374→      if (/[\u0A80-\u0AFF]/.test(s)) return 'gujarati';
   375→      if (/[\u0B80-\u0BFF]/.test(s)) return 'tamil';
   376→      if (/[\u0C00-\u0C7F]/.test(s)) return 'telugu';
   377→      if (/[\u0980-\u09FF]/.test(s)) return 'bengali';
   378→      return 'english';
   379→    }
   380→  }
   381→  return undefined;
   382→}

   384→/** Last user message text (for source retrieval + action detection + memory). */
   385→function lastUserMessage(messages: OracleChatRequest['messages']): string {
   386→  for (let i = messages.length - 1; i >= 0; i--) {
   387→    if (messages[i].role === 'user') return messages[i].content ?? '';
   388→  }
   389→  return '';
   390→}

   394→export async function POST(request: Request) {
   395→  let body: OracleChatRequest;
   396→  try {
   397→    body = (await request.json()) as OracleChatRequest;
   398→  } catch {
   399→    return new Response(JSON.stringify({ error: 'Invalid request' }), {
   400→      status: 400,
   401→      headers: { 'Content-Type': 'application/json' },
   402→    });
   403→  }
   404→
   405→  const messages = body.messages ?? [];
   406→  if (!Array.isArray(messages) || messages.length === 0) {
   407→    return new Response(JSON.stringify({ error: 'Invalid request' }), {
   408→      status: 400,
   409→      headers: { 'Content-Type': 'application/json' },
   410→    });
   411→  }
   412→
   413→  // ── Live-data context enrichment (silent failure) ─────────────────────────
   414→  let liveData: OracleLiveData | undefined = body.context?.liveData;
   415→  if (!liveData) {
   416→    try {
   417→      const { loadOracleLiveData } = await import('@/lib/connections');
   418→      liveData = await loadOracleLiveData();
   419→    } catch {
   420→      /* ignore — chat still works without live context */
   421→    }
   422→  }
   423→  const enrichedBody: OracleChatRequest = {
   424→    ...body,
   425→    context: { ...body.context, liveData },
   426→  };
   427→
   428→  // ── System prompt (with memory + documents) ───────────────────────────────
   429→  let promptData: { prompt: string; memoryBlock: string } | null = null;
   430→  try {
   431→    promptData = await buildSystemPrompt(enrichedBody);
   432→  } catch {
   433→    // Memory/source load failed — fall back to a minimal prompt so chat still works.
   434→    promptData = {
   435→      prompt: `${BRAND_IDENTITY_PROMPT_BLOCK}\n\nYou are GSTPilot Oracle™ — the AI Financial Officer for Indian businesses. Reply in the user's language. Be professional, warm, executive. Cite GST law where relevant. Use ₹ and Indian number formatting.`,
   436→      memoryBlock: '',
   437→    };
   438→  }
   439→
   440→  const systemPrompt = promptData.prompt;
   441→  const languageHint = inferLanguageHint(messages);
   442→  const userText = lastUserMessage(messages);
   443→
   444→  // ── Source retrieval (silent failure) ─────────────────────────────────────
   445→  let relevantSources: Awaited<ReturnType<typeof retrieveSources>> = [];
   446→  try {
   447→    if (userText) relevantSources = await retrieveSources(userText, 3);
   448→  } catch {
   449→    /* ignore */
   450→  }
   451→  const sourcesBlock = relevantSources.length > 0 ? renderSourcesBlock(relevantSources) : '';
   452→  const finalSystemPrompt = sourcesBlock ? `${systemPrompt}\n\n${sourcesBlock}` : systemPrompt;
   453→
   454→  const modelMessages: { role: 'assistant' | 'user' | 'system'; content: string }[] = [
   455→    { role: 'assistant', content: finalSystemPrompt },
   456→    ...toModelMessages(messages),
   457→  ];
   458→
   459→  // ── Acquire the upstream stream from the SDK ───────────────────────────────
   460→  let upstream: ReadableStream<Uint8Array> | null = null;
   461→  try {
   462→    const zai = await ZAI.create();
   463→    const result = await zai.chat.completions.create({
   464→      messages: modelMessages,
   465→      stream: true,
   466→      thinking: { type: 'disabled' },
   467→    });
   468→    if (result && typeof (result as ReadableStream<Uint8Array>).getReader === 'function') {
   469→      upstream = result as ReadableStream<Uint8Array>;
   470→    } else {
   471→      // Non-streaming fallback: emit the full text as one chunk then close.
   472→      const text =
   473→        (result as { choices?: { message?: { content?: string } }[] })?.choices?.[0]?.message
   474→          ?.content ?? '';
   475→      upstream = new ReadableStream<Uint8Array>({
   476→        start(controller) {
   477→          if (text) controller.enqueue(sseChunk({ token: text }));
   478→          controller.close();
   479→        },
   480→      });
   481→    }
   482→  } catch {
   483→    // RELIABILITY: never surface the underlying 401 / token / quota error.
   484→    // Return a graceful, user-facing message instead.
   485→    const stream = new ReadableStream<Uint8Array>({
   486→      start(controller) {
   487→        controller.enqueue(
   488→          sseChunk({
   489→            token:
   490→              'Oracle is temporarily unavailable. Please try again in a moment — your conversation is safe.',
   491→          }),
   492→        );
   493→        controller.enqueue(sseChunk({ done: true }));
   494→        controller.close();
   495→      },
   496→    });
   497→    return new Response(stream, { status: 200, headers: sseHeaders() });
   498→  }
   499→
   500→  // ── Transform the upstream SSE stream into our token stream ────────────────
   501→  const decoder = new TextDecoder();
   502→  const transformed = new ReadableStream<Uint8Array>({
   503→    async start(controller) {
   504→      if (languageHint) controller.enqueue(sseChunk({ language: languageHint }));
   505→
   506→      const reader = upstream!.getReader();
   507→      let buffer = '';
   508→      let emittedAny = false;
   509→      let fullText = '';
   510→      try {
   511→        while (true) {
   512→          const { done, value } = await reader.read();
   513→          if (done) break;
   514→          buffer += decoder.decode(value, { stream: true });
   515→
   516→          const lines = buffer.split('\n');
   517→          buffer = lines.pop() ?? '';
   518→
   519→          for (const rawLine of lines) {
   520→            const line = rawLine.trim();
   521→            if (!line || !line.startsWith('data:')) continue;
   522→            const data = line.slice(5).trim();
   523→            if (data === '[DONE]' || !data) continue;
   524→            try {
   525→              const json = JSON.parse(data);
   526→              const token: string =
   527→                json?.choices?.[0]?.delta?.content ??
   528→                json?.choices?.[0]?.message?.content ??
   529→                '';
   530→              if (token) {
   531→                emittedAny = true;
   532→                fullText += token;
   533→                controller.enqueue(sseChunk({ token }));
   534→              }
   535→            } catch {
   536→              // Partial JSON across a chunk boundary — resolves on next read.
   537→            }
   538→          }
   539→        }
   540→        // Flush any trailing buffered line.
   541→        const tail = buffer.trim();
   542→        if (tail.startsWith('data:')) {
   543→          const data = tail.slice(5).trim();
   544→          if (data && data !== '[DONE]') {
   545→            try {
   546→              const json = JSON.parse(data);
   547→              const token: string =
   548→                json?.choices?.[0]?.delta?.content ??
   549→                json?.choices?.[0]?.message?.content ??
   550→                '';
   551→              if (token) {
   552→                emittedAny = true;
   553→                fullText += token;
   554→                controller.enqueue(sseChunk({ token }));
   555→              }
   556→            } catch {
   557→              /* ignore */
   558→            }
   559→          }
   560→        }
   561→        if (!emittedAny) {
   562→          fullText =
   563→            "I'm here. Based on current GST rules and the information available, I'd be glad to help — could you share a bit more about what you're looking to do?";
   564→          controller.enqueue(sseChunk({ token: fullText }));
   565→        }
   566→
   567→        // ── Emit sources (if any) before closing ────────────────────────────
   568→        if (relevantSources.length > 0) {
   569→          controller.enqueue(sseChunk({ sources: relevantSources }));
   570→        }
   571→
   572→        // ── Detect + emit autopilot actions ─────────────────────────────────
   573→        let detectedActions: ReturnType<typeof detectActions> = [];
   574→        try {
   575→          detectedActions = detectActions({
   576→            userMessage: userText,
   577→            oracleResponse: fullText,
   578→            liveData,
   579→          });
   580→          if (detectedActions.length > 0) {
   581→            controller.enqueue(sseChunk({ actions: detectedActions }));
   582→          }
   583→        } catch {
   584→          /* ignore */
   585→        }
   586→
   587→        // ── Persist long-term memory from this exchange (fire-and-forget) ──
   588→        try {
   589→          extractAndPersistFacts({
   590→            userMessage: userText,
   591→            oracleResponse: fullText,
   592→            knownUserName: body.memory?.userName,
   593→            knownFirmName: body.memory?.firmName,
   594→            knownGstin: body.memory?.gstin,
   595→          }).catch(() => undefined);
   596→        } catch {
   597→          /* ignore */
   598→        }
   599→
   600→        controller.enqueue(sseChunk({ done: true }));
   601→        controller.close();
   602→      } catch {
   603→        // RELIABILITY: stream interrupted — never leak the underlying error.
   604→        if (!emittedAny) {
   605→          controller.enqueue(
   606→            sseChunk({
   607→              token:
   608→                'Oracle is temporarily unavailable. Please try again in a moment — your conversation is safe.',
   609→            }),
   610→          );
   611→        }
   612→        controller.enqueue(sseChunk({ done: true }));
   613→        controller.close();
   614→      }
   615→    },
   616→    cancel() {
   617→      upstream?.cancel?.().catch(() => undefined);
   618→    },
   619→  });
   620→
   621→  return new Response(transformed, { status: 200, headers: sseHeaders() });
   622→}
