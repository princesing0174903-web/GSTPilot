   116→const RECENT_STORAGE_KEY = 'gstpilot-recent-commands';
   117→const FAVORITES_STORAGE_KEY = 'gstpilot-favorite-commands';
   118→const MAX_RECENT = 5;
   119→
   120→// ═══════════════════════════════════════════════════════════════════════════════
   121→// LOCAL STORAGE HELPERS
   122→// ═══════════════════════════════════════════════════════════════════════════════
   123→
   124→function loadRecent(): RecentItem[] {
   125→  try {
   126→    const raw = localStorage.getItem(RECENT_STORAGE_KEY);
   127→    return raw ? JSON.parse(raw) : [];
   128→  } catch {
   129→    return [];
   130→  }
   131→}
   132→
   133→function saveRecent(items: RecentItem[]) {
   134→  try {
   135→    localStorage.setItem(RECENT_STORAGE_KEY, JSON.stringify(items.slice(0, MAX_RECENT)));
   136→  } catch {
   137→    // ignore quota errors
   138→  }
   139→}
   140→
   141→function loadFavorites(): FavoriteItem[] {
   142→  try {
   143→    const raw = localStorage.getItem(FAVORITES_STORAGE_KEY);
   144→    return raw ? JSON.parse(raw) : [];
   145→  } catch {
   146→    return [];
   147→  }
   148→}
   149→
   150→function saveFavorites(items: FavoriteItem[]) {
   151→  try {
   152→    localStorage.setItem(FAVORITES_STORAGE_KEY, JSON.stringify(items));
   153→  } catch {
   154→    // ignore quota errors
   155→  }
   156→}
