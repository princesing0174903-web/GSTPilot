    65→    65→function fmtINR(n: number): string {
    66→    66→  if (!isFinite(n)) return '₹0';
    67→    67→  const abs = Math.abs(n);
    68→    68→  if (abs >= 10000000) return `₹${(n / 10000000).toFixed(2)}Cr`;
    69→    69→  if (abs >= 100000) return `₹${(n / 100000).toFixed(2)}L`;
    70→    70→  if (abs >= 1000) return `₹${(n / 1000).toFixed(1)}K`;
    71→    71→  return `₹${Math.round(n).toLocaleString('en-IN')}`;
    72→    72→}
    73→    73→
    74→    74→function fmtNum(n: number): string {
    75→    75→  if (n >= 10000000) return `${(n / 10000000).toFixed(2)}Cr`;
    76→    76→  if (n >= 100000) return `${(n / 100000).toFixed(2)}L`;
    77→    77→  if (n >= 1000) return `${(n / 1000).toFixed(1)}K`;
    78→    78→  return n.toLocaleString('en-IN');
    79→    79→}

    81→    81→function timeAgo(iso: string | null): string {
    82→    82→  if (!iso) return 'never';
    83→    83→  const diff = Date.now() - new Date(iso).getTime();
    84→    84→  const s = Math.floor(diff / 1000);
    85→    85→  if (s < 60) return 'just now';
    86→    86→  const m = Math.floor(s / 60);
    87→    87→  if (m < 60) return `${m}m ago`;
    88→    88→  const h = Math.floor(m / 60);
    89→    89→  if (h < 24) return `${h}h ago`;
    90→    90→  const d = Math.floor(h / 24);
    91→    91→  if (d < 30) return `${d}d ago`;
    92→    92→  return new Date(iso).toLocaleDateString('en-IN');
    93→    93→}
