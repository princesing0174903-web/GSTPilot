    52→    52→function getDaysRemaining(dueDateStr: string): number {
    53→    53→  const dueDate = new Date(dueDateStr);
    54→    54→  const today = new Date();
    55→    55→  today.setHours(0, 0, 0, 0);
    56→    56→  dueDate.setHours(0, 0, 0, 0);
    57→    57→  return Math.ceil((dueDate.getTime() - today.getTime()) / (1000 * 60 * 60 * 24));
    58→    58→}
    59→    59→
    60→    60→function formatDaysRemaining(days: number): string {
    61→    61→  if (days < 0) return `${Math.abs(days)}d overdue`;
    62→    62→  if (days === 0) return 'Due today';
    63→    63→  if (days === 1) return 'Tomorrow';
    64→    64→  return `${days}d left`;
    65→    65→}
    66→    66→
    67→    67→function getInitials(name: string): string {
    68→    68→  return name.split(' ').map(w => w[0]).join('').slice(0, 2).toUpperCase();
    69→    69→}
    70→    70→
    71→    71→function formatINR(amount: number): string {
    72→    72→  return amount.toLocaleString('en-IN');
    73→    73→}
    74→    74→
    75→    75→function timeAgo(dateStr: string): string {
    76→    76→  const now = new Date();
    77→    77→  const date = new Date(dateStr);
    78→    78→  const seconds = Math.floor((now.getTime() - date.getTime()) / 1000);
    79→    79→  if (seconds < 60) return 'just now';
    80→    80→  const minutes = Math.floor(seconds / 60);
    81→    81→  if (minutes < 60) return `${minutes}m ago`;
    82→    82→  const hours = Math.floor(minutes / 60);
    83→    83→  if (hours < 24) return `${hours}h ago`;
    84→    84→  const days = Math.floor(hours / 24);
    85→    85→  if (days < 7) return `${days}d ago`;
    86→    86→  return date.toLocaleDateString('en-IN', { day: 'numeric', month: 'short' });
    87→    87→}
