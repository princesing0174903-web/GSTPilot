    35→function CollectionMethodChart({ data }: { data: { method: string; amount: number; color: string }[] }) {

    42→        return (
    43→          <div key={d.method}>
    44→            <div className="flex justify-between text-xs mb-1">
    45→              <span className="text-muted-foreground">{d.method}</span>
    46→              <span className="font-medium">{fmtINR(d.amount)} ({pct.toFixed(0)}%)</span>

    58→function PaymentTrendChart({ data }: { data: { week: string; collected: number; paid: number }[] }) {
    59→  const max = Math.max(...data.map(d => Math.max(d.collected, d.paid)))
    60→  const barW = 20
    61→  const gap = 12
    62→  const h = 70
    63→  return (
    64→    <svg width={data.length * (barW * 2 + gap)} height={h + 20} className="overflow-visible">
    65→      {data.map((d, i) => {
    66→        const x = i * (barW * 2 + gap)
    67→        const hC = (d.collected / max) * (h - 8)
    68→        const hP = (d.paid / max) * (h - 8)
    69→        return (
    70→          <g key={d.week}>
    71→            <rect x={x} y={h - hC} width={barW} height={hC} rx={3} fill="#10b981" opacity={0.7} />
    72→            <rect x={x + barW + 2} y={h - hP} width={barW} height={hP} rx={3} fill="#f59e0b" opacity={0.6} />
    73→            <text x={x + barW + 1} y={h + 14} textAnchor="middle" className="text-[9px] fill-muted-foreground">{d.week}</text>
    74→          </g>
    75→        )
    76→      })}
    77→    </svg>
    78→  )
    79→}

   163→function formatTxnDate(iso: string): string {
   164→  if (!iso) return '—'
   165→  const d = new Date(iso)
   166→  if (Number.isNaN(d.getTime())) return iso
   167→  const dd = String(d.getDate()).padStart(2, '0')
   168→  const mm = String(d.getMonth() + 1).padStart(2, '0')
   169→  const yyyy = d.getFullYear()
   170→  return `${dd}/${mm}/${yyyy}`
   171→}
   172→
   173→function titleCaseMode(mode: string): string {
   174→  if (!mode) return 'Bank'
   175→  const lower = mode.toLowerCase()
   176→  if (lower === 'upi') return 'UPI'
   177→  if (lower === 'bank') return 'Net Banking'
   178→  if (lower === 'card') return 'Card'
   179→  if (lower === 'cheque') return 'Cheque'
   180→  if (lower === 'cash') return 'Cash'
   181→  return mode.charAt(0).toUpperCase() + mode.slice(1)
   182→}
