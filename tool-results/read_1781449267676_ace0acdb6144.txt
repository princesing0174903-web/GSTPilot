   109→   109→function formatFileSize(bytes: number): string {
   110→   110→  if (bytes === 0) return '0 B';
   111→   111→  const k = 1024;
   112→   112→  const sizes = ['B', 'KB', 'MB', 'GB'];
   113→   113→  const i = Math.floor(Math.log(bytes) / Math.log(k));
   114→   114→  return parseFloat((bytes / Math.pow(k, i)).toFixed(1)) + ' ' + sizes[i];
   115→   115→}

   149→   149→function useAnimatedNumber(target: number, duration: number = 1200) {
   150→   150→  const [current, setCurrent] = useState(0);
   151→   151→  const ref = useRef<number | null>(null);
   152→   152→  const startTime = useRef<number | null>(null);
   153→   153→
   154→   154→  useEffect(() => {
   155→   155→    startTime.current = null;
   156→   156→    const startValue = current;
   157→   157→
   158→   158→    function step(timestamp: number) {
   159→   159→      if (!startTime.current) startTime.current = timestamp;
   160→   160→      const elapsed = timestamp - startTime.current;
   161→   161→      const progress = Math.min(elapsed / duration, 1);
   162→   162→      const eased = 1 - Math.pow(1 - progress, 3);
   163→   163→      setCurrent(Math.round(startValue + (target - startValue) * eased));
   164→   164→      if (progress < 1) {
   165→   165→        ref.current = requestAnimationFrame(step);
   166→   166→      }
   167→   167→    }
   168→   168→
   169→   169→    ref.current = requestAnimationFrame(step);
   170→   170→    return () => {
   171→   171→      if (ref.current) cancelAnimationFrame(ref.current);
   172→   172→    };
   173→   173→  }, [target, duration]);
   174→   174→
   175→   175→  return current;
   176→   176→}
