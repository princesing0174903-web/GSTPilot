    85→// ─── Permission-error detection (graceful degradation) ──────────────────
    86→// Mirrors the helper in src/hooks/use-firestore.ts so the Settings page can
    87→// detect Firestore permission-denied errors and surface a friendly message
    88→// instead of the raw "Missing or insufficient permissions" wall.
    89→function isPermissionError(err: unknown): boolean {
    90→  if (!err) return false
    91→  const e = err as { code?: string; message?: string; name?: string }
    92→  const code = (e.code || '').toLowerCase()
    93→  const message = (e.message || '').toLowerCase()
    94→  if (
    95→    code === 'permission-denied' ||
    96→    code === 'unauthenticated' ||
    97→    code === 'auth/operation-not-allowed' ||
    98→    code === 'auth/user-not-found'
    99→  ) {
   100→    return true
   101→  }
   102→  if (
   103→    message.includes('missing or insufficient permissions') ||
   104→    message.includes('permission-denied') ||
   105→    message.includes('insufficient permissions') ||
   106→    message.includes('not authorized') ||
   107→    message.includes('unauthenticated')
   108→  ) {
   109→    return true
   110→  }
   111→  return false
   112→}

   153→// ─── Team Member Type ───────────────────────────────────────────────────
   154→// The Settings UI surfaces a simplified 4-role taxonomy (Admin / Manager /
   155→// Staff / Viewer). The Firestore `organization_members` collection stores the
   156→// richer `OrgRole` enum (owner / admin / accountant / employee / auditor /
   157→// viewer). The two helpers below translate between the two representations so
   158→// the UI keeps its existing labels while the persistence layer uses the
   159→// canonical org-role vocabulary.
   160→interface TeamMember {
   161→  id: string
   162→  /** Firestore Auth uid — used as the key for role/status mutations. */
   163→  userId: string
   164→  name: string
   165→  email: string
   166→  role: 'Admin' | 'Manager' | 'Staff' | 'Viewer'
   167→  status: 'Active' | 'Invited'
   168→  initials: string
   169→}
   170→
   171→/**
   172→ * Map the Firestore `OrgRole` enum to the four UI role labels. The org owner
   173→ * and admin both surface as "Admin" so the existing UI rule (hide Edit/Remove
   174→ * for Admin rows) protects the owner from accidental demotion.
   175→ */
   176→function orgRoleToUiRole(role: string): TeamMember['role'] {
   177→  switch (role) {
   178→    case 'owner':
   179→    case 'admin':
   180→      return 'Admin'
   181→    case 'accountant':
   182→      return 'Manager'
   183→    case 'employee':
   184→      return 'Staff'
   185→    case 'auditor':
   186→    case 'viewer':
   187→    default:
   188→      return 'Viewer'
   189→  }
   190→}
   191→
   192→/** Map a UI role label back to the Firestore `OrgRole` enum for persistence. */
   193→function uiRoleToOrgRole(role: TeamMember['role']): OrgRole {
   194→  switch (role) {
   195→    case 'Admin':
   196→      return 'admin'
   197→    case 'Manager':
   198→      return 'accountant'
   199→    case 'Staff':
   200→      return 'employee'
   201→    case 'Viewer':
   202→    default:
   203→      return 'viewer'
   204→  }
   205→}
   206→
   207→/** Map the Firestore `MemberStatus` to the two-state UI label. */
   208→function memberStatusToUi(status: string): TeamMember['status'] {
   209→  return status === 'invited' ? 'Invited' : 'Active'
   210→}

   212→/** Derive 2-letter initials for the avatar fallback. */
   213→function getInitials(name: string, email: string): string {
   214→  if (name && name.trim()) {
   215→    const parts = name.trim().split(/\s+/).filter(Boolean)
   216→    if (parts.length >= 2) {
   217→      return (parts[0][0] + parts[1][0]).toUpperCase()
   218→    }
   219→    return parts[0].slice(0, 2).toUpperCase()
   220→  }
   221→  return (email.split('@')[0] || '?').slice(0, 2).toUpperCase()
   222→}

   357→// ─── Password Strength Helper ──────────────────────────────────────────
   358→function getPasswordStrength(password: string): { score: number; label: string; color: string } {
   359→  if (!password) return { score: 0, label: '', color: '' }
   360→  let score = 0
   361→  if (password.length >= 8) score++
   362→  if (password.length >= 12) score++
   363→  if (/[A-Z]/.test(password)) score++
   364→  if (/[0-9]/.test(password)) score++
   365→  if (/[^A-Za-z0-9]/.test(password)) score++
   366→
   367→  if (score <= 1) return { score: 20, label: 'Weak', color: 'bg-red-500' }
   368→  if (score <= 2) return { score: 40, label: 'Fair', color: 'bg-orange-500' }
   369→  if (score <= 3) return { score: 60, label: 'Good', color: 'bg-yellow-500' }
   370→  if (score <= 4) return { score: 80, label: 'Strong', color: 'bg-emerald-500' }
   371→  return { score: 100, label: 'Excellent', color: 'bg-emerald-600' }
   372→}
