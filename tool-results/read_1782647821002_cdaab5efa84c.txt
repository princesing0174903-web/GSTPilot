   112→function timeAgo(iso: string | null): string {
   113→  if (!iso) return 'never'
   114→  const diff = Date.now() - new Date(iso).getTime()
   115→  if (diff < 60000) return 'just now'
   116→  if (diff < 3600000) return `${Math.floor(diff / 60000)}m ago`
   117→  if (diff < 86400000) return `${Math.floor(diff / 3600000)}h ago`
   118→  return `${Math.floor(diff / 86400000)}d ago`
   119→}

   357→// INSTALLED TAB — my integrations with connect/sync/test
   358→// ═══════════════════════════════════════════════════════════════════════════════
   359→function InstalledTab({ onRefresh }: { onRefresh: () => void }) {
   360→  const [installed, setInstalled] = useState<InstalledIntegration[]>([])
   361→  const [loading, setLoading] = useState(true)
   362→  const [busy, setBusy] = useState<string | null>(null)
   363→  const [toast, setToast] = useState<string | null>(null)
   364→
   365→  const load = useCallback(async () => {
   366→    setLoading(true)
   367→    try {
   368→      const res = await apiGet<{ installed: InstalledIntegration[] }>('/integrations/installed')
   369→      setInstalled(res.installed)
   370→    } catch { /* ignore */ } finally { setLoading(false) }
   371→  }, [])
   372→
   373→  useEffect(() => { load() }, [load])
   374→
   375→  const action = async (id: string, actionType: 'connect' | 'disconnect' | 'sync' | 'test' | 'uninstall', name: string) => {
   376→    setBusy(`${id}-${actionType}`)
   377→    try {
   378→      if (actionType === 'connect') {
   379→        await apiPost('/integrations/connect', { installationId: id, connectedAccountId: `user@${name.toLowerCase().replace(/\s+/g, '')}.com`, scopes: [] })
   380→        setToast(`✅ ${name} connected.`)
   381→      } else if (actionType === 'disconnect') {
   382→        await apiPost('/integrations/disconnect', { installationId: id })
   383→        setToast(`⚠️ ${name} disconnected.`)
   384→      } else if (actionType === 'sync') {
   385→        const res = await apiPost<{ job: SyncJob }>('/integrations/sync', { installationId: id })
   386→        setToast(`🔄 ${name} sync ${res.job.status}: ${res.job.recordsProcessed} records in ${res.job.durationMs}ms.`)
   387→      } else if (actionType === 'test') {
   388→        const res = await apiPost<{ success: boolean; latencyMs: number; message: string }>('/integrations/test', { installationId: id })
   389→        setToast(`${res.success ? '✅' : '❌'} ${res.message}`)
   390→      } else if (actionType === 'uninstall') {
   391→        await apiPost('/integrations/uninstall', { installationId: id })
   392→        setToast(`🗑️ ${name} uninstalled.`)
   393→      }
   394→      onRefresh()
   395→      await load()
   396→      setTimeout(() => setToast(null), 4000)
   397→    } catch (e) {
   398→      setToast(`❌ ${(e as Error).message}`)
   399→      setTimeout(() => setToast(null), 4000)
   400→    } finally { setBusy(null) }
   401→  }
   402→
