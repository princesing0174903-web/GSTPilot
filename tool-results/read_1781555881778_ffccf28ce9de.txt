    77→function getDaysRemaining(dueDateStr: string): number {
    78→  const dueDate = new Date(dueDateStr);
    79→  const today = new Date();
    80→  today.setHours(0, 0, 0, 0);
    81→  dueDate.setHours(0, 0, 0, 0);
    82→  return Math.ceil((dueDate.getTime() - today.getTime()) / (1000 * 60 * 60 * 24));
    83→}
    84→
    85→function formatDaysRemaining(days: number): string {
    86→  if (days < 0) return `${Math.abs(days)}d overdue`;
    87→  if (days === 0) return 'Due today';
    88→  if (days === 1) return 'Tomorrow';
    89→  return `${days}d left`;
    90→}
    91→
    92→function formatIndianDate(dateStr: string): string {
    93→  if (!dateStr) return '—';
    94→  const d = new Date(dateStr);
    95→  if (isNaN(d.getTime())) return '—';
    96→  const day = String(d.getDate()).padStart(2, '0');
    97→  const month = String(d.getMonth() + 1).padStart(2, '0');
    98→  const year = d.getFullYear();
    99→  return `${day}/${month}/${year}`;
   100→}
   101→
   102→function timeAgo(dateStr: string): string {
   103→  if (!dateStr) return '';
   104→  const now = new Date();
   105→  const date = new Date(dateStr);
   106→  const seconds = Math.floor((now.getTime() - date.getTime()) / 1000);
   107→  if (seconds < 60) return 'just now';
   108→  const minutes = Math.floor(seconds / 60);
   109→  if (minutes < 60) return `${minutes}m ago`;
   110→  const hours = Math.floor(minutes / 60);
   111→  if (hours < 24) return `${hours}h ago`;
   112→  const days = Math.floor(hours / 24);
   113→  if (days < 7) return `${days}d ago`;
   114→  return date.toLocaleDateString('en-IN', { day: 'numeric', month: 'short' });
   115→}

   167→function MiniSparkline({ data, color = '#059669', width = 120, height = 32 }: {
   168→  data: number[];
   169→  color?: string;
   170→  width?: number;
   171→  height?: number;
   172→}) {
   173→  if (data.length < 2) return null;
   174→  const min = Math.min(...data);
   175→  const max = Math.max(...data);
   176→  const range = max - min || 1;
   177→  const padding = 2;
   178→
   179→  const points = data.map((v, i) => {
   180→    const x = padding + (i / (data.length - 1)) * (width - padding * 2);
   181→    const y = height - padding - ((v - min) / range) * (height - padding * 2);
   182→    return `${x},${y}`;
   183→  });
   184→
   185→  const pathD = `M ${points.join(' L ')}`;
   186→  const areaD = `${pathD} L ${width - padding},${height - padding} L ${padding},${height - padding} Z`;
   187→
   188→  return (
   189→    <svg width={width} height={height} className="overflow-visible">
   190→      <defs>
   191→        <linearGradient id="sparkGrad" x1="0" x2="0" y1="0" y2="1">
   192→          <stop offset="0%" stopColor={color} stopOpacity="0.2" />
   193→          <stop offset="100%" stopColor={color} stopOpacity="0" />
   194→        </linearGradient>
   195→      </defs>
   196→      <path d={areaD} fill="url(#sparkGrad)" />
   197→      <path d={pathD} fill="none" stroke={color} strokeWidth="1.5" strokeLinecap="round" strokeLinejoin="round" />
   198→      <circle cx={width - padding} cy={height - padding - ((data[data.length - 1] - min) / range) * (height - padding * 2)} r="2.5" fill={color} />
   199→    </svg>
   200→  );
   201→}

   207→function CircularGauge({ score, size = 160, strokeWidth = 10 }: {
   208→  score: number;
   209→  size?: number;
   210→  strokeWidth?: number;
   211→}) {
   212→  const radius = (size - strokeWidth) / 2;
   213→  const circumference = 2 * Math.PI * radius;
   214→  const offset = circumference - (score / 100) * circumference;
   215→  const color = getHealthStroke(score);
   216→  const center = size / 2;
   217→
   218→  return (
   219→    <div className="relative inline-flex items-center justify-center">
   220→      <svg width={size} height={size} className="-rotate-90">
   221→        <circle
   222→          cx={center}
   223→          cy={center}
   224→          r={radius}
   225→          fill="none"
   226→          stroke="currentColor"
   227→          strokeWidth={strokeWidth}
   228→          className="text-muted/30"
   229→        />
   230→        <motion.circle
   231→          cx={center}
   232→          cy={center}
   233→          r={radius}
   234→          fill="none"
   235→          stroke={color}
   236→          strokeWidth={strokeWidth}
   237→          strokeLinecap="round"
   238→          strokeDasharray={circumference}
   239→          initial={{ strokeDashoffset: circumference }}
   240→          animate={{ strokeDashoffset: offset }}
   241→          transition={{ duration: 1.2, ease: 'easeOut' }}
   242→        />
   243→      </svg>
   244→      <div className="absolute inset-0 flex flex-col items-center justify-center">
   245→        <motion.span
   246→          className={`text-3xl font-bold ${getHealthColor(score)}`}
   247→          initial={{ opacity: 0, scale: 0.5 }}
   248→          animate={{ opacity: 1, scale: 1 }}
   249→          transition={{ duration: 0.6, delay: 0.4 }}
   250→        >
   251→          {score}
   252→        </motion.span>
   253→        <span className="text-[10px] font-medium text-muted-foreground uppercase tracking-wider">out of 100</span>
   254→      </div>
   255→    </div>
   256→  );
   257→}

   260→// BAR CHART (Cash Flow)

   262→
   263→function CashFlowBarChart({ data }: {
   264→  data: Array<{ month: string; payable: number; receivable: number }>;
   265→}) {
   266→  if (data.length === 0) return null;
   267→
   268→  const maxVal = Math.max(...data.flatMap(d => [d.payable, d.receivable]), 1);
   269→  const barWidth = 24;
   270→  const groupGap = 40;
   271→  const chartHeight = 140;
   272→  const chartWidth = data.length * (barWidth * 2 + 8) + (data.length - 1) * groupGap + 60;
   273→
   274→  return (
   275→    <svg width="100%" height={chartHeight + 40} viewBox={`0 0 ${chartWidth} ${chartHeight + 40}`} className="overflow-visible">
   276→      {data.map((d, i) => {
   277→        const x = 40 + i * (barWidth * 2 + 8 + groupGap);
   278→        const payH = (d.payable / maxVal) * chartHeight;
   279→        const recH = (d.receivable / maxVal) * chartHeight;
   280→        return (
   281→          <g key={d.month}>
   282→            {/* Payable bar */}
   283→            <motion.rect
   284→              x={x}
   285→              y={chartHeight - payH + 10}
   286→              width={barWidth}
   287→              height={payH}
   288→              rx={3}
   289→              fill="#ef4444"
   290→              initial={{ height: 0, y: chartHeight + 10 }}
   291→              animate={{ height: payH, y: chartHeight - payH + 10 }}
   292→              transition={{ duration: 0.6, delay: i * 0.1 }}
   293→              opacity={0.8}
   294→            />
   295→            {/* Receivable bar */}
   296→            <motion.rect
   297→              x={x + barWidth + 4}
   298→              y={chartHeight - recH + 10}
   299→              width={barWidth}
   300→              height={recH}
   301→              rx={3}
   302→              fill="#059669"
   303→              initial={{ height: 0, y: chartHeight + 10 }}
   304→              animate={{ height: recH, y: chartHeight - recH + 10 }}
   305→              transition={{ duration: 0.6, delay: i * 0.1 + 0.05 }}
   306→              opacity={0.8}
   307→            />
   308→            {/* Month label */}
   309→            <text x={x + barWidth + 2} y={chartHeight + 28} textAnchor="middle" className="fill-muted-foreground text-[10px]">
   310→              {d.month}
   311→            </text>
   312→          </g>
   313→        );
   314→      })}
   315→      {/* Baseline */}
   316→      <line x1="30" y1={chartHeight + 10} x2={chartWidth - 10} y2={chartHeight + 10} stroke="currentColor" className="text-muted/30" strokeWidth="1" />
   317→    </svg>
   318→  );
   319→}

  1187→// 8. CASH FLOW FORECAST CARD

  1190→const CashFlowForecastCard = React.memo(function CashFlowForecastCard({
  1191→  invoices,
  1192→  returns,
  1193→  onViewAnalytics,
  1194→}: {
  1195→  invoices: Array<FirestoreInvoice & { id: string }>;
  1196→  returns: Array<FirestoreReturn & { id: string }>;
  1197→  onViewAnalytics: () => void;
  1198→}) {
  1199→  const { chartData, netCashFlow, projectionData } = useMemo(() => {
  1200→    const now = new Date();
  1201→    const months = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];
  1202→
  1203→    const data: Array<{ month: string; payable: number; receivable: number }> = [];
  1204→    let totalPayable = 0;
  1205→    let totalReceivable = 0;
  1206→
  1207→    for (let i = 2; i >= 0; i--) {
  1208→      const mDate = new Date(now.getFullYear(), now.getMonth() - i, 1);
  1209→      const mLabel = months[mDate.getMonth()];
  1210→      const mStart = new Date(mDate.getFullYear(), mDate.getMonth(), 1);
  1211→      const mEnd = new Date(mDate.getFullYear(), mDate.getMonth() + 1, 0);
  1212→
  1213→      // Payable: total tax from invoices in that period
  1214→      const mInvoices = invoices.filter(inv => {
  1215→        const d = new Date(inv.createdAt as string || 0);
  1216→        return d >= mStart && d <= mEnd;
  1217→      });
  1218→      const payable = mInvoices.reduce((s, inv) => s + (inv.igst || 0) + (inv.cgst || 0) + (inv.sgst || 0) + (inv.cess || 0), 0);
  1219→
  1220→      // Receivable: ITC from filed returns
  1221→      const mReturns = returns.filter(r => {
  1222→        if (r.status !== 'filed') return false;
  1223→        const d = new Date(r.filedDate || r.createdAt as string || 0);
  1224→        return d >= mStart && d <= mEnd;
  1225→      });
  1226→      const receivable = mReturns.reduce((s, r) => s + (r.totalTax || 0), 0);
  1227→
  1228→      totalPayable += payable;
  1229→      totalReceivable += receivable;
  1230→
  1231→      data.push({ month: mLabel, payable, receivable });
  1232→    }
  1233→
  1234→    // Simple 3-month projection (avg of last 3 months)
  1235→    const avgPayable = totalPayable / 3;
  1236→    const avgReceivable = totalReceivable / 3;
  1237→    const projectionData: Array<{ month: string; payable: number; receivable: number }> = [];
  1238→    for (let i = 1; i <= 3; i++) {
  1239→      const mDate = new Date(now.getFullYear(), now.getMonth() + i, 1);
  1240→      projectionData.push({
  1241→        month: months[mDate.getMonth()],
  1242→        payable: Math.round(avgPayable * (1 + (Math.random() - 0.5) * 0.1)),
  1243→        receivable: Math.round(avgReceivable * (1 + (Math.random() - 0.5) * 0.1)),
  1244→      });
  1245→    }
  1246→
  1247→    return { chartData: data, netCashFlow: totalReceivable - totalPayable, projectionData };
  1248→  }, [invoices, returns]);
  1249→
  1250→  return (
  1251→    <Card className="h-full">
  1252→      <CardHeader className="pb-2">
  1253→        <div className="flex items-center justify-between">
  1254→          <div>
  1255→            <CardTitle className="text-sm font-semibold flex items-center gap-2">
  1256→              <BarChart3 className="h-4 w-4 text-emerald-500" />
  1257→              Cash Flow Forecast
  1258→            </CardTitle>
  1259→            <CardDescription className="text-xs mt-0.5">GST payable vs receivable</CardDescription>
  1260→          </div>
  1261→          <Button variant="ghost" size="sm" className="text-xs text-emerald-600 hover:text-emerald-700 h-7" onClick={onViewAnalytics}>
  1262→            View Analytics <ChevronRight className="h-3 w-3 ml-0.5" />
  1263→          </Button>
  1264→        </div>
  1265→      </CardHeader>
  1266→      <CardContent>
  1267→        {/* Net Cash Flow */}
  1268→        <div className="flex items-center justify-between mb-4">
  1269→          <div>
  1270→            <p className="text-[10px] font-semibold text-muted-foreground uppercase tracking-widest">Net Cash Flow (3m)</p>
  1271→            <p className={`text-xl font-bold tracking-tight mt-0.5 ${netCashFlow >= 0 ? 'text-emerald-600' : 'text-red-600'}`}>
  1272→              {netCashFlow >= 0 ? '+' : ''}{formatCurrency(netCashFlow)}
  1273→            </p>
  1274→          </div>
  1275→          <div className="flex items-center gap-3 text-[10px]">
  1276→            <div className="flex items-center gap-1.5">
  1277→              <span className="h-2.5 w-2.5 rounded-sm bg-red-500 opacity-80" />
  1278→              <span className="text-muted-foreground">Payable</span>
  1279→            </div>
  1280→            <div className="flex items-center gap-1.5">
  1281→              <span className="h-2.5 w-2.5 rounded-sm bg-emerald-500 opacity-80" />
  1282→              <span className="text-muted-foreground">Receivable</span>
  1283→            </div>
  1284→          </div>
  1285→        </div>
  1286→
  1287→        {/* Chart */}
  1288→        <div className="overflow-x-auto">
  1289→          <CashFlowBarChart data={chartData} />
  1290→        </div>
  1291→
  1292→        {/* Projection line */}
  1293→        <div className="mt-3 p-2 rounded-lg border border-dashed border-emerald-200 bg-emerald-50/30">
  1294→          <div className="flex items-center gap-1.5 mb-1.5">
  1295→            <TrendingUp className="h-3 w-3 text-emerald-500" />
  1296→            <span className="text-[10px] font-medium text-emerald-700">3-Month Projection</span>
  1297→          </div>
  1298→          <div className="flex items-center gap-4">
  1299→            {projectionData.map(p => (
  1300→              <div key={p.month} className="flex-1">
  1301→                <p className="text-[10px] font-semibold text-muted-foreground">{p.month}</p>
  1302→                <p className="text-[10px] text-red-600">↓ {formatCurrency(p.payable)}</p>
  1303→                <p className="text-[10px] text-emerald-600">↑ {formatCurrency(p.receivable)}</p>
  1304→              </div>
  1305→            ))}
  1306→          </div>
  1307→        </div>
  1308→      </CardContent>
  1309→    </Card>
  1310→  );
  1311→});
