   112→const RECENT_STORAGE_KEY = 'gstpilot-recent-commands';
   113→const FAVORITES_STORAGE_KEY = 'gstpilot-favorite-commands';
   114→const MAX_RECENT = 5;
   115→
   116→// ═══════════════════════════════════════════════════════════════════════════════
   117→// LOCAL STORAGE HELPERS
   118→// ═══════════════════════════════════════════════════════════════════════════════
   119→
   120→function loadRecent(): RecentItem[] {
   121→  try {
   122→    const raw = localStorage.getItem(RECENT_STORAGE_KEY);
   123→    return raw ? JSON.parse(raw) : [];
   124→  } catch {
   125→    return [];
   126→  }
   127→}
   128→
   129→function saveRecent(items: RecentItem[]) {
   130→  try {
   131→    localStorage.setItem(RECENT_STORAGE_KEY, JSON.stringify(items.slice(0, MAX_RECENT)));
   132→  } catch {
   133→    // ignore quota errors
   134→  }
   135→}
   136→
   137→function loadFavorites(): FavoriteItem[] {
   138→  try {
   139→    const raw = localStorage.getItem(FAVORITES_STORAGE_KEY);
   140→    return raw ? JSON.parse(raw) : [];
   141→  } catch {
   142→    return [];
   143→  }
   144→}
   145→
   146→function saveFavorites(items: FavoriteItem[]) {
   147→  try {
   148→    localStorage.setItem(FAVORITES_STORAGE_KEY, JSON.stringify(items));
   149→  } catch {
   150→    // ignore quota errors
   151→  }
   152→}
