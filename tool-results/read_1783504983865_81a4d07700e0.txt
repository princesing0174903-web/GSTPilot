   105→export function OracleWorkspace({
   106→  open, onClose, onNavigate, userName, firmName, gstin, userId, initialPrompt,
   107→}: OracleWorkspaceProps) {
   108→  const [store, setStore] = useState<ConversationStore>({ conversations: [], activeId: null });
   109→  // Portal guard: only render into document.body after mount to avoid SSR
   110→  // hydration mismatch (server has no document.body).
   111→  const [mounted, setMounted] = useState(false);
   112→  useEffect(() => setMounted(true), []);
   113→  const [messages, setMessages] = useState<OracleMessage[]>([]);
   114→  const [activeId, setActiveId] = useState<string | null>(null);
   115→  const [input, setInput] = useState('');
   116→  const [isStreaming, setIsStreaming] = useState(false);
   117→  const [lastEmotion, setLastEmotion] = useState<OracleMessage['emotion']>('helpful');
   118→  const [activeLanguage, setActiveLanguage] = useState<OracleMessage['language']>('english');
   119→  // Mobile: history drawer open state
   120→  const [historyOpen, setHistoryOpen] = useState(false);
   121→
   122→  // ── Oracle Context Engine™ — live dashboard metrics from Firestore are
   123→  //    forwarded to the API as context.dashboardMetrics so the model can
   124→  //    reason about the user's real compliance / return / invoice state.
   125→  const { metrics: dashboardMetrics } = useLiveDashboardMetrics();
   126→
   127→  const scrollRef = useRef<HTMLDivElement>(null);
   128→  const inputRef = useRef<HTMLTextAreaElement>(null);
   129→  const abortRef = useRef<AbortController | null>(null);
   130→  const userPinnedUpRef = useRef(false);
   131→  const streamingIdRef = useRef<string | null>(null);
   132→  // Tracks the last initialPrompt we auto-sent, so we never fire it twice.
   133→  const initialPromptSentRef = useRef<string | null>(null);
   134→
   135→  // ─── Load + migrate store on open ──────────────────────────────────────────
   136→  useEffect(() => {
   137→    if (!open) return;
   138→    const loaded = loadStore();
   139→    if (loaded.conversations.length === 0) {
   140→      // Migrate legacy single-conversation store if present.
   141→      try {
   142→        const raw = localStorage.getItem(LEGACY_KEY);
   143→        if (raw) {
   144→          const parsed = JSON.parse(raw) as OracleMessage[];
   145→          if (Array.isArray(parsed) && parsed.length > 0) {
   146→            const now = new Date().toISOString();
   147→            const conv: OracleConversation = {
   148→              id: cryptoId(),
   149→              title: deriveTitle(parsed),
   150→              messages: parsed.map((m) => (m.streaming ? { ...m, streaming: false } : m)),
   151→              createdAt: parsed[0]?.createdAt ?? now,
   152→              updatedAt: now,
   153→            };
   154→            const next: ConversationStore = { conversations: [conv], activeId: conv.id };
   155→            setStore(next);
   156→            setActiveId(conv.id);
   157→            setMessages(conv.messages);
   158→            saveStore(next);
   159→            return;
   160→          }
   161→        }
   162→      } catch {
   163→        /* ignore */
   164→      }
   165→      // Nothing to migrate — start fresh with one empty conversation.
   166→      const conv = newConversation();
   167→      const next: ConversationStore = { conversations: [conv], activeId: conv.id };
   168→      setStore(next);
   169→      setActiveId(conv.id);
   170→      setMessages([]);
   171→      saveStore(next);
   172→      return;
   173→    }
   174→    setStore(loaded);
   175→    const active = loaded.conversations.find((c) => c.id === loaded.activeId) ?? loaded.conversations[0];
   176→    if (active) {
   177→      setActiveId(active.id);
   178→      setMessages(active.messages);
   179→    }
   180→  }, [open]);
   181→
   182→  // ─── Persist active conversation whenever messages change ───────────────────
   183→  useEffect(() => {
   184→    if (!open || !activeId) return;
   185→    setStore((prev) => {
   186→      if (!prev.conversations.length) return prev;
   187→      const exists = prev.conversations.some((c) => c.id === activeId);
   188→      let conversations: OracleConversation[];
   189→      let nextActiveId = prev.activeId;
   190→      if (exists) {
   191→        conversations = prev.conversations.map((c) =>
   192→          c.id === activeId
   193→            ? {
   194→                ...c,
   195→                messages,
   196→                title: deriveTitle(messages) || c.title,
   197→                updatedAt: new Date().toISOString(),
   198→              }
   199→            : c,
   200→        );
   201→      } else {
   202→        // Active conversation doesn't exist yet — create it on the fly.
   203→        const conv: OracleConversation = {
   204→          id: activeId,
   205→          title: deriveTitle(messages) || 'New chat',
   206→          messages,
   207→          createdAt: new Date().toISOString(),
   208→          updatedAt: new Date().toISOString(),
   209→        };
   210→        conversations = [conv, ...prev.conversations];
   211→        nextActiveId = activeId;
   212→      }
   213→      const next = { conversations, activeId: nextActiveId };
   214→      saveStore(next);
   215→      return next;
   216→    });
   217→  }, [messages, open, activeId]);
   218→
   219→  // ─── Body scroll lock while open ───────────────────────────────────────────
   220→  useEffect(() => {
   221→    if (!open) return;
   222→    const prev = document.body.style.overflow;
   223→    document.body.style.overflow = 'hidden';
   224→    return () => {
   225→      document.body.style.overflow = prev;
   226→    };
   227→  }, [open]);
   228→
   229→  // ─── Escape to close ───────────────────────────────────────────────────────
   230→  useEffect(() => {
   231→    if (!open) return;
   232→    const onKey = (e: globalThis.KeyboardEvent) => {
   233→      if (e.key === 'Escape' && !isStreaming) onClose();
   234→    };
   235→    window.addEventListener('keydown', onKey);
   236→    return () => window.removeEventListener('keydown', onKey);
   237→  }, [open, isStreaming, onClose]);
   238→
   239→  // ─── Smart auto-scroll ─────────────────────────────────────────────────────
   240→  const handleScroll = useCallback(() => {
   241→    const el = scrollRef.current;
   242→    if (!el) return;
   243→    const distanceFromBottom = el.scrollHeight - el.scrollTop - el.clientHeight;
   244→    userPinnedUpRef.current = distanceFromBottom > 120;
   245→  }, []);
   246→
   247→  const scrollToBottom = useCallback((smooth = false) => {
   248→    const el = scrollRef.current;
   249→    if (!el) return;
   250→    el.scrollTo({ top: el.scrollHeight, behavior: smooth ? 'smooth' : 'auto' });
   251→  }, []);
   252→
   253→  useEffect(() => {
   254→    if (!userPinnedUpRef.current) {
   255→      requestAnimationFrame(() => scrollToBottom(false));
   256→    }
   257→  }, [messages, scrollToBottom]);
   258→
   259→  // ─── Conversation actions ──────────────────────────────────────────────────
   260→  const handleNewChat = useCallback(() => {
   261→    if (isStreaming) return;
   262→    const conv = newConversation();
   263→    setStore((prev) => {
   264→      const next = { conversations: [conv, ...prev.conversations], activeId: conv.id };
   265→      saveStore(next);
   266→      return next;
   267→    });
   268→    setActiveId(conv.id);
   269→    setMessages([]);
   270→    setInput('');
   271→    userPinnedUpRef.current = false;
   272→    setHistoryOpen(false);
   273→    requestAnimationFrame(() => inputRef.current?.focus());
   274→  }, [isStreaming]);
   275→
   276→  const handleSwitchConversation = useCallback(
   277→    (id: string) => {
   278→      if (isStreaming) return;
   279→      const conv = store.conversations.find((c) => c.id === id);
   280→      if (!conv) return;
   281→      setActiveId(id);
   282→      setMessages(conv.messages);
   283→      setStore((prev) => {
   284→        const next = { ...prev, activeId: id };
   285→        saveStore(next);
   286→        return next;
   287→      });
   288→      setHistoryOpen(false);
   289→      userPinnedUpRef.current = false;
   290→      requestAnimationFrame(() => scrollToBottom(false));
   291→    },
   292→    [isStreaming, store.conversations, scrollToBottom],
   293→  );
   294→
   295→  const handleDeleteConversation = useCallback(
   296→    (id: string) => {
   297→      if (isStreaming) return;
   298→      setStore((prev) => {
   299→        const filtered = prev.conversations.filter((c) => c.id !== id);
   300→        let nextActive = prev.activeId;
   301→        let nextMessages = messages;
   302→        if (prev.activeId === id) {
   303→          if (filtered.length > 0) {
   304→            nextActive = filtered[0].id;
   305→            nextMessages = filtered[0].messages;
   306→          } else {
   307→            const conv = newConversation();
   308→            filtered.unshift(conv);
   309→            nextActive = conv.id;
   310→            nextMessages = [];
   311→          }
   312→          setActiveId(nextActive);
   313→          setMessages(nextMessages);
   314→        }
   315→        const next = { conversations: filtered, activeId: nextActive };
   316→        saveStore(next);
   317→        return next;
   318→      });
   319→    },
   320→    [isStreaming, messages],
   321→  );
   322→
   323→  // ─── Send flow (preserved from previous implementation) ─────────────────────
   324→  const sendMessage = useCallback(
   325→    async (raw: string) => {
   326→      const text = raw.trim();
   327→      if (!text || isStreaming) return;
   328→
   329→      // ── Brand-question short-circuit ───────────────────────────────────────
   330→      const brand = detectBrandQuestion(text);
   331→      if (brand.matched && brand.answer) {
   332→        const userMsg: OracleMessage = {
   333→          id: cryptoId(),
   334→          role: 'user',
   335→          content: text,
   336→          language: detectLanguage(text),
   337→          createdAt: new Date().toISOString(),
   338→        };
   339→        const oracleMsg: OracleMessage = {
   340→          id: cryptoId(),
   341→          role: 'oracle',
   342→          content: brand.answer,
   343→          language: detectLanguage(text),
   344→          emotion: 'success',
   345→          createdAt: new Date().toISOString(),
   346→          streaming: false,
   347→          followUps: ['Who founded GSTPilot?', 'What can Oracle do?', 'GST kya hota hai?'],
   348→        };
   349→        setMessages((prev) => [...prev, userMsg, oracleMsg]);
   350→        setLastEmotion('success');
   351→        setActiveLanguage(detectLanguage(text));
   352→        setInput('');
   353→        if (inputRef.current) inputRef.current.style.height = 'auto';
   354→        userPinnedUpRef.current = false;
   355→        requestAnimationFrame(() => scrollToBottom(true));
   356→        return;
   357→      }
   358→
   359→      const userMsg: OracleMessage = {
   360→        id: cryptoId(),
   361→        role: 'user',
   362→        content: text,
   363→        language: detectLanguage(text),
   364→        createdAt: new Date().toISOString(),
   365→      };
   366→      const oracleId = cryptoId();
   367→      const oraclePlaceholder: OracleMessage = {
   368→        id: oracleId,
   369→        role: 'oracle',
   370→        content: '',
   371→        language: userMsg.language,
   372→        createdAt: new Date().toISOString(),
   373→        streaming: true,
   374→      };
   375→      setMessages((prev) => [...prev, userMsg, oraclePlaceholder]);
   376→      streamingIdRef.current = oracleId;
   377→      setIsStreaming(true);
   378→      setActiveLanguage(userMsg.language);
   379→      setInput('');
   380→      userPinnedUpRef.current = false;
   381→      if (inputRef.current) inputRef.current.style.height = 'auto';
   382→
   383→      const history: OracleChatRequest['messages'] = [
   384→        ...messages
   385→          .filter((m) => m.content.trim().length > 0)
   386→          .map((m) => ({ role: m.role, content: m.content })),
   387→        { role: 'user', content: text },
   388→      ];
   389→
   390→      const payload: OracleChatRequest = {
   391→        messages: history,
   392→        memory: {
   393→          userName,
   394→          firmName,
   395→          gstin,
   396→          userId,
   397→          preferredLanguage: activeLanguage,
   398→          recentTopics: messages
   399→            .filter((m) => m.role === 'user')
   400→            .slice(-4)
   401→            .map((m) => m.content.slice(0, 60)),
   402→        },
   403→        // Oracle Context Engine™ — forward live dashboard metrics so the
   404→        // server can ground its response in the user's actual compliance /
   405→        // return / invoice state (built into the system prompt as
   406→        // "LIVE DASHBOARD DATA (legacy)").
   407→        context: {
   408→          dashboardMetrics: {
   409→            totalClients: dashboardMetrics.totalClients,
   410→            activeClients: dashboardMetrics.activeClients,
   411→            totalInvoices: dashboardMetrics.totalInvoices,
   412→            totalTaxVolume: dashboardMetrics.totalTaxVolume,
   413→            filedReturns: dashboardMetrics.filedReturns,
   414→            pendingReturns: dashboardMetrics.pendingReturns,
   415→            overdueReturns: dashboardMetrics.overdueReturns,
   416→            readyToFile: dashboardMetrics.readyToFile,
   417→            criticalIssues: dashboardMetrics.criticalIssues,
   418→            warnings: dashboardMetrics.warnings,
   419→            averageHealthScore: dashboardMetrics.averageHealthScore,
   420→            matchPercentage: dashboardMetrics.matchPercentage,
   421→            riskPercentage: dashboardMetrics.riskPercentage,
   422→            documentsProcessed: dashboardMetrics.documentsProcessed,
   423→            extractionsPending: dashboardMetrics.extractionsPending,
   424→          },
   425→        },
   426→      };
   427→
   428→      const controller = new AbortController();
   429→      abortRef.current = controller;
   430→
   431→      try {
   432→        const res = await fetch('/api/oracle/chat', {
   433→          method: 'POST',
   434→          headers: { 'Content-Type': 'application/json' },
   435→          body: JSON.stringify(payload),
   436→          signal: controller.signal,
   437→        });
   438→        if (!res.ok || !res.body) {
   439→          throw new Error(`Request failed (${res.status})`);
   440→        }
   441→
   442→        const reader = res.body.getReader();
   443→        const decoder = new TextDecoder();
   444→        let buffer = '';
   445→        let acc = '';
   446→
   447→        while (true) {
   448→          const { done, value } = await reader.read();
   449→          if (done) break;
   450→          buffer += decoder.decode(value, { stream: true });
   451→
   452→          const lines = buffer.split('\n');
   453→          buffer = lines.pop() ?? '';
   454→
   455→          for (const rawLine of lines) {
   456→            const line = rawLine.trim();
   457→            if (!line || !line.startsWith('data:')) continue;
   458→            const data = line.slice(5).trim();
   459→            if (!data || data === '[DONE]') continue;
   460→            let chunk: OracleStreamChunk;
   461→            try {
   462→              chunk = JSON.parse(data) as OracleStreamChunk;
   463→            } catch {
   464→              continue;
   465→            }
   466→            if (chunk.language) setActiveLanguage(chunk.language);
   467→            if (chunk.token) {
   468→              acc += chunk.token;
   469→              setMessages((prev) =>
   470→                prev.map((m) => (m.id === oracleId ? { ...m, content: acc } : m)),
   471→              );
   472→            }
   473→            if (chunk.done) {
   474→              setMessages((prev) =>
   475→                prev.map((m) =>
   476→                  m.id === oracleId
   477→                    ? {
   478→                        ...m,
   479→                        content: acc,
   480→                        streaming: false,
   481→                        emotion: detectEmotion(acc),
   482→                        followUps: buildFollowUps(acc),
   483→                        actions: buildActionChips(acc),
   484→                      }
   485→                    : m,
   486→                ),
   487→              );
   488→              setLastEmotion(detectEmotion(acc));
   489→            }
   490→            if (chunk.error) {
   491→              setMessages((prev) =>
   492→                prev.map((m) =>
   493→                  m.id === oracleId
   494→                    ? {
   495→                        ...m,
   496→                        content:
   497→                          acc ||
   498→                          "I ran into a temporary issue reaching my reasoning service. Please try that again — your conversation is safe.",
   499→                        streaming: false,
   500→                        emotion: 'warning',
   501→                      }
   502→                    : m,
   503→                ),
   504→              );
   505→              setLastEmotion('warning');
   506→            }
   507→          }
   508→        }
   509→
   510→        setMessages((prev) =>
   511→          prev.map((m) =>
   512→            m.id === oracleId && m.streaming
   513→              ? {
   514→                  ...m,
   515→                  streaming: false,
   516→                  emotion: detectEmotion(m.content),
   517→                  followUps: buildFollowUps(m.content),
   518→                  actions: buildActionChips(m.content),
   519→                }
   520→              : m,
   521→          ),
   522→        );
   523→      } catch (err) {
   524→        const aborted = err instanceof DOMException && err.name === 'AbortError';
   525→        setMessages((prev) =>
   526→          prev.map((m) =>
   527→            m.id === oracleId
   528→              ? {
   529→                  ...m,
   530→                  streaming: false,
   531→                  content:
   532→                    m.content ||
   533→                    (aborted
   534→                      ? 'Stopped.'
   535→                      : "I had trouble reaching my reasoning service. Please try again in a moment."),
   536→                  emotion: aborted ? m.emotion : 'warning',
   537→                }
   538→              : m,
   539→          ),
   540→        );
   541→        if (!aborted) setLastEmotion('warning');
   542→      } finally {
   543→        setIsStreaming(false);
   544→        streamingIdRef.current = null;
   545→        abortRef.current = null;
   546→        requestAnimationFrame(() => inputRef.current?.focus());
   547→      }
   548→    },
   549→    [isStreaming, messages, userName, firmName, gstin, userId, activeLanguage, scrollToBottom, dashboardMetrics],
   550→  );
   551→
   552→  // ─── Stop streaming ────────────────────────────────────────────────────────
   553→  const handleStop = useCallback(() => {
   554→    abortRef.current?.abort();
   555→  }, []);
   556→
   557→  // ─── Input handling ────────────────────────────────────────────────────────
   558→  const handleInputChange = useCallback((e: React.ChangeEvent<HTMLTextAreaElement>) => {
   559→    setInput(e.target.value);
   560→    const el = e.target;
   561→    el.style.height = 'auto';
   562→    el.style.height = `${Math.min(el.scrollHeight, 160)}px`;
   563→  }, []);
   564→
   565→  const handleKeyDown = useCallback(
   566→    (e: KeyboardEvent<HTMLTextAreaElement>) => {
   567→      if (e.key === 'Enter' && !e.shiftKey) {
   568→        e.preventDefault();
   569→        sendMessage(input);
   570→      }
   571→    },
   572→    [input, sendMessage],
   573→  );
   574→
   575→  // ─── Focus input on open ───────────────────────────────────────────────────
   576→  useEffect(() => {
   577→    if (open) {
   578→      requestAnimationFrame(() => inputRef.current?.focus());
   579→    }
   580→  }, [open]);

  1281→interface ConversationGroups {
  1282→  today: OracleConversation[];
  1283→  yesterday: OracleConversation[];

  1288→function groupConversations(convos: OracleConversation[]): ConversationGroups {
  1289→  const now = new Date();
  1290→  const startOfToday = new Date(now.getFullYear(), now.getMonth(), now.getDate()).getTime();
  1291→  const startOfYesterday = startOfToday - 86_400_000;
  1292→  const startOf7Days = startOfToday - 7 * 86_400_000;
  1293→
  1294→  const today: OracleConversation[] = [];
  1295→  const yesterday: OracleConversation[] = [];
  1296→  const prev7: OracleConversation[] = [];
  1297→  const older: OracleConversation[] = [];
  1298→
  1299→  const sorted = [...convos].sort(
  1300→    (a, b) => new Date(b.updatedAt).getTime() - new Date(a.updatedAt).getTime(),
  1301→  );
  1302→
  1303→  for (const c of sorted) {
  1304→    const t = new Date(c.updatedAt).getTime();
  1305→    if (isNaN(t)) {
  1306→      older.push(c);
  1307→    } else if (t >= startOfToday) {
  1308→      today.push(c);
  1309→    } else if (t >= startOfYesterday) {
  1310→      yesterday.push(c);
  1311→    } else if (t >= startOf7Days) {
  1312→      prev7.push(c);
  1313→    } else {
  1314→      older.push(c);
  1315→    }
  1316→  }
  1317→
  1318→  return { today, yesterday, prev7, older };
  1319→}

  1321→/** Generate contextual follow-up chips from the response content. */
  1322→function buildFollowUps(content: string): string[] | undefined {
  1323→  if (!content || content.length < 30) return undefined;
  1324→  const lower = content.toLowerCase();
  1325→
  1326→  const pool: string[] = [];
  1327→  if (/(gstr-3b|3b)/.test(lower)) pool.push('GSTR-3B की last date क्या है?', 'Late fee कितनी लगेगी?');
  1328→  if (/(gstr-1|gstr 1)/.test(lower)) pool.push('GSTR-1 कैसे file करें?', 'B2B और B2C में अंतर?');
  1329→  if (/(itc|input tax credit)/.test(lower)) pool.push('ITC claim कैसे करें?', 'Blocked ITC के rules?');
  1330→  if (/(late fee|penalty|overdue)/.test(lower)) pool.push('Late fee waiver मिल सकती है?', 'How to avoid this next time?');
  1331→  if (/(cash flow|collections)/.test(lower)) pool.push('Receivables कैसे recover करें?', 'Working capital optimize करें');
  1332→  if (/(reverse charge|rcm)/.test(lower)) pool.push('RCM किन पर लागू है?', 'How to report RCM in GSTR-3B?');
  1333→  if (/(refund)/.test(lower)) pool.push('Refund process क्या है?', 'Refund timeline कितनी है?');
  1334→
  1335→  pool.push('GST kya hota hai?', 'Explain ITC rules', 'How can Oracle help me daily?');
  1336→
  1337→  const seen = new Set<string>();
  1338→  const picks: string[] = [];
  1339→  for (const p of pool) {
  1340→    if (seen.has(p)) continue;
  1341→    seen.add(p);
  1342→    picks.push(p);
  1343→    if (picks.length >= 3) break;
  1344→  }
  1345→  return picks;
