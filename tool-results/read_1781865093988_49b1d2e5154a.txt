   365→   365→            mode,
   366→   366→            workingMemory: wm,
   367→   367→            history: priorMessages,
   368→   368→            investigationHints: investigationHints.length > 0 ? investigationHints : undefined,
   369→   369→          }),
   370→   370→          signal: controller.signal,
   371→   371→        });
   372→   372→
   373→   373→        if (!res.ok || !res.body) {
   374→   374→          throw new Error(`Oracle request failed: ${res.status}`);
   375→   375→        }
   376→   376→
   377→   377→        const reader = res.body.getReader();
   378→   378→        const decoder = new TextDecoder();
   379→   379→        let buffer = '';
   380→   380→        let currentParts: OracleResponseParts = { ...EMPTY_PARTS };
   381→   381→        let finalMode: OracleModeId = mode;
   382→   382→
   383→   383→        const patch = (p: Partial<OracleMessage>) => {
   384→   384→          setActiveConv((prev) => {
   385→   385→            if (!prev) return prev;
   386→   386→            return updateLastOracleMessage(prev, p);
   387→   387→          });
   388→   388→        };
   389→   389→
   390→   390→        const flushBuffer = () => {
   391→   391→          const events = buffer.split('\n\n');
   392→   392→          buffer = events.pop() ?? '';
   393→   393→          for (const evt of events) {
   394→   394→            const lines = evt.split('\n');
   395→   395→            let eventName = '';
   396→   396→            let dataStr = '';
   397→   397→            for (const line of lines) {
   398→   398→              if (line.startsWith('event: ')) eventName = line.slice(7).trim();
   399→   399→              else if (line.startsWith('data: ')) dataStr = line.slice(6);
   400→   400→            }
   401→   401→            if (!eventName || !dataStr) continue;
   402→   402→            try {
   403→   403→              const data = JSON.parse(dataStr);
   404→   404→              handleEvent(eventName, data);
   405→   405→            } catch {
   406→   406→              // Skip malformed event.
   407→   407→            }
   408→   408→          }
   409→   409→        };
   410→   410→
   411→   411→        const handleEvent = (eventName: string, data: unknown) => {
   412→   412→          const d = data as Record<string, unknown>;
   413→   413→          if (eventName === 'phase') {
   414→   414→            const phase = d.phase as OraclePhase;
   415→   415→            patch({ phase });
   416→   416→          } else if (eventName === 'part') {
   417→   417→            const part = d.part as string;
   418→   418→            const textChunk = (d.text as string) ?? '';
   419→   419→            const isItem = d.item === true;
   420→   420→            const source = d.source as { kind: string; title: string; snippet: string } | undefined;
   421→   421→            const card = d.card as { kind: string; title: string; description: string; impact?: string } | undefined;
   422→   422→
   423→   423→            if (part === 'keyInsight') {
   424→   424→              currentParts.keyInsight += textChunk;
   425→   425→            } else if (part === 'analysis') {
   426→   426→              currentParts.analysis += textChunk;
   427→   427→            } else if (part === 'recommendedActions') {
   428→   428→              if (isItem) currentParts.recommendedActions.push(textChunk);
   429→   429→              else {
   430→   430→                const last = currentParts.recommendedActions.length - 1;
   431→   431→                if (last >= 0) currentParts.recommendedActions[last] += textChunk;
   432→   432→                else currentParts.recommendedActions.push(textChunk);
   433→   433→              }
   434→   434→            } else if (part === 'potentialRisks') {
   435→   435→              if (isItem) currentParts.potentialRisks.push(textChunk);
   436→   436→              else {
   437→   437→                const last = currentParts.potentialRisks.length - 1;
   438→   438→                if (last >= 0) currentParts.potentialRisks[last] += textChunk;
   439→   439→                else currentParts.potentialRisks.push(textChunk);
   440→   440→              }
   441→   441→            } else if (part === 'nextBestStep') {
   442→   442→              currentParts.nextBestStep += textChunk;
   443→   443→            } else if (part === 'investigation') {
   444→   444→              if (!currentParts.investigation) currentParts.investigation = [];
   445→   445→              if (isItem) currentParts.investigation.push(textChunk);
   446→   446→              else {
   447→   447→                const last = currentParts.investigation.length - 1;
   448→   448→                if (last >= 0) currentParts.investigation[last] += textChunk;
   449→   449→                else currentParts.investigation.push(textChunk);
   450→   450→              }
   451→   451→            } else if (part === 'sources') {
   452→   452→              if (source) {
   453→   453→                currentParts.sources.push({
   454→   454→                  kind: (source.kind as OracleResponseParts['sources'][number]['kind']) ?? 'business-data',
   455→   455→                  title: source.title,
   456→   456→                  snippet: source.snippet,
   457→   457→                });
   458→   458→              }
   459→   459→            } else if (part === 'followUps') {
   460→   460→              if (isItem) currentParts.followUps.push(textChunk);
   461→   461→              else {
   462→   462→                const last = currentParts.followUps.length - 1;
   463→   463→                if (last >= 0) currentParts.followUps[last] += textChunk;
   464→   464→                else currentParts.followUps.push(textChunk);
   465→   465→              }
   466→   466→            } else if (part === 'actionCards') {
   467→   467→              if (card) {
   468→   468→                const kind = (card.kind as OracleActionKind) ?? 'create-report';
   469→   469→                currentParts.actionCards.push({
   470→   470→                  kind,
   471→   471→                  title: card.title,
   472→   472→                  description: card.description,
   473→   473→                  impact: card.impact,
   474→   474→                });
   475→   475→              }
   476→   476→            }
   477→   477→
   478→   478→            patch({ parts: { ...currentParts }, phase: 'streaming' });
   479→   479→          } else if (eventName === 'done') {
   480→   480→            finalMode = (d.mode as OracleModeId) ?? mode;
   481→   481→            // If the LLM didn't propose any action cards, fall back to defaults
   482→   482→            // so the "always 3 cards" contract holds.
   483→   483→            if (currentParts.actionCards.length === 0) {
   484→   484→              currentParts.actionCards = DEFAULT_FALLBACK_CARDS;
   485→   485→            }
   486→   486→            patch({
   487→   487→              parts: { ...currentParts },
   488→   488→              phase: 'done',
   489→   489→              mode: finalMode,
   490→   490→              actions: currentParts.recommendedActions.map((s) => mapActionKind(s)),
   491→   491→            });
   492→   492→          } else if (eventName === 'error') {
   493→   493→            patch({
   494→   494→              phase: 'error',
   495→   495→              error: (d.message as string) ?? 'Unknown error',
   496→   496→            });
   497→   497→          }
   498→   498→        };
   499→   499→
   500→   500→        while (true) {
   501→   501→          const { done, value } = await reader.read();
   502→   502→          if (done) break;
   503→   503→          buffer += decoder.decode(value, { stream: true });
   504→   504→          flushBuffer();
   505→   505→        }
   506→   506→        flushBuffer();
   507→   507→      } catch (err) {
   508→   508→        if ((err as Error).name === 'AbortError') {
   509→   509→          setActiveConv((prev) =>
   510→   510→            prev ? updateLastOracleMessage(prev, { phase: 'done' }) : prev,
   511→   511→          );
   512→   512→        } else {
   513→   513→          setActiveConv((prev) =>
   514→   514→            prev
   515→   515→              ? updateLastOracleMessage(prev, {
   516→   516→                  phase: 'error',
   517→   517→                  error: err instanceof Error ? err.message : 'Unknown error',
   518→   518→                })
   519→   519→              : prev,
   520→   520→          );
   521→   521→        }
   522→   522→      } finally {
   523→   523→        setIsStreaming(false);
   524→   524→        abortRef.current = null;
   525→   525→        refreshConversationList();
   526→   526→      }
   527→   527→    },
   528→   528→    [activeConv, isStreaming, userMemory, businessMemory, pinnedInsights, refreshConversationList],
   529→   529→  );

   555→   555→        return;
   556→   556→      }

   561→   561→      });

   568→   568→      const stepDuration = Math.max(400, Math.round(meta.durationMs / Math.max(totalSteps, 1)));
   569→   569→      let currentStepIdx = 0;
   570→   570→      let progress = 0;
   571→   571→
   572→   572→      const tick = () => {
   573→   573→        // Advance progress within the current step.
   574→   574→        const stepIncrement = Math.round(100 / Math.max(totalSteps, 1));
   575→   575→        progress = Math.min(100, progress + Math.max(8, Math.round(stepIncrement / 3)));
   576→   576→
   577→   577→        // If we've crossed into the next step boundary, advance the step.
   578→   578→        const expectedStep = Math.min(totalSteps - 1, Math.floor((progress / 100) * totalSteps));
   579→   579→        if (expectedStep > currentStepIdx) {

   584→   584→              | 'done'
   585→   585→              | 'active'
   586→   586→              | 'pending',
   587→   587→          }));

   591→   591→            prev.map((t) => (t.id === task.id ? { ...t, steps: newSteps, progress } : t)),
   592→   592→          );
   593→   593→        } else {
   594→   594→          updateTask(task.id, { progress });

   596→   596→        }
   597→   597→
   598→   598→        if (progress >= 100) {
   599→   599→          // Complete the task.
   600→   600→          const result = formatActionResult(card.kind, {
   601→   601→            count: Math.floor(Math.random() * 5) + 2,
   602→   602→            amount: Math.floor(Math.random() * 50 + 10) * 10000,
   603→   603→          });

   605→   605→          updateTask(task.id, {
   606→   606→            status: 'completed',
   607→   607→            progress: 100,
   608→   608→            steps: finalSteps,
   609→   609→            result,
   610→   610→          });
   611→   611→          setTasks((prev) =>
   612→   612→            prev.map((t) =>
   613→   613→              t.id === task.id
   614→   614→                ? { ...t, status: 'completed', progress: 100, steps: finalSteps, result }
   615→   615→                : t,
   616→   616→            ),
   617→   617→          );
   618→   618→          setAgentStatus('completed');
   619→   619→          setWorkingTitle(undefined);
   620→   620→          // Return to idle after a brief celebration.
   621→   621→          setTimeout(() => setAgentStatus('idle'), 3000);
   622→   622→          return;
   623→   623→        }
   624→   624→
   625→   625→        // Schedule the next tick.
   626→   626→        setTimeout(tick, stepDuration / 3);
   627→   627→      };
   628→   628→
   629→   629→      // Kick off the execution loop.
   630→   630→      setTimeout(tick, stepDuration / 3);

   957→   957→  if (l.includes('forecast') || l.includes('predict')) return 'generate-forecast';
   958→   958→  if (l.includes('notice') || l.includes('scn')) return 'create-notice';

   961→   961→  if (l.includes('export') || l.includes('pdf')) return 'export-pdf';
   962→   962→  return 'create-report';
