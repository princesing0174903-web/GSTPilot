    66→    66→export async function POST(req: NextRequest) {
    67→    67→  let question = ''
    68→    68→  let context = ''
    69→    69→  try {
    70→    70→    const body = await req.json()
    71→    71→    question = (body.question || '').trim()
    72→    72→    context = (body.context || '').trim()
    73→    73→    const conversationHistory: Array<{ role: string; content: string }> =
    74→    74→      body.conversationHistory || []
    75→    75→
    76→    76→    if (!question) {
    77→    77→      return NextResponse.json({ error: 'Question is required' }, { status: 400 })
    78→    78→    }
    79→    79→
    80→    80→    // 1) Deterministic, fast intent + actions + thinking steps
    81→    81→    const intent = detectIntent(question)
    82→    82→    const thinkingSteps = generateThinkingSteps(intent)
    83→    83→    const detectedActions = detectActions(question, intent)
    84→    84→
    85→    85→    // 2) Determine whether real data is available in context.
    86→    86→    //    The frontend builds context as a structured string starting with
    87→    87→    //    "LIVE DATA SNAPSHOT" and a list of "- key: value" lines.
    88→    88→    const hasLiveData = contextHasLiveData(context)
    89→    89→
    90→    90→    // 3) Build sources from ACTUAL context content (no fabrication)
    91→    91→    const sources = hasLiveData ? buildSources(context) : EMPTY_CONTEXT_SOURCES
    92→    92→
    93→    93→    // 4) Generate the answer via LLM (or fallback)
    94→    94→    let answer: string
    95→    95→    if (!hasLiveData) {
    96→    96→      // Skip LLM call entirely when no live data — guarantees no fabrication
    97→    97→      answer = EMPTY_CONTEXT_ANSWER
    98→    98→    } else {
    99→    99→      try {
   100→   100→        answer = await generateLLMAnswer(question, context, conversationHistory)
   101→   101→      } catch (llmError) {
   102→   102→        console.error(
   103→   103→          '[Oracle] LLM error:',
   104→   104→          llmError instanceof Error ? llmError.message : llmError,
   105→   105→        )
   106→   106→        answer = generateContextualFallback(question, context, intent)
   107→   107→      }
   108→   108→      if (!answer || !answer.trim()) {
   109→   109→        answer = generateContextualFallback(question, context, intent)
   110→   110→      }
   111→   111→    }
   112→   112→
   113→   113→    // 5) Extract structured insights from the LLM answer (or derive from context)
   114→   114→    const insights = hasLiveData
   115→   115→      ? extractInsights(answer, context, intent)
   116→   116→      : [
   117→   117→          {
   118→   118→            text: 'Connect your business data to unlock AI insights.',
   119→   119→            tone: 'neutral' as InsightTone,
   120→   120→          },
   121→   121→        ]
   122→   122→
   123→   123→    // 6) Suggested follow-ups based on intent (and data availability)
   124→   124→    const suggestedPrompts = hasLiveData
   125→   125→      ? suggestFollowUps(intent, question)
   126→   126→      : [
   127→   127→          'Show pending returns',
   128→   128→          'Which clients are risky?',
   129→   129→          'What revenue will I make next month?',
   130→   130→        ]
   131→   131→
   132→   132→    const responseBody: OracleResponse = {
   133→   133→      answer,
   134→   134→      insights,
   135→   135→      sources,
   136→   136→      actions: detectedActions,
   137→   137→      suggestedPrompts,
   138→   138→      intent,
   139→   139→      thinkingSteps,
   140→   140→    }
   141→   141→
   142→   142→    return NextResponse.json(responseBody)
   143→   143→  } catch (error: unknown) {
   144→   144→    console.error(
   145→   145→      '[Oracle] Fatal:',
   146→   146→      error instanceof Error ? error.message : error,
   147→   147→    )
   148→   148→    // Always return the structured shape — frontend never has to guard
   149→   149→    const fallbackResponse: OracleResponse = {
   150→   150→      answer: EMPTY_CONTEXT_ANSWER,
   151→   151→      insights: [
   152→   152→        {
   153→   153→          text: 'I had trouble reaching my reasoning engine. Please try again.',
   154→   154→          tone: 'warning',
   155→   155→        },
   156→   156→      ],
   157→   157→      sources: EMPTY_CONTEXT_SOURCES,
   158→   158→      actions: [],
   159→   159→      suggestedPrompts: [
   160→   160→        'Show pending returns',
   161→   161→        'Which clients are risky?',
   162→   162→        'What revenue will I make next month?',
   163→   163→      ],
   164→   164→      intent: 'fallback',
   165→   165→      thinkingSteps: [
   166→   166→        { label: 'Thinking...', duration: 400 },
   167→   167→        { label: 'Generating answer...', duration: 500 },
   168→   168→      ],
   169→   169→    }
   170→   170→    return NextResponse.json(fallbackResponse)
   171→   171→  }
   172→   172→}

   227→   227→function detectIntent(question: string): string {
   228→   228→  const q = question.toLowerCase()

   234→   234→    q.includes('run my business') ||
   235→   235→    q.includes('run my company')
   236→   236→  )
   237→   237→    return 'run_automation'
   238→   238→  if (
   239→   239→    q.includes('overload') ||
   240→   240→    q.includes('overloaded') ||
   241→   241→    q.includes('workload') ||
   242→   242→    q.includes('capacity') ||
   243→   243→    q.includes('bandwidth')
   244→   244→  )
   245→   245→    return 'workload'
   246→   246→  if (
   247→   247→    q.includes('revenue') ||
   248→   248→    q.includes('income') ||
   249→   249→    q.includes('earn') ||
   250→   250→    q.includes('next month') ||
   251→   251→    q.includes('forecast') ||
   252→   252→    q.includes('predict')
   253→   253→  )
   254→   254→    return 'revenue_forecast'
   255→   255→  if (
   256→   256→    q.includes('collection') ||
   257→   257→    q.includes('drop') ||
   258→   258→    q.includes('decline') ||
   259→   259→    q.includes('fell') ||
   260→   260→    q.includes('why did')
   261→   261→  )
   262→   262→    return 'collection_analysis'
   263→   263→  if (q.includes('risk') || q.includes('risky') || q.includes('health') || q.includes('at risk'))
   264→   264→    return 'risk_analysis'
   265→   265→  if (
   266→   266→    q.includes('pending') ||
   267→   267→    q.includes('overdue') ||
   268→   268→    q.includes('filing') ||
   269→   269→    q.includes('gstr') ||
   270→   270→    q.includes('return')
   271→   271→  )
   272→   272→    return 'pending_returns'
   273→   273→  if (

   275→   275→    q.includes('payment') ||
   276→   276→    q.includes('outstanding') ||
   277→   277→    q.includes('collect')

   281→   281→  if (q.includes('compliance') || q.includes('score') || q.includes('deadline')) return 'compliance'
   282→   282→  if (q.includes('client') || q.includes('customer')) return 'client_overview'
   283→   283→  if (q.includes('today') || q.includes('priority') || q.includes('should i') || q.includes('to do'))
   284→   284→    return 'daily_priority'
   285→   285→  if (q.includes('compare') || q.includes('last month') || q.includes('vs') || q.includes('trend'))
   286→   286→    return 'comparison'
   287→   287→  if (q.includes('task') || q.includes('reminder') || q.includes('notify')) return 'task_management'
   288→   288→  if (q.includes('report') || q.includes('summary') || q.includes('statement')) return 'report'
   289→   289→  return 'general'
   290→   290→}

   524→   524→/**
   525→   525→ * The frontend composes a context string starting with "LIVE DATA SNAPSHOT"
   526→   526→ * followed by "- key: value" lines. We parse those lines to surface the
   527→   527→ * real sources + record counts the LLM consulted. NO fabrication — if a
   528→   528→ * section is missing or zero, the source is omitted.

   530→   530→function buildSources(context: string): OracleSource[] {
   531→   531→  const sources: OracleSource[] = []
   532→   532→
   533→   533→  const clients = parseCount(context, /-?\s*Total clients:\s*(\d+)/i)
   534→   534→  if (clients > 0) {
   535→   535→    sources.push({ name: 'Clients', count: clients, icon: 'users' })
   536→   536→  }
   537→   537→
   538→   538→  const invoices = parseCount(context, /-?\s*Active invoices:\s*(\d+)/i)
   539→   539→  if (invoices > 0) {
   540→   540→    sources.push({ name: 'Invoices', count: invoices, icon: 'file-text' })
   541→   541→  }
   542→   542→
   543→   543→  const filed = parseCount(context, /-?\s*Filed returns:\s*(\d+)/i)
   544→   544→  // Frontend emits "- Pending returns: 3 GSTR-1, 2 GSTR-3B"
   545→   545→  // The count comes BEFORE the return-type label, so capture the leading digit.
   546→   546→  const gstr1Pending = parseCount(context, /-?\s*Pending returns:\s*(\d+)\s*GSTR-1/i)
   547→   547→  const gstr3bPending = parseCount(context, /GSTR-1,?\s*(\d+)\s*GSTR-3B/i)
   548→   548→  const totalReturns = filed + gstr1Pending + gstr3bPending
   549→   549→  if (totalReturns > 0) {
   550→   550→    sources.push({ name: 'GST Returns', count: totalReturns, icon: 'receipt' })
   551→   551→  }
   552→   552→
   553→   553→  const unread = parseCount(context, /-?\s*Unread notifications:\s*(\d+)/i)
   554→   554→  if (unread > 0) {
   555→   555→    sources.push({ name: 'Notifications', count: unread, icon: 'bell' })
   556→   556→  }
   557→   557→
   558→   558→  const activities = parseCount(context, /-?\s*Recent activities:\s*(\d+)/i)
   559→   559→  if (activities > 0) {
   560→   560→    sources.push({ name: 'Activities', count: activities, icon: 'activity' })
   561→   561→  }
   562→   562→
   563→   563→  // If we somehow couldn't parse anything but the context clearly has live data,
   564→   564→  // return a single generic source so the UI has something to show.
   565→   565→  if (sources.length === 0 && contextHasLiveData(context)) {
   566→   566→    sources.push({ name: 'Business Data', count: 0, icon: 'database' })
   567→   567→  }
   568→   568→
   569→   569→  return sources
   570→   570→}

   572→   572→function parseCount(context: string, pattern: RegExp): number {
   573→   573→  const match = context.match(pattern)
   574→   574→  if (!match || !match[1]) return 0
   575→   575→  const n = parseInt(match[1], 10)
   576→   576→  return Number.isFinite(n) ? n : 0
   577→   577→}

   579→   579→function contextHasLiveData(context: string): boolean {
   580→   580→  if (!context || context.length < 60) return false
   581→   581→  // The frontend always emits this header when live data is available
   582→   582→  if (!/LIVE DATA SNAPSHOT/i.test(context)) return false
   583→   583→  // If the snapshot explicitly says data unavailable, treat as empty
   584→   584→  if (/Data currently unavailable/i.test(context)) return false
   585→   585→  // Must have at least one "- key: value" data line beyond the header
   586→   586→  const dataLines = context
   587→   587→    .split('\n')
   588→   588→    .filter((l) => /^\s*-\s+\w/.test(l))
   589→   589→  return dataLines.length >= 2
   590→   590→}

   592→   592→// ─── Insight Extractor (parses LLM answer for 2-4 bullets) ────────────────────
   593→   593→
   594→   594→/**
   595→   595→ * Extracts 2-4 insight bullets from the LLM answer. Falls back to
   596→   596→ * context-derived insights when the answer has no parseable bullets.
   597→   597→ */
   598→   598→function extractInsights(
   599→   599→  answer: string,
   600→   600→  context: string,
   601→   601→  intent: string,
   602→   602→): OracleInsight[] {
   603→   603→  // Parse bullet/numbered lines from the answer body
   604→   604→  const rawLines = answer
   605→   605→    .split('\n')
   606→   606→    .map((l) => l.trim())
   607→   607→    .filter(Boolean)
   608→   608→
   609→   609→  const bullets: string[] = []
   610→   610→  for (const line of rawLines) {
   611→   611→    let text: string | null = null
   612→   612→    if (line.startsWith('• ') || line.startsWith('•')) {
   613→   613→      text = line.replace(/^•\s*/, '').trim()
   614→   614→    } else if (line.startsWith('- ')) {
   615→   615→      text = line.replace(/^-\s*/, '').trim()
   616→   616→    } else if (/^\d+\.\s/.test(line)) {
   617→   617→      text = line.replace(/^\d+\.\s*/, '').trim()
   618→   618→    }
   619→   619→    if (text && text.length >= 12 && text.length <= 240) {
   620→   620→      // Strip markdown bold/italic markers for cleaner insight text
   621→   621→      const cleaned = text.replace(/\*\*/g, '').replace(/__/g, '').trim()
   622→   622→      if (cleaned.length >= 12) bullets.push(cleaned)
   623→   623→    }
   624→   624→    if (bullets.length >= 4) break
   625→   625→  }
   626→   626→
   627→   627→  if (bullets.length >= 2) {
   628→   628→    return bullets.slice(0, 4).map((text) => ({ text, tone: detectTone(text) }))
   629→   629→  }
   630→   630→
   631→   631→  // Fallback: derive insights from real context numbers
   632→   632→  const derived = deriveInsightsFromContext(context, intent)
   633→   633→  if (derived.length >= 1) {
   634→   634→    // If we got 1 bullet from the answer, combine with 1 derived for 2 total
   635→   635→    const combined: OracleInsight[] = []
   636→   636→    if (bullets.length === 1) {
   637→   637→      combined.push({ text: bullets[0], tone: detectTone(bullets[0]) })
   638→   638→    }
   639→   639→    for (const d of derived) {
   640→   640→      if (combined.length >= 2) break
   641→   641→      if (!combined.some((c) => c.text === d.text)) combined.push(d)
   642→   642→    }
   643→   643→    if (combined.length >= 1) return combined.slice(0, 4)
   644→   644→  }
   645→   645→
   646→   646→  // Last-resort fallback: synthesize a single neutral insight from the answer
   647→   647→  const firstSentence = answer
   648→   648→    .split(/(?<=[.!?])\s+/)[0]
   649→   649→    ?.replace(/[*#]/g, '')
   650→   650→    .trim()
   651→   651→  if (firstSentence && firstSentence.length >= 12) {
   652→   652→    return [{ text: firstSentence.slice(0, 240), tone: 'neutral' }]
   653→   653→  }
   654→   654→  return [{ text: 'Insights will appear here once live data is connected.', tone: 'neutral' }]
   655→   655→}
   656→   656→
   657→   657→function detectTone(text: string): InsightTone {
   658→   658→  const t = text.toLowerCase()
   659→   659→  // Warning signals — risk/decline/overdue keywords
   660→   660→  if (
   661→   661→    /\b(overdue|drop|decline|declined|fell|missed|urgent|risk|risky|late|below|fail|failed|exposed|shortfall|delay|delayed|warning|alert|critical|breach|non-compliant|mismatch|gap)\b/.test(
   662→   662→      t,
   663→   663→    )
   664→   664→  ) {
   665→   665→    return 'warning'
   666→   666→  }
   667→   667→  // Positive signals — growth/improvement keywords
   668→   668→  if (
   669→   669→    /\b(growth|grew|increase|increased|up|healthy|profit|saved|completed|filed|recovered|improved|improvement|surplus|above|strong|stable|on track|on-time)\b/.test(
   670→   670→      t,
   671→   671→    )
   672→   672→  ) {
   673→   673→    return 'positive'
   674→   674→  }
   675→   675→  return 'neutral'
   676→   676→}

   678→   678→function deriveInsightsFromContext(context: string, intent: string): OracleInsight[] {
   679→   679→  const insights: OracleInsight[] = []
   680→   680→
   681→   681→  const clients = parseCount(context, /-?\s*Total clients:\s*(\d+)/i)
   682→   682→  const activeClients = parseCount(context, /-?\s*Active clients:\s*(\d+)/i)
   683→   683→  const invoices = parseCount(context, /-?\s*Active invoices:\s*(\d+)/i)
   684→   684→  const overdueInvoices = parseCount(context, /-?\s*Overdue invoices:\s*(\d+)/i)
   685→   685→  const filed = parseCount(context, /-?\s*Filed returns:\s*(\d+)/i)
   686→   686→  const gstr1Pending = parseCount(context, /-?\s*Pending returns:\s*(\d+)\s*GSTR-1/i)
   687→   687→
   688→   688→  // High-risk clients line (text after the colon)
   689→   689→  const riskMatch = context.match(/-?\s*High-risk clients:\s*(.+)/i)
   690→   690→  const riskLine = riskMatch?.[1]?.trim()
   691→   691→  const highRiskCount = riskLine
   692→   692→    ? (riskLine.match(/,/g)?.length ?? 0) + (riskLine.length > 0 ? 1 : 0)
   693→   693→    : 0
   694→   694→
   695→   695→  // Top client line
   696→   696→  const topMatch = context.match(/-?\s*Top 5 clients by revenue:\s*(.+)/i)
   697→   697→  const topLine = topMatch?.[1]?.trim()
   698→   698→
   699→   699→  if (intent === 'risk_analysis' && highRiskCount > 0 && riskLine) {
   700→   700→    insights.push({
   701→   701→      text: `${highRiskCount} client${highRiskCount > 1 ? 's' : ''} flagged high-risk — ${riskLine.slice(0, 140)}.`,
   702→   702→      tone: 'warning',
   703→   703→    })
   704→   704→  }
   705→   705→
   706→   706→  if (intent === 'pending_returns' && (gstr1Pending > 0 || filed > 0)) {
   707→   707→    insights.push({
   708→   708→      text: `${gstr1Pending} GSTR-1 ${gstr1Pending === 1 ? 'return' : 'returns'} pending, ${filed} already filed.`,
   709→   709→      tone: gstr1Pending > 0 ? 'warning' : 'positive',
   710→   710→    })
   711→   711→  }
   712→   712→
   713→   713→  if (
   714→   714→    (intent === 'invoice_collection' || intent === 'collection_analysis') &&
   715→   715→    (overdueInvoices > 0 || invoices > 0)
   716→   716→  ) {
   717→   717→    insights.push({
   718→   718→      text: `${overdueInvoices} overdue invoice${overdueInvoices === 1 ? '' : 's'} out of ${invoices} active.`,
   719→   719→      tone: overdueInvoices > 0 ? 'warning' : 'positive',
   720→   720→    })
   721→   721→  }
   722→   722→
   723→   723→  if (intent === 'client_overview' && clients > 0) {
   724→   724→    insights.push({
   725→   725→      text: `${clients} total client${clients === 1 ? '' : 's'} (${activeClients} active).`,
   726→   726→      tone: 'neutral',
   727→   727→    })
   728→   728→  }
   729→   729→
   730→   730→  if (intent === 'revenue_forecast' && topLine) {
   731→   731→    insights.push({
   732→   732→      text: `Top revenue clients: ${topLine.slice(0, 140)}.`,
   733→   733→      tone: 'positive',
   734→   734→    })
   735→   735→  }
   736→   736→
   737→   737→  if (intent === 'compliance' && filed > 0) {
   738→   738→    insights.push({
   739→   739→      text: `${filed} return${filed === 1 ? '' : 's'} filed so far this period.`,
   740→   740→      tone: 'positive',
   741→   741→    })
   742→   742→  }
   743→   743→
   744→   744→  return insights
   745→   745→}
