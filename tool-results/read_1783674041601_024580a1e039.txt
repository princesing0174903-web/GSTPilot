    56→/**
    57→ * Returns the current organization id (the canonical tenant scope) or null.
    58→ *
    59→ * The OrgContext persists the active organization id to localStorage under
    60→ * `gstpilot_org_id` whenever it resolves. This non-React module reads that
    61→ * value so CRUD functions can stamp every document with `organizationId` —
    62→ * the field the Firestore security rules and the onSnapshot hooks require.
    63→ *
    64→ * Falls back to the legacy `firmId` in `gstpilot_session` for compatibility
    65→ * with older sessions.
    66→ */
    67→function currentOrgId(): string | null {
    68→  try {
    69→    // 1. Canonical: org id persisted by OrgContext.
    70→    const orgId = localStorage.getItem('gstpilot_org_id');
    71→    if (orgId) return orgId;
    72→    // 2. Legacy: firmId embedded in the auth session.
    73→    const stored = localStorage.getItem('gstpilot_session');
    74→    if (stored) {
    75→      const parsed = JSON.parse(stored);
    76→      if (parsed.firmId) return parsed.firmId;
    77→    }
    78→  } catch { /* ignore */ }
    79→  return null;
    80→}

    87→/**
    88→ * Convert a Firestore snapshot ({ id, data() }) into a typed object with the
    89→ * document id prepended, recursively converting Firestore Timestamps to ISO
    90→ * strings. Mirrors the conversion logic in `withId` (firestore-schema.ts) but
    91→ * drops the `T extends Record<string, unknown>` constraint so it accepts
    92→ * strongly-typed interfaces (which do not have an implicit index signature).
    93→ */
    94→function docToData<T>(d: { id: string; data: () => Record<string, unknown> }): T & { id: string } {
    95→  const data = d.data();
    96→  const converted: Record<string, unknown> = {};
    97→  for (const [key, value] of Object.entries(data)) {
    98→    if (value && typeof value === 'object' && 'toDate' in value && typeof (value as { toDate: () => Date }).toDate === 'function') {
    99→      converted[key] = (value as { toDate: () => Date }).toDate().toISOString();
   100→    } else {
   101→      converted[key] = value;
   102→    }
   103→  }
   104→  return { id: d.id, ...converted } as T & { id: string };
   105→}

   471→export async function fileReturn(returnId: string): Promise<string> {
   472→  // CRITICAL: This function MUST NOT fake the ARN.
   473→  // GSTPilot never simulates successful government filings.
   474→  //
   475→  // The filing is delegated to the server-side API route which:
   476→  //   1. Checks for an active GSTN connection (Firestore gst_connections)
   477→  //   2. If no connection: returns 400 "GSTN connection required"
   478→  //   3. If mock provider: returns 400 "Cannot file in mock mode"
   479→  //   4. If official provider: calls provider.fileReturn() → real ARN
   480→  //   5. Only marks as "filed" if a real ARN is returned
   481→  //
   482→  // The return status transitions:
   483→  //   draft → prepared → validated → reviewed → submitted (awaiting GSTN ack)
   484→  //   submitted → filed (only when real ARN received from GSTN)
   485→  const response = await fetch(`/api/gstr-filing/${returnId}/file`, {
   486→    method: 'POST',
   487→    headers: { 'Content-Type': 'application/json' },
   488→  });
   489→
   490→  if (!response.ok) {
   491→    const error = await response.json().catch(() => ({ error: 'Filing failed' }));
   492→    throw new Error(error.error || 'Failed to file return with GSTN');
   493→  }
   494→
   495→  const result = await response.json();
   496→
   497→  // If the filing was submitted but not yet acknowledged (no real ARN),
   498→  // the return is marked as "submitted" — not "filed".
   499→  if (!result.acknowledgmentNumber) {
   500→    throw new Error(
   501→      'Return submitted to GSTN but not yet acknowledged. ' +
   502→      'The return status is now "submitted" — it will become "filed" when ' +
   503→      'GSTN returns an acknowledgment number (ARN). Check back after syncing.'
   504→    );
   505→  }
   506→
   507→  return result.acknowledgmentNumber as string;
   508→}

   510→/** Auto-create draft returns for current period.
   511→ *  Period format MUST be "YYYY-MM" to match gst-utils.ts (getFilingDueDate /
   512→ *  isOverdue / periodToLabel) and the dashboard's parsePeriod. Writing
   513→ *  "MM-YYYY" here previously caused "undefined 6" labels and inflated
   514→ *  overdue counts on the dashboard.
   515→ */
   516→async function autoCreateDraftReturns(firmId: string, clientId: string): Promise<void> {
   517→  const now = new Date();
   518→  const period = `${now.getFullYear()}-${String(now.getMonth() + 1).padStart(2, '0')}`;
   519→  const fy = now.getMonth() >= 3 ? `${now.getFullYear()}-${String(now.getFullYear() + 1).slice(2)}` : `${now.getFullYear() - 1}-${String(now.getFullYear()).slice(2)}`;
   520→
   521→  for (const returnType of ['GSTR-1', 'GSTR-3B'] as const) {
   522→    // Check if return already exists
   523→    const q = query(
   524→      collection(db, COLLECTIONS.RETURNS),
   525→      where('clientId', '==', clientId),
   526→      where('returnType', '==', returnType),
   527→      where('period', '==', period),
   528→      limit(1)
   529→    );
   530→    const existing = await getDocs(q);
   531→    if (existing.empty) {
   532→      await createReturn({
   533→        clientId,
   534→        returnType,
   535→        period,
   536→        financialYear: fy,
   537→        status: 'draft',
   538→        filedDate: null,
   539→        acknowledgmentNumber: null,
   540→        totalInvoices: 0,
   541→        readyForFiling: 0,
   542→        issuesFound: 0,
   543→        criticalErrors: 0,
   544→        warnings: 0,
   545→        totalTaxableValue: 0,
   546→        totalTax: 0,
   547→        jsonPayload: null,
   548→        assignedTo: null,
   549→        reviewedBy: null,
   550→      });
   551→    }
   552→  }
   553→}

   556→async function updateDraftReturnForInvoice(firmId: string, clientId: string, returnType: string, period: string, taxableValue: number, taxAmount: number): Promise<void> {
   557→  try {
   558→    const q = query(

   669→export async function resolveMismatch(reconId: string, invoiceNumber: string): Promise<void> {
   670→  const reconRef = doc(db, COLLECTIONS.RECONCILIATIONS, reconId);
   671→  const reconSnap = await getDoc(reconRef);

   673→

   676→    m.invoiceNumber === invoiceNumber
   677→      ? { ...m, resolved: true, resolvedBy: currentUserId(), resolvedAt: new Date().toISOString() }
   678→      : m
   679→  );
   680→
   681→  const resolvedCount = updatedMismatches.filter(m => m.resolved).length;
   682→  const newUnmatched = updatedMismatches.filter(m => !m.resolved && m.matchStatus === 'missing_in_books').length;
   683→  const newPartial = updatedMismatches.filter(m => !m.resolved && m.matchStatus === 'partial_match').length;

   685→  await updateDoc(reconRef, {
   686→    mismatches: updatedMismatches,
   687→    unmatched: newUnmatched,
   688→    partialMatches: newPartial,
   689→    matched: data.totalRecords - newUnmatched - newPartial,
   690→    updatedAt: serverTimestamp(),
   691→  });
   692→
   693→  addActivity({
   694→    type: 'mismatch_resolved',
   695→    title: 'Mismatch resolved',
   696→    description: `Reconciliation mismatch for ${invoiceNumber} resolved`,
   697→    clientId: data.clientId,
   698→    entityType: COLLECTIONS.RECONCILIATIONS,
   699→    entityId: reconId,
   700→  });
   701→}

   877→async function recalculateComplianceScore(clientId: string): Promise<void> {
   878→  try {
   879→    const clientRef = doc(db, COLLECTIONS.CLIENTS, clientId);
   880→    const clientSnap = await getDoc(clientRef);
   881→    if (!clientSnap.exists()) return;
   882→

   884→    let score = 100;
   885→
   886→    // Deductions
   887→    score -= Math.min(client.complianceProfile.overdueReturns * 15, 45);
   888→    score -= Math.min(client.complianceProfile.averageFilingDelay * 2, 20);
   889→    if (!client.complianceProfile.gstinValidity) score -= 20;
   890→    if (client.pendingReturnCount > 3) score -= 10;
   891→
   892→    score = Math.max(0, Math.min(100, score));
   893→
   894→    await updateDoc(clientRef, {
   895→      healthScore: score,
   896→      complianceProfile: {
   897→        ...client.complianceProfile,
   898→        filingCompliance: score,
   899→      },
   900→      updatedAt: serverTimestamp(),
   901→    });
   902→  } catch (e) {
   903→    console.warn('[Workflow] Compliance recalc failed:', e);

   909→/** Subscribe to a Firestore collection with firm scoping */
   910→export function subscribeToCollection<T>(
   911→  collectionName: CollectionName,
   912→  callback: (data: Array<T & { id: string }>) => void,
   913→  ...constraints: QueryConstraint[]
   914→): Unsubscribe {
   915→  const firmId = currentFirmId();
   916→
   917→  const q = firmId
   918→    ? query(collection(db, collectionName), where('firmId', '==', firmId), ...constraints)
   919→    : query(collection(db, collectionName), ...constraints);
   920→
   921→  return onSnapshot(q, (snap) => {
   922→    const data = snap.docs.map(d => {
   923→      const docData = d.data();
   924→      const converted: Record<string, unknown> = { id: d.id };
   925→      for (const [key, value] of Object.entries(docData)) {
   926→        if (value && typeof value === 'object' && 'toDate' in value && typeof (value as { toDate: () => Date }).toDate === 'function') {
   927→          converted[key] = (value as { toDate: () => Date }).toDate().toISOString();
   928→        } else {
   929→          converted[key] = value;
   930→        }
   931→      }
   932→      return converted as T & { id: string };
   933→    });
   934→    callback(data);
   935→  }, (error) => {
   936→    console.warn(`[Firestore] Subscription error for ${collectionName}:`, error);
   937→    callback([]);
   938→  });
   939→}

   941→/** Subscribe to a single Firestore document */
   942→export function subscribeToDoc<T>(
   943→  collectionName: CollectionName,
   944→  docId: string,
   945→  callback: (data: (T & { id: string }) | null) => void,
   946→): Unsubscribe {
   947→  return onSnapshot(doc(db, collectionName, docId), (snap) => {
   948→    if (!snap.exists()) {
   949→      callback(null);
   950→      return;
   951→    }
   952→    const docData = snap.data();
   953→    const converted: Record<string, unknown> = { id: snap.id };
   954→    for (const [key, value] of Object.entries(docData)) {
   955→      if (value && typeof value === 'object' && 'toDate' in value && typeof (value as { toDate: () => Date }).toDate === 'function') {
   956→        converted[key] = (value as { toDate: () => Date }).toDate().toISOString();
   957→      } else {
   958→        converted[key] = value;
   959→      }
   960→    }
   961→    callback(converted as T & { id: string });
   962→  }, (error) => {
   963→    console.warn(`[Firestore] Doc subscription error for ${collectionName}/${docId}:`, error);
   964→    callback(null);
   965→  });
   966→}

   968→/** Compute live dashboard metrics from Firestore data */
   969→export function computeDashboardMetrics(
   970→  clients: Array<FirestoreClient & { id: string }>,
   971→  invoices: Array<FirestoreInvoice & { id: string }>,
   972→  returns: Array<FirestoreReturn & { id: string }>,
   973→  documents: Array<FirestoreDocument & { id: string }>,
   974→  activities: Array<FirestoreActivity & { id: string }>,
   975→): LiveDashboardMetrics {
   976→  const activeClients = clients.filter(c => c.status === 'active').length;
   977→  const filedReturns = returns.filter(r => r.status === 'filed').length;
   978→  const pendingReturns = returns.filter(r => r.status !== 'filed').length;
   979→  const readyToFile = returns.filter(r => ['validated', 'reviewed', 'generated'].includes(r.status)).length;
   980→  const overdueReturns = returns.filter(r => {
   981→    if (r.status === 'filed') return false;
   982→    const periodDate = parsePeriod(r.period);
   983→    return periodDate && periodDate < new Date();
   984→  }).length;
   985→
   986→  const totalTaxVolume = invoices.reduce((sum, inv) => sum + (inv.totalAmount || 0), 0);
   987→  const totalTax = invoices.reduce((sum, inv) => sum + (inv.cgst || 0) + (inv.sgst || 0) + (inv.igst || 0) + (inv.cess || 0), 0);
   988→
   989→  const criticalIssues = invoices.filter(inv => inv.riskLevel === 'critical').length +
   990→    returns.filter(r => r.criticalErrors > 0).length;
   991→  const warnings = invoices.filter(inv => inv.riskLevel === 'high' || inv.riskLevel === 'medium').length +
   992→    returns.filter(r => r.warnings > 0).length;
   993→
   994→  const averageHealthScore = clients.length > 0
   995→    ? Math.round(clients.reduce((sum, c) => sum + c.healthScore, 0) / clients.length)
   996→    : 0;
   997→
   998→  const matchedInvoices = invoices.filter(inv => inv.matchStatus === 'perfect_match').length;
   999→  const matchPercentage = invoices.length > 0 ? Math.round((matchedInvoices / invoices.length) * 100) : 100;
  1000→
  1001→  const riskyInvoices = invoices.filter(inv => ['high', 'critical'].includes(inv.riskLevel)).length;
  1002→  const riskPercentage = invoices.length > 0 ? Math.round((riskyInvoices / invoices.length) * 100) : 0;
  1003→
  1004→  const documentsProcessed = documents.filter(d => d.extractionStatus === 'completed').length;
  1005→  const extractionsPending = documents.filter(d => d.extractionStatus === 'pending' || d.extractionStatus === 'in_progress').length;
  1006→
  1007→  const recentActivities = [...activities]
  1008→    .sort((a, b) => {
  1009→      const aTime = typeof a.createdAt === 'string' ? new Date(a.createdAt).getTime() : 0;
  1010→      const bTime = typeof b.createdAt === 'string' ? new Date(b.createdAt).getTime() : 0;
  1011→      return bTime - aTime;
  1012→    })
  1013→    .slice(0, 10);
  1014→
  1015→  const upcomingFilings = returns
  1016→    .filter(r => r.status !== 'filed')
  1017→    .sort((a, b) => {
  1018→      const aTime = typeof a.createdAt === 'string' ? new Date(a.createdAt).getTime() : 0;
  1019→      const bTime = typeof b.createdAt === 'string' ? new Date(b.createdAt).getTime() : 0;
  1020→      return aTime - bTime;
  1021→    })
  1022→    .slice(0, 5);
  1023→
  1024→  return {
  1025→    totalClients: clients.length,
  1026→    activeClients,
  1027→    totalInvoices: invoices.length,
  1028→    totalTaxVolume,
  1029→    filedReturns,
  1030→    pendingReturns,
  1031→    overdueReturns,
  1032→    readyToFile,
  1033→    criticalIssues,
  1034→    warnings,
  1035→    averageHealthScore,
  1036→    matchPercentage,
  1037→    riskPercentage,
  1038→    documentsProcessed,
  1039→    extractionsPending,
  1040→    recentActivities,
  1041→    upcomingFilings,
  1042→  };
  1043→}

  1045→function parsePeriod(period: string | null): Date | null {
  1046→  if (!period) return null;
  1047→  try {
  1048→    const [mm, yyyy] = period.split('-').map(Number);
  1049→    if (!mm || !yyyy) return null;
  1050→    return new Date(yyyy, mm, 20); // Due date is typically 20th of next month
  1051→  } catch {
  1052→    return null;
  1053→  }
  1054→}

  1637→export async function createNotice(
  1638→  data: Omit<FirestoreNotice, 'noticeId' | 'firmId' | 'organizationId' | 'createdAt' | 'updatedAt'>,
  1639→): Promise<string> {

  1648→    organizationId: firmId,
  1649→    createdAt: serverTimestamp(),
  1650→    updatedAt: serverTimestamp(),
  1651→  };
  1652→  await setDoc(ref, noticeData);
  1653→
  1654→  // Side effects — notify the firm + log activity
  1655→  addActivity({
  1656→    type: 'system',
  1657→    title: 'Notice added',
  1658→    description: `${data.subject}${data.clientTradeName ? ` — ${data.clientTradeName}` : ''}`,
  1659→    clientId: data.clientId || null,
  1660→    entityType: COLLECTIONS.NOTICES,
  1661→    entityId: noticeId,
  1662→  });

  1664→    type: 'issue_detected',
  1665→    priority: data.priority === 'urgent' ? 'urgent' : 'high',
  1666→    title: 'New notice added',
  1667→    message: `${data.noticeType.replace(/_/g, ' ')}: ${data.subject}`,
  1668→    entityType: COLLECTIONS.NOTICES,
  1669→    entityId: noticeId,
  1670→  });
  1671→
  1672→  return noticeId;
  1673→}
