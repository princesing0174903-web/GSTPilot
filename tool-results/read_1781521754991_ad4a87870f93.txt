   158→function getKanbanColumn(status: FilingStatus): KanbanColumn {
   159→  switch (status) {
   160→    case 'draft':
   161→    case 'prepared':
   162→      return 'draft';
   163→    case 'validated':
   164→    case 'reviewed':
   165→    case 'generated':
   166→      return 'ready';
   167→    case 'filed':
   168→      return 'filed';
   169→    case 'reopened':
   170→      return 'attention';
   171→    default:
   172→      return 'draft';
   173→  }
   174→}
   175→
   176→function getTimelineStepIndex(status: FilingStatus): number {
   177→  switch (status) {
   178→    case 'draft':
   179→    case 'prepared':
   180→      return 0;
   181→    case 'validated':
   182→    case 'reviewed':
   183→      return 1;
   184→    case 'generated':
   185→      return 2;
   186→    case 'filed':
   187→      return 3;
   188→    case 'reopened':
   189→      return -1;
   190→    default:
   191→      return 0;
   192→  }
   193→}
   194→
   195→function getAttentionSummary(filing: GSTRFiling): string {
   196→  if (filing.issuesFound <= 0) return 'No issues';
   197→  const parts: string[] = [];
   198→  if (filing.criticalErrors > 0) parts.push(`${filing.criticalErrors} critical error${filing.criticalErrors > 1 ? 's' : ''}`);
   199→  if (filing.warnings > 0) parts.push(`${filing.warnings} warning${filing.warnings > 1 ? 's' : ''}`);
   200→  if (filing.issuesFound > 0 && parts.length === 0) parts.push(`${filing.issuesFound} issue${filing.issuesFound > 1 ? 's' : ''}`);
   201→  return parts.join(' · ');
   202→}

   211→// ─── Section Breakdown Computation ──────────────────────────────────────────────
   212→
   213→function computeSectionsForFiling(
   214→  returnType: string,
   215→  invoices: { taxableValue: number; cgst: number; sgst: number; igst: number; buyerGstin?: string | null; status: string }[]
   216→): SectionBreakdown[] {
   217→  if (returnType === 'GSTR-1') {
   218→    const b2b = invoices.filter(inv => inv.buyerGstin);
   219→    const b2cLarge = invoices.filter(inv => !inv.buyerGstin && inv.taxableValue >= 250000);
   220→    const b2cSmall = invoices.filter(inv => !inv.buyerGstin && inv.taxableValue < 250000);
   221→
   222→    return [
   223→      {
   224→        section: 'B2B',
   225→        invoiceCount: b2b.length,
   226→        taxableValue: b2b.reduce((s, i) => s + i.taxableValue, 0),
   227→        taxAmount: b2b.reduce((s, i) => s + i.cgst + i.sgst + i.igst, 0),
   228→      },
   229→      {
   230→        section: 'B2C Large',
   231→        invoiceCount: b2cLarge.length,
   232→        taxableValue: b2cLarge.reduce((s, i) => s + i.taxableValue, 0),
   233→        taxAmount: b2cLarge.reduce((s, i) => s + i.cgst + i.sgst + i.igst, 0),
   234→      },
   235→      {
   236→        section: 'B2C Small',
   237→        invoiceCount: b2cSmall.length,
   238→        taxableValue: b2cSmall.reduce((s, i) => s + i.taxableValue, 0),
   239→        taxAmount: b2cSmall.reduce((s, i) => s + i.cgst + i.sgst + i.igst, 0),
   240→      },
   241→      {
   242→        section: 'CDNR',
   243→        invoiceCount: 0,
   244→        taxableValue: 0,
   245→        taxAmount: 0,
   246→      },
   247→    ];
   248→  }
   249→
   250→  // GSTR-3B
   251→  const totalTax = invoices.reduce((s, i) => s + i.cgst + i.sgst + i.igst, 0);
   252→  const totalTaxable = invoices.reduce((s, i) => s + i.taxableValue, 0);
   253→  return [
   254→    {
   255→      section: 'Outward Supplies',
   256→      invoiceCount: invoices.length,
   257→      taxableValue: totalTaxable,
   258→      taxAmount: totalTax,
   259→    },
   260→    {
   261→      section: 'Inward Supplies (RC)',
   262→      invoiceCount: 0,
   263→      taxableValue: 0,
   264→      taxAmount: 0,
   265→    },
   266→    {
   267→      section: 'ITC Claims',
   268→      invoiceCount: 0,
   269→      taxableValue: 0,
   270→      taxAmount: Math.round(-totalTax * 0.6),
   271→    },
   272→    {
   273→      section: 'Tax Paid',
   274→      invoiceCount: 0,
   275→      taxableValue: 0,
   276→      taxAmount: Math.round(totalTax * 0.75),
   277→    },
   278→  ];
   279→}
