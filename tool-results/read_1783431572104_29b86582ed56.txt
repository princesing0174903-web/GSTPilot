    31→function SparklineChart({ data, color = '#10b981', height = 40 }: { data: number[]; color?: string; height?: number }) {
    32→  const max = Math.max(...data)
    33→  const min = Math.min(...data)
    34→  const range = max - min || 1
    35→  const w = 120
    36→  const pts = data.map((v, i) => `${(i / (data.length - 1)) * w},${height - ((v - min) / range) * (height - 4) - 2}`)
    37→  return (
    38→    <svg width={w} height={height} className="overflow-visible">
    39→      <defs>
    40→        <linearGradient id={`grad-${color.replace('#', '')}`} x1="0" y1="0" x2="0" y2="1">
    41→          <stop offset="0%" stopColor={color} stopOpacity="0.3" />
    42→          <stop offset="100%" stopColor={color} stopOpacity="0.02" />
    43→        </linearGradient>
    44→      </defs>
    45→      <polygon
    46→        points={`0,${height} ${pts.join(' ')} ${w},${height}`}
    47→        fill={`url(#grad-${color.replace('#', '')})`}
    48→      />
    49→      <polyline points={pts.join(' ')} fill="none" stroke={color} strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" />
    50→    </svg>
    51→  )
    52→}

    54→function BarChartMini({ data, labels, color = '#10b981' }: { data: number[]; labels: string[]; color?: string }) {
    55→  const max = Math.max(...data)
    56→  const barW = 28
    57→  const gap = 8
    58→  const h = 80
    59→  return (
    60→    <svg width={data.length * (barW + gap)} height={h + 20} className="overflow-visible">
    61→      {data.map((v, i) => {
    62→        const barH = (v / max) * (h - 10)
    63→        return (
    64→          <g key={i}>
    65→            <rect x={i * (barW + gap)} y={h - barH} width={barW} height={barH} rx={4} fill={color} opacity={0.8} />
    66→            <text x={i * (barW + gap) + barW / 2} y={h + 14} textAnchor="middle" className="text-[9px] fill-muted-foreground">{labels[i]}</text>
    67→          </g>
    68→        )
    69→      })}
    70→    </svg>
    71→  )
    72→}
