    53→    53→function getDaysRemaining(dueDateStr: string): number {
    54→    54→  const dueDate = new Date(dueDateStr);
    55→    55→  const today = new Date();
    56→    56→  today.setHours(0, 0, 0, 0);
    57→    57→  dueDate.setHours(0, 0, 0, 0);
    58→    58→  return Math.ceil((dueDate.getTime() - today.getTime()) / (1000 * 60 * 60 * 24));
    59→    59→}
    60→    60→
    61→    61→function formatDaysRemaining(days: number): string {
    62→    62→  if (days < 0) return `${Math.abs(days)}d overdue`;
    63→    63→  if (days === 0) return 'Due today';
    64→    64→  if (days === 1) return 'Tomorrow';
    65→    65→  return `${days}d left`;
    66→    66→}
    67→    67→
    68→    68→function getInitials(name: string): string {
    69→    69→  return name.split(' ').map(w => w[0]).join('').slice(0, 2).toUpperCase();
    70→    70→}
    71→    71→
    72→    72→function formatINR(amount: number): string {
    73→    73→  return amount.toLocaleString('en-IN');
    74→    74→}
    75→    75→
    76→    76→function timeAgo(dateStr: string): string {
    77→    77→  const now = new Date();
    78→    78→  const date = new Date(dateStr);
    79→    79→  const seconds = Math.floor((now.getTime() - date.getTime()) / 1000);
    80→    80→  if (seconds < 60) return 'just now';
    81→    81→  const minutes = Math.floor(seconds / 60);
    82→    82→  if (minutes < 60) return `${minutes}m ago`;
    83→    83→  const hours = Math.floor(minutes / 60);
    84→    84→  if (hours < 24) return `${hours}h ago`;
    85→    85→  const days = Math.floor(hours / 24);
    86→    86→  if (days < 7) return `${days}d ago`;
    87→    87→  return date.toLocaleDateString('en-IN', { day: 'numeric', month: 'short' });
    88→    88→}
