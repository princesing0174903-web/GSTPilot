   155→function formatFileSize(bytes: number): string {
   156→  if (bytes < 1024) return `${bytes} B`;
   157→  if (bytes < 1048576) return `${(bytes / 1024).toFixed(1)} KB`;
   158→  return `${(bytes / 1048576).toFixed(1)} MB`;
   159→}

   161→function getConfidenceColor(confidence: number): string {
   162→  if (confidence >= 95) return 'text-emerald-600';
   163→  if (confidence >= 80) return 'text-amber-600';
   164→  return 'text-red-600';
   165→}

   347→  // ── Upload Handlers ──
   348→  // Deterministic file sizes based on file name hash
   349→  function getDeterministicSize(name: string): number {
   350→    let hash = 0;
   351→    for (let i = 0; i < name.length; i++) {
   352→      hash = ((hash << 5) - hash) + name.charCodeAt(i);
   353→      hash |= 0;
   354→    }

   395→              prev.map((f, i) =>
   396→                i === prev.length - newFiles.length + fileIndex

   398→                  : f
   399→              )
   400→            );
   401→
   402→            // Phase 3: Validating -> Complete

   405→                prev.map((f, i) =>

   425→              i === prev.length - newFiles.length + fileIndex

   430→        }

   540→        {/* ════════════════════════════════════════════════════════════════════
