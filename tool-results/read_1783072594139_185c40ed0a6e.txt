   184→function formatFileSize(bytes: number): string {
   185→  if (bytes < 1024) return `${bytes} B`;
   186→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`;
   187→  return `${(bytes / (1024 * 1024)).toFixed(1)} MB`;
   188→}
   189→
   190→function getFileIcon(type: string) {
   191→  if (type.includes('sheet') || type.includes('excel') || type.includes('csv'))
   192→    return <FileSpreadsheet className="size-5 text-emerald-600" />;
   193→  if (type.includes('image')) return <FileImage className="size-5 text-teal-600" />;
   194→  if (type.includes('pdf')) return <FileText className="size-5 text-red-500" />;
   195→  return <File className="size-5 text-slate-500" />;
   196→}
   197→
   198→function getOverdueFilings(filings: GSTRFiling[]): Set<string> {
   199→  const now = new Date();
   200→  const overdueIds = new Set<string>();
   201→  filings.forEach((f) => {
   202→    if (f.status !== 'filed' && f.status !== 'reopened') {
   203→      const [year, month] = f.period.split('-').map(Number);
   204→      if (year && month) {
   205→        const dueDate = new Date(year, month, 11);
   206→        if (now > dueDate) overdueIds.add(f.id);
   207→      }
   208→    }
   209→  });
   210→  return overdueIds;
   211→}
