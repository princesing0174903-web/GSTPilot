    60→function inrShort(n: number): string {
    61→  if (!isFinite(n) || isNaN(n)) return '₹0';
    62→  const abs = Math.abs(n);
    63→  if (abs >= 10000000) return `₹${(n / 10000000).toFixed(2)} Cr`;
    64→  if (abs >= 100000) return `₹${(n / 100000).toFixed(2)} L`;
    65→  if (abs >= 1000) return `₹${(n / 1000).toFixed(1)}K`;
    66→  return '₹' + Math.round(n).toLocaleString('en-IN');
    67→}

  1089→  messages: OracleChatRequest['messages'],

  1097→/** Server-side language hint derived from the latest user message. */
  1098→function inferLanguageHint(messages: OracleChatRequest['messages']): OracleLanguageId | undefined {
  1099→  for (let i = messages.length - 1; i >= 0; i--) {
  1100→    if (messages[i].role === 'user' && messages[i].content?.trim()) {
  1101→      const s = messages[i].content.slice(0, 500);
  1102→      if (/[\u0900-\u097F]/.test(s)) return 'hindi';
  1103→      if (/[\u0600-\u06FF]/.test(s)) return 'urdu';
  1104→      if (/[\u0A00-\u0A7F]/.test(s)) return 'punjabi';
  1105→      if (/[\u0A80-\u0AFF]/.test(s)) return 'gujarati';
  1106→      if (/[\u0B80-\u0BFF]/.test(s)) return 'tamil';
  1107→      if (/[\u0C00-\u0C7F]/.test(s)) return 'telugu';
  1108→      if (/[\u0980-\u09FF]/.test(s)) return 'bengali';
  1109→      return 'english';
  1110→    }
  1111→  }
  1112→  return undefined;
  1113→}
