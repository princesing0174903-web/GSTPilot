    99→    99→function NotificationBell() {
   100→   100→  const { data: unreadNotifs } = useFireUnreadNotifications()
   101→   101→  const { data: allNotifs } = useFireNotifications()

   103→   103→  const [open, setOpen] = useState(false)
   104→   104→
   105→   105→  const handleMarkRead = async (notifId: string) => {
   106→   106→    try { await markNotificationRead(notifId) } catch {}
   107→   107→  }

   110→   110→    try {
   111→   111→      for (const n of unreadNotifs) {
   112→   112→        await markNotificationRead(n.notifId || n.id)

   114→   114→    } catch {}
   115→   115→  }
   116→   116→
   117→   117→  const priorityColor = (p: string) => {
   118→   118→    switch (p) {
   119→   119→      case 'urgent': return 'text-red-600 bg-red-50 border-red-200'
   120→   120→      case 'high': return 'text-orange-600 bg-orange-50 border-orange-200'
   121→   121→      case 'normal': return 'text-blue-600 bg-blue-50 border-blue-200'
   122→   122→      default: return 'text-slate-600 bg-slate-50 border-slate-200'
   123→   123→    }
   124→   124→  }
   125→   125→

   127→   127→    switch (p) {
   128→   128→      case 'urgent': return 'bg-red-500'

   130→   130→      case 'normal': return 'bg-blue-500'
   131→   131→      default: return 'bg-slate-400'
   132→   132→    }
   133→   133→  }
   134→   134→
   135→   135→  const typeIcon = (t: string) => {
   136→   136→    switch (t) {

   141→   141→      case 'issue_detected': return <AlertTriangle className="h-3.5 w-3.5" />

   171→   171→                {unreadCount} new

   173→   173→            )}
   174→   174→          </div>

   176→   176→            <Button variant="ghost" size="sm" onClick={handleMarkAllRead} className="h-6 text-[10px] text-emerald-600 hover:text-emerald-700 px-2">
   177→   177→              Mark all read

   202→   202→                        {notif.title}
   203→   203→                      </p>
   204→   204→                      {!notif.read && <span className={`h-1.5 w-1.5 rounded-full ${priorityBadge(notif.priority)} shrink-0`} />}

   222→   222→              size="sm"

   261→   261→// KEYBOARD SHORTCUTS HOOK
   262→   262→// ═══════════════════════════════════════════════════════════════════════════════
   263→   263→
   264→   264→function useKeyboardShortcuts() {
   265→   265→  const { setCurrentView, setCommandPaletteOpen, commandPaletteOpen } = useApp()
   266→   266→  const gPressed = useRef(false)
   267→   267→  const gTimer = useRef<NodeJS.Timeout | null>(null)
   268→   268→
   269→   269→  useEffect(() => {
   270→   270→    const handler = (e: KeyboardEvent) => {
   271→   271→      // Ctrl+K → Toggle Command Palette
   272→   272→      if ((e.metaKey || e.ctrlKey) && e.key === 'k') {
   273→   273→        e.preventDefault()
   274→   274→        setCommandPaletteOpen(!commandPaletteOpen)
   275→   275→        return
   276→   276→      }
   277→   277→
   278→   278→      // G prefix shortcuts (G+D, G+C, G+I, G+R)
   279→   279→      if (e.key === 'g' && !e.metaKey && !e.ctrlKey && !e.altKey && !(e.target instanceof HTMLInputElement || e.target instanceof HTMLTextAreaElement)) {
   280→   280→        if (gPressed.current) return
   281→   281→        gPressed.current = true
   282→   282→        gTimer.current = setTimeout(() => { gPressed.current = false }, 500)
   283→   283→        return
   284→   284→      }
   285→   285→
   286→   286→      if (gPressed.current && !e.metaKey && !e.ctrlKey && !e.altKey) {
   287→   287→        gPressed.current = false
   288→   288→        if (gTimer.current) clearTimeout(gTimer.current)
   289→   289→
   290→   290→        switch (e.key) {
   291→   291→          case 'd': setCurrentView('dashboard'); break
   292→   292→          case 'c': setCurrentView('clients'); break
   293→   293→          case 'i': setCurrentView('invoices'); break
   294→   294→          case 'r': setCurrentView('returns'); break
   295→   295→        }
   296→   296→      }
   297→   297→    }
   298→   298→
   299→   299→    window.addEventListener('keydown', handler)
   300→   300→    return () => window.removeEventListener('keydown', handler)
   301→   301→  }, [setCurrentView, setCommandPaletteOpen, commandPaletteOpen])
   302→   302→}

   478→   478→// ═══════════════════════════════════════════════════════════════════════════════
   479→   479→
   480→   480→function OnboardingScreen() {
   481→   481→  const { user, markOnboardingComplete } = useAuth()
   482→   482→  const { setCurrentScreen, setCurrentView } = useApp()
   483→   483→  const [saveError, setSaveError] = React.useState<string | null>(null)
   484→   484→
   485→   485→  const saveOnboardingToFirestore = async (
   486→   486→    data: import('@/components/onboarding/OnboardingFlow').OnboardingData,
   487→   487→    firmId: string
   488→   488→  ) => {
   489→   489→    try {
   490→   490→      const { doc, setDoc, serverTimestamp } = await import('firebase/firestore')
   491→   491→      const { db } = await import('@/lib/firebase')
   492→   492→      if (!user) return
   493→   493→
   494→   494→      await setDoc(doc(db, 'users', user.id), {
   495→   495→        uid: user.id, fullName: data.fullName, email: data.email, phone: data.phone,
   496→   496→        ageGroup: data.ageGroup, profession: data.profession, experience: data.experience,
   497→   497→        firmId: firmId || null, firmName: data.firmName, onboardingCompleted: true,
   498→   498→        clientCount: data.clientCount, monthlyReturns: data.monthlyReturns,
   499→   499→        gstServices: data.gstServices, painPoints: data.painPoints,
   500→   500→        referralSource: data.referralSource, trialReasons: data.trialReasons,
   501→   501→        wantsUpdates: data.wantsUpdates, updatedAt: serverTimestamp(),
   502→   502→      }, { merge: true })
   503→   503→
   504→   504→      await setDoc(doc(db, 'onboarding', user.id), {
   505→   505→        ...data, firmId: firmId || null, completedAt: serverTimestamp(),
   506→   506→      })
   507→   507→    } catch (error) {
   508→   508→      console.warn('[Onboarding] Background Firestore save failed:', error)
   509→   509→    }
   510→   510→  }
   511→   511→
   512→   512→  const handleOnboardingComplete = (
   513→   513→    data: import('@/components/onboarding/OnboardingFlow').OnboardingData,
   514→   514→    destination?: import('@/components/onboarding/OnboardingFlow').OnboardingDestination
   515→   515→  ) => {
   516→   516→    setSaveError(null)
   517→   517→    if (!user) { setSaveError('No authenticated user found. Please sign in again.'); return }
   518→   518→
   519→   519→    markOnboardingComplete(undefined, data.firmName)
   520→   520→    setCurrentView(destination || 'dashboard')
   521→   521→    setCurrentScreen('app')
   522→   522→
   523→   523→    ;(async () => {
   524→   524→      try {
   525→   525→        const { collection, addDoc, serverTimestamp } = await import('firebase/firestore')
   526→   526→        const { db } = await import('@/lib/firebase')
   527→   527→        let firmId = ''
   528→   528→        try {
   529→   529→          const firmRef = await addDoc(collection(db, 'firms'), {
   530→   530→            ownerId: user.id, firmName: data.firmName, gstin: data.gstin || null,
   531→   531→            state: data.state, stateCode: data.stateCode,
   532→   532→            organizationType: data.organizationType,
   533→   533→            icaiMembershipNo: data.icaiMembershipNo || null,
   534→   534→            officeAddress: data.officeAddress || null, createdAt: serverTimestamp(),
   535→   535→          })
   536→   536→          firmId = firmRef.id
   537→   537→          markOnboardingComplete(firmId, data.firmName)
   538→   538→        } catch (firmError) {
   539→   539→          console.warn('[Onboarding] Firm creation failed:', firmError)
   540→   540→        }
   541→   541→        await saveOnboardingToFirestore(data, firmId)
   542→   542→      } catch (error) {
   543→   543→        console.warn('[Onboarding] Background save error:', error)
   544→   544→      }
   545→   545→    })()
   546→   546→  }
   547→   547→
   548→   548→  const handleSkip = () => {
   549→   549→    setSaveError(null)
   550→   550→    if (!user) { setSaveError('No authenticated user found.'); return }
   551→   551→
   552→   552→    markOnboardingComplete()
   553→   553→    setCurrentView('dashboard')
   554→   554→    setCurrentScreen('app')
   555→   555→
   556→   556→    ;(async () => {
   557→   557→      try {
   558→   558→        const { doc, setDoc, serverTimestamp } = await import('firebase/firestore')
   559→   559→        const { db } = await import('@/lib/firebase')
   560→   560→        await setDoc(doc(db, 'users', user.id), {
   561→   561→          onboardingCompleted: true, updatedAt: serverTimestamp(),
   562→   562→        }, { merge: true })
   563→   563→      } catch (error) {
   564→   564→        console.warn('[Onboarding] Background skip save failed:', error)
   565→   565→      }
   566→   566→    })()
   567→   567→  }
   568→   568→
   569→   569→  return (
   570→   570→    <OnboardingFlow
   571→   571→      onComplete={handleOnboardingComplete}
   572→   572→      onSkip={handleSkip}
   573→   573→      userEmail={user?.email}
   574→   574→      userName={user?.name}
   575→   575→      error={saveError}
   576→   576→      onDismissError={() => setSaveError(null)}
   577→   577→    />
   578→   578→  )
   579→   579→}

   585→   585→function AppRouter() {
   586→   586→  const { currentScreen, setCurrentScreen } = useApp()
   587→   587→  const { isAuthenticated, isInitializing, needsOnboarding, needsEmailVerification } = useAuth()
   588→   588→

   619→   619→
   620→   620→  if (isAuthenticated && needsOnboarding) {
   621→   621→    return <OnboardingScreen />
   622→   622→  }
   623→   623→
   624→   624→  if (currentScreen === 'app' && isAuthenticated) {
   625→   625→    return (
   626→   626→      <div className="min-h-screen flex flex-col">
   627→   627→        {needsEmailVerification && <EmailVerificationBanner />}
   628→   628→        <div className="flex-1 flex">
   629→   629→          <DashboardContent />
   630→   630→        </div>
   631→   631→      </div>
   632→   632→    )
   633→   633→  }
   634→   634→
   635→   635→  if (currentScreen === 'login') {
   636→   636→    return <LoginPage onBack={handleBackToLanding} onGetStarted={handleGetStarted} />
   637→   637→  }
   638→   638→
   639→   639→  return <LandingPage onGetStarted={handleGetStarted} onBookDemo={handleBookDemo} />
   640→   640→}
