    64→    64→function getDaysRemaining(dueDateStr: string): number {
    65→    65→  const dueDate = new Date(dueDateStr);
    66→    66→  const today = new Date();
    67→    67→  today.setHours(0, 0, 0, 0);
    68→    68→  dueDate.setHours(0, 0, 0, 0);
    69→    69→  return Math.ceil((dueDate.getTime() - today.getTime()) / (1000 * 60 * 60 * 24));
    70→    70→}
    71→    71→
    72→    72→function formatDaysRemaining(days: number): string {
    73→    73→  if (days < 0) return `${Math.abs(days)}d overdue`;
    74→    74→  if (days === 0) return 'Due today';
    75→    75→  if (days === 1) return 'Tomorrow';
    76→    76→  return `${days}d left`;
    77→    77→}

    83→    83→function formatDateIN(dateStr: string): string {
    84→    84→  const d = new Date(dateStr);
    85→    85→  if (Number.isNaN(d.getTime())) return '—';
    86→    86→  const dd = String(d.getDate()).padStart(2, '0');
    87→    87→  const mm = String(d.getMonth() + 1).padStart(2, '0');
    88→    88→  const yyyy = d.getFullYear();
    89→    89→  return `${dd}/${mm}/${yyyy}`;
    90→    90→}

    92→    92→function timeAgo(dateStr: string): string {
    93→    93→  const now = new Date();
    94→    94→  const date = new Date(dateStr);
    95→    95→  const seconds = Math.floor((now.getTime() - date.getTime()) / 1000);
    96→    96→  if (seconds < 60) return 'just now';
    97→    97→  const minutes = Math.floor(seconds / 60);
    98→    98→  if (minutes < 60) return `${minutes}m ago`;
    99→    99→  const hours = Math.floor(minutes / 60);
   100→   100→  if (hours < 24) return `${hours}h ago`;
   101→   101→  const days = Math.floor(hours / 24);
   102→   102→  if (days < 7) return `${days}d ago`;
   103→   103→  return date.toLocaleDateString('en-IN', { day: 'numeric', month: 'short' });
   104→   104→}

   119→   119→// One-sentence AI insight derived from live metrics.
   120→   120→function buildInsightSentence(
   121→   121→  metrics: LiveDashboardMetrics,
   122→   122→  pendingCollection: number,
   123→   123→): string {
   124→   124→  const parts: string[] = [];
   125→   125→  if (metrics.overdueReturns > 0) {
   126→   126→    parts.push(
   127→   127→      `${metrics.overdueReturns} overdue return${metrics.overdueReturns > 1 ? 's' : ''}`,
   128→   128→    );
   129→   129→  } else if (metrics.pendingReturns > 0) {
   130→   130→    parts.push(
   131→   131→      `${metrics.pendingReturns} return${metrics.pendingReturns > 1 ? 's' : ''} to file`,
   132→   132→    );
   133→   133→  }
   134→   134→  if (pendingCollection > 0) {
   135→   135→    parts.push(`₹${formatINR(pendingCollection)} pending collection`);
   136→   136→  }
   137→   137→  if (metrics.criticalIssues > 0) {
   138→   138→    parts.push(
   139→   139→      `${metrics.criticalIssues} critical issue${metrics.criticalIssues > 1 ? 's' : ''}`,
   140→   140→    );
   141→   141→  }
   142→   142→  if (parts.length === 0) {
   143→   143→    return `All clear — ${metrics.filedReturns} returns filed and ${metrics.totalClients} clients in good standing.`;
   144→   144→  }
   145→   145→  if (parts.length === 1) return `${parts[0]}.`;
   146→   146→  const last = parts.pop();
   147→   147→  return `${parts.join(', ')} and ${last}.`;
   148→   148→}

   267→   267→function healthTier(score: number): { label: string; tone: string } {
