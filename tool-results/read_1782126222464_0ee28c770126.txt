   209→export function OnboardingFlow({

   211→  onSkip,
   212→  userEmail = '',
   213→  userName = '',
   214→  error = null,
   215→  onDismissError,
   216→}: OnboardingFlowProps) {
   217→  const [currentStep, setCurrentStep] = useState(0); // 0=welcome, 1-4=steps, 5=final
   218→  const [direction, setDirection] = useState(1);
   219→
   220→  const [formData, setFormData] = useState<OnboardingData>({
   221→    fullName: userName,
   222→    email: userEmail,
   223→    phone: '',
   224→    ageGroup: '',
   225→    profession: '',
   226→    experience: '',
   227→    firmName: '',
   228→    organizationType: '',
   229→    gstin: '',
   230→    state: '',
   231→    stateCode: '',
   232→    icaiMembershipNo: '',
   233→    officeAddress: '',
   234→    clientCount: '',
   235→    monthlyReturns: '',
   236→    gstServices: [],
   237→    painPoints: [],
   238→    referralSource: '',
   239→    trialReasons: [],
   240→    wantsUpdates: true,
   241→  });
   242→
   243→  const [errors, setErrors] = useState<Record<string, string>>({});
   244→
   245→  // ─── Helpers ──────────────────────────────────────────────────────────────
   246→
   247→  const updateField = useCallback(
   248→    <K extends keyof OnboardingData>(field: K, value: OnboardingData[K]) => {
   249→      setFormData((prev) => ({ ...prev, [field]: value }));
   250→      // Clear error on change
   251→      setErrors((prev) => {
   252→        const next = { ...prev };
   253→        delete next[field];
   254→        return next;
   255→      });
   256→    },
   257→    []
   258→  );
   259→
   260→  const toggleArrayItem = useCallback(
   261→    (field: 'gstServices' | 'painPoints' | 'trialReasons', item: string) => {
   262→      setFormData((prev) => {
   263→        const arr = prev[field] as string[];
   264→        const next = arr.includes(item)
   265→          ? arr.filter((i) => i !== item)
   266→          : [...arr, item];
   267→        return { ...prev, [field]: next };
   268→      });
   269→    },
   270→    []
   271→  );
   272→
   273→  const goNext = useCallback(() => {
   274→    setDirection(1);
   275→    setCurrentStep((prev) => Math.min(prev + 1, 5));
   276→  }, []);
   277→
   278→  const goBack = useCallback(() => {
   279→    setDirection(-1);
   280→    setCurrentStep((prev) => Math.max(prev - 1, 0));
   281→  }, []);
   282→
   283→  // ─── Validation ───────────────────────────────────────────────────────────
   284→
   285→  const validateStep = useCallback(
   286→    (step: number): boolean => {
   287→      const newErrors: Record<string, string> = {};
   288→
   289→      if (step === 1) {
   290→        if (!formData.fullName.trim()) newErrors.fullName = 'Full name is required';
   291→        if (!formData.email.trim()) newErrors.email = 'Email is required';
   292→        else if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(formData.email))
   293→          newErrors.email = 'Enter a valid email';
   294→        if (!formData.phone.trim()) newErrors.phone = 'Mobile number is required';
   295→        else if (!/^\d{10}$/.test(formData.phone.replace(/\s/g, '')))
   296→          newErrors.phone = 'Enter a valid 10-digit mobile number';
   297→      }
   298→
   299→      if (step === 2) {
   300→        if (!formData.firmName.trim()) newErrors.firmName = 'Firm name is required';
   301→      }
   302→
   303→      setErrors(newErrors);
   304→      return Object.keys(newErrors).length === 0;
   305→    },
   306→    [formData]
   307→  );
   308→
   309→  const handleNext = useCallback(() => {
   310→    if (validateStep(currentStep)) {
   311→      goNext();
   312→    }
   313→  }, [currentStep, validateStep, goNext]);
   314→
   315→  // ─── GSTIN Validation ─────────────────────────────────────────────────────
   316→
   317→  const getGstinStatus = useCallback(() => {
   318→    const val = formData.gstin.trim();
   319→    if (!val) return null;
   320→    return GSTIN_REGEX.test(val) ? 'valid' : 'invalid';
   321→  }, [formData.gstin]);
   322→
   323→  // ─── State Code Lookup ────────────────────────────────────────────────────
   324→
   325→  const handleStateChange = useCallback(
   326→    (stateName: string) => {
   327→      const stateObj = INDIAN_STATES.find((s) => s.name === stateName);
   328→      setFormData((prev) => ({
   329→        ...prev,
   330→        state: stateName,
   331→        stateCode: stateObj?.code ?? '',
   332→      }));
   333→      setErrors((prev) => {
   334→        const next = { ...prev };
   335→        delete next.state;
   336→        return next;
   337→      });
   338→    },
   339→    []
   340→  );
   341→
   342→  // ─── Progress ─────────────────────────────────────────────────────────────
   343→
   344→  const totalSteps = 5;
   345→  const progressPercent =
   346→    currentStep === 0 ? 0 : (currentStep / totalSteps) * 100;
   347→
   348→  // ─── Step Icons ───────────────────────────────────────────────────────────
   349→
   350→  const stepIcons = [Zap, User, Building2, Briefcase, Target];
   351→
   352→  // ─── Render Step Content ──────────────────────────────────────────────────
   353→
   354→  const renderStep = () => {

   356→    if (currentStep === 0) {
   357→      return (
   358→        <motion.div
   359→          key="step-0"
   360→          custom={direction}

   362→          initial="enter"

   364→          exit="exit"
   365→          transition={stepTransition}

   368→          <motion.div

   370→            animate={{ scale: 1, opacity: 1 }}
   371→            transition={{ type: 'spring', stiffness: 200, damping: 15, delay: 0.1 }}

   373→          >

   375→              <Zap className="size-10 text-white" />
   376→            </div>
   377→          </motion.div>
   378→          <motion.h1
   379→            initial={{ y: 20, opacity: 0 }}

   381→            transition={{ delay: 0.2 }}
   382→            className="text-2xl font-bold tracking-tight text-foreground"
   383→          >
   384→            Welcome to GSTPilot
   385→          </motion.h1>
   386→          <motion.p

   390→            className="text-muted-foreground mt-2 max-w-sm"
   391→          >
   392→            Let&apos;s set up your workspace in less than 2 minutes.
   393→          </motion.p>
   394→          <motion.div

   396→            animate={{ y: 0, opacity: 1 }}
   397→            transition={{ delay: 0.4 }}
   398→            className="mt-8"

   400→            <Button
   401→              onClick={goNext}
   402→              size="lg"
   403→              className="bg-emerald-600 hover:bg-emerald-700 text-white gap-2 px-8"
   404→            >
   405→              Get Started
   406→              <ArrowRight className="size-4" />
   407→            </Button>
   408→          </motion.div>
   409→        </motion.div>
   410→      );
   411→    }
   412→
   413→    // Step 1 — About You
   414→    if (currentStep === 1) {
   415→      return (
   416→        <motion.div
   417→          key="step-1"
   418→          custom={direction}
   419→          variants={slideVariants}
   420→          initial="enter"
   421→          animate="center"
   422→          exit="exit"
   423→          transition={stepTransition}
   424→          className="py-4"
   425→        >
   426→          <div className="flex items-center gap-3 mb-6">
   427→            <div className="w-10 h-10 rounded-xl bg-emerald-100 flex items-center justify-center">
   428→              <User className="size-5 text-emerald-600" />
   429→            </div>
   430→            <div>
   431→              <h2 className="text-2xl font-bold">About You</h2>
   432→              <p className="text-sm text-muted-foreground">
   433→                Tell us about yourself so we can personalize your experience
   434→              </p>
   435→            </div>
   436→          </div>
   437→
   438→          <div className="space-y-5">
   439→            {/* Full Name */}
   440→            <div className="space-y-2">
   441→              <Label htmlFor="fullName">
   442→                Full Name <span className="text-destructive">*</span>
   443→              </Label>
   444→              <Input
   445→                id="fullName"
   446→                placeholder="Rajesh Kumar Sharma"
   447→                value={formData.fullName}
   448→                onChange={(e) => updateField('fullName', e.target.value)}
   449→                aria-invalid={!!errors.fullName}
   450→                className={errors.fullName ? 'border-destructive' : ''}
   451→              />
   452→              {errors.fullName && (
   453→                <p className="text-xs text-destructive">{errors.fullName}</p>

   456→
   457→            {/* Work Email */}

   465→                placeholder="rajesh@yourfirm.com"

   467→                onChange={(e) => updateField('email', e.target.value)}
   468→                aria-invalid={!!errors.email}
   469→                className={errors.email ? 'border-destructive' : ''}
   470→              />
   471→              {errors.email && (
   472→                <p className="text-xs text-destructive">{errors.email}</p>
   473→              )}
   474→            </div>
   475→
   476→            {/* Mobile Number */}

   491→                    updateField('phone', e.target.value.replace(/[^\d]/g, '').slice(0, 10))

   493→                  aria-invalid={!!errors.phone}
   494→                  className={`rounded-l-none ${errors.phone ? 'border-destructive' : ''}`}

   496→              </div>
   497→              {errors.phone && (
   498→                <p className="text-xs text-destructive">{errors.phone}</p>
   499→              )}
   500→            </div>
   501→
   502→            {/* Age Group */}
   503→            <div className="space-y-2">
   504→              <Label>Age Group</Label>
   505→              <Select
   506→                value={formData.ageGroup}
   507→                onValueChange={(v) => updateField('ageGroup', v)}

   510→                  <SelectValue placeholder="Select age group" />

   512→                <SelectContent>
   513→                  {AGE_GROUPS.map((g) => (
   514→                    <SelectItem key={g} value={g}>
   515→                      {g}
   516→                    </SelectItem>
   517→                  ))}
   518→                </SelectContent>
   519→              </Select>
   520→            </div>
   521→
   522→            {/* Profession */}
   523→            <div className="space-y-2">
   524→              <Label>Profession</Label>
   525→              <Select
   526→                value={formData.profession}
   527→                onValueChange={(v) => updateField('profession', v)}
   528→              >
   529→                <SelectTrigger className="w-full">
   530→                  <SelectValue placeholder="Select your profession" />
   531→                </SelectTrigger>
   532→                <SelectContent>
   533→                  {PROFESSIONS.map((p) => (
   534→                    <SelectItem key={p} value={p}>
   535→                      {p}
   536→                    </SelectItem>
   537→                  ))}
   538→                </SelectContent>
   539→              </Select>
   540→            </div>
   541→
   542→            {/* Years of Experience */}
   543→            <div className="space-y-2">
   544→              <Label>Years of Experience</Label>

   546→                value={formData.experience}
   547→                onValueChange={(v) => updateField('experience', v)}
   548→              >
   549→                <SelectTrigger className="w-full">
   550→                  <SelectValue placeholder="Select experience" />
   551→                </SelectTrigger>
   552→                <SelectContent>
   553→                  {EXPERIENCE_LEVELS.map((e) => (
   554→                    <SelectItem key={e} value={e}>
   555→                      {e}
   556→                    </SelectItem>
   557→                  ))}
   558→                </SelectContent>
   559→              </Select>
   560→            </div>
   561→          </div>
   562→        </motion.div>
   563→      );
   564→    }
   565→
   566→    // Step 2 — Firm Information
   567→    if (currentStep === 2) {
   568→      const gstinStatus = getGstinStatus();
   569→      return (
   570→        <motion.div
   571→          key="step-2"
   572→          custom={direction}
   573→          variants={slideVariants}
   574→          initial="enter"
   575→          animate="center"
   576→          exit="exit"
   577→          transition={stepTransition}
   578→          className="py-4"
   579→        >
   580→          <div className="flex items-center gap-3 mb-6">
   581→            <div className="w-10 h-10 rounded-xl bg-emerald-100 flex items-center justify-center">
   582→              <Building2 className="size-5 text-emerald-600" />
   583→            </div>
   584→            <div>
   585→              <h2 className="text-2xl font-bold">Firm Information</h2>
   586→              <p className="text-sm text-muted-foreground">
   587→                Details about your CA firm or practice
   588→              </p>
   589→            </div>
   590→          </div>
   591→
   592→          <div className="space-y-5">
   593→            {/* Firm Name */}
   594→            <div className="space-y-2">
   595→              <Label htmlFor="firmName">
   596→                Firm Name <span className="text-destructive">*</span>
   597→              </Label>
   598→              <Input
   599→                id="firmName"
   600→                placeholder="Sharma & Associates"
   601→                value={formData.firmName}
   602→                onChange={(e) => updateField('firmName', e.target.value)}
   603→                aria-invalid={!!errors.firmName}
   604→                className={errors.firmName ? 'border-destructive' : ''}
   605→              />
   606→              {errors.firmName && (
   607→                <p className="text-xs text-destructive">{errors.firmName}</p>
   608→              )}
   609→            </div>
   610→
   611→            {/* Type of Organization */}

   618→                <SelectTrigger className="w-full">
