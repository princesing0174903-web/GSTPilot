    57→// Relative time formatter — defensively handles ISO string / Date / Firestore Timestamp.
    58→function timeAgo(createdAt: unknown): string {
    59→  if (!createdAt) return 'just now';
    60→  let date: Date;
    61→  if (createdAt instanceof Date) {
    62→    date = createdAt;
    63→  } else if (typeof createdAt === 'string') {
    64→    date = new Date(createdAt);
    65→  } else if (
    66→    typeof createdAt === 'object' &&
    67→    createdAt !== null &&
    68→    'toDate' in createdAt &&
    69→    typeof (createdAt as { toDate: () => Date }).toDate === 'function'
    70→  ) {
    71→    date = (createdAt as { toDate: () => Date }).toDate();
    72→  } else {
    73→    return 'just now';
    74→  }
    75→  if (isNaN(date.getTime())) return 'just now';
    76→  const seconds = Math.floor((Date.now() - date.getTime()) / 1000);
    77→  if (seconds < 60) return 'just now';
    78→  const minutes = Math.floor(seconds / 60);
    79→  if (minutes < 60) return `${minutes}m ago`;
    80→  const hours = Math.floor(minutes / 60);
    81→  if (hours < 24) return `${hours}h ago`;
    82→  const days = Math.floor(hours / 24);
    83→  if (days < 7) return `${days}d ago`;
    84→  const weeks = Math.floor(days / 7);
    85→  if (weeks < 5) return `${weeks}w ago`;
    86→  const months = Math.floor(days / 30);
    87→  if (months < 12) return `${months}mo ago`;
    88→  const years = Math.floor(days / 365);
    89→  return `${years}y ago`;
    90→}

    92→// ─── Business Score computation ───────────────────────────────────────────────
    93→// Start at 100. Subtract weighted penalties for risk signals. Clamp 0–100.
    94→// If there is no data at all, return null (UI shows premium empty state).
    95→
    96→function computeBusinessScore(opts: {
    97→  totalClients: number;
    98→  criticalIssues: number;
    99→  warnings: number;
   100→  pendingReturns: number;
   101→  overdueReturns: number;
   102→  averageHealthScore: number;
   103→  matchPercentage: number;
   104→}): number | null {
   105→  const { totalClients, criticalIssues, warnings, pendingReturns, overdueReturns, averageHealthScore, matchPercentage } = opts;
   106→  if (totalClients === 0 && pendingReturns === 0 && overdueReturns === 0) return null;
   107→
   108→  let score = 100;
   109→  score -= criticalIssues * 6;
   110→  score -= warnings * 2;
   111→  score -= overdueReturns * 8;
   112→  score -= pendingReturns * 2;
   113→  if (averageHealthScore > 0) {
   114→    score -= Math.max(0, 80 - averageHealthScore) * 0.6;
   115→  }
   116→  score -= Math.max(0, 100 - matchPercentage) * 0.3;
   117→  return Math.max(0, Math.min(100, Math.round(score)));
   118→}

   120→// V16 palette — only Emerald / Cyan / Blue. No amber, no red.
   121→function scoreTier(score: number | null): { label: string; color: string } {
   122→  if (score === null) return { label: '—', color: 'text-muted-foreground' };
   123→  if (score >= 85) return { label: 'Excellent', color: 'text-emerald-400' };
   124→  if (score >= 70) return { label: 'Healthy', color: 'text-cyan-400' };
   125→  if (score >= 50) return { label: 'At Risk', color: 'text-blue-400' };
   126→  return { label: 'Critical', color: 'text-blue-300' };
   127→}

   143→function buildInsights(opts: {
   144→  metrics: {
   145→    totalClients: number;
   146→    pendingReturns: number;
   147→    overdueReturns: number;
   148→    criticalIssues: number;
   149→    warnings: number;
   150→    totalTaxVolume: number;
   151→    averageHealthScore: number;
   152→    matchPercentage: number;
   153→    upcomingFilings: FirestoreReturn[];
   154→  };
   155→  clients: Array<FirestoreClient & { id: string }>;
   156→}): AIInsight[] {
   157→  const { metrics, clients } = opts;
   158→  const insights: AIInsight[] = [];
   159→
   160→  // 1. Overdue returns — most urgent
   161→  if (metrics.overdueReturns > 0) {
   162→    insights.push({
   163→      id: 'overdue-returns',
   164→      icon: AlertTriangle,
   165→      tone: 'risk',
   166→      what: `${metrics.overdueReturns} GST return${metrics.overdueReturns > 1 ? 's are' : ' is'} overdue`,
   167→      why: 'Overdue returns trigger notices and late fees. File immediately to avoid penalties.',
   168→      actionLabel: 'File Now',
   169→      actionView: 'returns',
   170→    });
   171→  }
   172→
   173→  // 2. Pending returns due soon
   174→  const nextFiling = metrics.upcomingFilings[0];
   175→  if (nextFiling && metrics.pendingReturns > 0 && metrics.overdueReturns === 0) {
   176→    insights.push({
   177→      id: 'gst-due',
   178→      icon: Clock,
   179→      tone: 'info',
   180→      what: `GST filing due — ${nextFiling.returnType} for ${nextFiling.period}`,
   181→      why: `${metrics.pendingReturns} return${metrics.pendingReturns > 1 ? 's' : ''} pending. Filing on time keeps compliance score high.`,
   182→      actionLabel: 'File GST',
   183→      actionView: 'returns',
   184→    });
   185→  }
   186→
   187→  // 3. Critical issues / risky clients
   188→  const riskyClients = clients.filter(c => (c.healthScore ?? 100) < 60);
   189→  if (metrics.criticalIssues > 0 || riskyClients.length > 0) {
   190→    insights.push({
   191→      id: 'churn-risk',
   192→      icon: TrendingDown,
   193→      tone: 'risk',
   194→      what: riskyClients.length > 0
   195→        ? `${riskyClients.length} client${riskyClients.length > 1 ? 's may' : ' may'} churn`
   196→        : `${metrics.criticalIssues} critical issue${metrics.criticalIssues > 1 ? 's' : ''} need attention`,
   197→      why: riskyClients.length > 0
   198→        ? 'Low health scores signal dissatisfaction. Reach out before they leave.'
   199→        : 'Critical issues compound into bigger problems if left unaddressed.',
   200→      actionLabel: riskyClients.length > 0 ? 'View Clients' : 'Review Issues',
   201→      actionView: riskyClients.length > 0 ? 'clients' : 'ai-operating-room',
   202→    });
   203→  }
   204→
   205→  // 4. Cash position / collections
   206→  if (metrics.totalTaxVolume > 0) {
   207→    insights.push({
   208→      id: 'collections',
   209→      icon: Wallet,
   210→      tone: metrics.matchPercentage < 90 ? 'risk' : 'info',
   211→      what: metrics.matchPercentage < 90
   212→        ? 'Collections may drop — reconciliation gaps detected'
   213→        : 'Cash position is healthy',
   214→      why: metrics.matchPercentage < 90
   215→        ? `${metrics.matchPercentage.toFixed(0)}% of invoices reconciled. Unmatched invoices mean uncollected revenue.`
   216→        : `${metrics.matchPercentage.toFixed(0)}% invoice match rate. Keep chasing pending collections to maintain runway.`,
   217→      actionLabel: 'Fix Collections',
   218→      actionView: 'reconcile',
   219→    });
   220→  }
   221→
   222→  // 5. Compliance score insight
   223→  if (metrics.averageHealthScore > 0) {
   224→    const tier = metrics.averageHealthScore >= 80 ? 'success' : 'info';
   225→    insights.push({
   226→      id: 'compliance',
   227→      icon: ShieldAlert,
   228→      tone: tier,
   229→      what: `Compliance health at ${metrics.averageHealthScore.toFixed(0)}%`,
   230→      why: metrics.averageHealthScore >= 80
   231→        ? 'Strong compliance posture. Maintain by filing on time and resolving warnings.'
   232→        : 'Below optimal. Address pending returns and warnings to lift your score.',
   233→      actionLabel: 'View Compliance',
   234→      actionView: 'returns',
   235→    });
   236→  }
   237→
   238→  // 6. Always-on "all clear" nudge if nothing urgent
   239→  if (insights.length === 0) {
   240→    insights.push({
   241→      id: 'all-clear',
   242→      icon: Sparkles,
   243→      tone: 'success',
   244→      what: 'Everything looks good today',
   245→      why: 'No urgent issues. Hand control to Oracle and let it run the routine work.',
   246→      actionLabel: 'Run My Business',
   247→      actionView: 'run-my-business',
   248→    });
   249→  }
   250→
   251→  return insights.slice(0, 6);
   252→}

   256→function ScoreGauge({ score }: { score: number | null }) {
   257→  const size = 220;
   258→  const stroke = 14;
   259→  const r = (size - stroke) / 2;
   260→  const c = 2 * Math.PI * r;
   261→  const pct = score === null ? 0 : score / 100;
   262→  const offset = c * (1 - pct);
   263→
   264→  return (
   265→    <div className="relative" style={{ width: size, height: size }}>
   266→      <svg width={size} height={size} className="-rotate-90">
   267→        <defs>
   268→          <linearGradient id="missionScoreGradient" x1="0%" y1="0%" x2="100%" y2="100%">
   269→            <stop offset="0%" stopColor="#10b981" />
   270→            <stop offset="50%" stopColor="#06b6d4" />
   271→            <stop offset="100%" stopColor="#3b82f6" />
   272→          </linearGradient>
   273→        </defs>
   274→        <circle
   275→          cx={size / 2}
   276→          cy={size / 2}
   277→          r={r}
   278→          fill="none"
   279→          stroke="rgba(255,255,255,0.06)"
   280→          strokeWidth={stroke}
   281→        />
   282→        {score !== null && (
   283→          <motion.circle
   284→            cx={size / 2}
   285→            cy={size / 2}
   286→            r={r}
   287→            fill="none"
   288→            stroke="url(#missionScoreGradient)"
   289→            strokeWidth={stroke}
   290→            strokeLinecap="round"
   291→            strokeDasharray={c}
   292→            initial={{ strokeDashoffset: c }}
   293→            animate={{ strokeDashoffset: offset }}
   294→            transition={{ duration: 1.2, ease: 'easeOut', delay: 0.2 }}
   295→          />
   296→        )}
   297→      </svg>
   298→      <div className="absolute inset-0 flex flex-col items-center justify-center">
   299→        {score === null ? (
   300→          <span className="text-4xl font-bold text-muted-foreground">—</span>
   301→        ) : (
   302→          <motion.span
   303→            initial={{ opacity: 0, scale: 0.8 }}
   304→            animate={{ opacity: 1, scale: 1 }}
   305→            transition={{ duration: 0.5, delay: 0.6 }}
   306→            className={`text-6xl font-bold tracking-tight ${scoreTier(score).color}`}
   307→          >
   308→            {score}
   309→          </motion.span>
   310→        )}
   311→        <span className="text-xs text-muted-foreground tracking-wider uppercase mt-1">
   312→          / 100
   313→        </span>
   314→      </div>
   315→    </div>
   316→  );
   317→}
