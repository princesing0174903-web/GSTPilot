    66→export async function POST(req: NextRequest) {
    67→  let question = ''
    68→  let context = ''
    69→  try {
    70→    const body = await req.json()
    71→    question = (body.question || '').trim()
    72→    context = (body.context || '').trim()
    73→    const conversationHistory: Array<{ role: string; content: string }> =
    74→      body.conversationHistory || []
    75→
    76→    if (!question) {
    77→      return NextResponse.json({ error: 'Question is required' }, { status: 400 })
    78→    }
    79→
    80→    // 1) Deterministic, fast intent + actions + thinking steps
    81→    const intent = detectIntent(question)
    82→    const thinkingSteps = generateThinkingSteps(intent)
    83→    const detectedActions = detectActions(question, intent)
    84→
    85→    // 2) Determine whether real data is available in context.
    86→    //    The frontend builds context as a structured string starting with
    87→    //    "LIVE DATA SNAPSHOT" and a list of "- key: value" lines.
    88→    const hasLiveData = contextHasLiveData(context)
    89→
    90→    // 3) Build sources from ACTUAL context content (no fabrication)
    91→    const sources = hasLiveData ? buildSources(context) : EMPTY_CONTEXT_SOURCES
    92→
    93→    // 4) Generate the answer via LLM (or fallback)
    94→    let answer: string
    95→    if (!hasLiveData) {
    96→      // Skip LLM call entirely when no live data — guarantees no fabrication
    97→      answer = EMPTY_CONTEXT_ANSWER
    98→    } else {
    99→      try {
   100→        answer = await generateLLMAnswer(question, context, conversationHistory)
   101→      } catch (llmError) {
   102→        console.error(
   103→          '[Oracle] LLM error:',
   104→          llmError instanceof Error ? llmError.message : llmError,
   105→        )
   106→        answer = generateContextualFallback(question, context, intent)
   107→      }
   108→      if (!answer || !answer.trim()) {
   109→        answer = generateContextualFallback(question, context, intent)
   110→      }
   111→    }
   112→
   113→    // 5) Extract structured insights from the LLM answer (or derive from context)
   114→    const insights = hasLiveData
   115→      ? extractInsights(answer, context, intent)
   116→      : [
   117→          {
   118→            text: 'Connect your business data to unlock AI insights.',
   119→            tone: 'neutral' as InsightTone,
   120→          },
   121→        ]
   122→
   123→    // 6) Suggested follow-ups based on intent (and data availability)
   124→    const suggestedPrompts = hasLiveData
   125→      ? suggestFollowUps(intent, question)
   126→      : [
   127→          'Show pending returns',
   128→          'Which clients are risky?',
   129→          'What revenue will I make next month?',
   130→        ]
   131→
   132→    const responseBody: OracleResponse = {
   133→      answer,
   134→      insights,
   135→      sources,
   136→      actions: detectedActions,
   137→      suggestedPrompts,
   138→      intent,
   139→      thinkingSteps,
   140→    }
   141→
   142→    return NextResponse.json(responseBody)
   143→  } catch (error: unknown) {
   144→    console.error(
   145→      '[Oracle] Fatal:',
   146→      error instanceof Error ? error.message : error,
   147→    )
   148→    // Always return the structured shape — frontend never has to guard
   149→    const fallbackResponse: OracleResponse = {
   150→      answer: EMPTY_CONTEXT_ANSWER,
   151→      insights: [
   152→        {
   153→          text: 'I had trouble reaching my reasoning engine. Please try again.',
   154→          tone: 'warning',
   155→        },
   156→      ],
   157→      sources: EMPTY_CONTEXT_SOURCES,
   158→      actions: [],
   159→      suggestedPrompts: [
   160→        'Show pending returns',
   161→        'Which clients are risky?',
   162→        'What revenue will I make next month?',
   163→      ],
   164→      intent: 'fallback',
   165→      thinkingSteps: [
   166→        { label: 'Thinking...', duration: 400 },
   167→        { label: 'Generating answer...', duration: 500 },
   168→      ],
   169→    }
   170→    return NextResponse.json(fallbackResponse)
   171→  }
   172→}

   176→async function generateLLMAnswer(
   177→  question: string,
   178→  context: string,
   179→  conversationHistory: Array<{ role: string; content: string }>,
   180→): Promise<string> {
   181→  const ZAI = (await import('z-ai-web-dev-sdk')).default
   182→  const zai = await ZAI.create()
   183→
   184→  const systemPrompt = `You are GSTPilot Intelligence™ — the AI Brain of the user's Indian CA firm / business.
   185→
   186→You have LIVE access to the firm's Firestore data (clients, invoices, returns, payments, notifications, activities). The current data snapshot is provided below as "Live Business Context".
   187→
   188→## Your job
   189→1. Answer the user's question accurately using ONLY the live data below.
   190→2. Reference SPECIFIC NUMBERS, CLIENT NAMES, AMOUNTS, and DATES from the context.
   191→3. Use Indian number formatting (₹1,23,456 — lakhs/crores, no millions) and DD/MM/YYYY dates.
   192→4. Structure your answer as:
   193→   • A concise 2-3 sentence direct answer FIRST (no header, no preamble).
   194→   • Then 2-4 bullet points (use "• " prefix) with the most important insights,
   195→     numbers, or breakdowns supporting your answer.
   196→   • Then an optional "Recommended Actions:" line if there is a clear next step.
   197→5. Match the user's tone — professional but warm, like a senior CA partner.
   198→6. If the user asks about running automation ("run my firm", "run my business"),
   199→   describe what would happen step-by-step in 5-7 numbered steps.
   200→7. NEVER invent data. If the context does not contain what you need, say so
   201→   explicitly: "I don't have enough live data right now" and suggest what to
   202→   connect or check.
   203→8. Do NOT use markdown headers (#), do NOT use emojis, do NOT use tables.
   204→9. Keep total response under 200 words.
   205→
   206→Live Business Context:
   207→${context}
   208→
   209→Conversation history (most recent last):
   210→${conversationHistory
   211→  .slice(-4)
   212→  .map((m) => `${m.role}: ${m.content}`)
   213→  .join('\n') || '[No prior conversation]'}`
   214→
   215→  const response = await zai.chat.completions.create({
   216→    messages: [
   217→      { role: 'system', content: systemPrompt },
   218→      { role: 'user', content: question },
   219→    ],
   220→  })
   221→
   222→  return response.choices[0]?.message?.content || ''
   223→}

   227→function detectIntent(question: string): string {
   228→  const q = question.toLowerCase()
   229→  if (
   230→    /\b(run|execute|start|launch|automate)\b.*\b(firm|business|company|pipeline|workflow)\b/.test(
   231→      q,
   232→    ) ||
   233→    q.includes('run my firm') ||
   234→    q.includes('run my business') ||
   235→    q.includes('run my company')
   236→  )
   237→    return 'run_automation'
   238→  if (
   239→    q.includes('overload') ||
   240→    q.includes('overloaded') ||
   241→    q.includes('workload') ||
   242→    q.includes('capacity') ||
   243→    q.includes('bandwidth')
   244→  )
   245→    return 'workload'
   246→  if (
   247→    q.includes('revenue') ||
   248→    q.includes('income') ||
   249→    q.includes('earn') ||
   250→    q.includes('next month') ||
   251→    q.includes('forecast') ||
   252→    q.includes('predict')
   253→  )
   254→    return 'revenue_forecast'
   255→  if (
   256→    q.includes('collection') ||
   257→    q.includes('drop') ||
   258→    q.includes('decline') ||
   259→    q.includes('fell') ||
   260→    q.includes('why did')
   261→  )
   262→    return 'collection_analysis'
   263→  if (q.includes('risk') || q.includes('risky') || q.includes('health') || q.includes('at risk'))
   264→    return 'risk_analysis'
   265→  if (
   266→    q.includes('pending') ||
   267→    q.includes('overdue') ||
   268→    q.includes('filing') ||
   269→    q.includes('gstr') ||
   270→    q.includes('return')
   271→  )
   272→    return 'pending_returns'
   273→  if (
   274→    q.includes('invoice') ||
   275→    q.includes('payment') ||
   276→    q.includes('outstanding') ||
   277→    q.includes('collect')
   278→  )
   279→    return 'invoice_collection'
   280→  if (q.includes('cash') || q.includes('flow') || q.includes('liquidity')) return 'cash_flow'
   281→  // GST notices / show-cause / scrutiny — restore dedicated notices intent
   282→  if (
   283→    q.includes('notice') ||
   284→    q.includes('notices') ||
   285→    q.includes('show cause') ||
   286→    q.includes('show-cause') ||
   287→    q.includes('scrutiny') ||
   288→    q.includes('asd') ||
   289→    q.includes('drc-01') ||
   290→    q.includes('drc01')
   291→  )
   292→    return 'notices'
   293→  // ITC (Input Tax Credit) — restore compliance intent for ITC queries
   294→  if (
   295→    q.includes('itc') ||
   296→    q.includes('input tax credit') ||
   297→    q.includes('input credit') ||
   298→    q.includes('itc loss') ||
   299→    q.includes('itc mismatch') ||
   300→    q.includes('itc suggestion') ||
   301→    q.includes('itc optim')
   302→  )
   303→    return 'compliance'
   304→  if (q.includes('compliance') || q.includes('score') || q.includes('deadline')) return 'compliance'
   305→  if (q.includes('client') || q.includes('customer')) return 'client_overview'
   306→  if (q.includes('today') || q.includes('priority') || q.includes('should i') || q.includes('to do'))
   307→    return 'daily_priority'
   308→  if (q.includes('compare') || q.includes('last month') || q.includes('vs') || q.includes('trend'))
   309→    return 'comparison'
   310→  if (q.includes('task') || q.includes('reminder') || q.includes('notify')) return 'task_management'
   311→  if (q.includes('report') || q.includes('summary') || q.includes('statement')) return 'report'
   312→  return 'general'
   313→}

   589→// ─── Sources Builder (parses LIVE context string for real counts) ─────────────
   590→
   591→/**
   592→ * The frontend composes a context string starting with "LIVE DATA SNAPSHOT"
   593→ * followed by "- key: value" lines. We parse those lines to surface the
   594→ * real sources + record counts the LLM consulted. NO fabrication — if a
   595→ * section is missing or zero, the source is omitted.
   596→ */
   597→function buildSources(context: string): OracleSource[] {
   598→  const sources: OracleSource[] = []
   599→
   600→  const clients = parseCount(context, /-?\s*Total clients:\s*(\d+)/i)
   601→  if (clients > 0) {
   602→    sources.push({ name: 'Clients', count: clients, icon: 'users' })
   603→  }
   604→
   605→  const invoices = parseCount(context, /-?\s*Active invoices:\s*(\d+)/i)
   606→  if (invoices > 0) {
   607→    sources.push({ name: 'Invoices', count: invoices, icon: 'file-text' })
   608→  }
   609→
   610→  const filed = parseCount(context, /-?\s*Filed returns:\s*(\d+)/i)
   611→  // Frontend emits "- Pending returns: 3 GSTR-1, 2 GSTR-3B"
   612→  // The count comes BEFORE the return-type label, so capture the leading digit.
   613→  const gstr1Pending = parseCount(context, /-?\s*Pending returns:\s*(\d+)\s*GSTR-1/i)
   614→  const gstr3bPending = parseCount(context, /GSTR-1,?\s*(\d+)\s*GSTR-3B/i)
   615→  const totalReturns = filed + gstr1Pending + gstr3bPending
   616→  if (totalReturns > 0) {
   617→    sources.push({ name: 'GST Returns', count: totalReturns, icon: 'receipt' })
   618→  }
   619→
   620→  const unread = parseCount(context, /-?\s*Unread notifications:\s*(\d+)/i)
   621→  if (unread > 0) {
   622→    sources.push({ name: 'Notifications', count: unread, icon: 'bell' })
   623→  }
   624→
   625→  const activities = parseCount(context, /-?\s*Recent activities:\s*(\d+)/i)
   626→  if (activities > 0) {
   627→    sources.push({ name: 'Activities', count: activities, icon: 'activity' })
   628→  }
   629→
   630→  // If we somehow couldn't parse anything but the context clearly has live data,
   631→  // return a single generic source so the UI has something to show.
   632→  if (sources.length === 0 && contextHasLiveData(context)) {
   633→    sources.push({ name: 'Business Data', count: 0, icon: 'database' })
   634→  }
   635→
   636→  return sources
   637→}

   639→function parseCount(context: string, pattern: RegExp): number {
   640→  const match = context.match(pattern)
   641→  if (!match || !match[1]) return 0
   642→  const n = parseInt(match[1], 10)
   643→  return Number.isFinite(n) ? n : 0
   644→}

   646→function contextHasLiveData(context: string): boolean {
   647→  if (!context || context.length < 60) return false
   648→  // The frontend always emits this header when live data is available
   649→  if (!/LIVE DATA SNAPSHOT/i.test(context)) return false
   650→  // If the snapshot explicitly says data unavailable, treat as empty
   651→  if (/Data currently unavailable/i.test(context)) return false
   652→  // Must have at least one "- key: value" data line beyond the header
   653→  const dataLines = context
   654→    .split('\n')
   655→    .filter((l) => /^\s*-\s+\w/.test(l))
   656→  return dataLines.length >= 2
   657→}

   659→// ─── Insight Extractor (parses LLM answer for 2-4 bullets) ────────────────────

   661→/**
   662→ * Extracts 2-4 insight bullets from the LLM answer. Falls back to
   663→ * context-derived insights when the answer has no parseable bullets.
   664→ */
   665→function extractInsights(
   666→  answer: string,
   667→  context: string,
   668→  intent: string,
   669→): OracleInsight[] {
   670→  // Parse bullet/numbered lines from the answer body
   671→  const rawLines = answer
   672→    .split('\n')
   673→    .map((l) => l.trim())
   674→    .filter(Boolean)
   675→
   676→  const bullets: string[] = []
   677→  for (const line of rawLines) {
   678→    let text: string | null = null
   679→    if (line.startsWith('• ') || line.startsWith('•')) {
   680→      text = line.replace(/^•\s*/, '').trim()
   681→    } else if (line.startsWith('- ')) {
   682→      text = line.replace(/^-\s*/, '').trim()
   683→    } else if (/^\d+\.\s/.test(line)) {
   684→      text = line.replace(/^\d+\.\s*/, '').trim()
   685→    }
   686→    if (text && text.length >= 12 && text.length <= 240) {
   687→      // Strip markdown bold/italic markers for cleaner insight text
   688→      const cleaned = text.replace(/\*\*/g, '').replace(/__/g, '').trim()
   689→      if (cleaned.length >= 12) bullets.push(cleaned)
   690→    }
   691→    if (bullets.length >= 4) break
   692→  }
   693→
   694→  if (bullets.length >= 2) {
   695→    return bullets.slice(0, 4).map((text) => ({ text, tone: detectTone(text) }))
   696→  }
   697→
   698→  // Fallback: derive insights from real context numbers
   699→  const derived = deriveInsightsFromContext(context, intent)
   700→  if (derived.length >= 1) {
   701→    // If we got 1 bullet from the answer, combine with 1 derived for 2 total
   702→    const combined: OracleInsight[] = []
   703→    if (bullets.length === 1) {
   704→      combined.push({ text: bullets[0], tone: detectTone(bullets[0]) })
   705→    }
   706→    for (const d of derived) {
   707→      if (combined.length >= 2) break
   708→      if (!combined.some((c) => c.text === d.text)) combined.push(d)
   709→    }
   710→    if (combined.length >= 1) return combined.slice(0, 4)
   711→  }
   712→
   713→  // Last-resort fallback: synthesize a single neutral insight from the answer
   714→  const firstSentence = answer
   715→    .split(/(?<=[.!?])\s+/)[0]
   716→    ?.replace(/[*#]/g, '')
   717→    .trim()
   718→  if (firstSentence && firstSentence.length >= 12) {
   719→    return [{ text: firstSentence.slice(0, 240), tone: 'neutral' }]
   720→  }
   721→  return [{ text: 'Insights will appear here once live data is connected.', tone: 'neutral' }]
   722→}
   723→
   724→function detectTone(text: string): InsightTone {
   725→  const t = text.toLowerCase()
   726→  // Warning signals — risk/decline/overdue keywords
   727→  if (
   728→    /\b(overdue|drop|decline|declined|fell|missed|urgent|risk|risky|late|below|fail|failed|exposed|shortfall|delay|delayed|warning|alert|critical|breach|non-compliant|mismatch|gap)\b/.test(
   729→      t,
   730→    )
   731→  ) {
   732→    return 'warning'
   733→  }
   734→  // Positive signals — growth/improvement keywords
   735→  if (
   736→    /\b(growth|grew|increase|increased|up|healthy|profit|saved|completed|filed|recovered|improved|improvement|surplus|above|strong|stable|on track|on-time)\b/.test(
   737→      t,
   738→    )
   739→  ) {
   740→    return 'positive'
   741→  }
   742→  return 'neutral'
   743→}

   745→function deriveInsightsFromContext(context: string, intent: string): OracleInsight[] {
   746→  const insights: OracleInsight[] = []
   747→
   748→  const clients = parseCount(context, /-?\s*Total clients:\s*(\d+)/i)
   749→  const activeClients = parseCount(context, /-?\s*Active clients:\s*(\d+)/i)
   750→  const invoices = parseCount(context, /-?\s*Active invoices:\s*(\d+)/i)
   751→  const overdueInvoices = parseCount(context, /-?\s*Overdue invoices:\s*(\d+)/i)
   752→  const filed = parseCount(context, /-?\s*Filed returns:\s*(\d+)/i)
   753→  const gstr1Pending = parseCount(context, /-?\s*Pending returns:\s*(\d+)\s*GSTR-1/i)
   754→
   755→  // High-risk clients line (text after the colon)
   756→  const riskMatch = context.match(/-?\s*High-risk clients:\s*(.+)/i)
   757→  const riskLine = riskMatch?.[1]?.trim()
   758→  const highRiskCount = riskLine
   759→    ? (riskLine.match(/,/g)?.length ?? 0) + (riskLine.length > 0 ? 1 : 0)
   760→    : 0
   761→
   762→  // Top client line
   763→  const topMatch = context.match(/-?\s*Top 5 clients by revenue:\s*(.+)/i)
   764→  const topLine = topMatch?.[1]?.trim()
   765→
   766→  if (intent === 'risk_analysis' && highRiskCount > 0 && riskLine) {
   767→    insights.push({
   768→      text: `${highRiskCount} client${highRiskCount > 1 ? 's' : ''} flagged high-risk — ${riskLine.slice(0, 140)}.`,
   769→      tone: 'warning',
   770→    })
   771→  }
   772→
   773→  if (intent === 'pending_returns' && (gstr1Pending > 0 || filed > 0)) {
   774→    insights.push({
   775→      text: `${gstr1Pending} GSTR-1 ${gstr1Pending === 1 ? 'return' : 'returns'} pending, ${filed} already filed.`,
   776→      tone: gstr1Pending > 0 ? 'warning' : 'positive',
   777→    })
   778→  }
   779→
   780→  if (
   781→    (intent === 'invoice_collection' || intent === 'collection_analysis') &&
   782→    (overdueInvoices > 0 || invoices > 0)
   783→  ) {
   784→    insights.push({
   785→      text: `${overdueInvoices} overdue invoice${overdueInvoices === 1 ? '' : 's'} out of ${invoices} active.`,
   786→      tone: overdueInvoices > 0 ? 'warning' : 'positive',
   787→    })
   788→  }
   789→
   790→  if (intent === 'client_overview' && clients > 0) {
   791→    insights.push({
   792→      text: `${clients} total client${clients === 1 ? '' : 's'} (${activeClients} active).`,
   793→      tone: 'neutral',
   794→    })
   795→  }
   796→
   797→  if (intent === 'revenue_forecast' && topLine) {
   798→    insights.push({
   799→      text: `Top revenue clients: ${topLine.slice(0, 140)}.`,
   800→      tone: 'positive',
   801→    })
   802→  }
   803→
   804→  if (intent === 'compliance' && filed > 0) {
   805→    insights.push({
   806→      text: `${filed} return${filed === 1 ? '' : 's'} filed so far this period.`,
   807→      tone: 'positive',
   808→    })
   809→  }
   810→
   811→  if (intent === 'notices' && clients > 0) {
   812→    insights.push({
   813→      text: `Notice Center tracks ASD, DRC-01, show-cause, and scrutiny notices across ${clients} client${clients === 1 ? '' : 's'}.`,
   814→      tone: 'neutral',
   815→    })
   816→    if (gstr1Pending > 0) {
   817→      insights.push({
   818→        text: `${gstr1Pending} pending GSTR-1 ${gstr1Pending === 1 ? 'return' : 'returns'} — late filings often trigger auto-notices.`,
   819→        tone: 'warning',
   820→      })
   821→    }
   822→  }
   823→
   824→  return insights
   825→}
