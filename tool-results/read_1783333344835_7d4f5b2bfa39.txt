     2→     2→// GSTPilot — Real-time Firestore Hooks
     3→     3→// React hooks with onSnapshot listeners for live data + optimistic UI.
     4→     4→//
     5→     5→// ARCHITECTURE (Production-grade multi-tenant):
     6→     6→//   • Every query is scoped by `organizationId` from OrgContext — NEVER by the
     7→     7→//     legacy `firmId`. This matches the Firestore security rules, which require
     8→     8→//     every tenant-scoped document to carry `organizationId` and only allow
     9→     9→//     reads where `isOrgMember(resource.data.organizationId)` is true.
    10→    10→//   • If there is no organization (preview / demo mode, or not yet loaded),
    11→    11→//     hooks return empty data WITHOUT subscribing — so the UI renders premium
    12→    12→//     empty states instead of throwing permission errors.
    13→    13→//   • Permission errors (permission-denied / unauthenticated) are treated as
    14→    14→//     "no data available" — the hook clears its error and returns an empty
    15→    15→//     array. This is the graceful-degradation contract: the app NEVER shows a
    16→    16→//     "Missing or insufficient permissions" wall. Genuine errors (network,
    17→    17→//     index missing) still surface as `error`.

    45→    45→// ─── Permission-error detection (graceful degradation) ───────────────────────
    46→    46→
    47→    47→/**
    48→    48→ * True if the given Firestore error is a permission / auth failure that we
    49→    49→ * should treat as "no data available" rather than a hard error. This is the
    50→    50→ * core of the never-show-a-permission-wall contract.
    51→    51→ */
    52→    52→function isPermissionError(err: unknown): boolean {
    53→    53→  if (!err) return false;
    54→    54→  const e = err as { code?: string; message?: string };
    55→    55→  const code = (e.code || '').toLowerCase();
    56→    56→  const message = (e.message || '').toLowerCase();
    57→    57→  if (
    58→    58→    code === 'permission-denied' ||
    59→    59→    code === 'unauthenticated' ||
    60→    60→    code === 'auth/operation-not-allowed' ||
    61→    61→    code === 'auth/user-not-found'
    62→    62→  ) {
    63→    63→    return true;
    64→    64→  }
    65→    65→  if (
    66→    66→    message.includes('missing or insufficient permissions') ||
    67→    67→    message.includes('permission-denied') ||
    68→    68→    message.includes('insufficient permissions') ||
    69→    69→    message.includes('not authorized') ||
    70→    70→    message.includes('unauthenticated')
    71→    71→  ) {
    72→    72→    return true;
    73→    73→  }
    74→    74→  return false;
    75→    75→}

    77→    77→// ─── Timestamp Converter ─────────────────────────────────────────────────────
    78→    78→
    79→    79→function convertDoc<T extends Record<string, unknown>>(snapData: Record<string, unknown>): T {
    80→    80→  const converted: Record<string, unknown> = {};
    81→    81→  for (const [key, value] of Object.entries(snapData)) {
    82→    82→    if (value && typeof value === 'object' && 'toDate' in value && typeof (value as { toDate: () => Date }).toDate === 'function') {
    83→    83→      converted[key] = (value as { toDate: () => Date }).toDate().toISOString();
    84→    84→    } else if (value && typeof value === 'object' && !Array.isArray(value) && !(value instanceof Date)) {
    85→    85→      // Recursively convert nested objects
    86→    86→      converted[key] = convertDoc(value as Record<string, unknown>);
    87→    87→    } else {
    88→    88→      converted[key] = value;
    89→    89→    }
    90→    90→  }
    91→    91→  return converted as T;
    92→    92→}

    94→    94→// ─── Generic Collection Hook ─────────────────────────────────────────────────
    95→    95→//
    96→    96→// The hook resolves the current `organizationId` from OrgContext. If the org
    97→    97→// isn't resolved yet (loading) OR we're in preview mode (no org), the hook
    98→    98→// returns empty data without subscribing. This guarantees no permission errors
    99→    99→// ever reach the UI in preview/offline mode.
   100→   100→
   101→   101→function useFirestoreCollection<T>(
   102→   102→  collectionName: CollectionName,
   103→   103→  constraints: QueryConstraint[] = [],
   104→   104→  deps: unknown[] = [],
   105→   105→): { data: Array<T & { id: string }>; loading: boolean; error: string | null } {
   106→   106→  const { user } = useAuth();
   107→   107→  const { organization, isPreviewMode, loading: orgLoading } = useOrg();
   108→   108→  const organizationId = organization?.id ?? null;
   109→   109→
   110→   110→  const [data, setData] = useState<Array<T & { id: string }>>([]);
   111→   111→  const [loading, setLoading] = useState(true);
   112→   112→  const [error, setError] = useState<string | null>(null);
   113→   113→
   114→   114→  useEffect(() => {
   115→   115→    // No user → nothing to subscribe to (empty data).
   116→   116→    if (!user) {
   117→   117→      setData([]);
   118→   118→      setLoading(false);
   119→   119→      setError(null);
   120→   120→      return;
   121→   121→    }
   122→   122→
   123→   123→    // Org still resolving — stay in loading state, but don't subscribe yet.
   124→   124→    if (orgLoading && !organizationId) {
   125→   125→      return;
   126→   126→    }
   127→   127→
   128→   128→    // Preview mode (no real org) OR no organization at all → empty data,
   129→   129→    // no subscription. This is the preview-mode contract.
   130→   130→    if (!organizationId || isPreviewMode || organizationId === 'preview-org') {
   131→   131→      setData([]);
   132→   132→      setLoading(false);
   133→   133→      setError(null);
   134→   134→      return;
   135→   135→    }
   136→   136→
   137→   137→    let unsub: Unsubscribe | null = null;
   138→   138→
   139→   139→    // Scope EVERY query by organizationId. This is required by the Firestore
   140→   140→    // security rules — unbounded collection reads are denied.
   141→   141→    const baseConstraints: QueryConstraint[] = [
   142→   142→      where('organizationId', '==', organizationId),
   143→   143→      ...constraints,
   144→   144→    ];
   145→   145→
   146→   146→    const q = query(collection(db, collectionName), ...baseConstraints);
   147→   147→
   148→   148→    unsub = onSnapshot(q, (snap) => {
   149→   149→      const items = snap.docs.map(d => ({
   150→   150→        id: d.id,
   151→   151→        ...convertDoc<T>(d.data() as Record<string, unknown>),
   152→   152→      }));
   153→   153→      setData(items);
   154→   154→      setLoading(false);
   155→   155→      setError(null);
   156→   156→    }, (err) => {
   157→   157→      console.warn(`[Firestore] ${collectionName} subscription error:`, err);
   158→   158→      // Permission errors → graceful degradation (empty data, no error wall).
   159→   159→      if (isPermissionError(err)) {
   160→   160→        setData([]);
   161→   161→        setError(null);
   162→   162→        setLoading(false);
   163→   163→        return;
   164→   164→      }
   165→   165→      // Genuine errors (network, missing index) still surface.
   166→   166→      setError(err.message);
   167→   167→      setLoading(false);
   168→   168→    });
   169→   169→
   170→   170→    return () => {
   171→   171→      if (unsub) unsub();
   172→   172→    };
   173→   173→    // eslint-disable-next-line react-hooks/exhaustive-deps
   174→   174→  }, [user, organizationId, isPreviewMode, orgLoading, ...deps]);
   175→   175→
   176→   176→  // When user is null, return empty data (not loading)
   177→   177→  if (!user && loading) {
   178→   178→    return { data: [], loading: false, error: null };
   179→   179→  }
   180→   180→
   181→   181→  return { data, loading, error };
   182→   182→}

   184→   184→// ─── Generic Document Hook ───────────────────────────────────────────────────
   185→   185→
   186→   186→function useFirestoreDoc<T>(
   187→   187→  collectionName: CollectionName,
   188→   188→  docId: string | null,
   189→   189→): { data: (T & { id: string }) | null; loading: boolean; error: string | null } {
   190→   190→  const { user } = useAuth();
   191→   191→  const { organization, isPreviewMode } = useOrg();
   192→   192→  const organizationId = organization?.id ?? null;
   193→   193→
   194→   194→  const [data, setData] = useState<(T & { id: string }) | null>(null);
   195→   195→  const [loading, setLoading] = useState(!!docId);
   196→   196→  const [error, setError] = useState<string | null>(null);
   197→   197→
   198→   198→  useEffect(() => {
   199→   199→    if (!docId) {
   200→   200→      setData(null);
   201→   201→      setLoading(false);
   202→   202→      return;
   203→   203→    }
   204→   204→
   205→   205→    // Preview mode (no real org) → no doc, no subscription.
   206→   206→    if (!user || !organizationId || isPreviewMode || organizationId === 'preview-org') {
   207→   207→      setData(null);
   208→   208→      setLoading(false);
   209→   209→      setError(null);
   210→   210→      return;
   211→   211→    }
   212→   212→
   213→   213→    const unsub = onSnapshot(doc(db, collectionName, docId), (snap) => {
   214→   214→      if (!snap.exists()) {
   215→   215→        setData(null);
   216→   216→        setLoading(false);
   217→   217→        return;
   218→   218→      }
   219→   219→      setData({
   220→   220→        id: snap.id,
   221→   221→        ...convertDoc<T>(snap.data() as Record<string, unknown>),
   222→   222→      });
   223→   223→      setLoading(false);
   224→   224→      setError(null);
   225→   225→    }, (err) => {
   226→   226→      console.warn(`[Firestore] ${collectionName}/${docId} doc error:`, err);
   227→   227→      if (isPermissionError(err)) {
   228→   228→        setData(null);
   229→   229→        setError(null);
   230→   230→        setLoading(false);
   231→   231→        return;
   232→   232→      }
   233→   233→      setError(err.message);
   234→   234→      setLoading(false);
   235→   235→    });
   236→   236→
   237→   237→    return () => {
   238→   238→      if (unsub) unsub();
   239→   239→    };
   240→   240→  }, [collectionName, docId, user, organizationId, isPreviewMode]);
   241→   241→
   242→   242→  return { data, loading, error };
   243→   243→}

   460→   460→// ─── Executive Scores (computed from live data) ─────────────────────────────
   461→   461→
   462→   462→export function useFirmExecutiveScores(): { scores: FirmExecutiveScores; loading: boolean } {
   463→   463→  const { data: clients } = useFireClients();
   464→   464→  const { data: returns } = useFireReturns();
   465→   465→  const { data: invoices } = useFireInvoices();
   466→   466→  const { data: reconciliations } = useFireReconciliations();
   467→   467→  const { data: activities } = useFireRecentActivities(50);
   468→   468→
   469→   469→  const loading = clients.length === 0 && returns.length === 0;
   470→   470→
   471→   471→  const scores = useMemo<FirmExecutiveScores>(() => {
   472→   472→    if (loading) {
   473→   473→      return { firmHealth: 0, revenue: 0, compliance: 0, teamEfficiency: 0, clientSatisfaction: 0, cashFlow: 0 };
   474→   474→    }
   475→   475→
   476→   476→    // Firm Health: weighted avg of client health scores
   477→   477→    const activeClients = clients.filter(c => c.status === 'active');
   478→   478→    const avgHealth = activeClients.length > 0
   479→   479→      ? activeClients.reduce((s, c) => s + (c.healthScore || 0), 0) / activeClients.length
   480→   480→      : 50;
   481→   481→
   482→   482→    // Revenue Score: based on tax volume
   483→   483→    const totalTax = invoices.reduce((s, i) => s + (i.totalTax || 0) + (i.cgst || 0) + (i.sgst || 0) + (i.igst || 0), 0);
   484→   484→    const revenueScore = Math.min(100, Math.round((totalTax / 1000000) * 100));
   485→   485→
   486→   486→    // Compliance Score: filed vs total returns
   487→   487→    const filedReturns = returns.filter(r => r.status === 'filed').length;
   488→   488→    const complianceScore = returns.length > 0 ? Math.round((filedReturns / returns.length) * 100) : 50;
   489→   489→
   490→   490→    // Team Efficiency: based on completed activities
   491→   491→    const completedActivities = activities.filter(a => a.type.includes('filed') || a.type.includes('completed') || a.type.includes('resolved')).length;
   492→   492→    const teamEfficiency = activities.length > 0 ? Math.min(100, Math.round((completedActivities / Math.max(activities.length, 1)) * 100)) : 50;
   493→   493→
   494→   494→    // Client Satisfaction: inverse of overdue + health
   495→   495→    const overdueClients = activeClients.filter(c => (c.pendingReturnCount || 0) > 0).length;
   496→   496→    const clientSatisfaction = activeClients.length > 0 ? Math.round(((activeClients.length - overdueClients) / activeClients.length) * 100) : 50;
   497→   497→
   498→   498→    // Cash Flow: based on reconciliation match rate
   499→   499→    const matchedRecons = reconciliations.filter(r => r.matched > 0);
   500→   500→    const totalMatched = matchedRecons.reduce((s, r) => s + r.matched, 0);
   501→   501→    const totalRecords = matchedRecons.reduce((s, r) => s + r.totalRecords, 0);
   502→   502→    const cashFlow = totalRecords > 0 ? Math.round((totalMatched / totalRecords) * 100) : 50;
   503→   503→
   504→   504→    return {
   505→   505→      firmHealth: Math.round(avgHealth) || 50,
   506→   506→      revenue: revenueScore || 50,
   507→   507→      compliance: complianceScore || 50,
   508→   508→      teamEfficiency: teamEfficiency || 50,
   509→   509→      clientSatisfaction: clientSatisfaction || 50,
   510→   510→      cashFlow: cashFlow || 50,
   511→   511→    };
   512→   512→  }, [clients, returns, invoices, reconciliations, activities, loading]);
   513→   513→
   514→   514→  return { scores, loading };
   515→   515→}
