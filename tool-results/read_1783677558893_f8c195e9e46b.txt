    63→// Relative time formatter — defensively handles ISO string / Date / Firestore Timestamp.
    64→function timeAgo(createdAt: unknown): string {
    65→  if (!createdAt) return 'just now';
    66→  let date: Date;
    67→  if (createdAt instanceof Date) {
    68→    date = createdAt;
    69→  } else if (typeof createdAt === 'string') {
    70→    date = new Date(createdAt);
    71→  } else if (
    72→    typeof createdAt === 'object' &&
    73→    createdAt !== null &&
    74→    'toDate' in createdAt &&
    75→    typeof (createdAt as { toDate: () => Date }).toDate === 'function'
    76→  ) {
    77→    date = (createdAt as { toDate: () => Date }).toDate();
    78→  } else {
    79→    return 'just now';
    80→  }
    81→  if (isNaN(date.getTime())) return 'just now';
    82→  const seconds = Math.floor((Date.now() - date.getTime()) / 1000);
    83→  if (seconds < 60) return 'just now';
    84→  const minutes = Math.floor(seconds / 60);
    85→  if (minutes < 60) return `${minutes}m ago`;
    86→  const hours = Math.floor(minutes / 60);
    87→  if (hours < 24) return `${hours}h ago`;
    88→  const days = Math.floor(hours / 24);
    89→  if (days < 7) return `${days}d ago`;
    90→  const weeks = Math.floor(days / 7);
    91→  if (weeks < 5) return `${weeks}w ago`;
    92→  const months = Math.floor(days / 30);
    93→  if (months < 12) return `${months}mo ago`;
    94→  const years = Math.floor(days / 365);
    95→  return `${years}y ago`;
    96→}

    98→// ─── Business Score computation ───────────────────────────────────────────────
    99→// Start at 100. Subtract weighted penalties for risk signals. Clamp 0–100.
   100→// If there is no data at all, return null (UI shows premium empty state).
   101→
   102→function computeBusinessScore(opts: {
   103→  totalClients: number;
   104→  criticalIssues: number;
   105→  warnings: number;
   106→  pendingReturns: number;
   107→  overdueReturns: number;
   108→  averageHealthScore: number;
   109→  matchPercentage: number;
   110→}): number | null {
   111→  const { totalClients, criticalIssues, warnings, pendingReturns, overdueReturns, averageHealthScore, matchPercentage } = opts;
   112→  if (totalClients === 0 && pendingReturns === 0 && overdueReturns === 0) return null;
   113→
   114→  let score = 100;
   115→  score -= criticalIssues * 6;
   116→  score -= warnings * 2;
   117→  score -= overdueReturns * 8;
   118→  score -= pendingReturns * 2;
   119→  if (averageHealthScore > 0) {
   120→    score -= Math.max(0, 80 - averageHealthScore) * 0.6;
   121→  }
   122→  score -= Math.max(0, 100 - matchPercentage) * 0.3;
   123→  return Math.max(0, Math.min(100, Math.round(score)));
   124→}

   126→// V16 palette — only Emerald / Cyan / Blue. No amber, no red.
   127→function scoreTier(score: number | null): { label: string; color: string } {
   128→  if (score === null) return { label: '—', color: 'text-muted-foreground' };
   129→  if (score >= 85) return { label: 'Excellent', color: 'text-emerald-400' };
   130→  if (score >= 70) return { label: 'Healthy', color: 'text-cyan-400' };
   131→  if (score >= 50) return { label: 'At Risk', color: 'text-blue-400' };
   132→  return { label: 'Critical', color: 'text-blue-300' };
   133→}

   149→function buildInsights(opts: {
   150→  metrics: {
   151→    totalClients: number;
   152→    pendingReturns: number;
   153→    overdueReturns: number;
   154→    criticalIssues: number;
   155→    warnings: number;
   156→    totalTaxVolume: number;
   157→    averageHealthScore: number;
   158→    matchPercentage: number;
   159→    upcomingFilings: FirestoreReturn[];
   160→  };
   161→  clients: Array<FirestoreClient & { id: string }>;
   162→  /** True when the firm has real Firestore data. Gates the "all clear" insight
   163→   *  so we never say "Everything looks good today" when there's no data at all. */
   164→  hasData: boolean;
   165→}): AIInsight[] {
   166→  const { metrics, clients, hasData } = opts;
   167→  const insights: AIInsight[] = [];
   168→
   169→  // 1. Overdue returns — most urgent
   170→  if (metrics.overdueReturns > 0) {
   171→    insights.push({
   172→      id: 'overdue-returns',
   173→      icon: AlertTriangle,
   174→      tone: 'risk',
   175→      what: `${metrics.overdueReturns} GST return${metrics.overdueReturns > 1 ? 's are' : ' is'} overdue`,
   176→      why: 'Overdue returns trigger notices and late fees. File immediately to avoid penalties.',
   177→      actionLabel: 'File Now',
   178→      actionView: 'returns',
   179→    });
   180→  }
   181→
   182→  // 2. Pending returns due soon
   183→  const nextFiling = metrics.upcomingFilings[0];
   184→  if (nextFiling && metrics.pendingReturns > 0 && metrics.overdueReturns === 0) {
   185→    insights.push({
   186→      id: 'gst-due',
   187→      icon: Clock,
   188→      tone: 'info',
   189→      what: `GST filing due — ${nextFiling.returnType} for ${nextFiling.period}`,
   190→      why: `${metrics.pendingReturns} return${metrics.pendingReturns > 1 ? 's' : ''} pending. Filing on time keeps compliance score high.`,
   191→      actionLabel: 'File GST',
   192→      actionView: 'returns',
   193→    });
   194→  }
   195→
   196→  // 3. Critical issues / risky clients
   197→  const riskyClients = clients.filter(c => (c.healthScore ?? 100) < 60);
   198→  if (metrics.criticalIssues > 0 || riskyClients.length > 0) {
   199→    insights.push({
   200→      id: 'churn-risk',
   201→      icon: TrendingDown,
   202→      tone: 'risk',
   203→      what: riskyClients.length > 0
   204→        ? `${riskyClients.length} client${riskyClients.length > 1 ? 's may' : ' may'} churn`
   205→        : `${metrics.criticalIssues} critical issue${metrics.criticalIssues > 1 ? 's' : ''} need attention`,
   206→      why: riskyClients.length > 0
   207→        ? 'Low health scores signal dissatisfaction. Reach out before they leave.'
   208→        : 'Critical issues compound into bigger problems if left unaddressed.',
   209→      actionLabel: riskyClients.length > 0 ? 'View Clients' : 'Review Issues',
   210→      actionView: riskyClients.length > 0 ? 'clients' : 'ai-operating-room',
   211→    });
   212→  }
   213→
   214→  // 4. Cash position / collections
   215→  if (metrics.totalTaxVolume > 0) {
   216→    insights.push({
   217→      id: 'collections',
   218→      icon: Wallet,
   219→      tone: metrics.matchPercentage < 90 ? 'risk' : 'info',
   220→      what: metrics.matchPercentage < 90
   221→        ? 'Collections may drop — reconciliation gaps detected'
   222→        : 'Cash position is healthy',
   223→      why: metrics.matchPercentage < 90
   224→        ? `${metrics.matchPercentage.toFixed(0)}% of invoices reconciled. Unmatched invoices mean uncollected revenue.`
   225→        : `${metrics.matchPercentage.toFixed(0)}% invoice match rate. Keep chasing pending collections to maintain runway.`,
   226→      actionLabel: 'Fix Collections',
   227→      actionView: 'reconcile',
   228→    });
   229→  }
   230→
   231→  // 5. Compliance score insight
   232→  if (metrics.averageHealthScore > 0) {
   233→    const tier = metrics.averageHealthScore >= 80 ? 'success' : 'info';
   234→    insights.push({
   235→      id: 'compliance',
   236→      icon: ShieldAlert,
   237→      tone: tier,
   238→      what: `Compliance health at ${metrics.averageHealthScore.toFixed(0)}%`,
   239→      why: metrics.averageHealthScore >= 80
   240→        ? 'Strong compliance posture. Maintain by filing on time and resolving warnings.'
   241→        : 'Below optimal. Address pending returns and warnings to lift your score.',
   242→      actionLabel: 'View Compliance',
   243→      actionView: 'returns',
   244→    });
   245→  }
   246→
   247→  // 6. Always-on "all clear" nudge — only when we actually have data.
   248→  // Never say "Everything looks good today" when there is no data at all;
   249→  // in that case the widget shows its own honest empty state instead.
   250→  if (hasData && insights.length === 0) {
   251→    insights.push({
   252→      id: 'all-clear',
   253→      icon: Sparkles,
   254→      tone: 'success',
   255→      what: 'Everything looks good today',
   256→      why: 'No urgent issues. Hand control to Oracle and let it run the routine work.',
   257→      actionLabel: 'Run My Business',
   258→      actionView: 'run-my-business',
   259→    });
   260→  }
   261→
   262→  return insights.slice(0, 6);
   263→}

   267→function ScoreGauge({ score }: { score: number | null }) {
   268→  const size = 220;
   269→  const stroke = 14;
   270→  const r = (size - stroke) / 2;
   271→  const c = 2 * Math.PI * r;
   272→  const pct = score === null ? 0 : score / 100;
   273→  const offset = c * (1 - pct);
   274→
   275→  return (
   276→    <div className="relative" style={{ width: size, height: size }}>
   277→      <svg width={size} height={size} className="-rotate-90">
   278→        <defs>
   279→          <linearGradient id="missionScoreGradient" x1="0%" y1="0%" x2="100%" y2="100%">
   280→            <stop offset="0%" stopColor="#10b981" />
   281→            <stop offset="50%" stopColor="#06b6d4" />
   282→            <stop offset="100%" stopColor="#3b82f6" />
   283→          </linearGradient>
   284→        </defs>
   285→        <circle
   286→          cx={size / 2}
   287→          cy={size / 2}
   288→          r={r}
   289→          fill="none"
   290→          stroke="rgba(255,255,255,0.06)"
   291→          strokeWidth={stroke}
   292→        />
   293→        {score !== null && (
   294→          <motion.circle
   295→            cx={size / 2}
   296→            cy={size / 2}
   297→            r={r}
   298→            fill="none"
   299→            stroke="url(#missionScoreGradient)"
   300→            strokeWidth={stroke}
   301→            strokeLinecap="round"
   302→            strokeDasharray={c}
   303→            initial={{ strokeDashoffset: c }}
   304→            animate={{ strokeDashoffset: offset }}
   305→            transition={{ duration: 1.2, ease: 'easeOut' as const, delay: 0.2 }}
   306→          />
   307→        )}
   308→      </svg>
   309→      <div className="absolute inset-0 flex flex-col items-center justify-center">
   310→        {score === null ? (
   311→          <span className="text-4xl font-bold text-muted-foreground">—</span>
   312→        ) : (
   313→          <motion.span
   314→            initial={{ opacity: 0, scale: 0.8 }}
   315→            animate={{ opacity: 1, scale: 1 }}
   316→            transition={{ duration: 0.5, delay: 0.6 }}
   317→            className={`text-6xl font-bold tracking-tight ${scoreTier(score).color}`}
   318→          >
   319→            {score}
   320→          </motion.span>
   321→        )}
   322→        <span className="text-xs text-muted-foreground tracking-wider uppercase mt-1">
   323→          / 100
   324→        </span>
   325→      </div>
   326→    </div>
   327→  );
   328→}
