    84→const inrFmt = (n: number) => Math.round(n).toLocaleString('en-IN');
    85→const inrShort = (n: number) => {
    86→  if (!isFinite(n) || isNaN(n)) return '₹0';
    87→  const abs = Math.abs(n);
    88→  if (abs >= 10000000) return `₹${(n / 10000000).toFixed(2)} Cr`;
    89→  if (abs >= 100000) return `₹${(n / 100000).toFixed(2)} L`;
    90→  if (abs >= 1000) return `₹${(n / 1000).toFixed(1)}K`;
    91→  return '₹' + Math.round(n).toLocaleString('en-IN');
    92→};
    93→const nowISO = () => new Date().toISOString();
    94→const uid = (p: string) => `${p}_${Math.random().toString(36).slice(2, 10)}`;
    95→const daysAgo = (d: number) => {
    96→  const x = new Date();
    97→  x.setDate(x.getDate() - d);
    98→  return x.toISOString();
    99→};

   101→function scoreToLevel(score: number): RiskLevel {
   102→  if (score >= 75) return 'critical';
   103→  if (score >= 55) return 'high';
   104→  if (score >= 30) return 'medium';
   105→  return 'low';

   108→function levelRank(l: RiskLevel): number {
   109→  return { low: 0, medium: 1, high: 2, critical: 3 }[l];

   112→function filingDueDate(returnType: string, period: string): Date | null {
   113→  const m = period.match(/^(\d{4})-(\d{2})$/);
   114→  if (!m) return null;
   115→  const year = parseInt(m[1], 10);
   116→  const month = parseInt(m[2], 10);
   117→  if (month === 12) return new Date(year + 1, 0, 11);
   118→  const day = returnType === 'GSTR-1' ? 11 : 20;
   119→  return new Date(year, month, day);
   120→}

   258→export function buildKnowledgeGraph(rows: RawRows, cfo: CFOResponse): KnowledgeGraph {
   259→  const nodes: GraphNode[] = [];
   260→  const edges: { id: string; source: string; target: string; type: RelationshipType; amount?: number; label?: string }[] = [];
   261→
   262→  const pushNode = (n: GraphNode) => nodes.push(n);
   263→  const pushEdge = (source: string, target: string, type: RelationshipType, amount?: number, label?: string) => {
   264→    edges.push({ id: uid('e'), source, target, type, amount, label });
   265→  };
   266→
   267→  // 1. Business node — the firm itself (the user's business). Single root.
   268→  const businessId = 'business:firm';
   269→  const firmName = 'Your Business';
   270→  pushNode({
   271→    id: businessId,
   272→    type: 'business',
   273→    entityId: 'firm',
   274→    label: firmName,
   275→    subtitle: 'The firm at the center of the graph',
   276→    amount: cfo.dashboard.revenue.thisMonth,
   277→    meta: { healthScore: cfo.dashboard.healthScore.overall, cash: cfo.dashboard.cash.currentBalance },
   278→    x: 0, y: 0,
   279→  });
   280→
   281→  // 2. Bank account node — derived from cash position
   282→  const bankId = 'bank-account:primary';
   283→  pushNode({
   284→    id: bankId,
   285→    type: 'bank-account',
   286→    entityId: 'primary',
   287→    label: 'Primary Bank Account',
   288→    subtitle: `Balance ${inrShort(cfo.dashboard.cash.currentBalance)}`,
   289→    amount: cfo.dashboard.cash.currentBalance,
   290→    meta: { runway: cfo.dashboard.cash.runwayDays },
   291→    x: -200, y: 0,
   292→  });
   293→  pushEdge(businessId, bankId, 'OWNS');
   294→
   295→  // 3. Employees — REAL data from Employee table (Phase 6: no more hardcoded names)
   296→  const activeEmployees = rows.employees.filter((e) => e.status === 'active' || e.status !== 'inactive');
   297→  activeEmployees.slice(0, 25).forEach((emp, i) => {
   298→    const empId = `employee:${emp.id}`;
   299→    pushNode({
   300→      id: empId,
   301→      type: 'employee',
   302→      entityId: emp.id,
   303→      label: emp.name,
   304→      subtitle: emp.designation || emp.department || 'Employee',
   305→      meta: {
   306→        designation: emp.designation || '',
   307→        department: emp.department || '',
   308→        status: emp.status,
   309→      },
   310→      x: -250 + (i % 5) * 60, y: 220 + Math.floor(i / 5) * 50,
   311→    });
   312→    pushEdge(businessId, empId, 'OWNS');
   313→  });
   314→
   315→  // 4. Clients — REAL data from Prisma
   316→  rows.clients.slice(0, 500).forEach((c, i) => {
   317→    const angle = (i / Math.max(rows.clients.length, 1)) * 2 * Math.PI;
   318→    const r = 320;
   319→    const nodeId = `client:${c.id}`;
   320→    pushNode({
   321→      id: nodeId,
   322→      type: 'client',
   323→      entityId: c.id,
   324→      label: c.tradeName || c.gstin,
   325→      subtitle: c.gstin,
   326→      meta: { status: c.status, healthScore: c.healthScore, gstin: c.gstin },
   327→      x: Math.cos(angle) * r,
   328→      y: Math.sin(angle) * r,
   329→    });
   330→    pushEdge(businessId, nodeId, 'OWNS');
   331→  });
   332→
   333→  // 5. Invoices — REAL data; link to clients (by buyerGstin match) or business
   334→  rows.invoices.slice(0, 2000).forEach((inv, i) => {
   335→    const angle = (i / Math.max(rows.invoices.length, 1)) * 2 * Math.PI;
   336→    const r = 480;
   337→    const invNodeId = `invoice:${inv.id}`;
   338→    pushNode({
   339→      id: invNodeId,
   340→      type: 'invoice',
   341→      entityId: inv.id,
   342→      label: inv.invoiceNumber,
   343→      subtitle: `${inrShort(inv.totalAmount)} · ${inv.paymentStatus || inv.status}`,
   344→      amount: inv.totalAmount,
   345→      meta: {
   346→        status: inv.status, period: inv.period ?? '', date: inv.invoiceDate,
   347→        paymentStatus: inv.paymentStatus, balanceAmount: inv.balanceAmount,
   348→        buyerGstin: inv.buyerGstin ?? '',
   349→      },
   350→      x: Math.cos(angle) * r,
   351→      y: Math.sin(angle) * r,
   352→    });
   353→    const matchedClient = rows.clients.find((c) => c.gstin === inv.buyerGstin);
   354→    if (matchedClient) {
   355→      // Client RECEIVES invoice (Phase 6: new relationship type)
   356→      pushEdge(`client:${matchedClient.id}`, invNodeId, 'RECEIVES', inv.totalAmount);
   357→      // Invoice GENERATES revenue for business
   358→      pushEdge(businessId, invNodeId, 'GENERATES', inv.totalAmount);
   359→    } else {
   360→      pushEdge(businessId, invNodeId, 'GENERATES', inv.totalAmount);
   361→    }
   362→  });
   363→
   364→  // 6. GST Returns — REAL data; link to clients + business
   365→  rows.filings.slice(0, 1000).forEach((f, i) => {
   366→    const angle = (i / Math.max(rows.filings.length, 1)) * 2 * Math.PI;
   367→    const r = 580;
   368→    const nodeId = `gst-return:${f.id}`;
   369→    const due = filingDueDate(f.returnType, f.period);
   370→    const daysLeft = due ? Math.ceil((due.getTime() - Date.now()) / (1000 * 60 * 60 * 24)) : null;
   371→    pushNode({
   372→      id: nodeId,
   373→      type: 'gst-return',
   374→      entityId: f.id,
   375→      label: `${f.returnType} · ${f.period}`,
   376→      subtitle: `${f.status} · tax ${inrShort(f.totalTax)}`,
   377→      amount: f.totalTax,
   378→      meta: { status: f.status, period: f.period, returnType: f.returnType, daysLeft: daysLeft ?? 0 },
   379→      x: Math.cos(angle) * r,
   380→      y: Math.sin(angle) * r,
   381→    });
   382→    pushEdge(`client:${f.clientId}`, nodeId, 'FILES', f.totalTax);
   383→    // GST Return GENERATES_LIABILITY for the business (Phase 6: new relationship)
   384→    if (f.status !== 'filed' && f.totalTax > 0) {
   385→      pushEdge(nodeId, businessId, 'GENERATES_LIABILITY', f.totalTax);
   386→    }
   387→  });
   388→
   389→  // 7. Notices — REAL data; link to clients + business + compliance
   390→  rows.notices.slice(0, 500).forEach((n, i) => {
   391→    const angle = (i / Math.max(rows.notices.length, 1)) * 2 * Math.PI;
   392→    const r = 660;
   393→    const nodeId = `notice:${n.id}`;
   394→    pushNode({
   395→      id: nodeId,
   396→      type: 'notice',
   397→      entityId: n.id,
   398→      label: n.noticeType,
   399→      subtitle: `${n.status} · due ${n.dueDate || 'n/a'}`,
   400→      meta: { status: n.status, noticeDate: n.noticeDate ?? '', dueDate: n.dueDate ?? '' },
   401→      x: Math.cos(angle) * r,
   402→      y: Math.sin(angle) * r,
   403→    });
   404→    pushEdge(`client:${n.clientId}`, nodeId, 'RESPONDS_TO');
   405→    // Notice AFFECTS business compliance (Phase 6: new relationship)
   406→    pushEdge(nodeId, businessId, 'AFFECTS');
   407→  });
   408→
   409→  // 8. Vendors — REAL data derived from PurchaseBill table
   410→  //    Group purchase bills by vendorName to create one vendor node per unique vendor
   411→  const vendorMap: Map<string, {
   412→    name: string; gstin: string | null; totalSpend: number; pendingPayable: number;
   413→    overdueCount: number; billCount: number;
   414→  }> = new Map();
   415→  rows.purchaseBills.forEach((b) => {
   416→    const key = b.vendorGstin || b.vendorName;
   417→    if (!vendorMap.has(key)) {
   418→      vendorMap.set(key, {
   419→        name: b.vendorName, gstin: b.vendorGstin, totalSpend: 0, pendingPayable: 0, overdueCount: 0, billCount: 0,
   420→      });
   421→    }
   422→    const v = vendorMap.get(key)!;
   423→    v.totalSpend += b.totalAmount;
   424→    v.pendingPayable += b.balanceAmount;
   425→    v.billCount += 1;
   426→    if (b.paymentStatus === 'overdue' || (b.dueDate && b.balanceAmount > 0 && new Date(b.dueDate).getTime() < Date.now())) {
   427→      v.overdueCount += 1;
   428→    }
   429→  });
   430→
   431→  const vendorEntries = Array.from(vendorMap.entries())
   432→    .sort((a, b) => b[1].totalSpend - a[1].totalSpend)
   433→    .slice(0, 100); // top 100 vendors by spend
   434→
   435→  vendorEntries.forEach(([, v], i) => {
   436→    const vendorId = `vendor:${v.gstin || v.name.replace(/\s+/g, '-').toLowerCase()}`;
   437→    const dependencyScore = Math.min(90, 30 + Math.log10(v.totalSpend + 1) * 8 + (v.overdueCount > 0 ? 15 : 0));
   438→    pushNode({
   439→      id: vendorId,
   440→      type: 'vendor',
   441→      entityId: v.gstin || v.name,
   442→      label: v.name,
   443→      subtitle: `Spend ${inrShort(v.totalSpend)} · ${v.billCount} bills`,
   444→      amount: v.totalSpend,
   445→      meta: {
   446→        gstin: v.gstin || '',
   447→        totalSpend: v.totalSpend,
   448→        pendingPayable: v.pendingPayable,
   449→        overdueCount: v.overdueCount,
   450→        billCount: v.billCount,
   451→        dependencyScore,
   452→        reliabilityScore: Math.max(0, 100 - v.overdueCount * 15),
   453→      },
   454→      x: -400 + (i % 6) * 70, y: -240 + Math.floor(i / 6) * 50,
   455→    });
   456→    pushEdge(businessId, vendorId, 'PAYS', v.totalSpend);
   457→  });
   458→
   459→  // 9. Purchase bills → become INVOICE nodes (vendor invoices) + ITC records

   719→      pushNode({
   720→        id: msgNodeId,
   721→        type: 'conversation',
   722→        entityId: rec.id,
   723→        label: (rec.title || 'Email').slice(0, 42),

   731→      pushNode({
   732→        id: msgNodeId,
   733→        type: 'conversation',
   734→        entityId: rec.id,

   746→        type: 'transaction',

   758→  // Index edges by type and nodes by type for stats

   766→    edges: edges.map((e) => ({ ...e, weight: e.amount ? Math.min(10, Math.max(1, Math.log10(e.amount + 1))) : 1 })),
   767→    nodeCountByType,
   768→    edgeCountByType,
   769→  };
   770→}

   772→// ─── Module 3: Risk Graph ────────────────────────────────────────────────────
   773→
   774→export function buildRiskGraph(graph: KnowledgeGraph, cfo: CFOResponse): RiskGraph {
   775→  const riskNodes: RiskNode[] = [];
   776→
   777→  // 1. Late-paying clients — from CFO memory client behavior
   778→  cfo.memory.clientBehavior.forEach((c) => {
   779→    const score =
   780→      c.riskLabel === 'High' ? 78 :
   781→      c.riskLabel === 'Medium' ? 50 :
   782→      c.riskLabel === 'Low' ? 18 : 8;
   783→    if (score < 30) return; // skip low-risk
   784→    const clientNode = graph.nodes.find((n) => n.type === 'client' && n.label === c.clientName);
   785→    const level = scoreToLevel(score);
   786→    riskNodes.push({
   787→      id: uid('risk'),
   788→      nodeId: clientNode?.id || `client:${c.clientName}`,
   789→      entityName: c.clientName,
   790→      entityType: 'client',
   791→      category: 'late_payment',
   792→      level,
   793→      score,
   794→      reasons: [
   795→        `${c.delays} delayed payment(s) recorded`,
   796→        `Average delay ${c.averageDelayDays} days`,
   797→        `Outstanding ${inrShort(c.totalOutstanding)}`,
   798→      ],
   799→      impact: `If ${c.clientName} delays further, cash flow impact up to ${inrShort(c.totalOutstanding)}.`,
   800→      amountAtRisk: c.totalOutstanding,
   801→      recommendation:
   802→        level === 'critical' || level === 'high'
   803→          ? 'Escalate to tier-2 follow-up; consider revised payment terms.'
   804→          : 'Send automated reminders 7 days before due date.',
   805→    });
   806→  });
   807→
   808→  // 2. GST notice risk — for each open notice
   809→  graph.nodes.filter((n) => n.type === 'notice').forEach((n) => {
   810→    const score = 70;
   811→    riskNodes.push({
   812→      id: uid('risk'),
   813→      nodeId: n.id,
   814→      entityName: n.label,
   815→      entityType: 'notice',
   816→      category: 'gst_notice',
   817→      level: 'high',
   818→      score,
   819→      reasons: [
   820→        `Notice type: ${n.label}`,
   821→        'Statutory response required within timeline',
   822→        'Non-response → penalty + escalation',
   823→      ],
   824→      impact: 'Non-response may trigger penalty and freeze ITC claims.',
   825→      recommendation: 'Respond within 7 days. Engage GST counsel if escalation tier.',
   826→    });
   827→  });
   828→
   829→  // 3. Cash flow risk — from CFO risks
   830→  const cashRisk = cfo.risks.find((r) => r.category === 'cash' && r.level !== 'low');
   831→  if (cashRisk) {
   832→    const score = cashRisk.score;
   833→    riskNodes.push({
   834→      id: uid('risk'),
   835→      nodeId: 'bank-account:primary',
   836→      entityName: 'Primary Bank Account',
   837→      entityType: 'bank-account',
   838→      category: 'cash_flow',
   839→      level: scoreToLevel(score),
   840→      score,
   841→      reasons: cashRisk.reasons,
   842→      impact: `Runway ${cfo.dashboard.cash.runwayDays || '∞'} days; daily burn ${inrShort(cfo.dashboard.cash.burnRatePerDay)}.`,
   843→      amountAtRisk: cfo.dashboard.cash.currentBalance,
   844→      recommendation: cashRisk.recommendation || 'Review cash position and accelerate collections.',
   845→    });
   846→  }
   847→
   848→  // 4. Compliance risk
   849→  const complianceRisk = cfo.risks.find((r) => r.category === 'compliance' && r.level !== 'low');
   850→  if (complianceRisk) {
   851→    const score = complianceRisk.score;
   852→    riskNodes.push({
   853→      id: uid('risk'),
   854→      nodeId: 'business:firm',
   855→      entityName: 'Business (Compliance)',
   856→      entityType: 'business',
   857→      category: 'compliance',
   858→      level: scoreToLevel(score),
   859→      score,
   860→      reasons: complianceRisk.reasons,
   861→      impact: 'Late fee ₹50/day + 18% interest p.a. on overdue returns.',
   862→      amountAtRisk: cfo.dashboard.gst.liability,
   863→      recommendation: complianceRisk.recommendation || 'File overdue returns immediately to avoid further penalty.',
   864→    });
   865→  }
   866→
   867→  // 5. Revenue concentration — if one client > 35% of revenue
   868→  const totalRevenue = cfo.dashboard.revenue.thisMonth || 1;
   869→  graph.nodes
   870→    .filter((n) => n.type === 'client')
   871→    .forEach((cNode) => {
   872→      const clientInvoices = graph.edges
   873→        .filter((e) => e.source === cNode.id && e.type === 'GENERATES')
   874→        .reduce((sum, e) => sum + (e.amount || 0), 0);
   875→      const concentration = clientInvoices / totalRevenue;
   876→      if (concentration > 0.35 && clientInvoices > 0) {
   877→        const score = Math.min(85, Math.round(concentration * 100));
   878→        riskNodes.push({
   879→          id: uid('risk'),
   880→          nodeId: cNode.id,
   881→          entityName: cNode.label,
   882→          entityType: 'client',
   883→          category: 'revenue_concentration',
   884→          level: scoreToLevel(score),
   885→          score,
   886→          reasons: [
   887→            `Generates ${(concentration * 100).toFixed(0)}% of revenue`,
   888→            `Revenue exposure ${inrShort(clientInvoices)}`,
   889→            'Single-point-of-failure risk',
   890→          ],
   891→          impact: `Loss of this client → ${(concentration * 100).toFixed(0)}% revenue drop.`,
   892→          amountAtRisk: clientInvoices,
   893→          recommendation: 'Diversify client base; lock in multi-year contract.',
   894→        });
   895→      }
   896→    });
   897→
   898→  // 6. Vendor dependency — REAL data from vendor nodes (Phase 6: uses real spend + overdue)
   899→  graph.nodes.filter((n) => n.type === 'vendor').forEach((v) => {
   900→    const depScore = (v.meta?.dependencyScore as number) || 40;
   901→    const overdueCount = (v.meta?.overdueCount as number) || 0;
   902→    if (depScore < 50 && overdueCount === 0) return;
   903→    const finalScore = Math.min(90, Math.max(depScore, 50 + overdueCount * 10));
   904→    riskNodes.push({
   905→      id: uid('risk'),
   906→      nodeId: v.id,
   907→      entityName: v.label,
   908→      entityType: 'vendor',
   909→      category: 'vendor_dependency',
   910→      level: scoreToLevel(finalScore),
   911→      score: finalScore,
   912→      reasons: [
   913→        `Annual spend ${inrShort(v.amount || 0)}`,
   914→        `${overdueCount} overdue bill(s)`,
   915→        `Reliability score ${(v.meta?.reliabilityScore as number) || 100}/100`,
   916→      ],
   917→      impact: `Disruption → procurement delay, potential revenue loss ${inrShort((v.amount || 0) * 0.3)}.`,
   918→      amountAtRisk: v.amount,
   919→      recommendation: overdueCount > 2
   920→        ? 'Identify backup vendor; renegotiate payment terms.'
   921→        : 'Maintain 60-day stock buffer; monitor vendor health monthly.',
   922→    });
   923→  });
   924→
   925→  // 7. Fraud risk — heuristic: invoice with riskLevel 'high' in invoice book
   926→  const hasCritical = riskNodes.some((r) => r.level === 'critical');
   927→  if (!hasCritical) {
   928→    const suspiciousInvoices = graph.nodes.filter((n) => n.type === 'invoice' && (n.meta?.status === 'flagged'));
   929→    if (suspiciousInvoices.length > 0) {
   930→      suspiciousInvoices.slice(0, 2).forEach((inv) => {
   931→        riskNodes.push({
   932→          id: uid('risk'),
   933→          nodeId: inv.id,
   934→          entityName: inv.label,
   935→          entityType: 'invoice',
   936→          category: 'fraud',
   937→          level: 'high',
   938→          score: 65,
   939→          reasons: ['Round-trip pattern detected', 'Vendor mismatch in 2B'],
   940→          impact: 'Potential ITC reversal + penalty up to 100% of tax.',
   941→          amountAtRisk: inv.amount,
   942→          recommendation: 'Hold ITC claim; verify vendor GSTIN status.',
   943→        });
   944→      });
   945→    }
   946→  }
   947→
   948→  // Aggregate stats
   949→  const countByLevel = { low: 0, medium: 0, high: 0, critical: 0 } as Record<RiskLevel, number>;
   950→  const countByCategory = {} as Record<RiskCategory, number>;
   951→  riskNodes.forEach((r) => {
   952→    countByLevel[r.level]++;
   953→    countByCategory[r.category] = (countByCategory[r.category] || 0) + 1;
   954→  });
   955→  const overallLevel = riskNodes.reduce<RiskLevel>(
   956→    (max, r) => (levelRank(r.level) > levelRank(max) ? r.level : max),
   957→    'low',
   958→  );
   959→  const topRisks = [...riskNodes].sort((a, b) => b.score - a.score).slice(0, 5);
   960→
   961→  // Stamp risk on graph nodes
   962→  riskNodes.forEach((r) => {
   963→    const node = graph.nodes.find((n) => n.id === r.nodeId);
   964→    if (node) {
   965→      node.riskScore = r.score;
   966→      node.riskLevel = r.level;
   967→    }
   968→  });
   969→
   970→  return { nodes: riskNodes, countByLevel, countByCategory, overallLevel, topRisks };
   971→}
