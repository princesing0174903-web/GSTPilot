   793→   793→/** Convert Firestore timestamps to ISO strings for UI consumption */
   794→   794→export function withId<T extends Record<string, unknown>>(doc: { id: string; data: () => T }): T & { id: string } {
   795→   795→  const data = doc.data();
   796→   796→  const converted: Record<string, unknown> = {};
   797→   797→  for (const [key, value] of Object.entries(data)) {
   798→   798→    if (value && typeof value === 'object' && 'toDate' in value && typeof (value as { toDate: () => Date }).toDate === 'function') {
   799→   799→      converted[key] = (value as { toDate: () => Date }).toDate().toISOString();
   800→   800→    } else {
   801→   801→      converted[key] = value;
   802→   802→    }
   803→   803→  }
   804→   804→  return { id: doc.id, ...converted } as T & { id: string };
   805→   805→}
   806→   806→
   807→   807→/** Convert an array of Firestore docs with timestamp conversion */
   808→   808→export function convertDocs<T extends Record<string, unknown>>(docs: Array<{ id: string; data: () => T }>): Array<T & { id: string }> {
   809→   809→  return docs.map(withId);
   810→   810→}
