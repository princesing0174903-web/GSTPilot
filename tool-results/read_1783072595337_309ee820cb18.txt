   117→  const { selectedClientId, setSelectedClientId, setCurrentView } = useApp();
   118→
   119→  // Data state
   120→  const [filings, setFilings] = useState<GSTRFiling[]>([]);
   121→  const [calendarItems, setCalendarItems] = useState<FilingCalendarItem[]>([]);
   122→  const [clients, setClients] = useState<ClientOption[]>([]);
   123→  const [loading, setLoading] = useState(true);
   124→
   125→  // Navigation state
   126→  const [currentMonth, setCurrentMonth] = useState(new Date());
   127→
   128→  // Filter state
   129→  const [filterClient, setFilterClient] = useState<string>('all');
   130→  const [filterReturnType, setFilterReturnType] = useState<string>('all');
   131→
   132→  // Detail dialog
   133→  const [selectedItem, setSelectedItem] = useState<FilingCalendarItem | null>(
   134→    null
   135→  );
   136→  const [detailOpen, setDetailOpen] = useState(false);
   137→
   138→  // ─── Fetch data ──────────────────────────────────────────────────────────
   139→  const fetchFilings = useCallback(async () => {
   140→    try {
   141→      setLoading(true);
   142→      const [filingsRes, dashboardRes] = await Promise.all([
   143→        fetch('/api/gstr-filing'),
   144→        fetch('/api/dashboard'),
   145→      ]);
   146→
   147→      if (filingsRes.ok) {
   148→        const filingsData = await filingsRes.json();
   149→        setFilings(filingsData.filings ?? []);
   150→      }
   151→
   152→      if (dashboardRes.ok) {
   153→        const dashData = await dashboardRes.json();
   154→        setCalendarItems(dashData.filingCalendar ?? []);
   155→      }
   156→    } catch {
   157→      // silently handle
   158→    } finally {
   159→      setLoading(false);
   160→    }
   161→  }, []);
   162→
   163→  const fetchClients = useCallback(async () => {
   164→    try {
   165→      const res = await fetch('/api/clients');
   166→      if (!res.ok) return;
   167→      const data = await res.json();
   168→      setClients(
   169→        (data.clients ?? []).map((c: { id: string; tradeName: string }) => ({
   170→          id: c.id,
   171→          tradeName: c.tradeName,
   172→        }))
   173→      );
   174→    } catch {
   175→      // silently ignore
   176→    }
   177→  }, []);
   178→
   179→  useEffect(() => {
   180→    fetchClients();
   181→    fetchFilings();
   182→  }, [fetchClients, fetchFilings]);
   183→
   184→  // Sync with global client selection
   185→  useEffect(() => {
   186→    if (selectedClientId) {
   187→      setFilterClient(selectedClientId);
   188→    }
   189→  }, [selectedClientId]);
   190→
   191→  // ─── Build calendar items from filings if API doesn't provide them ──────
   192→  const allCalendarItems = useMemo<FilingCalendarItem[]>(() => {
   193→    if (calendarItems.length > 0) {
   194→      return calendarItems;
   195→    }
   196→    // Fallback: build from filings
   197→    return filings.map((f) => {
   198→      const dueDate = getFilingDueDate(f.returnType, f.period);
   199→      const overdue = isOverdue(f.period);
   200→      let status: FilingCalendarItem['status'];
   201→
   202→      if (f.status === 'filed') {
   203→        status = 'filed';
   204→      } else if (overdue) {
   205→        status = 'overdue';
   206→      } else {
   207→        const dueDateObj = new Date(dueDate);
   208→        const now = new Date();
   209→        const sevenDaysFromNow = addDays(now, 7);
   210→        if (isBefore(dueDateObj, sevenDaysFromNow) && isAfter(dueDateObj, now)) {
   211→          status = 'upcoming';
   212→        } else {
   213→          status = 'pending';
   214→        }
   215→      }
   216→
   217→      return {
   218→        id: f.id,
   219→        returnType: f.returnType,
   220→        period: f.period,
   221→        dueDate,
   222→        status,
   223→        clientId: f.clientId,
   224→        clientName: f.client?.tradeName ?? 'Unknown',
   225→      };
   226→    });
   227→  }, [calendarItems, filings]);
   228→
   229→  // ─── Apply filters ──────────────────────────────────────────────────────
   230→  const filteredItems = useMemo(() => {
   231→    return allCalendarItems.filter((item) => {
   232→      if (filterClient !== 'all' && item.clientId !== filterClient)
   233→        return false;
   234→      if (
   235→        filterReturnType !== 'all' &&
   236→        item.returnType !== filterReturnType
   237→      )
   238→        return false;
   239→      return true;
   240→    });
   241→  }, [allCalendarItems, filterClient, filterReturnType]);
   242→
   243→  // ─── Summary counts ─────────────────────────────────────────────────────
   244→  const now = new Date();
   245→  const sevenDaysFromNow = addDays(now, 7);
   246→
   247→  const upcomingCount = filteredItems.filter((item) => {
   248→    const dueDate = new Date(item.dueDate);
   249→    return (
   250→      item.status !== 'filed' &&
   251→      isAfter(dueDate, now) &&
   252→      isBefore(dueDate, sevenDaysFromNow)
   253→    );
   254→  }).length;
   255→
   256→  const pendingCount = filteredItems.filter(
   257→    (item) => item.status === 'pending'
   258→  ).length;
   259→
   260→  const overdueCount = filteredItems.filter(
   261→    (item) => item.status === 'overdue'
   262→  ).length;
   263→
   264→  const filedThisMonthCount = filteredItems.filter((item) => {
   265→    if (item.status !== 'filed') return false;
   266→    const dueDate = new Date(item.dueDate);
   267→    return (
   268→      dueDate.getMonth() === now.getMonth() &&
   269→      dueDate.getFullYear() === now.getFullYear()

   271→  }).length;
   272→
   273→  // ─── Calendar grid data ─────────────────────────────────────────────────

   275→  const monthEnd = endOfMonth(currentMonth);

   280→  // Map due dates to items for quick lookup

   284→      const key = item.dueDate;
   285→      const existing = map.get(key) ?? [];

   287→      map.set(key, existing);
   288→    });
   289→    return map;
   290→  }, [filteredItems]);
   291→
   292→  // ─── Categorized filing lists ───────────────────────────────────────────

   294→    const dueDate = new Date(item.dueDate);
   295→    return (
   296→      item.status !== 'filed' &&
   297→      isAfter(dueDate, now) &&
   298→      isBefore(dueDate, sevenDaysFromNow)
   299→    );
   300→  });
   301→
   302→  const pendingFilings = filteredItems.filter(
   303→    (item) => item.status === 'pending'
   304→  );
   305→
   306→  const overdueFilings = filteredItems.filter(
   307→    (item) => item.status === 'overdue'
   308→  );
   309→
   310→  const filedFilings = filteredItems.filter(
   311→    (item) => item.status === 'filed'
   312→  );

   314→  // ─── Helpers ────────────────────────────────────────────────────────────
