    63→/**
    64→ * Convert a Firestore snapshot ({ id, data() }) into a typed object with the
    65→ * document id prepended, recursively converting Firestore Timestamps to ISO
    66→ * strings. Mirrors the conversion logic in `withId` (firestore-schema.ts) but
    67→ * drops the `T extends Record<string, unknown>` constraint so it accepts
    68→ * strongly-typed interfaces (which do not have an implicit index signature).
    69→ */
    70→function docToData<T>(d: { id: string; data: () => Record<string, unknown> }): T & { id: string } {
    71→  const data = d.data();
    72→  const converted: Record<string, unknown> = {};
    73→  for (const [key, value] of Object.entries(data)) {
    74→    if (value && typeof value === 'object' && 'toDate' in value && typeof (value as { toDate: () => Date }).toDate === 'function') {
    75→      converted[key] = (value as { toDate: () => Date }).toDate().toISOString();
    76→    } else {
    77→      converted[key] = value;
    78→    }
    79→  }
    80→  return { id: d.id, ...converted } as T & { id: string };
    81→}

   416→export async function updateReturnStatus(returnId: string, status: FilingStatus, extras?: Partial<FirestoreReturn>): Promise<void> {

   425→    prepared: 'return_prepared',
   426→    validated: 'return_prepared',

   431→  };

   433→  const activityType = statusActivityMap[status] || 'return_prepared';
   434→  const statusLabel = status.charAt(0).toUpperCase() + status.slice(1);
   435→
   436→  addActivity({
   437→    type: activityType,
   438→    title: `Return ${statusLabel}`,
   439→    description: `${data.returnType} for ${data.period} moved to ${statusLabel}`,
   440→    clientId: data.clientId,
   441→    entityType: COLLECTIONS.RETURNS,
   442→    entityId: returnId,

   445→  if (status === 'reviewed' || status === 'validated' || status === 'generated') {
   446→    addNotification({
   447→      type: 'return_reviewed',
   448→      priority: 'normal',
   449→      title: `${data.returnType} ready to file`,
   450→      message: `${data.returnType} for ${data.period} has been ${statusLabel} and is ready for filing.`,
   451→      entityType: COLLECTIONS.RETURNS,
   452→      entityId: returnId,
   453→    });
   454→
   455→    // Update compliance score
   456→    recalculateComplianceScore(data.clientId);
   457→  }
   458→}

   508→/** Auto-create draft returns for current period */
   509→async function autoCreateDraftReturns(firmId: string, clientId: string): Promise<void> {
   510→  const now = new Date();
   511→  const period = `${String(now.getMonth() + 1).padStart(2, '0')}-${now.getFullYear()}`;
   512→  const fy = now.getMonth() >= 3 ? `${now.getFullYear()}-${String(now.getFullYear() + 1).slice(2)}` : `${now.getFullYear() - 1}-${String(now.getFullYear()).slice(2)}`;
   513→
   514→  for (const returnType of ['GSTR-1', 'GSTR-3B'] as const) {
   515→    // Check if return already exists
   516→    const q = query(
   517→      collection(db, COLLECTIONS.RETURNS),
   518→      where('clientId', '==', clientId),
   519→      where('returnType', '==', returnType),
   520→      where('period', '==', period),
   521→      limit(1)
   522→    );
   523→    const existing = await getDocs(q);
   524→    if (existing.empty) {
   525→      await createReturn({
   526→        clientId,
   527→        returnType,
   528→        period,
   529→        financialYear: fy,
   530→        status: 'draft',
   531→        filedDate: null,
   532→        acknowledgmentNumber: null,
   533→        totalInvoices: 0,
   534→        readyForFiling: 0,
   535→        issuesFound: 0,
   536→        criticalErrors: 0,
   537→        warnings: 0,
   538→        totalTaxableValue: 0,
   539→        totalTax: 0,
   540→        jsonPayload: null,
   541→        assignedTo: null,
   542→        reviewedBy: null,
   543→      });
   544→    }
   545→  }
   546→}

   548→/** Update draft return when an invoice is added */
   549→async function updateDraftReturnForInvoice(firmId: string, clientId: string, returnType: string, period: string, taxableValue: number, taxAmount: number): Promise<void> {
   550→  try {
   551→    const q = query(
   552→      collection(db, COLLECTIONS.RETURNS),
   553→      where('clientId', '==', clientId),
   554→      where('returnType', '==', returnType),
   555→      where('period', '==', period),
   556→      limit(1)
   557→    );
   558→    const snap = await getDocs(q);
   559→    if (!snap.empty) {
   560→      const returnDoc = snap.docs[0];
   561→      const data = returnDoc.data() as FirestoreReturn;
   562→      await updateDoc(returnDoc.ref, {
   563→        totalInvoices: (data.totalInvoices || 0) + 1,
   564→        totalTaxableValue: (data.totalTaxableValue || 0) + taxableValue,
   565→        totalTax: (data.totalTax || 0) + taxAmount,
   566→        updatedAt: serverTimestamp(),
   567→      });
   568→    }
   569→  } catch (e) {
   570→    console.warn('[Workflow] Failed to update draft return:', e);
   571→  }

   580→export async function createReconciliation(data: {

   585→  const firmId = currentFirmId();

   588→
   589→  // Simulate reconciliation results

   591→  const matched = Math.floor(totalRecords * (0.6 + Math.random() * 0.25));
   592→  const partialMatches = Math.floor((totalRecords - matched) * 0.4);
   593→  const unmatched = totalRecords - matched - partialMatches;
   594→  const highRisk = Math.floor(unmatched * 0.3);
   595→
   596→  const mismatches: ReconMismatch[] = [];
   597→  for (let i = 0; i < unmatched + partialMatches; i++) {

   600→    mismatches.push({

   602→      invoiceDate: new Date().toISOString().split('T')[0],

   606→      matchScore: i < partialMatches ? 70 + Math.floor(Math.random() * 20) : 0,

   608→      portalAmount: booksAmount + (Math.random() > 0.5 ? diff : -diff),
   609→      difference: diff,

   611→      resolved: false,
   612→      resolvedBy: null,
   613→      resolvedAt: null,
   614→    });
   615→  }
   616→
   617→  const gstDifference = mismatches.reduce((sum, m) => sum + m.difference, 0);
   618→
   619→  const reconData: FirestoreReconciliation = {

   621→    firmId: firmId || '',

   632→    mismatches,
   633→    runBy: currentUserId(),
   634→    createdAt: serverTimestamp(),
   635→    updatedAt: serverTimestamp(),
   636→  };
   637→
   638→  await setDoc(reconRef, reconData);
   639→
   640→  addActivity({
   641→    type: 'reconciliation_run',

   644→    clientId: data.clientId,
   645→    entityType: COLLECTIONS.RECONCILIATIONS,
   646→    entityId: reconId,
   647→    metadata: { matched, unmatched, gstDifference },

   660→    entityType: COLLECTIONS.RECONCILIATIONS,
   661→    entityId: reconId,
   662→  });
   663→
   664→  return reconId;
   665→}

   904→/** Subscribe to a Firestore collection with firm scoping */
   905→export function subscribeToCollection<T>(
   906→  collectionName: CollectionName,
   907→  callback: (data: Array<T & { id: string }>) => void,
   908→  ...constraints: QueryConstraint[]
   909→): Unsubscribe {
   910→  const firmId = currentFirmId();
   911→
   912→  const q = firmId
   913→    ? query(collection(db, collectionName), where('firmId', '==', firmId), ...constraints)
   914→    : query(collection(db, collectionName), ...constraints);
   915→
   916→  return onSnapshot(q, (snap) => {
   917→    const data = snap.docs.map(d => {
   918→      const docData = d.data();
   919→      const converted: Record<string, unknown> = { id: d.id };
   920→      for (const [key, value] of Object.entries(docData)) {
   921→        if (value && typeof value === 'object' && 'toDate' in value && typeof (value as { toDate: () => Date }).toDate === 'function') {
   922→          converted[key] = (value as { toDate: () => Date }).toDate().toISOString();
   923→        } else {
   924→          converted[key] = value;
   925→        }
   926→      }
   927→      return converted as T & { id: string };
   928→    });
   929→    callback(data);
   930→  }, (error) => {
   931→    console.warn(`[Firestore] Subscription error for ${collectionName}:`, error);
   932→    callback([]);
   933→  });
   934→}

   936→/** Subscribe to a single Firestore document */
   937→export function subscribeToDoc<T>(
   938→  collectionName: CollectionName,
   939→  docId: string,
   940→  callback: (data: (T & { id: string }) | null) => void,
   941→): Unsubscribe {
   942→  return onSnapshot(doc(db, collectionName, docId), (snap) => {
   943→    if (!snap.exists()) {
   944→      callback(null);
   945→      return;
   946→    }
   947→    const docData = snap.data();
   948→    const converted: Record<string, unknown> = { id: snap.id };
   949→    for (const [key, value] of Object.entries(docData)) {
   950→      if (value && typeof value === 'object' && 'toDate' in value && typeof (value as { toDate: () => Date }).toDate === 'function') {
   951→        converted[key] = (value as { toDate: () => Date }).toDate().toISOString();
   952→      } else {
   953→        converted[key] = value;
   954→      }
   955→    }
   956→    callback(converted as T & { id: string });
   957→  }, (error) => {
   958→    console.warn(`[Firestore] Doc subscription error for ${collectionName}/${docId}:`, error);
   959→    callback(null);
   960→  });
   961→}

   963→/** Compute live dashboard metrics from Firestore data */
   964→export function computeDashboardMetrics(
   965→  clients: Array<FirestoreClient & { id: string }>,
   966→  invoices: Array<FirestoreInvoice & { id: string }>,
   967→  returns: Array<FirestoreReturn & { id: string }>,
   968→  documents: Array<FirestoreDocument & { id: string }>,
   969→  activities: Array<FirestoreActivity & { id: string }>,
   970→): LiveDashboardMetrics {
   971→  const activeClients = clients.filter(c => c.status === 'active').length;
   972→  const filedReturns = returns.filter(r => r.status === 'filed').length;
   973→  const pendingReturns = returns.filter(r => r.status !== 'filed').length;
   974→  const readyToFile = returns.filter(r => ['validated', 'reviewed', 'generated'].includes(r.status)).length;
   975→  const overdueReturns = returns.filter(r => {
   976→    if (r.status === 'filed') return false;
   977→    const periodDate = parsePeriod(r.period);
   978→    return periodDate && periodDate < new Date();
   979→  }).length;
   980→
   981→  const totalTaxVolume = invoices.reduce((sum, inv) => sum + (inv.totalAmount || 0), 0);
   982→  const totalTax = invoices.reduce((sum, inv) => sum + (inv.cgst || 0) + (inv.sgst || 0) + (inv.igst || 0) + (inv.cess || 0), 0);
   983→
   984→  const criticalIssues = invoices.filter(inv => inv.riskLevel === 'critical').length +
   985→    returns.filter(r => r.criticalErrors > 0).length;
   986→  const warnings = invoices.filter(inv => inv.riskLevel === 'high' || inv.riskLevel === 'medium').length +
   987→    returns.filter(r => r.warnings > 0).length;
   988→
   989→  const averageHealthScore = clients.length > 0
   990→    ? Math.round(clients.reduce((sum, c) => sum + c.healthScore, 0) / clients.length)
   991→    : 0;
   992→
   993→  const matchedInvoices = invoices.filter(inv => inv.matchStatus === 'perfect_match').length;
   994→  const matchPercentage = invoices.length > 0 ? Math.round((matchedInvoices / invoices.length) * 100) : 100;
   995→
   996→  const riskyInvoices = invoices.filter(inv => ['high', 'critical'].includes(inv.riskLevel)).length;
   997→  const riskPercentage = invoices.length > 0 ? Math.round((riskyInvoices / invoices.length) * 100) : 0;
   998→
   999→  const documentsProcessed = documents.filter(d => d.extractionStatus === 'completed').length;
  1000→  const extractionsPending = documents.filter(d => d.extractionStatus === 'pending' || d.extractionStatus === 'in_progress').length;
  1001→
  1002→  const recentActivities = [...activities]
  1003→    .sort((a, b) => {
  1004→      const aTime = typeof a.createdAt === 'string' ? new Date(a.createdAt).getTime() : 0;
  1005→      const bTime = typeof b.createdAt === 'string' ? new Date(b.createdAt).getTime() : 0;
  1006→      return bTime - aTime;
  1007→    })
  1008→    .slice(0, 10);
  1009→
  1010→  const upcomingFilings = returns
  1011→    .filter(r => r.status !== 'filed')
  1012→    .sort((a, b) => {
  1013→      const aTime = typeof a.createdAt === 'string' ? new Date(a.createdAt).getTime() : 0;
  1014→      const bTime = typeof b.createdAt === 'string' ? new Date(b.createdAt).getTime() : 0;
  1015→      return aTime - bTime;
  1016→    })
  1017→    .slice(0, 5);
  1018→
  1019→  return {
  1020→    totalClients: clients.length,
  1021→    activeClients,
  1022→    totalInvoices: invoices.length,
  1023→    totalTaxVolume,
  1024→    filedReturns,
  1025→    pendingReturns,
  1026→    overdueReturns,
  1027→    readyToFile,
  1028→    criticalIssues,
  1029→    warnings,
  1030→    averageHealthScore,
  1031→    matchPercentage,
  1032→    riskPercentage,
  1033→    documentsProcessed,
  1034→    extractionsPending,
  1035→    recentActivities,
  1036→    upcomingFilings,
  1037→  };
  1038→}

  1040→function parsePeriod(period: string | null): Date | null {
  1041→  if (!period) return null;
  1042→  try {
  1043→    const [mm, yyyy] = period.split('-').map(Number);
  1044→    if (!mm || !yyyy) return null;
  1045→    return new Date(yyyy, mm, 20); // Due date is typically 20th of next month
  1046→  } catch {
  1047→    return null;
  1048→  }
  1049→}
