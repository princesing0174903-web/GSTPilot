    62→    62→function BalanceTrendChart({ data }: { data: { day: string; balance: number }[] }) {
    63→    63→  const max = Math.max(...data.map(d => d.balance))
    64→    64→  const min = Math.min(...data.map(d => d.balance))
    65→    65→  const range = max - min || 1
    66→    66→  const w = 280
    67→    67→  const h = 60
    68→    68→  const pts = data.map((d, i) => `${(i / (data.length - 1)) * w},${h - ((d.balance - min) / range) * (h - 8) - 4}`)
    69→    69→  return (
    70→    70→    <svg width={w} height={h + 16} className="overflow-visible">
    71→    71→      <defs>
    72→    72→        <linearGradient id="balGrad" x1="0" y1="0" x2="0" y2="1">
    73→    73→          <stop offset="0%" stopColor="#10b981" stopOpacity="0.2" />
    74→    74→          <stop offset="100%" stopColor="#10b981" stopOpacity="0.02" />
    75→    75→        </linearGradient>
    76→    76→      </defs>
    77→    77→      <polygon points={`0,${h} ${pts.join(' ')} ${w},${h}`} fill="url(#balGrad)" />
    78→    78→      <polyline points={pts.join(' ')} fill="none" stroke="#10b981" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" />
    79→    79→      {data.filter((_, i) => i % 2 === 0).map((d, i) => {
    80→    80→        const x = (data.indexOf(d) / (data.length - 1)) * w
    81→    81→        return <text key={i} x={x} y={h + 12} textAnchor="middle" className="text-[8px] fill-muted-foreground">{d.day}</text>
    82→    82→      })}
    83→    83→    </svg>
    84→    84→  )
    85→    85→}

   180→   180→function formatSyncDate(iso: string | null): string {
   181→   181→  if (!iso) return '—'
   182→   182→  const d = new Date(iso)
   183→   183→  if (Number.isNaN(d.getTime())) return '—'
   184→   184→  const dd = String(d.getDate()).padStart(2, '0')
   185→   185→  const mm = String(d.getMonth() + 1).padStart(2, '0')
   186→   186→  const yyyy = d.getFullYear()
   187→   187→  const hh = String(d.getHours()).padStart(2, '0')
   188→   188→  const min = String(d.getMinutes()).padStart(2, '0')
   189→   189→  return `${dd}/${mm}/${yyyy} ${hh}:${min}`
   190→   190→}
   191→   191→
   192→   192→function formatTxnDate(iso: string): string {
   193→   193→  if (!iso) return '—'
   194→   194→  const d = new Date(iso)
   195→   195→  if (Number.isNaN(d.getTime())) return iso
   196→   196→  const dd = String(d.getDate()).padStart(2, '0')
   197→   197→  const mm = String(d.getMonth() + 1).padStart(2, '0')
   198→   198→  const yyyy = d.getFullYear()
   199→   199→  return `${dd}/${mm}/${yyyy}`
   200→   200→}

   206→   206→// Map Firestore BankAccount → BankAccount shape used by this page.
   207→   207→function mapBankAccount(acc: FirestoreBankAccount & { id: string }): BankAccount {
   208→   208→  const accountType = (acc.accountType || 'current') as string
   209→   209→  return {
   210→   210→    id: acc.id,
   211→   211→    bank: acc.bankName || 'Bank Account',
   212→   212→    account: acc.accountNumberMasked || '****',
   213→   213→    type: accountType.charAt(0).toUpperCase() + accountType.slice(1),
   214→   214→    balance: Number(acc.currentBalance ?? acc.availableBalance ?? 0) || 0,
   215→   215→    lastSync: formatSyncDate((acc.lastSyncAt as string | null) ?? null),
   216→   216→    status: (acc.status as string) || 'connected',
   217→   217→  }
   218→   218→}

   220→   220→// Map Firestore BankTransaction → BankTransaction shape used by this page.
   221→   221→function mapBankTxnToTxn(t: FirestoreBankTransaction & { id: string }): BankTransaction {
   222→   222→  const type = t.type === 'credit' ? 'credit' : 'debit'
   223→   223→  return {
   224→   224→    id: t.id,
   225→   225→    date: formatTxnDate(t.date || ''),
   226→   226→    description: t.description || 'Bank Transaction',
   227→   227→    amount: Math.abs(Number(t.amount) || 0),
   228→   228→    type,
   229→   229→    balance: t.balanceAfter == null ? null : Number(t.balanceAfter),
   230→   230→    account: t.referenceNo ? t.referenceNo.toUpperCase() : 'BANK',
   231→   231→    category: (t.category as string) || (type === 'credit' ? 'Revenue' : 'Purchase'),
   232→   232→  }
   233→   233→}

   235→   235→// Map Firestore Payment row → BankTransaction shape used by this page.
   236→   236→function mapPaymentToTxn(p: FirestorePayment & { id: string }): BankTransaction {
   237→   237→  const isVendor = (p.partyType || 'customer') === 'vendor'
   238→   238→  return {
   239→   239→    id: p.id,
   240→   240→    date: formatTxnDate(p.paymentDate || ''),
   241→   241→    description: `${upperMode(p.paymentMode || 'bank')} - ${p.partyName || 'Unknown'}`,
   242→   242→    amount: Number(p.amount) || 0,
   243→   243→    type: isVendor ? 'debit' : 'credit',
   244→   244→    balance: null,
   245→   245→    account: upperMode(p.paymentMode || 'bank'),
   246→   246→    category: isVendor ? 'Purchase' : 'Revenue',
   247→   247→  }
   248→   248→}

   250→   250→// Map Firestore Payment row → ReconciliationEntry shape used by this page.
   251→   251→function mapPaymentToRecon(p: FirestorePayment & { id: string }): ReconciliationEntry {
   252→   252→  let status: ReconciliationEntry['status'] = 'unmatched'
   253→   253→  if (p.reconciled) status = 'matched'
   254→   254→  else if (p.status === 'failed') status = 'disputed'
   255→   255→  return {
   256→   256→    id: p.id,
   257→   257→    date: formatTxnDate(p.paymentDate || ''),
   258→   258→    bankTxn: p.referenceNo || `${upperMode(p.paymentMode || 'bank')}-${p.id.slice(-6)}`,
   259→   259→    bookEntry: p.invoiceId || null,
   260→   260→    amount: Number(p.amount) || 0,
   261→   261→    status,
   262→   262→    account: upperMode(p.paymentMode || 'bank'),
   263→   263→  }
   264→   264→}
