   125→function getDaysRemaining(dueDateStr: string): number {
   126→  const dueDate = new Date(dueDateStr);
   127→  const today = new Date('2025-07-08');
   128→  today.setHours(0, 0, 0, 0);
   129→  dueDate.setHours(0, 0, 0, 0);
   130→  return Math.ceil((dueDate.getTime() - today.getTime()) / (1000 * 60 * 60 * 24));
   131→}
   132→
   133→function formatDaysRemaining(days: number): string {
   134→  if (days < 0) return `${Math.abs(days)}d overdue`;
   135→  if (days === 0) return 'Due today';
   136→  if (days === 1) return 'Tomorrow';
   137→  return `${days}d left`;
   138→}
   139→
   140→function getInitials(name: string): string {
   141→  return name
   142→    .split(' ')
   143→    .map(w => w[0])
   144→    .join('')
   145→    .slice(0, 2)
   146→    .toUpperCase();
   147→}

   182→// Map Issue category to BlockingIssue category
   183→function mapIssueCategory(issue: Issue): BlockingIssue['category'] {
   184→  const cat = issue.category?.toLowerCase() ?? '';
   185→  if (cat.includes('gstin')) return 'gstin_error';
   186→  if (cat.includes('missing') || cat.includes('invoice')) return 'missing_invoice';
   187→  if (cat.includes('recon') || cat.includes('mismatch')) return 'recon_mismatch';
   188→  return 'validation_failure';
   189→}
