   210→export function OnboardingFlow({
   211→  onComplete,
   212→  onSkip,
   213→  userEmail = '',
   214→  userName = '',
   215→  saving = false,
   216→  error = null,
   217→  onDismissError,
   218→}: OnboardingFlowProps) {
   219→  const [currentStep, setCurrentStep] = useState(0); // 0=welcome, 1-4=steps, 5=final
   220→  const [direction, setDirection] = useState(1);
   221→
   222→  const [formData, setFormData] = useState<OnboardingData>({
   223→    fullName: userName,
   224→    email: userEmail,
   225→    phone: '',
   226→    ageGroup: '',
   227→    profession: '',
   228→    experience: '',
   229→    firmName: '',
   230→    organizationType: '',
   231→    gstin: '',
   232→    state: '',
   233→    stateCode: '',
   234→    icaiMembershipNo: '',
   235→    officeAddress: '',
   236→    clientCount: '',
   237→    monthlyReturns: '',
   238→    gstServices: [],
   239→    painPoints: [],
   240→    referralSource: '',
   241→    trialReasons: [],
   242→    wantsUpdates: true,
   243→  });
   244→
   245→  const [errors, setErrors] = useState<Record<string, string>>({});
   246→
   247→  // ─── Helpers ──────────────────────────────────────────────────────────────
   248→
   249→  const updateField = useCallback(
   250→    <K extends keyof OnboardingData>(field: K, value: OnboardingData[K]) => {
   251→      setFormData((prev) => ({ ...prev, [field]: value }));
   252→      // Clear error on change
   253→      setErrors((prev) => {
   254→        const next = { ...prev };
   255→        delete next[field];
   256→        return next;
   257→      });
   258→    },
   259→    []
   260→  );
   261→
   262→  const toggleArrayItem = useCallback(
   263→    (field: 'gstServices' | 'painPoints' | 'trialReasons', item: string) => {
   264→      setFormData((prev) => {
   265→        const arr = prev[field] as string[];
   266→        const next = arr.includes(item)
   267→          ? arr.filter((i) => i !== item)
   268→          : [...arr, item];
   269→        return { ...prev, [field]: next };
   270→      });
   271→    },
   272→    []
   273→  );
   274→
   275→  const goNext = useCallback(() => {
   276→    setDirection(1);
   277→    setCurrentStep((prev) => Math.min(prev + 1, 5));
   278→  }, []);
   279→
   280→  const goBack = useCallback(() => {
   281→    setDirection(-1);
   282→    setCurrentStep((prev) => Math.max(prev - 1, 0));
   283→  }, []);
   284→
   285→  // ─── Validation ───────────────────────────────────────────────────────────
   286→
   287→  const validateStep = useCallback(
   288→    (step: number): boolean => {
   289→      const newErrors: Record<string, string> = {};
   290→
   291→      if (step === 1) {
   292→        if (!formData.fullName.trim()) newErrors.fullName = 'Full name is required';
   293→        if (!formData.email.trim()) newErrors.email = 'Email is required';
   294→        else if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(formData.email))
   295→          newErrors.email = 'Enter a valid email';
   296→        if (!formData.phone.trim()) newErrors.phone = 'Mobile number is required';
   297→        else if (!/^\d{10}$/.test(formData.phone.replace(/\s/g, '')))
   298→          newErrors.phone = 'Enter a valid 10-digit mobile number';
   299→      }
   300→
   301→      if (step === 2) {
   302→        if (!formData.firmName.trim()) newErrors.firmName = 'Firm name is required';
   303→      }
   304→
   305→      setErrors(newErrors);
   306→      return Object.keys(newErrors).length === 0;
   307→    },
   308→    [formData]
   309→  );
   310→
   311→  const handleNext = useCallback(() => {
   312→    if (validateStep(currentStep)) {
   313→      goNext();
   314→    }
   315→  }, [currentStep, validateStep, goNext]);
   316→
   317→  // ─── GSTIN Validation ─────────────────────────────────────────────────────
   318→
   319→  const getGstinStatus = useCallback(() => {
   320→    const val = formData.gstin.trim();
   321→    if (!val) return null;
   322→    return GSTIN_REGEX.test(val) ? 'valid' : 'invalid';
   323→  }, [formData.gstin]);
   324→
   325→  // ─── State Code Lookup ────────────────────────────────────────────────────
   326→
   327→  const handleStateChange = useCallback(
   328→    (stateName: string) => {
   329→      const stateObj = INDIAN_STATES.find((s) => s.name === stateName);
   330→      setFormData((prev) => ({
   331→        ...prev,
   332→        state: stateName,
   333→        stateCode: stateObj?.code ?? '',
   334→      }));
   335→      setErrors((prev) => {
   336→        const next = { ...prev };
   337→        delete next.state;
   338→        return next;
   339→      });
   340→    },
   341→    []
   342→  );
   343→
   344→  // ─── Progress ─────────────────────────────────────────────────────────────
   345→
   346→  const totalSteps = 5;
   347→  const progressPercent =
   348→    currentStep === 0 ? 0 : (currentStep / totalSteps) * 100;
   349→
   350→  // ─── Step Icons ───────────────────────────────────────────────────────────
   351→
   352→  const stepIcons = [Zap, User, Building2, Briefcase, Target];
   353→
   354→  // ─── Render Step Content ──────────────────────────────────────────────────
   355→
   356→  const renderStep = () => {
   357→    // Step 0 — Welcome
   358→    if (currentStep === 0) {
   359→      return (
   360→        <motion.div
   361→          key="step-0"
   362→          custom={direction}
   363→          variants={slideVariants}
   364→          initial="enter"
   365→          animate="center"
   366→          exit="exit"
   367→          transition={stepTransition}
   368→          className="flex flex-col items-center justify-center text-center py-8"
   369→        >
   370→          <motion.div
   371→            initial={{ scale: 0.5, opacity: 0 }}
   372→            animate={{ scale: 1, opacity: 1 }}
   373→            transition={{ type: 'spring', stiffness: 200, damping: 15, delay: 0.1 }}
   374→            className="mb-8"
   375→          >
   376→            <div className="w-20 h-20 rounded-2xl bg-gradient-to-br from-emerald-400 to-teal-600 flex items-center justify-center shadow-lg shadow-emerald-200">
   377→              <Zap className="size-10 text-white" />
   378→            </div>
   379→          </motion.div>
   380→          <motion.h1
   381→            initial={{ y: 20, opacity: 0 }}
   382→            animate={{ y: 0, opacity: 1 }}
   383→            transition={{ delay: 0.2 }}
   384→            className="text-2xl font-bold tracking-tight text-foreground"
   385→          >
   386→            Welcome to GSTPilot
   387→          </motion.h1>
   388→          <motion.p
   389→            initial={{ y: 20, opacity: 0 }}
   390→            animate={{ y: 0, opacity: 1 }}
   391→            transition={{ delay: 0.3 }}
   392→            className="text-muted-foreground mt-2 max-w-sm"
   393→          >
   394→            Let&apos;s set up your workspace in less than 2 minutes.
   395→          </motion.p>
   396→          <motion.div
   397→            initial={{ y: 20, opacity: 0 }}
   398→            animate={{ y: 0, opacity: 1 }}
   399→            transition={{ delay: 0.4 }}
   400→            className="mt-8"
   401→          >
   402→            <Button
   403→              onClick={goNext}
   404→              size="lg"
   405→              className="bg-emerald-600 hover:bg-emerald-700 text-white gap-2 px-8"
   406→            >
   407→              Get Started
   408→              <ArrowRight className="size-4" />
   409→            </Button>
   410→          </motion.div>
   411→        </motion.div>
   412→      );
   413→    }
   414→
   415→    // Step 1 — About You
   416→    if (currentStep === 1) {
   417→      return (
   418→        <motion.div
   419→          key="step-1"
   420→          custom={direction}
   421→          variants={slideVariants}
   422→          initial="enter"
   423→          animate="center"
   424→          exit="exit"
   425→          transition={stepTransition}
   426→          className="py-4"
   427→        >
   428→          <div className="flex items-center gap-3 mb-6">
   429→            <div className="w-10 h-10 rounded-xl bg-emerald-100 flex items-center justify-center">
   430→              <User className="size-5 text-emerald-600" />
   431→            </div>
   432→            <div>
   433→              <h2 className="text-2xl font-bold">About You</h2>
   434→              <p className="text-sm text-muted-foreground">
   435→                Tell us about yourself so we can personalize your experience
   436→              </p>
   437→            </div>
   438→          </div>
   439→
   440→          <div className="space-y-5">
   441→            {/* Full Name */}
   442→            <div className="space-y-2">
   443→              <Label htmlFor="fullName">
   444→                Full Name <span className="text-destructive">*</span>
   445→              </Label>
   446→              <Input
   447→                id="fullName"
   448→                placeholder="Rajesh Kumar Sharma"
   449→                value={formData.fullName}
   450→                onChange={(e) => updateField('fullName', e.target.value)}
   451→                aria-invalid={!!errors.fullName}
   452→                className={errors.fullName ? 'border-destructive' : ''}
   453→              />
   454→              {errors.fullName && (
   455→                <p className="text-xs text-destructive">{errors.fullName}</p>
   456→              )}
   457→            </div>
   458→
   459→            {/* Work Email */}
   460→            <div className="space-y-2">

   464→              <Input
   465→                id="email"
   466→                type="email"
   467→                placeholder="rajesh@yourfirm.com"
   468→                value={formData.email}
   469→                onChange={(e) => updateField('email', e.target.value)}
   470→                aria-invalid={!!errors.email}
   471→                className={errors.email ? 'border-destructive' : ''}
   472→              />
   473→              {errors.email && (
   474→                <p className="text-xs text-destructive">{errors.email}</p>
   475→              )}
   476→            </div>
   477→
   478→            {/* Mobile Number */}
   479→            <div className="space-y-2">
   480→              <Label htmlFor="phone">
   481→                Mobile Number <span className="text-destructive">*</span>
   482→              </Label>
   483→              <div className="flex">
   484→                <div className="flex items-center rounded-l-md border border-r-0 bg-muted px-3 text-sm text-muted-foreground h-9">
   485→                  +91
   486→                </div>
   487→                <Input
   488→                  id="phone"
   489→                  type="tel"
   490→                  placeholder="9876543210"
   491→                  value={formData.phone}
   492→                  onChange={(e) =>
   493→                    updateField('phone', e.target.value.replace(/[^\d]/g, '').slice(0, 10))

   495→                  aria-invalid={!!errors.phone}
   496→                  className={`rounded-l-none ${errors.phone ? 'border-destructive' : ''}`}
   497→                />
   498→              </div>
   499→              {errors.phone && (
   500→                <p className="text-xs text-destructive">{errors.phone}</p>
   501→              )}
   502→            </div>
   503→
   504→            {/* Age Group */}
   505→            <div className="space-y-2">
   506→              <Label>Age Group</Label>
   507→              <Select
   508→                value={formData.ageGroup}
   509→                onValueChange={(v) => updateField('ageGroup', v)}
   510→              >
   511→                <SelectTrigger className="w-full">
   512→                  <SelectValue placeholder="Select age group" />
   513→                </SelectTrigger>
   514→                <SelectContent>
   515→                  {AGE_GROUPS.map((g) => (
   516→                    <SelectItem key={g} value={g}>
   517→                      {g}
   518→                    </SelectItem>
   519→                  ))}
   520→                </SelectContent>
   521→              </Select>
   522→            </div>
   523→
   524→            {/* Profession */}
   525→            <div className="space-y-2">
   526→              <Label>Profession</Label>
   527→              <Select
   528→                value={formData.profession}
   529→                onValueChange={(v) => updateField('profession', v)}
   530→              >
   531→                <SelectTrigger className="w-full">
   532→                  <SelectValue placeholder="Select your profession" />
   533→                </SelectTrigger>
   534→                <SelectContent>
   535→                  {PROFESSIONS.map((p) => (
   536→                    <SelectItem key={p} value={p}>
   537→                      {p}
   538→                    </SelectItem>
   539→                  ))}
   540→                </SelectContent>
   541→              </Select>
   542→            </div>
   543→
   544→            {/* Years of Experience */}
   545→            <div className="space-y-2">
   546→              <Label>Years of Experience</Label>
   547→              <Select
   548→                value={formData.experience}
   549→                onValueChange={(v) => updateField('experience', v)}
   550→              >
   551→                <SelectTrigger className="w-full">
   552→                  <SelectValue placeholder="Select experience" />
   553→                </SelectTrigger>
   554→                <SelectContent>
   555→                  {EXPERIENCE_LEVELS.map((e) => (
   556→                    <SelectItem key={e} value={e}>
   557→                      {e}
   558→                    </SelectItem>
   559→                  ))}
   560→                </SelectContent>
   561→              </Select>
   562→            </div>
   563→          </div>
   564→        </motion.div>
   565→      );
   566→    }
   567→
   568→    // Step 2 — Firm Information
   569→    if (currentStep === 2) {
   570→      const gstinStatus = getGstinStatus();
   571→      return (
   572→        <motion.div
   573→          key="step-2"
   574→          custom={direction}
   575→          variants={slideVariants}
   576→          initial="enter"
   577→          animate="center"
   578→          exit="exit"
   579→          transition={stepTransition}
   580→          className="py-4"
   581→        >
   582→          <div className="flex items-center gap-3 mb-6">
   583→            <div className="w-10 h-10 rounded-xl bg-emerald-100 flex items-center justify-center">
   584→              <Building2 className="size-5 text-emerald-600" />
   585→            </div>
   586→            <div>
   587→              <h2 className="text-2xl font-bold">Firm Information</h2>
   588→              <p className="text-sm text-muted-foreground">
   589→                Details about your CA firm or practice
   590→              </p>
   591→            </div>
   592→          </div>
   593→
   594→          <div className="space-y-5">
   595→            {/* Firm Name */}
   596→            <div className="space-y-2">
   597→              <Label htmlFor="firmName">
   598→                Firm Name <span className="text-destructive">*</span>
   599→              </Label>
   600→              <Input
   601→                id="firmName"
   602→                placeholder="Sharma & Associates"
   603→                value={formData.firmName}
   604→                onChange={(e) => updateField('firmName', e.target.value)}
   605→                aria-invalid={!!errors.firmName}
   606→                className={errors.firmName ? 'border-destructive' : ''}
   607→              />
   608→              {errors.firmName && (
   609→                <p className="text-xs text-destructive">{errors.firmName}</p>
   610→              )}
   611→            </div>
   612→
   613→            {/* Type of Organization */}
   614→            <div className="space-y-2">
   615→              <Label>Type of Organization</Label>

   617→                value={formData.organizationType}
   618→                onValueChange={(v) => updateField('organizationType', v)}
   619→              >
   620→                <SelectTrigger className="w-full">
   621→                  <SelectValue placeholder="Select organization type" />

   624→                  {ORGANIZATION_TYPES.map((t) => (

   633→            {/* GSTIN */}

   637→                <Input
   638→                  id="gstin"
   639→                  placeholder="22AAAAA0000A1Z5"

   642→                    updateField(
   643→                      'gstin',
