   164→function formatFileSize(bytes: number): string {
   165→  if (bytes < 1024) return `${bytes} B`;
   166→  if (bytes < 1048576) return `${(bytes / 1024).toFixed(1)} KB`;
   167→  return `${(bytes / 1048576).toFixed(1)} MB`;
   168→}

   170→function getConfidenceColor(confidence: number): string {
   171→  if (confidence >= 95) return 'text-emerald-600';
   172→  if (confidence >= 80) return 'text-amber-600';
   173→  return 'text-red-600';
   174→}

   176→function getConfidenceBarColor(confidence: number): string {

   227→// ─── Map API Invoice to local ExtractedInvoice ────────────────────────────────
   228→
   229→function mapInvoiceStatus(status: string): ValidationStatus {
   230→  if (status === 'approved' || status === 'filed') return 'validated';
   231→  if (status === 'draft') return 'warning';
   232→  return 'error';
   233→}
   234→
   235→function mapInvoiceToExtracted(inv: Invoice, clientName: string): ExtractedInvoice {
   236→  const vStatus = mapInvoiceStatus(inv.status);
   237→  return {
   238→    id: inv.id,
   239→    invoiceNumber: inv.invoiceNumber,
   240→    clientName,
   241→    invoiceDate: inv.invoiceDate,
   242→    section: inv.gstr1Section,
   243→    sectionLabel: GSTR1_SECTION_LABELS[inv.gstr1Section] ?? inv.gstr1Section.toUpperCase(),
   244→    taxableValue: inv.taxableValue,
   245→    cgst: inv.cgst,
   246→    sgst: inv.sgst,
   247→    igst: inv.igst,
   248→    totalTax: inv.cgst + inv.sgst + inv.igst,
   249→    totalAmount: inv.totalAmount,
   250→    validationStatus: vStatus,
   251→    ocrConfidence: vStatus === 'error' ? 75.0 : vStatus === 'warning' ? 92.0 : 98.5,
   252→    missingFields: inv.aiExplanation ? [inv.aiExplanation.split(' ').slice(0, 2).join(' ')] : [],
   253→    validationMessages: inv.aiExplanation ? [inv.aiExplanation] : [],
   254→  };
   255→}

   313→      const clientName = clientMap.get(inv.clientId)?.tradeName ?? 'Unknown';

   331→  // ── Loading ──

   333→

   335→  const pipelineCounts = useMemo(() => {
   336→    const uploading = processingFiles.filter(f => f.status === 'uploading').length;
   337→    const extracting = processingFiles.filter(f => f.status === 'extracting' || f.status === 'validating').length;

   339→    const errors = invoices.filter(i => i.validationStatus === 'error').length;
   340→    return { uploading, extracting, validated, errors };
   341→  }, [processingFiles, invoices]);
   342→
   343→  // ── Validation Issues from API ──
   344→  const apiIssues = issuesData?.issues ?? [];
   345→
   346→  const validationIssues: ValidationIssueItem[] = useMemo(() => {
   347→    // First, derive issues from invoice validation status
   348→    const issues: ValidationIssueItem[] = [];
   349→    for (const inv of invoices) {
   350→      if (inv.validationStatus === 'error') {
   351→        for (const msg of inv.validationMessages) {
   352→          issues.push({
   353→            id: `${inv.id}-err-${issues.length}`,
   354→            severity: 'critical',
   355→            description: msg,
   356→            invoiceNumber: inv.invoiceNumber,
   357→            invoiceId: inv.id,
   358→          });
   359→        }
   360→      }
   361→      if (inv.validationStatus === 'warning') {

   363→          issues.push({
   364→            id: `${inv.id}-warn-${issues.length}`,
   365→            severity: 'warning',
   366→            description: msg,
   367→            invoiceNumber: inv.invoiceNumber,
   368→            invoiceId: inv.id,
   369→          });
   370→        }
   371→      }
   372→      for (const field of inv.missingFields) {
   373→        issues.push({

   378→          invoiceId: inv.id,
   379→        });
   380→      }
   381→    }

   386→          id: issue.id,

   390→          invoiceId: issue.invoiceId ?? '',

   392→      }
   393→    }

   539→
   540→  // ── Loading Skeleton ──
   541→  if (loading) {

   543→      <div className="space-y-6 p-4 md:p-6">
   544→        <div className="flex items-center justify-between">
   545→          <div className="space-y-2">

   561→  // ── Empty state when no invoices exist ──
   562→  if (invoices.length === 0 && processingFiles.length === 0) {

   568→            animate={{ opacity: 1, y: 0 }}
   569→            transition={{ duration: 0.4 }}
   570→            className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between"
   571→          >
   572→            <div>
   573→              <h1 className="text-2xl font-bold tracking-tight text-slate-900 sm:text-3xl">

   577→                Upload, extract, and validate invoices

   580→          </motion.div>
   581→          <EmptyState
   582→            icon={FileUp}
   583→            title="No invoices uploaded"

   585→            action={{
