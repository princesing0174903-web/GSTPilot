   120→// ──────────────────────────────────────────────
   121→// Period generator (last 12 months)
   122→// ──────────────────────────────────────────────
   123→function getRecentPeriods(count: number = 12): string[] {
   124→  const periods: string[] = [];
   125→  const now = new Date();
   126→  for (let i = 0; i < count; i++) {
   127→    const d = new Date(now.getFullYear(), now.getMonth() - i, 1);
   128→    const yyyy = d.getFullYear();
   129→    const mm = String(d.getMonth() + 1).padStart(2, '0');
   130→    periods.push(`${yyyy}-${mm}`);
   131→  }
   132→  return periods;
   133→}

   135→function periodLabel(period: string): string {
   136→  const months = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];
   137→  const [year, month] = period.split('-').map(Number);
   138→  return `${months[month - 1]} ${year}`;
   139→}

   181→// ──────────────────────────────────────────────
   182→// Match Rate Ring
   183→// ──────────────────────────────────────────────
   184→function MatchRateRing({
   185→  matchPercent,
   186→  partialPercent,
   187→  size = 140,
   188→  strokeWidth = 12,
   189→}: {
   190→  matchPercent: number;
   191→  partialPercent: number;
   192→  size?: number;
   193→  strokeWidth?: number;
   194→}) {
   195→  const radius = (size - strokeWidth) / 2;
   196→  const circumference = 2 * Math.PI * radius;
   197→  const mismatchPercent = Math.max(0, 100 - matchPercent - partialPercent);
   198→
   199→  const matchLen = (matchPercent / 100) * circumference;
   200→  const partialLen = (partialPercent / 100) * circumference;
   201→  const mismatchLen = (mismatchPercent / 100) * circumference;
   202→
   203→  const partialOffset = -(matchLen);
   204→  const mismatchOffset = -(matchLen + partialLen);
   205→
   206→  return (
   207→    <div className="relative inline-flex items-center justify-center" style={{ width: size, height: size }}>
   208→      <svg width={size} height={size} className="-rotate-90">
   209→        <circle cx={size / 2} cy={size / 2} r={radius} fill="none" stroke="#f1f5f9" strokeWidth={strokeWidth} />
   210→        <defs>
   211→          <filter id="ringGlow" x="-20%" y="-20%" width="140%" height="140%">
   212→            <feGaussianBlur stdDeviation="2" result="blur" />
   213→            <feMerge><feMergeNode in="blur" /><feMergeNode in="SourceGraphic" /></feMerge>
   214→          </filter>
   215→        </defs>
   216→        {matchPercent > 0 && (
   217→          <motion.circle
   218→            cx={size / 2} cy={size / 2} r={radius} fill="none" stroke="#10b981"
   219→            strokeWidth={strokeWidth} strokeLinecap="round"
   220→            strokeDasharray={`${matchLen} ${circumference - matchLen}`} strokeDashoffset={0}
   221→            initial={{ strokeDasharray: `0 ${circumference}` }}
   222→            animate={{ strokeDasharray: `${matchLen} ${circumference - matchLen}` }}
   223→            transition={{ duration: 1.2, ease: [0.25, 0.46, 0.45, 0.94] }}
   224→            filter="url(#ringGlow)"
   225→          />
   226→        )}
   227→        {partialPercent > 0 && (
   228→          <motion.circle
   229→            cx={size / 2} cy={size / 2} r={radius} fill="none" stroke="#f59e0b"
   230→            strokeWidth={strokeWidth} strokeLinecap="round"
   231→            strokeDasharray={`${partialLen} ${circumference - partialLen}`} strokeDashoffset={partialOffset}
   232→            initial={{ strokeDasharray: `0 ${circumference}`, strokeDashoffset: 0 }}
   233→            animate={{ strokeDasharray: `${partialLen} ${circumference - partialLen}`, strokeDashoffset: partialOffset }}
   234→            transition={{ duration: 1.2, ease: [0.25, 0.46, 0.45, 0.94], delay: 0.2 }}
   235→            filter="url(#ringGlow)"
   236→          />
   237→        )}
   238→        {mismatchPercent > 0 && (
   239→          <motion.circle
   240→            cx={size / 2} cy={size / 2} r={radius} fill="none" stroke="#ef4444"
   241→            strokeWidth={strokeWidth} strokeLinecap="round"
   242→            strokeDasharray={`${mismatchLen} ${circumference - mismatchLen}`} strokeDashoffset={mismatchOffset}
   243→            initial={{ strokeDasharray: `0 ${circumference}`, strokeDashoffset: 0 }}
   244→            animate={{ strokeDasharray: `${mismatchLen} ${circumference - mismatchLen}`, strokeDashoffset: mismatchOffset }}
   245→            transition={{ duration: 1.2, ease: [0.25, 0.46, 0.45, 0.94], delay: 0.4 }}
   246→            filter="url(#ringGlow)"
   247→          />
   248→        )}
   249→      </svg>
   250→      <div className="absolute inset-0 flex flex-col items-center justify-center">
   251→        <motion.span
   252→          className="text-3xl font-bold tracking-tight text-foreground"
   253→          initial={{ opacity: 0, scale: 0.5 }}
   254→          animate={{ opacity: 1, scale: 1 }}
   255→          transition={{ delay: 0.7, duration: 0.4, type: 'spring' }}
   256→        >
   257→          {matchPercent}%
   258→        </motion.span>
   259→        <span className="text-[10px] font-medium text-muted-foreground">Match Rate</span>
   260→      </div>
   261→    </div>
   262→  );
   263→}
