    72→function getStatusBadge(status: string) {
    73→  switch (status) {
    74→    case 'approved': case 'validated': return <Badge className="bg-emerald-50 text-emerald-700 border-emerald-200 text-[10px] px-1.5 py-0 h-5 gap-0.5"><CheckCircle2 className="size-2.5" />Validated</Badge>;
    75→    case 'draft': return <Badge className="bg-slate-50 text-slate-700 border-slate-200 text-[10px] px-1.5 py-0 h-5 gap-0.5"><Clock className="size-2.5" />Draft</Badge>;
    76→    case 'cancelled': return <Badge className="bg-red-50 text-red-700 border-red-200 text-[10px] px-1.5 py-0 h-5 gap-0.5"><XCircle className="size-2.5" />Cancelled</Badge>;
    77→    case 'filed': return <Badge className="bg-emerald-50 text-emerald-700 border-emerald-200 text-[10px] px-1.5 py-0 h-5 gap-0.5"><CheckCircle2 className="size-2.5" />Filed</Badge>;
    78→    default: return <Badge variant="secondary" className="text-[10px]">{status}</Badge>;
    79→  }
    80→}
    81→
    82→function getMatchBadge(status: string) {
    83→  switch (status) {
    84→    case 'perfect_match': return <Badge className="bg-emerald-50 text-emerald-700 border-emerald-200 text-[10px] px-1.5 py-0">Matched</Badge>;
    85→    case 'partial_match': return <Badge className="bg-amber-50 text-amber-700 border-amber-200 text-[10px] px-1.5 py-0">Partial</Badge>;
    86→    case 'mismatch': case 'missing_in_books': case 'missing_in_gstr': return <Badge className="bg-red-50 text-red-700 border-red-200 text-[10px] px-1.5 py-0">Mismatch</Badge>;
    87→    default: return <Badge variant="secondary" className="text-[10px]">{status}</Badge>;
    88→  }
    89→}
    90→
    91→function getRiskColor(risk: string) {
    92→  switch (risk) {
    93→    case 'critical': return 'text-red-600 bg-red-50';
    94→    case 'high': return 'text-orange-600 bg-orange-50';
    95→    case 'medium': return 'text-amber-600 bg-amber-50';
    96→    default: return 'text-slate-600 bg-slate-50';
    97→  }
    98→}

   113→function StepProgressBar({ currentStep }: { currentStep: number }) {
   114→  const percent = Math.round(((currentStep + 1) / PREP_STEPS.length) * 100);
   115→
   116→  return (
   117→    <div className="border border-border/60 rounded-xl p-4 md:p-6">
   118→      <div className="flex items-center justify-between mb-4">
   119→        <h3 className="text-sm font-semibold text-foreground uppercase tracking-wider">Preparation Progress</h3>
   120→        <span className="text-xs font-medium text-muted-foreground">{percent}% Complete</span>
   121→      </div>
   122→      <div className="h-2 rounded-full bg-slate-100 mb-6 overflow-hidden">
   123→        <motion.div className="h-full rounded-full bg-emerald-500" animate={{ width: `${percent}%` }} transition={{ duration: 0.6, ease: 'easeOut' }} />
   124→      </div>
   125→      <div className="flex items-start">
   126→        {PREP_STEPS.map((step, idx) => {
   127→          const isCompleted = idx < currentStep;
   128→          const isActive = idx === currentStep;
   129→          const Icon = step.icon;
   130→          return (
   131→            <React.Fragment key={step.id}>
   132→              <div className="flex flex-col items-center" style={{ minWidth: idx === 0 || idx === PREP_STEPS.length - 1 ? '60px' : '80px', flex: '1 1 0' }}>
   133→                <motion.div
   134→                  className={`flex items-center justify-center h-9 w-9 rounded-full shrink-0 text-sm font-semibold transition-all duration-500 ${
   135→                    isCompleted ? 'bg-emerald-500 text-white shadow-sm shadow-emerald-500/30'
   136→                      : isActive ? 'bg-emerald-50 text-emerald-700 border-2 border-emerald-500 shadow-sm'
   137→                        : 'bg-slate-100 text-slate-400'
   138→                  }`}
   139→                  animate={isActive ? { scale: [1, 1.08, 1] } : {}}
   140→                  transition={isActive ? { duration: 2, repeat: Infinity, ease: 'easeInOut' } : {}}
   141→                >
   142→                  {isCompleted ? <Check className="h-4 w-4" /> : <Icon className="h-4 w-4" />}
   143→                </motion.div>
   144→                <span className={`text-[10px] text-center leading-tight mt-2 whitespace-nowrap ${
   145→                  isCompleted ? 'text-emerald-700 font-medium' : isActive ? 'text-foreground font-semibold' : 'text-muted-foreground'
   146→                }`}>
   147→                  {step.label}
   148→                </span>
   149→              </div>
   150→              {idx < PREP_STEPS.length - 1 && (
   151→                <div className="flex items-center pt-[18px] flex-1 min-w-[12px]">
   152→                  <motion.div
   153→                    className={`h-[3px] w-full rounded-full transition-colors duration-500 ${isCompleted ? 'bg-emerald-400' : 'bg-slate-200'}`}
   154→                    animate={isActive ? { opacity: [0.5, 1, 0.5] } : {}}
   155→                    transition={isActive ? { duration: 2, repeat: Infinity, ease: 'easeInOut' } : {}}
   156→                  />
   157→                </div>
   158→              )}
   159→            </React.Fragment>
   160→          );
   161→        })}
   162→      </div>
   163→    </div>
   164→  );
   165→}

   171→export default function ReturnPrepWorkspace() {
   172→  const { returnPrepCtx, setCurrentView } = useApp();
   173→  const clientId = returnPrepCtx.clientId;
   174→  const returnType = returnPrepCtx.returnType;
   175→  const period = returnPrepCtx.period;
   176→

   180→  const { data: returns, loading: returnsLoading } = useFireReturns(clientId);
   181→  const { data: documents, loading: documentsLoading } = useFireDocuments(clientId);

   185→  const returnsList = returns ?? [];
   186→  const documentsList = documents ?? [];

   202→    returnsList.find(r => r.returnType === returnType && r.period === period),
   203→    [returnsList, returnType, period]

   207→    if (invoiceFilter === 'all') return invoicesList;

   218→  const draftCount = invoicesList.filter(i => i.status === 'draft').length;
   219→  const highRiskCount = invoicesList.filter(i => i.riskLevel === 'high' || i.riskLevel === 'critical').length;
   220→
   221→  const matchedCount = invoicesList.filter(i => i.matchStatus === 'perfect_match').length;
   222→  const partialCount = invoicesList.filter(i => i.matchStatus === 'partial_match').length;

   224→
   225→  const validationScore = invoicesList.length > 0 ? Math.min(100, Math.round((approvedCount / invoicesList.length) * 100)) : 0;

   227→  const complianceScore = client?.healthScore ?? 0;
   228→
   229→  const allChecksPass = validationScore >= 95 && highRiskCount === 0;
   230→  const isGSTR1 = returnType === 'GSTR-1';
   231→  const loading = clientLoading || invoicesLoading;
   232→
   233→  // ── Computed step based on data ──
   234→  const computedStep = useMemo(() => {

   236→    if (currentReturn?.status === 'generated') return 4;
   237→    if (currentReturn?.status === 'reviewed' || currentReturn?.status === 'validated') return 3;
   238→    if (documentsList.some(d => d.extractionStatus === 'completed')) return 1;
   239→    if (documentsList.length > 0) return 0;
   240→    return 0;
   241→  }, [currentReturn, documentsList]);
   242→
   243→  const effectiveStep = Math.max(currentStep, computedStep);
   244→
   245→  // ── Actions ──
   246→  const handleBack = () => setCurrentView('client-workspace');
   247→
   248→  const handleAdvanceStep = useCallback((step: number) => {
   249→    setCurrentStep(step);
   250→  }, []);

   256→      // Create return if not exists
   257→      let returnId = currentReturn?.id;

   260→          returnId: '',
   261→          clientId,
   262→          returnType,
   263→          period,
   264→          financialYear: period.split('-')[0],
   265→          status: 'prepared',
   266→          filedDate: null,
   267→          acknowledgmentNumber: null,
   268→          totalInvoices: invoicesList.length,
   269→          readyForFiling: approvedCount,
   270→          issuesFound: highRiskCount,
   271→          criticalErrors: 0,
   272→          warnings: 0,
   273→          totalTaxableValue: totalTaxable,

   275→          jsonPayload: null,
   276→          assignedTo: null,
   277→          reviewedBy: null,
   278→        });

   283→      toast.success('Validation Complete', { description: `${invoicesList.length} invoices processed` });
   284→    } catch (err) {
   285→      toast.error('Validation failed', { description: err instanceof Error ? err.message : 'Unknown error' });
   286→    } finally {
   287→      setActionLoading(false);
   288→    }
   289→  }, [clientId, currentReturn, returnType, period, invoicesList, approvedCount, highRiskCount, totalTaxable, totalTax, handleAdvanceStep]);

   293→      toast.warning('Complete validation first');
   294→      return;
   295→    }

   297→    toast.success('Reconciliation Complete', { description: 'Books vs GSTR-2B matching finished.' });
   298→  }, [effectiveStep, handleAdvanceStep]);

   300→  const handleMarkReady = useCallback(async () => {
   301→    if (!currentReturn?.id) {
   302→      toast.error('No return found to mark ready');
   303→      return;
   304→    }
   305→    if (highRiskCount > 0) {
   306→      toast.warning('Cannot mark ready', { description: `${highRiskCount} high-risk invoices must be resolved first` });
   307→      return;
   308→    }
   309→    setActionLoading(true);
   310→    try {
   311→      await updateReturnStatus(currentReturn.id, 'generated');
   312→      handleAdvanceStep(4);
   313→      toast.success('Return marked as Ready to File');
   314→    } catch (err) {
   315→      toast.error('Failed to mark ready', { description: err instanceof Error ? err.message : 'Unknown error' });
   316→    } finally {
   317→      setActionLoading(false);
   318→    }
   319→  }, [currentReturn, highRiskCount, handleAdvanceStep]);
   320→
   321→  const handleFileReturn = useCallback(() => {
   322→    if (!currentReturn?.id) {
   323→      toast.error('No return to file');
   324→      return;
   325→    }
   326→    setFilingModalOpen(true);
   327→    setFilingProgress('validating');
   328→
   329→    setTimeout(() => setFilingProgress('generating'), 1500);
   330→    setTimeout(() => setFilingProgress('submitting'), 3000);
   331→    setTimeout(async () => {
   332→      try {
   333→        const arn = await fileReturn(currentReturn.id);
   334→        setFilingProgress('success');
   335→        handleAdvanceStep(5);
   336→        toast.success(`${returnType} Filed Successfully!`, { description: `Period: ${periodToLabel(period)} · ARN: ${arn}` });
   337→      } catch (err) {
   338→        setFilingModalOpen(false);
   339→        setFilingProgress('idle');
   340→        toast.error('Filing failed', { description: err instanceof Error ? err.message : 'Unknown error' });
   341→      }
   342→    }, 4500);
   343→  }, [currentReturn, returnType, period, handleAdvanceStep]);
   344→
   345→  // ── Return Summary ──

   347→    const b2bSales = invoicesList.filter(i => i.invoiceType === 'B2B').reduce((s, i) => s + i.taxableValue, 0);

   351→  }, [invoicesList, totalTaxable]);

   356→    const netTaxPayable = Math.max(0, outputTax - itcAvailable);
   357→    return { taxableSupplies: totalTaxable, itcAvailable, outputTax, netTaxPayable };
   358→  }, [totalTaxable, totalTax]);
   359→
   360→  // ── Loading skeleton ──
   361→  if (loading && !client) {
   362→    return (
   363→      <div className="max-w-7xl mx-auto px-4 md:px-6 py-6 space-y-6">
   364→        <Skeleton className="h-6 w-40" />
   365→        <div className="border rounded-xl p-6 space-y-4">
   366→          <div className="flex gap-4"><Skeleton className="h-10 w-10 rounded-lg" /><div className="space-y-2 flex-1"><Skeleton className="h-6 w-64" /><Skeleton className="h-4 w-48" /></div></div>
   367→        </div>
   368→        <div className="flex gap-2">{Array.from({ length: 6 }).map((_, i) => <Skeleton key={i} className="h-10 flex-1 rounded-lg" />)}</div>
   369→      </div>
   370→    );
   371→  }
   372→
   373→  return (
   374→    <div className="max-w-7xl mx-auto px-4 md:px-6 py-6 space-y-8">
   375→
   376→      {/* ═══ INVOICE DETAIL DIALOG ═══ */}
   377→      <Dialog open={!!selectedInvoice} onOpenChange={(open) => { if (!open) setSelectedInvoice(null); }}>
   378→        <DialogContent className="sm:max-w-lg">
   379→          <DialogHeader>
   380→            <DialogTitle className="flex items-center gap-2">
   381→              <FileText className="h-4 w-4 text-emerald-600" /> Invoice Detail
   382→            </DialogTitle>
   383→          </DialogHeader>
   384→          {selectedInvoice && (
   385→            <div className="space-y-4 mt-2">
   386→              <div className="grid grid-cols-2 gap-3">
   387→                <div><span className="text-[10px] text-muted-foreground uppercase tracking-wider">Invoice Number</span><p className="text-sm font-mono font-semibold">{selectedInvoice.invoiceNumber}</p></div>
   388→                <div><span className="text-[10px] text-muted-foreground uppercase tracking-wider">Date</span><p className="text-sm">{selectedInvoice.invoiceDate}</p></div>
   389→                <div><span className="text-[10px] text-muted-foreground uppercase tracking-wider">Seller GSTIN</span><p className="text-sm font-mono">{selectedInvoice.sellerGstin}</p></div>
   390→                <div><span className="text-[10px] text-muted-foreground uppercase tracking-wider">Buyer</span><p className="text-sm">{selectedInvoice.buyerName || '—'}</p></div>
   391→                <div><span className="text-[10px] text-muted-foreground uppercase tracking-wider">HSN Code</span><p className="text-sm font-mono">{selectedInvoice.hsnCode || 'Missing'}</p></div>
   392→                <div><span className="text-[10px] text-muted-foreground uppercase tracking-wider">Place of Supply</span><p className="text-sm">{selectedInvoice.placeOfSupply || '—'}</p></div>
   393→              </div>
   394→              <Separator />
   395→              <div className="grid grid-cols-2 gap-3">
   396→                <div><span className="text-[10px] text-muted-foreground uppercase tracking-wider">Taxable Value</span><p className="text-sm font-bold">{formatCurrency(selectedInvoice.taxableValue)}</p></div>
   397→                <div><span className="text-[10px] text-muted-foreground uppercase tracking-wider">Status</span><div className="mt-0.5">{getStatusBadge(selectedInvoice.status)}</div></div>

   399→                <div><span className="text-[10px] text-muted-foreground uppercase tracking-wider">Risk Level</span><div className="mt-0.5"><span className={`inline-flex items-center text-[10px] font-medium px-1.5 py-0.5 rounded ${getRiskColor(selectedInvoice.riskLevel)}`}>{selectedInvoice.riskLevel}</span></div></div>
   400→                <div><span className="text-[10px] text-muted-foreground uppercase tracking-wider">CGST</span><p className="text-sm">{selectedInvoice.cgst > 0 ? formatCurrency(selectedInvoice.cgst) : '—'}</p></div>
   401→                <div><span className="text-[10px] text-muted-foreground uppercase tracking-wider">SGST</span><p className="text-sm">{selectedInvoice.sgst > 0 ? formatCurrency(selectedInvoice.sgst) : '—'}</p></div>
   402→                <div><span className="text-[10px] text-muted-foreground uppercase tracking-wider">IGST</span><p className="text-sm">{selectedInvoice.igst > 0 ? formatCurrency(selectedInvoice.igst) : '—'}</p></div>
   403→                <div><span className="text-[10px] text-muted-foreground uppercase tracking-wider">Total Amount</span><p className="text-sm font-bold text-emerald-700">{formatCurrency(selectedInvoice.totalAmount)}</p></div>
   404→              </div>
   405→              {selectedInvoice.aiExplanation && (
   406→                <>
   407→                  <Separator />
   408→                  <div className="p-3 rounded-lg bg-amber-50 border border-amber-200">
   409→                    <p className="text-xs font-semibold text-amber-700 mb-1">AI Insight</p>
   410→                    <p className="text-[11px] text-amber-800 leading-relaxed">{selectedInvoice.aiExplanation}</p>
   411→                  </div>
   412→                </>
   413→              )}
   414→            </div>
   415→          )}
   416→        </DialogContent>
   417→      </Dialog>
   418→
