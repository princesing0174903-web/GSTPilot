    27→function formatINR(amount: number): string {
    28→  const parts = amount.toFixed(0).split('.');
    29→  let intPart = parts[0];
    30→  const lastThree = intPart.slice(-3);
    31→  const otherNumbers = intPart.slice(0, -3);
    32→  if (otherNumbers !== '' && otherNumbers !== '-') {
    33→    intPart = otherNumbers.replace(/\B(?=(\d{2})+(?!\d))/g, ',') + ',' + lastThree;
    34→  } else {
    35→    intPart = lastThree;
    36→  }
    37→  return '₹' + intPart;
    38→}

    40→function formatDate(dateStr: string): string {
    41→  const d = new Date(dateStr);
    42→  const dd = String(d.getDate()).padStart(2, '0');
    43→  const mm = String(d.getMonth() + 1).padStart(2, '0');
    44→  const yyyy = d.getFullYear();
    45→  return `${dd}/${mm}/${yyyy}`;
    46→}

   117→// SCORE COMPUTATION
   118→// ═══════════════════════════════════════════════════════════════════════════════
   119→
   120→function computeScores(invoices: typeof demoInvoices) {
   121→  const total = invoices.length;
   122→  const unpaid = invoices.filter(i => i.status === 'unpaid').length;
   123→  const overdue = invoices.filter(i => i.status === 'overdue').length;
   124→  const totalAmount = invoices.reduce((s, i) => s + i.amount, 0);
   125→  const overdueAmount = invoices.filter(i => i.status === 'overdue').reduce((s, i) => s + i.amount, 0);
   126→  const avgAge = invoices.reduce((s, i) => s + i.age, 0) / total;
   127→  const eligibleCount = invoices.filter(i => i.eligible).length;
   128→  const eligibleAmount = invoices.filter(i => i.eligible).reduce((s, i) => s + i.amount, 0);
   129→
   130→  // Cash Flow Score: f(invoice aging, collection speed, bank balance trend)
   131→  const ageFactor = Math.max(0, 100 - avgAge * 0.8);
   132→  const collectionFactor = ((total - overdue) / total) * 100;
   133→  const cashFlow = Math.round(ageFactor * 0.4 + collectionFactor * 0.6);
   134→
   135→  // Credit Score: f(payment regularity, GST compliance, default history)
   136→  const paymentRegularity = Math.min(100, ((unpaid + 1) / (overdue + 1)) * 50);
   137→  const gstCompliance = 85; // assumed
   138→  const credit = Math.round(paymentRegularity * 0.5 + gstCompliance * 0.5);
   139→
   140→  // Collection Score: f(on-time collections %, overdue amount ratio, DSO)
   141→  const onTimePct = ((total - overdue) / total) * 100;
   142→  const overdueRatio = overdueAmount / totalAmount;
   143→  const dso = avgAge;
   144→  const dsoFactor = Math.max(0, 100 - (dso - 30) * 1.5);
   145→  const collection = Math.round(onTimePct * 0.4 + (1 - overdueRatio) * 100 * 0.35 + dsoFactor * 0.25);
   146→
   147→  // Business Health: weighted average
   148→  const health = Math.round(cashFlow * 0.3 + credit * 0.25 + collection * 0.25 + gstCompliance * 0.2);
   149→
   150→  return {
   151→    cashFlow: Math.min(100, Math.max(0, cashFlow)),
   152→    credit: Math.min(100, Math.max(0, credit)),
   153→    collection: Math.min(100, Math.max(0, collection)),
   154→    health: Math.min(100, Math.max(0, health)),
   155→    eligibleCount,
   156→    eligibleAmount,
   157→    totalAmount,
   158→    overdueAmount,
   159→  };
   160→}

   166→function ScoreGauge({ score, label, icon: Icon, size = 140 }: {
   167→  score: number; label: string; icon: React.ElementType; size?: number;
   168→}) {
   169→  const r = (size - 20) / 2;
   170→  const cx = size / 2;
   171→  const cy = size / 2;
   172→  const circumference = 2 * Math.PI * r;
   173→  const strokeDashoffset = circumference * (1 - score / 100);
   174→
   175→  const getColor = (s: number) => {
   176→    if (s >= 70) return { stroke: '#10b981', fill: '#ecfdf5', text: 'text-emerald-600', bg: 'bg-emerald-50' };
   177→    if (s >= 40) return { stroke: '#f59e0b', fill: '#fffbeb', text: 'text-amber-600', bg: 'bg-amber-50' };
   178→    return { stroke: '#ef4444', fill: '#fef2f2', text: 'text-red-600', bg: 'bg-red-50' };
   179→  };
   180→
   181→  const color = getColor(score);
   182→
   183→  return (
   184→    <div className="flex flex-col items-center gap-2">
   185→      <div className="relative" style={{ width: size, height: size }}>
   186→        <svg width={size} height={size} className="transform -rotate-90">
   187→          <circle cx={cx} cy={cy} r={r} fill="none" stroke="#e2e8f0" strokeWidth="8" />
   188→          <motion.circle
   189→            cx={cx} cy={cy} r={r} fill="none"
   190→            stroke={color.stroke}
   191→            strokeWidth="8"
   192→            strokeLinecap="round"
   193→            strokeDasharray={circumference}
   194→            initial={{ strokeDashoffset: circumference }}
   195→            animate={{ strokeDashoffset }}
   196→            transition={{ duration: 1.5, ease: 'easeOut', delay: 0.2 }}
   197→          />
   198→        </svg>
   199→        <div className="absolute inset-0 flex flex-col items-center justify-center">
   200→          <motion.span
   201→            className="text-2xl font-bold"
   202→            initial={{ opacity: 0, scale: 0.5 }}
   203→            animate={{ opacity: 1, scale: 1 }}
   204→            transition={{ duration: 0.5, delay: 0.8 }}
   205→          >
   206→            <span className={color.text}>{score}</span>
   207→            <span className="text-xs text-slate-400 font-normal">/100</span>
   208→          </motion.span>
   209→        </div>
   210→        <div className={`absolute top-1 left-1/2 -translate-x-1/2 rounded-full p-1 ${color.bg}`}>
   211→          <Icon className={`h-3.5 w-3.5 ${color.text}`} />
   212→        </div>
   213→      </div>
   214→      <span className="text-xs font-semibold text-slate-600 text-center">{label}</span>
   215→    </div>
   216→  );
   217→}

   223→function Sparkline({ data, color = '#10b981', width = 120, height = 32 }: {
   224→  data: number[]; color?: string; width?: number; height?: number;
   225→}) {
   226→  const max = Math.max(...data);
   227→  const min = Math.min(...data);
   228→  const range = max - min || 1;
   229→  const padding = 2;
   230→
   231→  const points = data.map((v, i) => {
   232→    const x = padding + (i / (data.length - 1)) * (width - padding * 2);
   233→    const y = padding + (1 - (v - min) / range) * (height - padding * 2);
   234→    return `${x},${y}`;
   235→  }).join(' ');
   236→
   237→  const areaPoints = `${padding},${height - padding} ${points} ${width - padding},${height - padding}`;
   238→
   239→  return (
   240→    <svg width={width} height={height} className="overflow-visible">
   241→      <motion.polygon
   242→        points={areaPoints}
   243→        fill={color}
   244→        fillOpacity={0.1}
   245→        initial={{ opacity: 0 }}
   246→        animate={{ opacity: 1 }}
   247→        transition={{ duration: 0.8, delay: 0.5 }}
   248→      />
   249→      <motion.polyline
   250→        points={points}
   251→        fill="none"
   252→        stroke={color}
   253→        strokeWidth="2"
   254→        strokeLinecap="round"
   255→        strokeLinejoin="round"
   256→        initial={{ pathLength: 0 }}
   257→        animate={{ pathLength: 1 }}
   258→        transition={{ duration: 1.2, delay: 0.3 }}
   259→      />
   260→      {data.length > 0 && (
   261→        <motion.circle
   262→          cx={padding + ((data.length - 1) / (data.length - 1)) * (width - padding * 2)}
   263→          cy={padding + (1 - (data[data.length - 1] - min) / range) * (height - padding * 2)}
   264→          r="3"
   265→          fill={color}
   266→          initial={{ scale: 0 }}
   267→          animate={{ scale: 1 }}
   268→          transition={{ duration: 0.3, delay: 1.5 }}
   269→        />
   270→      )}
   271→    </svg>
   272→  );
   273→}

   279→function RevenueAreaChart({ data }: { data: typeof monthlyRevenue }) {
   280→  const width = 700;
   281→  const height = 180;
   282→  const padding = { top: 10, right: 10, bottom: 30, left: 70 };
   283→  const chartW = width - padding.left - padding.right;
   284→  const chartH = height - padding.top - padding.bottom;
   285→
   286→  const maxVal = Math.max(...data.map(d => d.value));
   287→  const minVal = Math.min(...data.map(d => d.value)) * 0.85;
   288→
   289→  const points = data.map((d, i) => ({
   290→    x: padding.left + (i / (data.length - 1)) * chartW,
   291→    y: padding.top + (1 - (d.value - minVal) / (maxVal - minVal)) * chartH,
   292→  }));
   293→
   294→  const linePath = points.map((p, i) => `${i === 0 ? 'M' : 'L'}${p.x},${p.y}`).join(' ');
   295→  const areaPath = `${linePath} L${points[points.length - 1].x},${padding.top + chartH} L${points[0].x},${padding.top + chartH} Z`;
   296→
   297→  return (
   298→    <svg viewBox={`0 0 ${width} ${height}`} className="w-full" preserveAspectRatio="xMidYMid meet">
   299→      <defs>
   300→        <linearGradient id="revenueGrad" x1="0" y1="0" x2="0" y2="1">
   301→          <stop offset="0%" stopColor="#10b981" stopOpacity={0.3} />
   302→          <stop offset="100%" stopColor="#10b981" stopOpacity={0.02} />
   303→        </linearGradient>
   304→      </defs>
   305→      {/* Grid lines */}
   306→      {[0, 0.25, 0.5, 0.75, 1].map((pct) => {
   307→        const y = padding.top + pct * chartH;
   308→        const val = maxVal - pct * (maxVal - minVal);
   309→        return (
   310→          <g key={pct}>
   311→            <line x1={padding.left} y1={y} x2={width - padding.right} y2={y} stroke="#e2e8f0" strokeDasharray="3,3" />
   312→            <text x={padding.left - 8} y={y + 4} textAnchor="end" className="text-[9px] fill-slate-400">{formatINR(val)}</text>
   313→          </g>
   314→        );
   315→      })}
   316→      {/* X labels */}
   317→      {data.map((d, i) => {
   318→        const x = padding.left + (i / (data.length - 1)) * chartW;
   319→        return (
   320→          <text key={i} x={x} y={height - 5} textAnchor="middle" className="text-[8px] fill-slate-400">
   321→            {d.month.split(' ')[0]}
   322→          </text>
   323→        );
   324→      })}
   325→      <motion.path
   326→        d={areaPath}
   327→        fill="url(#revenueGrad)"
   328→        initial={{ opacity: 0 }}
   329→        animate={{ opacity: 1 }}
   330→        transition={{ duration: 1, delay: 0.3 }}
   331→      />
   332→      <motion.path
   333→        d={linePath}
   334→        fill="none"
   335→        stroke="#10b981"
   336→        strokeWidth="2.5"
   337→        strokeLinecap="round"
   338→        strokeLinejoin="round"
   339→        initial={{ pathLength: 0 }}
   340→        animate={{ pathLength: 1 }}
   341→        transition={{ duration: 1.5, delay: 0.2 }}
   342→      />
   343→      {points.map((p, i) => (
   344→        <motion.circle
   345→          key={i}
   346→          cx={p.x} cy={p.y} r="3.5"
   347→          fill="white" stroke="#10b981" strokeWidth="2"
   348→          initial={{ scale: 0 }}
   349→          animate={{ scale: 1 }}
   350→          transition={{ duration: 0.3, delay: 0.5 + i * 0.08 }}
   351→        />
   352→      ))}
   353→    </svg>
   354→  );
   355→}

   361→function EMIChart({ principal, rate, tenure }: { principal: number; rate: number; tenure: number }) {
   362→  const monthlyRate = rate / 12 / 100;
   363→  const emi = monthlyRate > 0
   364→    ? principal * monthlyRate * Math.pow(1 + monthlyRate, tenure) / (Math.pow(1 + monthlyRate, tenure) - 1)
   365→    : principal / tenure;
   366→  const totalPayment = emi * tenure;
   367→  const totalInterest = totalPayment - principal;
   368→
   369→  const size = 160;
   370→  const cx = size / 2;
   371→  const cy = size / 2;
   372→  const r = 55;
   373→  const circumference = 2 * Math.PI * r;
   374→  const principalPct = principal / totalPayment;
   375→  const principalDash = circumference * principalPct;
   376→
   377→  return (
   378→    <div className="flex items-center gap-4">
   379→      <svg width={size} height={size}>
   380→        <circle cx={cx} cy={cy} r={r} fill="none" stroke="#e2e8f0" strokeWidth="16" />
   381→        <motion.circle
   382→          cx={cx} cy={cy} r={r} fill="none"
   383→          stroke="#10b981" strokeWidth="16"
   384→          strokeDasharray={`${principalDash} ${circumference}`}
   385→          strokeLinecap="round"
   386→          transform={`rotate(-90 ${cx} ${cy})`}
   387→          initial={{ strokeDasharray: `0 ${circumference}` }}
   388→          animate={{ strokeDasharray: `${principalDash} ${circumference}` }}
   389→          transition={{ duration: 1, delay: 0.3 }}
   390→        />
   391→        <motion.circle
   392→          cx={cx} cy={cy} r={r} fill="none"
   393→          stroke="#f59e0b" strokeWidth="16"
   394→          strokeDasharray={`${circumference - principalDash} ${circumference}`}
   395→          strokeDashoffset={-principalDash}
   396→          strokeLinecap="round"
   397→          transform={`rotate(-90 ${cx} ${cy})`}
   398→          initial={{ opacity: 0 }}
   399→          animate={{ opacity: 1 }}
   400→          transition={{ duration: 0.5, delay: 1 }}
   401→        />
   402→        <text x={cx} y={cy - 6} textAnchor="middle" className="text-xs font-bold fill-slate-700">{formatINR(Math.round(emi))}</text>
   403→        <text x={cx} y={cy + 10} textAnchor="middle" className="text-[9px] fill-slate-400">/month</text>
   404→      </svg>
   405→      <div className="flex flex-col gap-2 text-xs">
   406→        <div className="flex items-center gap-2">
   407→          <div className="h-3 w-3 rounded-sm bg-emerald-500" />
   408→          <span className="text-slate-600">Principal: {formatINR(principal)}</span>
   409→        </div>
   410→        <div className="flex items-center gap-2">
   411→          <div className="h-3 w-3 rounded-sm bg-amber-500" />
   412→          <span className="text-slate-600">Interest: {formatINR(Math.round(totalInterest))}</span>
   413→        </div>

   849→  // GST filing regularity = filedReturns / (filedReturns + pendingReturns + overdueReturns)
   850→  const { data: invoicesResp, isLoading: invoicesLoading } = useQuery<{ invoices: Array<{ taxableValue: number; invoiceDate: string }> }>({
   851→    queryKey: ['invoices', 'wc-loans'],

   859→  const annualInvoiceVolume = useMemo(() => {
   860→    const invoices = invoicesResp?.invoices ?? [];
   861→    if (invoices.length === 0) return 0;
   862→    const oneYearAgo = Date.now() - 365 * 24 * 60 * 60 * 1000;
   863→    return invoices
   864→      .filter((inv) => {
   865→        if (!inv.invoiceDate) return true;
   866→        const d = new Date(inv.invoiceDate).getTime();
   867→        return Number.isNaN(d) || d >= oneYearAgo;
   868→      })
   869→      .reduce((sum, inv) => sum + (Number(inv.taxableValue) || 0), 0);
   870→  }, [invoicesResp]);
   871→
   872→  const filingRegularity = useMemo(() => {
   873→    const filed = dashboard?.filedReturns ?? 0;
   874→    const pending = dashboard?.pendingReturns ?? 0;
