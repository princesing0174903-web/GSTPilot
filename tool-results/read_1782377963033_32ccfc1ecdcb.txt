   104→export function OracleWorkspace({
   105→  open, onClose, onNavigate, userName, firmName, gstin, userId, initialPrompt,
   106→}: OracleWorkspaceProps) {
   107→  const [store, setStore] = useState<ConversationStore>({ conversations: [], activeId: null });
   108→  // Portal guard: only render into document.body after mount to avoid SSR
   109→  // hydration mismatch (server has no document.body).
   110→  const [mounted, setMounted] = useState(false);
   111→  useEffect(() => setMounted(true), []);
   112→  const [messages, setMessages] = useState<OracleMessage[]>([]);
   113→  const [activeId, setActiveId] = useState<string | null>(null);
   114→  const [input, setInput] = useState('');
   115→  const [isStreaming, setIsStreaming] = useState(false);
   116→  const [lastEmotion, setLastEmotion] = useState<OracleMessage['emotion']>('helpful');
   117→  const [activeLanguage, setActiveLanguage] = useState<OracleMessage['language']>('english');
   118→  // Mobile: history drawer open state
   119→  const [historyOpen, setHistoryOpen] = useState(false);
   120→
   121→  const scrollRef = useRef<HTMLDivElement>(null);
   122→  const inputRef = useRef<HTMLTextAreaElement>(null);
   123→  const abortRef = useRef<AbortController | null>(null);
   124→  const userPinnedUpRef = useRef(false);
   125→  const streamingIdRef = useRef<string | null>(null);
   126→  // Tracks the last initialPrompt we auto-sent, so we never fire it twice.
   127→  const initialPromptSentRef = useRef<string | null>(null);
   128→
   129→  // ─── Load + migrate store on open ──────────────────────────────────────────
   130→  useEffect(() => {
   131→    if (!open) return;
   132→    const loaded = loadStore();
   133→    if (loaded.conversations.length === 0) {
   134→      // Migrate legacy single-conversation store if present.
   135→      try {
   136→        const raw = localStorage.getItem(LEGACY_KEY);
   137→        if (raw) {
   138→          const parsed = JSON.parse(raw) as OracleMessage[];
   139→          if (Array.isArray(parsed) && parsed.length > 0) {
   140→            const now = new Date().toISOString();
   141→            const conv: OracleConversation = {
   142→              id: cryptoId(),
   143→              title: deriveTitle(parsed),
   144→              messages: parsed.map((m) => (m.streaming ? { ...m, streaming: false } : m)),
   145→              createdAt: parsed[0]?.createdAt ?? now,
   146→              updatedAt: now,
   147→            };
   148→            const next: ConversationStore = { conversations: [conv], activeId: conv.id };
   149→            setStore(next);
   150→            setActiveId(conv.id);
   151→            setMessages(conv.messages);
   152→            saveStore(next);
   153→            return;
   154→          }
   155→        }
   156→      } catch {
   157→        /* ignore */
   158→      }
   159→      // Nothing to migrate — start fresh with one empty conversation.
   160→      const conv = newConversation();
   161→      const next: ConversationStore = { conversations: [conv], activeId: conv.id };
   162→      setStore(next);
   163→      setActiveId(conv.id);
   164→      setMessages([]);
   165→      saveStore(next);
   166→      return;
   167→    }
   168→    setStore(loaded);
   169→    const active = loaded.conversations.find((c) => c.id === loaded.activeId) ?? loaded.conversations[0];
   170→    if (active) {
   171→      setActiveId(active.id);
   172→      setMessages(active.messages);
   173→    }
   174→  }, [open]);
   175→
   176→  // ─── Persist active conversation whenever messages change ───────────────────
   177→  useEffect(() => {
   178→    if (!open || !activeId) return;
   179→    setStore((prev) => {
   180→      if (!prev.conversations.length) return prev;
   181→      const exists = prev.conversations.some((c) => c.id === activeId);
   182→      let conversations: OracleConversation[];
   183→      let nextActiveId = prev.activeId;
   184→      if (exists) {
   185→        conversations = prev.conversations.map((c) =>
   186→          c.id === activeId
   187→            ? {
   188→                ...c,
   189→                messages,
   190→                title: deriveTitle(messages) || c.title,
   191→                updatedAt: new Date().toISOString(),
   192→              }
   193→            : c,
   194→        );
   195→      } else {
   196→        // Active conversation doesn't exist yet — create it on the fly.
   197→        const conv: OracleConversation = {
   198→          id: activeId,
   199→          title: deriveTitle(messages) || 'New chat',
   200→          messages,
   201→          createdAt: new Date().toISOString(),
   202→          updatedAt: new Date().toISOString(),
   203→        };
   204→        conversations = [conv, ...prev.conversations];
   205→        nextActiveId = activeId;
   206→      }
   207→      const next = { conversations, activeId: nextActiveId };
   208→      saveStore(next);
   209→      return next;
   210→    });
   211→  }, [messages, open, activeId]);
   212→
   213→  // ─── Body scroll lock while open ───────────────────────────────────────────
   214→  useEffect(() => {
   215→    if (!open) return;
   216→    const prev = document.body.style.overflow;
   217→    document.body.style.overflow = 'hidden';
   218→    return () => {
   219→      document.body.style.overflow = prev;
   220→    };
   221→  }, [open]);
   222→
   223→  // ─── Escape to close ───────────────────────────────────────────────────────
   224→  useEffect(() => {
   225→    if (!open) return;
   226→    const onKey = (e: globalThis.KeyboardEvent) => {
   227→      if (e.key === 'Escape' && !isStreaming) onClose();
   228→    };
   229→    window.addEventListener('keydown', onKey);
   230→    return () => window.removeEventListener('keydown', onKey);
   231→  }, [open, isStreaming, onClose]);
   232→
   233→  // ─── Smart auto-scroll ─────────────────────────────────────────────────────
   234→  const handleScroll = useCallback(() => {
   235→    const el = scrollRef.current;
   236→    if (!el) return;
   237→    const distanceFromBottom = el.scrollHeight - el.scrollTop - el.clientHeight;
   238→    userPinnedUpRef.current = distanceFromBottom > 120;
   239→  }, []);
   240→
   241→  const scrollToBottom = useCallback((smooth = false) => {
   242→    const el = scrollRef.current;
   243→    if (!el) return;
   244→    el.scrollTo({ top: el.scrollHeight, behavior: smooth ? 'smooth' : 'auto' });
   245→  }, []);
   246→
   247→  useEffect(() => {
   248→    if (!userPinnedUpRef.current) {
   249→      requestAnimationFrame(() => scrollToBottom(false));
   250→    }
   251→  }, [messages, scrollToBottom]);
   252→
   253→  // ─── Conversation actions ──────────────────────────────────────────────────
   254→  const handleNewChat = useCallback(() => {
   255→    if (isStreaming) return;
   256→    const conv = newConversation();
   257→    setStore((prev) => {
   258→      const next = { conversations: [conv, ...prev.conversations], activeId: conv.id };
   259→      saveStore(next);
   260→      return next;
   261→    });
   262→    setActiveId(conv.id);
   263→    setMessages([]);
   264→    setInput('');
   265→    userPinnedUpRef.current = false;
   266→    setHistoryOpen(false);
   267→    requestAnimationFrame(() => inputRef.current?.focus());
   268→  }, [isStreaming]);
   269→
   270→  const handleSwitchConversation = useCallback(
   271→    (id: string) => {
   272→      if (isStreaming) return;
   273→      const conv = store.conversations.find((c) => c.id === id);
   274→      if (!conv) return;
   275→      setActiveId(id);
   276→      setMessages(conv.messages);
   277→      setStore((prev) => {
   278→        const next = { ...prev, activeId: id };
   279→        saveStore(next);
   280→        return next;
   281→      });
   282→      setHistoryOpen(false);
   283→      userPinnedUpRef.current = false;
   284→      requestAnimationFrame(() => scrollToBottom(false));
   285→    },
   286→    [isStreaming, store.conversations, scrollToBottom],
   287→  );
   288→
   289→  const handleDeleteConversation = useCallback(
   290→    (id: string) => {
   291→      if (isStreaming) return;
   292→      setStore((prev) => {
   293→        const filtered = prev.conversations.filter((c) => c.id !== id);
   294→        let nextActive = prev.activeId;
   295→        let nextMessages = messages;
   296→        if (prev.activeId === id) {
   297→          if (filtered.length > 0) {
   298→            nextActive = filtered[0].id;
   299→            nextMessages = filtered[0].messages;
   300→          } else {
   301→            const conv = newConversation();
   302→            filtered.unshift(conv);
   303→            nextActive = conv.id;
   304→            nextMessages = [];
   305→          }
   306→          setActiveId(nextActive);
   307→          setMessages(nextMessages);
   308→        }
   309→        const next = { conversations: filtered, activeId: nextActive };
   310→        saveStore(next);
   311→        return next;
   312→      });
   313→    },
   314→    [isStreaming, messages],
   315→  );
   316→
   317→  // ─── Send flow (preserved from previous implementation) ─────────────────────
   318→  const sendMessage = useCallback(
   319→    async (raw: string) => {
   320→      const text = raw.trim();
   321→      if (!text || isStreaming) return;
   322→
   323→      // ── Brand-question short-circuit ───────────────────────────────────────
   324→      const brand = detectBrandQuestion(text);
   325→      if (brand.matched && brand.answer) {
   326→        const userMsg: OracleMessage = {
   327→          id: cryptoId(),
   328→          role: 'user',
   329→          content: text,
   330→          language: detectLanguage(text),
   331→          createdAt: new Date().toISOString(),
   332→        };
   333→        const oracleMsg: OracleMessage = {
   334→          id: cryptoId(),
   335→          role: 'oracle',
   336→          content: brand.answer,
   337→          language: detectLanguage(text),
   338→          emotion: 'success',
   339→          createdAt: new Date().toISOString(),
   340→          streaming: false,
   341→          followUps: ['Who founded GSTPilot?', 'What can Oracle do?', 'GST kya hota hai?'],
   342→        };
   343→        setMessages((prev) => [...prev, userMsg, oracleMsg]);
   344→        setLastEmotion('success');
   345→        setActiveLanguage(detectLanguage(text));
   346→        setInput('');
   347→        if (inputRef.current) inputRef.current.style.height = 'auto';
   348→        userPinnedUpRef.current = false;
   349→        requestAnimationFrame(() => scrollToBottom(true));
   350→        return;
   351→      }
   352→
   353→      const userMsg: OracleMessage = {
   354→        id: cryptoId(),
   355→        role: 'user',
   356→        content: text,
   357→        language: detectLanguage(text),
   358→        createdAt: new Date().toISOString(),
   359→      };
   360→      const oracleId = cryptoId();
   361→      const oraclePlaceholder: OracleMessage = {
   362→        id: oracleId,
   363→        role: 'oracle',
   364→        content: '',
   365→        language: userMsg.language,
   366→        createdAt: new Date().toISOString(),
   367→        streaming: true,
   368→      };
   369→      setMessages((prev) => [...prev, userMsg, oraclePlaceholder]);
   370→      streamingIdRef.current = oracleId;
   371→      setIsStreaming(true);
   372→      setActiveLanguage(userMsg.language);
   373→      setInput('');
   374→      userPinnedUpRef.current = false;
   375→      if (inputRef.current) inputRef.current.style.height = 'auto';
   376→
   377→      const history: OracleChatRequest['messages'] = [
   378→        ...messages
   379→          .filter((m) => m.content.trim().length > 0)
   380→          .map((m) => ({ role: m.role, content: m.content })),
   381→        { role: 'user', content: text },
   382→      ];
   383→
   384→      const payload: OracleChatRequest = {
   385→        messages: history,
   386→        memory: {
   387→          userName,
   388→          firmName,
   389→          gstin,
   390→          userId,
   391→          preferredLanguage: activeLanguage,
   392→          recentTopics: messages
   393→            .filter((m) => m.role === 'user')
   394→            .slice(-4)
   395→            .map((m) => m.content.slice(0, 60)),
   396→        },
   397→      };
   398→
   399→      const controller = new AbortController();
   400→      abortRef.current = controller;
   401→
   402→      try {
   403→        const res = await fetch('/api/oracle/chat', {
   404→          method: 'POST',
   405→          headers: { 'Content-Type': 'application/json' },
   406→          body: JSON.stringify(payload),
   407→          signal: controller.signal,
   408→        });
   409→        if (!res.ok || !res.body) {
   410→          throw new Error(`Request failed (${res.status})`);
   411→        }
   412→
   413→        const reader = res.body.getReader();
   414→        const decoder = new TextDecoder();
   415→        let buffer = '';
   416→        let acc = '';
   417→
   418→        while (true) {
   419→          const { done, value } = await reader.read();
   420→          if (done) break;
   421→          buffer += decoder.decode(value, { stream: true });
   422→
   423→          const lines = buffer.split('\n');
   424→          buffer = lines.pop() ?? '';
   425→
   426→          for (const rawLine of lines) {
   427→            const line = rawLine.trim();
   428→            if (!line || !line.startsWith('data:')) continue;
   429→            const data = line.slice(5).trim();
   430→            if (!data || data === '[DONE]') continue;
   431→            let chunk: OracleStreamChunk;
   432→            try {
   433→              chunk = JSON.parse(data) as OracleStreamChunk;
   434→            } catch {
   435→              continue;
   436→            }
   437→            if (chunk.language) setActiveLanguage(chunk.language);
   438→            if (chunk.token) {
   439→              acc += chunk.token;
   440→              setMessages((prev) =>
   441→                prev.map((m) => (m.id === oracleId ? { ...m, content: acc } : m)),
   442→              );
   443→            }
   444→            if (chunk.done) {
   445→              setMessages((prev) =>
   446→                prev.map((m) =>
   447→                  m.id === oracleId
   448→                    ? {
   449→                        ...m,
   450→                        content: acc,
   451→                        streaming: false,
   452→                        emotion: detectEmotion(acc),
   453→                        followUps: buildFollowUps(acc),
   454→                      }
   455→                    : m,
   456→                ),
   457→              );
   458→              setLastEmotion(detectEmotion(acc));
   459→            }
   460→            if (chunk.error) {
   461→              setMessages((prev) =>
   462→                prev.map((m) =>
   463→                  m.id === oracleId
   464→                    ? {
   465→                        ...m,
   466→                        content:
   467→                          acc ||
   468→                          "I ran into a temporary issue reaching my reasoning service. Please try that again — your conversation is safe.",
   469→                        streaming: false,
   470→                        emotion: 'warning',
   471→                      }
   472→                    : m,
   473→                ),
   474→              );
   475→              setLastEmotion('warning');
   476→            }
   477→          }
   478→        }
   479→
   480→        setMessages((prev) =>
   481→          prev.map((m) =>
   482→            m.id === oracleId && m.streaming
   483→              ? {
   484→                  ...m,
   485→                  streaming: false,
   486→                  emotion: detectEmotion(m.content),
   487→                  followUps: buildFollowUps(m.content),
   488→                }
   489→              : m,
   490→          ),
   491→        );
   492→      } catch (err) {
   493→        const aborted = err instanceof DOMException && err.name === 'AbortError';
   494→        setMessages((prev) =>
   495→          prev.map((m) =>
   496→            m.id === oracleId
   497→              ? {
   498→                  ...m,
   499→                  streaming: false,
   500→                  content:
   501→                    m.content ||
   502→                    (aborted
   503→                      ? 'Stopped.'
   504→                      : "I had trouble reaching my reasoning service. Please try again in a moment."),
   505→                  emotion: aborted ? m.emotion : 'warning',
   506→                }
   507→              : m,
   508→          ),
   509→        );
   510→        if (!aborted) setLastEmotion('warning');
   511→      } finally {
   512→        setIsStreaming(false);
   513→        streamingIdRef.current = null;
   514→        abortRef.current = null;
   515→        requestAnimationFrame(() => inputRef.current?.focus());
   516→      }
   517→    },
   518→    [isStreaming, messages, userName, firmName, gstin, userId, activeLanguage, scrollToBottom],
   519→  );
   520→
   521→  // ─── Stop streaming ────────────────────────────────────────────────────────
   522→  const handleStop = useCallback(() => {
   523→    abortRef.current?.abort();
   524→  }, []);
   525→
   526→  // ─── Input handling ────────────────────────────────────────────────────────
   527→  const handleInputChange = useCallback((e: React.ChangeEvent<HTMLTextAreaElement>) => {
   528→    setInput(e.target.value);
   529→    const el = e.target;
   530→    el.style.height = 'auto';
   531→    el.style.height = `${Math.min(el.scrollHeight, 160)}px`;
   532→  }, []);
   533→
   534→  const handleKeyDown = useCallback(
   535→    (e: KeyboardEvent<HTMLTextAreaElement>) => {
   536→      if (e.key === 'Enter' && !e.shiftKey) {
   537→        e.preventDefault();
   538→        sendMessage(input);
   539→      }
   540→    },
   541→    [input, sendMessage],
   542→  );
   543→
   544→  // ─── Focus input on open ───────────────────────────────────────────────────
   545→  useEffect(() => {
   546→    if (open) {
   547→      requestAnimationFrame(() => inputRef.current?.focus());
   548→    }
   549→  }, [open]);
   550→
   551→  // ─── Auto-send a prefilled prompt (from the CommandBar / other launchers) ───
   552→  // Waits one tick so the conversation-load effect (also keyed on `open`) has
   553→  // settled messages/activeId before we send. Guarded by a ref so a given
   554→  // prompt is never sent twice.
   555→  useEffect(() => {
   556→    if (!open || !initialPrompt) return;
   557→    if (initialPromptSentRef.current === initialPrompt) return;
   558→    const t = setTimeout(() => {
   559→      initialPromptSentRef.current = initialPrompt;
   560→      sendMessage(initialPrompt);
   561→    }, 60);
   562→    return () => clearTimeout(t);
   563→  }, [open, initialPrompt, sendMessage]);
   564→
   565→  // ─── Rail navigation ───────────────────────────────────────────────────────
   566→  const handleRailClick = useCallback(
   567→    (item: RailItem) => {
   568→      if (item.id === 'oracle') return; // already here
   569→      if (isStreaming) return;
   570→      onNavigate(item.id);
   571→      onClose();
   572→    },
   573→    [isStreaming, onNavigate, onClose],
   574→  );
   575→
   576→  // ─── Derived state ─────────────────────────────────────────────────────────
   577→  const avatarState = useMemo(
   578→    () => deriveAvatarState({ isStreaming, hasInput: input.trim().length > 0, lastEmotion }),
   579→    [isStreaming, input, lastEmotion],
   580→  );
   581→
   582→  const isEmpty = messages.length === 0;
   583→  const grouped = useMemo(() => groupConversations(store.conversations), [store.conversations]);
   584→

  1224→interface ConversationGroups {
  1225→  today: OracleConversation[];
  1226→  yesterday: OracleConversation[];
  1227→  prev7: OracleConversation[];
  1228→  older: OracleConversation[];

  1231→function groupConversations(convos: OracleConversation[]): ConversationGroups {
  1232→  const now = new Date();
  1233→  const startOfToday = new Date(now.getFullYear(), now.getMonth(), now.getDate()).getTime();
  1234→  const startOfYesterday = startOfToday - 86_400_000;
  1235→  const startOf7Days = startOfToday - 7 * 86_400_000;
  1236→
  1237→  const today: OracleConversation[] = [];
  1238→  const yesterday: OracleConversation[] = [];
  1239→  const prev7: OracleConversation[] = [];
  1240→  const older: OracleConversation[] = [];
  1241→
  1242→  const sorted = [...convos].sort(
  1243→    (a, b) => new Date(b.updatedAt).getTime() - new Date(a.updatedAt).getTime(),
  1244→  );
  1245→
  1246→  for (const c of sorted) {
  1247→    const t = new Date(c.updatedAt).getTime();
  1248→    if (isNaN(t)) {
  1249→      older.push(c);
  1250→    } else if (t >= startOfToday) {
  1251→      today.push(c);
  1252→    } else if (t >= startOfYesterday) {
  1253→      yesterday.push(c);
  1254→    } else if (t >= startOf7Days) {
  1255→      prev7.push(c);
  1256→    } else {
  1257→      older.push(c);
  1258→    }
  1259→  }
  1260→
  1261→  return { today, yesterday, prev7, older };
  1262→}

  1264→/** Generate contextual follow-up chips from the response content. */
  1265→function buildFollowUps(content: string): string[] | undefined {
  1266→  if (!content || content.length < 30) return undefined;
  1267→  const lower = content.toLowerCase();
  1268→
  1269→  const pool: string[] = [];
  1270→  if (/(gstr-3b|3b)/.test(lower)) pool.push('GSTR-3B की last date क्या है?', 'Late fee कितनी लगेगी?');
  1271→  if (/(gstr-1|gstr 1)/.test(lower)) pool.push('GSTR-1 कैसे file करें?', 'B2B और B2C में अंतर?');
  1272→  if (/(itc|input tax credit)/.test(lower)) pool.push('ITC claim कैसे करें?', 'Blocked ITC के rules?');
  1273→  if (/(late fee|penalty|overdue)/.test(lower)) pool.push('Late fee waiver मिल सकती है?', 'How to avoid this next time?');
  1274→  if (/(cash flow|collections)/.test(lower)) pool.push('Receivables कैसे recover करें?', 'Working capital optimize करें');
  1275→  if (/(reverse charge|rcm)/.test(lower)) pool.push('RCM किन पर लागू है?', 'How to report RCM in GSTR-3B?');
  1276→  if (/(refund)/.test(lower)) pool.push('Refund process क्या है?', 'Refund timeline कितनी है?');
  1277→
  1278→  pool.push('GST kya hota hai?', 'Explain ITC rules', 'How can Oracle help me daily?');
  1279→
  1280→  const seen = new Set<string>();
  1281→  const picks: string[] = [];
  1282→  for (const p of pool) {
  1283→    if (seen.has(p)) continue;

  1285→    picks.push(p);

  1289→}
