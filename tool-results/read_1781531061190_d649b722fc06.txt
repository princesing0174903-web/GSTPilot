    88→    88→function NotificationBell() {
    89→    89→  const { data: unreadNotifs } = useFireUnreadNotifications()
    90→    90→  const { data: allNotifs } = useFireNotifications()

    92→    92→  const [open, setOpen] = useState(false)
    93→    93→
    94→    94→  const handleMarkRead = async (notifId: string) => {
    95→    95→    try { await markNotificationRead(notifId) } catch {}
    96→    96→  }
    97→    97→
    98→    98→  const handleMarkAllRead = async () => {
    99→    99→    try {
   100→   100→      for (const n of unreadNotifs) {
   101→   101→        await markNotificationRead(n.notifId || n.id)
   102→   102→      }
   103→   103→    } catch {}
   104→   104→  }
   105→   105→
   106→   106→  const priorityColor = (p: string) => {
   107→   107→    switch (p) {
   108→   108→      case 'urgent': return 'text-red-600 bg-red-50 border-red-200'
   109→   109→      case 'high': return 'text-orange-600 bg-orange-50 border-orange-200'
   110→   110→      case 'normal': return 'text-blue-600 bg-blue-50 border-blue-200'
   111→   111→      default: return 'text-slate-600 bg-slate-50 border-slate-200'
   112→   112→    }
   113→   113→  }
   114→   114→
   115→   115→  const priorityBadge = (p: string) => {
   116→   116→    switch (p) {
   117→   117→      case 'urgent': return 'bg-red-500'
   118→   118→      case 'high': return 'bg-orange-500'
   119→   119→      case 'normal': return 'bg-blue-500'
   120→   120→      default: return 'bg-slate-400'
   121→   121→    }
   122→   122→  }
   123→   123→
   124→   124→  const typeIcon = (t: string) => {
   125→   125→    switch (t) {
   126→   126→      case 'filing_due': case 'filing_overdue': return <Clock className="h-3.5 w-3.5" />
   127→   127→      case 'mismatch_found': return <AlertTriangle className="h-3.5 w-3.5" />
   128→   128→      case 'filing_completed': return <Check className="h-3.5 w-3.5" />
   129→   129→      case 'extraction_complete': return <FileText className="h-3.5 w-3.5" />
   130→   130→      case 'issue_detected': return <AlertTriangle className="h-3.5 w-3.5" />
   131→   131→      case 'ai_insight': return <Sparkles className="h-3.5 w-3.5" />
   132→   132→      default: return <Bell className="h-3.5 w-3.5" />
   133→   133→    }
   134→   134→  }
   135→   135→
   136→   136→  const displayed = (allNotifs || []).slice(0, 15)
   137→   137→
   138→   138→  return (
   139→   139→    <Popover open={open} onOpenChange={setOpen}>
   140→   140→      <PopoverTrigger asChild>
   141→   141→        <Button variant="ghost" size="sm" className="relative h-8 w-8 p-0 rounded-lg hover:bg-accent">
   142→   142→          {unreadCount > 0 ? (
   143→   143→            <BellRing className="h-4 w-4 text-amber-600" />
   144→   144→          ) : (
   145→   145→            <Bell className="h-4 w-4 text-muted-foreground" />
   146→   146→          )}
   147→   147→          {unreadCount > 0 && (
   148→   148→            <span className="absolute -top-0.5 -right-0.5 flex h-4 min-w-4 items-center justify-center rounded-full bg-red-500 px-1 text-[9px] font-bold text-white">
   149→   149→              {unreadCount > 9 ? '9+' : unreadCount}
   150→   150→            </span>
   151→   151→          )}
   152→   152→        </Button>
   153→   153→      </PopoverTrigger>
   154→   154→      <PopoverContent align="end" className="w-80 sm:w-96 p-0 gap-0" sideOffset={8}>
   155→   155→        <div className="flex items-center justify-between px-4 py-3 border-b">
   156→   156→          <div className="flex items-center gap-2">
   157→   157→            <h3 className="text-sm font-semibold">Notifications</h3>
   158→   158→            {unreadCount > 0 && (
   159→   159→              <Badge variant="secondary" className="h-5 px-1.5 text-[10px] bg-red-100 text-red-700">
   160→   160→                {unreadCount} new
   161→   161→              </Badge>
   162→   162→            )}
   163→   163→          </div>
   164→   164→          {unreadCount > 0 && (
   165→   165→            <Button variant="ghost" size="sm" onClick={handleMarkAllRead} className="h-6 text-[10px] text-emerald-600 hover:text-emerald-700 px-2">
   166→   166→              Mark all read
   167→   167→            </Button>
   168→   168→          )}
   169→   169→        </div>
   170→   170→        <ScrollArea className="max-h-80">
   171→   171→          {displayed.length === 0 ? (
   172→   172→            <div className="flex flex-col items-center justify-center py-8 px-4 text-center">
   173→   173→              <Bell className="h-8 w-8 text-muted-foreground/30 mb-2" />
   174→   174→              <p className="text-sm text-muted-foreground">No notifications yet</p>
   175→   175→              <p className="text-xs text-muted-foreground/60 mt-1">Notifications will appear here when returns are due, mismatches are found, or documents are processed.</p>
   176→   176→            </div>
   177→   177→          ) : (
   178→   178→            <div className="divide-y">
   179→   179→              {displayed.map((notif: FirestoreNotification & { id: string }) => (
   180→   180→                <div
   181→   181→                  key={notif.id}
   182→   182→                  className={`flex gap-3 px-4 py-3 hover:bg-accent/50 transition-colors cursor-pointer ${!notif.read ? 'bg-emerald-50/50 dark:bg-emerald-950/20' : ''}`}
   183→   183→                  onClick={() => !notif.read && handleMarkRead(notif.notifId || notif.id)}
   184→   184→                >
   185→   185→                  <div className={`mt-0.5 flex items-center justify-center h-7 w-7 rounded-full shrink-0 ${priorityColor(notif.priority)}`}>
   186→   186→                    {typeIcon(notif.type)}
   187→   187→                  </div>
   188→   188→                  <div className="flex-1 min-w-0">
   189→   189→                    <div className="flex items-center gap-1.5">
   190→   190→                      <p className={`text-xs font-medium truncate ${!notif.read ? 'text-foreground' : 'text-muted-foreground'}`}>
   191→   191→                        {notif.title}
   192→   192→                      </p>
   193→   193→                      {!notif.read && <span className={`h-1.5 w-1.5 rounded-full ${priorityBadge(notif.priority)} shrink-0`} />}
   194→   194→                    </div>
   195→   195→                    <p className="text-[11px] text-muted-foreground line-clamp-2 mt-0.5">{notif.message}</p>
   196→   196→                    <div className="flex items-center gap-2 mt-1">

   199→   199→                      </span>
   200→   200→                    </div>

   203→   203→              ))}
   204→   204→            </div>
   205→   205→          )}
   206→   206→        </ScrollArea>
   207→   207→        {displayed.length > 0 && (
   208→   208→          <div className="border-t px-4 py-2">
   209→   209→            <Button
   210→   210→              variant="ghost"
   211→   211→              size="sm"
   212→   212→              className="w-full text-xs text-emerald-600 hover:text-emerald-700"
   213→   213→              onClick={() => {
   214→   214→                setOpen(false)

   216→   216→            >
   217→   217→              View All Notifications

   221→   221→      </PopoverContent>
   222→   222→    </Popover>

   229→   229→
   230→   230→function GlobalSearchBar() {
   231→   231→  const { setCurrentView, setSelectedClientId } = useApp()
   232→   232→  const [open, setOpen] = useState(false)
   233→   233→  const [query, setQuery] = useState('')
   234→   234→  const [selectedIndex, setSelectedIndex] = useState(0)
   235→   235→  const inputRef = useRef<HTMLInputElement>(null)
   236→   236→  const resultsRef = useRef<HTMLDivElement>(null)
   237→   237→
   238→   238→  const { data: clients } = useFireClients()
   239→   239→  const { data: invoices } = useFireInvoices()
   240→   240→  const { data: returns } = useFireReturns()
   241→   241→  const { data: activities } = useFireRecentActivities(20)
   242→   242→
   243→   243→  const results = useMemo(() => {
   244→   244→    if (!query.trim()) return []
   245→   245→    const q = query.toLowerCase()
   246→   246→    const found: Array<{ type: string; id: string; title: string; subtitle: string; clientId?: string }> = []
   247→   247→
   248→   248→    for (const c of clients) {
   249→   249→      if (c.tradeName?.toLowerCase().includes(q) || c.gstin?.toLowerCase().includes(q) || c.legalName?.toLowerCase().includes(q)) {
   250→   250→        found.push({ type: 'client', id: c.id, title: c.tradeName, subtitle: `${c.gstin}${c.state ? ` · ${c.state}` : ''}`, clientId: c.id })
   251→   251→      }
   252→   252→    }
   253→   253→    for (const inv of invoices) {
   254→   254→      if (inv.invoiceNumber?.toLowerCase().includes(q) || inv.buyerName?.toLowerCase().includes(q) || inv.sellerGstin?.toLowerCase().includes(q)) {
   255→   255→        found.push({ type: 'invoice', id: inv.id, title: inv.invoiceNumber, subtitle: `${inv.invoiceType} · ₹${(inv.totalAmount || 0).toLocaleString('en-IN')}`, clientId: inv.clientId })
   256→   256→      }
   257→   257→    }
   258→   258→    for (const r of returns) {
   259→   259→      if (r.returnType?.toLowerCase().includes(q) || r.period?.toLowerCase().includes(q) || r.status?.toLowerCase().includes(q)) {
   260→   260→        found.push({ type: 'return', id: r.id, title: `${r.returnType} · ${r.period}`, subtitle: `${r.status}`, clientId: r.clientId })
   261→   261→      }
   262→   262→    }
   263→   263→    for (const a of activities) {
   264→   264→      if (a.title?.toLowerCase().includes(q) || a.description?.toLowerCase().includes(q)) {
   265→   265→        found.push({ type: 'activity', id: a.id, title: a.title, subtitle: a.description?.slice(0, 60) || '', clientId: a.clientId || undefined })
   266→   266→      }
   267→   267→    }
   268→   268→    return found.slice(0, 12)
   269→   269→  }, [query, clients, invoices, returns, activities])
   270→   270→
   271→   271→  const handleSelect = (result: typeof results[0]) => {
   272→   272→    setOpen(false)
   273→   273→    setQuery('')
   274→   274→    if (result.type === 'client' && result.clientId) {
   275→   275→      setSelectedClientId(result.clientId)
   276→   276→      setCurrentView('client-workspace')
   277→   277→    } else if (result.type === 'invoice') {
   278→   278→      setCurrentView('invoices')
   279→   279→    } else if (result.type === 'return') {
   280→   280→      setCurrentView('returns')
   281→   281→    }
   282→   282→  }
   283→   283→
   284→   284→  const handleKeyDown = (e: React.KeyboardEvent) => {
   285→   285→    if (e.key === 'ArrowDown') {
   286→   286→      e.preventDefault()
   287→   287→      setSelectedIndex(i => Math.min(i + 1, results.length - 1))
   288→   288→    } else if (e.key === 'ArrowUp') {
   289→   289→      e.preventDefault()
   290→   290→      setSelectedIndex(i => Math.max(i - 1, 0))
   291→   291→    } else if (e.key === 'Enter' && results[selectedIndex]) {
   292→   292→      handleSelect(results[selectedIndex])
   293→   293→    } else if (e.key === 'Escape') {
   294→   294→      setOpen(false)
   295→   295→      setQuery('')
   296→   296→    }
   297→   297→  }
   298→   298→
   299→   299→  const typeConfig: Record<string, { label: string; color: string; icon: React.ElementType }> = {
   300→   300→    client: { label: 'Client', color: 'text-emerald-600', icon: Building2 },
   301→   301→    invoice: { label: 'Invoice', color: 'text-amber-600', icon: FileText },
   302→   302→    return: { label: 'Return', color: 'text-blue-600', icon: FileText },
   303→   303→    activity: { label: 'Activity', color: 'text-purple-600', icon: Clock },
   304→   304→  }
   305→   305→
   306→   306→  return (
   307→   307→    <>
   308→   308→      {/* Search Trigger Button */}
   309→   309→      <button
   310→   310→        onClick={() => { setOpen(true); setTimeout(() => inputRef.current?.focus(), 100) }}
   311→   311→        className="flex items-center gap-2 h-8 rounded-lg border border-border/60 bg-muted/40 px-3 text-muted-foreground hover:bg-muted/60 hover:border-border transition-colors w-full max-w-xs sm:max-w-sm"
   312→   312→      >
   313→   313→        <Search className="h-3.5 w-3.5 shrink-0" />
   314→   314→        <span className="text-xs truncate flex-1 text-left">Search clients, invoices, returns...</span>
   315→   315→        <kbd className="hidden sm:inline-flex items-center gap-0.5 rounded border bg-background px-1 py-0.5 text-[9px] font-mono text-muted-foreground/60">
   316→   316→          <Command className="h-2.5 w-2.5" />K
   317→   317→        </kbd>
   318→   318→      </button>
   319→   319→
   320→   320→      {/* Search Dialog Overlay */}
   321→   321→      <AnimatePresence>
   322→   322→        {open && (
   323→   323→          <>
   324→   324→            <motion.div
   325→   325→              initial={{ opacity: 0 }}
   326→   326→              animate={{ opacity: 1 }}
   327→   327→              exit={{ opacity: 0 }}
   328→   328→              className="fixed inset-0 z-50 bg-black/40 backdrop-blur-sm"
   329→   329→              onClick={() => { setOpen(false); setQuery('') }}
   330→   330→            />
   331→   331→            <motion.div
   332→   332→              initial={{ opacity: 0, scale: 0.97, y: -10 }}
   333→   333→              animate={{ opacity: 1, scale: 1, y: 0 }}
   334→   334→              exit={{ opacity: 0, scale: 0.97, y: -10 }}
   335→   335→              transition={{ duration: 0.15 }}
   336→   336→              className="fixed left-1/2 top-[15%] z-50 w-full max-w-lg -translate-x-1/2 rounded-xl border bg-background shadow-2xl overflow-hidden"
   337→   337→            >
   338→   338→              <div className="flex items-center gap-3 px-4 py-3 border-b">
   339→   339→                <Search className="h-4 w-4 text-muted-foreground shrink-0" />
   340→   340→                <input
   341→   341→                  ref={inputRef}
   342→   342→                  type="text"
   343→   343→                  placeholder="Search clients, GSTIN, invoices, returns, activities..."
   344→   344→                  value={query}
   345→   345→                  onChange={e => { setQuery(e.target.value); setSelectedIndex(0) }}
   346→   346→                  onKeyDown={handleKeyDown}
   347→   347→                  className="flex-1 text-sm bg-transparent outline-none placeholder:text-muted-foreground"
   348→   348→                  autoFocus
   349→   349→                />
   350→   350→                <button onClick={() => { setOpen(false); setQuery('') }} className="text-muted-foreground hover:text-foreground">
   351→   351→                  <X className="h-4 w-4" />
   352→   352→                </button>
   353→   353→              </div>
   354→   354→
   355→   355→              <div ref={resultsRef} className="max-h-72 overflow-y-auto">
   356→   356→                {!query.trim() && (
   357→   357→                  <div className="px-4 py-6 text-center text-sm text-muted-foreground">
   358→   358→                    <Search className="h-6 w-6 text-muted-foreground/30 mx-auto mb-2" />
   359→   359→                    Start typing to search across all modules
   360→   360→                  </div>
   361→   361→                )}
   362→   362→                {query.trim() && results.length === 0 && (
   363→   363→                  <div className="px-4 py-6 text-center text-sm text-muted-foreground">
   364→   364→                    No results for &quot;{query}&quot;
   365→   365→                  </div>
   366→   366→                )}
   367→   367→                {results.map((result, i) => {
   368→   368→                  const cfg = typeConfig[result.type] || typeConfig.activity
   369→   369→                  const Icon = cfg.icon
   370→   370→                  return (
   371→   371→                    <button
   372→   372→                      key={`${result.type}-${result.id}`}
   373→   373→                      className={`w-full flex items-center gap-3 px-4 py-2.5 text-left transition-colors ${i === selectedIndex ? 'bg-emerald-50 dark:bg-emerald-950/30' : 'hover:bg-muted/50'}`}
   374→   374→                      onClick={() => handleSelect(result)}
   375→   375→                      onMouseEnter={() => setSelectedIndex(i)}
   376→   376→                    >
   377→   377→                      <div className="flex items-center justify-center h-7 w-7 rounded-lg bg-muted/50 shrink-0">
   378→   378→                        <Icon className={`h-3.5 w-3.5 ${cfg.color}`} />
   379→   379→                      </div>
   380→   380→                      <div className="flex-1 min-w-0">
   381→   381→                        <p className="text-sm font-medium truncate">{result.title}</p>
   382→   382→                        <p className="text-[11px] text-muted-foreground truncate">{result.subtitle}</p>
   383→   383→                      </div>
   384→   384→                      <span className="text-[10px] font-medium text-muted-foreground uppercase tracking-wider shrink-0">{cfg.label}</span>
   385→   385→                    </button>
   386→   386→                  )
   387→   387→                })}
   388→   388→              </div>
   389→   389→
   390→   390→              <div className="px-4 py-2 border-t bg-muted/20 flex items-center gap-4 text-[10px] text-muted-foreground">

   407→   407→// ═══════════════════════════════════════════════════════════════════════════════
   408→   408→
   409→   409→function useKeyboardShortcuts(onSearchOpen: () => void) {
   410→   410→  const { setCurrentView } = useApp()
   411→   411→  const gPressed = useRef(false)
   412→   412→  const gTimer = useRef<NodeJS.Timeout | null>(null)
   413→   413→
   414→   414→  useEffect(() => {
   415→   415→    const handler = (e: KeyboardEvent) => {
   416→   416→      // Ctrl+K → Open Search
   417→   417→      if ((e.metaKey || e.ctrlKey) && e.key === 'k') {
   418→   418→        e.preventDefault()
   419→   419→        onSearchOpen()
   420→   420→        return
   421→   421→      }
   422→   422→
   423→   423→      // G prefix shortcuts (G+D, G+C, G+I, G+R)
   424→   424→      if (e.key === 'g' && !e.metaKey && !e.ctrlKey && !e.altKey && !(e.target instanceof HTMLInputElement || e.target instanceof HTMLTextAreaElement)) {
   425→   425→        if (gPressed.current) return
   426→   426→        gPressed.current = true
   427→   427→        gTimer.current = setTimeout(() => { gPressed.current = false }, 500)
   428→   428→        return
   429→   429→      }
   430→   430→
   431→   431→      if (gPressed.current && !e.metaKey && !e.ctrlKey && !e.altKey) {
   432→   432→        gPressed.current = false
   433→   433→        if (gTimer.current) clearTimeout(gTimer.current)
   434→   434→
   435→   435→        switch (e.key) {
   436→   436→          case 'd': setCurrentView('dashboard'); break
   437→   437→          case 'c': setCurrentView('clients'); break
   438→   438→          case 'i': setCurrentView('invoices'); break
   439→   439→          case 'r': setCurrentView('returns'); break
   440→   440→        }
   441→   441→      }
   442→   442→    }
   443→   443→
   444→   444→    window.addEventListener('keydown', handler)
   445→   445→    return () => window.removeEventListener('keydown', handler)
   446→   446→  }, [setCurrentView, onSearchOpen])
   447→   447→}

   618→   618→function OnboardingScreen() {

   620→   620→  const { setCurrentScreen, setCurrentView } = useApp()
   621→   621→  const [saveError, setSaveError] = React.useState<string | null>(null)
   622→   622→
   623→   623→  const saveOnboardingToFirestore = async (
   624→   624→    data: import('@/components/onboarding/OnboardingFlow').OnboardingData,
   625→   625→    firmId: string
   626→   626→  ) => {
   627→   627→    try {
   628→   628→      const { doc, setDoc, serverTimestamp } = await import('firebase/firestore')
   629→   629→      const { db } = await import('@/lib/firebase')
   630→   630→      if (!user) return
   631→   631→
   632→   632→      await setDoc(doc(db, 'users', user.id), {
   633→   633→        uid: user.id, fullName: data.fullName, email: data.email, phone: data.phone,
   634→   634→        ageGroup: data.ageGroup, profession: data.profession, experience: data.experience,
   635→   635→        firmId: firmId || null, firmName: data.firmName, onboardingCompleted: true,
   636→   636→        clientCount: data.clientCount, monthlyReturns: data.monthlyReturns,
   637→   637→        gstServices: data.gstServices, painPoints: data.painPoints,
   638→   638→        referralSource: data.referralSource, trialReasons: data.trialReasons,
   639→   639→        wantsUpdates: data.wantsUpdates, updatedAt: serverTimestamp(),
   640→   640→      }, { merge: true })
   641→   641→
   642→   642→      await setDoc(doc(db, 'onboarding', user.id), {
   643→   643→        ...data, firmId: firmId || null, completedAt: serverTimestamp(),
   644→   644→      })
   645→   645→    } catch (error) {
   646→   646→      console.warn('[Onboarding] Background Firestore save failed:', error)
   647→   647→    }
   648→   648→  }
   649→   649→
   650→   650→  const handleOnboardingComplete = (
   651→   651→    data: import('@/components/onboarding/OnboardingFlow').OnboardingData,
   652→   652→    destination?: import('@/components/onboarding/OnboardingFlow').OnboardingDestination
   653→   653→  ) => {
   654→   654→    setSaveError(null)
   655→   655→    if (!user) { setSaveError('No authenticated user found. Please sign in again.'); return }
   656→   656→
   657→   657→    markOnboardingComplete(undefined, data.firmName)
   658→   658→    setCurrentView(destination || 'dashboard')
   659→   659→    setCurrentScreen('app')
   660→   660→
   661→   661→    ;(async () => {
   662→   662→      try {
   663→   663→        const { collection, addDoc, serverTimestamp } = await import('firebase/firestore')
   664→   664→        const { db } = await import('@/lib/firebase')
   665→   665→        let firmId = ''

   667→   667→          const firmRef = await addDoc(collection(db, 'firms'), {

   670→   670→            organizationType: data.organizationType,

   675→   675→          markOnboardingComplete(firmId, data.firmName)
   676→   676→        } catch (firmError) {
   677→   677→          console.warn('[Onboarding] Firm creation failed:', firmError)
   678→   678→        }
   679→   679→        await saveOnboardingToFirestore(data, firmId)
   680→   680→      } catch (error) {
   681→   681→        console.warn('[Onboarding] Background save error:', error)
   682→   682→      }
   683→   683→    })()

   690→   690→    markOnboardingComplete()
   691→   691→    setCurrentView('dashboard')
   692→   692→    setCurrentScreen('app')
   693→   693→
   694→   694→    ;(async () => {

   696→   696→        const { doc, setDoc, serverTimestamp } = await import('firebase/firestore')
   697→   697→        const { db } = await import('@/lib/firebase')
   698→   698→        await setDoc(doc(db, 'users', user.id), {
   699→   699→          onboardingCompleted: true, updatedAt: serverTimestamp(),
   700→   700→        }, { merge: true })
   701→   701→      } catch (error) {
   702→   702→        console.warn('[Onboarding] Background skip save failed:', error)
   703→   703→      }
   704→   704→    })()
   705→   705→  }
