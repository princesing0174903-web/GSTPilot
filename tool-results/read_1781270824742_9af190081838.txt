   114→   114→function getDaysRemaining(dueDateStr: string): number {
   115→   115→  const dueDate = new Date(dueDateStr);
   116→   116→  const today = new Date('2025-07-08');
   117→   117→  today.setHours(0, 0, 0, 0);
   118→   118→  dueDate.setHours(0, 0, 0, 0);
   119→   119→  return Math.ceil((dueDate.getTime() - today.getTime()) / (1000 * 60 * 60 * 24));
   120→   120→}
   121→   121→
   122→   122→function formatDaysRemaining(days: number): string {
   123→   123→  if (days < 0) return `${Math.abs(days)}d overdue`;
   124→   124→  if (days === 0) return 'Due today';
   125→   125→  if (days === 1) return 'Tomorrow';
   126→   126→  return `${days}d left`;
   127→   127→}
   128→   128→
   129→   129→function getInitials(name: string): string {
   130→   130→  return name

   133→   133→    .join('')
   134→   134→    .slice(0, 2)
   135→   135→    .toUpperCase();
   136→   136→}
