   209→   209→export function OnboardingFlow({
   210→   210→  onComplete,
   211→   211→  onSkip,
   212→   212→  userEmail = '',
   213→   213→  userName = '',
   214→   214→  error = null,
   215→   215→  onDismissError,
   216→   216→}: OnboardingFlowProps) {
   217→   217→  const [currentStep, setCurrentStep] = useState(0); // 0=welcome, 1-4=steps, 5=final
   218→   218→  const [direction, setDirection] = useState(1);
   219→   219→
   220→   220→  const [formData, setFormData] = useState<OnboardingData>({
   221→   221→    fullName: userName,
   222→   222→    email: userEmail,
   223→   223→    phone: '',
   224→   224→    ageGroup: '',
   225→   225→    profession: '',
   226→   226→    experience: '',
   227→   227→    firmName: '',
   228→   228→    organizationType: '',
   229→   229→    gstin: '',
   230→   230→    state: '',
   231→   231→    stateCode: '',
   232→   232→    icaiMembershipNo: '',
   233→   233→    officeAddress: '',
   234→   234→    clientCount: '',
   235→   235→    monthlyReturns: '',
   236→   236→    gstServices: [],
   237→   237→    painPoints: [],
   238→   238→    referralSource: '',
   239→   239→    trialReasons: [],
   240→   240→    wantsUpdates: true,
   241→   241→  });
   242→   242→
   243→   243→  const [errors, setErrors] = useState<Record<string, string>>({});
   244→   244→
   245→   245→  // ─── Helpers ──────────────────────────────────────────────────────────────
   246→   246→
   247→   247→  const updateField = useCallback(
   248→   248→    <K extends keyof OnboardingData>(field: K, value: OnboardingData[K]) => {
   249→   249→      setFormData((prev) => ({ ...prev, [field]: value }));
   250→   250→      // Clear error on change
   251→   251→      setErrors((prev) => {
   252→   252→        const next = { ...prev };
   253→   253→        delete next[field];
   254→   254→        return next;
   255→   255→      });
   256→   256→    },
   257→   257→    []
   258→   258→  );
   259→   259→
   260→   260→  const toggleArrayItem = useCallback(
   261→   261→    (field: 'gstServices' | 'painPoints' | 'trialReasons', item: string) => {
   262→   262→      setFormData((prev) => {
   263→   263→        const arr = prev[field] as string[];
   264→   264→        const next = arr.includes(item)
   265→   265→          ? arr.filter((i) => i !== item)
   266→   266→          : [...arr, item];
   267→   267→        return { ...prev, [field]: next };
   268→   268→      });
   269→   269→    },
   270→   270→    []
   271→   271→  );
   272→   272→
   273→   273→  const goNext = useCallback(() => {
   274→   274→    setDirection(1);
   275→   275→    setCurrentStep((prev) => Math.min(prev + 1, 5));
   276→   276→  }, []);
   277→   277→
   278→   278→  const goBack = useCallback(() => {
   279→   279→    setDirection(-1);
   280→   280→    setCurrentStep((prev) => Math.max(prev - 1, 0));
   281→   281→  }, []);
   282→   282→
   283→   283→  // ─── Validation ───────────────────────────────────────────────────────────
   284→   284→
   285→   285→  const validateStep = useCallback(
   286→   286→    (step: number): boolean => {
   287→   287→      const newErrors: Record<string, string> = {};
   288→   288→
   289→   289→      if (step === 1) {
   290→   290→        if (!formData.fullName.trim()) newErrors.fullName = 'Full name is required';
   291→   291→        if (!formData.email.trim()) newErrors.email = 'Email is required';
   292→   292→        else if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(formData.email))
   293→   293→          newErrors.email = 'Enter a valid email';
   294→   294→        if (!formData.phone.trim()) newErrors.phone = 'Mobile number is required';
   295→   295→        else if (!/^\d{10}$/.test(formData.phone.replace(/\s/g, '')))
   296→   296→          newErrors.phone = 'Enter a valid 10-digit mobile number';
   297→   297→      }
   298→   298→
   299→   299→      if (step === 2) {
   300→   300→        if (!formData.firmName.trim()) newErrors.firmName = 'Firm name is required';
   301→   301→      }
   302→   302→
   303→   303→      setErrors(newErrors);
   304→   304→      return Object.keys(newErrors).length === 0;
   305→   305→    },
   306→   306→    [formData]
   307→   307→  );
   308→   308→
   309→   309→  const handleNext = useCallback(() => {
   310→   310→    if (validateStep(currentStep)) {
   311→   311→      goNext();
   312→   312→    }
   313→   313→  }, [currentStep, validateStep, goNext]);
   314→   314→
   315→   315→  // ─── GSTIN Validation ─────────────────────────────────────────────────────
   316→   316→
   317→   317→  const getGstinStatus = useCallback(() => {
   318→   318→    const val = formData.gstin.trim();
   319→   319→    if (!val) return null;
   320→   320→    return GSTIN_REGEX.test(val) ? 'valid' : 'invalid';
   321→   321→  }, [formData.gstin]);
   322→   322→
   323→   323→  // ─── State Code Lookup ────────────────────────────────────────────────────
   324→   324→
   325→   325→  const handleStateChange = useCallback(
   326→   326→    (stateName: string) => {
   327→   327→      const stateObj = INDIAN_STATES.find((s) => s.name === stateName);
   328→   328→      setFormData((prev) => ({
   329→   329→        ...prev,
   330→   330→        state: stateName,
   331→   331→        stateCode: stateObj?.code ?? '',
   332→   332→      }));
   333→   333→      setErrors((prev) => {
   334→   334→        const next = { ...prev };
   335→   335→        delete next.state;
   336→   336→        return next;
   337→   337→      });
   338→   338→    },
   339→   339→    []
   340→   340→  );
   341→   341→
   342→   342→  // ─── Progress ─────────────────────────────────────────────────────────────
   343→   343→
   344→   344→  const totalSteps = 5;
   345→   345→  const progressPercent =
   346→   346→    currentStep === 0 ? 0 : (currentStep / totalSteps) * 100;
   347→   347→
   348→   348→  // ─── Step Icons ───────────────────────────────────────────────────────────
   349→   349→
   350→   350→  const stepIcons = [Zap, User, Building2, Briefcase, Target];
   351→   351→
   352→   352→  // ─── Render Step Content ──────────────────────────────────────────────────
   353→   353→
   354→   354→  const renderStep = () => {
   355→   355→    // Step 0 — Welcome
   356→   356→    if (currentStep === 0) {
   357→   357→      return (
   358→   358→        <motion.div
   359→   359→          key="step-0"
   360→   360→          custom={direction}
   361→   361→          variants={slideVariants}
   362→   362→          initial="enter"
   363→   363→          animate="center"
   364→   364→          exit="exit"
   365→   365→          transition={stepTransition}
   366→   366→          className="flex flex-col items-center justify-center text-center py-8"
   367→   367→        >
   368→   368→          <motion.div
   369→   369→            initial={{ scale: 0.5, opacity: 0 }}
   370→   370→            animate={{ scale: 1, opacity: 1 }}
   371→   371→            transition={{ type: 'spring', stiffness: 200, damping: 15, delay: 0.1 }}
   372→   372→            className="mb-8"
   373→   373→          >
   374→   374→            <div className="w-20 h-20 rounded-2xl bg-gradient-to-br from-emerald-400 to-teal-600 flex items-center justify-center shadow-lg shadow-emerald-200">
   375→   375→              <Zap className="size-10 text-white" />
   376→   376→            </div>
   377→   377→          </motion.div>
   378→   378→          <motion.h1
   379→   379→            initial={{ y: 20, opacity: 0 }}
   380→   380→            animate={{ y: 0, opacity: 1 }}
   381→   381→            transition={{ delay: 0.2 }}
   382→   382→            className="text-2xl font-bold tracking-tight text-foreground"
   383→   383→          >
   384→   384→            Welcome to GSTPilot
   385→   385→          </motion.h1>
   386→   386→          <motion.p
   387→   387→            initial={{ y: 20, opacity: 0 }}
   388→   388→            animate={{ y: 0, opacity: 1 }}
   389→   389→            transition={{ delay: 0.3 }}
   390→   390→            className="text-muted-foreground mt-2 max-w-sm"
   391→   391→          >
   392→   392→            Let&apos;s set up your workspace in less than 2 minutes.
   393→   393→          </motion.p>
   394→   394→          <motion.div
   395→   395→            initial={{ y: 20, opacity: 0 }}
   396→   396→            animate={{ y: 0, opacity: 1 }}
   397→   397→            transition={{ delay: 0.4 }}
   398→   398→            className="mt-8"
   399→   399→          >
   400→   400→            <Button
   401→   401→              onClick={goNext}
   402→   402→              size="lg"
   403→   403→              className="bg-emerald-600 hover:bg-emerald-700 text-white gap-2 px-8"

   405→   405→              Get Started

   407→   407→            </Button>
   408→   408→          </motion.div>
   409→   409→        </motion.div>
   410→   410→      );
   411→   411→    }
   412→   412→
   413→   413→    // Step 1 — About You
   414→   414→    if (currentStep === 1) {
   415→   415→      return (
   416→   416→        <motion.div
   417→   417→          key="step-1"
   418→   418→          custom={direction}
   419→   419→          variants={slideVariants}
   420→   420→          initial="enter"
   421→   421→          animate="center"
   422→   422→          exit="exit"
   423→   423→          transition={stepTransition}
   424→   424→          className="py-4"
   425→   425→        >
   426→   426→          <div className="flex items-center gap-3 mb-6">
   427→   427→            <div className="w-10 h-10 rounded-xl bg-emerald-100 flex items-center justify-center">
   428→   428→              <User className="size-5 text-emerald-600" />
   429→   429→            </div>
   430→   430→            <div>
   431→   431→              <h2 className="text-2xl font-bold">About You</h2>
   432→   432→              <p className="text-sm text-muted-foreground">
   433→   433→                Tell us about yourself so we can personalize your experience
   434→   434→              </p>
   435→   435→            </div>
   436→   436→          </div>
   437→   437→
   438→   438→          <div className="space-y-5">
   439→   439→            {/* Full Name */}
   440→   440→            <div className="space-y-2">
   441→   441→              <Label htmlFor="fullName">

   443→   443→              </Label>
   444→   444→              <Input
   445→   445→                id="fullName"
   446→   446→                placeholder="Rajesh Kumar Sharma"
   447→   447→                value={formData.fullName}
   448→   448→                onChange={(e) => updateField('fullName', e.target.value)}
   449→   449→                aria-invalid={!!errors.fullName}
   450→   450→                className={errors.fullName ? 'border-destructive' : ''}
   451→   451→              />
   452→   452→              {errors.fullName && (
   453→   453→                <p className="text-xs text-destructive">{errors.fullName}</p>
   454→   454→              )}
   455→   455→            </div>
   456→   456→
   457→   457→            {/* Work Email */}
   458→   458→            <div className="space-y-2">
   459→   459→              <Label htmlFor="email">
   460→   460→                Work Email <span className="text-destructive">*</span>
   461→   461→              </Label>
   462→   462→              <Input
   463→   463→                id="email"
   464→   464→                type="email"
   465→   465→                placeholder="rajesh@yourfirm.com"
   466→   466→                value={formData.email}
   467→   467→                onChange={(e) => updateField('email', e.target.value)}
   468→   468→                aria-invalid={!!errors.email}
   469→   469→                className={errors.email ? 'border-destructive' : ''}
   470→   470→              />
   471→   471→              {errors.email && (
   472→   472→                <p className="text-xs text-destructive">{errors.email}</p>
   473→   473→              )}
   474→   474→            </div>
   475→   475→
   476→   476→            {/* Mobile Number */}

   480→   480→              </Label>
   481→   481→              <div className="flex">
   482→   482→                <div className="flex items-center rounded-l-md border border-r-0 bg-muted px-3 text-sm text-muted-foreground h-9">
   483→   483→                  +91

   485→   485→                <Input

   493→   493→                  aria-invalid={!!errors.phone}

   500→   500→            </div>
   501→   501→
   502→   502→            {/* Age Group */}
