   219→function getGstr1DueDate(year: number, month: number): Date {
   220→  // 11th of following month
   221→  const nextMonth = month === 12 ? 1 : month + 1;
   222→  const nextYear = month === 12 ? year + 1 : year;
   223→  return new Date(nextYear, nextMonth - 1, 11);
   224→}
   225→
   226→function getGstr3BDueDate(year: number, month: number): Date {
   227→  // 20th of following month
   228→  const nextMonth = month === 12 ? 1 : month + 1;
   229→  const nextYear = month === 12 ? year + 1 : year;
   230→  return new Date(nextYear, nextMonth - 1, 20);
   231→}
   232→
   233→function getGstr2BDueDate(year: number, month: number): Date {
   234→  // 13th of following month
   235→  const nextMonth = month === 12 ? 1 : month + 1;
   236→  const nextYear = month === 12 ? year + 1 : year;
   237→  return new Date(nextYear, nextMonth - 1, 13);
   238→}
   239→
   240→function getAnnualReturnDueDate(year: number): Date {
   241→  // December 31st of following year
   242→  return new Date(year + 1, 11, 31);
   243→}
   244→
   245→function getTDSTCSDueDate(year: number, month: number): Date {
   246→  // 10th of following month
   247→  const nextMonth = month === 12 ? 1 : month + 1;
   248→  const nextYear = month === 12 ? year + 1 : year;
   249→  return new Date(nextYear, nextMonth - 1, 10);
   250→}
   251→
   252→function getDaysRemaining(dueDate: Date): number {
   253→  const today = new Date();
   254→  today.setHours(0, 0, 0, 0);
   255→  const due = new Date(dueDate);
   256→  due.setHours(0, 0, 0, 0);
   257→  return differenceInDays(due, today);
   258→}
   259→
   260→function getDaysOverdue(dueDate: Date): number {
   261→  const today = new Date();
   262→  today.setHours(0, 0, 0, 0);
   263→  const due = new Date(dueDate);
   264→  due.setHours(0, 0, 0, 0);
   265→  return differenceInDays(today, due);
   266→}
   267→
   268→function getStatusForDeadline(dueDate: Date, filedCount: number, totalCount: number): 'filed' | 'pending' | 'overdue' | 'upcoming' {
   269→  if (filedCount >= totalCount && totalCount > 0) return 'filed';
   270→  const daysRem = getDaysRemaining(dueDate);
   271→  if (daysRem < 0) return 'overdue';
   272→  if (daysRem <= 7) return 'pending';
   273→  return 'upcoming';
   274→}
   275→
   276→function formatDaysRemaining(days: number): string {
   277→  if (days < 0) return `${Math.abs(days)}d overdue`;
   278→  if (days === 0) return 'Due today';
   279→  if (days === 1) return '1 day left';
   280→  return `${days} days left`;
   281→}

   332→              <span

   363→  // ── Data Fetching ────────────────────────────────────────────────────────

   365→  useEffect(() => {
   366→    async function fetchData() {
   367→      try {
   368→        setLoading(true);

   370→          fetch('/api/dashboard'),

   379→        if (filingRes.ok) {

   384→        console.error('Error fetching deadline data:', err);
   385→      } finally {
   386→        setLoading(false);
   387→      }
   388→    }

   390→    fetchData();

   410→    const entries: DeadlineEntry[] = [...deadlineEntries];
   411→
   412→    // Add default GST deadlines if not covered by API data

   414→    const defaultReturnTypes = ['GSTR-1', 'GSTR-3B', 'GSTR-2B', 'TDS/TCS'];
