   569→function EmailVerificationBanner() {

   620→function OnboardingScreen() {
   621→  const { user, markOnboardingComplete } = useAuth()
   622→  const { setCurrentScreen, setCurrentView } = useApp()
   623→  const [saveError, setSaveError] = React.useState<string | null>(null)
   624→
   625→  // ── Fire-and-forget: save onboarding data to Firestore in the background ──
   626→  const saveOnboardingToFirestore = async (
   627→    data: import('@/components/onboarding/OnboardingFlow').OnboardingData,
   628→    firmId: string
   629→  ) => {
   630→    try {
   631→      const { doc, setDoc, serverTimestamp } = await import('firebase/firestore')
   632→      const { db } = await import('@/lib/firebase')
   633→
   634→      if (!user) return
   635→
   636→      // Update user document
   637→      await setDoc(doc(db, 'users', user.id), {
   638→        uid: user.id,
   639→        fullName: data.fullName,
   640→        email: data.email,
   641→        phone: data.phone,
   642→        ageGroup: data.ageGroup,
   643→        profession: data.profession,
   644→        experience: data.experience,
   645→        firmId: firmId || null,
   646→        firmName: data.firmName,
   647→        onboardingCompleted: true,
   648→        clientCount: data.clientCount,
   649→        monthlyReturns: data.monthlyReturns,
   650→        gstServices: data.gstServices,
   651→        painPoints: data.painPoints,
   652→        referralSource: data.referralSource,
   653→        trialReasons: data.trialReasons,
   654→        wantsUpdates: data.wantsUpdates,
   655→        updatedAt: serverTimestamp(),
   656→      }, { merge: true })
   657→      console.log('[Onboarding] ✅ User document saved to Firestore')
   658→
   659→      // Save onboarding record
   660→      await setDoc(doc(db, 'onboarding', user.id), {
   661→        ...data,
   662→        firmId: firmId || null,
   663→        completedAt: serverTimestamp(),
   664→      })
   665→      console.log('[Onboarding] ✅ Onboarding record saved to Firestore')
   666→    } catch (error) {
   667→      console.warn('[Onboarding] ⚠️ Background Firestore save failed:', error)
   668→    }
   669→  }
   670→
   671→  const handleOnboardingComplete = (
   672→    data: import('@/components/onboarding/OnboardingFlow').OnboardingData,
   673→    destination?: import('@/components/onboarding/OnboardingFlow').OnboardingDestination
   674→  ) => {
   675→    setSaveError(null)
   676→
   677→    if (!user) {
   678→      setSaveError('No authenticated user found. Please sign in again.')
   679→      return
   680→    }
   681→
   682→    // ── INSTANT: Mark onboarding complete locally & navigate ──
   683→    // This happens synchronously — user sees dashboard immediately
   684→    console.log('[Onboarding] 🚀 Navigating to', destination || 'dashboard', '(optimistic)')
   685→    markOnboardingComplete(undefined, data.firmName)
   686→    setCurrentView(destination || 'dashboard')
   687→    setCurrentScreen('app')
   688→
   689→    // ── BACKGROUND: Create firm doc + save everything to Firestore ──
   690→    // These happen async — user is already in the dashboard
   691→    console.log('[Onboarding] 📝 Starting background Firestore writes...')
   692→    ;(async () => {
   693→      try {
   694→        const { collection, addDoc, serverTimestamp } = await import('firebase/firestore')
   695→        const { db } = await import('@/lib/firebase')
   696→
   697→        // Create firm document
   698→        let firmId = ''
   699→        try {
   700→          const firmRef = await addDoc(collection(db, 'firms'), {
   701→            ownerId: user.id,
   702→            firmName: data.firmName,
   703→            gstin: data.gstin || null,
   704→            state: data.state,
   705→            stateCode: data.stateCode,
   706→            organizationType: data.organizationType,
   707→            icaiMembershipNo: data.icaiMembershipNo || null,
   708→            officeAddress: data.officeAddress || null,
   709→            createdAt: serverTimestamp(),
   710→          })
   711→          firmId = firmRef.id
   712→          console.log('[Onboarding] ✅ Firm created:', firmId)
   713→
   714→          // Update the local user with firmId
   715→          markOnboardingComplete(firmId, data.firmName)
   716→        } catch (firmError) {
   717→          console.warn('[Onboarding] ⚠️ Firm creation failed:', firmError)
   718→        }
   719→
   720→        // Save remaining data
   721→        await saveOnboardingToFirestore(data, firmId)
   722→      } catch (error) {
   723→        console.warn('[Onboarding] ⚠️ Background save error:', error)
   724→      }
   725→    })()
   726→  }
   727→
   728→  const handleSkip = () => {
   729→    setSaveError(null)
   730→
   731→    if (!user) {
   732→      setSaveError('No authenticated user found. Please sign in again.')
   733→      return
   734→    }
   735→
   736→    // ── INSTANT: Navigate immediately ──
   737→    console.log('[Onboarding] 🚀 Skipping onboarding (optimistic)')
   738→    markOnboardingComplete()
   739→    setCurrentView('dashboard')
   740→    setCurrentScreen('app')
   741→
   742→    // ── BACKGROUND: Mark as completed in Firestore ──
   743→    ;(async () => {
   744→      try {
   745→        const { doc, setDoc, serverTimestamp } = await import('firebase/firestore')
   746→        const { db } = await import('@/lib/firebase')
   747→        await setDoc(doc(db, 'users', user.id), {
   748→          onboardingCompleted: true,
   749→          updatedAt: serverTimestamp(),
   750→        }, { merge: true })
   751→        console.log('[Onboarding] ✅ Skip saved to Firestore')
   752→      } catch (error) {
   753→        console.warn('[Onboarding] ⚠️ Background skip save failed:', error)
   754→      }
   755→    })()
   756→  }
   757→
   758→  return (
   759→    <OnboardingFlow
   760→      onComplete={handleOnboardingComplete}
   761→      onSkip={handleSkip}
   762→      userEmail={user?.email}
   763→      userName={user?.name}
   764→      error={saveError}
   765→      onDismissError={() => setSaveError(null)}
   766→    />
   767→  )
   768→}

   770→function AppRouter() {
   771→  const { currentScreen, setCurrentScreen } = useApp()
   772→  const { isAuthenticated, isInitializing, needsOnboarding, needsEmailVerification } = useAuth()
   773→
   774→  useEffect(() => {
   775→    if (isInitializing) return
   776→    // Only auto-redirect to app if authenticated AND onboarding is complete
   777→    // Don't redirect if user still needs onboarding — the OnboardingScreen handles its own navigation
   778→    if (isAuthenticated && !needsOnboarding && currentScreen !== 'app') {
   779→      setCurrentScreen('app')
   780→    }
   781→  }, [isAuthenticated, isInitializing, currentScreen, setCurrentScreen, needsOnboarding])
   782→
   783→  useEffect(() => {
   784→    if (isInitializing) return
   785→    if (!isAuthenticated && currentScreen === 'app') {
   786→      setCurrentScreen('landing')
   787→    }
   788→  }, [isAuthenticated, isInitializing, currentScreen, setCurrentScreen])
   789→
   790→  const handleGetStarted = () => setCurrentScreen('login')
   791→  const handleBookDemo = () => setCurrentScreen('login')
   792→  const handleBackToLanding = () => setCurrentScreen('landing')
   793→
   794→  if (isInitializing && !isAuthenticated) {
   795→    return (
   796→      <div className="min-h-screen flex items-center justify-center bg-black">
   797→        <div className="flex flex-col items-center gap-4">
   798→          <div className="flex h-11 w-11 items-center justify-center rounded-2xl glass-surface motion-pulse">
   799→            <Zap className="h-5 w-5 accent-text" />
   800→          </div>
   801→          <span className="text-sm text-white/55 font-medium">Loading GSTPilot…</span>
   802→        </div>
   803→      </div>
   804→    )
   805→  }
   806→
   807→  // Show onboarding if user hasn't completed it
   808→  if (isAuthenticated && needsOnboarding) {
   809→    return <OnboardingScreen />
   810→  }
   811→
   812→  if (currentScreen === 'app' && isAuthenticated) {
   813→    return (
   814→      <div className="flex min-h-screen flex-col">
   815→        {needsEmailVerification && <EmailVerificationBanner />}
   816→        <DashboardContent />
   817→      </div>
   818→    )
   819→  }
   820→
   821→  if (currentScreen === 'login') {
   822→    return <LoginPage onBack={handleBackToLanding} onGetStarted={handleGetStarted} />
   823→  }
   824→
   825→  return <LandingPage onGetStarted={handleGetStarted} onBookDemo={handleBookDemo} />
   826→}

   828→export default function Home() {
   829→  return <AppRouter />
