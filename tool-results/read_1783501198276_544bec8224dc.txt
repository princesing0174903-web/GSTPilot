    67→/**
    68→ * Convert a Firestore snapshot ({ id, data() }) into a typed object with the
    69→ * document id prepended, recursively converting Firestore Timestamps to ISO
    70→ * strings. Mirrors the conversion logic in `withId` (firestore-schema.ts) but
    71→ * drops the `T extends Record<string, unknown>` constraint so it accepts
    72→ * strongly-typed interfaces (which do not have an implicit index signature).
    73→ */
    74→function docToData<T>(d: { id: string; data: () => Record<string, unknown> }): T & { id: string } {
    75→  const data = d.data();
    76→  const converted: Record<string, unknown> = {};
    77→  for (const [key, value] of Object.entries(data)) {
    78→    if (value && typeof value === 'object' && 'toDate' in value && typeof (value as { toDate: () => Date }).toDate === 'function') {
    79→      converted[key] = (value as { toDate: () => Date }).toDate().toISOString();
    80→    } else {
    81→      converted[key] = value;
    82→    }
    83→  }
    84→  return { id: d.id, ...converted } as T & { id: string };
    85→}

   447→export async function fileReturn(returnId: string): Promise<string> {
   448→  // CRITICAL: This function MUST NOT fake the ARN.
   449→  // GSTPilot never simulates successful government filings.
   450→  //
   451→  // The filing is delegated to the server-side API route which:
   452→  //   1. Checks for an active GSTN connection (Firestore gst_connections)
   453→  //   2. If no connection: returns 400 "GSTN connection required"
   454→  //   3. If mock provider: returns 400 "Cannot file in mock mode"
   455→  //   4. If official provider: calls provider.fileReturn() → real ARN
   456→  //   5. Only marks as "filed" if a real ARN is returned
   457→  //
   458→  // The return status transitions:
   459→  //   draft → prepared → validated → reviewed → submitted (awaiting GSTN ack)
   460→  //   submitted → filed (only when real ARN received from GSTN)
   461→  const response = await fetch(`/api/gstr-filing/${returnId}/file`, {
   462→    method: 'POST',
   463→    headers: { 'Content-Type': 'application/json' },
   464→  });
   465→
   466→  if (!response.ok) {
   467→    const error = await response.json().catch(() => ({ error: 'Filing failed' }));
   468→    throw new Error(error.error || 'Failed to file return with GSTN');
   469→  }
   470→
   471→  const result = await response.json();
   472→
   473→  // If the filing was submitted but not yet acknowledged (no real ARN),
   474→  // the return is marked as "submitted" — not "filed".
   475→  if (!result.acknowledgmentNumber) {
   476→    throw new Error(
   477→      'Return submitted to GSTN but not yet acknowledged. ' +
   478→      'The return status is now "submitted" — it will become "filed" when ' +
   479→      'GSTN returns an acknowledgment number (ARN). Check back after syncing.'
   480→    );
   481→  }
   482→
   483→  return result.acknowledgmentNumber as string;
   484→}

   486→/** Auto-create draft returns for current period.
   487→ *  Period format MUST be "YYYY-MM" to match gst-utils.ts (getFilingDueDate /
   488→ *  isOverdue / periodToLabel) and the dashboard's parsePeriod. Writing
   489→ *  "MM-YYYY" here previously caused "undefined 6" labels and inflated
   490→ *  overdue counts on the dashboard.
   491→ */
   492→async function autoCreateDraftReturns(firmId: string, clientId: string): Promise<void> {
   493→  const now = new Date();
   494→  const period = `${now.getFullYear()}-${String(now.getMonth() + 1).padStart(2, '0')}`;
   495→  const fy = now.getMonth() >= 3 ? `${now.getFullYear()}-${String(now.getFullYear() + 1).slice(2)}` : `${now.getFullYear() - 1}-${String(now.getFullYear()).slice(2)}`;
   496→
   497→  for (const returnType of ['GSTR-1', 'GSTR-3B'] as const) {
   498→    // Check if return already exists
   499→    const q = query(
   500→      collection(db, COLLECTIONS.RETURNS),
   501→      where('clientId', '==', clientId),
   502→      where('returnType', '==', returnType),
   503→      where('period', '==', period),
   504→      limit(1)
   505→    );
   506→    const existing = await getDocs(q);
   507→    if (existing.empty) {
   508→      await createReturn({
   509→        clientId,
   510→        returnType,
   511→        period,
   512→        financialYear: fy,
   513→        status: 'draft',
   514→        filedDate: null,
   515→        acknowledgmentNumber: null,
   516→        totalInvoices: 0,
   517→        readyForFiling: 0,
   518→        issuesFound: 0,
   519→        criticalErrors: 0,
   520→        warnings: 0,
   521→        totalTaxableValue: 0,
   522→        totalTax: 0,
   523→        jsonPayload: null,
   524→        assignedTo: null,
   525→        reviewedBy: null,
   526→      });
   527→    }
   528→  }
   529→}

   532→async function updateDraftReturnForInvoice(firmId: string, clientId: string, returnType: string, period: string, taxableValue: number, taxAmount: number): Promise<void> {
   533→  try {
   534→    const q = query(
   535→      collection(db, COLLECTIONS.RETURNS),
   536→      where('clientId', '==', clientId),
   537→      where('returnType', '==', returnType),
   538→      where('period', '==', period),
   539→      limit(1)
   540→    );
   541→    const snap = await getDocs(q);
   542→    if (!snap.empty) {
   543→      const returnDoc = snap.docs[0];
   544→      const data = returnDoc.data() as FirestoreReturn;
   545→      await updateDoc(returnDoc.ref, {
   546→        totalInvoices: (data.totalInvoices || 0) + 1,
   547→        totalTaxableValue: (data.totalTaxableValue || 0) + taxableValue,
   548→        totalTax: (data.totalTax || 0) + taxAmount,
   549→        updatedAt: serverTimestamp(),
   550→      });
   551→    }
   552→  } catch (e) {
   553→    console.warn('[Workflow] Failed to update draft return:', e);
   554→  }
   555→}

   563→export async function createReconciliation(data: {

   570→  const reconRef = doc(db, COLLECTIONS.RECONCILIATIONS, reconId);
   571→
   572→  // ── Production reconciliation ──
   573→  // Do NOT fabricate random mismatch records. Compute real counts from the
   574→  // books (Firestore invoices for this client+period) and the portal
   575→  // (GSTR-2B sync records if available). If portal data hasn't been synced
   576→  // yet, the reconciliation is created with status "pending" and zeroed
   577→  // counters — the real reconciliation engine (lib/banking/reconcile.ts or
   578→  // lib/gstn/reconcile.ts) populates the mismatches when it runs.
   579→  const booksQuery = query(
   580→    collection(db, COLLECTIONS.INVOICES),

   582→    where('period', '==', data.period)
   583→  );

   586→  const matched = 0;
   587→  const partialMatches = 0;
   588→  const unmatched = 0;
   589→  const highRisk = 0;
   590→  const mismatches: ReconMismatch[] = [];

   594→    reconId,
   595→    firmId: firmId || '',
   596→    clientId: data.clientId,
   597→    period: data.period,
   598→    sources: data.sources,
   599→    // Status is "pending" until the real reconciliation engine runs and
   600→    // populates mismatches. Previously this was hard-coded to "completed"
   601→    // with fabricated random counts.
   602→    status: 'pending',
   603→    totalRecords,
   604→    matched,
   605→    unmatched,

   623→    entityId: reconId,
   624→    metadata: { matched, unmatched, gstDifference, totalRecords },

   627→  // Do NOT auto-generate AI recommendations with random confidence scores.
   628→  // Real recommendations are generated by the reconciliation engine after

   637→    entityType: COLLECTIONS.RECONCILIATIONS,

   639→  });

   644→export async function resolveMismatch(reconId: string, invoiceNumber: string): Promise<void> {
   645→  const reconRef = doc(db, COLLECTIONS.RECONCILIATIONS, reconId);
   646→  const reconSnap = await getDoc(reconRef);
   647→  if (!reconSnap.exists()) return;
   648→
   649→  const data = reconSnap.data() as FirestoreReconciliation;
   650→  const updatedMismatches = data.mismatches.map(m =>
   651→    m.invoiceNumber === invoiceNumber
   652→      ? { ...m, resolved: true, resolvedBy: currentUserId(), resolvedAt: new Date().toISOString() }
   653→      : m
   654→  );
   655→
   656→  const resolvedCount = updatedMismatches.filter(m => m.resolved).length;
   657→  const newUnmatched = updatedMismatches.filter(m => !m.resolved && m.matchStatus === 'missing_in_books').length;
   658→  const newPartial = updatedMismatches.filter(m => !m.resolved && m.matchStatus === 'partial_match').length;
   659→
   660→  await updateDoc(reconRef, {
   661→    mismatches: updatedMismatches,
   662→    unmatched: newUnmatched,
   663→    partialMatches: newPartial,
   664→    matched: data.totalRecords - newUnmatched - newPartial,
   665→    updatedAt: serverTimestamp(),
   666→  });
   667→
   668→  addActivity({
   669→    type: 'mismatch_resolved',
   670→    title: 'Mismatch resolved',
   671→    description: `Reconciliation mismatch for ${invoiceNumber} resolved`,
   672→    clientId: data.clientId,
   673→    entityType: COLLECTIONS.RECONCILIATIONS,
   674→    entityId: reconId,
   675→  });
   676→}

   784→      suggestedAction: getSuggestedAction(recType),

   849→async function recalculateComplianceScore(clientId: string): Promise<void> {

   851→    const clientRef = doc(db, COLLECTIONS.CLIENTS, clientId);

   864→    score = Math.max(0, Math.min(100, score));

   866→    await updateDoc(clientRef, {

   868→      complianceProfile: {
   869→        ...client.complianceProfile,
   870→        filingCompliance: score,
   871→      },

   873→    });
   874→  } catch (e) {

   881→/** Subscribe to a Firestore collection with firm scoping */
   882→export function subscribeToCollection<T>(
   883→  collectionName: CollectionName,
   884→  callback: (data: Array<T & { id: string }>) => void,
   885→  ...constraints: QueryConstraint[]
   886→): Unsubscribe {
   887→  const firmId = currentFirmId();
   888→
   889→  const q = firmId
   890→    ? query(collection(db, collectionName), where('firmId', '==', firmId), ...constraints)
   891→    : query(collection(db, collectionName), ...constraints);
   892→
   893→  return onSnapshot(q, (snap) => {
   894→    const data = snap.docs.map(d => {
   895→      const docData = d.data();
   896→      const converted: Record<string, unknown> = { id: d.id };
   897→      for (const [key, value] of Object.entries(docData)) {
   898→        if (value && typeof value === 'object' && 'toDate' in value && typeof (value as { toDate: () => Date }).toDate === 'function') {
   899→          converted[key] = (value as { toDate: () => Date }).toDate().toISOString();
   900→        } else {
   901→          converted[key] = value;
   902→        }
   903→      }
   904→      return converted as T & { id: string };
   905→    });
   906→    callback(data);
   907→  }, (error) => {
   908→    console.warn(`[Firestore] Subscription error for ${collectionName}:`, error);
   909→    callback([]);
   910→  });
   911→}

   913→/** Subscribe to a single Firestore document */
   914→export function subscribeToDoc<T>(
   915→  collectionName: CollectionName,
   916→  docId: string,
   917→  callback: (data: (T & { id: string }) | null) => void,
   918→): Unsubscribe {
   919→  return onSnapshot(doc(db, collectionName, docId), (snap) => {
   920→    if (!snap.exists()) {
   921→      callback(null);
   922→      return;
   923→    }
   924→    const docData = snap.data();
   925→    const converted: Record<string, unknown> = { id: snap.id };
   926→    for (const [key, value] of Object.entries(docData)) {
   927→      if (value && typeof value === 'object' && 'toDate' in value && typeof (value as { toDate: () => Date }).toDate === 'function') {
   928→        converted[key] = (value as { toDate: () => Date }).toDate().toISOString();
   929→      } else {
   930→        converted[key] = value;
   931→      }
   932→    }
   933→    callback(converted as T & { id: string });
   934→  }, (error) => {
   935→    console.warn(`[Firestore] Doc subscription error for ${collectionName}/${docId}:`, error);
   936→    callback(null);
   937→  });
   938→}

   940→/** Compute live dashboard metrics from Firestore data */
   941→export function computeDashboardMetrics(
   942→  clients: Array<FirestoreClient & { id: string }>,
   943→  invoices: Array<FirestoreInvoice & { id: string }>,
   944→  returns: Array<FirestoreReturn & { id: string }>,
   945→  documents: Array<FirestoreDocument & { id: string }>,
   946→  activities: Array<FirestoreActivity & { id: string }>,
   947→): LiveDashboardMetrics {
   948→  const activeClients = clients.filter(c => c.status === 'active').length;
   949→  const filedReturns = returns.filter(r => r.status === 'filed').length;
   950→  const pendingReturns = returns.filter(r => r.status !== 'filed').length;
   951→  const readyToFile = returns.filter(r => ['validated', 'reviewed', 'generated'].includes(r.status)).length;
   952→  const overdueReturns = returns.filter(r => {
   953→    if (r.status === 'filed') return false;
   954→    const periodDate = parsePeriod(r.period);
   955→    return periodDate && periodDate < new Date();
   956→  }).length;
   957→
   958→  const totalTaxVolume = invoices.reduce((sum, inv) => sum + (inv.totalAmount || 0), 0);
   959→  const totalTax = invoices.reduce((sum, inv) => sum + (inv.cgst || 0) + (inv.sgst || 0) + (inv.igst || 0) + (inv.cess || 0), 0);
   960→
   961→  const criticalIssues = invoices.filter(inv => inv.riskLevel === 'critical').length +
   962→    returns.filter(r => r.criticalErrors > 0).length;
   963→  const warnings = invoices.filter(inv => inv.riskLevel === 'high' || inv.riskLevel === 'medium').length +
   964→    returns.filter(r => r.warnings > 0).length;
   965→
   966→  const averageHealthScore = clients.length > 0
   967→    ? Math.round(clients.reduce((sum, c) => sum + c.healthScore, 0) / clients.length)
   968→    : 0;
   969→
   970→  const matchedInvoices = invoices.filter(inv => inv.matchStatus === 'perfect_match').length;
   971→  const matchPercentage = invoices.length > 0 ? Math.round((matchedInvoices / invoices.length) * 100) : 100;
   972→
   973→  const riskyInvoices = invoices.filter(inv => ['high', 'critical'].includes(inv.riskLevel)).length;
   974→  const riskPercentage = invoices.length > 0 ? Math.round((riskyInvoices / invoices.length) * 100) : 0;
   975→
   976→  const documentsProcessed = documents.filter(d => d.extractionStatus === 'completed').length;
   977→  const extractionsPending = documents.filter(d => d.extractionStatus === 'pending' || d.extractionStatus === 'in_progress').length;
   978→
   979→  const recentActivities = [...activities]
   980→    .sort((a, b) => {
   981→      const aTime = typeof a.createdAt === 'string' ? new Date(a.createdAt).getTime() : 0;
   982→      const bTime = typeof b.createdAt === 'string' ? new Date(b.createdAt).getTime() : 0;
   983→      return bTime - aTime;
   984→    })
   985→    .slice(0, 10);
   986→
   987→  const upcomingFilings = returns
   988→    .filter(r => r.status !== 'filed')
   989→    .sort((a, b) => {
   990→      const aTime = typeof a.createdAt === 'string' ? new Date(a.createdAt).getTime() : 0;
   991→      const bTime = typeof b.createdAt === 'string' ? new Date(b.createdAt).getTime() : 0;
   992→      return aTime - bTime;
   993→    })
   994→    .slice(0, 5);
   995→
   996→  return {
   997→    totalClients: clients.length,
   998→    activeClients,
   999→    totalInvoices: invoices.length,
  1000→    totalTaxVolume,
  1001→    filedReturns,
  1002→    pendingReturns,
  1003→    overdueReturns,
  1004→    readyToFile,
  1005→    criticalIssues,
  1006→    warnings,
  1007→    averageHealthScore,
  1008→    matchPercentage,
  1009→    riskPercentage,
  1010→    documentsProcessed,
  1011→    extractionsPending,
  1012→    recentActivities,
  1013→    upcomingFilings,
  1014→  };
  1015→}

  1017→function parsePeriod(period: string | null): Date | null {
  1018→  if (!period) return null;
  1019→  try {
  1020→    const [mm, yyyy] = period.split('-').map(Number);
  1021→    if (!mm || !yyyy) return null;
  1022→    return new Date(yyyy, mm, 20); // Due date is typically 20th of next month
  1023→  } catch {
  1024→    return null;
  1025→  }
  1026→}
