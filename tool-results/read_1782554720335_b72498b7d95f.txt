    68→// COUNT-UP ANIMATION HOOK
    69→// ═══════════════════════════════════════════════════════════════════════════════
    70→
    71→function useCountUp(target: number, duration = 1500) {
    72→  const [count, setCount] = useState(0)
    73→  const prevTarget = useRef(0)
    74→  const countRef = useRef(0)
    75→
    76→  useEffect(() => {
    77→    if (target === prevTarget.current) return
    78→    prevTarget.current = target
    79→
    80→    const start = countRef.current
    81→    const startTime = Date.now()
    82→    const diff = target - start
    83→
    84→    const step = () => {
    85→      const elapsed = Date.now() - startTime
    86→      const progress = Math.min(elapsed / duration, 1)
    87→      const eased = 1 - Math.pow(1 - progress, 3)
    88→      const newVal = Math.round(start + diff * eased)
    89→      countRef.current = newVal
    90→      setCount(newVal)
    91→      if (progress < 1) requestAnimationFrame(step)
    92→    }
    93→    requestAnimationFrame(step)
    94→  }, [target, duration])
    95→
    96→  return count
    97→}

   580→function RadarChart({ dimensions, size = 280 }: {
   581→  dimensions: BusinessDimension[]
   582→  size?: number
   583→}) {
   584→  const cx = size / 2
   585→  const cy = size / 2
   586→  const r = size / 2 - 40
   587→  const n = dimensions.length
   588→  const angleStep = (2 * Math.PI) / n
   589→
   590→  const getPoint = (index: number, value: number) => {
   591→    const angle = angleStep * index - Math.PI / 2
   592→    const dist = (value / 100) * r
   593→    return {
   594→      x: cx + dist * Math.cos(angle),
   595→      y: cy + dist * Math.sin(angle),
   596→    }
   597→  }
   598→
   599→  // Grid rings
   600→  const rings = [20, 40, 60, 80, 100]
   601→
   602→  return (
   603→    <svg width={size} height={size} className="overflow-visible">
   604→      {/* Grid rings */}
   605→      {rings.map((ring) => (
   606→        <polygon
   607→          key={ring}
   608→          points={Array.from({ length: n }, (_, i) => {
   609→            const p = getPoint(i, ring)
   610→            return `${p.x},${p.y}`
   611→          }).join(' ')}
   612→          fill="none"
   613→          stroke="rgba(148,163,184,0.15)"
   614→          strokeWidth={1}
   615→        />
   616→      ))}
   617→
   618→      {/* Axis lines */}
   619→      {dimensions.map((_, i) => {
   620→        const p = getPoint(i, 100)
   621→        return (
   622→          <line
   623→            key={i}
   624→            x1={cx} y1={cy}
   625→            x2={p.x} y2={p.y}
   626→            stroke="rgba(148,163,184,0.12)"
   627→            strokeWidth={1}
   628→          />
   629→        )
   630→      })}
   631→
   632→      {/* Data polygon — glow effect */}
   633→      <polygon
   634→        points={dimensions.map((d, i) => {
   635→          const p = getPoint(i, d.score)
   636→          return `${p.x},${p.y}`
   637→        }).join(' ')}
   638→        fill="rgba(16, 185, 129, 0.12)"
   639→        stroke="rgba(16, 185, 129, 0.6)"
   640→        strokeWidth={2}
   641→        className="drop-shadow-[0_0_8px_rgba(16,185,129,0.4)]"
   642→      />
   643→
   644→      {/* Data polygon — main */}
   645→      <polygon
   646→        points={dimensions.map((d, i) => {
   647→          const p = getPoint(i, d.score)
   648→          return `${p.x},${p.y}`
   649→        }).join(' ')}
   650→        fill="rgba(16, 185, 129, 0.08)"
   651→        stroke="#10b981"
   652→        strokeWidth={2}
   653→      />
   654→
   655→      {/* Data points */}
   656→      {dimensions.map((d, i) => {
   657→        const p = getPoint(i, d.score)
   658→        const color = d.inverted
   659→          ? d.score >= 60 ? '#10b981' : d.score >= 40 ? '#f59e0b' : '#ef4444'
   660→          : d.score >= 80 ? '#10b981' : d.score >= 60 ? '#f59e0b' : '#ef4444'
   661→        return (
   662→          <g key={i}>
   663→            <circle cx={p.x} cy={p.y} r={5} fill={color} className="drop-shadow-[0_0_6px_rgba(16,185,129,0.5)]" />
   664→            <circle cx={p.x} cy={p.y} r={8} fill={color} opacity={0.2}>
   665→              <animate attributeName="r" values="6;12;6" dur="2s" repeatCount="indefinite" />
   666→              <animate attributeName="opacity" values="0.3;0.05;0.3" dur="2s" repeatCount="indefinite" />
   667→            </circle>
   668→          </g>
   669→        )
   670→      })}
   671→
   672→      {/* Labels */}
   673→      {dimensions.map((d, i) => {
   674→        const p = getPoint(i, 115)
   675→        return (
   676→          <text
   677→            key={i}

   747→function BeforeAfterBar({ label, before, after, format = 'number' }: {
   748→  label: string; before: number; after: number; format?: 'number' | 'percent' | 'inr'
   749→}) {
   750→  const maxVal = Math.max(Math.abs(before), Math.abs(after), 1)
   751→  const beforeW = Math.abs(before) / maxVal * 100
   752→  const afterW = Math.abs(after) / maxVal * 100
   753→  const isNeg = after < before
   754→
   755→  const formatVal = (v: number) => {
   756→    if (format === 'inr') return fmtINRFull(v)
   757→    if (format === 'percent') return pct(v)
   758→    return v.toLocaleString('en-IN')
   759→  }
   760→
   761→  return (
   762→    <div className="space-y-1.5">
   763→      <div className="flex items-center justify-between text-xs">
   764→        <span className="text-slate-400 font-medium">{label}</span>
   765→        <span className={`text-[11px] font-semibold ${isNeg ? 'text-red-400' : 'text-emerald-400'}`}>
   766→          {isNeg ? <ArrowDownRight className="inline h-3 w-3" /> : <ArrowUpRight className="inline h-3 w-3" />}
   767→          {formatVal(after)}
   768→        </span>
   769→      </div>
   770→      <div className="space-y-1">
   771→        <div className="flex items-center gap-2">
   772→          <span className="text-[9px] text-slate-500 w-8 shrink-0">Before</span>
   773→          <div className="flex-1 h-2 bg-slate-800 rounded-full overflow-hidden">
   774→            <motion.div
   775→              className="h-full bg-slate-500 rounded-full"
   776→              initial={{ width: 0 }}
   777→              animate={{ width: `${beforeW}%` }}
   778→              transition={{ duration: 1, delay: 0.2 }}
   779→            />
   780→          </div>
   781→        </div>
   782→        <div className="flex items-center gap-2">
   783→          <span className="text-[9px] text-slate-500 w-8 shrink-0">After</span>
   784→          <div className="flex-1 h-2 bg-slate-800 rounded-full overflow-hidden">
   785→            <motion.div
   786→              className={`h-full rounded-full ${isNeg ? 'bg-red-500' : 'bg-emerald-500'}`}
   787→              initial={{ width: 0 }}
   788→              animate={{ width: `${afterW}%` }}
   789→              transition={{ duration: 1, delay: 0.4 }}
   790→            />
   791→          </div>
   792→        </div>
   793→      </div>
   794→    </div>
   795→  )
   796→}

   799→// SVG CHART: PREDICTION ACCURACY

   801→
   802→function PredictionAccuracyChart({ data, w = 320, h = 160 }: {
   803→  data: { month: string; predicted: number; actual: number }[]
   804→  w?: number; h?: number
   805→}) {
   806→  const padding = { top: 20, right: 20, bottom: 30, left: 40 }
   807→  const chartW = w - padding.left - padding.right
   808→  const chartH = h - padding.top - padding.bottom
   809→  const allValues = data.flatMap(d => [d.predicted, d.actual])
   810→  const minVal = Math.min(...allValues) - 2
   811→  const maxVal = Math.max(...allValues) + 2
   812→  const range = maxVal - minVal || 1
   813→
   814→  const getX = (i: number) => padding.left + (i / (data.length - 1)) * chartW
   815→  const getY = (v: number) => padding.top + chartH - ((v - minVal) / range) * chartH
   816→
   817→  const predictedLine = data.map((d, i) => `${i === 0 ? 'M' : 'L'}${getX(i)},${getY(d.predicted)}`).join(' ')
   818→  const actualLine = data.map((d, i) => `${i === 0 ? 'M' : 'L'}${getX(i)},${getY(d.actual)}`).join(' ')
   819→
   820→  // Confidence band
   821→  const bandTop = data.map((d, i) => `${getX(i)},${getY(d.predicted + 3)}`).join(' L')
   822→  const bandBottom = data.map((d, i) => `${getX(i)},${getY(d.predicted - 3)}`).join(' L')
   823→  const bandPath = `M${bandTop} L${[...data].reverse().map((d, i) => `${getX(data.length - 1 - i)},${getY(d.predicted - 3)}`).join(' L')} Z`
   824→
   825→  return (
   826→    <svg width={w} height={h} className="overflow-visible">
   827→      {/* Confidence band */}
   828→      <path d={bandPath} fill="rgba(16,185,129,0.06)" stroke="none" />
   829→
   830→      {/* Grid lines */}
   831→      {[0.25, 0.5, 0.75].map((frac) => (
   832→        <line
   833→          key={frac}
   834→          x1={padding.left}
   835→          y1={padding.top + chartH * frac}
   836→          x2={w - padding.right}
   837→          y2={padding.top + chartH * frac}
   838→          stroke="rgba(148,163,184,0.08)"
   839→          strokeWidth={1}
   840→        />
   841→      ))}

   883→
   884→function MiniCompareChart({ current, last, label, w = 120, h = 40 }: {
   885→  current: number; last: number; label: string; w?: number; h?: number
   886→}) {
   887→  const generatePoints = (score: number) => {
   888→    const base = score / 100
   889→    return Array.from({ length: 8 }, (_, i) => {
   890→      const x = (i / 7) * w
   891→      const noise = Math.sin(i * 1.5) * 0.1 + Math.cos(i * 0.7) * 0.05
   892→      const y = h - (base + noise) * (h - 4) - 2
   893→      return { x, y: Math.max(2, Math.min(h - 2, y)) }
   894→    })
   895→  }
   896→
   897→  const currentPts = generatePoints(current)
   898→  const lastPts = generatePoints(last)
   899→
   900→  const toPath = (pts: { x: number; y: number }[]) =>
   901→    pts.map((p, i) => `${i === 0 ? 'M' : 'L'}${p.x},${p.y}`).join(' ')
   902→
   903→  const diff = current - last
   904→
   905→  return (
   906→    <div className="space-y-1">
   907→      <div className="flex items-center justify-between">
   908→        <span className="text-[10px] text-slate-400">{label}</span>
   909→        <span className={`text-[10px] font-semibold ${diff > 0 ? 'text-emerald-400' : diff < 0 ? 'text-red-400' : 'text-slate-400'}`}>
   910→          {diff > 0 ? '+' : ''}{diff}
   911→        </span>
   912→      </div>
   913→      <svg width={w} height={h} className="overflow-visible">
   914→        <path d={toPath(lastPts)} fill="none" stroke="rgba(148,163,184,0.3)" strokeWidth={1.5} />
   915→        <path d={toPath(currentPts)} fill="none" stroke="#10b981" strokeWidth={1.5} />
   916→      </svg>
   917→    </div>
   918→  )
   919→}
