   854→   854→  // ── PART 7 ── While the org context is resolving after auth, show a brief
   855→   855→  // loader so protected pages never mount with a null organization.
   856→   856→  if (isAuthenticated && orgLoading && !needsOnboarding) {
   857→   857→    return (
   858→   858→      <div className="min-h-screen flex items-center justify-center bg-black">
   859→   859→        <div className="flex flex-col items-center gap-4">
   860→   860→          <div className="flex h-11 w-11 items-center justify-center rounded-2xl glass-surface motion-pulse">
   861→   861→            <Zap className="h-5 w-5 accent-text" />
   862→   862→          </div>
   863→   863→          <span className="text-sm text-white/55 font-medium">Loading your workspace…</span>
   864→   864→        </div>
   865→   865→      </div>
   866→   866→    )
   867→   867→  }
   868→   868→
   869→   869→  // ── PART 2 ── Authenticated but no organization → onboarding creates one.
   870→   870→  // BUT: if there was a load error (e.g. Firestore unreachable), show a retry
   871→   871→  // screen instead of onboarding — onboarding would also fail to write.
   872→   872→  if (needsOnboarding && !orgError) {
   873→   873→    return <OnboardingScreen />
   874→   874→  }
   875→   875→
   876→   876→  // ── PART 8 ── Org context failed to load after retries. Show a friendly
   877→   877→  // error with a retry button. Never crash, never expose Firebase errors.
   878→   878→  if (isAuthenticated && !orgLoading && orgError && !organization) {
   879→   879→    return (
   880→   880→      <div className="min-h-screen flex items-center justify-center bg-black p-6">
   881→   881→        <div className="flex flex-col items-center gap-5 max-w-md text-center">
   882→   882→          <div className="flex h-12 w-12 items-center justify-center rounded-2xl bg-red-500/10 border border-red-500/20">
   883→   883→            <Zap className="h-6 w-6 text-red-400" />
   884→   884→          </div>
   885→   885→          <h2 className="text-xl font-bold text-white">Couldn&apos;t load your workspace</h2>
   886→   886→          <p className="text-sm text-white/55 leading-relaxed">
   887→   887→            {orgError}
   888→   888→          </p>
   889→   889→          <button
   890→   890→            onClick={() => reloadOrg()}
   891→   891→            className="mt-2 inline-flex items-center gap-2 rounded-xl bg-white px-5 py-2.5 text-sm font-semibold text-black hover:bg-white/90 transition-colors"
   892→   892→          >
   893→   893→            Try again
   894→   894→          </button>
   895→   895→        </div>
   896→   896→      </div>
   897→   897→    )
   898→   898→  }
   899→   899→
   900→   900→  // ── PART 5 ── Protected route: only render the app shell when authenticated
   901→   901→  // AND the organization context is fully loaded.
   902→   902→  if (currentScreen === 'app' && isAuthenticated && organization) {
   903→   903→    return (
   904→   904→      <div className="flex min-h-screen flex-col">
   905→   905→        {needsEmailVerification && <EmailVerificationBanner />}
   906→   906→        <DashboardContent />
   907→   907→      </div>
   908→   908→    )
   909→   909→  }
   910→   910→
   911→   911→  if (currentScreen === 'login') {
   912→   912→    return <LoginPage onBack={handleBackToLanding} onGetStarted={handleGetStarted} />
   913→   913→  }
   914→   914→
   915→   915→  return <LandingPage onGetStarted={handleGetStarted} onBookDemo={handleBookDemo} />
   916→   916→}
