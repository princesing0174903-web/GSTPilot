   163→function formatFileSize(bytes: number): string {
   164→  if (bytes < 1024) return `${bytes} B`;
   165→  if (bytes < 1048576) return `${(bytes / 1024).toFixed(1)} KB`;
   166→  return `${(bytes / 1048576).toFixed(1)} MB`;
   167→}

   169→function getConfidenceColor(confidence: number): string {
   170→  if (confidence >= 95) return 'text-emerald-600';
   171→  if (confidence >= 80) return 'text-amber-600';

   173→}

   226→// ─── Map API Invoice to local ExtractedInvoice ────────────────────────────────
   227→
   228→function mapInvoiceStatus(status: string): ValidationStatus {
   229→  if (status === 'approved' || status === 'filed') return 'validated';
   230→  if (status === 'draft') return 'warning';
   231→  return 'error';
   232→}
   233→
   234→function mapInvoiceToExtracted(inv: Invoice, clientName: string): ExtractedInvoice {
   235→  const vStatus = mapInvoiceStatus(inv.status);
   236→  return {
   237→    id: inv.id,
   238→    invoiceNumber: inv.invoiceNumber,
   239→    clientName,
   240→    invoiceDate: inv.invoiceDate,
   241→    section: inv.gstr1Section,
   242→    sectionLabel: GSTR1_SECTION_LABELS[inv.gstr1Section] ?? inv.gstr1Section.toUpperCase(),
   243→    taxableValue: inv.taxableValue,
   244→    cgst: inv.cgst,
   245→    sgst: inv.sgst,
   246→    igst: inv.igst,
   247→    totalTax: inv.cgst + inv.sgst + inv.igst,
   248→    totalAmount: inv.totalAmount,
   249→    validationStatus: vStatus,
   250→    ocrConfidence: vStatus === 'error' ? 75.0 : vStatus === 'warning' ? 92.0 : 98.5,
   251→    missingFields: inv.aiExplanation ? [inv.aiExplanation.split(' ').slice(0, 2).join(' ')] : [],
   252→    validationMessages: inv.aiExplanation ? [inv.aiExplanation] : [],
   253→  };
   254→}

   346→    // First, derive issues from invoice validation status

   350→        for (const msg of inv.validationMessages) {

   565→        {/* ════════════════════════════════════════════════════════════════════

   568→        <motion.div

   571→          transition={{ duration: 0.4 }}

   573→        >
