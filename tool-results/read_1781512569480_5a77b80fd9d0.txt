    42→    42→function formatRelativeTime(ts: string) {
    43→    43→  const diff = Date.now() - new Date(ts).getTime();
    44→    44→  const m = Math.floor(diff / 60000);
    45→    45→  if (m < 1) return 'Just now';
    46→    46→  if (m < 60) return `${m}m ago`;
    47→    47→  const h = Math.floor(m / 60);
    48→    48→  if (h < 24) return `${h}h ago`;
    49→    49→  const d = Math.floor(h / 24);
    50→    50→  if (d < 7) return `${d}d ago`;
    51→    51→  return new Date(ts).toLocaleDateString();
    52→    52→}

    54→    54→function formatFileSize(bytes: number) {
    55→    55→  if (bytes < 1024) return `${bytes} B`;
    56→    56→  if (bytes < 1048576) return `${(bytes / 1024).toFixed(1)} KB`;
    57→    57→  return `${(bytes / 1048576).toFixed(1)} MB`;
    58→    58→}

    90→    90→function CircularProgress({ value, size = 80, strokeWidth = 6 }: { value: number; size?: number; strokeWidth?: number }) {
    91→    91→  const radius = (size - strokeWidth) / 2;
    92→    92→  const circumference = 2 * Math.PI * radius;
    93→    93→  const offset = circumference - (value / 100) * circumference;
    94→    94→  const color = value >= 80 ? '#10b981' : value >= 50 ? '#f59e0b' : '#ef4444';
    95→    95→  const textColor = value >= 80 ? 'text-emerald-600' : value >= 50 ? 'text-amber-600' : 'text-red-600';
    96→    96→
    97→    97→  return (
    98→    98→    <div className="relative shrink-0" style={{ width: size, height: size }}>
    99→    99→      <svg width={size} height={size} className="-rotate-90">
   100→   100→        <circle cx={size / 2} cy={size / 2} r={radius} fill="none" stroke="currentColor" strokeWidth={strokeWidth} className="text-slate-100" />
   101→   101→        <circle cx={size / 2} cy={size / 2} r={radius} fill="none" stroke={color} strokeWidth={strokeWidth} strokeDasharray={circumference} strokeDashoffset={offset} strokeLinecap="round" className="transition-all duration-700 ease-out" />
   102→   102→      </svg>
   103→   103→      <div className="absolute inset-0 flex items-center justify-center">
   104→   104→        <span className={`text-lg font-bold ${textColor}`}>{value > 0 ? `${value}%` : '—'}</span>
   105→   105→      </div>
   106→   106→    </div>
   107→   107→  );
   108→   108→}
