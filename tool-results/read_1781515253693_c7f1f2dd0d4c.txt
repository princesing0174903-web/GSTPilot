   193→function SaveButton({ onSave }: { onSave: () => Promise<void> }) {
   194→  const [state, setState] = useState<'idle' | 'saving' | 'saved'>('idle')
   195→
   196→  const handleClick = async () => {
   197→    setState('saving')
   198→    try {
   199→      await onSave()
   200→      setState('saved')
   201→      setTimeout(() => setState('idle'), 2000)
   202→    } catch {
   203→      setState('idle')
   204→    }
   205→  }
   206→
   207→  return (
   208→    <Button
   209→      onClick={handleClick}
   210→      disabled={state === 'saving'}
   211→      className="bg-emerald-600 hover:bg-emerald-700 text-white gap-2 min-w-[110px]"
   212→    >
   213→      <AnimatePresence mode="wait">
   214→        {state === 'idle' && (
   215→          <motion.span
   216→            key="idle"
   217→            initial={{ opacity: 0, scale: 0.8 }}
   218→            animate={{ opacity: 1, scale: 1 }}
   219→            exit={{ opacity: 0, scale: 0.8 }}
   220→            className="flex items-center gap-2"
   221→          >
   222→            <Save className="h-4 w-4" />
   223→            Save
   224→          </motion.span>
   225→        )}

   227→          <motion.span
   228→            key="saving"
   229→            initial={{ opacity: 0, scale: 0.8 }}

   255→// ─── Password Strength Helper ──────────────────────────────────────────
   256→function getPasswordStrength(password: string): { score: number; label: string; color: string } {
   257→  if (!password) return { score: 0, label: '', color: '' }
   258→  let score = 0
   259→  if (password.length >= 8) score++
   260→  if (password.length >= 12) score++
   261→  if (/[A-Z]/.test(password)) score++
   262→  if (/[0-9]/.test(password)) score++
   263→  if (/[^A-Za-z0-9]/.test(password)) score++
   264→
   265→  if (score <= 1) return { score: 20, label: 'Weak', color: 'bg-red-500' }
   266→  if (score <= 2) return { score: 40, label: 'Fair', color: 'bg-orange-500' }
   267→  if (score <= 3) return { score: 60, label: 'Good', color: 'bg-yellow-500' }
   268→  if (score <= 4) return { score: 80, label: 'Strong', color: 'bg-emerald-500' }
   269→  return { score: 100, label: 'Excellent', color: 'bg-emerald-600' }
   270→}
