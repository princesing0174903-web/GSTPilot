    53→function getDaysRemaining(dueDateStr: string): number {
    54→  const dueDate = new Date(dueDateStr);
    55→  const today = new Date();
    56→  today.setHours(0, 0, 0, 0);
    57→  dueDate.setHours(0, 0, 0, 0);
    58→  return Math.ceil((dueDate.getTime() - today.getTime()) / (1000 * 60 * 60 * 24));
    59→}
    60→
    61→function formatDaysRemaining(days: number): string {
    62→  if (days < 0) return `${Math.abs(days)}d overdue`;
    63→  if (days === 0) return 'Due today';
    64→  if (days === 1) return 'Tomorrow';
    65→  return `${days}d left`;
    66→}
    67→
    68→function getInitials(name: string): string {
    69→  return name.split(' ').map(w => w[0]).join('').slice(0, 2).toUpperCase();
    70→}
    71→
    72→function formatINR(amount: number): string {
    73→  return amount.toLocaleString('en-IN');
    74→}
    75→
    76→function timeAgo(dateStr: string): string {
    77→  const now = new Date();
    78→  const date = new Date(dateStr);
    79→  const seconds = Math.floor((now.getTime() - date.getTime()) / 1000);
    80→  if (seconds < 60) return 'just now';
    81→  const minutes = Math.floor(seconds / 60);
    82→  if (minutes < 60) return `${minutes}m ago`;
    83→  const hours = Math.floor(minutes / 60);
    84→  if (hours < 24) return `${hours}h ago`;
    85→  const days = Math.floor(hours / 24);
    86→  if (days < 7) return `${days}d ago`;
    87→  return date.toLocaleDateString('en-IN', { day: 'numeric', month: 'short' });
    88→}

   135→function activityIcon(type: string): { icon: React.ReactNode; color: string; bg: string } {
   136→  if (type.includes('filed')) return { icon: <CheckCircle2 className="h-3.5 w-3.5" />, color: 'text-emerald-600', bg: 'bg-emerald-50' };
   137→  if (type.includes('upload') || type.includes('document')) return { icon: <Upload className="h-3.5 w-3.5" />, color: 'text-blue-600', bg: 'bg-blue-50' };
   138→  if (type.includes('review')) return { icon: <ClipboardCheck className="h-3.5 w-3.5" />, color: 'text-violet-600', bg: 'bg-violet-50' };
   139→  if (type.includes('client')) return { icon: <Users className="h-3.5 w-3.5" />, color: 'text-sky-600', bg: 'bg-sky-50' };
   140→  if (type.includes('reconcil') || type.includes('mismatch')) return { icon: <AlertTriangle className="h-3.5 w-3.5" />, color: 'text-amber-600', bg: 'bg-amber-50' };
   141→  if (type.includes('invoice') || type.includes('extract')) return { icon: <FileText className="h-3.5 w-3.5" />, color: 'text-teal-600', bg: 'bg-teal-50' };
   142→  return { icon: <Activity className="h-3.5 w-3.5" />, color: 'text-slate-500', bg: 'bg-slate-50' };
   143→}
