    59→function getDaysRemaining(dueDateStr: string): number {
    60→  const dueDate = new Date(dueDateStr);
    61→  const today = new Date();
    62→  today.setHours(0, 0, 0, 0);
    63→  dueDate.setHours(0, 0, 0, 0);
    64→  return Math.ceil((dueDate.getTime() - today.getTime()) / (1000 * 60 * 60 * 24));
    65→}
    66→
    67→function formatDaysRemaining(days: number): string {
    68→  if (days < 0) return `${Math.abs(days)}d overdue`;
    69→  if (days === 0) return 'Due today';
    70→  if (days === 1) return 'Tomorrow';
    71→  return `${days}d left`;
    72→}
    73→
    74→function formatINR(amount: number): string {
    75→  return amount.toLocaleString('en-IN');
    76→}
    77→
    78→function formatDateIN(dateStr: string): string {
    79→  const d = new Date(dateStr);
    80→  if (Number.isNaN(d.getTime())) return '—';
    81→  const dd = String(d.getDate()).padStart(2, '0');
    82→  const mm = String(d.getMonth() + 1).padStart(2, '0');
    83→  const yyyy = d.getFullYear();
    84→  return `${dd}/${mm}/${yyyy}`;
    85→}
    86→
    87→function timeAgo(dateStr: string): string {
    88→  const now = new Date();
    89→  const date = new Date(dateStr);
    90→  const seconds = Math.floor((now.getTime() - date.getTime()) / 1000);
    91→  if (seconds < 60) return 'just now';
    92→  const minutes = Math.floor(seconds / 60);
    93→  if (minutes < 60) return `${minutes}m ago`;
    94→  const hours = Math.floor(minutes / 60);
    95→  if (hours < 24) return `${hours}h ago`;
    96→  const days = Math.floor(hours / 24);
    97→  if (days < 7) return `${days}d ago`;
    98→  return date.toLocaleDateString('en-IN', { day: 'numeric', month: 'short' });
    99→}

   114→// One-sentence AI insight derived from live metrics.
   115→function buildInsightSentence(
   116→  metrics: LiveDashboardMetrics,
   117→  pendingCollection: number,
   118→): string {
   119→  const parts: string[] = [];
   120→  if (metrics.overdueReturns > 0) {
   121→    parts.push(
   122→      `${metrics.overdueReturns} overdue return${metrics.overdueReturns > 1 ? 's' : ''}`,
   123→    );
   124→  } else if (metrics.pendingReturns > 0) {
   125→    parts.push(
   126→      `${metrics.pendingReturns} return${metrics.pendingReturns > 1 ? 's' : ''} to file`,
   127→    );
   128→  }
   129→  if (pendingCollection > 0) {
   130→    parts.push(`₹${formatINR(pendingCollection)} pending collection`);
   131→  }
   132→  if (metrics.criticalIssues > 0) {
   133→    parts.push(
   134→      `${metrics.criticalIssues} critical issue${metrics.criticalIssues > 1 ? 's' : ''}`,
   135→    );
   136→  }
   137→  if (parts.length === 0) {
   138→    return `All clear — ${metrics.filedReturns} returns filed and ${metrics.totalClients} clients in good standing.`;
   139→  }
   140→  if (parts.length === 1) return `${parts[0]}.`;
   141→  const last = parts.pop();
   142→  return `${parts.join(', ')} and ${last}.`;
   143→}

   262→function healthTier(score: number): { label: string; tone: string } {
   263→  if (score >= 85) return { label: 'Excellent', tone: 'text-emerald-400' };
   264→  if (score >= 70) return { label: 'Healthy', tone: 'text-emerald-400' };
   265→  if (score >= 50) return { label: 'At Risk', tone: 'text-amber-400' };
   266→  return { label: 'Critical', tone: 'text-amber-400' };
   267→}
