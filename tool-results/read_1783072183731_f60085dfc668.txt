   664→function OnboardingScreen() {
   665→  const { user, markOnboardingComplete } = useAuth()
   666→  const { setCurrentScreen, setCurrentView } = useApp()
   667→  const [saveError, setSaveError] = React.useState<string | null>(null)
   668→
   669→  // ── Fire-and-forget: save onboarding data to Firestore in the background ──
   670→  const saveOnboardingToFirestore = async (
   671→    data: import('@/components/onboarding/OnboardingFlow').OnboardingData,
   672→    firmId: string
   673→  ) => {
   674→    try {
   675→      const { doc, setDoc, serverTimestamp } = await import('firebase/firestore')
   676→      const { db } = await import('@/lib/firebase')
   677→
   678→      if (!user) return
   679→
   680→      // Update user document
   681→      await setDoc(doc(db, 'users', user.id), {
   682→        uid: user.id,
   683→        fullName: data.fullName,
   684→        email: data.email,
   685→        phone: data.phone,
   686→        ageGroup: data.ageGroup,
   687→        profession: data.profession,
   688→        experience: data.experience,
   689→        firmId: firmId || null,
   690→        firmName: data.firmName,
   691→        onboardingCompleted: true,
   692→        clientCount: data.clientCount,
   693→        monthlyReturns: data.monthlyReturns,
   694→        gstServices: data.gstServices,
   695→        painPoints: data.painPoints,
   696→        referralSource: data.referralSource,
   697→        trialReasons: data.trialReasons,
   698→        wantsUpdates: data.wantsUpdates,
   699→        updatedAt: serverTimestamp(),
   700→      }, { merge: true })
   701→      console.log('[Onboarding] ✅ User document saved to Firestore')
   702→
   703→      // Save onboarding record
   704→      await setDoc(doc(db, 'onboarding', user.id), {
   705→        ...data,
   706→        firmId: firmId || null,
   707→        completedAt: serverTimestamp(),
   708→      })
   709→      console.log('[Onboarding] ✅ Onboarding record saved to Firestore')
   710→    } catch (error) {
   711→      console.warn('[Onboarding] ⚠️ Background Firestore save failed:', error)
   712→    }
   713→  }
   714→
   715→  const handleOnboardingComplete = (
   716→    data: import('@/components/onboarding/OnboardingFlow').OnboardingData,
   717→    destination?: import('@/components/onboarding/OnboardingFlow').OnboardingDestination
   718→  ) => {
   719→    setSaveError(null)
   720→
   721→    if (!user) {
   722→      setSaveError('No authenticated user found. Please sign in again.')
   723→      return
   724→    }
   725→
   726→    // ── INSTANT: Mark onboarding complete locally & navigate ──
   727→    // This happens synchronously — user sees dashboard immediately
   728→    console.log('[Onboarding] 🚀 Navigating to', destination || 'dashboard', '(optimistic)')
   729→    markOnboardingComplete(undefined, data.firmName)
   730→    setCurrentView(destination || 'dashboard')
   731→    setCurrentScreen('app')
   732→
   733→    // ── BACKGROUND: Create firm doc + save everything to Firestore ──
   734→    // These happen async — user is already in the dashboard
   735→    console.log('[Onboarding] 📝 Starting background Firestore writes...')
   736→    ;(async () => {
   737→      try {
   738→        const { collection, addDoc, serverTimestamp } = await import('firebase/firestore')
   739→        const { db } = await import('@/lib/firebase')
   740→
   741→        // Create firm document
   742→        let firmId = ''
   743→        try {
   744→          const firmRef = await addDoc(collection(db, 'firms'), {
   745→            ownerId: user.id,
   746→            firmName: data.firmName,
   747→            gstin: data.gstin || null,
   748→            state: data.state,
   749→            stateCode: data.stateCode,
   750→            organizationType: data.organizationType,
   751→            icaiMembershipNo: data.icaiMembershipNo || null,
   752→            officeAddress: data.officeAddress || null,
   753→            createdAt: serverTimestamp(),
   754→          })
   755→          firmId = firmRef.id
   756→          console.log('[Onboarding] ✅ Firm created:', firmId)
   757→
   758→          // Update the local user with firmId
   759→          markOnboardingComplete(firmId, data.firmName)
   760→        } catch (firmError) {
   761→          console.warn('[Onboarding] ⚠️ Firm creation failed:', firmError)
   762→        }
   763→
   764→        // Save remaining data
   765→        await saveOnboardingToFirestore(data, firmId)
   766→      } catch (error) {
   767→        console.warn('[Onboarding] ⚠️ Background save error:', error)
   768→      }
   769→    })()
   770→  }
   771→
   772→  const handleSkip = () => {
   773→    setSaveError(null)
   774→
   775→    if (!user) {
   776→      setSaveError('No authenticated user found. Please sign in again.')
   777→      return
   778→    }
   779→
   780→    // ── INSTANT: Navigate immediately ──
   781→    console.log('[Onboarding] 🚀 Skipping onboarding (optimistic)')
   782→    markOnboardingComplete()
   783→    setCurrentView('dashboard')
   784→    setCurrentScreen('app')
   785→
   786→    // ── BACKGROUND: Mark as completed in Firestore ──
   787→    ;(async () => {
   788→      try {
   789→        const { doc, setDoc, serverTimestamp } = await import('firebase/firestore')
   790→        const { db } = await import('@/lib/firebase')
   791→        await setDoc(doc(db, 'users', user.id), {
   792→          onboardingCompleted: true,
   793→          updatedAt: serverTimestamp(),
   794→        }, { merge: true })
   795→        console.log('[Onboarding] ✅ Skip saved to Firestore')
   796→      } catch (error) {
   797→        console.warn('[Onboarding] ⚠️ Background skip save failed:', error)
   798→      }
   799→    })()
   800→  }
   801→
   802→  return (
   803→    <OnboardingFlow
   804→      onComplete={handleOnboardingComplete}
   805→      onSkip={handleSkip}
   806→      userEmail={user?.email}
   807→      userName={user?.name}
   808→      error={saveError}
   809→      onDismissError={() => setSaveError(null)}
   810→    />
   811→  )
   812→}

   814→function AppRouter() {
   815→  const { currentScreen, setCurrentScreen } = useApp()
   816→  const { isAuthenticated, isInitializing, needsOnboarding, needsEmailVerification } = useAuth()
   817→
   818→  useEffect(() => {
   819→    if (isInitializing) return
   820→    // Only auto-redirect to app if authenticated AND onboarding is complete
   821→    // Don't redirect if user still needs onboarding — the OnboardingScreen handles its own navigation
   822→    if (isAuthenticated && !needsOnboarding && currentScreen !== 'app') {
   823→      setCurrentScreen('app')
   824→    }
   825→  }, [isAuthenticated, isInitializing, currentScreen, setCurrentScreen, needsOnboarding])
   826→
   827→  useEffect(() => {
   828→    if (isInitializing) return
   829→    if (!isAuthenticated && currentScreen === 'app') {
   830→      setCurrentScreen('landing')
   831→    }
   832→  }, [isAuthenticated, isInitializing, currentScreen, setCurrentScreen])
   833→
   834→  const handleGetStarted = () => setCurrentScreen('login')
   835→  const handleBookDemo = () => setCurrentScreen('login')
   836→  const handleBackToLanding = () => setCurrentScreen('landing')
   837→
   838→  if (isInitializing && !isAuthenticated) {
   839→    return (
   840→      <div className="min-h-screen flex items-center justify-center bg-black">
   841→        <div className="flex flex-col items-center gap-4">
   842→          <div className="flex h-11 w-11 items-center justify-center rounded-2xl glass-surface motion-pulse">
   843→            <Zap className="h-5 w-5 accent-text" />
   844→          </div>
   845→          <span className="text-sm text-white/55 font-medium">Loading GSTPilot…</span>
   846→        </div>
   847→      </div>
   848→    )
   849→  }
   850→
   851→  // Show onboarding if user hasn't completed it
   852→  if (isAuthenticated && needsOnboarding) {
   853→    return <OnboardingScreen />
   854→  }
   855→
   856→  if (currentScreen === 'app' && isAuthenticated) {
   857→    return (
   858→      <div className="flex min-h-screen flex-col">
   859→        {needsEmailVerification && <EmailVerificationBanner />}
   860→        <DashboardContent />
   861→      </div>
   862→    )
   863→  }
   864→
   865→  if (currentScreen === 'login') {
   866→    return <LoginPage onBack={handleBackToLanding} onGetStarted={handleGetStarted} />
   867→  }
   868→
   869→  return <LandingPage onGetStarted={handleGetStarted} onBookDemo={handleBookDemo} />
   870→}
