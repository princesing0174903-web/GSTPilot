   103→   103→function NotificationBell() {
   104→   104→  const { data: unreadNotifs } = useFireUnreadNotifications()

   107→   107→  const [open, setOpen] = useState(false)
   108→   108→
   109→   109→  const handleMarkRead = async (notifId: string) => {
   110→   110→    try { await markNotificationRead(notifId) } catch {}
   111→   111→  }

   114→   114→    try {
   115→   115→      for (const n of unreadNotifs) {

   118→   118→    } catch {}
   119→   119→  }

   121→   121→  const priorityColor = (p: string) => {
   122→   122→    switch (p) {
   123→   123→      case 'urgent': return 'text-red-600 bg-red-50 border-red-200'
   124→   124→      case 'high': return 'text-orange-600 bg-orange-50 border-orange-200'
   125→   125→      case 'normal': return 'text-blue-600 bg-blue-50 border-blue-200'
   126→   126→      default: return 'text-slate-600 bg-slate-50 border-slate-200'
   127→   127→    }
   128→   128→  }

   131→   131→    switch (p) {
   132→   132→      case 'urgent': return 'bg-red-500'

   134→   134→      case 'normal': return 'bg-blue-500'
   135→   135→      default: return 'bg-slate-400'
   136→   136→    }
   137→   137→  }
   138→   138→
   139→   139→  const typeIcon = (t: string) => {

   144→   144→      case 'extraction_complete': return <FileText className="h-3.5 w-3.5" />

   265→   265→// KEYBOARD SHORTCUTS HOOK
   266→   266→// ═══════════════════════════════════════════════════════════════════════════════
   267→   267→
   268→   268→function useKeyboardShortcuts() {
   269→   269→  const { setCurrentView, setCommandPaletteOpen, commandPaletteOpen } = useApp()
   270→   270→  const gPressed = useRef(false)
   271→   271→  const gTimer = useRef<NodeJS.Timeout | null>(null)
   272→   272→
   273→   273→  useEffect(() => {
   274→   274→    const handler = (e: KeyboardEvent) => {
   275→   275→      // Ctrl+K → Toggle Command Palette
   276→   276→      if ((e.metaKey || e.ctrlKey) && e.key === 'k') {
   277→   277→        e.preventDefault()
   278→   278→        setCommandPaletteOpen(!commandPaletteOpen)
   279→   279→        return
   280→   280→      }
   281→   281→
   282→   282→      // G prefix shortcuts (G+D, G+C, G+I, G+R)
   283→   283→      if (e.key === 'g' && !e.metaKey && !e.ctrlKey && !e.altKey && !(e.target instanceof HTMLInputElement || e.target instanceof HTMLTextAreaElement)) {
   284→   284→        if (gPressed.current) return
   285→   285→        gPressed.current = true
   286→   286→        gTimer.current = setTimeout(() => { gPressed.current = false }, 500)
   287→   287→        return
   288→   288→      }
   289→   289→
   290→   290→      if (gPressed.current && !e.metaKey && !e.ctrlKey && !e.altKey) {
   291→   291→        gPressed.current = false
   292→   292→        if (gTimer.current) clearTimeout(gTimer.current)
   293→   293→
   294→   294→        switch (e.key) {
   295→   295→          case 'd': setCurrentView('dashboard'); break
   296→   296→          case 'c': setCurrentView('clients'); break
   297→   297→          case 'i': setCurrentView('invoices'); break
   298→   298→          case 'r': setCurrentView('returns'); break
   299→   299→        }
   300→   300→      }
   301→   301→    }
   302→   302→
   303→   303→    window.addEventListener('keydown', handler)
   304→   304→    return () => window.removeEventListener('keydown', handler)
   305→   305→  }, [setCurrentView, setCommandPaletteOpen, commandPaletteOpen])
   306→   306→}

   483→   483→// ONBOARDING SCREEN
   484→   484→// ═══════════════════════════════════════════════════════════════════════════════
   485→   485→
   486→   486→function OnboardingScreen() {
   487→   487→  const { user, markOnboardingComplete } = useAuth()
   488→   488→  const { setCurrentScreen, setCurrentView } = useApp()
   489→   489→  const [saveError, setSaveError] = React.useState<string | null>(null)
   490→   490→
   491→   491→  const saveOnboardingToFirestore = async (
   492→   492→    data: import('@/components/onboarding/OnboardingFlow').OnboardingData,
   493→   493→    firmId: string
   494→   494→  ) => {
   495→   495→    try {
   496→   496→      const { doc, setDoc, serverTimestamp } = await import('firebase/firestore')
   497→   497→      const { db } = await import('@/lib/firebase')
   498→   498→      if (!user) return
   499→   499→
   500→   500→      await setDoc(doc(db, 'users', user.id), {
   501→   501→        uid: user.id, fullName: data.fullName, email: data.email, phone: data.phone,
   502→   502→        ageGroup: data.ageGroup, profession: data.profession, experience: data.experience,
   503→   503→        firmId: firmId || null, firmName: data.firmName, onboardingCompleted: true,
   504→   504→        clientCount: data.clientCount, monthlyReturns: data.monthlyReturns,
   505→   505→        gstServices: data.gstServices, painPoints: data.painPoints,
   506→   506→        referralSource: data.referralSource, trialReasons: data.trialReasons,
   507→   507→        wantsUpdates: data.wantsUpdates, updatedAt: serverTimestamp(),
   508→   508→      }, { merge: true })
   509→   509→
   510→   510→      await setDoc(doc(db, 'onboarding', user.id), {
   511→   511→        ...data, firmId: firmId || null, completedAt: serverTimestamp(),
   512→   512→      })
   513→   513→    } catch (error) {
   514→   514→      console.warn('[Onboarding] Background Firestore save failed:', error)
   515→   515→    }
   516→   516→  }
   517→   517→
   518→   518→  const handleOnboardingComplete = (
   519→   519→    data: import('@/components/onboarding/OnboardingFlow').OnboardingData,
   520→   520→    destination?: import('@/components/onboarding/OnboardingFlow').OnboardingDestination
   521→   521→  ) => {
   522→   522→    setSaveError(null)
   523→   523→    if (!user) { setSaveError('No authenticated user found. Please sign in again.'); return }
   524→   524→
   525→   525→    markOnboardingComplete(undefined, data.firmName)
   526→   526→    setCurrentView(destination || 'dashboard')
   527→   527→    setCurrentScreen('app')
   528→   528→
   529→   529→    ;(async () => {
   530→   530→      try {
   531→   531→        const { collection, addDoc, serverTimestamp } = await import('firebase/firestore')
   532→   532→        const { db } = await import('@/lib/firebase')
   533→   533→        let firmId = ''
   534→   534→        try {
   535→   535→          const firmRef = await addDoc(collection(db, 'firms'), {
   536→   536→            ownerId: user.id, firmName: data.firmName, gstin: data.gstin || null,
   537→   537→            state: data.state, stateCode: data.stateCode,
   538→   538→            organizationType: data.organizationType,
   539→   539→            icaiMembershipNo: data.icaiMembershipNo || null,
   540→   540→            officeAddress: data.officeAddress || null, createdAt: serverTimestamp(),
   541→   541→          })
   542→   542→          firmId = firmRef.id
   543→   543→          markOnboardingComplete(firmId, data.firmName)
   544→   544→        } catch (firmError) {
   545→   545→          console.warn('[Onboarding] Firm creation failed:', firmError)
   546→   546→        }
   547→   547→        await saveOnboardingToFirestore(data, firmId)
   548→   548→      } catch (error) {
   549→   549→        console.warn('[Onboarding] Background save error:', error)
   550→   550→      }
   551→   551→    })()
   552→   552→  }
   553→   553→
   554→   554→  const handleSkip = () => {
   555→   555→    setSaveError(null)
   556→   556→    if (!user) { setSaveError('No authenticated user found.'); return }
   557→   557→
   558→   558→    markOnboardingComplete()
   559→   559→    setCurrentView('dashboard')
   560→   560→    setCurrentScreen('app')
   561→   561→
   562→   562→    ;(async () => {
   563→   563→      try {
   564→   564→        const { doc, setDoc, serverTimestamp } = await import('firebase/firestore')
   565→   565→        const { db } = await import('@/lib/firebase')
   566→   566→        await setDoc(doc(db, 'users', user.id), {
   567→   567→          onboardingCompleted: true, updatedAt: serverTimestamp(),
   568→   568→        }, { merge: true })
   569→   569→      } catch (error) {
   570→   570→        console.warn('[Onboarding] Background skip save failed:', error)
   571→   571→      }
   572→   572→    })()
   573→   573→  }
   574→   574→
   575→   575→  return (
   576→   576→    <OnboardingFlow
   577→   577→      onComplete={handleOnboardingComplete}
   578→   578→      onSkip={handleSkip}
   579→   579→      userEmail={user?.email}
   580→   580→      userName={user?.name}
   581→   581→      error={saveError}
   582→   582→      onDismissError={() => setSaveError(null)}
   583→   583→    />
   584→   584→  )
   585→   585→}

   589→   589→// ═══════════════════════════════════════════════════════════════════════════════
   590→   590→
   591→   591→function AppRouter() {
   592→   592→  const { currentScreen, setCurrentScreen } = useApp()
   593→   593→  const { isAuthenticated, isInitializing, needsOnboarding, needsEmailVerification } = useAuth()
   594→   594→
   595→   595→  useEffect(() => {
   596→   596→    if (isInitializing) return
   597→   597→    if (isAuthenticated && !needsOnboarding && currentScreen !== 'app') {
   598→   598→      setCurrentScreen('app')
   599→   599→    }
   600→   600→  }, [isAuthenticated, isInitializing, currentScreen, setCurrentScreen, needsOnboarding])
   601→   601→
   602→   602→  useEffect(() => {
   603→   603→    if (isInitializing) return
   604→   604→    if (!isAuthenticated && currentScreen === 'app') {
   605→   605→      setCurrentScreen('landing')
   606→   606→    }
   607→   607→  }, [isAuthenticated, isInitializing, currentScreen, setCurrentScreen])
   608→   608→
   609→   609→  const handleGetStarted = () => setCurrentScreen('login')
   610→   610→  const handleBookDemo = () => setCurrentScreen('login')
   611→   611→  const handleBackToLanding = () => setCurrentScreen('landing')
   612→   612→
   613→   613→  if (isInitializing && !isAuthenticated) {
   614→   614→    return (
   615→   615→      <div className="min-h-screen flex items-center justify-center bg-white">
   616→   616→        <div className="flex flex-col items-center gap-3">
   617→   617→          <div className="flex h-10 w-10 items-center justify-center rounded-lg bg-gradient-to-br from-emerald-500 to-emerald-600 shadow-lg shadow-emerald-600/20">
   618→   618→            <Zap className="h-6 w-6 text-white animate-pulse" />
   619→   619→          </div>
   620→   620→          <span className="text-sm text-slate-500 font-medium">Loading GSTPilot...</span>
   621→   621→        </div>
   622→   622→      </div>
   623→   623→    )
   624→   624→  }
   625→   625→
   626→   626→  if (isAuthenticated && needsOnboarding) {
   627→   627→    return <OnboardingScreen />
   628→   628→  }
   629→   629→
   630→   630→  if (currentScreen === 'app' && isAuthenticated) {
   631→   631→    return (
   632→   632→      <div className="min-h-screen flex flex-col">
   633→   633→        {needsEmailVerification && <EmailVerificationBanner />}
   634→   634→        <div className="flex-1 flex">
   635→   635→          <DashboardContent />
   636→   636→        </div>
   637→   637→      </div>
   638→   638→    )
   639→   639→  }
   640→   640→
   641→   641→  if (currentScreen === 'login') {
   642→   642→    return <LoginPage onBack={handleBackToLanding} onGetStarted={handleGetStarted} />
   643→   643→  }
   644→   644→
   645→   645→  return <LandingPage onGetStarted={handleGetStarted} onBookDemo={handleBookDemo} />
   646→   646→}

   650→   650→}
