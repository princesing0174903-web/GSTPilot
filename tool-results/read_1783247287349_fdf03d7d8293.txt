     4→     4→// GSTPilot — Firestore Security Rules (Multi-Tenant Isolation)
     5→     5→//
     6→     6→// GOVERNANCE PRINCIPLE (PART 9):
     7→     7→//   Every tenant-scoped document carries an `organizationId` field. A user
     8→     8→//   may only read / write documents whose `organizationId` matches an
     9→     9→//   organization they are an active member of.
    10→    10→//
    11→    11→//   Membership is resolved by reading `organization_members/{orgId}_{uid}`.
    12→    12→//   This guarantees no user can ever see another organization's data — even
    13→    13→//   if client-side code has a bug.
    14→    14→//
    15→    15→// COLLECTIONS:
    16→    16→//   users                    — user profiles (self-owned)
    17→    17→//   organizations            — tenant roots (members only)
    18→    18→//   organization_members     — membership join (members only)
    19→    19→//   onboarding               — onboarding questionnaires (self-owned)
    20→    20→//   ── Tenant-scoped (all require matching organizationId) ──
    21→    21→//   clients, invoices, payments, expenses, bank_accounts, bank_transactions,
    22→    22→//   gst_profiles, gst_returns, notices, reports, tasks, ai_memory,
    23→    23→//   notifications, documents, returns, reconciliations, activities,
    24→    24→//   aiRecommendations, leads, deals, meetings, predictions, priorityQueue
    25→    25→// ═══════════════════════════════════════════════════════════════════════════════

    30→    30→    // ─── Helper Functions ─────────────────────────────────────────────────
    31→    31→
    32→    32→    function isSignedIn() {
    33→    33→      return request.auth != null;
    34→    34→    }
    35→    35→
    36→    36→    function uid() {
    37→    37→      return request.auth.uid;
    38→    38→    }
    39→    39→
    40→    40→    // The membership document id for a given org + user.
    41→    41→    function memberDocId(orgId) {
    42→    42→      return orgId + '_' + uid();
    43→    43→    }
    44→    44→
    45→    45→    // True if the user is an active member of the given org.
    46→    46→    function isOrgMember(orgId) {
    47→    47→      return isSignedIn()
    48→    48→        && exists(/databases/$(database)/documents/organization_members/$(memberDocId(orgId)))
    49→    49→        && get(/databases/$(database)/documents/organization_members/$(memberDocId(orgId)))
    50→    50→             .data.status == 'active';
    51→    51→    }
    52→    52→
    53→    53→    // The user's role within the given org (or empty string).
    54→    54→    function roleIn(orgId) {
    55→    55→      return get(/databases/$(database)/documents/organization_members/$(memberDocId(orgId)))
    56→    56→        .data.role;
    57→    57→    }
    58→    58→
    59→    59→    function isOwnerOrAdmin(orgId) {
    60→    60→      return isOrgMember(orgId)
    61→    61→        && roleIn(orgId) in ['owner', 'admin'];
    62→    62→    }
    63→    63→
    64→    64→    function canMutate(orgId) {
    65→    65→      return isOrgMember(orgId)
    66→    66→        && roleIn(orgId) in ['owner', 'admin', 'accountant', 'employee'];
    67→    67→    }
    68→    68→
    69→    69→    // True if the stored resource's organizationId matches an org the user
    70→    70→    // belongs to. Used for reads / updates / deletes.
    71→    71→    function resourceBelongsToUserOrg() {
    72→    72→      return isOrgMember(resource.data.organizationId);
    73→    73→    }
    74→    74→
    75→    75→    // True if the incoming write's organizationId matches an org the user
    76→    76→    // belongs to. Used on create / update.
    77→    77→    function writeScopedToUserOrg() {
    78→    78→      return isOrgMember(request.resource.data.organizationId);
    79→    79→    }
    80→    80→
    81→    81→    // True on update if the organizationId is unchanged (prevents moving a
    82→    82→    // document across tenants).
    83→    83→    function orgIdUnchanged() {
    84→    84→      return resource.data.organizationId == request.resource.data.organizationId;
    85→    85→    }
    86→    86→

   113→   113→

   123→   123→            || uid() == resource.data.userId);
