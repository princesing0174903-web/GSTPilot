   235→function daysUntil(dateStr: string): number {
   236→  const parts = dateStr.split('/')
   237→  const date = new Date(parseInt(parts[2]), parseInt(parts[1]) - 1, parseInt(parts[0]))
   238→  const now = new Date()
   239→  const diff = date.getTime() - now.getTime()
   240→  return Math.ceil(diff / (1000 * 60 * 60 * 24))
   241→}

   243→// Circular gauge SVG component
   244→function ComplianceGauge({ score, size = 80 }: { score: number; size?: number }) {
   245→  const strokeWidth = 6
   246→  const radius = (size - strokeWidth) / 2
   247→  const circumference = 2 * Math.PI * radius
   248→  const offset = circumference - (score / 100) * circumference
   249→  const color = getScoreStroke(score)
   250→
   251→  return (
   252→    <div className="relative" style={{ width: size, height: size }}>
   253→      <svg width={size} height={size} className="-rotate-90">
   254→        <circle
   255→          cx={size / 2}
   256→          cy={size / 2}
   257→          r={radius}
   258→          fill="none"
   259→          stroke="currentColor"
   260→          strokeWidth={strokeWidth}
   261→          className="text-slate-100 dark:text-slate-800"
   262→        />
   263→        <circle
   264→          cx={size / 2}
   265→          cy={size / 2}
   266→          r={radius}
   267→          fill="none"
   268→          stroke={color}
   269→          strokeWidth={strokeWidth}
   270→          strokeDasharray={circumference}
   271→          strokeDashoffset={offset}
   272→          strokeLinecap="round"
   273→          className="transition-all duration-1000 ease-out"
   274→        />
   275→      </svg>
   276→      <div className="absolute inset-0 flex items-center justify-center">
   277→        <span className={`text-lg font-bold ${getScoreColor(score)}`}>{score}</span>
   278→      </div>
   279→    </div>
   280→  )
   281→}
