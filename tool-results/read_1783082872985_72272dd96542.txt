   120→function mapFileTypeToReportFormat(fileType: string): ReportFormat {
   121→  if (fileType === 'json') return 'json';

   127→function parseFileSizeBytes(fileSize: string): number {
   128→  // Accept formats like "12.3 KB", "1.5 MB", or "PDF" (non-numeric → 0)
   129→  const match = fileSize.match(/([\d.]+)\s*(KB|MB|GB)?/i);
   130→  if (!match) return 0;
   131→  const num = parseFloat(match[1]);
   132→  const unit = (match[2] ?? '').toUpperCase();
   133→  if (unit === 'KB') return Math.round(num * 1024);
   134→  if (unit === 'MB') return Math.round(num * 1024 * 1024);
   135→  if (unit === 'GB') return Math.round(num * 1024 * 1024 * 1024);
   136→  return Math.round(num);
   137→}

   144→function loadHistory(): RecentExport[] {
   145→  if (typeof window === 'undefined') return [];
   146→  try {
   147→    const raw = window.localStorage.getItem(HISTORY_KEY);
   148→    if (!raw) return [];
   149→    const parsed = JSON.parse(raw);
   150→    if (!Array.isArray(parsed)) return [];
   151→    return parsed.slice(0, HISTORY_LIMIT);
   152→  } catch {
   153→    return [];
   154→  }
   155→}

   157→function saveHistory(items: RecentExport[]): void {
   158→  if (typeof window === 'undefined') return;
   159→  try {
   160→    window.localStorage.setItem(HISTORY_KEY, JSON.stringify(items.slice(0, HISTORY_LIMIT)));
   161→  } catch {
   162→    /* ignore quota errors */
   163→  }
   164→}

   179→function escapeHtml(s: string): string {
   180→  return String(s)
   181→    .replace(/&/g, '&amp;')
   182→    .replace(/</g, '&lt;')
   183→    .replace(/>/g, '&gt;')
   184→    .replace(/"/g, '&quot;')
   185→    .replace(/'/g, '&#39;');
   186→}

   202→            <tr><th>${escapeHtml(r.label)}</th><td>${escapeHtml(r.value)}</td></tr>

   281→function openPrintWindow(html: string): boolean {
   282→  if (typeof window === 'undefined') return false;
   283→  const printWindow = window.open('', '_blank', 'width=900,height=720');
   284→  if (!printWindow) {
   285→    // Pop-up blocked — fall back to writing into a hidden iframe
   286→    const iframe = document.createElement('iframe');
   287→    iframe.style.position = 'fixed';
   288→    iframe.style.right = '0';
   289→    iframe.style.bottom = '0';
   290→    iframe.style.width = '0';
   291→    iframe.style.height = '0';
   292→    iframe.style.border = '0';
   293→    document.body.appendChild(iframe);
   294→    const doc = iframe.contentWindow?.document;
   295→    if (!doc) {
   296→      document.body.removeChild(iframe);
   297→      return false;
   298→    }
   299→    doc.open();
   300→    doc.write(html);
   301→    doc.close();
   302→    iframe.contentWindow?.focus();
   303→    setTimeout(() => {
   304→      try { iframe.contentWindow?.print(); } catch { /* ignore */ }
   305→      setTimeout(() => document.body.removeChild(iframe), 1000);
   306→    }, 400);
   307→    return true;
   308→  }
   309→  printWindow.document.open();
   310→  printWindow.document.write(html);
   311→  printWindow.document.close();
   312→  printWindow.focus();
   313→  return true;
   314→}

   588→      const format = mapFileTypeToReportFormat(exp.fileType);

   598→          }
   599→        } catch { /* ignore encoding errors */ }

   602→        clientId: null,
   603→        clientTradeName: exp.clientName,
   604→        reportType,
   605→        format,
   606→        title: exp.exportType,
   607→        period: exp.period,
   608→        description: `Generated via ${exp.exportType} export`,
   609→        status: 'ready',
   610→        fileSize: parseFileSizeBytes(exp.fileSize),

   618→          fileType: exp.fileType,
