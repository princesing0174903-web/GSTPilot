   237→  // ─── Phase 3 — Agent State ──────────────────────────────────────────────

   258→
   259→  // ─── Phase 3 — Proactive Oracle ─────────────────────────────────────────
   260→  // Build the proactive queue from business memory. Whenever business memory

   262→  useEffect(() => {
   263→    const queue = buildProactiveNotifications(businessMemory);

   266→
   267→  // Pick the next proactive notification to surface (on mount + when queue changes).
   268→  useEffect(() => {
   269→    if (proactive) return; // Don't override an active notification.
   270→    const next = pickNextProactive(businessMemory);
   271→    if (next) {
   272→      // Delay slightly so it doesn't feel jarring on workspace open.
   273→      const timer = setTimeout(() => setProactive(next), 1500);
   274→      return () => clearTimeout(timer);
   275→    }
   276→  }, [businessMemory, proactive]);
   277→

   279→    // Dismiss by the notification title (stable key derived in the proactive engine).
   280→    dismissProactive(n.title);
   281→    setProactive(null);
   282→    // Pick the next one after a short delay.
   283→    setTimeout(() => {
   284→      const next = pickNextProactive(businessMemory);

   286→    }, 800);
   287→  }, [businessMemory]);
   288→
   289→  // ─── Streaming state ────────────────────────────────────────────────────
   290→  const [isStreaming, setIsStreaming] = useState(false);
   291→  const abortRef = useRef<AbortController | null>(null);
   292→  const scrollRef = useRef<HTMLDivElement>(null);
   293→
   294→  useEffect(() => {
   295→    const el = scrollRef.current;
   296→    if (!el) return;
   297→    el.scrollTo({ top: el.scrollHeight, behavior: 'smooth' });
   298→  }, [activeConv?.messages]);
   299→
   300→  // Sync agent status with streaming state.
   301→  useEffect(() => {
   302→    if (isStreaming) {
   303→      setAgentStatus('thinking');
   304→    } else if (agentStatus === 'thinking') {
   305→      // After streaming ends, briefly show 'completed' then return to 'idle'.
   306→      setAgentStatus('completed');
   307→      const timer = setTimeout(() => setAgentStatus('idle'), 2500);
   308→      return () => clearTimeout(timer);
   309→    }
   310→  }, [isStreaming, agentStatus]);
   311→
   312→  // ─── Send Message (orchestrates the full Oracle flow) ───────────────────
   313→  const sendMessage = useCallback(
   314→    async (text: string, forcedMode?: OracleModeId) => {
   315→      if (!activeConv || isStreaming) return;
   316→
   317→      const mode = forcedMode ?? detectOracleMode(text).id;
   318→
   319→      const priorMessages = activeConv.messages.map((m) => ({
   320→        role: (m.role === 'user' ? 'user' : 'oracle') as 'user' | 'oracle',
   321→        content: m.content,
   322→        createdAt: m.createdAt,
   323→      }));
   324→      const wm: WorkingMemory = buildWorkingMemory({
   325→        user: userMemory,
   326→        business: businessMemory,
   327→        priorMessages,
   328→        pinnedInsights,
   329→      });
   330→
   331→      const investigationHints = detectInvestigation(text, businessMemory);
   332→
   333→      const userMsg: OracleMessage = {
   334→        id: newMessageId(),
   335→        role: 'user',
   336→        content: text,
   337→        createdAt: new Date().toISOString(),
   338→        mode,
   339→      };
   340→      const oracleMsg: OracleMessage = {
   341→        id: newMessageId(),
   342→        role: 'oracle',
   343→        content: '',
   344→        createdAt: new Date().toISOString(),
   345→        mode,
   346→        parts: { ...EMPTY_PARTS },
   347→        phase: 'thinking',
   348→      };
   349→
   350→      const convWithUser = appendMessage(activeConv, userMsg);
   351→      const convWithOracle = appendMessage(convWithUser, oracleMsg);
   352→      setActiveConv(convWithOracle);
   353→      refreshConversationList();
   354→      setIsStreaming(true);
   355→
   356→      const controller = new AbortController();
   357→      abortRef.current = controller;
   358→
   359→      try {
   360→        const res = await fetch('/api/oracle/chat', {
   361→          method: 'POST',
   362→          headers: { 'Content-Type': 'application/json' },
   363→          body: JSON.stringify({
   364→            question: text,
   365→            mode,
   366→            workingMemory: wm,
   367→            history: priorMessages,
   368→            investigationHints: investigationHints.length > 0 ? investigationHints : undefined,
   369→          }),
   370→          signal: controller.signal,
   371→        });
   372→
   373→        if (!res.ok || !res.body) {
   374→          throw new Error(`Oracle request failed: ${res.status}`);
   375→        }
   376→
   377→        const reader = res.body.getReader();
   378→        const decoder = new TextDecoder();
   379→        let buffer = '';
   380→        let currentParts: OracleResponseParts = { ...EMPTY_PARTS };
   381→        let finalMode: OracleModeId = mode;
   382→
   383→        const patch = (p: Partial<OracleMessage>) => {
   384→          setActiveConv((prev) => {
   385→            if (!prev) return prev;
   386→            return updateLastOracleMessage(prev, p);
   387→          });
   388→        };
   389→
   390→        const flushBuffer = () => {
   391→          const events = buffer.split('\n\n');
   392→          buffer = events.pop() ?? '';
   393→          for (const evt of events) {
   394→            const lines = evt.split('\n');
   395→            let eventName = '';
   396→            let dataStr = '';
   397→            for (const line of lines) {
   398→              if (line.startsWith('event: ')) eventName = line.slice(7).trim();
   399→              else if (line.startsWith('data: ')) dataStr = line.slice(6);
   400→            }
   401→            if (!eventName || !dataStr) continue;
   402→            try {
   403→              const data = JSON.parse(dataStr);
   404→              handleEvent(eventName, data);
   405→            } catch {
   406→              // Skip malformed event.
   407→            }
   408→          }
   409→        };
   410→
   411→        const handleEvent = (eventName: string, data: unknown) => {
   412→          const d = data as Record<string, unknown>;
   413→          if (eventName === 'phase') {
   414→            const phase = d.phase as OraclePhase;
   415→            patch({ phase });
   416→          } else if (eventName === 'part') {
   417→            const part = d.part as string;
   418→            const textChunk = (d.text as string) ?? '';
   419→            const isItem = d.item === true;
   420→            const source = d.source as { kind: string; title: string; snippet: string } | undefined;
   421→            const card = d.card as { kind: string; title: string; description: string; impact?: string } | undefined;
   422→
   423→            if (part === 'keyInsight') {
   424→              currentParts.keyInsight += textChunk;
   425→            } else if (part === 'analysis') {
   426→              currentParts.analysis += textChunk;
   427→            } else if (part === 'recommendedActions') {
   428→              if (isItem) currentParts.recommendedActions.push(textChunk);
   429→              else {
   430→                const last = currentParts.recommendedActions.length - 1;
   431→                if (last >= 0) currentParts.recommendedActions[last] += textChunk;
   432→                else currentParts.recommendedActions.push(textChunk);
   433→              }
   434→            } else if (part === 'potentialRisks') {
   435→              if (isItem) currentParts.potentialRisks.push(textChunk);
   436→              else {
   437→                const last = currentParts.potentialRisks.length - 1;
   438→                if (last >= 0) currentParts.potentialRisks[last] += textChunk;
   439→                else currentParts.potentialRisks.push(textChunk);
   440→              }
   441→            } else if (part === 'nextBestStep') {
   442→              currentParts.nextBestStep += textChunk;
   443→            } else if (part === 'investigation') {
   444→              if (!currentParts.investigation) currentParts.investigation = [];
   445→              if (isItem) currentParts.investigation.push(textChunk);
   446→              else {
   447→                const last = currentParts.investigation.length - 1;
   448→                if (last >= 0) currentParts.investigation[last] += textChunk;
   449→                else currentParts.investigation.push(textChunk);
   450→              }
   451→            } else if (part === 'sources') {
   452→              if (source) {
   453→                currentParts.sources.push({
   454→                  kind: (source.kind as OracleResponseParts['sources'][number]['kind']) ?? 'business-data',
   455→                  title: source.title,
   456→                  snippet: source.snippet,
   457→                });
   458→              }
   459→            } else if (part === 'followUps') {
   460→              if (isItem) currentParts.followUps.push(textChunk);
   461→              else {
   462→                const last = currentParts.followUps.length - 1;
   463→                if (last >= 0) currentParts.followUps[last] += textChunk;
   464→                else currentParts.followUps.push(textChunk);
   465→              }
   466→            } else if (part === 'actionCards') {
   467→              if (card) {
   468→                const kind = (card.kind as OracleActionKind) ?? 'create-report';
   469→                currentParts.actionCards.push({
   470→                  kind,
   471→                  title: card.title,
   472→                  description: card.description,
   473→                  impact: card.impact,
   474→                });
   475→              }
   476→            }
   477→
   478→            patch({ parts: { ...currentParts }, phase: 'streaming' });
   479→          } else if (eventName === 'done') {
   480→            finalMode = (d.mode as OracleModeId) ?? mode;
   481→            // If the LLM didn't propose any action cards, fall back to defaults
   482→            // so the "always 3 cards" contract holds.
   483→            if (currentParts.actionCards.length === 0) {
   484→              currentParts.actionCards = DEFAULT_FALLBACK_CARDS;
   485→            }
   486→            patch({
   487→              parts: { ...currentParts },
   488→              phase: 'done',
   489→              mode: finalMode,
   490→              actions: currentParts.recommendedActions.map((s) => mapActionKind(s)),
   491→            });
   492→          } else if (eventName === 'error') {
   493→            patch({
   494→              phase: 'error',
   495→              error: (d.message as string) ?? 'Unknown error',
   496→            });
   497→          }
   498→        };
   499→
   500→        while (true) {
   501→          const { done, value } = await reader.read();
   502→          if (done) break;
   503→          buffer += decoder.decode(value, { stream: true });
   504→          flushBuffer();
   505→        }
   506→        flushBuffer();
   507→      } catch (err) {
   508→        if ((err as Error).name === 'AbortError') {
   509→          setActiveConv((prev) =>
   510→            prev ? updateLastOracleMessage(prev, { phase: 'done' }) : prev,
   511→          );
   512→        } else {
   513→          setActiveConv((prev) =>
   514→            prev
   515→              ? updateLastOracleMessage(prev, {
   516→                  phase: 'error',
   517→                  error: err instanceof Error ? err.message : 'Unknown error',
   518→                })
   519→              : prev,
   520→          );
   521→        }
   522→      } finally {
   523→        setIsStreaming(false);
   524→        abortRef.current = null;
   525→        refreshConversationList();
   526→      }
   527→    },
   528→    [activeConv, isStreaming, userMemory, businessMemory, pinnedInsights, refreshConversationList],
   529→  );
   530→
   531→  const handleStop = useCallback(() => {
   532→    abortRef.current?.abort();
   533→  }, []);
   534→
   535→  const handleExit = useCallback(() => {
   536→    if (onExit) onExit();
   537→    else setCurrentView('dashboard' as AppView);
   538→  }, [onExit, setCurrentView]);
   539→
   540→  // ─── Phase 3 — Action Engine (spawns + executes Oracle Tasks) ───────────
   541→  const handleRunAction = useCallback(
   542→    (card: OracleActionCard) => {
   543→      const meta = ACTION_LIBRARY[card.kind];
   544→      if (!meta) return;
   545→
   546→      // Don't allow spawning a second concurrent task (keeps the UX clean).
   547→      if (tasks.some((t) => t.status === 'running')) {
   548→        // Queue it as scheduled instead.
   549→        const scheduledFor = new Date(Date.now() + 5 * 60 * 1000).toISOString();
   550→        const task = createTask(card.kind, card.title, card.description, {
   551→          scheduledFor,
   552→          resultView: meta.resultView,
   553→        });
   554→        setTasks((prev) => [task, ...prev]);
   555→        return;
   556→      }
   557→
   558→      // Spawn the task (status: 'running').
   559→      const task = createTask(card.kind, card.title, card.description, {
   560→        resultView: meta.resultView,
   561→      });
   562→      setTasks((prev) => [task, ...prev]);
   563→      setAgentStatus('working');
   564→      setWorkingTitle(card.title);
   565→
   566→      // Simulate execution: step through the steps, increment progress.
   567→      const totalSteps = task.steps.length;
   568→      const stepDuration = Math.max(400, Math.round(meta.durationMs / Math.max(totalSteps, 1)));
   569→      let currentStepIdx = 0;
   570→      let progress = 0;
   571→
   572→      const tick = () => {
   573→        // Advance progress within the current step.
   574→        const stepIncrement = Math.round(100 / Math.max(totalSteps, 1));
   575→        progress = Math.min(100, progress + Math.max(8, Math.round(stepIncrement / 3)));
   576→
   577→        // If we've crossed into the next step boundary, advance the step.
   578→        const expectedStep = Math.min(totalSteps - 1, Math.floor((progress / 100) * totalSteps));
   579→        if (expectedStep > currentStepIdx) {
   580→          // Mark previous steps as done, new one as active.
   581→          const newSteps = task.steps.map((s, i) => ({
   582→            ...s,
   583→            state: (i < expectedStep ? 'done' : i === expectedStep ? 'active' : 'pending') as
   584→              | 'done'
   585→              | 'active'
   586→              | 'pending',
   587→          }));
   588→          currentStepIdx = expectedStep;
   589→          updateTask(task.id, { steps: newSteps, progress });
   590→          setTasks((prev) =>
   591→            prev.map((t) => (t.id === task.id ? { ...t, steps: newSteps, progress } : t)),
   592→          );
   593→        } else {
   594→          updateTask(task.id, { progress });
   595→          setTasks((prev) => prev.map((t) => (t.id === task.id ? { ...t, progress } : t)));
   596→        }
   597→
   598→        if (progress >= 100) {
   599→          // Complete the task.
   600→          const result = formatActionResult(card.kind, {
   601→            count: Math.floor(Math.random() * 5) + 2,
   602→            amount: Math.floor(Math.random() * 50 + 10) * 10000,
   603→          });
   604→          const finalSteps = task.steps.map((s) => ({ ...s, state: 'done' as const }));
   605→          updateTask(task.id, {
   606→            status: 'completed',
   607→            progress: 100,
   608→            steps: finalSteps,
   609→            result,
   610→          });
   611→          setTasks((prev) =>
   612→            prev.map((t) =>
   613→              t.id === task.id
   614→                ? { ...t, status: 'completed', progress: 100, steps: finalSteps, result }
   615→                : t,
   616→            ),
   617→          );
   618→          setAgentStatus('completed');
   619→          setWorkingTitle(undefined);
   620→          // Return to idle after a brief celebration.
   621→          setTimeout(() => setAgentStatus('idle'), 3000);
   622→          return;
   623→        }
   624→
   625→        // Schedule the next tick.
   626→        setTimeout(tick, stepDuration / 3);
   627→      };
   628→
   629→      // Kick off the execution loop.
   630→      setTimeout(tick, stepDuration / 3);
   631→    },
   632→    [tasks],
   633→  );
   634→
   635→  // Stable ref to handleRunAction so handleActProactive doesn't need to re-bind.
   636→  const runActionRef = useRef(handleRunAction);
   637→  useEffect(() => {
   638→    runActionRef.current = handleRunAction;
   639→  }, [handleRunAction]);
   640→
   641→  // Re-bind handleActProactive to always use the latest ref.
   642→  const handleActProactiveBound = useCallback(
   643→    (card: OracleActionCard, n: ProactiveNotification) => {

   648→  );
   649→
   650→  // ─── Task panel handlers ────────────────────────────────────────────────
   651→  const handleClearTasks = useCallback(() => {

   669→      const actionKind = kind as OracleActionKind;
   670→      if (ACTION_LIBRARY[actionKind]) {

   682→        'generate-gst-return': 'returns',
   683→        'recover-collections': 'reconcile',
   684→        'open-reconcile': 'reconcile',
   685→        'open-returns': 'returns',
   686→        'open-invoices': 'invoices',
   687→        'open-clients': 'clients',
   688→        'create-reminder': 'autopilot',
   689→        'open-analytics': 'business-dna',
   690→        'generate-forecast': 'ai-predictions',
   691→        'export-pdf': 'business-dna',
   692→        'create-checklist': 'autopilot',
   693→      };
   694→      const view = viewMap[kind] ?? 'dashboard';
   695→      void label;
   696→      setCurrentView(view);
   697→    },
   698→    [setCurrentView, handleRunAction],
   699→  );
   700→
   701→  const handleNextBestStep = useCallback(
   702→    (step: string) => {
   703→      handleAction(mapActionKind(step), step);
   704→    },
   705→    [handleAction],
   706→  );
   707→
   708→  const handleInsightClick = useCallback(
   709→    (_key: string, insight: { label: string; detail: string }) => {
   710→      const prompt = `Tell me more about "${insight.label}" — ${insight.detail}`;
   711→      sendMessage(prompt);
   712→    },
   713→    [sendMessage],
   714→  );
   715→
   716→  const activeMode = activeConv?.messages
   717→    .slice()
   718→    .reverse()
   719→    .find((m) => m.role === 'oracle' && m.mode)?.mode;
   720→
   721→  const modeMeta = activeMode ? ORACLE_MODES[activeMode] : null;
   722→
   723→  // ─── Render ─────────────────────────────────────────────────────────────
   724→  const hasMessages = (activeConv?.messages.length ?? 0) > 0;
   725→  const hasRunningTask = tasks.some((t) => t.status === 'running');
   726→
   727→  return (
   728→    <motion.div
   729→      initial={{ opacity: 0 }}
   730→      animate={{ opacity: 1 }}
   731→      transition={{ duration: 0.4 }}
   732→      className="flex h-full w-full gap-3 p-3"
   733→    >
   734→      {/* ─── LEFT SIDEBAR (Memory + History) ──────────────────────────────── */}
   735→      <div className="hidden w-[280px] shrink-0 lg:block">
   736→        <OracleSidebar
   737→          conversations={conversations}
   738→          activeConversationId={activeId}
   739→          userMemory={userMemory}
   740→          businessMemory={businessMemory}
   741→          pinnedInsights={pinnedInsights}
   742→          onSelect={handleSelect}
   743→          onNew={handleNewChat}
   744→          onDelete={handleDelete}
   745→          onChanged={refreshConversationList}
   746→        />
   747→      </div>
   748→
   749→      {/* ─── MAIN WORKSPACE ──────────────────────────────────────────────── */}
   750→      <div className="glass-surface relative flex min-w-0 flex-1 flex-col rounded-3xl">
   751→        {/* ─── Header ─────────────────────────────────────────────────────── */}
   752→        <header className="flex h-14 shrink-0 items-center gap-3 border-b border-border px-4">
   753→          <button
   754→            onClick={handleExit}
   755→            aria-label="Back to dashboard"
   756→            className="flex h-8 w-8 items-center justify-center rounded-lg text-muted-foreground transition-colors hover:bg-card/[0.5] hover:text-foreground"
   757→          >
   758→            <ArrowLeft className="h-4 w-4" />
   759→          </button>
   760→
   761→          <div className="flex h-8 w-8 items-center justify-center rounded-lg accent-gradient shadow-lg shadow-emerald-500/20">
   762→            <InfinitySymbol size={18} />

   766→            <div className="flex items-baseline gap-1">
   767→              <span className="text-sm font-semibold text-foreground">GSTPilot Oracle</span>
   768→              <sup className="text-[9px] font-medium text-muted-foreground">™</sup>

   771→              <span className="relative flex h-1.5 w-1.5">
   772→                <span className="absolute inline-flex h-full w-full animate-ping rounded-full bg-emerald-400 opacity-75" />

   780→
   781→          {/* Active Mode Indicator */}
   782→          {modeMeta && (
   783→            <div
   784→              className="hidden items-center gap-1.5 rounded-full px-2.5 py-1 text-[10px] font-semibold sm:flex"
   785→              style={{ backgroundColor: modeMeta.tint, color: modeMeta.color }}
   786→            >
   787→              <span
   788→                className="h-1.5 w-1.5 rounded-full"
   789→                style={{ backgroundColor: modeMeta.color }}
   790→              />
   791→              {modeMeta.name}
   792→            </div>

   794→
   795→          {/* Proactive Bell — shows when there are undismissed proactive signals */}

   804→          )}
   805→

   807→          <button
   808→            onClick={() => setAgentPanelOpen((v) => !v)}
   809→            aria-label={agentPanelOpen ? 'Hide Oracle Tasks' : 'Show Oracle Tasks'}
   810→            title={agentPanelOpen ? 'Hide Oracle Tasks' : 'Show Oracle Tasks'}
   811→            className={cn(
   812→              'flex h-8 w-8 items-center justify-center rounded-lg transition-colors',
   813→              agentPanelOpen
   814→                ? 'accent-text bg-card/[0.5]'
   815→                : 'text-muted-foreground hover:bg-card/[0.5] hover:text-foreground',
   816→            )}
   817→          >
   818→            {agentPanelOpen ? (
   819→              <PanelRightClose className="h-4 w-4" />
   820→            ) : (
   821→              <PanelRightOpen className="h-4 w-4" />
   822→            )}
   823→          </button>

   826→          <button
   827→            onClick={handleNewChat}
   828→            aria-label="New chat"
   829→            className="flex h-8 items-center gap-1.5 rounded-lg border border-border bg-card/[0.4] px-2.5 text-xs text-muted-foreground transition-colors hover:bg-card/[0.6] hover:text-foreground lg:hidden"
   830→          >
   831→            <Sparkles className="h-3.5 w-3.5" />
   832→            New
   833→          </button>
   834→        </header>
   835→
   836→        {/* ─── Status Bar (Phase 3) ──────────────────────────────────────── */}
   837→        <div className="shrink-0 border-b border-border bg-background/30 px-3 py-1.5 backdrop-blur-xl">
   838→          <OracleStatusBar
   839→            status={agentStatus}
   840→            workingTitle={workingTitle}

   890→
   891→        {/* ─── Insights Strip (live, always visible above input) ──────────── */}
   892→        <div className="shrink-0 border-t border-border bg-background/30 px-4 py-2 backdrop-blur-xl">
   893→          <div className="mx-auto w-full max-w-3xl">
   894→            <div className="mb-1.5 flex items-center gap-1.5">
   895→              <Sparkles className="h-2.5 w-2.5 accent-text" />

   897→                Live Insights · Updated from your business memory

   899→            </div>
   900→            <OracleInsightsPanel
   901→              insights={insights}
   902→              variant="strip"
   903→              onInsightClick={(key, insight) => handleInsightClick(key, insight)}
   904→            />
   905→          </div>
   906→        </div>
   907→
   908→        {/* ─── Input Bar (sticky bottom) ──────────────────────────────────── */}
   909→        <div className="shrink-0 border-t border-border bg-background/40 p-4 backdrop-blur-xl">
   910→          <OracleInputBar
   911→            onSubmit={(text) => sendMessage(text)}
   912→            disabled={isStreaming}

   914→            onStop={handleStop}
   915→          />
   916→        </div>
   917→
   918→        {/* ─── Phase 3 — Proactive Notifications (overlay, top-right) ─────── */}
   919→        <OracleNotifications
   920→          notification={proactive}
   921→          onDismiss={handleDismissProactive}

   923→        />
   924→      </div>
   925→
   926→      {/* ─── RIGHT AGENT PANEL (Phase 3 — Oracle Tasks) ──────────────────── */}

   930→            key="agent-panel"
   931→            initial={{ opacity: 0, width: 0, x: 20 }}

   933→            exit={{ opacity: 0, width: 0, x: 20 }}

   935→            className="hidden shrink-0 overflow-hidden xl:block"
   936→          >
   937→            <OracleAgentPanel
   938→              tasks={tasks}
   939→              onClear={handleClearTasks}
   940→              onTaskClick={handleTaskClick}
   941→            />
   942→          </motion.div>
   943→        )}
   944→      </AnimatePresence>
   945→    </motion.div>
   946→  );
   947→}

   951→function mapActionKind(label: string): OracleActionKind {
   952→  const l = label.toLowerCase();
   953→  if (l.includes('file') || l.includes('return')) return 'generate-gst-return';
   954→  if (l.includes('recover') || l.includes('collection')) return 'recover-collections';
   955→  if (l.includes('reconcil') || l.includes('2b')) return 'prepare-reconciliation';
   956→  if (l.includes('cash') && l.includes('flow')) return 'cash-flow-forecast';
   957→  if (l.includes('forecast') || l.includes('predict')) return 'generate-forecast';
   958→  if (l.includes('notice') || l.includes('scn')) return 'create-notice';
   959→  if (l.includes('report')) return 'create-report';
   960→  if (l.includes('reminder') || l.includes('checklist')) return 'create-checklist';
   961→  if (l.includes('export') || l.includes('pdf')) return 'export-pdf';
   962→  return 'create-report';
   963→}
