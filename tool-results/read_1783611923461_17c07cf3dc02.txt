    57→function formatINR(n: number): string {
    58→  return new Intl.NumberFormat('en-IN', { style: 'currency', currency: 'INR', maximumFractionDigits: 0 }).format(n)
    59→}
    60→
    61→function formatDate(d: string | unknown): string {
    62→  if (!d) return '—'
    63→  try {
    64→    const date = typeof d === 'string' ? new Date(d) : new Date()
    65→    return date.toLocaleDateString('en-IN', { day: 'numeric', month: 'short', year: 'numeric' })
    66→  } catch { return '—' }
    67→}
    68→
    69→function formatTime(d: string | unknown): string {
    70→  if (!d) return ''
    71→  try {
    72→    const date = typeof d === 'string' ? new Date(d) : new Date()
    73→    return date.toLocaleTimeString('en-IN', { hour: '2-digit', minute: '2-digit' })
    74→  } catch { return '' }
    75→}
    76→
    77→function daysUntil(d: string | unknown): number {
    78→  if (!d) return 999
    79→  try {
    80→    const date = typeof d === 'string' ? new Date(d) : new Date()
    81→    return Math.ceil((date.getTime() - Date.now()) / (1000 * 60 * 60 * 24))
    82→  } catch { return 999 }
    83→}

    85→function scoreColor(score: number): string {
    86→  if (score >= 70) return 'text-emerald-600 bg-emerald-50 border-emerald-200'
    87→  if (score >= 40) return 'text-amber-600 bg-amber-50 border-amber-200'
    88→  return 'text-red-600 bg-red-50 border-red-200'
    89→}
    90→
    91→function scoreBgGradient(score: number): string {
    92→  if (score >= 70) return 'from-emerald-500 to-emerald-600'
    93→  if (score >= 40) return 'from-amber-500 to-amber-600'
    94→  return 'from-red-500 to-red-600'
    95→}

   259→function KanbanLeadCard({ lead, onStatusChange, onConvert, onDelete }: {
   260→  lead: FirestoreLead & { id: string }

   268→
   269→  return (
   270→    <motion.div

   276→      onDragStart={() => setIsDragging(true)}

   280→        shadow-sm hover:shadow-md transition-all duration-200
   281→        ${isDragging ? 'ring-2 ring-emerald-400 ring-offset-2 shadow-lg scale-[1.02]' : ''}

   298→                aria-label={`Delete lead ${lead.contactName}`}
   299→                onClick={(e) => e.stopPropagation()}
   300→              >
   301→                <Trash2 className="h-3 w-3" />
   302→              </Button>
   303→            </AlertDialogTrigger>
   304→            <AlertDialogContent>
   305→              <AlertDialogHeader>
   306→                <AlertDialogTitle>Delete lead?</AlertDialogTitle>
   307→                <AlertDialogDescription>

   320→            </AlertDialogContent>

   323→            {lead.leadScore}
   324→          </Badge>
   325→        </div>
   326→      </div>
   327→
   328→      <div className="flex items-center gap-3 mb-2.5">

   340→            {tag}
   341→          </span>
   342→        ))}

   348→      <div className="flex items-center justify-between">
   349→        <div className="flex items-center gap-1.5">
   350→          <Avatar className="h-5 w-5">
   351→            <AvatarFallback className="text-[8px] bg-emerald-100 text-emerald-700">
   352→              {lead.assignedTo?.split(' ').map(w => w[0]).join('') || '?'}
   353→            </AvatarFallback>
   354→          </Avatar>
   355→          <span className="text-[10px] text-muted-foreground truncate max-w-[80px]">{lead.assignedTo}</span>
   356→        </div>
   357→
   358→        {lead.nextFollowUp && (
   359→          <span className={`text-[10px] font-medium ${days <= 1 ? 'text-red-600' : days <= 3 ? 'text-amber-600' : 'text-muted-foreground'}`}>
   360→            {days <= 0 ? 'Overdue' : days === 1 ? 'Tomorrow' : `${days}d`}
   361→          </span>
   362→        )}
   363→      </div>
   364→
   365→      {lead.status !== 'converted' && lead.status !== 'lost' && (
   366→        <div className="mt-2 pt-2 border-t border-border/40 flex items-center gap-1.5 opacity-0 group-hover:opacity-100 transition-opacity">
   367→          {lead.status === 'negotiation' && (
   368→            <Button
   369→              variant="ghost"
   370→              size="sm"
   371→              className="h-6 text-[10px] px-2 text-emerald-600 hover:text-emerald-700 hover:bg-emerald-50"
   372→              onClick={() => onConvert(lead.leadId)}
   373→            >
   374→              <UserPlus className="h-3 w-3 mr-1" /> Convert
   375→            </Button>
   376→          )}
   377→          {PIPELINE_COLUMNS.findIndex(c => c.status === lead.status) < PIPELINE_COLUMNS.length - 2 && (
   378→            <Button
   379→              variant="ghost"
   380→              size="sm"
   381→              className="h-6 text-[10px] px-2 text-blue-600 hover:text-blue-700 hover:bg-blue-50"
   382→              onClick={() => {
   383→                const nextStatuses = PIPELINE_COLUMNS.map(c => c.status)
   384→                const currentIdx = nextStatuses.indexOf(lead.status)
   385→                if (currentIdx < nextStatuses.length - 2) {
   386→                  onStatusChange(lead.leadId, nextStatuses[currentIdx + 1] as LeadStatus)
   387→                }
   388→              }}
   389→            >
   390→              Next <ChevronRight className="h-3 w-3 ml-0.5" />
   391→            </Button>
   392→          )}
   393→        </div>
   394→      )}
   395→    </motion.div>
   396→  )
   397→}

   575→  const [sortDir, setSortDir] = useState<'asc' | 'desc'>('desc')

   578→  const [deleteDealId, setDeleteDealId] = useState<string | null>(null)
   579→
   580→  const sortedDeals = useMemo(() => {
   581→    return [...deals].sort((a, b) => {
   582→      const dir = sortDir === 'asc' ? 1 : -1
   583→      if (sortField === 'value' || sortField === 'probability') {
   584→        return ((a[sortField] as number) - (b[sortField] as number)) * dir
   585→      }
   586→      const aDate = typeof a.expectedCloseDate === 'string' ? new Date(a.expectedCloseDate).getTime() : 0
   587→      const bDate = typeof b.expectedCloseDate === 'string' ? new Date(b.expectedCloseDate).getTime() : 0
   588→      return (aDate - bDate) * dir
   589→    })
   590→  }, [deals, sortField, sortDir])
   591→
   592→  const handleSort = (field: typeof sortField) => {
   593→    if (sortField === field) setSortDir(d => d === 'asc' ? 'desc' : 'asc')
   594→    else { setSortField(field); setSortDir('desc') }
   595→  }
   596→

   599→      await createDeal({

   603→        stage: newDeal.stage,
   604→        probability: Number(newDeal.probability) || 30,
   605→        expectedCloseDate: newDeal.expectedCloseDate ? new Date(newDeal.expectedCloseDate) : null,

  1270→function RevenueForecastView({ leads, deals, loading }: { leads: Array<FirestoreLead & { id: string }>; deals: Array<FirestoreDeal & { id: string }>; loading: boolean }) {
  1271→  const stageValues = useMemo(() => {
  1272→    const stages: Record<string, { label: string; value: number; color: string }> = {
  1273→      new: { label: 'New', value: 0, color: '#94a3b8' },
  1274→      contacted: { label: 'Contacted', value: 0, color: '#3b82f6' },
  1275→      qualified: { label: 'Qualified', value: 0, color: '#8b5cf6' },

  1277→      negotiation: { label: 'Negotiation', value: 0, color: '#f97316' },
  1278→    }
  1279→    leads.filter(l => l.status !== 'converted' && l.status !== 'lost').forEach(lead => {
  1280→      if (stages[lead.status]) stages[lead.status].value += lead.estimatedValue
  1281→    })
  1282→    return Object.values(stages)
  1283→  }, [leads])
  1284→
  1285→  const maxBarValue = useMemo(() => Math.max(...stageValues.map(s => s.value), 1), [stageValues])
  1286→
  1287→  const weightedPipeline = useMemo(() => {
  1288→    return deals.filter(d => d.stage !== 'closed_lost').reduce((sum, d) => sum + d.value * d.probability / 100, 0)
  1289→  }, [deals])
  1290→
  1291→  const totalPipeline = useMemo(() => {
  1292→    return deals.filter(d => d.stage !== 'closed_lost').reduce((sum, d) => sum + d.value, 0)
  1293→  }, [deals])
  1294→
  1295→  const monthlyTrend = useMemo(() => {
  1296→    // Compute from real deals — no Math.random fabrication.
  1297→    // Returns empty array when no deals exist (honest empty state).
  1298→    if (!deals || deals.length === 0) return []
  1299→    const months = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun']
  1300→    return months.map(m => {
  1301→      const monthDeals = deals.filter((d: any) => {
  1302→        const dDate = new Date(d.closeDate ?? d.createdAt ?? 0)
  1303→        return dDate.getMonth() === months.indexOf(m) && (d.status === 'won' || d.status === 'closed')
  1304→      })
