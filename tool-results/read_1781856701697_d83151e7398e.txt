   280→function loadThread(): ChatMessage[] {
   281→  if (typeof window === 'undefined') return [];
   282→  try {
   283→    const raw = window.localStorage.getItem(THREAD_STORAGE_KEY);
   284→    if (!raw) return [];
   285→    const parsed = JSON.parse(raw);
   286→    if (!Array.isArray(parsed)) return [];
   287→    return parsed.filter(
   288→      (m) => m && typeof m.id === 'string' && typeof m.content === 'string',
   289→    );
   290→  } catch {
   291→    return [];
   292→  }
   293→}

   296→  if (typeof window === 'undefined') return;
   297→  try {
   298→    // Keep only the last 40 messages to avoid storage bloat
   299→    const trimmed = messages.slice(-40);
   300→    window.localStorage.setItem(THREAD_STORAGE_KEY, JSON.stringify(trimmed));
   301→  } catch {
   302→    // localStorage may be full or disabled — silently ignore
   303→  }
   304→}

   358→function formatDate(ts: number | string | undefined): string {
   359→  try {
   360→    const d = typeof ts === 'number' ? new Date(ts) : new Date(ts as string);
   361→    if (isNaN(d.getTime())) return '—';
   362→    const dd = String(d.getDate()).padStart(2, '0');
   363→    const mm = String(d.getMonth() + 1).padStart(2, '0');
   364→    const yyyy = d.getFullYear();
   365→    return `${dd}/${mm}/${yyyy}`;
   366→  } catch {
   367→    return '—';
   368→  }
   369→}

   371→function formatTime(ts: number): string {
   372→  const d = new Date(ts);
   373→  let h = d.getHours();
   374→  const m = String(d.getMinutes()).padStart(2, '0');
   375→  const ampm = h >= 12 ? 'PM' : 'AM';
   376→  h = h % 12 || 12;
   377→  return `${h}:${m} ${ampm}`;
   378→}

   384→// ─── Live Data Context Builder ────────────────────────────────────────────────
   385→// Safely composes a context string from live Firestore hooks. Never throws.
   386→
   387→function buildLiveDataContext(opts: {
   388→  clients: Array<Record<string, unknown>>;
   389→  invoices: Array<Record<string, unknown>>;
   390→  returns: Array<Record<string, unknown>>;
   391→  notifications: Array<Record<string, unknown>>;
   392→  activities: Array<Record<string, unknown>>;
   393→}): string {
   394→  try {
   395→    const { clients, invoices, returns, notifications, activities } = opts;
   396→    const now = new Date();
   397→    const dd = String(now.getDate()).padStart(2, '0');
   398→    const mm = String(now.getMonth() + 1).padStart(2, '0');
   399→    const yyyy = now.getFullYear();
   400→    const hh = String(now.getHours()).padStart(2, '0');
   401→    const min = String(now.getMinutes()).padStart(2, '0');
   402→    const stamp = `${dd}/${mm}/${yyyy} ${hh}:${min}`;
   403→
   404→    const lines: string[] = [`LIVE DATA SNAPSHOT (${stamp}):`];
   405→
   406→    // Clients
   407→    const totalClients = clients.length;
   408→    const activeClients = clients.filter((c) => c.status === 'active').length;
   409→    lines.push(`- Total clients: ${totalClients}`);
   410→    lines.push(`- Active clients: ${activeClients}`);
   411→
   412→    // Invoices
   413→    const activeInvoices = invoices.filter(
   414→      (i) => i.status !== 'cancelled' && i.status !== 'draft',
   415→    );
   416→    const totalInvoiceValue = activeInvoices.reduce((s, i) => {
   417→      const v =
   418→        (i.totalAmount as number) ||
   419→        (i.grandTotal as number) ||
   420→        (i.total as number) ||
   421→        ((i.subTotal as number) || 0) + ((i.totalTax as number) || 0);
   422→      return s + (typeof v === 'number' ? v : 0);
   423→    }, 0);
   424→    const overdueInvoices = activeInvoices.filter((i) => {
   425→      const status = i.status as string;
   426→      return status === 'overdue' || status === 'unpaid';
   427→    });
   428→    const overdueValue = overdueInvoices.reduce((s, i) => {
   429→      const v =
   430→        (i.totalAmount as number) ||
   431→        (i.grandTotal as number) ||
   432→        (i.total as number) ||
   433→        0;
   434→      return s + (typeof v === 'number' ? v : 0);
   435→    }, 0);
   436→    lines.push(
   437→      `- Active invoices: ${activeInvoices.length} (${formatINR(totalInvoiceValue)} total)`,
   438→    );
   439→    lines.push(
   440→      `- Overdue invoices: ${overdueInvoices.length} (${formatINR(overdueValue)})`,
   441→    );
   442→
   443→    // Returns
   444→    const gstr1Pending = returns.filter(
   445→      (r) => (r.returnType as string) === 'GSTR-1' && (r.status as string) !== 'filed',
   446→    ).length;
   447→    const gstr3bPending = returns.filter(
   448→      (r) => (r.returnType as string) === 'GSTR-3B' && (r.status as string) !== 'filed',
   449→    ).length;
   450→    const filedReturns = returns.filter((r) => r.status === 'filed').length;
   451→    lines.push(
   452→      `- Pending returns: ${gstr1Pending} GSTR-1, ${gstr3bPending} GSTR-3B`,
   453→    );
   454→    lines.push(`- Filed returns: ${filedReturns}`);
   455→
   456→    // Notifications + Activities
   457→    const unread = notifications.filter((n) => n.read === false).length;
   458→    lines.push(`- Unread notifications: ${unread}`);
   459→    lines.push(`- Recent activities: ${activities.length} in last 24h`);
   460→
   461→    // Top 5 clients by revenue
   462→    const clientRevenue = new Map<string, number>();
   463→    invoices.forEach((inv) => {
   464→      const cid = (inv.clientId as string) || (inv.clientName as string) || 'unknown';
   465→      const v =
   466→        (inv.totalAmount as number) ||
   467→        (inv.grandTotal as number) ||
   468→        (inv.total as number) ||
   469→        0;
   470→      clientRevenue.set(cid, (clientRevenue.get(cid) || 0) + (typeof v === 'number' ? v : 0));
   471→    });
   472→    const top5 = [...clientRevenue.entries()]
   473→      .sort((a, b) => b[1] - a[1])
   474→      .slice(0, 5);
   475→    if (top5.length > 0) {
   476→      const top5Str = top5
   477→        .map(([name, rev]) => {
   478→          const client = clients.find(
   479→            (c) => c.id === name || (c.tradeName as string) === name,
   480→          );
   481→          const label = (client?.tradeName as string) || name;
   482→          return `${label} (${formatINR(rev)})`;
   483→        })
   484→        .join(', ');
   485→      lines.push(`- Top 5 clients by revenue: ${top5Str}`);
   486→    }
   487→
   488→    // High-risk clients (health score < 60)
   489→    const highRisk = clients.filter((c) => {
   490→      const score = (c.healthScore as number) ?? 100;
   491→      return typeof score === 'number' && score < 60;
   492→    });
   493→    if (highRisk.length > 0) {
   494→      const riskStr = highRisk
   495→        .slice(0, 5)
   496→        .map(
   497→          (c) =>
   498→            `${(c.tradeName as string) || (c.name as string) || 'Unknown'} (health ${c.healthScore ?? '?'})`,
   499→        )
   500→        .join(', ');
   501→      lines.push(`- High-risk clients: ${riskStr}`);
   502→    }
   503→
   504→    return lines.join('\n');
   505→  } catch {
   506→    return 'LIVE DATA SNAPSHOT: Data currently unavailable.';
   507→  }
   508→}

   510→// ─── Sub-component: Message Content (mini-markdown) ───────────────────────────
   511→// Renders **bold**, line breaks, bullet points (•), and numbered lists.
   512→
   513→function MessageContent({ text }: { text: string }) {
   514→  const blocks = useMemo(() => {
   515→    const paragraphs = text.split('\n');
   516→    return paragraphs.map((line) => {
   517→      const trimmed = line.trim();
   518→      if (trimmed.startsWith('• ') || trimmed.startsWith('- ')) {
   519→        return { type: 'bullet' as const, content: trimmed.slice(2) };
   520→      }
   521→      if (/^\d+\.\s/.test(trimmed)) {
   522→        return { type: 'numbered' as const, content: trimmed.replace(/^\d+\.\s/, '') };
   523→      }
   524→      return { type: 'text' as const, content: trimmed };
   525→    });
   526→  }, [text]);
   527→
   528→  const renderInline = (s: string, keyBase: string) => {
   529→    // Split on **bold** markers
   530→    const parts = s.split(/(\*\*[^*]+\*\*)/g);
   531→    return parts.map((part, i) => {
   532→      if (part.startsWith('**') && part.endsWith('**')) {
   533→        return (
   534→          <strong key={`${keyBase}-${i}`} className="font-semibold text-foreground">
   535→            {part.slice(2, -2)}
   536→          </strong>
   537→        );
   538→      }
   539→      return <span key={`${keyBase}-${i}`}>{part}</span>;
   540→    });
   541→  };
   542→
   543→  return (
   544→    <div className="space-y-1.5">
   545→      {blocks.map((b, idx) => {
   546→        if (b.type === 'bullet') {
   547→          return (
   548→            <div key={idx} className="flex gap-2">
   549→              <span className="mt-0.5 text-emerald-400">•</span>
   550→              <span className="text-foreground/90">{renderInline(b.content, `b${idx}`)}</span>
   551→            </div>
   552→          );
   553→        }
   554→        if (b.type === 'numbered') {
   555→          return (
   556→            <div key={idx} className="flex gap-2">
   557→              <span className="font-medium accent-text">{idx + 1}.</span>
   558→              <span className="text-foreground/90">{renderInline(b.content, `n${idx}`)}</span>
   559→            </div>
   560→          );
   561→        }
   562→        if (!b.content) return <div key={idx} className="h-1" />;
   563→        return (
   564→          <p key={idx} className="text-foreground/90">
   565→            {renderInline(b.content, `t${idx}`)}

   615→// ─── Hook: Typing Effect ──────────────────────────────────────────────────────
   616→// Reveals text progressively in small chunks for a "typing" feel.
   617→// v10: 2 chars / 10ms (faster than the old 3 chars / 16ms).
   618→
   619→function useTypingEffect(fullText: string, active: boolean, speed = 10, chunkSize = 2) {
   620→  // revealLength is only mutated inside the interval callback (never
   621→  // synchronously in the effect body), which keeps React's
   622→  // set-state-in-effect lint rule happy and avoids cascading renders.
   623→  const [revealLength, setRevealLength] = useState(0);
   624→
   625→  useEffect(() => {
   626→    if (!active) return;
   627→    let i = 0;
   628→    const interval = setInterval(() => {
   629→      i += chunkSize;
   630→      setRevealLength(i);
   631→      if (i >= fullText.length) {
   632→        clearInterval(interval);
   633→      }
   634→    }, speed);
   635→    return () => clearInterval(interval);
   636→  }, [fullText, active, speed, chunkSize]);
   637→
   638→  // When not active (e.g. older messages), show the full text instantly
   639→  // via a derived value — no state mutation required.
   640→  const displayed = active ? fullText.slice(0, revealLength) : fullText;
   641→  const completed = displayed.length >= fullText.length;
   642→
   643→  return { displayed, completed };
   644→}

   646→// ─── Streaming Thinking Phases (V19 — spec-exact) ─────────────────────────────
   647→// Shows each phase sequentially with a checkmark when complete. The final phase

   660→  }, [phaseIndex]);
   661→
