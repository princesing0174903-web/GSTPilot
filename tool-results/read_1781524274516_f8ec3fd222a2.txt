   165→              'bg-white/95 border-border'

   167→          >
   168→            <div className={`mt-0.5 shrink-0 ${

   178→            </div>
   179→            <button className="shrink-0 text-muted-foreground hover:text-foreground" onClick={() => onDismiss(toast.id)}>

   193→function DesktopProgressBar({ completedStep }: { completedStep: number }) {
   194→  const completedCount = completedStep + 1;
   195→  const percent = Math.round((completedCount / PREP_STEPS.length) * 100);
   196→
   197→  return (
   198→    <div className="hidden md:block border border-border/60 rounded-xl p-6">
   199→      <div className="flex items-center justify-between mb-4">
   200→        <h3 className="text-sm font-semibold text-foreground uppercase tracking-wider">Preparation Progress</h3>
   201→        <span className="text-xs font-medium text-muted-foreground">{percent}% Complete</span>
   202→      </div>
   203→      <div className="h-2 rounded-full bg-slate-100 mb-6 overflow-hidden">
   204→        <motion.div
   205→          className="h-full rounded-full bg-emerald-500"
   206→          animate={{ width: `${percent}%` }}
   207→          transition={{ duration: 0.6, ease: 'easeOut' }}
   208→        />
   209→      </div>
   210→      <div className="flex items-start">
   211→        {PREP_STEPS.map((step, idx) => {
   212→          const isCompleted = idx <= completedStep;
   213→          const isActive = idx === completedStep + 1 && idx < PREP_STEPS.length;
   214→          const isFuture = idx > completedStep + 1;
   215→          return (
   216→            <React.Fragment key={step.id}>
   217→              <div className="flex flex-col items-center" style={{ minWidth: idx === 0 || idx === PREP_STEPS.length - 1 ? '80px' : '100px', flex: '1 1 0' }}>
   218→                <motion.div
   219→                  className={`flex items-center justify-center h-9 w-9 rounded-full shrink-0 text-sm font-semibold transition-all duration-500 ${
   220→                    isCompleted
   221→                      ? 'bg-emerald-500 text-white shadow-sm shadow-emerald-500/30'
   222→                      : isActive
   223→                        ? 'bg-emerald-50 text-emerald-700 border-2 border-emerald-500 shadow-sm shadow-emerald-200/50'
   224→                        : 'bg-slate-100 text-slate-400'
   225→                  }`}
   226→                  animate={isActive ? { scale: [1, 1.08, 1] } : {}}
   227→                  transition={isActive ? { duration: 2, repeat: Infinity, ease: 'easeInOut' } : {}}
   228→                >
   229→                  {isCompleted ? <Check className="h-4 w-4" /> : idx + 1}
   230→                </motion.div>
   231→                <span className={`text-xs text-center leading-tight mt-2 whitespace-nowrap ${
   232→                  isCompleted ? 'text-emerald-700 font-medium' : isActive ? 'text-foreground font-semibold' : 'text-muted-foreground'
   233→                }`}>
   234→                  {step.label}
   235→                </span>
   236→              </div>
   237→              {idx < PREP_STEPS.length - 1 && (
   238→                <div className="flex items-center pt-[18px] flex-1 min-w-[16px]">
   239→                  <motion.div
   240→                    className={`h-[3px] w-full rounded-full transition-colors duration-500 ${isCompleted ? 'bg-emerald-400' : 'bg-slate-200'}`}
   241→                    animate={isActive ? { opacity: [0.5, 1, 0.5] } : {}}
   242→                    transition={isActive ? { duration: 2, repeat: Infinity, ease: 'easeInOut' } : {}}
   243→                  />
   244→                </div>
   245→              )}
   246→            </React.Fragment>
   247→          );
   248→        })}
   249→      </div>
   250→    </div>
   251→  );
   252→}

   284→export default function ReturnPrepWorkspace() {
   285→  const { selectedClientId, returnPrepCtx, setCurrentView } = useApp();
   286→  const updateFilingStatus = useUpdateFilingStatus();
   287→  const fileReturnMutation = useFileReturn();
   288→
   289→  // Resolve client ID
   290→  const rawClientId = selectedClientId ?? returnPrepCtx.clientId ?? '';
   291→  const clientId = rawClientId;
   292→
   293→  const returnType = returnPrepCtx.returnType;
   294→  const period = returnPrepCtx.period;
   295→
   296→  // ── API data via React Query ──
   297→  const { data: clientData, isLoading: clientLoading } = useClient(clientId);
   298→  const { data: invoicesData, isLoading: invoicesLoading } = useInvoices(clientId);
   299→  const { data: issuesData, isLoading: issuesLoading } = useIssues(clientId);
   300→  const { data: filingsData } = useFilings(clientId);
   301→  const { data: reconData } = useReconRuns(clientId);
   302→
   303→  const client = clientData?.client ?? null;
   304→  const invoices: Invoice[] = invoicesData?.invoices?.filter((i: Invoice) => i.clientId === clientId) ?? [];
   305→  const validationIssues: Issue[] = issuesData?.issues?.filter((i: Issue) => i.clientId === clientId) ?? [];
   306→  const filings = filingsData?.filings?.filter((f: GSTRFiling) => f.clientId === clientId) ?? [];
   307→  const reconRuns = reconData?.runs ?? [];
   308→
   309→  // ── Local UI state ──
   310→  const [currentPrepStep, setCurrentPrepStep] = useState(1);
   311→  const [invoiceFilter, setInvoiceFilter] = useState<'all' | 'validated' | 'warning' | 'error'>('all');
   312→  const [toasts, setToasts] = useState<ToastMessage[]>([]);
   313→  const [selectedInvoice, setSelectedInvoice] = useState<Invoice | null>(null);
   314→  const [selectedReconCategory, setSelectedReconCategory] = useState<string | null>(null);
   315→  const [filingModalOpen, setFilingModalOpen] = useState(false);
   316→  const [filingProgress, setFilingProgress] = useState<'idle' | 'validating' | 'generating' | 'submitting' | 'success'>('idle');
   317→
   318→  // ── Derived state ──
   319→  const clientName = client?.tradeName ?? 'Select a Client';
   320→  const clientGSTIN = client?.gstin ?? '';
   321→
   322→  const filteredInvoices = useMemo(() => {
   323→    if (invoiceFilter === 'all') return invoices;
   324→    return invoices.filter(inv => inv.status === invoiceFilter);
   325→  }, [invoices, invoiceFilter]);
   326→
   327→  // Derived from API data
   328→  const unresolvedIssues = validationIssues.filter(i => i.status === 'open').length;
   329→
   330→  const totalTaxable = invoices.reduce((sum, inv) => sum + inv.taxableValue, 0);
   331→  const totalCGST = invoices.reduce((sum, inv) => sum + inv.cgst, 0);
   332→  const totalSGST = invoices.reduce((sum, inv) => sum + inv.sgst, 0);
   333→  const totalIGST = invoices.reduce((sum, inv) => sum + inv.igst, 0);
   334→  const totalTax = totalCGST + totalSGST + totalIGST;
   335→
   336→  const errorCount = invoices.filter(i => i.status === 'error').length;
   337→  const warningCount = invoices.filter(i => i.status === 'warning').length;
   338→  const validatedCount = invoices.filter(i => i.status === 'validated' || i.status === 'approved').length;
   339→  const allValidated = errorCount === 0 && warningCount === 0;
   340→
   341→  const validationScore = invoices.length > 0 ? Math.min(100, Math.round((validatedCount / invoices.length) * 100)) : 100;
   342→
   343→  // Compute match rate from reconciliation runs
   344→  const matchRate = useMemo(() => {
   345→    if (reconRuns.length === 0) return 0;
   346→    const latestRun = reconRuns[0];
   347→    return latestRun.totalRecords > 0 ? Math.round((latestRun.matched / latestRun.totalRecords) * 100) : 0;
   348→  }, [reconRuns]);
   349→
   350→  // Compute compliance score from client health and validation score
   351→  const complianceScore = client?.healthScore ?? validationScore;
   352→
   353→  const allChecksPass = validationScore >= 95 && unresolvedIssues === 0;
   354→  const isGSTR1 = returnType === 'GSTR-1';
   355→
   356→  // Use local step state
   357→  const effectiveCompletedStep = currentPrepStep;
   358→
   359→  // ── Toast helper ──
   360→  const addToast = useCallback((title: string, description: string, type: ToastMessage['type'] = 'success') => {
   361→    const id = `toast-${Date.now()}`;
   362→    setToasts(prev => [...prev, { id, title, description, type }]);
   363→    setTimeout(() => setToasts(prev => prev.filter(t => t.id !== id)), 4000);
   364→  }, []);
   365→
   366→  const dismissToast = useCallback((id: string) => {
   367→    setToasts(prev => prev.filter(t => t.id !== id));
   368→  }, []);
   369→
   370→  // ── Actions ──
   371→  const handleBack = () => setCurrentView('client-workspace');
   372→
   373→  const handleApproveInvoice = (invId: string) => {
   374→    const inv = invoices.find(i => i.id === invId);
   375→    if (!inv || inv.status === 'validated') return;
   376→    updateFilingStatus.mutate({ id: invId, status: 'validated' } as any, {
   377→      onSuccess: () => toast.success(`${inv.invoiceNumber} approved`),
   378→      onError: (err: Error) => toast.error(err.message),
   379→    });
   380→  };
   381→
   382→  const handleFixIssue = (issueId: string) => {
   383→    const issue = validationIssues.find(i => i.id === issueId);
   384→    if (!issue || issue.status === 'resolved') return;
   385→    updateFilingStatus.mutate({ id: issueId, status: 'resolved' } as any, {
   386→      onSuccess: () => toast.success(`${issue.category} resolved`),
   387→      onError: (err: Error) => toast.error(err.message),
   388→    });
   389→  };
   390→
   391→  const handleDismissInsight = (insightId: string) => {
   392→    toast.info('Insight dismissed');
   393→  };
   394→
   395→  // Run Validation: mark all invoices as validated via API
   396→  const handleRunValidation = () => {
   397→    const unresolved = validationIssues.filter(i => i.status === 'open');
   398→    // Mark all non-validated invoices as validated
   399→    invoices.forEach(inv => {
   400→      if (inv.status !== 'validated' && inv.status !== 'approved') {
   401→        updateFilingStatus.mutate({ id: inv.id, status: 'validated' } as any);
   402→      }
   403→    });
   404→    setCurrentPrepStep(2);
   405→    toast.success('Validation Complete', { description: `All ${unresolved.length} issues resolved. ${invoices.length - validatedCount} invoices updated.` });
   406→  };
   407→
   408→  // Run Reconciliation: advance step to 3
   409→  const handleRunReconciliation = () => {
   410→    if (currentPrepStep < 2) {
   411→      toast.warning('Complete validation first', { description: 'All invoices must be validated before reconciliation' });
   412→      return;
   413→    }
   414→    setCurrentPrepStep(3);
   415→    toast.success('Reconciliation Complete', { description: 'Books vs GSTR-2B matching finished.' });
   416→  };
   417→
   418→  // Mark Ready: advance step to 5 (Ready to File)
   419→  const handleMarkReady = () => {
   420→    if (errorCount > 0 || warningCount > 0) {
   421→      toast.warning('Cannot mark ready', { description: `${errorCount} errors and ${warningCount} warnings must be resolved first` });
   422→      return;
   423→    }
   424→    setCurrentPrepStep(5);
   425→    toast.success('Return marked as Ready to File', { description: 'All validations passed. You can now file this return.' });
   426→  };
   427→
   428→  // File Return Simulation
   429→  const handleFileReturn = () => {
   430→    if (!allChecksPass) return;
   431→    setFilingModalOpen(true);
   432→    setFilingProgress('validating');
   433→
   434→    setTimeout(() => setFilingProgress('generating'), 1500);
   435→    setTimeout(() => setFilingProgress('submitting'), 3000);
   436→    setTimeout(() => {
   437→      setFilingProgress('success');
   438→      setCurrentPrepStep(6);
   439→      // Find the matching filing and file it via API
   440→      const matchingFiling = filings.find(f => f.returnType === returnType && f.period === period);
   441→      if (matchingFiling) {
   442→        fileReturnMutation.mutate(matchingFiling.id, {
   443→          onSuccess: () => toast.success(`${returnType} Filed Successfully!`, { description: `Period: ${periodToLabel(period)} · Tax: ${formatCurrency(totalTax)}` }),
   444→          onError: (err: Error) => toast.error(err.message),
   445→        });
   446→      } else {
   447→        toast.success(`${returnType} Filed Successfully!`, { description: `Period: ${periodToLabel(period)} · Tax: ${formatCurrency(totalTax)}` });
   448→      }
