   164→// ─── Save Button with Animated States ──────────────────────────────────
   165→function SaveButton({ onSave }: { onSave: () => Promise<void> }) {
   166→  const [state, setState] = useState<'idle' | 'saving' | 'saved'>('idle')
   167→
   168→  const handleClick = async () => {
   169→    setState('saving')
   170→    try {
   171→      await onSave()
   172→      setState('saved')
   173→      setTimeout(() => setState('idle'), 2000)
   174→    } catch {
   175→      setState('idle')
   176→    }
   177→  }
   178→
   179→  return (
   180→    <Button
   181→      onClick={handleClick}
   182→      disabled={state === 'saving'}
   183→      className="bg-emerald-600 hover:bg-emerald-700 text-white gap-2 min-w-[110px]"
   184→    >
   185→      <AnimatePresence mode="wait">
   186→        {state === 'idle' && (
   187→          <motion.span
   188→            key="idle"
   189→            initial={{ opacity: 0, scale: 0.8 }}
   190→            animate={{ opacity: 1, scale: 1 }}
   191→            exit={{ opacity: 0, scale: 0.8 }}
   192→            className="flex items-center gap-2"
   193→          >
   194→            <Save className="h-4 w-4" />
   195→            Save
   196→          </motion.span>
   197→        )}
   198→        {state === 'saving' && (
   199→          <motion.span
   200→            key="saving"
   201→            initial={{ opacity: 0, scale: 0.8 }}
   202→            animate={{ opacity: 1, scale: 1 }}
   203→            exit={{ opacity: 0, scale: 0.8 }}
   204→            className="flex items-center gap-2"
   205→          >
   206→            <Loader2 className="h-4 w-4 animate-spin" />

   227→// ─── Password Strength Helper ──────────────────────────────────────────
   228→function getPasswordStrength(password: string): { score: number; label: string; color: string } {
   229→  if (!password) return { score: 0, label: '', color: '' }
   230→  let score = 0
   231→  if (password.length >= 8) score++
   232→  if (password.length >= 12) score++
   233→  if (/[A-Z]/.test(password)) score++
   234→  if (/[0-9]/.test(password)) score++
   235→  if (/[^A-Za-z0-9]/.test(password)) score++
   236→
   237→  if (score <= 1) return { score: 20, label: 'Weak', color: 'bg-red-500' }
   238→  if (score <= 2) return { score: 40, label: 'Fair', color: 'bg-orange-500' }
   239→  if (score <= 3) return { score: 60, label: 'Good', color: 'bg-yellow-500' }
   240→  if (score <= 4) return { score: 80, label: 'Strong', color: 'bg-emerald-500' }
   241→  return { score: 100, label: 'Excellent', color: 'bg-emerald-600' }
   242→}
