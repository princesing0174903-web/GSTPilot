    64→// ─── Helpers ──────────────────────────────────────────────────────────────────
    65→function fmtINR(n: number): string {
    66→  if (!isFinite(n)) return '₹0';
    67→  const abs = Math.abs(n);
    68→  if (abs >= 10000000) return `₹${(n / 10000000).toFixed(2)}Cr`;
    69→  if (abs >= 100000) return `₹${(n / 100000).toFixed(2)}L`;
    70→  if (abs >= 1000) return `₹${(n / 1000).toFixed(1)}K`;
    71→  return `₹${Math.round(n).toLocaleString('en-IN')}`;
    72→}
    73→
    74→function fmtNum(n: number): string {
    75→  if (n >= 10000000) return `${(n / 10000000).toFixed(2)}Cr`;
    76→  if (n >= 100000) return `${(n / 100000).toFixed(2)}L`;
    77→  if (n >= 1000) return `${(n / 1000).toFixed(1)}K`;
    78→  return n.toLocaleString('en-IN');
    79→}

    81→function timeAgo(iso: string | null): string {
    82→  if (!iso) return 'never';
    83→  const diff = Date.now() - new Date(iso).getTime();
    84→  const s = Math.floor(diff / 1000);
    85→  if (s < 60) return 'just now';
    86→  const m = Math.floor(s / 60);
    87→  if (m < 60) return `${m}m ago`;
    88→  const h = Math.floor(m / 60);
    89→  if (h < 24) return `${h}h ago`;
    90→  const d = Math.floor(h / 24);
    91→  if (d < 30) return `${d}d ago`;
    92→  return new Date(iso).toLocaleDateString('en-IN');
    93→}
