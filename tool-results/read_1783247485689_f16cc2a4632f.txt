   578→function CodeSnippet({ code, language }: { code: string; language: string }) {
   579→  const [copied, setCopied] = useState(false);
   580→
   581→  const handleCopy = () => {
   582→    navigator.clipboard.writeText(code);
   583→    setCopied(true);
   584→    setTimeout(() => setCopied(false), 2000);
   585→  };
   586→
   587→  // Simple syntax highlighting via color spans
   588→  const highlight = (text: string, lang: string) => {
   589→    if (lang === 'curl') {
   590→      return text
   591→        .replace(/(curl)/g, '<span class="text-emerald-400 font-semibold">$1</span>')
   592→        .replace(/(-[A-Z])/g, '<span class="text-amber-400">$1</span>')
   593→        .replace(/(https?:\/\/[^\s'"]+)/g, '<span class="text-sky-400">$1</span>')
   594→        .replace(/("[^"]*")/g, '<span class="text-orange-300">$1</span>')
   595→        .replace(/(gpi_live_sk_\w+)/g, '<span class="text-pink-400">$1</span>');
   596→    }
   597→    if (lang === 'javascript' || lang === 'typescript') {
   598→      return text
   599→        .replace(/(import|from|const|await|new)/g, '<span class="text-purple-400">$1</span>')
   600→        .replace(/(Gstpilot|gstpilot)/g, '<span class="text-emerald-400">$1</span>')
   601→        .replace(/('[^']*')/g, '<span class="text-orange-300">$1</span>')
   602→        .replace(/("[^"]*")/g, '<span class="text-orange-300">$1</span>')
   603→        .replace(/(\b\d+\b)/g, '<span class="text-sky-400">$1</span>')
   604→        .replace(/(\/\/.*)/g, '<span class="text-slate-500">$1</span>')
   605→        .replace(/(gpi_live_sk_\w+)/g, '<span class="text-pink-400">$1</span>');
   606→    }
   607→    if (lang === 'python') {
   608→      return text
   609→        .replace(/(import|from|def|await|print)/g, '<span class="text-purple-400">$1</span>')
   610→        .replace(/(Gstpilot|gstpilot)/g, '<span class="text-emerald-400">$1</span>')
   611→        .replace(/('[^']*')/g, '<span class="text-orange-300">$1</span>')
   612→        .replace(/("[^"]*")/g, '<span class="text-orange-300">$1</span>')
   613→        .replace(/(#.*)/g, '<span class="text-slate-500">$1</span>')
   614→        .replace(/(gpi_live_sk_\w+)/g, '<span class="text-pink-400">$1</span>');
   615→    }
   616→    return text;
   617→  };
   618→
   619→  return (
   620→    <div className="relative group">
   621→      <pre className="bg-slate-900 text-slate-200 p-4 rounded-lg text-xs font-mono overflow-x-auto leading-relaxed">
   622→        <code dangerouslySetInnerHTML={{ __html: highlight(code, language) }} />
   623→      </pre>
   624→      <button
   625→        onClick={handleCopy}
   626→        className="absolute top-2 right-2 p-1.5 rounded-md bg-slate-700/80 hover:bg-slate-600 text-slate-300 transition-colors opacity-0 group-hover:opacity-100"
   627→      >
   628→        {copied ? <Check className="h-3.5 w-3.5 text-emerald-400" /> : <Copy className="h-3.5 w-3.5" />}
   629→      </button>
   630→    </div>
   631→  );
   632→}
