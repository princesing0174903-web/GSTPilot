   878→   878→/** Subscribe to a Firestore collection with firm scoping */
   879→   879→export function subscribeToCollection<T>(
   880→   880→  collectionName: CollectionName,
   881→   881→  callback: (data: Array<T & { id: string }>) => void,
   882→   882→  ...constraints: QueryConstraint[]
   883→   883→): Unsubscribe {
   884→   884→  const firmId = currentFirmId();
   885→   885→
   886→   886→  const q = firmId
   887→   887→    ? query(collection(db, collectionName), where('firmId', '==', firmId), ...constraints)
   888→   888→    : query(collection(db, collectionName), ...constraints);
   889→   889→
   890→   890→  return onSnapshot(q, (snap) => {
   891→   891→    const data = snap.docs.map(d => {
   892→   892→      const docData = d.data();
   893→   893→      const converted: Record<string, unknown> = { id: d.id };
   894→   894→      for (const [key, value] of Object.entries(docData)) {
   895→   895→        if (value && typeof value === 'object' && 'toDate' in value && typeof (value as { toDate: () => Date }).toDate === 'function') {
   896→   896→          converted[key] = (value as { toDate: () => Date }).toDate().toISOString();
   897→   897→        } else {
   898→   898→          converted[key] = value;
   899→   899→        }
   900→   900→      }
   901→   901→      return converted as T & { id: string };
   902→   902→    });
   903→   903→    callback(data);
   904→   904→  }, (error) => {
   905→   905→    console.warn(`[Firestore] Subscription error for ${collectionName}:`, error);
   906→   906→    callback([]);
   907→   907→  });
   908→   908→}
   909→   909→
   910→   910→/** Subscribe to a single Firestore document */
   911→   911→export function subscribeToDoc<T>(
   912→   912→  collectionName: CollectionName,
   913→   913→  docId: string,
   914→   914→  callback: (data: (T & { id: string }) | null) => void,
   915→   915→): Unsubscribe {
   916→   916→  return onSnapshot(doc(db, collectionName, docId), (snap) => {
   917→   917→    if (!snap.exists()) {
   918→   918→      callback(null);
   919→   919→      return;
   920→   920→    }
   921→   921→    const docData = snap.data();
   922→   922→    const converted: Record<string, unknown> = { id: snap.id };
   923→   923→    for (const [key, value] of Object.entries(docData)) {
   924→   924→      if (value && typeof value === 'object' && 'toDate' in value && typeof (value as { toDate: () => Date }).toDate === 'function') {
   925→   925→        converted[key] = (value as { toDate: () => Date }).toDate().toISOString();
   926→   926→      } else {
   927→   927→        converted[key] = value;
   928→   928→      }
   929→   929→    }
   930→   930→    callback(converted as T & { id: string });
   931→   931→  }, (error) => {

   934→   934→  });

   937→   937→/** Compute live dashboard metrics from Firestore data */
   938→   938→export function computeDashboardMetrics(
   939→   939→  clients: Array<FirestoreClient & { id: string }>,
   940→   940→  invoices: Array<FirestoreInvoice & { id: string }>,
   941→   941→  returns: Array<FirestoreReturn & { id: string }>,
   942→   942→  documents: Array<FirestoreDocument & { id: string }>,
   943→   943→  activities: Array<FirestoreActivity & { id: string }>,
   944→   944→): LiveDashboardMetrics {
   945→   945→  const activeClients = clients.filter(c => c.status === 'active').length;
   946→   946→  const filedReturns = returns.filter(r => r.status === 'filed').length;
   947→   947→  const pendingReturns = returns.filter(r => r.status !== 'filed').length;
   948→   948→  const readyToFile = returns.filter(r => ['validated', 'reviewed', 'generated'].includes(r.status)).length;
   949→   949→  const overdueReturns = returns.filter(r => {
   950→   950→    if (r.status === 'filed') return false;
   951→   951→    const periodDate = parsePeriod(r.period);
   952→   952→    return periodDate && periodDate < new Date();
   953→   953→  }).length;
   954→   954→
   955→   955→  const totalTaxVolume = invoices.reduce((sum, inv) => sum + (inv.totalAmount || 0), 0);
   956→   956→  const totalTax = invoices.reduce((sum, inv) => sum + (inv.cgst || 0) + (inv.sgst || 0) + (inv.igst || 0) + (inv.cess || 0), 0);
   957→   957→

   959→   959→    returns.filter(r => r.criticalErrors > 0).length;
   960→   960→  const warnings = invoices.filter(inv => inv.riskLevel === 'high' || inv.riskLevel === 'medium').length +
   961→   961→    returns.filter(r => r.warnings > 0).length;
   962→   962→
   963→   963→  const averageHealthScore = clients.length > 0
   964→   964→    ? Math.round(clients.reduce((sum, c) => sum + c.healthScore, 0) / clients.length)
   965→   965→    : 0;
   966→   966→
   967→   967→  const matchedInvoices = invoices.filter(inv => inv.matchStatus === 'perfect_match').length;
   968→   968→  const matchPercentage = invoices.length > 0 ? Math.round((matchedInvoices / invoices.length) * 100) : 100;
   969→   969→
   970→   970→  const riskyInvoices = invoices.filter(inv => ['high', 'critical'].includes(inv.riskLevel)).length;
   971→   971→  const riskPercentage = invoices.length > 0 ? Math.round((riskyInvoices / invoices.length) * 100) : 0;
   972→   972→
   973→   973→  const documentsProcessed = documents.filter(d => d.extractionStatus === 'completed').length;
   974→   974→  const extractionsPending = documents.filter(d => d.extractionStatus === 'pending' || d.extractionStatus === 'in_progress').length;
   975→   975→
   976→   976→  const recentActivities = [...activities]
   977→   977→    .sort((a, b) => {
   978→   978→      const aTime = typeof a.createdAt === 'string' ? new Date(a.createdAt).getTime() : 0;
   979→   979→      const bTime = typeof b.createdAt === 'string' ? new Date(b.createdAt).getTime() : 0;
   980→   980→      return bTime - aTime;
   981→   981→    })
   982→   982→    .slice(0, 10);
   983→   983→
   984→   984→  const upcomingFilings = returns
   985→   985→    .filter(r => r.status !== 'filed')
   986→   986→    .sort((a, b) => {
   987→   987→      const aTime = typeof a.createdAt === 'string' ? new Date(a.createdAt).getTime() : 0;
   988→   988→      const bTime = typeof b.createdAt === 'string' ? new Date(b.createdAt).getTime() : 0;
   989→   989→      return aTime - bTime;
   990→   990→    })
   991→   991→    .slice(0, 5);
   992→   992→
   993→   993→  return {
   994→   994→    totalClients: clients.length,
   995→   995→    activeClients,
   996→   996→    totalInvoices: invoices.length,
   997→   997→    totalTaxVolume,
   998→   998→    filedReturns,
   999→   999→    pendingReturns,
  1000→  1000→    overdueReturns,
  1001→  1001→    readyToFile,
  1002→  1002→    criticalIssues,
  1003→  1003→    warnings,
  1004→  1004→    averageHealthScore,
  1005→  1005→    matchPercentage,
  1006→  1006→    riskPercentage,
  1007→  1007→    documentsProcessed,
  1008→  1008→    extractionsPending,
  1009→  1009→    recentActivities,
  1010→  1010→    upcomingFilings,
  1011→  1011→  };
  1012→  1012→}

  1014→  1014→function parsePeriod(period: string | null): Date | null {
  1015→  1015→  if (!period) return null;
  1016→  1016→  try {
  1017→  1017→    const [mm, yyyy] = period.split('-').map(Number);
  1018→  1018→    if (!mm || !yyyy) return null;
  1019→  1019→    return new Date(yyyy, mm, 20); // Due date is typically 20th of next month
  1020→  1020→  } catch {
  1021→  1021→    return null;
  1022→  1022→  }
  1023→  1023→}
