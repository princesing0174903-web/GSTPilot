    43→function formatINR(n: number): string {
    44→  return new Intl.NumberFormat('en-IN', { style: 'currency', currency: 'INR', maximumFractionDigits: 0 }).format(n)
    45→}
    46→
    47→function formatDate(d: string | unknown): string {
    48→  if (!d) return '—'
    49→  try {
    50→    const date = typeof d === 'string' ? new Date(d) : new Date()
    51→    return date.toLocaleDateString('en-IN', { day: 'numeric', month: 'short', year: 'numeric' })
    52→  } catch { return '—' }
    53→}
    54→
    55→function formatTime(d: string | unknown): string {
    56→  if (!d) return ''
    57→  try {
    58→    const date = typeof d === 'string' ? new Date(d) : new Date()
    59→    return date.toLocaleTimeString('en-IN', { hour: '2-digit', minute: '2-digit' })
    60→  } catch { return '' }
    61→}
    62→
    63→function daysUntil(d: string | unknown): number {
    64→  if (!d) return 999
    65→  try {
    66→    const date = typeof d === 'string' ? new Date(d) : new Date()
    67→    return Math.ceil((date.getTime() - Date.now()) / (1000 * 60 * 60 * 24))
    68→  } catch { return 999 }
    69→}
    70→
    71→function scoreColor(score: number): string {
    72→  if (score >= 70) return 'text-emerald-600 bg-emerald-50 border-emerald-200'
    73→  if (score >= 40) return 'text-amber-600 bg-amber-50 border-amber-200'
    74→  return 'text-red-600 bg-red-50 border-red-200'
    75→}
    76→
    77→function scoreBgGradient(score: number): string {
    78→  if (score >= 70) return 'from-emerald-500 to-emerald-600'
    79→  if (score >= 40) return 'from-amber-500 to-amber-600'
    80→  return 'from-red-500 to-red-600'
    81→}

   349→}

   354→
   355→function PipelineView({ leads, loading }: { leads: Array<FirestoreLead & { id: string }>; loading: boolean }) {
   356→  const [searchTerm, setSearchTerm] = useState('')
   357→  const [sourceFilter, setSourceFilter] = useState<string>('all')
   358→  const [dragOverColumn, setDragOverColumn] = useState<string | null>(null)
   359→  const [convertingId, setConvertingId] = useState<string | null>(null)
   360→
   361→  const filteredLeads = useMemo(() => {
   362→    return leads.filter(lead => {
   363→      const matchesSearch = !searchTerm ||
   364→        lead.contactName.toLowerCase().includes(searchTerm.toLowerCase()) ||
   365→        lead.company.toLowerCase().includes(searchTerm.toLowerCase())
   366→      const matchesSource = sourceFilter === 'all' || lead.source === sourceFilter
   367→      return matchesSearch && matchesSource
   368→    })
   369→  }, [leads, searchTerm, sourceFilter])
   370→
   371→  const handleStatusChange = useCallback(async (leadId: string, newStatus: LeadStatus) => {
   372→    try {
   373→      await updateLead(leadId, { status: newStatus })
   374→    } catch (err) {
   375→      console.warn('Failed to update lead status:', err)
   376→    }
   377→  }, [])
   378→
   379→  const handleConvert = useCallback(async (leadId: string) => {
   380→    setConvertingId(leadId)
   381→    try {
   382→      await convertLeadToClient(leadId)
   383→    } catch (err) {
   384→      console.warn('Failed to convert lead:', err)
   385→    } finally {
   386→      setConvertingId(null)
   387→    }
   388→  }, [])
   389→
   390→  const handleDrop = useCallback((leadId: string, newStatus: LeadStatus) => {
   391→    handleStatusChange(leadId, newStatus)
   392→    setDragOverColumn(null)
   393→  }, [handleStatusChange])
   394→
   395→  if (loading) {
   396→    return (
   397→      <div className="flex gap-4 overflow-x-auto pb-4">
   398→        {PIPELINE_COLUMNS.map(col => (
   399→          <div key={col.status} className="min-w-[260px] flex-1">
   400→            <Skeleton className="h-8 w-32 mb-3" />
   401→            <div className="space-y-3">
   402→              {[1, 2, 3].map(i => (

  1107→function RevenueForecastView({ leads, deals, loading }: { leads: Array<FirestoreLead & { id: string }>; deals: Array<FirestoreDeal & { id: string }>; loading: boolean }) {
  1108→  const stageValues = useMemo(() => {
  1109→    const stages: Record<string, { label: string; value: number; color: string }> = {
  1110→      new: { label: 'New', value: 0, color: '#94a3b8' },
  1111→      contacted: { label: 'Contacted', value: 0, color: '#3b82f6' },
  1112→      qualified: { label: 'Qualified', value: 0, color: '#8b5cf6' },

  1115→    }
  1116→    leads.filter(l => l.status !== 'converted' && l.status !== 'lost').forEach(lead => {
  1117→      if (stages[lead.status]) stages[lead.status].value += lead.estimatedValue
  1118→    })
  1119→    return Object.values(stages)
  1120→  }, [leads])
  1121→
  1122→  const maxBarValue = useMemo(() => Math.max(...stageValues.map(s => s.value), 1), [stageValues])
  1123→
  1124→  const weightedPipeline = useMemo(() => {
  1125→    return deals.filter(d => d.stage !== 'closed_lost').reduce((sum, d) => sum + d.value * d.probability / 100, 0)
  1126→  }, [deals])
  1127→
  1128→  const totalPipeline = useMemo(() => {
  1129→    return deals.filter(d => d.stage !== 'closed_lost').reduce((sum, d) => sum + d.value, 0)
  1130→  }, [deals])
  1131→
  1132→  const monthlyTrend = useMemo(() => {
  1133→    // Compute from real deals — no Math.random fabrication.
  1134→    // Returns empty array when no deals exist (honest empty state).

  1138→      const monthDeals = deals.filter((d: any) => {
  1139→        const dDate = new Date(d.closeDate ?? d.createdAt ?? 0)
  1140→        return dDate.getMonth() === months.indexOf(m) && (d.status === 'won' || d.status === 'closed')
  1141→      })

  1143→        month: m,
  1144→        converted: monthDeals.length,
  1145→        value: monthDeals.reduce((s: number, d: any) => s + Number(d.value ?? 0), 0),
  1146→      }
  1147→    })

  1154→  return (
  1155→    <div className="space-y-6">
  1156→      {/* Summary */}
  1157→      <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
  1158→        <Card className="border-0 shadow-sm">
  1159→          <CardContent className="p-4">
  1160→            <p className="text-[11px] text-muted-foreground mb-1">Total Pipeline Value</p>
  1161→            <p className="text-xl font-bold text-foreground">{formatINR(totalPipeline)}</p>
  1162→          </CardContent>
  1163→        </Card>
  1164→        <Card className="border-0 shadow-sm">
  1165→          <CardContent className="p-4">
  1166→            <p className="text-[11px] text-muted-foreground mb-1">Weighted Pipeline</p>

  1169→        </Card>

  1177→
  1178→      {/* Pipeline by Stage (SVG Bar Chart) */}
  1179→      <Card className="border-0 shadow-sm">
  1180→        <CardHeader className="pb-3 pt-4 px-4">
  1181→          <CardTitle className="text-xs font-semibold flex items-center gap-2">
  1182→            <BarChart3 className="h-3.5 w-3.5 text-emerald-500" /> Pipeline Value by Stage
  1183→          </CardTitle>

  1185→        <CardContent className="px-4 pb-4">
  1186→          <div className="space-y-3">
