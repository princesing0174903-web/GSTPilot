   728→/** Helper to safely parse a JSON string field into a typed array. */
   729→export function parseJsonArray<T>(s: string | null | undefined, fallback: T[] = []): T[] {
   730→  if (!s) return fallback;
   731→  try {
   732→    const parsed = JSON.parse(s);
   733→    return Array.isArray(parsed) ? (parsed as T[]) : fallback;
   734→  } catch {
   735→    return fallback;
   736→  }
   737→}
   738→
   739→/** Helper to safely parse a JSON string field into a typed object. */
   740→export function parseJsonObject<T>(s: string | null | undefined, fallback: T): T {
   741→  if (!s) return fallback;
   742→  try {
   743→    return { ...fallback, ...(JSON.parse(s) as T) };
   744→  } catch {
   745→    return fallback;
   746→  }
   747→}
