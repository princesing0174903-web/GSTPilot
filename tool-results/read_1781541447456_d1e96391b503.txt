   122→// ─── CSS Bar Chart ──────────────────────────────────────────────────────────
   123→
   124→function CSSBarChart({
   125→  data,
   126→  valueKey,
   127→  labelKey,
   128→  formatValue,
   129→  color = 'emerald',
   130→  height = 160,
   131→}: {
   132→  data: Array<Record<string, unknown>>;
   133→  valueKey: string;
   134→  labelKey: string;
   135→  formatValue: (v: number) => string;
   136→  color?: string;
   137→  height?: number;
   138→}) {
   139→  const values = data.map((d) => Number(d[valueKey]) || 0);
   140→  const maxVal = Math.max(...values, 1);
   141→
   142→  const barColor: Record<string, string> = {
   143→    emerald: 'bg-emerald-500',
   144→    teal: 'bg-teal-500',
   145→    cyan: 'bg-cyan-500',
   146→    violet: 'bg-violet-500',
   147→    amber: 'bg-amber-500',
   148→  };
   149→
   150→  const hoverColor: Record<string, string> = {
   151→    emerald: 'hover:bg-emerald-400',
   152→    teal: 'hover:bg-teal-400',
   153→    cyan: 'hover:bg-cyan-400',
   154→    violet: 'hover:bg-violet-400',
   155→    amber: 'hover:bg-amber-400',
   156→  };
   157→
   158→  return (
   159→    <div className="flex items-end gap-1.5 sm:gap-2" style={{ height }}>
   160→      {data.map((item, idx) => {
   161→        const val = Number(item[valueKey]) || 0;
   162→        const pct = maxVal > 0 ? (val / maxVal) * 100 : 0;
   163→        return (
   164→          <div key={idx} className="flex flex-col items-center flex-1 min-w-0 group relative">
   165→            {/* Tooltip */}
   166→            <div className="absolute -top-8 left-1/2 -translate-x-1/2 opacity-0 group-hover:opacity-100 transition-opacity bg-emerald-900 text-white text-[10px] px-2 py-1 rounded whitespace-nowrap z-10 pointer-events-none">
   167→              {formatValue(val)}
   168→            </div>
   169→            <div
   170→              className={`w-full rounded-t-sm transition-all duration-500 ${barColor[color]} ${hoverColor[color]} min-h-[2px]`}
   171→              style={{ height: `${Math.max(pct, 1)}%` }}
   172→            />
   173→            <span className="text-[10px] text-muted-foreground mt-1.5 truncate w-full text-center">
   174→              {String(item[labelKey])}
   175→            </span>
   176→          </div>
   177→        );
   178→      })}
   179→    </div>
   180→  );
   181→}

   183→// ─── Progress Ring ──────────────────────────────────────────────────────────
   184→
   185→function ProgressRing({
   186→  value,
   187→  max = 100,
   188→  size = 120,
   189→  strokeWidth = 10,
   190→  label,
   191→}: {
   192→  value: number;
   193→  max?: number;
   194→  size?: number;
   195→  strokeWidth?: number;
   196→  label: string;
   197→}) {
   198→  const radius = (size - strokeWidth) / 2;
   199→  const circumference = 2 * Math.PI * radius;
   200→  const pct = max > 0 ? Math.min(value / max, 1) : 0;
   201→  const offset = circumference - pct * circumference;
   202→
   203→  const color =
   204→    pct >= 0.8
   205→      ? 'text-emerald-500'
   206→      : pct >= 0.5
   207→        ? 'text-amber-500'
   208→        : 'text-red-500';
   209→
   210→  const strokeColor =
   211→    pct >= 0.8
   212→      ? 'stroke-emerald-500'
   213→      : pct >= 0.5
   214→        ? 'stroke-amber-500'
   215→        : 'stroke-red-500';
   216→
   217→  return (
   218→    <div className="flex flex-col items-center gap-2">
   219→      <svg width={size} height={size} className="-rotate-90">
   220→        <circle
   221→          cx={size / 2}
   222→          cy={size / 2}
   223→          r={radius}
   224→          fill="none"
   225→          stroke="currentColor"
   226→          strokeWidth={strokeWidth}
   227→          className="text-emerald-100 dark:text-emerald-900/30"
   228→        />
   229→        <circle
   230→          cx={size / 2}
   231→          cy={size / 2}
   232→          r={radius}
   233→          fill="none"
   234→          strokeWidth={strokeWidth}
   235→          strokeLinecap="round"
   236→          strokeDasharray={circumference}
   237→          strokeDashoffset={offset}
   238→          className={`${strokeColor} transition-all duration-1000 ease-out`}
   239→        />
   240→      </svg>
   241→      <div className="absolute flex flex-col items-center justify-center" style={{ width: size, height: size }}>
   242→        <span className={`text-2xl font-bold ${color}`}>{Math.round(pct * 100)}%</span>
   243→        <span className="text-[10px] text-muted-foreground">{label}</span>
   244→      </div>
   245→    </div>
   246→  );
   247→}

   253→export default function AnalyticsPage() {
   254→  // ─── Live Firestore Data ──────────────────────────────────────────────────
   255→  const { data: clients, loading: clientsLoading } = useFireClients();
   256→  const { data: invoices, loading: invoicesLoading } = useFireInvoices();
   257→  const { data: returns, loading: returnsLoading } = useFireReturns();
   258→  const { data: reconciliations, loading: reconciliationsLoading } = useFireReconciliations();
   259→  const { data: firm, loading: firmLoading } = useFireFirm();
   260→
   261→  const isLoading =
   262→    clientsLoading || invoicesLoading || returnsLoading || reconciliationsLoading || firmLoading;
   263→
   264→  // ─── Computed: Monthly Tax Processed ──────────────────────────────────────
   265→  const monthlyTaxData: PeriodTax[] = useMemo(() => {
   266→    if (!invoices.length) return [];
   267→    const map = new Map<string, PeriodTax>();
   268→    for (const inv of invoices) {
   269→      const p = inv.period || 'unknown';
   270→      if (!map.has(p)) {
   271→        map.set(p, { period: p, label: p === 'unknown' ? 'Unknown' : periodToLabel(p), totalTax: 0, cgst: 0, sgst: 0, igst: 0 });
   272→      }
   273→      const entry = map.get(p)!;
   274→      entry.cgst += inv.cgst || 0;
   275→      entry.sgst += inv.sgst || 0;
   276→      entry.igst += inv.igst || 0;
   277→      entry.totalTax += (inv.cgst || 0) + (inv.sgst || 0) + (inv.igst || 0);
   278→    }
   279→    return Array.from(map.values()).sort((a, b) => a.period.localeCompare(b.period));
   280→  }, [invoices]);
   281→
   282→  // ─── Computed: Client Growth ──────────────────────────────────────────────
   283→  const clientStats = useMemo(() => {
   284→    const total = clients.length;
   285→    const active = clients.filter((c) => c.status === 'active').length;
   286→    const inactive = clients.filter((c) => c.status === 'inactive').length;
   287→    const suspended = clients.filter((c) => c.status === 'suspended').length;
   288→    return { total, active, inactive, suspended };
   289→  }, [clients]);
   290→
   291→  // ─── Computed: Return Filing Rate ─────────────────────────────────────────
   292→  const filingStats = useMemo(() => {
   293→    const total = returns.length;
   294→    const filed = returns.filter((r) => r.status === 'filed').length;
   295→    const pending = returns.filter((r) => r.status === 'draft' || r.status === 'in_progress').length;
   296→    const overdue = returns.filter((r) => r.status === 'overdue').length;
   297→    const rate = total > 0 ? (filed / total) * 100 : 0;
   298→    return { total, filed, pending, overdue, rate };
   299→  }, [returns]);
   300→
   301→  // ─── Computed: Compliance Trend ───────────────────────────────────────────
   302→  const complianceTrend: CompliancePoint[] = useMemo(() => {
   303→    if (!returns.length) return [];
   304→    const map = new Map<string, { totalScore: number; count: number }>();
   305→    for (const ret of returns) {
   306→      const p = ret.period || 'unknown';
   307→      if (!map.has(p)) {
   308→        map.set(p, { totalScore: 0, count: 0 });
   309→      }
   310→      const entry = map.get(p)!;
   311→      // Use issuesFound to derive a compliance indicator: fewer issues = higher compliance
   312→      const score = ret.totalInvoices > 0
   313→        ? Math.max(0, 100 - (ret.criticalErrors * 20 + ret.issuesFound * 5 + ret.warnings * 2))
   314→        : 100;
   315→      entry.totalScore += score;
   316→      entry.count += 1;
   317→    }
   318→    return Array.from(map.entries())
   319→      .map(([period, { totalScore, count }]) => ({
   320→        period,
   321→        label: period === 'unknown' ? 'Unknown' : periodToLabel(period),
   322→        score: count > 0 ? Math.round(totalScore / count) : 0,
   323→      }))
   324→      .sort((a, b) => a.period.localeCompare(b.period));
   325→  }, [returns]);
   326→
   327→  // ─── Computed: Revenue Trend ──────────────────────────────────────────────
   328→  const revenueTrend: PeriodRevenue[] = useMemo(() => {
   329→    if (!invoices.length) return [];
   330→    const map = new Map<string, number>();
   331→    for (const inv of invoices) {
   332→      const p = inv.period || 'unknown';
   333→      map.set(p, (map.get(p) || 0) + (inv.totalAmount || 0));
   334→    }
   335→    return Array.from(map.entries())
   336→      .map(([period, totalRevenue]) => ({
   337→        period,
   338→        label: period === 'unknown' ? 'Unknown' : periodToLabel(period),
   339→        totalRevenue,
   340→      }))
   341→      .sort((a, b) => a.period.localeCompare(b.period));
   342→  }, [invoices]);
   343→
   344→  // ─── Computed: Top Clients by Tax Volume ──────────────────────────────────
   345→  const topClients: TopClient[] = useMemo(() => {
   346→    return [...clients]
   347→      .sort((a, b) => (b.totalTaxPaid || 0) - (a.totalTaxPaid || 0))
   348→      .slice(0, 5)
   349→      .map((c) => ({
   350→        id: c.id,
   351→        tradeName: c.tradeName,
   352→        gstin: c.gstin,
   353→        totalTaxPaid: c.totalTaxPaid || 0,
   354→        invoiceCount: c.invoiceCount || 0,
   355→        healthScore: c.healthScore || 0,
   356→        status: c.status,
   357→      }));
   358→  }, [clients]);
   359→
   360→  // ─── Computed: Reconciliation Summary ─────────────────────────────────────
   361→  const reconStats = useMemo(() => {
   362→    const total = reconciliations.length;
   363→    const completed = reconciliations.filter((r) => r.status === 'completed').length;
   364→    const avgMatch = total > 0
   365→      ? Math.round(
   366→          reconciliations.reduce((sum, r) => {
   367→            const rate = r.totalRecords > 0 ? (r.matched / r.totalRecords) * 100 : 0;
   368→            return sum + rate;
   369→          }, 0) / total
   370→        )
   371→      : 0;
   372→    const totalITCDiff = reconciliations.reduce((sum, r) => sum + (r.gstDifference || 0), 0);
   373→    return { total, completed, avgMatch, totalITCDiff };
   374→  }, [reconciliations]);
   375→
   376→  // ─── Summary Metrics ─────────────────────────────────────────────────────
   377→  const totalTaxVolume = useMemo(
   378→    () => invoices.reduce((sum, inv) => sum + (inv.cgst || 0) + (inv.sgst || 0) + (inv.igst || 0) + (inv.cess || 0), 0),
   379→    [invoices]
   380→  );
   381→
   382→  const totalRevenue = useMemo(
   383→    () => invoices.reduce((sum, inv) => sum + (inv.totalAmount || 0), 0),
   384→    [invoices]
   385→  );
   386→
   387→  // ─── Render ───────────────────────────────────────────────────────────────
   388→
   389→  if (isLoading) {
   390→    return (
   391→      <div className="p-4 md:p-6 space-y-6">
   392→        {/* Header skeleton */}
   393→        <div className="space-y-2">
   394→          <Skeleton className="h-8 w-64" />
   395→          <Skeleton className="h-4 w-96" />
   396→        </div>
   397→        {/* Summary row */}
   398→        <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
   399→          {Array.from({ length: 4 }).map((_, i) => (
   400→            <Skeleton key={i} className="h-24 rounded-xl" />
   401→          ))}
   402→        </div>
   403→        {/* Chart grid */}
   404→        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-6">
   405→          {Array.from({ length: 6 }).map((_, i) => (
   406→            <SkeletonCard key={i} />
   407→          ))}
   408→        </div>
   409→      </div>
   410→    );
   411→  }
   412→
   413→  return (
   414→    <ScrollArea className="h-full">
   415→      <div className="p-4 md:p-6 space-y-6">
   416→        {/* ─── Header ──────────────────────────────────────────────────────── */}
   417→        <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-2">
   418→          <div>
   419→            <h1 className="text-2xl font-bold tracking-tight text-emerald-900 dark:text-emerald-100">
   420→              Analytics
   421→            </h1>
   422→            <p className="text-sm text-muted-foreground mt-1">
   423→              Real-time insights from your GST compliance data
   424→              {firm && (
   425→                <span className="text-emerald-600 dark:text-emerald-400 ml-1">
   426→                  &middot; {firm.firmName}
   427→                </span>
   428→              )}
   429→            </p>
   430→          </div>
   431→          <Badge
   432→            variant="outline"
   433→            className="w-fit border-emerald-200 dark:border-emerald-800 text-emerald-700 dark:text-emerald-300"
   434→          >
   435→            <Activity className="h-3 w-3 mr-1" />
   436→            Live Data
   437→          </Badge>
   438→        </div>
   439→
   440→        {/* ─── Summary Metric Cards ────────────────────────────────────────── */}
   441→        <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
   442→          <motion.div custom={0} variants={cardVariants} initial="hidden" animate="visible">
   443→            <Card className="border-emerald-100 dark:border-emerald-900/30 bg-gradient-to-br from-emerald-50 to-white dark:from-emerald-950/30 dark:to-card">
   444→              <CardContent className="pt-0">
   445→                <div className="flex items-center gap-2 mb-1">
   446→                  <div className="rounded-md bg-emerald-100 dark:bg-emerald-900/50 p-1.5">
   447→                    <IndianRupee className="h-4 w-4 text-emerald-600 dark:text-emerald-400" />
   448→                  </div>
   449→                  <span className="text-xs text-muted-foreground">Total Tax Volume</span>
   450→                </div>
   451→                <p className="text-xl font-bold text-emerald-900 dark:text-emerald-100">
   452→                  {formatCurrency(totalTaxVolume)}
   453→                </p>
   454→              </CardContent>
   455→            </Card>
   456→          </motion.div>
   457→
   458→          <motion.div custom={1} variants={cardVariants} initial="hidden" animate="visible">
   459→            <Card className="border-emerald-100 dark:border-emerald-900/30">
   460→              <CardContent className="pt-0">
   461→                <div className="flex items-center gap-2 mb-1">
   462→                  <div className="rounded-md bg-emerald-100 dark:bg-emerald-900/50 p-1.5">
   463→                    <Users className="h-4 w-4 text-emerald-600 dark:text-emerald-400" />
   464→                  </div>
   465→                  <span className="text-xs text-muted-foreground">Active Clients</span>
   466→                </div>
   467→                <p className="text-xl font-bold">
   468→                  {formatNumber(clientStats.active)}
   469→                  <span className="text-sm font-normal text-muted-foreground"> / {formatNumber(clientStats.total)}</span>
   470→                </p>
   471→              </CardContent>
   472→            </Card>
   473→          </motion.div>
   474→
   475→          <motion.div custom={2} variants={cardVariants} initial="hidden" animate="visible">
   476→            <Card className="border-emerald-100 dark:border-emerald-900/30">
   477→              <CardContent className="pt-0">
   478→                <div className="flex items-center gap-2 mb-1">
   479→                  <div className="rounded-md bg-emerald-100 dark:bg-emerald-900/50 p-1.5">
   480→                    <FileCheck className="h-4 w-4 text-emerald-600 dark:text-emerald-400" />
   481→                  </div>
   482→                  <span className="text-xs text-muted-foreground">Filing Rate</span>
   483→                </div>
   484→                <p className="text-xl font-bold">
   485→                  {filingStats.rate.toFixed(1)}%
   486→                  <span className="text-sm font-normal text-muted-foreground"> filed</span>
   487→                </p>
   488→              </CardContent>
   489→            </Card>
   490→          </motion.div>
   491→
   492→          <motion.div custom={3} variants={cardVariants} initial="hidden" animate="visible">
   493→            <Card className="border-emerald-100 dark:border-emerald-900/30">
   494→              <CardContent className="pt-0">
   495→                <div className="flex items-center gap-2 mb-1">
   496→                  <div className="rounded-md bg-emerald-100 dark:bg-emerald-900/50 p-1.5">
   497→                    <Shield className="h-4 w-4 text-emerald-600 dark:text-emerald-400" />
   498→                  </div>
   499→                  <span className="text-xs text-muted-foreground">Match Rate</span>
   500→                </div>
   501→                <p className="text-xl font-bold">
   502→                  {reconStats.avgMatch}%
   503→                  <span className="text-sm font-normal text-muted-foreground"> avg match</span>
   504→                </p>
   505→              </CardContent>
   506→            </Card>
   507→          </motion.div>
   508→        </div>
   509→
   510→        {/* ─── Chart Grid ──────────────────────────────────────────────────── */}
   511→        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-6">
   512→
   513→          {/* ── 1. Monthly Tax Processed ────────────────────────────────────── */}
   514→          <motion.div custom={4} variants={cardVariants} initial="hidden" animate="visible">
   515→            <Card className="border-emerald-100 dark:border-emerald-900/30">
   516→              <CardHeader>
   517→                <div className="flex items-center gap-2">
   518→                  <BarChart3 className="h-4 w-4 text-emerald-600 dark:text-emerald-400" />
   519→                  <CardTitle className="text-base">Monthly Tax Processed</CardTitle>
   520→                </div>
   521→                <p className="text-xs text-muted-foreground">
   522→                  CGST + SGST + IGST per return period
   523→                </p>
   524→              </CardHeader>
   525→              <CardContent>
   526→                {monthlyTaxData.length === 0 ? (
   527→                  <EmptyState
   528→                    icon={BarChart3}
   529→                    title="No tax data yet"
   530→                    description="Tax analytics will appear once invoices are processed"
