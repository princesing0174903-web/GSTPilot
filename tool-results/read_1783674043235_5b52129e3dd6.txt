    17→//     "Missing or insufficient permissions" wall. Genuine errors (network,

    47→// ─── Permission-error detection (graceful degradation) ───────────────────────
    48→
    49→/**
    50→ * True if the given Firestore error is a permission / auth failure that we
    51→ * should treat as "no data available" rather than a hard error. This is the
    52→ * core of the never-show-a-permission-wall contract.
    53→ */
    54→function isPermissionError(err: unknown): boolean {
    55→  if (!err) return false;
    56→  const e = err as { code?: string; message?: string };
    57→  const code = (e.code || '').toLowerCase();
    58→  const message = (e.message || '').toLowerCase();
    59→  if (
    60→    code === 'permission-denied' ||
    61→    code === 'unauthenticated' ||
    62→    code === 'auth/operation-not-allowed' ||
    63→    code === 'auth/user-not-found'
    64→  ) {
    65→    return true;
    66→  }
    67→  if (
    68→    message.includes('missing or insufficient permissions') ||
    69→    message.includes('permission-denied') ||
    70→    message.includes('insufficient permissions') ||
    71→    message.includes('not authorized') ||
    72→    message.includes('unauthenticated')
    73→  ) {
    74→    return true;
    75→  }
    76→  return false;
    77→}

    79→// ─── Timestamp Converter ─────────────────────────────────────────────────────
    80→
    81→function convertDoc<T extends Record<string, unknown>>(snapData: Record<string, unknown>): T {
    82→  const converted: Record<string, unknown> = {};
    83→  for (const [key, value] of Object.entries(snapData)) {
    84→    if (value && typeof value === 'object' && 'toDate' in value && typeof (value as { toDate: () => Date }).toDate === 'function') {
    85→      converted[key] = (value as { toDate: () => Date }).toDate().toISOString();
    86→    } else if (value && typeof value === 'object' && !Array.isArray(value) && !(value instanceof Date)) {
    87→      // Recursively convert nested objects
    88→      converted[key] = convertDoc(value as Record<string, unknown>);
    89→    } else {
    90→      converted[key] = value;
    91→    }
    92→  }
    93→  return converted as T;
    94→}

    96→// ─── Generic Collection Hook ─────────────────────────────────────────────────
    97→//
    98→// The hook resolves the current `organizationId` from OrgContext. If the org
    99→// isn't resolved yet (loading) OR we're in preview mode (no org), the hook
   100→// returns empty data without subscribing. This guarantees no permission errors
   101→// ever reach the UI in preview/offline mode.
   102→
   103→function useFirestoreCollection<T>(
   104→  collectionName: CollectionName,
   105→  constraints: QueryConstraint[] = [],
   106→  deps: unknown[] = [],
   107→): { data: Array<T & { id: string }>; loading: boolean; error: string | null } {
   108→  const { user } = useAuth();
   109→  const { organization, isPreviewMode, loading: orgLoading } = useOrg();
   110→  const organizationId = organization?.id ?? null;
   111→
   112→  const [data, setData] = useState<Array<T & { id: string }>>([]);
   113→  const [loading, setLoading] = useState(true);
   114→  const [error, setError] = useState<string | null>(null);
   115→
   116→  useEffect(() => {
   117→    // No user → nothing to subscribe to (empty data).
   118→    if (!user) {
   119→      setData([]);
   120→      setLoading(false);
   121→      setError(null);
   122→      return;
   123→    }
   124→
   125→    // Org still resolving — stay in loading state, but don't subscribe yet.
   126→    if (orgLoading && !organizationId) {
   127→      return;
   128→    }
   129→
   130→    // Preview mode (no real org) OR no organization at all → empty data,
   131→    // no subscription. This is the preview-mode contract.
   132→    if (!organizationId || isPreviewMode || organizationId === 'preview-org') {
   133→      setData([]);
   134→      setLoading(false);
   135→      setError(null);
   136→      return;
   137→    }
   138→
   139→    let unsub: Unsubscribe | null = null;
   140→
   141→    // Scope EVERY query by organizationId. This is required by the Firestore
   142→    // security rules — unbounded collection reads are denied.
   143→    const baseConstraints: QueryConstraint[] = [
   144→      where('organizationId', '==', organizationId),
   145→      ...constraints,
   146→    ];
   147→
   148→    const q = query(collection(db, collectionName), ...baseConstraints);
   149→
   150→    unsub = onSnapshot(q, (snap) => {
   151→      const items = snap.docs.map(d => ({
   152→        id: d.id,
   153→        ...convertDoc<T>(d.data() as Record<string, unknown>),
   154→      }));
   155→      setData(items);
   156→      setLoading(false);
   157→      setError(null);
   158→    }, (err) => {
   159→      console.warn(`[Firestore] ${collectionName} subscription error:`, err);
   160→      // Permission errors → graceful degradation (empty data, no error wall).
   161→      if (isPermissionError(err)) {
   162→        setData([]);
   163→        setError(null);
   164→        setLoading(false);
   165→        return;
   166→      }
   167→      // Genuine errors (network, missing index) still surface.
   168→      setError(err.message);
   169→      setLoading(false);
   170→    });
   171→
   172→    return () => {
   173→      if (unsub) unsub();
   174→    };
   175→  }, [user, organizationId, isPreviewMode, orgLoading, ...deps]);
   176→
   177→  // When user is null, return empty data (not loading)
   178→  if (!user && loading) {
   179→    return { data: [], loading: false, error: null };
   180→  }
   181→
   182→  return { data, loading, error };
   183→}

   185→// ─── Generic Document Hook ───────────────────────────────────────────────────
   186→
   187→function useFirestoreDoc<T>(
   188→  collectionName: CollectionName,
   189→  docId: string | null,
   190→): { data: (T & { id: string }) | null; loading: boolean; error: string | null } {
   191→  const { user } = useAuth();
   192→  const { organization, isPreviewMode } = useOrg();
   193→  const organizationId = organization?.id ?? null;
   194→
   195→  const [data, setData] = useState<(T & { id: string }) | null>(null);
   196→  const [loading, setLoading] = useState(!!docId);
   197→  const [error, setError] = useState<string | null>(null);
   198→
   199→  useEffect(() => {
   200→    if (!docId) {
   201→      setData(null);
   202→      setLoading(false);
   203→      return;
   204→    }
   205→
   206→    // Preview mode (no real org) → no doc, no subscription.
   207→    if (!user || !organizationId || isPreviewMode || organizationId === 'preview-org') {
   208→      setData(null);
   209→      setLoading(false);
   210→      setError(null);
   211→      return;
   212→    }
   213→
   214→    const unsub = onSnapshot(doc(db, collectionName, docId), (snap) => {
   215→      if (!snap.exists()) {
   216→        setData(null);
   217→        setLoading(false);
   218→        return;
   219→      }
   220→      setData({
   221→        id: snap.id,
   222→        ...convertDoc<T>(snap.data() as Record<string, unknown>),
   223→      });
   224→      setLoading(false);
   225→      setError(null);
   226→    }, (err) => {
   227→      console.warn(`[Firestore] ${collectionName}/${docId} doc error:`, err);
   228→      if (isPermissionError(err)) {
   229→        setData(null);
   230→        setError(null);
   231→        setLoading(false);
   232→        return;
   233→      }
   234→      setError(err.message);
   235→      setLoading(false);
   236→    });
   237→
   238→    return () => {
   239→      if (unsub) unsub();
   240→    };
   241→  }, [collectionName, docId, user, organizationId, isPreviewMode]);
   242→
   243→  return { data, loading, error };
   244→}

   503→// ─── Executive Scores (computed from live data) ─────────────────────────────
   504→
   505→export function useFirmExecutiveScores(): { scores: FirmExecutiveScores; loading: boolean } {
   506→  const { data: clients } = useFireClients();
   507→  const { data: returns } = useFireReturns();
   508→  const { data: invoices } = useFireInvoices();
   509→  const { data: reconciliations } = useFireReconciliations();
   510→  const { data: activities } = useFireRecentActivities(50);
   511→
   512→  const loading = clients.length === 0 && returns.length === 0;
   513→
   514→  const scores = useMemo<FirmExecutiveScores>(() => {
   515→    if (loading) {
   516→      return { firmHealth: 0, revenue: 0, compliance: 0, teamEfficiency: 0, clientSatisfaction: 0, cashFlow: 0 };
   517→    }
   518→
   519→    // Firm Health: weighted avg of client health scores
   520→    const activeClients = clients.filter(c => c.status === 'active');
   521→    const avgHealth = activeClients.length > 0
   522→      ? activeClients.reduce((s, c) => s + (c.healthScore || 0), 0) / activeClients.length
   523→      : 50;
   524→
   525→    // Revenue Score: based on tax volume
   526→    const totalTax = invoices.reduce((s, i) => s + (i.totalTax || 0) + (i.cgst || 0) + (i.sgst || 0) + (i.igst || 0), 0);
   527→    const revenueScore = Math.min(100, Math.round((totalTax / 1000000) * 100));
   528→
   529→    // Compliance Score: filed vs total returns
   530→    const filedReturns = returns.filter(r => r.status === 'filed').length;
   531→    const complianceScore = returns.length > 0 ? Math.round((filedReturns / returns.length) * 100) : 50;
   532→
   533→    // Team Efficiency: based on completed activities
   534→    const completedActivities = activities.filter(a => a.type.includes('filed') || a.type.includes('completed') || a.type.includes('resolved')).length;
   535→    const teamEfficiency = activities.length > 0 ? Math.min(100, Math.round((completedActivities / Math.max(activities.length, 1)) * 100)) : 50;
   536→
   537→    // Client Satisfaction: inverse of overdue + health
   538→    const overdueClients = activeClients.filter(c => (c.pendingReturnCount || 0) > 0).length;
   539→    const clientSatisfaction = activeClients.length > 0 ? Math.round(((activeClients.length - overdueClients) / activeClients.length) * 100) : 50;
   540→
   541→    // Cash Flow: based on reconciliation match rate
   542→    const matchedRecons = reconciliations.filter(r => r.matched > 0);
   543→    const totalMatched = matchedRecons.reduce((s, r) => s + r.matched, 0);
   544→    const totalRecords = matchedRecons.reduce((s, r) => s + r.totalRecords, 0);
   545→    const cashFlow = totalRecords > 0 ? Math.round((totalMatched / totalRecords) * 100) : 50;
   546→
   547→    return {
   548→      firmHealth: Math.round(avgHealth) || 50,
   549→      revenue: revenueScore || 50,
   550→      compliance: complianceScore || 50,
   551→      teamEfficiency: teamEfficiency || 50,
   552→      clientSatisfaction: clientSatisfaction || 50,
   553→      cashFlow: cashFlow || 50,
   554→    };
   555→  }, [clients, returns, invoices, reconciliations, activities, loading]);
   556→
   557→  return { scores, loading };
   558→}
