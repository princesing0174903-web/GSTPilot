    52→function getDaysRemaining(dueDateStr: string): number {
    53→  const dueDate = new Date(dueDateStr);
    54→  const today = new Date();
    55→  today.setHours(0, 0, 0, 0);
    56→  dueDate.setHours(0, 0, 0, 0);
    57→  return Math.ceil((dueDate.getTime() - today.getTime()) / (1000 * 60 * 60 * 24));
    58→}
    59→
    60→function formatDaysRemaining(days: number): string {
    61→  if (days < 0) return `${Math.abs(days)}d overdue`;
    62→  if (days === 0) return 'Due today';
    63→  if (days === 1) return 'Tomorrow';
    64→  return `${days}d left`;
    65→}
    66→
    67→function getInitials(name: string): string {
    68→  return name.split(' ').map(w => w[0]).join('').slice(0, 2).toUpperCase();
    69→}
    70→
    71→function formatINR(amount: number): string {
    72→  return amount.toLocaleString('en-IN');
    73→}
    74→
    75→function timeAgo(dateStr: string): string {
    76→  const now = new Date();
    77→  const date = new Date(dateStr);
    78→  const seconds = Math.floor((now.getTime() - date.getTime()) / 1000);
    79→  if (seconds < 60) return 'just now';
    80→  const minutes = Math.floor(seconds / 60);
    81→  if (minutes < 60) return `${minutes}m ago`;
    82→  const hours = Math.floor(minutes / 60);
    83→  if (hours < 24) return `${hours}h ago`;
    84→  const days = Math.floor(hours / 24);
    85→  if (days < 7) return `${days}d ago`;
    86→  return date.toLocaleDateString('en-IN', { day: 'numeric', month: 'short' });
    87→}

   134→function activityIcon(type: string): { icon: React.ReactNode; color: string; bg: string } {
   135→  if (type.includes('filed')) return { icon: <CheckCircle2 className="h-3.5 w-3.5" />, color: 'text-emerald-600', bg: 'bg-emerald-50' };
   136→  if (type.includes('upload') || type.includes('document')) return { icon: <Upload className="h-3.5 w-3.5" />, color: 'text-blue-600', bg: 'bg-blue-50' };
   137→  if (type.includes('review')) return { icon: <ClipboardCheck className="h-3.5 w-3.5" />, color: 'text-violet-600', bg: 'bg-violet-50' };
   138→  if (type.includes('client')) return { icon: <Users className="h-3.5 w-3.5" />, color: 'text-sky-600', bg: 'bg-sky-50' };
   139→  if (type.includes('reconcil') || type.includes('mismatch')) return { icon: <AlertTriangle className="h-3.5 w-3.5" />, color: 'text-amber-600', bg: 'bg-amber-50' };
   140→  if (type.includes('invoice') || type.includes('extract')) return { icon: <FileText className="h-3.5 w-3.5" />, color: 'text-teal-600', bg: 'bg-teal-50' };
   141→  return { icon: <Activity className="h-3.5 w-3.5" />, color: 'text-slate-500', bg: 'bg-slate-50' };
   142→}
