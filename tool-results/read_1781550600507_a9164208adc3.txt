   483→/** Auto-create draft returns for current period */
   484→async function autoCreateDraftReturns(firmId: string, clientId: string): Promise<void> {
   485→  const now = new Date();
   486→  const period = `${String(now.getMonth() + 1).padStart(2, '0')}-${now.getFullYear()}`;
   487→  const fy = now.getMonth() >= 3 ? `${now.getFullYear()}-${String(now.getFullYear() + 1).slice(2)}` : `${now.getFullYear() - 1}-${String(now.getFullYear()).slice(2)}`;
   488→
   489→  for (const returnType of ['GSTR-1', 'GSTR-3B'] as const) {
   490→    // Check if return already exists
   491→    const q = query(
   492→      collection(db, COLLECTIONS.RETURNS),
   493→      where('clientId', '==', clientId),
   494→      where('returnType', '==', returnType),
   495→      where('period', '==', period),
   496→      limit(1)
   497→    );
   498→    const existing = await getDocs(q);
   499→    if (existing.empty) {
   500→      await createReturn({
   501→        clientId,
   502→        returnType,
   503→        period,
   504→        financialYear: fy,
   505→        status: 'draft',
   506→        filedDate: null,
   507→        acknowledgmentNumber: null,
   508→        totalInvoices: 0,
   509→        readyForFiling: 0,
   510→        issuesFound: 0,
   511→        criticalErrors: 0,
   512→        warnings: 0,
   513→        totalTaxableValue: 0,
   514→        totalTax: 0,
   515→        jsonPayload: null,
   516→        assignedTo: null,
   517→        reviewedBy: null,
   518→      });
   519→    }
   520→  }
   521→}

   523→/** Update draft return when an invoice is added */
   524→async function updateDraftReturnForInvoice(firmId: string, clientId: string, returnType: string, period: string, taxableValue: number, taxAmount: number): Promise<void> {
   525→  try {
   526→    const q = query(
   527→      collection(db, COLLECTIONS.RETURNS),
   528→      where('clientId', '==', clientId),
   529→      where('returnType', '==', returnType),
   530→      where('period', '==', period),
   531→      limit(1)
   532→    );
   533→    const snap = await getDocs(q);
   534→    if (!snap.empty) {
   535→      const returnDoc = snap.docs[0];
   536→      const data = returnDoc.data() as FirestoreReturn;
   537→      await updateDoc(returnDoc.ref, {
   538→        totalInvoices: (data.totalInvoices || 0) + 1,
   539→        totalTaxableValue: (data.totalTaxableValue || 0) + taxableValue,
   540→        totalTax: (data.totalTax || 0) + taxAmount,
   541→        updatedAt: serverTimestamp(),
   542→      });
   543→    }
   544→  } catch (e) {
   545→    console.warn('[Workflow] Failed to update draft return:', e);
   546→  }
   547→}

   550→// Reconciliation:
   551→//   - Generates mismatch records

   555→export async function createReconciliation(data: {
   556→  clientId: string;
   557→  period: string;
   558→  sources: string;
   559→}): Promise<string> {
   560→  const firmId = currentFirmId();
   561→  const reconId = generateId();
   562→  const reconRef = doc(db, COLLECTIONS.RECONCILIATIONS, reconId);
   563→
   564→  // Simulate reconciliation results
   565→  const totalRecords = Math.floor(Math.random() * 50) + 10;
   566→  const matched = Math.floor(totalRecords * (0.6 + Math.random() * 0.25));
   567→  const partialMatches = Math.floor((totalRecords - matched) * 0.4);
   568→  const unmatched = totalRecords - matched - partialMatches;
   569→  const highRisk = Math.floor(unmatched * 0.3);
   570→
   571→  const mismatches: ReconMismatch[] = [];
   572→  for (let i = 0; i < unmatched + partialMatches; i++) {
   573→    const booksAmount = Math.floor(Math.random() * 100000) + 5000;
   574→    const diff = Math.floor(booksAmount * (Math.random() * 0.15));
   575→    mismatches.push({
   576→      invoiceNumber: `INV-${String(i + 1).padStart(4, '0')}`,
   577→      invoiceDate: new Date().toISOString().split('T')[0],
   578→      sourceGstin: '27AAACR5055K1ZB',
   579→      matchedGstin: i < partialMatches ? '27AAACR5055K1ZB' : null,
   580→      matchStatus: i < partialMatches ? 'partial_match' as MatchStatus : 'missing_in_books' as MatchStatus,
   581→      matchScore: i < partialMatches ? 70 + Math.floor(Math.random() * 20) : 0,
   582→      booksAmount,
   583→      portalAmount: booksAmount + (Math.random() > 0.5 ? diff : -diff),
   584→      difference: diff,
   585→      reason: i < partialMatches ? 'Amount mismatch' : 'Invoice not found in GSTR-2B',
   586→      resolved: false,
   587→      resolvedBy: null,
   588→      resolvedAt: null,
   589→    });
   590→  }
   591→
   592→  const gstDifference = mismatches.reduce((sum, m) => sum + m.difference, 0);
   593→
   594→  const reconData: FirestoreReconciliation = {
   595→    reconId,
   596→    firmId: firmId || '',
   597→    clientId: data.clientId,
   598→    period: data.period,
   599→    sources: data.sources,
   600→    status: 'completed',
   601→    totalRecords,
   602→    matched,
   603→    unmatched,
   604→    partialMatches,
   605→    highRisk,
   606→    gstDifference,
   607→    mismatches,
   608→    runBy: currentUserId(),
   609→    createdAt: serverTimestamp(),
   610→    updatedAt: serverTimestamp(),
   611→  };
   612→
   613→  await setDoc(reconRef, reconData);
   614→
   615→  addActivity({
   616→    type: 'reconciliation_run',
   617→    title: 'Reconciliation completed',
   618→    description: `${data.sources}: ${matched}/${totalRecords} matched, ${unmatched} unmatched`,
   619→    clientId: data.clientId,
   620→    entityType: COLLECTIONS.RECONCILIATIONS,
   621→    entityId: reconId,
   622→    metadata: { matched, unmatched, gstDifference },
   623→  });
   624→
   625→  // Generate AI recommendations for mismatches
   626→  if (highRisk > 0) {
   627→    generateAIRecommendations(firmId || '', data.clientId, reconId, mismatches);
   628→  }
   629→
   630→  addNotification({
   631→    type: 'mismatch_found',
   632→    priority: highRisk > 3 ? 'urgent' : 'high',
   633→    title: 'Reconciliation mismatches found',
   634→    message: `${unmatched} unmatched and ${partialMatches} partial matches found. ITC difference: ₹${gstDifference.toLocaleString('en-IN')}`,
   635→    entityType: COLLECTIONS.RECONCILIATIONS,
   636→    entityId: reconId,
   637→  });
   638→
   639→  return reconId;
   640→}

   642→export async function resolveMismatch(reconId: string, invoiceNumber: string): Promise<void> {
   643→  const reconRef = doc(db, COLLECTIONS.RECONCILIATIONS, reconId);

   654→  const resolvedCount = updatedMismatches.filter(m => m.resolved).length;

   847→async function recalculateComplianceScore(clientId: string): Promise<void> {
   848→  try {
   849→    const clientRef = doc(db, COLLECTIONS.CLIENTS, clientId);
   850→    const clientSnap = await getDoc(clientRef);
   851→    if (!clientSnap.exists()) return;
   852→
   853→    const client = clientSnap.data() as FirestoreClient;
   854→
   855→    // ═══ Client Health Engine ═══
   856→    // Score = GSTIN Validity + Filing Timeliness + Invoice Accuracy + Recon Match + Pending Returns
   857→    // Starting from 100, each factor deducts points
   858→
   859→    let score = 100;
   860→
   861→    // 1. GSTIN Validity (0-20 points deduction)
   862→    const gstinValid = client.complianceProfile.gstinValidity;
   863→    if (!gstinValid) score -= 20;
   864→
   865→    // 2. Filing Timeliness (0-30 points deduction)
   866→    const overdueReturns = client.complianceProfile.overdueReturns || 0;
   867→    const avgFilingDelay = client.complianceProfile.averageFilingDelay || 0;
   868→    score -= Math.min(overdueReturns * 10, 25);
   869→    score -= Math.min(avgFilingDelay * 2, 15);
   870→
   871→    // 3. Pending Returns (0-15 points deduction)
   872→    const pendingReturns = client.pendingReturnCount || 0;
   873→    if (pendingReturns > 5) score -= 15;
   874→    else if (pendingReturns > 3) score -= 10;
   875→    else if (pendingReturns > 0) score -= 3;
   876→
   877→    // 4. Invoice Accuracy & Reconciliation (0-20 points deduction)
   878→    // Use invoice count and document count as proxy
   879→    const invoiceCount = client.invoiceCount || 0;
   880→    const docCount = client.documentCount || 0;
   881→    if (invoiceCount > 0 && docCount === 0) score -= 10; // invoices but no docs = suspicious
   882→    if (invoiceCount > 100) score -= 5; // high volume = more risk
   883→
   884→    // 5. Filing compliance score from profile
   885→    const filingCompliance = client.complianceProfile.filingCompliance || 100;
   886→    if (filingCompliance < 50) score -= 15;
   887→    else if (filingCompliance < 80) score -= 5;
   888→
   889→    score = Math.max(0, Math.min(100, score));
   890→
   891→    // Determine risk category
   892→    const riskCategory = score >= 80 ? 'low' : score >= 60 ? 'medium' : 'high';
   893→
   894→    await updateDoc(clientRef, {
   895→      healthScore: score,
   896→      complianceProfile: {
   897→        ...client.complianceProfile,
   898→        filingCompliance: Math.min(filingCompliance, score),
   899→      },
   900→      updatedAt: serverTimestamp(),
   901→    });
   902→
   903→    // Generate AI recommendation for high-risk clients
   904→    if (riskCategory === 'high') {
   905→      try {
   906→        const aiRecs = await getDocs(query(
   907→          collection(db, COLLECTIONS.AI_RECOMMENDATIONS),
   908→          where('clientId', '==', clientId),
   909→          where('status', '==', 'active'),
   910→          limit(3),
   911→        ));
   912→        if (aiRecs.size === 0) {
   913→          addDoc(collection(db, COLLECTIONS.AI_RECOMMENDATIONS), {
   914→            recId: generateId(),
   915→            firmId: client.firmId,
   916→            clientId,
   917→            invoiceId: null,
   918→            reconId: null,
   919→            type: 'review_manually',
   920→            title: `Client at risk: ${client.tradeName}`,
   921→            description: `Health score is ${score}/100. ${overdueReturns > 0 ? `${overdueReturns} overdue returns. ` : ''}${pendingReturns > 0 ? `${pendingReturns} pending returns. ` : ''}${!gstinValid ? 'GSTIN may be invalid. ' : ''}Immediate attention recommended.`,
   922→            suggestedAction: 'Review client compliance status and file pending returns immediately.',
   923→            confidenceScore: 85,
   924→            riskLevel: 'high',
   925→            status: 'active',
   926→            dismissedBy: null,
   927→            appliedBy: null,
   928→            createdAt: serverTimestamp(),
   929→            updatedAt: serverTimestamp(),
   930→          });
   931→        }
   932→      } catch {}
   933→    }
   934→  } catch (e) {
   935→    console.warn('[Workflow] Compliance recalc failed:', e);
   936→  }
   937→}

   941→/** Subscribe to a Firestore collection with firm scoping */
   942→export function subscribeToCollection<T>(
   943→  collectionName: CollectionName,
   944→  callback: (data: Array<T & { id: string }>) => void,
   945→  ...constraints: QueryConstraint[]
   946→): Unsubscribe {
   947→  const firmId = currentFirmId();
   948→
   949→  const q = firmId
   950→    ? query(collection(db, collectionName), where('firmId', '==', firmId), ...constraints)
   951→    : query(collection(db, collectionName), ...constraints);
   952→
   953→  return onSnapshot(q, (snap) => {
   954→    const data = snap.docs.map(d => {
   955→      const docData = d.data();
   956→      const converted: Record<string, unknown> = { id: d.id };
   957→      for (const [key, value] of Object.entries(docData)) {
   958→        if (value && typeof value === 'object' && 'toDate' in value && typeof (value as { toDate: () => Date }).toDate === 'function') {
   959→          converted[key] = (value as { toDate: () => Date }).toDate().toISOString();
   960→        } else {
   961→          converted[key] = value;
   962→        }
   963→      }
   964→      return converted as T & { id: string };
   965→    });
   966→    callback(data);
   967→  }, (error) => {
   968→    console.warn(`[Firestore] Subscription error for ${collectionName}:`, error);
   969→    callback([]);
   970→  });
   971→}

   973→/** Subscribe to a single Firestore document */
   974→export function subscribeToDoc<T>(
   975→  collectionName: CollectionName,
   976→  docId: string,
   977→  callback: (data: (T & { id: string }) | null) => void,
   978→): Unsubscribe {

   980→    if (!snap.exists()) {
   981→      callback(null);
   982→      return;
   983→    }
   984→    const docData = snap.data();
   985→    const converted: Record<string, unknown> = { id: snap.id };
   986→    for (const [key, value] of Object.entries(docData)) {
   987→      if (value && typeof value === 'object' && 'toDate' in value && typeof (value as { toDate: () => Date }).toDate === 'function') {
   988→        converted[key] = (value as { toDate: () => Date }).toDate().toISOString();
   989→      } else {
   990→        converted[key] = value;
   991→      }
   992→    }
   993→    callback(converted as T & { id: string });
   994→  }, (error) => {
   995→    console.warn(`[Firestore] Doc subscription error for ${collectionName}/${docId}:`, error);
   996→    callback(null);
   997→  });
   998→}

  1000→/** Compute live dashboard metrics from Firestore data */
  1001→export function computeDashboardMetrics(
  1002→  clients: Array<FirestoreClient & { id: string }>,
  1003→  invoices: Array<FirestoreInvoice & { id: string }>,
  1004→  returns: Array<FirestoreReturn & { id: string }>,
  1005→  documents: Array<FirestoreDocument & { id: string }>,
  1006→  activities: Array<FirestoreActivity & { id: string }>,
  1007→): LiveDashboardMetrics {
  1008→  const activeClients = clients.filter(c => c.status === 'active').length;
  1009→  const filedReturns = returns.filter(r => r.status === 'filed').length;
  1010→  const pendingReturns = returns.filter(r => r.status !== 'filed').length;
  1011→  const readyToFile = returns.filter(r => ['validated', 'reviewed', 'generated'].includes(r.status)).length;
  1012→  const overdueReturns = returns.filter(r => {
  1013→    if (r.status === 'filed') return false;
  1014→    const periodDate = parsePeriod(r.period);
  1015→    return periodDate && periodDate < new Date();
  1016→  }).length;
  1017→
  1018→  const totalTaxVolume = invoices.reduce((sum, inv) => sum + (inv.totalAmount || 0), 0);
  1019→  const totalTax = invoices.reduce((sum, inv) => sum + (inv.cgst || 0) + (inv.sgst || 0) + (inv.igst || 0) + (inv.cess || 0), 0);
  1020→
  1021→  const criticalIssues = invoices.filter(inv => inv.riskLevel === 'critical').length +
  1022→    returns.filter(r => r.criticalErrors > 0).length;
  1023→  const warnings = invoices.filter(inv => inv.riskLevel === 'high' || inv.riskLevel === 'medium').length +
  1024→    returns.filter(r => r.warnings > 0).length;
  1025→
  1026→  const averageHealthScore = clients.length > 0
  1027→    ? Math.round(clients.reduce((sum, c) => sum + c.healthScore, 0) / clients.length)
  1028→    : 0;
  1029→
  1030→  const matchedInvoices = invoices.filter(inv => inv.matchStatus === 'perfect_match').length;
  1031→  const matchPercentage = invoices.length > 0 ? Math.round((matchedInvoices / invoices.length) * 100) : 100;
  1032→
  1033→  const riskyInvoices = invoices.filter(inv => ['high', 'critical'].includes(inv.riskLevel)).length;
  1034→  const riskPercentage = invoices.length > 0 ? Math.round((riskyInvoices / invoices.length) * 100) : 0;
  1035→
  1036→  const documentsProcessed = documents.filter(d => d.extractionStatus === 'completed').length;
  1037→  const extractionsPending = documents.filter(d => d.extractionStatus === 'pending' || d.extractionStatus === 'in_progress').length;
  1038→
  1039→  const recentActivities = [...activities]
  1040→    .sort((a, b) => {
  1041→      const aTime = typeof a.createdAt === 'string' ? new Date(a.createdAt).getTime() : 0;
  1042→      const bTime = typeof b.createdAt === 'string' ? new Date(b.createdAt).getTime() : 0;
  1043→      return bTime - aTime;
  1044→    })
  1045→    .slice(0, 10);
  1046→
  1047→  const upcomingFilings = returns
  1048→    .filter(r => r.status !== 'filed')
  1049→    .sort((a, b) => {
  1050→      const aTime = typeof a.createdAt === 'string' ? new Date(a.createdAt).getTime() : 0;
  1051→      const bTime = typeof b.createdAt === 'string' ? new Date(b.createdAt).getTime() : 0;
  1052→      return aTime - bTime;
  1053→    })
  1054→    .slice(0, 5);
  1055→
  1056→  return {
  1057→    totalClients: clients.length,
  1058→    activeClients,
  1059→    totalInvoices: invoices.length,
  1060→    totalTaxVolume,
  1061→    filedReturns,
  1062→    pendingReturns,
  1063→    overdueReturns,
  1064→    readyToFile,
  1065→    criticalIssues,
  1066→    warnings,
  1067→    averageHealthScore,
  1068→    matchPercentage,
  1069→    riskPercentage,
  1070→    documentsProcessed,
  1071→    extractionsPending,
  1072→    recentActivities,
  1073→    upcomingFilings,
  1074→  };
  1075→}

  1077→function parsePeriod(period: string | null): Date | null {
  1078→  if (!period) return null;
  1079→  try {
  1080→    const [mm, yyyy] = period.split('-').map(Number);
  1081→    if (!mm || !yyyy) return null;
  1082→    return new Date(yyyy, mm, 20); // Due date is typically 20th of next month
  1083→  } catch {
  1084→    return null;
  1085→  }
  1086→}

  1095→  factors: {
  1096→    gstinValidity: { score: number; maxScore: number; status: 'pass' | 'fail' };
  1097→    filingTimeliness: { score: number; maxScore: number; overdueReturns: number; avgDelay: number };
  1098→    pendingReturns: { score: number; maxScore: number; count: number };
  1099→    invoiceAccuracy: { score: number; maxScore: number; invoiceCount: number; docCount: number };
  1100→    reconciliation: { score: number; maxScore: number; matchRate: number };
  1101→  };
  1102→  recommendations: string[];
  1103→}
  1104→
  1105→export function computeClientHealthBreakdown(
  1106→  client: FirestoreClient & { id: string },
  1107→  invoices?: Array<FirestoreInvoice & { id: string }>,
  1108→  reconciliations?: Array<FirestoreReconciliation & { id: string }>,
  1109→): ClientHealthBreakdown {
  1110→  const recommendations: string[] = [];
  1111→
  1112→  // 1. GSTIN Validity (0-20)
  1113→  const gstinValid = client.complianceProfile?.gstinValidity ?? true;
  1114→  const gstinScore = gstinValid ? 20 : 0;
  1115→  if (!gstinValid) recommendations.push('Verify GSTIN status on the GST portal — it may be cancelled or suspended');
  1116→
  1117→  // 2. Filing Timeliness (0-30)
  1118→  const overdueReturns = client.complianceProfile?.overdueReturns || 0;
  1119→  const avgDelay = client.complianceProfile?.averageFilingDelay || 0;
  1120→  let filingScore = 30;
  1121→  filingScore -= Math.min(overdueReturns * 10, 25);
  1122→  filingScore -= Math.min(avgDelay * 2, 15);
  1123→  filingScore = Math.max(0, filingScore);
  1124→  if (overdueReturns > 0) recommendations.push(`${overdueReturns} overdue return(s) — file immediately to avoid late fees (₹50/day)`);
  1125→  if (avgDelay > 5) recommendations.push(`Average filing delay is ${avgDelay} days — set up reminders for upcoming deadlines`);
  1126→
  1127→  // 3. Pending Returns (0-15)
  1128→  const pendingCount = client.pendingReturnCount || 0;
  1129→  let pendingScore = 15;
  1130→  if (pendingCount > 5) pendingScore = 0;
  1131→  else if (pendingCount > 3) pendingScore = 5;
  1132→  else if (pendingCount > 0) pendingScore = 12;
  1133→  if (pendingCount > 3) recommendations.push(`${pendingCount} pending returns — prioritize filing to maintain compliance`);
  1134→
  1135→  // 4. Invoice Accuracy (0-20)
  1136→  const invoiceCount = client.invoiceCount || 0;
  1137→  const docCount = client.documentCount || 0;
  1138→  let invoiceScore = 20;
  1139→  if (invoiceCount > 0 && docCount === 0) { invoiceScore -= 10; recommendations.push('Invoices exist but no documents uploaded — upload source documents for audit trail'); }
  1140→  if (invoiceCount > 100) { invoiceScore -= 5; }
  1141→
  1142→  // 5. Reconciliation Match Rate (0-15)
  1143→  const clientRecos = reconciliations?.filter(r => r.clientId === client.id) || [];
  1144→  let matchRate = 100;
  1145→  let reconScore = 15;
  1146→  if (clientRecos.length > 0) {
  1147→    const totalRecords = clientRecos.reduce((sum, r) => sum + (r.totalRecords || 0), 0);
  1148→    const matched = clientRecos.reduce((sum, r) => sum + (r.matched || 0), 0);
  1149→    matchRate = totalRecords > 0 ? Math.round((matched / totalRecords) * 100) : 100;
  1150→    if (matchRate < 80) { reconScore = 5; recommendations.push(`Reconciliation match rate is ${matchRate}% — resolve mismatches to maximize ITC claims`); }
  1151→    else if (matchRate < 95) { reconScore = 10; recommendations.push(`Reconciliation match rate is ${matchRate}% — some mismatches need attention`); }
  1152→  }
  1153→
  1154→  const totalScore = gstinScore + filingScore + pendingScore + invoiceScore + reconScore;
  1155→  const riskCategory: 'low' | 'medium' | 'high' = totalScore >= 80 ? 'low' : totalScore >= 60 ? 'medium' : 'high';
  1156→
  1157→  if (recommendations.length === 0 && totalScore >= 80) {
  1158→    recommendations.push('Client is in good standing — continue regular monitoring');
  1159→  }
  1160→
  1161→  return {
  1162→    score: totalScore,
  1163→    riskCategory,
  1164→    factors: {
  1165→      gstinValidity: { score: gstinScore, maxScore: 20, status: gstinValid ? 'pass' : 'fail' },
  1166→      filingTimeliness: { score: filingScore, maxScore: 30, overdueReturns, avgDelay },
  1167→      pendingReturns: { score: pendingScore, maxScore: 15, count: pendingCount },
  1168→      invoiceAccuracy: { score: invoiceScore, maxScore: 20, invoiceCount, docCount },
  1169→      reconciliation: { score: reconScore, maxScore: 15, matchRate },
  1170→    },
  1171→    recommendations,
  1172→  };
  1173→}
