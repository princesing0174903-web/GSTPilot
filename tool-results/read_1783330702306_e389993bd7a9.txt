   674→function OnboardingScreen() {
   675→  const { user } = useAuth()
   676→  const { completeOnboarding } = useOrg()
   677→  const { setCurrentScreen, setCurrentView } = useApp()
   678→  const [saveError, setSaveError] = React.useState<string | null>(null)
   679→  const [isSubmitting, setIsSubmitting] = React.useState(false)
   680→
   681→  // ── Create the organization + owner membership via the org service ──
   682→  const createOrganizationForUser = async (
   683→    data: import('@/components/onboarding/OnboardingFlow').OnboardingData
   684→  ): Promise<{ orgId: string | null; error: string | null }> => {
   685→    if (!user) return { orgId: null, error: 'No authenticated user found.' }
   686→
   687→    const { createOrganization, updateUserProfile } = await import('@/lib/auth/organizations')
   688→    const { doc, setDoc, serverTimestamp } = await import('firebase/firestore')
   689→    const { db } = await import('@/lib/firebase')
   690→
   691→    // 1. Create the organization + owner membership + set currentOrganizationId.
   692→    const { organization, error: orgError } = await createOrganization({
   693→      name: data.firmName,
   694→      ownerId: user.id,
   695→      ownerEmail: user.email,
   696→      ownerDisplayName: data.fullName || user.name,
   697→      ownerPhotoURL: user.picture || null,
   698→      gstin: data.gstin || null,
   699→      plan: 'free',
   700→    })
   701→    if (orgError || !organization) {
   702→      return { orgId: null, error: orgError || 'Could not create your organization.' }
   703→    }
   704→
   705→    // 2. Enrich the user profile with onboarding metadata (best-effort).
   706→    try {
   707→      await updateUserProfile(user.id, {
   708→        displayName: data.fullName,
   709→        phone: data.phone,
   710→        company: data.firmName,
   711→        gstin: data.gstin || null,
   712→      })
   713→      // Persist the extended onboarding questionnaire for analytics.
   714→      await setDoc(doc(db, 'onboarding', user.id), {
   715→        ...data,
   716→        organizationId: organization.id,
   717→        completedAt: serverTimestamp(),
   718→      }, { merge: true })
   719→    } catch (err) {
   720→      console.warn('[Onboarding] Profile enrichment failed:', err)
   721→    }
   722→
   723→    return { orgId: organization.id, error: null }
   724→  }
   725→
   726→  const handleOnboardingComplete = async (
   727→    data: import('@/components/onboarding/OnboardingFlow').OnboardingData,
   728→    destination?: import('@/components/onboarding/OnboardingFlow').OnboardingDestination
   729→  ) => {
   730→    setSaveError(null)
   731→    if (!user) {
   732→      setSaveError('No authenticated user found. Please sign in again.')
   733→      return
   734→    }
   735→
   736→    setIsSubmitting(true)
   737→    try {
   738→      const { orgId, error: createError } = await createOrganizationForUser(data)
   739→      if (createError || !orgId) {
   740→        setSaveError(createError || 'Could not create your organization. Please try again.')
   741→        setIsSubmitting(false)
   742→        return
   743→      }
   744→
   745→      // Reload the org context so the rest of the app sees the new org.
   746→      await completeOnboarding(orgId)
   747→
   748→      // Navigate to the chosen destination.
   749→      setCurrentView(destination || 'dashboard')
   750→      setCurrentScreen('app')
   751→    } catch (err) {
   752→      setSaveError(
   753→        err instanceof Error
   754→          ? err.message
   755→          : 'Something went wrong while setting up your workspace.'
   756→      )
   757→    } finally {
   758→      setIsSubmitting(false)
   759→    }
   760→  }
   761→
   762→  const handleSkip = async () => {
   763→    setSaveError(null)
   764→    if (!user) {
   765→      setSaveError('No authenticated user found. Please sign in again.')
   766→      return
   767→    }
   768→
   769→    setIsSubmitting(true)
   770→    try {
   771→      // Skipping still requires an organization — create a default one
   772→      // using the user's name so the app is fully functional.
   773→      const { createOrganization } = await import('@/lib/auth/organizations')
   774→      const { organization, error: orgError } = await createOrganization({
   775→        name: `${user.name}'s Workspace`,
   776→        ownerId: user.id,
   777→        ownerEmail: user.email,
   778→        ownerDisplayName: user.name,
   779→        ownerPhotoURL: user.picture || null,
   780→        gstin: null,
   781→        plan: 'free',
   782→      })
   783→      if (orgError || !organization) {
   784→        setSaveError(orgError || 'Could not create your workspace. Please try again.')
   785→        setIsSubmitting(false)
   786→        return
   787→      }
   788→      await completeOnboarding(organization.id)
   789→      setCurrentView('dashboard')
   790→      setCurrentScreen('app')
   791→    } catch (err) {
   792→      setSaveError(
   793→        err instanceof Error
   794→          ? err.message
   795→          : 'Something went wrong while setting up your workspace.'
   796→      )
   797→    } finally {
   798→      setIsSubmitting(false)
   799→    }
   800→  }
   801→
   802→  return (
   803→    <OnboardingFlow
   804→      onComplete={handleOnboardingComplete}
   805→      onSkip={handleSkip}
   806→      userEmail={user?.email}
   807→      userName={user?.name}
   808→      error={saveError}
   809→      onDismissError={() => setSaveError(null)}
   810→    />
   811→  )
   812→}

   814→function AppRouter() {
   815→  const { currentScreen, setCurrentScreen } = useApp()
   816→  const { isAuthenticated, isInitializing, needsEmailVerification } = useAuth()
   817→  // ── PART 3 / PART 5 ── The org context is the source of truth for whether
   818→  // the user has an organization. `needsOrganization` becomes the new
   819→  // "needs onboarding" gate. `orgLoading` lets us hold the protected routes
   820→  // until the tenant context is fully resolved so no page ever renders with a
   821→  // null organizationId.
   822→  const { needsOrganization, loading: orgLoading, organization, error: orgError, reload: reloadOrg } = useOrg()
   823→
   824→  // Derived: a user needs onboarding when authenticated but without an org.
   825→  const needsOnboarding = isAuthenticated && needsOrganization
   826→
   827→  useEffect(() => {
   828→    if (isInitializing) return
   829→    // Only auto-redirect to the app when: authenticated, org context resolved,
   830→    // and the user actually has an organization. Otherwise the OnboardingScreen
   831→    // handles its own navigation.
   832→    if (isAuthenticated && !needsOnboarding && !orgLoading && currentScreen !== 'app') {
   833→      setCurrentScreen('app')
   834→    }
   835→  }, [isAuthenticated, isInitializing, currentScreen, setCurrentScreen, needsOnboarding, orgLoading])
   836→
   837→  useEffect(() => {
   838→    if (isInitializing) return
   839→    // ── PART 5 ── Protected routes: kick unauthenticated users back to landing.
   840→    if (!isAuthenticated && currentScreen === 'app') {
   841→      setCurrentScreen('landing')
   842→    }
   843→  }, [isAuthenticated, isInitializing, currentScreen, setCurrentScreen])
   844→
   845→  const handleGetStarted = () => setCurrentScreen('login')
   846→  const handleBookDemo = () => setCurrentScreen('login')
   847→  const handleBackToLanding = () => setCurrentScreen('landing')
   848→
   849→  // ── PART 7 ── Loading state while authentication initializes.
   850→  if (isInitializing && !isAuthenticated) {
   851→    return (
   852→      <div className="min-h-screen flex items-center justify-center bg-black">
   853→        <div className="flex flex-col items-center gap-4">
   854→          <div className="flex h-11 w-11 items-center justify-center rounded-2xl glass-surface motion-pulse">
   855→            <Zap className="h-5 w-5 accent-text" />
   856→          </div>
   857→          <span className="text-sm text-white/55 font-medium">Loading GSTPilot…</span>
   858→        </div>
   859→      </div>
   860→    )
   861→  }
   862→
   863→  // ── PART 7 ── While the org context is resolving after auth, show a brief
   864→  // loader so protected pages never mount with a null organization.
   865→  if (isAuthenticated && orgLoading && !needsOnboarding) {
   866→    return (
   867→      <div className="min-h-screen flex items-center justify-center bg-black">
   868→        <div className="flex flex-col items-center gap-4">
   869→          <div className="flex h-11 w-11 items-center justify-center rounded-2xl glass-surface motion-pulse">
   870→            <Zap className="h-5 w-5 accent-text" />
   871→          </div>
   872→          <span className="text-sm text-white/55 font-medium">Loading your workspace…</span>
   873→        </div>
   874→      </div>
   875→    )
   876→  }
   877→
   878→  // ── PART 2 ── Authenticated but no organization → onboarding creates one.
   879→  // BUT: if there was a load error (e.g. Firestore unreachable), show a retry
   880→  // screen instead of onboarding — onboarding would also fail to write.
   881→  if (needsOnboarding && !orgError) {
   882→    return <OnboardingScreen />
   883→  }
   884→
   885→  // ── PART 8 ── Org context failed to load after retries. Show a friendly
   886→  // error with a retry button — BUT never for permission errors. Permission
   887→  // errors mean the user is in preview/offline mode; OrgContext already falls
   888→  // back to a demo org in that case, so we treat any residual permission error
   889→  // as "preview mode" and render the app shell with empty states.
   890→  const isOrgPermissionError = !!orgError && (
   891→    /permission|insufficient|unauthenticated|not authorized|missing or/i.test(orgError)
   892→  )
   893→  if (isAuthenticated && !orgLoading && orgError && !organization && !isOrgPermissionError) {
   894→    return (
   895→      <div className="min-h-screen flex items-center justify-center bg-black p-6">
   896→        <div className="flex flex-col items-center gap-5 max-w-md text-center">
   897→          <div className="flex h-12 w-12 items-center justify-center rounded-2xl bg-red-500/10 border border-red-500/20">
   898→            <Zap className="h-6 w-6 text-red-400" />
   899→          </div>
   900→          <h2 className="text-xl font-bold text-white">Couldn&apos;t load your workspace</h2>
   901→          <p className="text-sm text-white/55 leading-relaxed">
   902→            {orgError}
   903→          </p>
   904→          <button
   905→            onClick={() => reloadOrg()}
   906→            className="mt-2 inline-flex items-center gap-2 rounded-xl bg-white px-5 py-2.5 text-sm font-semibold text-black hover:bg-white/90 transition-colors"
   907→          >
   908→            Try again
   909→          </button>
   910→        </div>
   911→      </div>
   912→    )
   913→  }
   914→
   915→  // ── PART 5 ── Protected route: render the app shell when authenticated AND
   916→  // (the organization is loaded OR we're in preview/permission-error mode).
   917→  // This guarantees the user never gets stuck behind a wall when Firestore is
   918→  // unreachable — they see the dashboard with premium empty states instead.
   919→  if (currentScreen === 'app' && isAuthenticated && (organization || isOrgPermissionError)) {
   920→    return (
   921→      <div className="flex min-h-screen flex-col">
   922→        {needsEmailVerification && <EmailVerificationBanner />}
   923→        <DashboardContent />
   924→      </div>
   925→    )
   926→  }
   927→
   928→  if (currentScreen === 'login') {
   929→    return <LoginPage onBack={handleBackToLanding} onGetStarted={handleGetStarted} />
   930→  }
   931→
   932→  return <LandingPage onGetStarted={handleGetStarted} onBookDemo={handleBookDemo} />
   933→}
