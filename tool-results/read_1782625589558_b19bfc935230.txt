    51→function formatINR(n: number): string {
    52→  if (!isFinite(n) || isNaN(n)) return '₹0';
    53→  const abs = Math.abs(n);
    54→  if (abs >= 10000000) return `₹${(n / 10000000).toFixed(2)} Cr`;
    55→  if (abs >= 100000) return `₹${(n / 100000).toFixed(2)} L`;
    56→  if (abs >= 1000) return `₹${(n / 1000).toFixed(1)}K`;
    57→  return '₹' + Math.round(n).toLocaleString('en-IN');
    58→}
    59→
    60→function formatINRFull(n: number): string {
    61→  if (!isFinite(n) || isNaN(n)) return '₹0';
    62→  return '₹' + Math.round(n).toLocaleString('en-IN');
    63→}

   127→function timeAgo(iso: string): string {
   128→  const diff = Date.now() - new Date(iso).getTime();
   129→  const s = Math.floor(diff / 1000);
   130→  if (s < 60) return 'just now';
   131→  const m = Math.floor(s / 60);
   132→  if (m < 60) return `${m}m ago`;
   133→  const h = Math.floor(m / 60);
   134→  if (h < 24) return `${h}h ago`;
   135→  return `${Math.floor(h / 24)}d ago`;
   136→}

   140→function useAnimatedNumber(target: number, duration = 1000) {
   141→  const [val, setVal] = useState(0);
   142→  useEffect(() => {
   143→    let raf = 0;
   144→    const start = performance.now();
   145→    const from = 0;
   146→    const animate = (now: number) => {
   147→      const t = Math.min((now - start) / duration, 1);
   148→      const eased = 1 - Math.pow(1 - t, 3);
   149→      setVal(from + (target - from) * eased);
   150→      if (t < 1) raf = requestAnimationFrame(animate);
   151→    };
   152→    raf = requestAnimationFrame(animate);
   153→    return () => cancelAnimationFrame(raf);
   154→  }, [target, duration]);
   155→  return val;
   156→}
