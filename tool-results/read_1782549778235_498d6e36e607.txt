    73→function formatINR(amount: number): string {
    74→  if (!isFinite(amount) || isNaN(amount)) return '₹0';
    75→  const abs = Math.abs(amount);
    76→  if (abs >= 10000000) return `₹${(amount / 10000000).toFixed(2)} Cr`;
    77→  if (abs >= 100000) return `₹${(amount / 100000).toFixed(2)} L`;
    78→  if (abs >= 1000) return `₹${(amount / 1000).toFixed(1)}K`;
    79→  return '₹' + Math.round(amount).toLocaleString('en-IN');
    80→}
    81→
    82→function formatINRFull(amount: number): string {
    83→  if (!isFinite(amount) || isNaN(amount)) return '₹0';
    84→  return '₹' + Math.round(amount).toLocaleString('en-IN');
    85→}

    87→function timeAgo(iso: string): string {
    88→  if (!iso) return '—';
    89→  const diff = Date.now() - new Date(iso).getTime();
    90→  const s = Math.floor(diff / 1000);
    91→  if (s < 0 || s < 60) return 'just now';
    92→  const m = Math.floor(s / 60);
    93→  if (m < 60) return `${m}m ago`;
    94→  const h = Math.floor(m / 60);
    95→  if (h < 24) return `${h}h ago`;
    96→  return `${Math.floor(h / 24)}d ago`;
    97→}

    99→function signedINR(amount: number): string {
   100→  if (!isFinite(amount) || isNaN(amount) || amount === 0) return '₹0';
   101→  const sign = amount > 0 ? '+' : '−';
   102→  return sign + formatINR(Math.abs(amount));
   103→}

   107→function nodeImportance(n: GraphNode): number {
   108→  let base = 30;
   109→  if (n.type === 'business') base = 100;
   110→  else if (n.type === 'client' || n.type === 'vendor' || n.type === 'employee' || n.type === 'bank-account') base = 75;
   111→  else if (n.type === 'invoice' || n.type === 'gst-return' || n.type === 'notice' || n.type === 'task' || n.type === 'report') base = 50;
   112→  else if (n.type === 'prediction' || n.type === 'conversation') base = 30;
   113→  if (n.riskScore) base += Math.min(20, n.riskScore / 5);
   114→  if (n.amount) base += Math.min(15, Math.log10(n.amount + 1));
   115→  return base;
   116→}

   258→function VisualGraphExplorer({ state, focusedNodeId, setFocusedNodeId }: ExplorerProps) {
   259→  const { toast } = useToast();
   260→
   261→  // View state
   262→  const [zoom, setZoom] = useState(1);
   263→  const [pan, setPan] = useState<{ x: number; y: number }>({ x: 0, y: 0 });
   264→  const [userOverrides, setUserOverrides] = useState<Record<string, NodePos>>({});
   265→  const [hiddenTypes, setHiddenTypes] = useState<Set<NodeType>>(new Set());
   266→  const [search, setSearch] = useState('');
   267→  const [riskOverlay, setRiskOverlay] = useState(false);
   268→  const [selectedNodeId, setSelectedNodeId] = useState<string | null>(null);
   269→
   270→  // Interaction state — refs for hot-path values, state for the cursor
   271→  const svgRef = useRef<SVGSVGElement | null>(null);
   272→  const interactionMode = useRef<'none' | 'pan' | 'node-drag'>('none');
   273→  const dragStart = useRef<{ x: number; y: number }>({ x: 0, y: 0 });
   274→  const panStart = useRef<{ x: number; y: number }>({ x: 0, y: 0 });
   275→  const dragNodeId = useRef<string | null>(null);
   276→  const movedRef = useRef(false);
   277→  const [isPanning, setIsPanning] = useState(false);
   278→
   279→  // Positions = engine-derived layout + any user drag overrides
   280→  const positions = useMemo(() => {
   281→    const pos: Record<string, NodePos> = {};
   282→    state.knowledgeGraph.nodes.forEach((n) => {
   283→      pos[n.id] = userOverrides[n.id] || { x: n.x ?? 0, y: n.y ?? 0 };
   284→    });
   285→    return pos;
   286→  }, [state, userOverrides]);
   287→
   288→  // ─── Visible nodes (filtered + limited) ─────────────────────────────────────
   289→  const visibleNodes = useMemo(() => {
   290→    const lower = search.trim().toLowerCase();
   291→    let nodes = state.knowledgeGraph.nodes.filter((n) => !hiddenTypes.has(n.type));
   292→
   293→    if (lower) {
   294→      const matches = nodes.filter((n) => n.label.toLowerCase().includes(lower));
   295→      const matchSet = new Set(matches.map((m) => m.id));
   296→      // Add 1-hop neighbors of matches
   297→      state.knowledgeGraph.edges.forEach((e) => {
   298→        if (matchSet.has(e.source)) matchSet.add(e.target);
   299→        if (matchSet.has(e.target)) matchSet.add(e.source);
   300→      });
   301→      nodes = nodes.filter((n) => matchSet.has(n.id));
   302→    }
   303→
   304→    // Always keep business + focused node
   305→    const alwaysKeep = new Set<string>();
   306→    state.knowledgeGraph.nodes.filter((n) => n.type === 'business').forEach((n) => alwaysKeep.add(n.id));
   307→    if (focusedNodeId) alwaysKeep.add(focusedNodeId);
   308→    if (selectedNodeId) alwaysKeep.add(selectedNodeId);
   309→
   310→    // Sort by importance and limit to top 45
   311→    const sorted = [...nodes].sort((a, b) => {
   312→      if (alwaysKeep.has(a.id) && !alwaysKeep.has(b.id)) return -1;
   313→      if (!alwaysKeep.has(a.id) && alwaysKeep.has(b.id)) return 1;
   314→      return nodeImportance(b) - nodeImportance(a);
   315→    });
   316→    return sorted.slice(0, 45);
   317→  }, [state, hiddenTypes, search, focusedNodeId, selectedNodeId]);
   318→
   319→  const visibleNodeIds = useMemo(() => new Set(visibleNodes.map((n) => n.id)), [visibleNodes]);
   320→
   321→  const visibleEdges = useMemo(() => {
   322→    return state.knowledgeGraph.edges.filter(
   323→      (e) => visibleNodeIds.has(e.source) && visibleNodeIds.has(e.target),
   324→    );
   325→  }, [state, visibleNodeIds]);
   326→
   327→  // ─── Focus neighbors (for highlighting) ─────────────────────────────────────
   328→  const focusNeighbors = useMemo(() => {
   329→    if (!focusedNodeId) return null;
   330→    const set = new Set<string>([focusedNodeId]);
   331→    state.knowledgeGraph.edges.forEach((e) => {
   332→      if (e.source === focusedNodeId) set.add(e.target);
   333→      if (e.target === focusedNodeId) set.add(e.source);
   334→    });
   335→    return set;
   336→  }, [focusedNodeId, state]);
   337→
   338→  // ─── Pointer interaction handlers ───────────────────────────────────────────
   339→  const handlePointerDown = useCallback((e: React.PointerEvent<SVGSVGElement>) => {
   340→    if (e.button !== 0 && e.pointerType === 'mouse') return;
   341→    const target = e.target as Element;
   342→    const nodeIdAttr = target.getAttribute('data-node-id');
   343→    (e.target as Element).setPointerCapture?.(e.pointerId);
   344→    movedRef.current = false;
   345→    dragStart.current = { x: e.clientX, y: e.clientY };
   346→    if (nodeIdAttr) {
   347→      interactionMode.current = 'node-drag';
   348→      dragNodeId.current = nodeIdAttr;
   349→      panStart.current = { ...positions[nodeIdAttr] };
   350→    } else {
   351→      interactionMode.current = 'pan';
   352→      setIsPanning(true);
   353→      panStart.current = { ...pan };
   354→    }
   355→  }, [pan, positions]);
   356→
   357→  const handlePointerMove = useCallback((e: React.PointerEvent<SVGSVGElement>) => {
   358→    if (interactionMode.current === 'none') return;
   359→    const dx = e.clientX - dragStart.current.x;
   360→    const dy = e.clientY - dragStart.current.y;
   361→    if (Math.abs(dx) > 3 || Math.abs(dy) > 3) movedRef.current = true;
   362→
   363→    if (interactionMode.current === 'pan') {
   364→      setPan({ x: panStart.current.x + dx, y: panStart.current.y + dy });
   365→    } else if (interactionMode.current === 'node-drag' && dragNodeId.current) {
   366→      const id = dragNodeId.current;
   367→      const newX = panStart.current.x + dx / zoom;
   368→      const newY = panStart.current.y + dy / zoom;
   369→      setUserOverrides((prev) => ({ ...prev, [id]: { x: newX, y: newY } }));
   370→    }
   371→  }, [zoom]);
   372→
   373→  const handlePointerUp = useCallback((e: React.PointerEvent<SVGSVGElement>) => {
   374→    const wasMode = interactionMode.current;
   375→    const targetId = dragNodeId.current;
   376→    interactionMode.current = 'none';
   377→    dragNodeId.current = null;
   378→    setIsPanning(false);
   379→    (e.target as Element).releasePointerCapture?.(e.pointerId);
   380→
   381→    if (!movedRef.current && wasMode === 'node-drag' && targetId) {
   382→      // Treat as click
   383→      if (selectedNodeId === targetId) {
   384→        setSelectedNodeId(null);
   385→      } else {
   386→        setSelectedNodeId(targetId);
   387→      }
   388→      if (focusedNodeId === targetId) {
   389→        setFocusedNodeId(null);
   390→      } else {
   391→        setFocusedNodeId(targetId);
   392→        // Scroll into view
   393→        if (typeof window !== 'undefined') {
   394→          window.scrollTo({ top: 0, behavior: 'smooth' });
   395→        }
   396→      }
   397→    } else if (!movedRef.current && wasMode === 'pan') {
   398→      // Background click — clear selection but keep focus if any
   399→      setSelectedNodeId(null);
   400→    }
   401→  }, [focusedNodeId, selectedNodeId, setFocusedNodeId]);
   402→
   403→  const handleWheel = useCallback((e: React.WheelEvent<SVGSVGElement>) => {
   404→    e.preventDefault();
   405→    const delta = e.deltaY > 0 ? -0.1 : 0.1;
   406→    setZoom((z) => Math.min(3, Math.max(0.3, z + delta)));
   407→  }, []);
   408→
   409→  const zoomIn = useCallback(() => setZoom((z) => Math.min(3, z + 0.2)), []);
   410→  const zoomOut = useCallback(() => setZoom((z) => Math.max(0.3, z - 0.2)), []);
   411→  const resetView = useCallback(() => {
   412→    setZoom(1);
   413→    setPan({ x: 0, y: 0 });
   414→    setFocusedNodeId(null);
   415→    setSelectedNodeId(null);
   416→    setUserOverrides({});
   417→  }, [setFocusedNodeId]);
   418→
   419→  const toggleType = useCallback((t: NodeType) => {
   420→    setHiddenTypes((prev) => {
   421→      const next = new Set(prev);
   422→      if (next.has(t)) next.delete(t);
   423→      else next.add(t);
   424→      return next;
   425→    });
   426→  }, []);
   427→
   428→  // ─── Selected node detail (neighbors) ───────────────────────────────────────
   429→  const selectedNode = useMemo(() => {
   430→    if (!selectedNodeId) return null;
   431→    return state.knowledgeGraph.nodes.find((n) => n.id === selectedNodeId) || null;
   432→  }, [selectedNodeId, state]);
   433→
   434→  const selectedNeighbors = useMemo(() => {
   435→    if (!selectedNodeId) return [];
   436→    const results: { node: GraphNode; edge: GraphEdge }[] = [];
   437→    state.knowledgeGraph.edges.forEach((e) => {
   438→      if (e.source === selectedNodeId) {
   439→        const n = state.knowledgeGraph.nodes.find((x) => x.id === e.target);
   440→        if (n) results.push({ node: n, edge: e });
   441→      } else if (e.target === selectedNodeId) {
   442→        const n = state.knowledgeGraph.nodes.find((x) => x.id === e.source);
   443→        if (n) results.push({ node: n, edge: e });
   444→      }
   445→    });
   446→    return results.slice(0, 12);
   447→  }, [selectedNodeId, state]);
   448→
   449→  const selectedRisk = useMemo(() => {
   450→    if (!selectedNodeId) return null;
   451→    return state.riskGraph.nodes.find((r) => r.nodeId === selectedNodeId) || null;
   452→  }, [selectedNodeId, state]);
   453→
   454→  // ─── SVG coordinate system ──────────────────────────────────────────────────
   455→  // Use a fixed viewBox that fits a typical radial layout (~720x540)
   456→  const VIEW_W = 800;
   457→  const VIEW_H = 540;
   458→
   459→  const focusNode = useCallback((id: string) => {
   460→    setFocusedNodeId(id);
   461→    setSelectedNodeId(id);
   462→    if (typeof window !== 'undefined') {
   463→      window.scrollTo({ top: 0, behavior: 'smooth' });
   464→    }
   465→  }, [setFocusedNodeId]);
   466→
   467→  return (
   468→    <FadeIn delay={0.05}>
   469→      <Card className="border-white/[0.06] bg-card/60 backdrop-blur-sm">
   470→        <CardHeader className="pb-3">
   471→          <div className="flex flex-wrap items-center justify-between gap-3">
   472→            <CardTitle className="flex items-center gap-2 text-base">
   473→              <Network className="h-4 w-4 accent-text" />
   474→              Visual Graph Explorer
   475→              <Badge variant="outline" className="ml-1 border-white/10 bg-white/[0.03] text-[10px]">
   476→                {visibleNodes.length} of {state.knowledgeGraph.nodes.length} nodes
   477→              </Badge>

  1714→function GraphApiModule() {
  1715→  const [results, setResults] = useState<Record<string, { loading: boolean; data: unknown; error?: string }>>({});
  1716→
  1717→  const tryIt = useCallback(async (ep: ApiEndpoint) => {
  1718→    setResults((prev) => ({ ...prev, [ep.path]: { loading: true, data: null } }));
  1719→    try {
  1720→      const res = ep.method === 'POST'
  1721→        ? await fetch(ep.url, {
  1722→            method: 'POST',
  1723→            headers: { 'Content-Type': 'application/json' },
  1724→            body: ep.body,
  1725→            cache: 'no-store',
  1726→          })
  1727→        : await fetch(ep.url, { cache: 'no-store' });
  1728→      if (!res.ok) throw new Error(`HTTP ${res.status}`);
  1729→      const json = await res.json();
  1730→      setResults((prev) => ({ ...prev, [ep.path]: { loading: false, data: json } }));
  1731→    } catch (e) {
  1732→      setResults((prev) => ({
  1733→        ...prev,
  1734→        [ep.path]: { loading: false, data: null, error: e instanceof Error ? e.message : 'Unknown error' },
  1735→      }));
  1736→    }
  1737→  }, []);
  1738→

  1740→    <div>

  1753→                  <div className="mb-2 flex items-start justify-between gap-2">

  1757→                          variant="outline"
  1758→                          className={`text-[9px] font-bold ${
  1759→                            ep.method === 'GET'
  1760→                              ? 'border-emerald-500/30 bg-emerald-500/[0.08] text-emerald-300'
  1761→                              : 'border-amber-500/30 bg-amber-500/[0.08] text-amber-300'
  1762→                          }`}
