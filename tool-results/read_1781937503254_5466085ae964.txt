  1246→interface ConversationGroups {
  1247→  today: OracleConversation[];
  1248→  yesterday: OracleConversation[];
  1249→  prev7: OracleConversation[];
  1250→  older: OracleConversation[];

  1253→function groupConversations(convos: OracleConversation[]): ConversationGroups {
  1254→  const now = new Date();
  1255→  const startOfToday = new Date(now.getFullYear(), now.getMonth(), now.getDate()).getTime();
  1256→  const startOfYesterday = startOfToday - 86_400_000;
  1257→  const startOf7Days = startOfToday - 7 * 86_400_000;
  1258→
  1259→  const today: OracleConversation[] = [];
  1260→  const yesterday: OracleConversation[] = [];
  1261→  const prev7: OracleConversation[] = [];
  1262→  const older: OracleConversation[] = [];
  1263→
  1264→  const sorted = [...convos].sort(
  1265→    (a, b) => new Date(b.updatedAt).getTime() - new Date(a.updatedAt).getTime(),
  1266→  );
  1267→
  1268→  for (const c of sorted) {
  1269→    const t = new Date(c.updatedAt).getTime();
  1270→    if (isNaN(t)) {
  1271→      older.push(c);
  1272→    } else if (t >= startOfToday) {
  1273→      today.push(c);
  1274→    } else if (t >= startOfYesterday) {
  1275→      yesterday.push(c);
  1276→    } else if (t >= startOf7Days) {
  1277→      prev7.push(c);
  1278→    } else {
  1279→      older.push(c);
  1280→    }
  1281→  }
  1282→
  1283→  return { today, yesterday, prev7, older };
  1284→}

  1286→/** Generate contextual follow-up chips from the response content. */
  1287→function buildFollowUps(content: string): string[] | undefined {
  1288→  if (!content || content.length < 30) return undefined;
  1289→  const lower = content.toLowerCase();
  1290→
  1291→  const pool: string[] = [];
  1292→  if (/(gstr-3b|3b)/.test(lower)) pool.push('GSTR-3B की last date क्या है?', 'Late fee कितनी लगेगी?');
  1293→  if (/(gstr-1|gstr 1)/.test(lower)) pool.push('GSTR-1 कैसे file करें?', 'B2B और B2C में अंतर?');
  1294→  if (/(itc|input tax credit)/.test(lower)) pool.push('ITC claim कैसे करें?', 'Blocked ITC के rules?');
  1295→  if (/(late fee|penalty|overdue)/.test(lower)) pool.push('Late fee waiver मिल सकती है?', 'How to avoid this next time?');
  1296→  if (/(cash flow|collections)/.test(lower)) pool.push('Receivables कैसे recover करें?', 'Working capital optimize करें');
  1297→  if (/(reverse charge|rcm)/.test(lower)) pool.push('RCM किन पर लागू है?', 'How to report RCM in GSTR-3B?');
  1298→  if (/(refund)/.test(lower)) pool.push('Refund process क्या है?', 'Refund timeline कितनी है?');
  1299→
  1300→  pool.push('GST kya hota hai?', 'Explain ITC rules', 'How can Oracle help me daily?');
  1301→
  1302→  const seen = new Set<string>();
  1303→  const picks: string[] = [];
  1304→  for (const p of pool) {
  1305→    if (seen.has(p)) continue;
  1306→    seen.add(p);
  1307→    picks.push(p);
  1308→    if (picks.length >= 3) break;

  1311→}
