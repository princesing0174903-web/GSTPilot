   249→      q,

   251→    q.includes('file my gst') ||
   252→    q.includes('itc rules') ||
   253→    q.includes('show filing steps')
   254→  ) {
   255→    return { persona: 'gst_expert', intent: 'gst_filing' }
   256→  }
   257→
   258→  // ── Compliance Assistant™ ──
   259→  if (
   260→    /\b(compliance|deadline|due date|overdue|penalt(y|ies)|notice|roc|statutory|audit|tds|advance tax|kyb|dpt-3|dir-3|mgt-7|aoc-4|advt|section 44ab|tax audit|late fees?|checklist)\b/.test(
   261→      q,

   263→    q.includes('explain penalties') ||
   264→    q.includes('generate checklist')
   265→  ) {
   266→    return { persona: 'compliance_assistant', intent: 'compliance' }
   267→  }
   268→
   269→  // ── Financial Analyst™ ──
   270→  if (
   271→    /\b(report|reports|analytics|analysis|trend|trends|benchmark|summary|statement|compare|comparison|variance|ratio|kpi|dashboard|insight|insights|visualiz|chart|graph|executive brief|mis)\b/.test(
   272→      q,

   277→    return { persona: 'financial_analyst', intent: 'report' }

   280→  // ── AI CFO™ ──

   288→    return { persona: 'ai_cfo', intent: 'finance' }

   306→function extractCompanyProfile(
   307→  context: string,
   308→  body: { companyProfile?: Partial<CompanyProfile> },
   309→): CompanyProfile {
   310→  const hasLiveData = contextHasLiveData(context)
   311→
   312→  // Parse live counts from context
   313→  const clientCount = parseCount(context, /-?\s*Total clients:\s*(\d+)/i)
   314→  const invoiceCount = parseCount(context, /-?\s*Active invoices:\s*(\d+)/i)
   315→  const gstr1Pending = parseCount(context, /-?\s*Pending returns:\s*(\d+)\s*GSTR-1/i)
   316→  const gstr3bPending = parseCount(
   317→    context,
   318→    /GSTR-1,?\s*(\d+)\s*GSTR-3B/i,
   319→  )
   320→  const pendingReturns = gstr1Pending + gstr3bPending
   321→
   322→  // Connected services: derive from what data is present
   323→  const connectedServices: string[] = []
   324→  if (clientCount > 0) connectedServices.push('Clients')
   325→  if (invoiceCount > 0) connectedServices.push('Invoices')
   326→  if (parseCount(context, /-?\s*Filed returns:\s*(\d+)/i) > 0 || pendingReturns > 0)
   327→    connectedServices.push('GST Returns')
   328→  if (parseCount(context, /-?\s*Unread notifications:\s*(\d+)/i) > 0)
   329→    connectedServices.push('Notifications')
   330→
   331→  return {
   332→    firmName: body.companyProfile?.firmName,
   333→    industry: body.companyProfile?.industry,
   334→    state: body.companyProfile?.state,
   335→    gstType: body.companyProfile?.gstType,
   336→    organizationType: body.companyProfile?.organizationType,
   337→    connectedServices: connectedServices.length > 0 ? connectedServices : ['None yet'],
   338→    hasLiveData,
   339→    clientCount,
   340→    invoiceCount,
   341→    pendingReturns,
   342→  }
   343→}

   347→function buildSources(context: string, hasLiveData: boolean): OracleSource[] {
   348→  if (!hasLiveData) {
   349→    // When no live data, show the knowledge sources Oracle consulted
   350→    return [
   351→      { name: 'GST Act & Rules', count: 1, icon: 'book' },
   352→      { name: 'CBIC Notifications', count: 1, icon: 'file-text' },
   353→      { name: 'India Tax Framework', count: 1, icon: 'landmark' },
   354→    ]
   355→  }
   356→
   357→  const sources: OracleSource[] = []
   358→
   359→  const clients = parseCount(context, /-?\s*Total clients:\s*(\d+)/i)
   360→  if (clients > 0) sources.push({ name: 'Clients', count: clients, icon: 'users' })
   361→
   362→  const invoices = parseCount(context, /-?\s*Active invoices:\s*(\d+)/i)
   363→  if (invoices > 0) sources.push({ name: 'Invoices', count: invoices, icon: 'file-text' })
   364→
   365→  const filed = parseCount(context, /-?\s*Filed returns:\s*(\d+)/i)
   366→  const gstr1Pending = parseCount(context, /-?\s*Pending returns:\s*(\d+)\s*GSTR-1/i)
   367→  const gstr3bPending = parseCount(context, /GSTR-1,?\s*(\d+)\s*GSTR-3B/i)
   368→  const totalReturns = filed + gstr1Pending + gstr3bPending
   369→  if (totalReturns > 0)
   370→    sources.push({ name: 'GST Returns', count: totalReturns, icon: 'receipt' })
   371→
   372→  const unread = parseCount(context, /-?\s*Unread notifications:\s*(\d+)/i)
   373→  if (unread > 0)
   374→    sources.push({ name: 'Notifications', count: unread, icon: 'bell' })
   375→
   376→  // Always include the knowledge base as a source
   377→  sources.push({ name: 'GST Knowledge Base', count: 1, icon: 'book' })
   378→
   379→  return sources
   380→}

   389→  const actions: OracleAction[] = []
   390→  const q = question.toLowerCase()

   392→  if (intent === 'run_automation') {
   393→    actions.push({
   394→      type: 'execute_workflow',
   395→      title: 'Run Autonomous Pipeline',
   396→      description:
   397→        'Execute the full autonomous pipeline: read data → reconcile → predict cash flow → file returns → notify teams → generate reports.',
   398→      payload: {
   399→        workflow: q.includes('company')
   400→          ? 'run_my_company'
   401→          : q.includes('india')
   402→            ? 'run_india_business'
   403→            : 'run_my_firm',
   404→      },
   405→    })
   406→  }
   407→
   408→  if (
   409→    persona === 'gst_expert' &&
   410→    (q.includes('file') || q.includes('filing') || q.includes('return') || intent === 'gst_filing')
   411→  ) {
   412→    actions.push({
   413→      type: 'navigate',
   414→      title: 'Open GST Returns',
   415→      description: 'Navigate to the GST Returns workspace to file pending returns.',
   416→      view: 'returns',
   417→      payload: { view: 'returns' },
   418→    })
   419→  }
   420→
   421→  if (q.includes('late fee') || q.includes('penalty') || q.includes('overdue')) {
   422→    actions.push({
   423→      type: 'calculate',
   424→      title: 'Calculate Late Fee',
   425→      description: 'Compute the exact late fee and interest based on delay and tax amount.',
   426→      payload: { calculation: 'late_fee' },
   427→    })
   428→  }
   429→
   430→  if (q.includes('checklist') || q.includes('steps') || q.includes('how to')) {
   431→    actions.push({
   432→      type: 'checklist',
   433→      title: 'Generate Checklist',
   434→      description: 'Create a step-by-step actionable checklist for this task.',
   435→      payload: { source: 'oracle' },
   436→    })
   437→  }
   438→
   439→  if (q.includes('report') || q.includes('summary') || persona === 'financial_analyst') {
   440→    actions.push({
   441→      type: 'generate_report',
   442→      title: 'Create Report',
   443→      description: 'Compile an executive report with the key metrics discussed.',
   444→      payload: { format: 'pdf' },
   445→    })
   446→  }
   447→
   448→  if (q.includes('remind') || q.includes('collection')) {
   449→    actions.push({
   450→      type: 'send_reminder',
   451→      title: 'Send Payment Reminders',
   452→      description: 'Send automated payment reminders to clients with overdue invoices.',
   453→      payload: { channel: 'all' },
   454→    })
   455→  }
   456→
   457→  if (persona === 'ai_cfo' && (q.includes('cash') || q.includes('flow'))) {
   458→    actions.push({
   459→      type: 'navigate',
   460→      title: 'Open Cash Flow',
   461→      description: 'Open the cash flow workspace to see inflow/outflow projections.',
   462→      view: 'working-capital',
   463→      payload: { view: 'working-capital' },
   464→    })
   465→  }
   466→
   467→  if (persona === 'financial_analyst') {
   468→    actions.push({
   469→      type: 'navigate',
   470→      title: 'Open Analytics',
   471→      description: 'Open the executive analytics dashboard for detailed insights.',
   472→      view: 'ai-predictions',
   473→      payload: { view: 'ai-predictions' },
   474→    })
   475→  }
   476→
   477→  // Cap at 3 actions
   478→  return actions.slice(0, 3)
   479→}

   590→function parseStructuredResponse(raw: string): ParsedLLMResponse {
   591→  // Find marker positions
   592→  const answerStart = raw.indexOf(MARKER_ANSWER)
   593→  const whyStart = raw.indexOf(MARKER_WHY)
   594→  const actionsStart = raw.indexOf(MARKER_ACTIONS)
   595→  const followupsStart = raw.indexOf(MARKER_FOLLOWUPS)
   596→
   597→  // If no markers found, treat the whole thing as the answer
   598→  if (answerStart === -1) {
   599→    return {
   600→      answer: raw.trim(),
   601→      whyItMatters: '',
   602→      recommendedActions: [],
   603→      followUpQuestions: [],
   604→    }
   605→  }
   606→
   607→  const extractSection = (start: number, end: number): string => {
   608→    if (start === -1) return ''
   609→    const headerEnd = start + raw.slice(start).indexOf('\n') + 1
   610→    const sectionEnd = end === -1 ? raw.length : end
   611→    return raw.slice(headerEnd, sectionEnd).trim()
   612→  }
   613→
   614→  const answer = extractSection(
   615→    answerStart,
   616→    [whyStart, actionsStart, followupsStart].filter((p) => p > answerStart).sort()[0] ?? -1,
   617→  )
   618→  const whyItMatters = extractSection(
   619→    whyStart,
   620→    [actionsStart, followupsStart].filter((p) => p > whyStart).sort()[0] ?? -1,
   621→  )
   622→  const actionsText = extractSection(whyStart > -1 ? whyStart : actionsStart, followupsStart > actionsStart ? followupsStart : -1)
   623→  const followupsText = extractSection(followupsStart, -1)
   624→
   625→  // Parse bullet lists
   626→  const parseBullets = (text: string): string[] => {
   627→    if (!text) return []
   628→    return text
   629→      .split('\n')
   630→      .map((l) => l.trim())
   631→      .filter((l) => l.startsWith('- ') || l.startsWith('• '))
   632→      .map((l) => l.replace(/^[-•]\s*/, '').trim())
   633→      .filter(Boolean)
   634→  }
   635→
   636→  return {
   637→    answer: answer || raw.trim(),
   638→    whyItMatters: whyItMatters || 'This directly impacts your compliance posture and financial health.',
   639→    recommendedActions: parseBullets(actionsText),
   640→    followUpQuestions: parseBullets(followupsText),
   641→  }
   642→}

   644→async function generateStructuredAnswer(
   645→  question: string,
   646→  persona: PersonaDef,
   647→  context: string,
   648→  profile: CompanyProfile,
   649→  conversationHistory: Array<{ role: string; content: string }>,
   650→): Promise<ParsedLLMResponse> {
   651→  const ZAI = (await import('z-ai-web-dev-sdk')).default
   652→  const zai = await ZAI.create()
   653→
   654→  const systemPrompt = buildSystemPrompt(persona, context, profile, conversationHistory)
   655→
   656→  const response = await zai.chat.completions.create({
   657→    messages: [
   658→      { role: 'system', content: systemPrompt },
   659→      { role: 'user', content: question },
   660→    ],
   661→  })
   662→
   663→  const raw = response.choices[0]?.message?.content || ''
   664→  return parseStructuredResponse(raw)
   665→}

   775→function parseCount(context: string, pattern: RegExp): number {
   776→  const match = context.match(pattern)
   777→  if (!match || !match[1]) return 0
   778→  const n = parseInt(match[1], 10)
   779→  return Number.isFinite(n) ? n : 0
   780→}

   782→function contextHasLiveData(context: string): boolean {
   783→  if (!context || context.length < 60) return false
   784→  if (!/LIVE DATA SNAPSHOT/i.test(context)) return false
   785→  if (/Data currently unavailable/i.test(context)) return false
   786→  const dataLines = context.split('\n').filter((l) => /^\s*-\s+\w/.test(l))
   787→  return dataLines.length >= 2
   788→}

   792→export async function POST(req: NextRequest) {
   793→  try {
   794→    const body = await req.json()
   795→    const question: string = (body.question || '').trim()
   796→    const context: string = (body.context || '').trim()
   797→    const conversationHistory: Array<{ role: string; content: string }> =
   798→      body.conversationHistory || []
   799→
   800→    if (!question) {
   801→      return NextResponse.json({ error: 'Question is required' }, { status: 400 })
   802→    }
   803→
   804→    // 1) Detect persona + intent
   805→    const { persona: personaId, intent } = detectPersona(question)
   806→    const persona = PERSONAS[personaId]
   807→
   808→    // 2) Build company profile (memory)
   809→    const profile = extractCompanyProfile(context, body)
   810→
   811→    // 3) Build sources (from live data OR knowledge base)
   812→    const sources = buildSources(context, profile.hasLiveData)
   813→
   814→    // 4) Thinking steps (from persona — spec-exact phases)
   815→    const thinkingSteps = persona.thinkingSteps
   816→
   817→    // 5) Detect actions
   818→    const actions = detectActions(question, personaId, intent)
   819→
   820→    // 6) Generate the structured answer via LLM (or fallback)
   821→    let parsed: ParsedLLMResponse
   822→    try {
   823→      parsed = await generateStructuredAnswer(
   824→        question,
   825→        persona,
   826→        context,
   827→        profile,
   828→        conversationHistory,
   829→      )
   830→    } catch (llmError) {
   831→      console.error(
   832→        '[Oracle] LLM error — using guidance fallback:',
   833→        llmError instanceof Error ? llmError.message : llmError,
   834→      )
   835→      parsed = generateGuidanceFallback(question, persona, profile)
   836→    }
   837→
   838→    // Validate + fill any empty sections
   839→    if (!parsed.answer || !parsed.answer.trim()) {
   840→      parsed = generateGuidanceFallback(question, persona, profile)

   844→        'This directly impacts your compliance posture, cash flow, and risk exposure.'
