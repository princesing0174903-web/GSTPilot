     1→// ═══════════════════════════════════════════════════════════════════════════════
     2→// GSTPilot ERP & Accounting Integrations™ — Client-Safe Firestore Service
     3→//
     4→// The single entry point for all ERP Firestore operations on the CLIENT side.
     5→// Mirrors the banking-provider + communication-provider service pattern:
     6→//   • Real-time subscriptions via onSnapshot (org-scoped)
     7→//   • CRUD writes via the Firebase client SDK (rules enforce org isolation)
     8→//   • Multi-tenant — every function filters on `organizationId`
     9→//
    10→// This module is CLIENT-SAFE — it only imports from `firebase/firestore` and
    11→// `@/lib/firebase` (the client SDK). It NEVER imports the provider, crypto, or
    12→// any server-only code.
    13→//
    14→// The flow for each ERP operation:
    15→//   1. Client hook calls the appropriate API route (/api/erp/*) for the
    16→//      provider work (connect, sync, etc.). The API route returns plain data +
    17→//      the encrypted connection.
    18→//   2. Client hook calls the appropriate function HERE to persist the result
    19→//      to Firestore.
    20→//   3. Real-time onSnapshot subscriptions surface the change to every
    21→//      connected client instantly.
    22→//
    23→// Firestore timeout guard: when the Firebase project has the Firestore API
    24→// disabled, the SDK retries PERMISSION_DENIED for ~120s before rejecting.
    25→// `withTimeout` races every op against a 6s deadline so the UI never hangs.
    26→// ═══════════════════════════════════════════════════════════════════════════════

    91→// ─── Firestore timeout guard ─────────────────────────────────────────────────

    93→const FIRESTORE_TIMEOUT_MS = 6000;
    94→
    95→function withTimeout<T>(promise: Promise<T>, label: string): Promise<T> {
    96→  promise.catch(() => { /* timed-out op — ignore late rejection */ });
    97→  return Promise.race([
    98→    promise,
    99→    new Promise<T>((_, reject) =>
   100→      setTimeout(
   101→        () =>
   102→          reject(
   103→            new ERPError(
   104→              `${label} timed out — Firestore may be unreachable.`,
   105→              { code: 'FIRESTORE_TIMEOUT', statusCode: 503, retryable: true },
   106→            ),
   107→          ),
   108→        FIRESTORE_TIMEOUT_MS,
   109→      ),
   110→    ),
   111→  ]);
   112→}
   113→
   114→const safeAddDoc = <T>(ref: ReturnType<typeof collection>, data: T) =>
   115→  withTimeout(addDoc(ref, data as Record<string, unknown>), 'ERP addDoc');
   116→const safeSetDoc = (ref: ReturnType<typeof doc>, data: unknown, opts?: { merge?: boolean }) =>
   117→  withTimeout(setDoc(ref, data as Record<string, unknown>, opts), 'ERP setDoc');
   118→const safeUpdateDoc = (ref: ReturnType<typeof doc>, data: unknown) =>
   119→  withTimeout(updateDoc(ref, data as Record<string, unknown>), 'ERP updateDoc');
   120→const safeDeleteDoc = (ref: ReturnType<typeof doc>) =>
   121→  withTimeout(deleteDoc(ref), 'ERP deleteDoc');
   122→const safeGetDoc = (ref: ReturnType<typeof doc>) =>
   123→  withTimeout(getDoc(ref), 'ERP getDoc');
   124→const safeGetDocs = (q: ReturnType<typeof query>) =>
   125→  withTimeout(getDocs(q), 'ERP getDocs');
   126→const safeCommit = (batch: ReturnType<typeof writeBatch>) =>
   127→  withTimeout(batch.commit(), 'ERP writeBatch.commit');

   129→// ─── Timestamp conversion ────────────────────────────────────────────────────
   130→
   131→function ts(value: unknown): string {
   132→  if (value && typeof value === 'object' && 'toDate' in value) {
   133→    return (value as { toDate: () => Date }).toDate().toISOString();
   134→  }
   135→  if (typeof value === 'string') return value;
   136→  return '';
   137→}
   138→
   139→function round2(n: number): number {
   140→  return Math.round(n * 100) / 100;
   141→}

   166→export function subscribeToConnections(
   167→  organizationId: string,
   168→  callback: (connections: ERPConnection[]) => void,
   169→  options?: { onError?: (err: Error) => void },
   170→): Unsubscribe {
   171→  assertOrg(organizationId);
   172→  const q = query(
   173→    collection(db, ERP_COLLECTIONS.CONNECTIONS),
   174→    where('organizationId', '==', organizationId),
   175→    orderBy('createdAt', 'desc'),
   176→  );
   177→  return onSnapshot(
   178→    q,
   179→    (snap) => callback(snap.docs.map((d) => toConnection(d.id, d.data() as Record<string, unknown>))),
   180→    (err) => options?.onError?.(err as Error),
   181→  );
   182→}

   184→export async function getConnections(organizationId: string): Promise<ERPConnection[]> {
   185→  assertOrg(organizationId);

   188→    where('organizationId', '==', organizationId),

   191→  const snap = await safeGetDocs(q).catch(() => ({ docs: [] as Array<{ id: string; data: () => Record<string, unknown> }> }));
   192→  return snap.docs.map((d) => toConnection(d.id, d.data() as Record<string, unknown>));
   193→}
   194→
   195→export async function getConnection(
   196→  organizationId: string,
   197→  connectionId: string,
   198→): Promise<ERPConnection | null> {
   199→  assertOrg(organizationId);
   200→  const ref = doc(db, ERP_COLLECTIONS.CONNECTIONS, connectionId);
   201→  const snap = await safeGetDoc(ref).catch(() => null);
   202→  if (!snap || !snap.exists()) return null;
   203→  return toConnection(snap.id, snap.data() as Record<string, unknown>);
   204→}

   206→export async function saveConnection(
   207→  organizationId: string,
   208→  data: Omit<ERPConnection, 'id' | 'createdAt' | 'updatedAt'>,
   209→): Promise<string> {
   210→  assertOrg(organizationId);
   211→  const ref = await safeAddDoc(collection(db, ERP_COLLECTIONS.CONNECTIONS), {
   212→    ...data,
   213→    organizationId,
   214→    createdAt: serverTimestamp(),
   215→    updatedAt: serverTimestamp(),
   216→  });
   217→  return ref.id;
   218→}

   323→// ─── Generic entity subscription + save ──────────────────────────────────────
   324→
   325→function subscribeToEntities<T>(
   326→  organizationId: string,
   327→  collName: string,
   328→  converter: (id: string, raw: Record<string, unknown>) => T,
   329→  callback: (records: T[]) => void,
   330→  options?: { onError?: (err: Error) => void; limitCount?: number; connectionId?: string; orderByField?: string; orderDir?: 'asc' | 'desc' },
   331→): Unsubscribe {
   332→  assertOrg(organizationId);
   333→  const constraints: QueryConstraint[] = [where('organizationId', '==', organizationId)];
   334→  if (options?.connectionId) constraints.push(where('connectionId', '==', options.connectionId));
   335→  if (options?.orderByField) {
   336→    constraints.push(orderBy(options.orderByField, options.orderDir ?? 'desc'));
   337→  }
   338→  if (options?.limitCount) constraints.push(limit(options.limitCount));
   339→  const q = query(collection(db, collName), ...constraints);
   340→  return onSnapshot(
   341→    q,
   342→    (snap) => callback(snap.docs.map((d) => converter(d.id, d.data() as Record<string, unknown>))),
   343→    (err) => options?.onError?.(err as Error),
   344→  );
   345→}
   346→
   347→async function saveEntities<T extends { id: string }>(
   348→  organizationId: string,
   349→  collName: string,
   350→  records: T[],
   351→): Promise<void> {
   352→  assertOrg(organizationId);
   353→  if (records.length === 0) return;
   354→  const batch = writeBatch(db);
   355→  for (const r of records) {
   356→    const { id: _id, ...data } = r;
   357→    void _id;
   358→    const ref = doc(db, collName, r.id);
   359→    batch.set(ref, { ...data, organizationId, updatedAt: serverTimestamp() }, { merge: true });
   360→  }
   361→  await safeCommit(batch);
   362→}
   363→
   364→async function deleteEntitiesForConnection(
   365→  organizationId: string,
   366→  collName: string,
   367→  connectionId: string,
   368→): Promise<void> {
   369→  assertOrg(organizationId);
   370→  const q = query(
   371→    collection(db, collName),
   372→    where('organizationId', '==', organizationId),
   373→    where('connectionId', '==', connectionId),
   374→  );
   375→  const snap = await safeGetDocs(q).catch(() => ({ docs: [] as Array<{ ref: import('firebase/firestore').DocumentReference }> }));
   376→  const batch = writeBatch(db);
   377→  snap.docs.forEach((d) => batch.delete(d.ref));
   378→  if (snap.docs.length > 0) await safeCommit(batch);
   379→}

   780→// ─── Disconnect cascade (delete all ERP data for a connection) ───────────────
   781→
   782→/**
   783→ * Delete ALL ERP data for a connection — used during disconnect.
   784→ * Order: delete child collections first, then sync jobs, then the connection.
   785→ */
   786→export async function cascadeDisconnect(
   787→  organizationId: string,
   788→  connectionId: string,
   789→): Promise<void> {
   790→  assertOrg(organizationId);
   791→  await Promise.all([
   792→    deleteEntitiesForConnection(organizationId, ERP_COLLECTIONS.CUSTOMERS, connectionId),
   793→    deleteEntitiesForConnection(organizationId, ERP_COLLECTIONS.VENDORS, connectionId),
   794→    deleteEntitiesForConnection(organizationId, ERP_COLLECTIONS.INVOICES, connectionId),
   795→    deleteEntitiesForConnection(organizationId, ERP_COLLECTIONS.INVENTORY, connectionId),
   796→    deleteEntitiesForConnection(organizationId, ERP_COLLECTIONS.LEDGERS, connectionId),
   797→    deleteEntitiesForConnection(organizationId, ERP_COLLECTIONS.PAYMENTS, connectionId),
   798→    deleteEntitiesForConnection(organizationId, ERP_COLLECTIONS.BANK_TRANSACTIONS, connectionId),
   799→    deleteEntitiesForConnection(organizationId, ERP_COLLECTIONS.TAXES, connectionId),
   800→    deleteEntitiesForConnection(organizationId, ERP_COLLECTIONS.SYNC_JOBS, connectionId),
   801→  ]).catch(() => { /* partial — continue */ });
   802→  await deleteConnection(organizationId, connectionId);
   803→}

   805→// ─── ERP summary (computed from Firestore data) ──────────────────────────────
   806→
   807→function emptySummary(): ERPSummary {
   808→  return {
   809→    connectedERPs: 0,
   810→    connectedProviders: [],
   811→    totalRevenue: 0,
   812→    totalExpenses: 0,
   813→    netProfit: 0,
   814→    outstandingReceivables: 0,
   815→    outstandingPayables: 0,
   816→    inventoryValue: 0,
   817→    cashPosition: 0,
   818→    lowStockItems: 0,
   819→    overdueInvoices: 0,
   820→    netTaxLiability: 0,
   821→    customerCount: 0,
   822→    vendorCount: 0,
   823→    inventoryItemCount: 0,
   824→    isConnected: false,
   825→    recentInvoices: [],
   826→    topCustomers: [],
   827→    topVendors: [],
   828→  };
   829→}
   830→
   831→/**
   832→ * Compute the ERPSummary from real-time Firestore data.
   833→ * Pure + deterministic — used by the useERP hook (memoized) and dashboard.
   834→ *
   835→ * Accepts the entity arrays (which the hook already has from subscriptions) so
   836→ * this is a pure function with no Firestore reads.
   837→ */
   838→export function computeERPSummary(
   839→  connections: ERPConnection[],
   840→  customers: ERPCustomer[],
   841→  vendors: ERPVendor[],
   842→  invoices: ERPInvoice[],
   843→  inventory: ERPInventoryItem[],
   844→  ledgers: ERPLedger[],
   845→  taxes: ERPTax[],
   846→): ERPSummary {
   847→  const connected = connections.filter((c) => c.connectionStatus === 'connected');
   848→  if (connections.length === 0 && customers.length === 0 && invoices.length === 0) {
   849→    return emptySummary();
   850→  }
   851→
   852→  // Revenue = Σ sales invoices grandTotal
   853→  const totalRevenue = round2(
   854→    invoices.filter((i) => i.invoiceType === 'sales').reduce((s, i) => s + i.grandTotal, 0),
   855→  );
   856→  // Expenses = Σ purchase invoices grandTotal
   857→  const totalExpenses = round2(
   858→    invoices.filter((i) => i.invoiceType === 'purchase').reduce((s, i) => s + i.grandTotal, 0),
   859→  );
   860→  const netProfit = round2(totalRevenue - totalExpenses);
   861→
   862→  const outstandingReceivables = round2(
   863→    customers.reduce((s, c) => s + c.outstandingBalance, 0),
   864→  );
   865→  const outstandingPayables = round2(
   866→    vendors.reduce((s, v) => s + v.outstandingPayable, 0),
   867→  );
   868→
   869→  const inventoryValue = round2(
   870→    inventory.reduce((s, i) => s + i.stockValue, 0),
   871→  );
   872→  const lowStockItems = inventory.filter(
   873→    (i) => i.stockStatus === 'low_stock' || i.stockStatus === 'out_of_stock',
   874→  ).length;
   875→
   876→  const now = new Date().toISOString();
   877→  const overdueInvoices = invoices.filter(
   878→    (i) => i.status === 'overdue' || (i.balanceDue > 0 && i.dueDate && i.dueDate < now.slice(0, 10)),
   879→  ).length;
   880→
   881→  // Cash position = Σ bank + cash ledger closing balances
   882→  const cashPosition = round2(
   883→    ledgers
   884→      .filter((l) => l.ledgerType === 'bank' || l.ledgerType === 'cash')
   885→      .reduce((s, l) => s + l.closingBalance, 0),
   886→  );
   887→
   888→  // Net tax liability = output tax - input tax (current period)
   889→  const outputTax = taxes.filter((t) => t.direction === 'output').reduce((s, t) => s + t.taxAmount, 0);
   890→  const inputTax = taxes.filter((t) => t.direction === 'input').reduce((s, t) => s + t.taxAmount, 0);
   891→  const netTaxLiability = round2(outputTax - inputTax);
   892→
   893→  const recentInvoices = invoices.slice(0, 10);
   894→
   895→  const topCustomers = [...customers]
   896→    .sort((a, b) => b.totalSales - a.totalSales)
   897→    .slice(0, 5)
   898→    .map((c) => ({ name: c.name, totalSales: round2(c.totalSales) }));
   899→
   900→  const topVendors = [...vendors]
   901→    .sort((a, b) => b.totalPurchases - a.totalPurchases)
   902→    .slice(0, 5)
   903→    .map((v) => ({ name: v.name, totalPurchases: round2(v.totalPurchases) }));
   904→
   905→  return {
   906→    connectedERPs: connected.length,
   907→    connectedProviders: connected.map((c) => c.provider),
   908→    totalRevenue,
   909→    totalExpenses,
   910→    netProfit,
   911→    outstandingReceivables,
   912→    outstandingPayables,
   913→    inventoryValue,
   914→    cashPosition,
   915→    lowStockItems,
   916→    overdueInvoices,
   917→    netTaxLiability,
   918→    customerCount: customers.length,
   919→    vendorCount: vendors.length,
   920→    inventoryItemCount: inventory.length,
   921→    isConnected: connected.length > 0,
   922→    recentInvoices,
   923→    topCustomers,
   924→    topVendors,
   925→  };
   926→}
