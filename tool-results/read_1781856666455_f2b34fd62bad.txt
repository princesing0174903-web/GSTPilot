   193→  const orbRef = useRef<HTMLDivElement>(null)
   194→
   195→  const handlePointerDown = (e: React.PointerEvent) => {
   196→    setIsDragging(true)
   197→    setHasMoved(false)
   198→    dragStart.current = {
   199→      x: e.clientX,
   200→      y: e.clientY,
   201→      posX: position.x,
   202→      posY: position.y,
   203→    }
   204→    ;(e.target as HTMLElement).setPointerCapture(e.pointerId)

   207→  const handlePointerMove = (e: React.PointerEvent) => {
   208→    if (!isDragging) return
   209→    const dx = e.clientX - dragStart.current.x

   213→      x: dragStart.current.posX + dx,

   385→// --- AI Thinking Indicator (V16 — Perplexity-style multi-step) ----------------
   386→
   387→function ThinkingIndicator({ steps }: { steps?: Array<{ label: string; duration: number }> }) {
   388→  const [currentStep, setCurrentStep] = useState(0)
   389→  const stepLabels = steps && steps.length > 0
   390→    ? steps.map(s => s.label)
   391→    : ['Thinking...', 'Generating answer...']
   392→
   393→  useEffect(() => {
   394→    if (!steps || steps.length === 0) return
   395→    let stepIdx = 0
   396→    const runStep = () => {
   397→      if (stepIdx >= steps.length - 1) return
   398→      setTimeout(() => {
   399→        stepIdx++
   400→        setCurrentStep(stepIdx)
   401→        runStep()
   402→      }, steps[stepIdx]?.duration || 500)
   403→    }
   404→    runStep()
   405→    return () => setCurrentStep(0)
   406→  }, [steps])
   407→
   408→  return (
   409→    <motion.div
   410→      initial={{ opacity: 0, y: 8 }}
   411→      animate={{ opacity: 1, y: 0 }}
   412→      className="flex gap-2.5"
   413→    >
   414→      <div className="shrink-0 h-8 w-8 rounded-full bg-gradient-to-br from-emerald-500 via-cyan-500 to-blue-500 flex items-center justify-center shadow-md">
   415→        <Brain className="h-4 w-4 text-foreground animate-pulse" />
   416→      </div>
   417→      <div className="flex-1">
   418→        <div className="rounded-2xl rounded-tl-sm glass-surface px-4 py-3">
   419→          <div className="flex items-center gap-2">
   420→            <div className="flex gap-1">
   421→              {[0, 1, 2].map((i) => (
   422→                <motion.div
   423→                  key={i}
   424→                  className="h-1.5 w-1.5 rounded-full bg-emerald-400"
   425→                  animate={{
   426→                    scale: [1, 1.4, 1],
   427→                    opacity: [0.4, 1, 0.4],
   428→                  }}
   429→                  transition={{
   430→                    duration: 1,
   431→                    repeat: Infinity,
   432→                    delay: i * 0.2,
   433→                    ease: 'easeInOut',
   434→                  }}
   435→                />
   436→              ))}
   437→            </div>
   438→            <motion.span
   439→              key={currentStep}
   440→              initial={{ opacity: 0, x: -4 }}
   441→              animate={{ opacity: 1, x: 0 }}
   442→              transition={{ duration: 0.3 }}
   443→              className="text-xs font-medium accent-text"
   444→            >
   445→              {stepLabels[currentStep] || stepLabels[0]}
   446→            </motion.span>
   447→          </div>
   448→          {/* Step progress dots */}
   449→          {stepLabels.length > 1 && (
   450→            <div className="mt-2 flex gap-1">
   451→              {stepLabels.map((_, i) => (
   452→                <div
   453→                  key={i}
   454→                  className={`h-0.5 flex-1 rounded-full transition-all duration-300 ${
   455→                    i <= currentStep ? 'accent-gradient' : 'bg-card/[0.08]'
   456→                  }`}
   457→                />
   458→              ))}
   459→            </div>
   460→          )}
   461→        </div>
   462→      </div>
   463→    </motion.div>
   464→  )
   465→}

   469→function useTypewriter(text: string, enabled: boolean, speed = 12) {
   470→  const [displayed, setDisplayed] = useState('')
   471→  const [done, setDone] = useState(false)
   472→
   473→  useEffect(() => {
   474→    if (!enabled) {
   475→      setDisplayed(text)
   476→      setDone(true)
   477→      return
   478→    }
   479→    setDisplayed('')
   480→    setDone(false)
   481→    let i = 0
   482→    const interval = setInterval(() => {
   483→      i++
   484→      setDisplayed(text.slice(0, i))
   485→      if (i >= text.length) {
   486→        clearInterval(interval)
   487→        setDone(true)
   488→      }
   489→    }, speed)
   490→    return () => clearInterval(interval)
   491→  }, [text, enabled, speed])
   492→
   493→  return { displayed, done }
   494→}

   498→function renderMarkdown(text: string) {
   499→  // Split by lines, render basic formatting
   500→  const lines = text.split('\n')
   501→  const elements: React.ReactNode[] = []
   502→  lines.forEach((line, idx) => {
   503→    if (!line.trim()) {
   504→      elements.push(<div key={idx} className="h-2" />)
   505→      return
   506→    }
   507→    // Headings (## or #)
   508→    if (line.startsWith('## ')) {
   509→      elements.push(
   510→        <div key={idx} className="font-bold text-sm mt-2 mb-1 text-foreground dark:text-secondary">
   511→          {line.replace(/^##\s+/, '')}
   512→        </div>
   513→      )
   514→      return
   515→    }
   516→    if (line.startsWith('# ')) {
   517→      elements.push(
   518→        <div key={idx} className="font-bold text-base mt-2 mb-1 text-foreground dark:text-secondary">
   519→          {line.replace(/^#\s+/, '')}
   520→        </div>
   521→      )
   522→      return
   523→    }
   524→    // Bullet points
   525→    if (line.startsWith('• ') || line.startsWith('- ')) {
   526→      const content = line.replace(/^[•-]\s+/, '')
   527→      elements.push(
   528→        <div key={idx} className="flex gap-2 text-sm leading-relaxed">
   529→          <span className="text-emerald-500 mt-0.5">•</span>
   530→          <span className="flex-1">{renderInline(content)}</span>
   531→        </div>
   532→      )
   533→      return
   534→    }
   535→    // Numbered list
   536→    const numMatch = line.match(/^(\d+)\.\s+(.*)/)
   537→    if (numMatch) {
   538→      elements.push(
   539→        <div key={idx} className="flex gap-2 text-sm leading-relaxed">
   540→          <span className="font-bold text-emerald-600 dark:text-emerald-400 min-w-[1.2rem]">{numMatch[1]}.</span>
   541→          <span className="flex-1">{renderInline(numMatch[2])}</span>
   542→        </div>
   543→      )
   544→      return
   545→    }
   546→    // Regular paragraph
   547→    elements.push(
   548→      <div key={idx} className="text-sm leading-relaxed">
   549→        {renderInline(line)}
   550→      </div>
   551→    )
   552→  })
   553→  return elements
   554→}
   555→
   556→function renderInline(text: string): React.ReactNode {
   557→  // Handle **bold** and `code`
   558→  const parts: React.ReactNode[] = []
   559→  const regex = /(\*\*[^*]+\*\*|`[^`]+`)/g
   560→  let lastIndex = 0
   561→  let match
   562→  let key = 0
   563→  while ((match = regex.exec(text)) !== null) {
   564→    if (match.index > lastIndex) {
   565→      parts.push(text.slice(lastIndex, match.index))
   566→    }
   567→    const token = match[0]
   568→    if (token.startsWith('**')) {
   569→      parts.push(
   570→        <strong key={key++} className="font-bold text-foreground dark:text-secondary">
   571→          {token.slice(2, -2)}
   572→        </strong>
   573→      )
   574→    } else if (token.startsWith('`')) {
   575→      parts.push(
   576→        <code key={key++} className="px-1 py-0.5 rounded bg-emerald-50 dark:bg-emerald-950/50 text-emerald-700 dark:text-emerald-300 text-xs font-mono">
   577→          {token.slice(1, -1)}
   578→        </code>
   579→      )
   580→    }
   581→    lastIndex = match.index + token.length
   582→  }
   583→  if (lastIndex < text.length) {
   584→    parts.push(text.slice(lastIndex))
   585→  }
   586→  return parts.length > 0 ? parts : text
   587→}

   623→        window.speechSynthesis.cancel()

   859→    const text = (overrideText ?? input).trim()

   861→
   862→    const userMsg: Message = {
   863→      id: `u-${Date.now()}`,
   864→      role: 'user',
   865→      content: text,
   866→      timestamp: Date.now(),

   936→          prev.map((m) => (m.id === assistantMsg.id ? { ...m, isTyping: false } : m))

  1283→export function GSTPilotIntelligence() {
  1284→  const [isOpen, setIsOpen] = useState(false)
  1285→  const [isFullscreen, setIsFullscreen] = useState(false)
  1286→  const [isListening, setIsListening] = useState(false)
  1287→  const [isThinking, setIsThinking] = useState(false)
  1288→
  1289→  // Ctrl+K shortcut
  1290→  useEffect(() => {
  1291→    const handler = (e: KeyboardEvent) => {
  1292→      if ((e.ctrlKey || e.metaKey) && e.key.toLowerCase() === 'k') {
  1293→        e.preventDefault()
  1294→        setIsOpen((prev) => !prev)
  1295→      }
  1296→      if (e.key === 'Escape' && isOpen) {
  1297→        setIsOpen(false)
  1298→      }
  1299→    }
  1300→    window.addEventListener('keydown', handler)
  1301→    return () => window.removeEventListener('keydown', handler)
  1302→  }, [isOpen])
  1303→
  1304→  const handleMinimize = () => {
  1305→    setIsOpen(false)
  1306→    setIsFullscreen(false)
  1307→  }
  1308→
  1309→  return (
  1310→    <>
  1311→      <FloatingOrb
  1312→        onClick={() => setIsOpen((prev) => !prev)}
  1313→        isOpen={isOpen}
  1314→        isListening={isListening}
  1315→        isThinking={isThinking}
  1316→      />
  1317→      <CommandCenter
  1318→        isOpen={isOpen}
  1319→        isFullscreen={isFullscreen}
  1320→        isListening={isListening}
  1321→        setIsListening={setIsListening}
  1322→        isThinking={isThinking}
  1323→        setIsThinking={setIsThinking}
  1324→        onClose={() => {
  1325→          setIsOpen(false)
  1326→          setIsFullscreen(false)
  1327→        }}
  1328→        onToggleFullscreen={() => setIsFullscreen((prev) => !prev)}
  1329→        onMinimize={handleMinimize}
  1330→      />
  1331→    </>
  1332→  )
  1333→}
  1334→
  1335→export default GSTPilotIntelligence
