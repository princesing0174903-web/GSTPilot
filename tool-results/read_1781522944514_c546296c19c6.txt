   158→   158→function getKanbanColumn(status: FilingStatus): KanbanColumn {
   159→   159→  switch (status) {
   160→   160→    case 'draft':
   161→   161→    case 'prepared':
   162→   162→      return 'draft';
   163→   163→    case 'validated':
   164→   164→    case 'reviewed':
   165→   165→    case 'generated':
   166→   166→      return 'ready';
   167→   167→    case 'filed':
   168→   168→      return 'filed';
   169→   169→    case 'reopened':
   170→   170→      return 'attention';
   171→   171→    default:
   172→   172→      return 'draft';
   173→   173→  }
   174→   174→}

   195→   195→function getAttentionSummary(filing: GSTRFiling): string {
   196→   196→  if (filing.issuesFound <= 0) return 'No issues';
   197→   197→  const parts: string[] = [];
   198→   198→  if (filing.criticalErrors > 0) parts.push(`${filing.criticalErrors} critical error${filing.criticalErrors > 1 ? 's' : ''}`);
   199→   199→  if (filing.warnings > 0) parts.push(`${filing.warnings} warning${filing.warnings > 1 ? 's' : ''}`);
   200→   200→  if (filing.issuesFound > 0 && parts.length === 0) parts.push(`${filing.issuesFound} issue${filing.issuesFound > 1 ? 's' : ''}`);
   201→   201→  return parts.join(' · ');
   202→   202→}

   211→   211→// ─── Section Breakdown Computation ──────────────────────────────────────────────
   212→   212→
   213→   213→function computeSectionsForFiling(
   214→   214→  returnType: string,
   215→   215→  invoices: { taxableValue: number; cgst: number; sgst: number; igst: number; buyerGstin?: string | null; status: string }[]
   216→   216→): SectionBreakdown[] {
   217→   217→  if (returnType === 'GSTR-1') {
   218→   218→    const b2b = invoices.filter(inv => inv.buyerGstin);
   219→   219→    const b2cLarge = invoices.filter(inv => !inv.buyerGstin && inv.taxableValue >= 250000);
   220→   220→    const b2cSmall = invoices.filter(inv => !inv.buyerGstin && inv.taxableValue < 250000);
   221→   221→
   222→   222→    return [
   223→   223→      {
   224→   224→        section: 'B2B',
   225→   225→        invoiceCount: b2b.length,
   226→   226→        taxableValue: b2b.reduce((s, i) => s + i.taxableValue, 0),
   227→   227→        taxAmount: b2b.reduce((s, i) => s + i.cgst + i.sgst + i.igst, 0),
   228→   228→      },
   229→   229→      {
   230→   230→        section: 'B2C Large',
   231→   231→        invoiceCount: b2cLarge.length,
   232→   232→        taxableValue: b2cLarge.reduce((s, i) => s + i.taxableValue, 0),
   233→   233→        taxAmount: b2cLarge.reduce((s, i) => s + i.cgst + i.sgst + i.igst, 0),
   234→   234→      },
   235→   235→      {
   236→   236→        section: 'B2C Small',
   237→   237→        invoiceCount: b2cSmall.length,
   238→   238→        taxableValue: b2cSmall.reduce((s, i) => s + i.taxableValue, 0),
   239→   239→        taxAmount: b2cSmall.reduce((s, i) => s + i.cgst + i.sgst + i.igst, 0),
   240→   240→      },
   241→   241→      {
   242→   242→        section: 'CDNR',
   243→   243→        invoiceCount: 0,
   244→   244→        taxableValue: 0,
   245→   245→        taxAmount: 0,
   246→   246→      },
   247→   247→    ];
   248→   248→  }
   249→   249→
   250→   250→  // GSTR-3B
   251→   251→  const totalTax = invoices.reduce((s, i) => s + i.cgst + i.sgst + i.igst, 0);
   252→   252→  const totalTaxable = invoices.reduce((s, i) => s + i.taxableValue, 0);
   253→   253→  return [
   254→   254→    {
   255→   255→      section: 'Outward Supplies',
   256→   256→      invoiceCount: invoices.length,
   257→   257→      taxableValue: totalTaxable,
   258→   258→      taxAmount: totalTax,
   259→   259→    },
   260→   260→    {
   261→   261→      section: 'Inward Supplies (RC)',
   262→   262→      invoiceCount: 0,
   263→   263→      taxableValue: 0,
   264→   264→      taxAmount: 0,
   265→   265→    },
   266→   266→    {
   267→   267→      section: 'ITC Claims',
   268→   268→      invoiceCount: 0,
   269→   269→      taxableValue: 0,
   270→   270→      taxAmount: Math.round(-totalTax * 0.6),
   271→   271→    },
   272→   272→    {
   273→   273→      section: 'Tax Paid',
   274→   274→      invoiceCount: 0,
   275→   275→      taxableValue: 0,
   276→   276→      taxAmount: Math.round(totalTax * 0.75),
   277→   277→    },
   278→   278→  ];
