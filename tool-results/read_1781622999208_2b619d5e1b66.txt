   151→function formatINR(amount: number): string {

   156→function formatDate(ts: number | string | undefined): string {
   157→  try {
   158→    const d = typeof ts === 'number' ? new Date(ts) : new Date(ts as string);
   159→    if (isNaN(d.getTime())) return '—';
   160→    const dd = String(d.getDate()).padStart(2, '0');
   161→    const mm = String(d.getMonth() + 1).padStart(2, '0');
   162→    const yyyy = d.getFullYear();
   163→    return `${dd}/${mm}/${yyyy}`;
   164→  } catch {
   165→    return '—';
   166→  }
   167→}

   169→function formatTime(ts: number): string {
   170→  const d = new Date(ts);
   171→  let h = d.getHours();
   172→  const m = String(d.getMinutes()).padStart(2, '0');
   173→  const ampm = h >= 12 ? 'PM' : 'AM';
   174→  h = h % 12 || 12;
   175→  return `${h}:${m} ${ampm}`;
   176→}

   185→function buildLiveDataContext(opts: {
   186→  clients: Array<Record<string, unknown>>;
   187→  invoices: Array<Record<string, unknown>>;
   188→  returns: Array<Record<string, unknown>>;
   189→  notifications: Array<Record<string, unknown>>;
   190→  activities: Array<Record<string, unknown>>;
   191→}): string {
   192→  try {
   193→    const { clients, invoices, returns, notifications, activities } = opts;
   194→    const now = new Date();
   195→    const dd = String(now.getDate()).padStart(2, '0');
   196→    const mm = String(now.getMonth() + 1).padStart(2, '0');
   197→    const yyyy = now.getFullYear();
   198→    const hh = String(now.getHours()).padStart(2, '0');
   199→    const min = String(now.getMinutes()).padStart(2, '0');
   200→    const stamp = `${dd}/${mm}/${yyyy} ${hh}:${min}`;
   201→
   202→    const lines: string[] = [`LIVE DATA SNAPSHOT (${stamp}):`];
   203→
   204→    // Clients
   205→    const totalClients = clients.length;
   206→    const activeClients = clients.filter((c) => c.status === 'active').length;
   207→    lines.push(`- Total clients: ${totalClients}`);
   208→    lines.push(`- Active clients: ${activeClients}`);
   209→
   210→    // Invoices
   211→    const activeInvoices = invoices.filter(
   212→      (i) => i.status !== 'cancelled' && i.status !== 'draft',
   213→    );
   214→    const totalInvoiceValue = activeInvoices.reduce((s, i) => {
   215→      const v =
   216→        (i.totalAmount as number) ||
   217→        (i.grandTotal as number) ||
   218→        (i.total as number) ||
   219→        ((i.subTotal as number) || 0) + ((i.totalTax as number) || 0);
   220→      return s + (typeof v === 'number' ? v : 0);
   221→    }, 0);
   222→    const overdueInvoices = activeInvoices.filter((i) => {
   223→      const status = i.status as string;
   224→      return status === 'overdue' || status === 'unpaid';
   225→    });
   226→    const overdueValue = overdueInvoices.reduce((s, i) => {
   227→      const v =
   228→        (i.totalAmount as number) ||
   229→        (i.grandTotal as number) ||
   230→        (i.total as number) ||
   231→        0;
   232→      return s + (typeof v === 'number' ? v : 0);
   233→    }, 0);
   234→    lines.push(
   235→      `- Active invoices: ${activeInvoices.length} (${formatINR(totalInvoiceValue)} total)`,
   236→    );
   237→    lines.push(
   238→      `- Overdue invoices: ${overdueInvoices.length} (${formatINR(overdueValue)})`,
   239→    );
   240→
   241→    // Returns
   242→    const gstr1Pending = returns.filter(
   243→      (r) => (r.returnType as string) === 'GSTR-1' && (r.status as string) !== 'filed',
   244→    ).length;
   245→    const gstr3bPending = returns.filter(
   246→      (r) => (r.returnType as string) === 'GSTR-3B' && (r.status as string) !== 'filed',
   247→    ).length;
   248→    const filedReturns = returns.filter((r) => r.status === 'filed').length;
   249→    lines.push(
   250→      `- Pending returns: ${gstr1Pending} GSTR-1, ${gstr3bPending} GSTR-3B`,
   251→    );
   252→    lines.push(`- Filed returns: ${filedReturns}`);
   253→
   254→    // Notifications + Activities
   255→    const unread = notifications.filter((n) => n.read === false).length;
   256→    lines.push(`- Unread notifications: ${unread}`);
   257→    lines.push(`- Recent activities: ${activities.length} in last 24h`);
   258→
   259→    // Top 5 clients by revenue
   260→    const clientRevenue = new Map<string, number>();
   261→    invoices.forEach((inv) => {
   262→      const cid = (inv.clientId as string) || (inv.clientName as string) || 'unknown';
   263→      const v =
   264→        (inv.totalAmount as number) ||
   265→        (inv.grandTotal as number) ||
   266→        (inv.total as number) ||
   267→        0;
   268→      clientRevenue.set(cid, (clientRevenue.get(cid) || 0) + (typeof v === 'number' ? v : 0));
   269→    });
   270→    const top5 = [...clientRevenue.entries()]
   271→      .sort((a, b) => b[1] - a[1])
   272→      .slice(0, 5);
   273→    if (top5.length > 0) {
   274→      const top5Str = top5
   275→        .map(([name, rev]) => {
   276→          const client = clients.find(
   277→            (c) => c.id === name || (c.tradeName as string) === name,
   278→          );
   279→          const label = (client?.tradeName as string) || name;
   280→          return `${label} (${formatINR(rev)})`;
   281→        })
   282→        .join(', ');
   283→      lines.push(`- Top 5 clients by revenue: ${top5Str}`);
   284→    }
   285→
   286→    // High-risk clients (health score < 60)
   287→    const highRisk = clients.filter((c) => {
   288→      const score = (c.healthScore as number) ?? 100;
   289→      return typeof score === 'number' && score < 60;
   290→    });
   291→    if (highRisk.length > 0) {
   292→      const riskStr = highRisk
   293→        .slice(0, 5)
   294→        .map(
   295→          (c) =>
   296→            `${(c.tradeName as string) || (c.name as string) || 'Unknown'} (health ${c.healthScore ?? '?'})`,
   297→        )
   298→        .join(', ');
   299→      lines.push(`- High-risk clients: ${riskStr}`);
   300→    }
   301→
   302→    return lines.join('\n');
   303→  } catch {
   304→    return 'LIVE DATA SNAPSHOT: Data currently unavailable.';
   305→  }
   306→}

   381→// ─── Sub-component: Message Content (mini-markdown) ───────────────────────────
   382→// Renders **bold**, line breaks, and bullet points (•).
   383→
   384→function MessageContent({ text }: { text: string }) {
   385→  const blocks = useMemo(() => {
   386→    const paragraphs = text.split('\n');
   387→    return paragraphs.map((line) => {
   388→      const trimmed = line.trim();
   389→      if (trimmed.startsWith('• ') || trimmed.startsWith('- ')) {
   390→        return { type: 'bullet' as const, content: trimmed.slice(2) };
   391→      }
   392→      if (/^\d+\.\s/.test(trimmed)) {
   393→        return { type: 'numbered' as const, content: trimmed.replace(/^\d+\.\s/, '') };
   394→      }
   395→      return { type: 'text' as const, content: trimmed };
   396→    });
   397→  }, [text]);
   398→
   399→  const renderInline = (s: string, keyBase: string) => {
   400→    // Split on **bold** markers
   401→    const parts = s.split(/(\*\*[^*]+\*\*)/g);
   402→    return parts.map((part, i) => {
   403→      if (part.startsWith('**') && part.endsWith('**')) {
   404→        return (
   405→          <strong key={`${keyBase}-${i}`} className="font-semibold text-foreground">
   406→            {part.slice(2, -2)}
   407→          </strong>
   408→        );
   409→      }
   410→      return <span key={`${keyBase}-${i}`}>{part}</span>;
   411→    });
   412→  };
   413→
   414→  return (
   415→    <div className="space-y-1">
   416→      {blocks.map((b, idx) => {
   417→        if (b.type === 'bullet') {
   418→          return (
   419→            <div key={idx} className="flex gap-2">
   420→              <span className="mt-0.5 text-emerald-400">•</span>
   421→              <span>{renderInline(b.content, `b${idx}`)}</span>
   422→            </div>
   423→          );
   424→        }
   425→        if (b.type === 'numbered') {
   426→          return (
   427→            <div key={idx} className="flex gap-2">
   428→              <span className="text-emerald-400 font-medium">{idx + 1}.</span>
   429→              <span>{renderInline(b.content, `n${idx}`)}</span>
   430→            </div>
   431→          );
   432→        }
   433→        if (!b.content) return <div key={idx} className="h-1" />;
   434→        return <p key={idx}>{renderInline(b.content, `t${idx}`)}</p>;
   435→      })}
   436→    </div>
   437→  );
   438→}

   482→// ─── Hook: Typing Effect ──────────────────────────────────────────────────────
   483→// Reveals text progressively in small chunks for a "typing" feel.
   484→
   485→function useTypingEffect(fullText: string, active: boolean, speed = 16, chunkSize = 3) {
   486→  // revealLength is only mutated inside the interval callback (never
   487→  // synchronously in the effect body), which keeps React's
   488→  // set-state-in-effect lint rule happy and avoids cascading renders.
   489→  const [revealLength, setRevealLength] = useState(0);
   490→
   491→  useEffect(() => {
   492→    if (!active) return;
   493→    let i = 0;
   494→    const interval = setInterval(() => {
   495→      i += chunkSize;
   496→      setRevealLength(i);
   497→      if (i >= fullText.length) {
   498→        clearInterval(interval);
   499→      }
   500→    }, speed);
   501→    return () => clearInterval(interval);
   502→  }, [fullText, active, speed, chunkSize]);
   503→
   504→  // When not active (e.g. older messages), show the full text instantly
   505→  // via a derived value — no state mutation required.
   506→  const displayed = active ? fullText.slice(0, revealLength) : fullText;
   507→  const completed = displayed.length >= fullText.length;
   508→
   509→  return { displayed, completed };
   510→}

   515→function TypingMessage({
   516→  message,
   517→  isLatest,
   518→  onTypingComplete,
   519→}: {
   520→  message: ChatMessage;
   521→  isLatest: boolean;
   522→  onTypingComplete: () => void;

   524→  const { displayed, completed } = useTypingEffect(
   525→    message.content,
   526→    isLatest,
   527→    16,
   528→    3,

   532→    if (completed && isLatest) {

   537→  return (
   538→    <div className="space-y-1">
   539→      <MessageContent text={displayed} />
   540→      {!completed && (
   541→        <motion.span

   543→          animate={{ opacity: [1, 0, 1] }}
   544→          transition={{ duration: 0.8, repeat: Infinity }}
   545→        />
   546→      )}
