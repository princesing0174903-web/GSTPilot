   222→   222→export interface TaxCalculation {
   223→   223→  taxType: string;
   224→   224→  rate: number;
   225→   225→  baseAmount: number;
   226→   226→  taxAmount: number;
   227→   227→  totalAmount: number;
   228→   228→}
   229→   229→
   230→   230→export const calculateTax = (
   231→   231→  amount: number,
   232→   232→  countryCode: CountryCode,
   233→   233→  taxType: 'indirect' | 'corporate' | 'payroll' | 'import' | 'export' = 'indirect'
   234→   234→): TaxCalculation => {
   235→   235→  const country = getCountry(countryCode);
   236→   236→  let rate = 0;
   237→   237→  let taxName = '';
   238→   238→  switch (taxType) {
   239→   239→    case 'indirect': rate = country.taxRate; taxName = country.primaryTaxName; break;
   240→   240→    case 'corporate': rate = country.corporateTaxRate; taxName = 'Corporate Tax'; break;
   241→   241→    case 'payroll': rate = country.payrollTaxRate; taxName = 'Payroll Tax'; break;
   242→   242→    case 'import': rate = country.importDuty; taxName = 'Import Duty'; break;
   243→   243→    case 'export': rate = country.exportDuty; taxName = 'Export Duty'; break;
   244→   244→  }
   245→   245→  return {
   246→   246→    taxType: taxName,
   247→   247→    rate,
   248→   248→    baseAmount: amount,
   249→   249→    taxAmount: (amount * rate) / 100,
   250→   250→    totalAmount: amount + (amount * rate) / 100,
   251→   251→  };
   252→   252→};
