   108→// ─── Helpers ───────────────────────────────────────────────────────────────
   109→function formatFileSize(bytes: number): string {
   110→  if (bytes === 0) return '0 B';
   111→  const k = 1024;
   112→  const sizes = ['B', 'KB', 'MB', 'GB'];
   113→  const i = Math.floor(Math.log(bytes) / Math.log(k));
   114→  return parseFloat((bytes / Math.pow(k, i)).toFixed(1)) + ' ' + sizes[i];
   115→}

   117→function getFileTypeColor(type: string): string {
   118→  const t = type.toLowerCase();
   119→  if (t === 'pdf') return 'bg-red-100 text-red-700 dark:bg-red-950/40 dark:text-red-400';
   120→  if (['xlsx', 'xls', 'csv'].includes(t)) return 'bg-emerald-100 text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-400';
   121→  if (['docx', 'doc'].includes(t)) return 'bg-blue-100 text-blue-700 dark:bg-blue-950/40 dark:text-blue-400';
   122→  if (['jpg', 'jpeg', 'png', 'gif', 'svg', 'webp'].includes(t)) return 'bg-purple-100 text-purple-700 dark:bg-purple-950/40 dark:text-purple-400';
   123→  if (['pptx', 'ppt'].includes(t)) return 'bg-amber-100 text-amber-700 dark:bg-amber-950/40 dark:text-amber-400';
   124→  return 'bg-slate-100 text-slate-700 dark:bg-slate-800 dark:text-slate-400';

   148→// ─── Animated Number Hook ──────────────────────────────────────────────────
   149→function useAnimatedNumber(target: number, duration: number = 1200) {
   150→  const [current, setCurrent] = useState(0);
   151→  const ref = useRef<number | null>(null);
   152→  const startTime = useRef<number | null>(null);
   153→
   154→  useEffect(() => {
   155→    startTime.current = null;
   156→    const startValue = current;
   157→
   158→    function step(timestamp: number) {
   159→      if (!startTime.current) startTime.current = timestamp;
   160→      const elapsed = timestamp - startTime.current;
   161→      const progress = Math.min(elapsed / duration, 1);
   162→      const eased = 1 - Math.pow(1 - progress, 3);
   163→      setCurrent(Math.round(startValue + (target - startValue) * eased));
   164→      if (progress < 1) {
   165→        ref.current = requestAnimationFrame(step);
   166→      }
   167→    }
   168→
   169→    ref.current = requestAnimationFrame(step);
   170→    return () => {
   171→      if (ref.current) cancelAnimationFrame(ref.current);
   172→    };
   173→  }, [target, duration]);
   174→
   175→  return current;
   176→}
