    88→// ═══════════════════════════════════════════════════════════════════════════════
    89→// LOCAL STORAGE HELPERS
    90→// ═══════════════════════════════════════════════════════════════════════════════
    91→
    92→function loadRecent(): RecentItem[] {
    93→  try {
    94→    const raw = localStorage.getItem(RECENT_STORAGE_KEY);
    95→    return raw ? JSON.parse(raw) : [];
    96→  } catch {
    97→    return [];
    98→  }
    99→}
   100→
   101→function saveRecent(items: RecentItem[]) {
   102→  try {
   103→    localStorage.setItem(RECENT_STORAGE_KEY, JSON.stringify(items.slice(0, MAX_RECENT)));
   104→  } catch {
   105→    // ignore quota errors
   106→  }
   107→}
   108→
   109→function loadFavorites(): FavoriteItem[] {
   110→  try {
   111→    const raw = localStorage.getItem(FAVORITES_STORAGE_KEY);
   112→    return raw ? JSON.parse(raw) : [];
   113→  } catch {
   114→    return [];
   115→  }
   116→}
   117→
   118→function saveFavorites(items: FavoriteItem[]) {
   119→  try {
   120→    localStorage.setItem(FAVORITES_STORAGE_KEY, JSON.stringify(items));
   121→  } catch {
   122→    // ignore quota errors
   123→  }
   124→}

   864→// COMMAND ITEM ROW — Reusable row for each palette item
   865→// ═══════════════════════════════════════════════════════════════════════════════
   866→
   867→interface CommandItemRowProps {
   868→  icon: React.ElementType;
   869→  label: string;
   870→  description?: string;
   871→  shortcut?: string;
   872→  isFavorite?: boolean;
   873→  onToggleFavorite?: () => void;
   874→  onSelect: () => void;
   875→}
   876→
   877→function CommandItemRow({
   878→  icon: Icon,
   879→  label,
   880→  description,
   881→  shortcut,
   882→  isFavorite,
   883→  onToggleFavorite,
   884→  onSelect,
   885→}: CommandItemRowProps) {
   886→  const [hovered, setHovered] = React.useState(false);
   887→
   888→  return (
   889→    <motion.div
   890→      className="group relative flex items-center gap-3 rounded-lg px-3 py-2 cursor-pointer transition-colors duration-150 hover:bg-white/[0.06]"
   891→      onClick={onSelect}
   892→      onMouseEnter={() => setHovered(true)}
   893→      onMouseLeave={() => setHovered(false)}
   894→      whileHover={{ x: 2 }}
   895→      transition={{ duration: 0.1 }}
   896→    >
   897→      {/* Active left accent bar */}
   898→      <span className="absolute left-0 top-1/2 h-5 w-[2px] -translate-y-1/2 rounded-full bg-emerald-400 opacity-0 group-hover:opacity-100 transition-opacity duration-150" />
   899→      <div className="flex h-7 w-7 shrink-0 items-center justify-center rounded-md bg-white/[0.04] transition-colors group-hover:bg-emerald-400/10">
   900→        <Icon className="h-3.5 w-3.5 text-muted-foreground transition-colors group-hover:text-emerald-400" />
   901→      </div>
   902→      <div className="flex-1 min-w-0">
   903→        <p className="text-sm font-medium text-white truncate">{label}</p>
   904→        {description && (
   905→          <p className="text-xs text-muted-foreground truncate">{description}</p>
   906→        )}
   907→      </div>
   908→      <div className="flex items-center gap-2 shrink-0">
   909→        {/* Favorite toggle for commands */}
   910→        {onToggleFavorite && (
   911→          <button
   912→            onClick={(e) => {
   913→              e.stopPropagation();
   914→              onToggleFavorite();

   916→            className="opacity-0 group-hover:opacity-100 transition-opacity p-0.5 hover:bg-white/[0.08] rounded"
   917→            aria-label={isFavorite ? 'Remove from favorites' : 'Add to favorites'}
   918→          >
   919→            <Star
   920→              className={`h-3 w-3 ${
   921→                isFavorite
   922→                  ? 'fill-cyan-400 text-cyan-400'

   924→              }`}
   925→            />
   926→          </button>
   927→        )}

   929→          <kbd className="badge-premium hidden sm:inline-flex font-mono text-[10px]">
   930→            {shortcut}
   931→          </kbd>

   934→      </div>
   935→    </motion.div>
   936→  );
   937→}
