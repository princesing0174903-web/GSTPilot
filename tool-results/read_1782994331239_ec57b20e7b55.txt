    88→function NoticeCategoryChart() {
    89→  const data = [
    90→    { label: 'GST', value: 3, color: '#10b981' },
    91→    { label: 'Income Tax', value: 2, color: '#f59e0b' },
    92→    { label: 'ROC', value: 1, color: '#6ee7b7' },
    93→    { label: 'Labour', value: 1, color: '#94a3b8' },
    94→    { label: 'Custom', value: 1, color: '#a7f3d0' },
    95→  ]
    96→  const cx = 75
    97→  const cy = 75
    98→  const r = 55
    99→  const innerR = 35
   100→  const total = data.reduce((s, d) => s + d.value, 0)
   101→
   102→  let startAngle = -90
   103→  const slices = data.map(d => {
   104→    const angle = (d.value / total) * 360
   105→    const endAngle = startAngle + angle
   106→    const slice = { ...d, startAngle, endAngle }
   107→    startAngle = endAngle
   108→    return slice
   109→  })
   110→
   111→  const describeArc = (cx: number, cy: number, r: number, startA: number, endA: number) => {
   112→    const rad = (a: number) => (a * Math.PI) / 180
   113→    const x1 = cx + r * Math.cos(rad(startA))
   114→    const y1 = cy + r * Math.sin(rad(startA))
   115→    const x2 = cx + r * Math.cos(rad(endA))
   116→    const y2 = cy + r * Math.sin(rad(endA))
   117→    const largeArc = endA - startA > 180 ? 1 : 0
   118→    return `M${x1},${y1} A${r},${r} 0 ${largeArc} 1 ${x2},${y2}`
   119→  }
   120→
   121→  return (
   122→    <div className="flex flex-col sm:flex-row items-center gap-4">
   123→      <svg viewBox="0 0 150 150" className="w-36 h-36 shrink-0">
   124→        {slices.map((s, i) => (
   125→          <path key={i} d={describeArc(cx, cy, r, s.startAngle, s.endAngle)} fill="none" stroke={s.color} strokeWidth="20" />
   126→        ))}
   127→        <circle cx={cx} cy={cy} r={innerR} fill="white" />
   128→        <text x={cx} y={cy - 4} className="text-[16px] font-bold fill-slate-900" textAnchor="middle">{total}</text>
   129→        <text x={cx} y={cy + 10} className="text-[7px] fill-slate-500" textAnchor="middle">Active</text>
   130→      </svg>
   131→      <div className="flex flex-col gap-2">
   132→        {data.map((d) => (
   133→          <div key={d.label} className="flex items-center gap-2">
   134→            <div className="h-2.5 w-2.5 rounded-full shrink-0" style={{ backgroundColor: d.color }} />
   135→            <span className="text-[11px] text-slate-600">{d.label}</span>
   136→            <span className="text-[11px] font-semibold text-slate-800 ml-auto">{d.value}</span>
   137→          </div>
   138→        ))}
   139→      </div>
   140→    </div>
   141→  )
   142→}

   144→function ResponseTimeChart() {
   145→  const months = ['Oct', 'Nov', 'Dec', 'Jan', 'Feb']
   146→  const avgDays = [9.5, 8.2, 7.8, 7.1, 6.2]
   147→  const w = 320
   148→  const h = 160
   149→  const padL = 35
   150→  const padB = 25
   151→  const padT = 10
   152→  const chartW = w - padL - 10
   153→  const chartH = h - padB - padT
   154→  const maxVal = 10
   155→  const minVal = 5
   156→
   157→  const points = avgDays.map((v, i) => {
   158→    const x = padL + (i / (avgDays.length - 1)) * chartW
   159→    const y = padT + chartH - ((v - minVal) / (maxVal - minVal)) * chartH
   160→    return { x, y, v }
   161→  })
   162→
   163→  const pathD = points.map((p, i) => `${i === 0 ? 'M' : 'L'}${p.x},${p.y}`).join(' ')
   164→  const areaD = pathD + ` L${points[points.length - 1].x},${h - padB} L${points[0].x},${h - padB} Z`
   165→
   166→  return (
   167→    <svg viewBox={`0 0 ${w} ${h}`} className="w-full h-auto">
   168→      {[0, 1, 2, 3, 4].map(i => {
   169→        const y = padT + (i / 4) * chartH
   170→        return <line key={i} x1={padL} y1={y} x2={w - 10} y2={y} stroke="#e2e8f0" strokeWidth="0.5" />
   171→      })}
   172→      <path d={areaD} fill="url(#responseGrad)" opacity="0.3" />
   173→      <path d={pathD} fill="none" stroke="#10b981" strokeWidth="2" />
   174→      {points.map((p, i) => (
   175→        <g key={i}>
   176→          <circle cx={p.x} cy={p.y} r="3.5" fill="#10b981" stroke="white" strokeWidth="1.5" />
   177→          <text x={p.x} y={p.y - 8} className="text-[8px] fill-slate-500" textAnchor="middle">{p.v}d</text>
   178→        </g>
   179→      ))}
   180→      {months.map((m, i) => (
   181→        <text key={m} x={padL + (i / (months.length - 1)) * chartW} y={h - 6} className="text-[9px] fill-slate-400" textAnchor="middle">{m}</text>
   182→      ))}
   183→      <text x={padL - 5} y={padT + 4} className="text-[8px] fill-slate-400" textAnchor="end">10d</text>
   184→      <text x={padL - 5} y={padT + chartH + 4} className="text-[8px] fill-slate-400" textAnchor="end">5d</text>
   185→      <defs>
   186→        <linearGradient id="responseGrad" x1="0" y1="0" x2="0" y2="1">
   187→          <stop offset="0%" stopColor="#10b981" />
   188→          <stop offset="100%" stopColor="#10b981" stopOpacity="0" />
   189→        </linearGradient>
   190→      </defs>
   191→    </svg>
   192→  )
   193→}
