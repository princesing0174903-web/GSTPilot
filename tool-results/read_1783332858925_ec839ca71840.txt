    28→// ─── API Fetch Helper ─────────────────────────────────────────────────────────
    29→
    30→async function apiFetch<T>(url: string, options?: RequestInit): Promise<T> {
    31→  const res = await fetch(url, {
    32→    headers: { 'Content-Type': 'application/json', ...options?.headers },
    33→    ...options,
    34→  });
    35→  if (!res.ok) {
    36→    const error = await res.json().catch(() => ({ error: 'Request failed' }));
    37→    throw new Error(error.error || `HTTP ${res.status}`);
    38→  }
    39→  return res.json();
    40→}

   248→export function useClient(
   249→  id: string | undefined,
   250→  options?: Omit<
   251→    UseQueryOptions<ClientsResponse, Error>,
   252→    'queryKey' | 'queryFn'
   253→  >
   254→) {
   255→  return useQuery<ClientsResponse, Error>({
   256→    queryKey: queryKeys.clients.detail(id ?? ''),
   257→    queryFn: () =>
   258→      apiFetch<ClientsResponse>('/api/clients').then((res) => {
   259→        // Filter to single client if id provided
   260→        if (id && res.clients) {
   261→          const client = res.clients.find((c: any) => c.id === id);
   262→          return { client: client ?? null, clients: res.clients };
   263→        }
   264→        return res;
   265→      }),
   266→    enabled: !!id,
   267→    ...options,
   268→  });
   269→}

   744→  >

   749→    mutationFn: (data) => {
   750→      if (data instanceof FormData) {
   751→        // FormData upload — don't set Content-Type (browser sets multipart boundary)
   752→        return fetch('/api/upload', {
   753→          method: 'POST',
   754→          body: data,
   755→        }).then(async (res) => {
   756→          if (!res.ok) {
   757→            const error = await res.json().catch(() => ({ error: 'Upload failed' }));
   758→            throw new Error(error.error || `HTTP ${res.status}`);
   759→          }
   760→          return res.json();
   761→        });
   762→      }
   763→      // JSON upload
   764→      return apiFetch<DocumentResponse>('/api/documents', {
   765→        method: 'POST',
   766→        body: JSON.stringify(data),
   767→      });
   768→    },
   769→    onSuccess: (data, variables, context) => {
   770→      const clientId = variables instanceof FormData ? variables.get('clientId') as string : (variables as any).clientId;
   771→      queryClient.invalidateQueries({
   772→        queryKey: queryKeys.documents.all(clientId),
   773→      });
   774→      queryClient.invalidateQueries({ queryKey: queryKeys.documents.all() });
   775→      options?.onSuccess?.(data, variables, context);
   776→    },
   777→    ...options,
   778→  });
