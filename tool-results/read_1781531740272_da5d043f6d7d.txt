    65→function getDaysRemaining(dueDateStr: string): number {
    66→  const dueDate = new Date(dueDateStr);
    67→  const today = new Date();
    68→  today.setHours(0, 0, 0, 0);
    69→  dueDate.setHours(0, 0, 0, 0);
    70→  return Math.ceil((dueDate.getTime() - today.getTime()) / (1000 * 60 * 60 * 24));
    71→}
    72→
    73→function formatDaysRemaining(days: number): string {
    74→  if (days < 0) return `${Math.abs(days)}d overdue`;
    75→  if (days === 0) return 'Due today';
    76→  if (days === 1) return 'Tomorrow';
    77→  return `${days}d left`;
    78→}
    79→
    80→function timeAgo(dateStr: string): string {
    81→  if (!dateStr) return '';
    82→  const now = new Date();
    83→  const date = new Date(dateStr);
    84→  const seconds = Math.floor((now.getTime() - date.getTime()) / 1000);
    85→  if (seconds < 60) return 'just now';
    86→  const minutes = Math.floor(seconds / 60);
    87→  if (minutes < 60) return `${minutes}m ago`;
    88→  const hours = Math.floor(minutes / 60);
    89→  if (hours < 24) return `${hours}h ago`;
    90→  const days = Math.floor(hours / 24);
    91→  if (days < 7) return `${days}d ago`;
    92→  return date.toLocaleDateString('en-IN', { day: 'numeric', month: 'short' });
    93→}

    99→interface SmartCardProps {
   100→  label: string;
   101→  value: string | number;
   102→  icon: React.ReactNode;
   103→  iconColor: string;
   104→  iconBg: string;
   105→  subtitle: string;

   112→
   113→function SmartCard({ label, value, icon, iconColor, iconBg, subtitle, trend, trendValue, pulse, index, onClick }: SmartCardProps) {
   114→  const trendIcon = trend === 'up' ? <TrendingUp className="h-3 w-3" /> : trend === 'down' ? <TrendingDown className="h-3 w-3" /> : <Minus className="h-3 w-3" />;
   115→  const trendColor = trend === 'up' ? 'text-emerald-600' : trend === 'down' ? 'text-red-500' : 'text-muted-foreground';
   116→
   117→  return (
   118→    <motion.div
   119→      initial={{ opacity: 0, y: 16 }}
   120→      animate={{ opacity: 1, y: 0 }}
   121→      transition={{ duration: 0.35, delay: index * 0.05 }}
   122→    >
   123→      <Card className={`hover:shadow-md transition-all duration-200 ${onClick ? 'cursor-pointer' : ''}`} onClick={onClick}>
   124→        <CardContent className="p-4">
   125→          <div className="flex items-start justify-between">
   126→            <div className="space-y-1 min-w-0 flex-1">
   127→              <p className="text-[10px] font-semibold text-muted-foreground uppercase tracking-widest truncate">
   128→                {label}
   129→              </p>
   130→              <div className="flex items-baseline gap-2">
   131→                <p className="text-2xl font-bold text-foreground tracking-tight">{value}</p>
   132→                {trend && trendValue && (
   133→                  <span className={`flex items-center gap-0.5 text-[10px] font-semibold ${trendColor}`}>
   134→                    {trendIcon} {trendValue}
   135→                  </span>
   136→                )}

   138→              <p className="text-[11px] text-muted-foreground truncate">{subtitle}</p>
   139→            </div>
   140→            <div className={`relative flex items-center justify-center h-9 w-9 rounded-lg ${iconBg} shrink-0 ml-2`}>
   141→              <span className={iconColor}>{icon}</span>
   142→              {pulse && (
   143→                <span className="absolute -top-0.5 -right-0.5 h-2.5 w-2.5 rounded-full bg-emerald-500 animate-pulse" />
   144→              )}
   145→            </div>
   146→          </div>
   147→        </CardContent>
   148→      </Card>
   149→    </motion.div>

   157→function activityIcon(type: string): { icon: React.ReactNode; color: string; bg: string } {
   158→  if (type.includes('filed')) return { icon: <CheckCircle2 className="h-3.5 w-3.5" />, color: 'text-emerald-600', bg: 'bg-emerald-50' };
   159→  if (type.includes('upload') || type.includes('document')) return { icon: <Upload className="h-3.5 w-3.5" />, color: 'text-blue-600', bg: 'bg-blue-50' };
   160→  if (type.includes('review')) return { icon: <ClipboardCheck className="h-3.5 w-3.5" />, color: 'text-violet-600', bg: 'bg-violet-50' };
   161→  if (type.includes('client')) return { icon: <Users className="h-3.5 w-3.5" />, color: 'text-sky-600', bg: 'bg-sky-50' };
   162→  if (type.includes('reconcil') || type.includes('mismatch')) return { icon: <AlertTriangle className="h-3.5 w-3.5" />, color: 'text-amber-600', bg: 'bg-amber-50' };
   163→  if (type.includes('invoice') || type.includes('extract')) return { icon: <FileText className="h-3.5 w-3.5" />, color: 'text-teal-600', bg: 'bg-teal-50' };
   164→  return { icon: <Activity className="h-3.5 w-3.5" />, color: 'text-slate-500', bg: 'bg-slate-50' };
   165→}

   269→    const todayStart = new Date(today);
   270→    todayStart.setHours(0, 0, 0, 0);

   272→      const d = new Date(inv.createdAt as string || 0);
   273→      return d >= todayStart;

   279→      if (r.status === 'filed') return false;
   280→      const due = getFilingDueDate(r.returnType, r.period);
