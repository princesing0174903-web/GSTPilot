     3→     3→// GSTPilot — Real-time Firestore Hooks
     4→     4→// React hooks with onSnapshot listeners for live data + optimistic UI.
     5→     5→//
     6→     6→// ARCHITECTURE (Production-grade multi-tenant):
     7→     7→//   • Every query is scoped by `organizationId` from OrgContext — NEVER by the
     8→     8→//     legacy `firmId`. This matches the Firestore security rules, which require
     9→     9→//     every tenant-scoped document to carry `organizationId` and only allow
    10→    10→//     reads where `isOrgMember(resource.data.organizationId)` is true.
    11→    11→//   • If there is no organization (preview / demo mode, or not yet loaded),
    12→    12→//     hooks return empty data WITHOUT subscribing — so the UI renders premium
    13→    13→//     empty states instead of throwing permission errors.
    14→    14→//   • Permission errors (permission-denied / unauthenticated) are treated as
    15→    15→//     "no data available" — the hook clears its error and returns an empty
    16→    16→//     array. This is the graceful-degradation contract: the app NEVER shows a
    17→    17→//     "Missing or insufficient permissions" wall. Genuine errors (network,
    18→    18→//     index missing) still surface as `error`.

    47→    47→// ─── Permission-error detection (graceful degradation) ───────────────────────
    48→    48→
    49→    49→/**
    50→    50→ * True if the given Firestore error is a permission / auth failure that we
    51→    51→ * should treat as "no data available" rather than a hard error. This is the
    52→    52→ * core of the never-show-a-permission-wall contract.
    53→    53→ */
    54→    54→function isPermissionError(err: unknown): boolean {
    55→    55→  if (!err) return false;
    56→    56→  const e = err as { code?: string; message?: string };
    57→    57→  const code = (e.code || '').toLowerCase();
    58→    58→  const message = (e.message || '').toLowerCase();
    59→    59→  if (
    60→    60→    code === 'permission-denied' ||
    61→    61→    code === 'unauthenticated' ||
    62→    62→    code === 'auth/operation-not-allowed' ||
    63→    63→    code === 'auth/user-not-found'
    64→    64→  ) {
    65→    65→    return true;
    66→    66→  }
    67→    67→  if (
    68→    68→    message.includes('missing or insufficient permissions') ||
    69→    69→    message.includes('permission-denied') ||
    70→    70→    message.includes('insufficient permissions') ||
    71→    71→    message.includes('not authorized') ||
    72→    72→    message.includes('unauthenticated')
    73→    73→  ) {
    74→    74→    return true;
    75→    75→  }
    76→    76→  return false;
    77→    77→}

    79→    79→// ─── Timestamp Converter ─────────────────────────────────────────────────────
    80→    80→
    81→    81→function convertDoc<T extends Record<string, unknown>>(snapData: Record<string, unknown>): T {
    82→    82→  const converted: Record<string, unknown> = {};
    83→    83→  for (const [key, value] of Object.entries(snapData)) {
    84→    84→    if (value && typeof value === 'object' && 'toDate' in value && typeof (value as { toDate: () => Date }).toDate === 'function') {
    85→    85→      converted[key] = (value as { toDate: () => Date }).toDate().toISOString();
    86→    86→    } else if (value && typeof value === 'object' && !Array.isArray(value) && !(value instanceof Date)) {
    87→    87→      // Recursively convert nested objects
    88→    88→      converted[key] = convertDoc(value as Record<string, unknown>);
    89→    89→    } else {
    90→    90→      converted[key] = value;
    91→    91→    }
    92→    92→  }
    93→    93→  return converted as T;
    94→    94→}

    96→    96→// ─── Generic Collection Hook ─────────────────────────────────────────────────
    97→    97→//
    98→    98→// The hook resolves the current `organizationId` from OrgContext. If the org
    99→    99→// isn't resolved yet (loading) OR we're in preview mode (no org), the hook
   100→   100→// returns empty data without subscribing. This guarantees no permission errors
   101→   101→// ever reach the UI in preview/offline mode.
   102→   102→
   103→   103→function useFirestoreCollection<T>(
   104→   104→  collectionName: CollectionName,
   105→   105→  constraints: QueryConstraint[] = [],
   106→   106→  deps: unknown[] = [],
   107→   107→): { data: Array<T & { id: string }>; loading: boolean; error: string | null } {
   108→   108→  const { user } = useAuth();
   109→   109→  const { organization, isPreviewMode, loading: orgLoading } = useOrg();
   110→   110→  const organizationId = organization?.id ?? null;
   111→   111→
   112→   112→  const [data, setData] = useState<Array<T & { id: string }>>([]);
   113→   113→  const [loading, setLoading] = useState(true);
   114→   114→  const [error, setError] = useState<string | null>(null);
   115→   115→
   116→   116→  useEffect(() => {
   117→   117→    // No user → nothing to subscribe to (empty data).
   118→   118→    if (!user) {
   119→   119→      setData([]);
   120→   120→      setLoading(false);
   121→   121→      setError(null);
   122→   122→      return;
   123→   123→    }
   124→   124→
   125→   125→    // Org still resolving — stay in loading state, but don't subscribe yet.
   126→   126→    if (orgLoading && !organizationId) {
   127→   127→      return;
   128→   128→    }
   129→   129→
   130→   130→    // Preview mode (no real org) OR no organization at all → empty data,
   131→   131→    // no subscription. This is the preview-mode contract.
   132→   132→    if (!organizationId || isPreviewMode || organizationId === 'preview-org') {
   133→   133→      setData([]);
   134→   134→      setLoading(false);
   135→   135→      setError(null);
   136→   136→      return;
   137→   137→    }
   138→   138→
   139→   139→    let unsub: Unsubscribe | null = null;
   140→   140→
   141→   141→    // Scope EVERY query by organizationId. This is required by the Firestore
   142→   142→    // security rules — unbounded collection reads are denied.
   143→   143→    const baseConstraints: QueryConstraint[] = [
   144→   144→      where('organizationId', '==', organizationId),
   145→   145→      ...constraints,
   146→   146→    ];
   147→   147→
   148→   148→    const q = query(collection(db, collectionName), ...baseConstraints);
   149→   149→
   150→   150→    unsub = onSnapshot(q, (snap) => {
   151→   151→      const items = snap.docs.map(d => ({
   152→   152→        id: d.id,
   153→   153→        ...convertDoc<T>(d.data() as Record<string, unknown>),
   154→   154→      }));
   155→   155→      setData(items);
   156→   156→      setLoading(false);
   157→   157→      setError(null);
   158→   158→    }, (err) => {
   159→   159→      console.warn(`[Firestore] ${collectionName} subscription error:`, err);
   160→   160→      // Permission errors → graceful degradation (empty data, no error wall).
   161→   161→      if (isPermissionError(err)) {
   162→   162→        setData([]);
   163→   163→        setError(null);
   164→   164→        setLoading(false);
   165→   165→        return;
   166→   166→      }
   167→   167→      // Genuine errors (network, missing index) still surface.
   168→   168→      setError(err.message);
   169→   169→      setLoading(false);
   170→   170→    });
   171→   171→
   172→   172→    return () => {
   173→   173→      if (unsub) unsub();
   174→   174→    };
   175→   175→  }, [user, organizationId, isPreviewMode, orgLoading, ...deps]);
   176→   176→
   177→   177→  // When user is null, return empty data (not loading)
   178→   178→  if (!user && loading) {
   179→   179→    return { data: [], loading: false, error: null };
   180→   180→  }
   181→   181→
   182→   182→  return { data, loading, error };
   183→   183→}

   185→   185→// ─── Generic Document Hook ───────────────────────────────────────────────────
   186→   186→
   187→   187→function useFirestoreDoc<T>(
   188→   188→  collectionName: CollectionName,
   189→   189→  docId: string | null,
   190→   190→): { data: (T & { id: string }) | null; loading: boolean; error: string | null } {
   191→   191→  const { user } = useAuth();
   192→   192→  const { organization, isPreviewMode } = useOrg();
   193→   193→  const organizationId = organization?.id ?? null;
   194→   194→
   195→   195→  const [data, setData] = useState<(T & { id: string }) | null>(null);
   196→   196→  const [loading, setLoading] = useState(!!docId);
   197→   197→  const [error, setError] = useState<string | null>(null);
   198→   198→
   199→   199→  useEffect(() => {
   200→   200→    if (!docId) {
   201→   201→      setData(null);
   202→   202→      setLoading(false);
   203→   203→      return;
   204→   204→    }
   205→   205→
   206→   206→    // Preview mode (no real org) → no doc, no subscription.
   207→   207→    if (!user || !organizationId || isPreviewMode || organizationId === 'preview-org') {
   208→   208→      setData(null);
   209→   209→      setLoading(false);
   210→   210→      setError(null);
   211→   211→      return;
   212→   212→    }
   213→   213→
   214→   214→    const unsub = onSnapshot(doc(db, collectionName, docId), (snap) => {
   215→   215→      if (!snap.exists()) {
   216→   216→        setData(null);
   217→   217→        setLoading(false);
   218→   218→        return;
   219→   219→      }
   220→   220→      setData({
   221→   221→        id: snap.id,
   222→   222→        ...convertDoc<T>(snap.data() as Record<string, unknown>),
   223→   223→      });
   224→   224→      setLoading(false);
   225→   225→      setError(null);
   226→   226→    }, (err) => {
   227→   227→      console.warn(`[Firestore] ${collectionName}/${docId} doc error:`, err);
   228→   228→      if (isPermissionError(err)) {
   229→   229→        setData(null);
   230→   230→        setError(null);
   231→   231→        setLoading(false);
   232→   232→        return;
   233→   233→      }
   234→   234→      setError(err.message);
   235→   235→      setLoading(false);
   236→   236→    });
   237→   237→
   238→   238→    return () => {
   239→   239→      if (unsub) unsub();
   240→   240→    };
   241→   241→  }, [collectionName, docId, user, organizationId, isPreviewMode]);
   242→   242→
   243→   243→  return { data, loading, error };
   244→   244→}

   503→   503→// ─── Executive Scores (computed from live data) ─────────────────────────────
   504→   504→
   505→   505→export function useFirmExecutiveScores(): { scores: FirmExecutiveScores; loading: boolean } {
   506→   506→  const { data: clients } = useFireClients();
   507→   507→  const { data: returns } = useFireReturns();
   508→   508→  const { data: invoices } = useFireInvoices();
   509→   509→  const { data: reconciliations } = useFireReconciliations();
   510→   510→  const { data: activities } = useFireRecentActivities(50);
   511→   511→
   512→   512→  const loading = clients.length === 0 && returns.length === 0;
   513→   513→
   514→   514→  const scores = useMemo<FirmExecutiveScores>(() => {
   515→   515→    if (loading) {
   516→   516→      return { firmHealth: 0, revenue: 0, compliance: 0, teamEfficiency: 0, clientSatisfaction: 0, cashFlow: 0 };
   517→   517→    }
   518→   518→
   519→   519→    // Firm Health: weighted avg of client health scores
   520→   520→    const activeClients = clients.filter(c => c.status === 'active');
   521→   521→    const avgHealth = activeClients.length > 0
   522→   522→      ? activeClients.reduce((s, c) => s + (c.healthScore || 0), 0) / activeClients.length
   523→   523→      : 50;
   524→   524→
   525→   525→    // Revenue Score: based on tax volume
   526→   526→    const totalTax = invoices.reduce((s, i) => s + (i.totalTax || 0) + (i.cgst || 0) + (i.sgst || 0) + (i.igst || 0), 0);
   527→   527→    const revenueScore = Math.min(100, Math.round((totalTax / 1000000) * 100));
   528→   528→
   529→   529→    // Compliance Score: filed vs total returns
   530→   530→    const filedReturns = returns.filter(r => r.status === 'filed').length;
   531→   531→    const complianceScore = returns.length > 0 ? Math.round((filedReturns / returns.length) * 100) : 50;
   532→   532→
   533→   533→    // Team Efficiency: based on completed activities
   534→   534→    const completedActivities = activities.filter(a => a.type.includes('filed') || a.type.includes('completed') || a.type.includes('resolved')).length;
   535→   535→    const teamEfficiency = activities.length > 0 ? Math.min(100, Math.round((completedActivities / Math.max(activities.length, 1)) * 100)) : 50;
   536→   536→
   537→   537→    // Client Satisfaction: inverse of overdue + health
   538→   538→    const overdueClients = activeClients.filter(c => (c.pendingReturnCount || 0) > 0).length;
   539→   539→    const clientSatisfaction = activeClients.length > 0 ? Math.round(((activeClients.length - overdueClients) / activeClients.length) * 100) : 50;
   540→   540→
   541→   541→    // Cash Flow: based on reconciliation match rate
   542→   542→    const matchedRecons = reconciliations.filter(r => r.matched > 0);
   543→   543→    const totalMatched = matchedRecons.reduce((s, r) => s + r.matched, 0);
   544→   544→    const totalRecords = matchedRecons.reduce((s, r) => s + r.totalRecords, 0);
   545→   545→    const cashFlow = totalRecords > 0 ? Math.round((totalMatched / totalRecords) * 100) : 50;
   546→   546→
   547→   547→    return {
   548→   548→      firmHealth: Math.round(avgHealth) || 50,
   549→   549→      revenue: revenueScore || 50,
   550→   550→      compliance: complianceScore || 50,
   551→   551→      teamEfficiency: teamEfficiency || 50,
   552→   552→      clientSatisfaction: clientSatisfaction || 50,
   553→   553→      cashFlow: cashFlow || 50,
   554→   554→    };
   555→   555→  }, [clients, returns, invoices, reconciliations, activities, loading]);
   556→   556→
   557→   557→  return { scores, loading };
   558→   558→}
