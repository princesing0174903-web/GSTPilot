   289→// ═══════════════════════════════════════════════════════════════════════════════
   290→// CONNECT MODAL — handles each connector type's connection flow
   291→// ═══════════════════════════════════════════════════════════════════════════════
   292→
   293→function ConnectModal({
   294→  type, userId, onClose, onConnected,
   295→}: {
   296→  type: ConnectorType;
   297→  userId?: string;
   298→  onClose: () => void;
   299→  onConnected: () => void;
   300→}) {
   301→  const def = CONNECTOR_DEFINITIONS.find(d => d.type === type)!;
   302→  const [connecting, setConnecting] = useState(false);
   303→  const [error, setError] = useState<string | null>(null);
   304→
   305→  // GSTN state
   306→  const [gstin, setGstin] = useState('');
   307→
   308→  // Bank state
   309→  const [bankCode, setBankCode] = useState('');
   310→  const [accountNumber, setAccountNumber] = useState('');
   311→  const [ifsc, setIfsc] = useState('');
   312→
   313→  // WhatsApp state
   314→  const [phone, setPhone] = useState('');
   315→
   316→  // Accounting state
   317→  const [companyName, setCompanyName] = useState('');
   318→  const [companyGstin, setCompanyGstin] = useState('');
   319→
   320→  const handleConnect = async () => {
   321→    if (!userId) {
   322→      setError('User not authenticated');
   323→      return;
   324→    }
   325→    setConnecting(true);
   326→    setError(null);
   327→
   328→    try {
   329→      let endpoint = '';
   330→      let body: Record<string, unknown> = { userId };
   331→
   332→      if (type === 'gstn') {
   333→        if (!gstin.trim()) { setError('GSTIN is required'); setConnecting(false); return; }
   334→        endpoint = '/api/connect/gstn';
   335→        body.gstin = gstin.trim();
   336→      } else if (type === 'bank') {
   337→        if (!bankCode) { setError('Select a bank'); setConnecting(false); return; }
   338→        if (!accountNumber.trim()) { setError('Account number is required'); setConnecting(false); return; }
   339→        endpoint = '/api/connect/bank';
   340→        body.bankCode = bankCode;
   341→        body.accountNumber = accountNumber.trim();
   342→        body.ifsc = ifsc.trim();
   343→      } else if (type === 'gmail') {
   344→        // Gmail uses Google Identity Services for OAuth
   345→        const accessToken = await getGmailAccessToken();
   346→        if (!accessToken) {
   347→          setError('Gmail authorization failed or was cancelled');
   348→          setConnecting(false);
   349→          return;
   350→        }
   351→        endpoint = '/api/connect/gmail';
   352→        body.accessToken = accessToken;
   353→      } else if (type === 'whatsapp') {
   354→        if (!phone.trim()) { setError('Phone number is required'); setConnecting(false); return; }
   355→        endpoint = '/api/connect/whatsapp';
   356→        body.phoneNumber = phone.trim();
   357→      } else if (type === 'tally' || type === 'zoho' || type === 'quickbooks') {
   358→        if (!companyName.trim()) { setError('Company name is required'); setConnecting(false); return; }
   359→        endpoint = '/api/connect/accounting';
   360→        body.software = type;
   361→        body.companyName = companyName.trim();
   362→        body.companyGstin = companyGstin.trim();
   363→      }
   364→
   365→      const res = await fetch(endpoint, {
   366→        method: 'POST',
   367→        headers: { 'Content-Type': 'application/json' },
   368→        body: JSON.stringify(body),
   369→      });
   370→      const data = await res.json();
   371→
   372→      if (!res.ok) {
   373→        setError(data.error || data.details?.join('; ') || 'Connection failed');
   374→        setConnecting(false);
   375→        return;
   376→      }
   377→
   378→      onConnected();
   379→    } catch (err) {
   380→      setError(err instanceof Error ? err.message : 'Connection failed');
   381→      setConnecting(false);
   382→    }
   383→  };
   384→
   385→  return (
   386→    <motion.div
   387→      initial={{ opacity: 0 }}
   388→      animate={{ opacity: 1 }}
   389→      exit={{ opacity: 0 }}
   390→      className="fixed inset-0 z-50 flex items-center justify-center bg-black/60 backdrop-blur-sm p-4"
   391→      onClick={onClose}
   392→    >
   393→      <motion.div
   394→        initial={{ scale: 0.95, opacity: 0 }}
   395→        animate={{ scale: 1, opacity: 1 }}
   396→        exit={{ scale: 0.95, opacity: 0 }}
   397→        transition={{ duration: 0.2, ease: EASE }}
   398→        onClick={e => e.stopPropagation()}
   399→        className="w-full max-w-md overflow-hidden rounded-2xl border border-white/[0.1] bg-[#0a0a0f] p-6 backdrop-blur-xl"
   400→      >
   401→        <h2 className="text-lg font-semibold text-white">{def.connectLabel}</h2>
   402→        <p className="mt-1 text-xs text-white/50">{def.description}</p>
   403→
   404→        <div className="mt-5 space-y-4">
   405→          {type === 'gstn' && (
   406→            <div>
   407→              <label className="mb-1.5 block text-xs font-medium text-white/70">GSTIN</label>
   408→              <input
   409→                value={gstin}
   410→                onChange={e => setGstin(e.target.value)}
   411→                placeholder="e.g. 27ABCDE1234F1Z5"
   412→                className="w-full rounded-lg border border-white/[0.08] bg-white/[0.03] px-3 py-2.5 text-sm text-white placeholder:text-white/30 focus:border-blue-500/40 focus:outline-none"
   413→              />
   414→              <p className="mt-1 text-[10px] text-white/35">
   415→                15-character GSTIN. Validated with real checksum algorithm (state code + PAN + entity + Z + checksum).
   416→              </p>
   417→            </div>
   418→          )}
   419→
   420→          {type === 'bank' && (
   421→            <>
   422→              <div>
   423→                <label className="mb-1.5 block text-xs font-medium text-white/70">Bank</label>
   424→                <select
   425→                  value={bankCode}
   426→                  onChange={e => setBankCode(e.target.value)}
   427→                  className="w-full rounded-lg border border-white/[0.08] bg-white/[0.03] px-3 py-2.5 text-sm text-white focus:border-blue-500/40 focus:outline-none"
   428→                >
   429→                  <option value="">Select your bank</option>
   430→                  {SUPPORTED_BANKS.map(b => (
   431→                    <option key={b.code} value={b.code} className="bg-[#0a0a0f]">{b.name}</option>
   432→                  ))}
   433→                </select>
   434→              </div>
   435→              <div>
   436→                <label className="mb-1.5 block text-xs font-medium text-white/70">Account Number</label>
   437→                <input
   438→                  value={accountNumber}
   439→                  onChange={e => setAccountNumber(e.target.value)}
   440→                  placeholder="e.g. 50100123456789"
   441→                  className="w-full rounded-lg border border-white/[0.08] bg-white/[0.03] px-3 py-2.5 text-sm text-white placeholder:text-white/30 focus:border-blue-500/40 focus:outline-none"
   442→                />
   443→              </div>
   444→              <div>
   445→                <label className="mb-1.5 block text-xs font-medium text-white/70">IFSC (optional)</label>
   446→                <input
   447→                  value={ifsc}
   448→                  onChange={e => setIfsc(e.target.value)}
   449→                  placeholder="e.g. HDFC0001234"
   450→                  className="w-full rounded-lg border border-white/[0.08] bg-white/[0.03] px-3 py-2.5 text-sm text-white placeholder:text-white/30 focus:border-blue-500/40 focus:outline-none"
   451→                />
   452→              </div>
   453→            </>
   454→          )}
   455→
   456→          {type === 'gmail' && (
   457→            <div className="rounded-lg border border-blue-500/20 bg-blue-500/[0.05] p-4">
   458→              <p className="text-sm text-white/70">
   459→                Click <strong className="text-white">Connect Gmail</strong> to open Google's secure OAuth consent screen.
   460→                GSTPilot will read GST-related emails (notices, invoices, tax communications) with{' '}
   461→                <code className="rounded bg-white/10 px-1 text-[11px]">gmail.readonly</code> scope.
   462→              </p>
   463→              <p className="mt-2 text-[11px] text-white/40">
   464→                Your access token is used once for sync, then discarded. Only parsed email data is stored.
   465→              </p>
   466→            </div>
   467→          )}
   468→
   469→          {type === 'whatsapp' && (
   470→            <div>
   471→              <label className="mb-1.5 block text-xs font-medium text-white/70">WhatsApp Business Phone Number</label>
   472→              <input
   473→                value={phone}
   474→                onChange={e => setPhone(e.target.value)}
   475→                placeholder="e.g. +91 98765 43210"
   476→                className="w-full rounded-lg border border-white/[0.08] bg-white/[0.03] px-3 py-2.5 text-sm text-white placeholder:text-white/30 focus:border-blue-500/40 focus:outline-none"
   477→              />
   478→              <p className="mt-1 text-[10px] text-white/35">
   479→                Indian mobile number. Validated for 10-digit format starting with 6-9.
   480→              </p>
   481→            </div>
   482→          )}
   483→
   484→          {(type === 'tally' || type === 'zoho' || type === 'quickbooks') && (
   485→            <>
   486→              <div>
   487→                <label className="mb-1.5 block text-xs font-medium text-white/70">Company Name</label>
   488→                <input
   489→                  value={companyName}
   490→                  onChange={e => setCompanyName(e.target.value)}
   491→                  placeholder="e.g. Acme Industries Pvt Ltd"
   492→                  className="w-full rounded-lg border border-white/[0.08] bg-white/[0.03] px-3 py-2.5 text-sm text-white placeholder:text-white/30 focus:border-blue-500/40 focus:outline-none"
   493→                />
   494→              </div>
   495→              <div>
   496→                <label className="mb-1.5 block text-xs font-medium text-white/70">Company GSTIN (optional)</label>
   497→                <input
   498→                  value={companyGstin}
   499→                  onChange={e => setCompanyGstin(e.target.value)}
   500→                  placeholder="e.g. 27ABCDE1234F1Z5"
   501→                  className="w-full rounded-lg border border-white/[0.08] bg-white/[0.03] px-3 py-2.5 text-sm text-white placeholder:text-white/30 focus:border-blue-500/40 focus:outline-none"
   502→                />
   503→              </div>
   504→            </>

   625→function loadGoogleIdentityServices(): Promise<void> {
   626→  return new Promise((resolve) => {
   627→    if (typeof window === 'undefined') return resolve();
   628→    if ((window as unknown as { google?: unknown }).google) return resolve();
   629→    const script = document.createElement('script');
   630→    script.src = 'https://accounts.google.com/gsi/client';
   631→    script.async = true;
   632→    script.defer = true;
   633→    script.onload = () => resolve();
   634→    document.head.appendChild(script);
   635→  });
   636→}

   639→  try {
   640→    await loadGoogleIdentityServices();
   641→    const google = (window as unknown as { google?: { accounts?: { oauth2?: { initTokenClient: (config: unknown) => { requestAccessToken: (c: unknown) => void } } } } }).google;
   642→    if (!google?.accounts?.oauth2) return null;
   643→
   644→    // Use the Firebase project's OAuth client ID (public, safe for frontend)

   646→
   647→    return new Promise<string | null>((resolve) => {
   648→      const client = google.accounts.oauth2.initTokenClient({

   665→function timeAgo(iso: string): string {
   666→  const date = new Date(iso);
   667→  const seconds = Math.floor((Date.now() - date.getTime()) / 1000);
   668→  if (seconds < 60) return 'just now';
   669→  const minutes = Math.floor(seconds / 60);
   670→  if (minutes < 60) return `${minutes}m ago`;
   671→  const hours = Math.floor(minutes / 60);
   672→  if (hours < 24) return `${hours}h ago`;
   673→  const days = Math.floor(hours / 24);
   674→  if (days < 7) return `${days}d ago`;
   675→  return date.toLocaleDateString('en-IN');
   676→}
