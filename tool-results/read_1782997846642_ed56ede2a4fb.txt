    33→function BalanceTrendChart({ data }: { data: { day: string; balance: number }[] }) {
    34→  const max = Math.max(...data.map(d => d.balance))
    35→  const min = Math.min(...data.map(d => d.balance))
    36→  const range = max - min || 1
    37→  const w = 280
    38→  const h = 60
    39→  const pts = data.map((d, i) => `${(i / (data.length - 1)) * w},${h - ((d.balance - min) / range) * (h - 8) - 4}`)
    40→  return (
    41→    <svg width={w} height={h + 16} className="overflow-visible">
    42→      <defs>
    43→        <linearGradient id="balGrad" x1="0" y1="0" x2="0" y2="1">
    44→          <stop offset="0%" stopColor="#10b981" stopOpacity="0.2" />
    45→          <stop offset="100%" stopColor="#10b981" stopOpacity="0.02" />
    46→        </linearGradient>
    47→      </defs>
    48→      <polygon points={`0,${h} ${pts.join(' ')} ${w},${h}`} fill="url(#balGrad)" />
    49→      <polyline points={pts.join(' ')} fill="none" stroke="#10b981" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" />
    50→      {data.filter((_, i) => i % 2 === 0).map((d, i) => {
    51→        const x = (data.indexOf(d) / (data.length - 1)) * w
    52→        return <text key={i} x={x} y={h + 12} textAnchor="middle" className="text-[8px] fill-muted-foreground">{d.day}</text>
    53→      })}
    54→    </svg>
    55→  )
    56→}

    58→function ReconcileDonut({ reconciled, unreconciled }: { reconciled: number; unreconciled: number }) {
    59→  const total = reconciled + unreconciled
    60→  const r = 36
    61→  const c = 2 * Math.PI * r
    62→  const reconcilePct = (reconciled / total) * 100
    63→  const dashOffset = c - (reconcilePct / 100) * c
    64→  return (
    65→    <svg width="90" height="90" className="overflow-visible">
    66→      <circle cx="45" cy="45" r={r} fill="none" stroke="currentColor" strokeWidth="8" className="text-slate-100 dark:text-slate-800" />
    67→      <circle cx="45" cy="45" r={r} fill="none" stroke="#10b981" strokeWidth="8" strokeLinecap="round" strokeDasharray={c} strokeDashoffset={dashOffset} transform="rotate(-90 45 45)" />
    68→      <text x="45" y="42" textAnchor="middle" className="text-sm font-bold fill-emerald-600 dark:fill-emerald-400">{reconcilePct.toFixed(0)}%</text>
    69→      <text x="45" y="54" textAnchor="middle" className="text-[7px] fill-muted-foreground">Reconciled</text>
    70→    </svg>
    71→  )
    72→}
