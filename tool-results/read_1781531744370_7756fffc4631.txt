    65→    65→function getDaysRemaining(dueDateStr: string): number {
    66→    66→  const dueDate = new Date(dueDateStr);
    67→    67→  const today = new Date();
    68→    68→  today.setHours(0, 0, 0, 0);
    69→    69→  dueDate.setHours(0, 0, 0, 0);
    70→    70→  return Math.ceil((dueDate.getTime() - today.getTime()) / (1000 * 60 * 60 * 24));
    71→    71→}
    72→    72→
    73→    73→function formatDaysRemaining(days: number): string {
    74→    74→  if (days < 0) return `${Math.abs(days)}d overdue`;
    75→    75→  if (days === 0) return 'Due today';
    76→    76→  if (days === 1) return 'Tomorrow';
    77→    77→  return `${days}d left`;
    78→    78→}
    79→    79→
    80→    80→function timeAgo(dateStr: string): string {
    81→    81→  if (!dateStr) return '';
    82→    82→  const now = new Date();
    83→    83→  const date = new Date(dateStr);
    84→    84→  const seconds = Math.floor((now.getTime() - date.getTime()) / 1000);
    85→    85→  if (seconds < 60) return 'just now';
    86→    86→  const minutes = Math.floor(seconds / 60);
    87→    87→  if (minutes < 60) return `${minutes}m ago`;
    88→    88→  const hours = Math.floor(minutes / 60);
    89→    89→  if (hours < 24) return `${hours}h ago`;
    90→    90→  const days = Math.floor(hours / 24);
    91→    91→  if (days < 7) return `${days}d ago`;
    92→    92→  return date.toLocaleDateString('en-IN', { day: 'numeric', month: 'short' });
    93→    93→}

   286→   286→    // Clients At Risk — health score < 60
