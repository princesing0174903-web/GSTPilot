    69→function getDaysRemaining(dueDateStr: string): number {
    70→  const dueDate = new Date(dueDateStr);
    71→  const today = new Date();
    72→  today.setHours(0, 0, 0, 0);
    73→  dueDate.setHours(0, 0, 0, 0);
    74→  return Math.ceil((dueDate.getTime() - today.getTime()) / (1000 * 60 * 60 * 24));
    75→}
    76→
    77→function formatDaysRemaining(days: number): string {
    78→  if (days < 0) return `${Math.abs(days)}d overdue`;
    79→  if (days === 0) return 'Due today';
    80→  if (days === 1) return 'Tomorrow';
    81→  return `${days}d left`;
    82→}
    83→
    84→function timeAgo(dateStr: string): string {
    85→  if (!dateStr) return '';
    86→  const now = new Date();
    87→  const date = new Date(dateStr);
    88→  const seconds = Math.floor((now.getTime() - date.getTime()) / 1000);
    89→  if (seconds < 60) return 'just now';
    90→  const minutes = Math.floor(seconds / 60);
    91→  if (minutes < 60) return `${minutes}m ago`;
    92→  const hours = Math.floor(minutes / 60);
    93→  if (hours < 24) return `${hours}h ago`;
    94→  const days = Math.floor(hours / 24);
    95→  if (days < 7) return `${days}d ago`;
    96→  return date.toLocaleDateString('en-IN', { day: 'numeric', month: 'short' });
    97→}

   103→interface SmartCardProps {
   104→  label: string;
   105→  value: string | number;
   106→  icon: React.ReactNode;
   107→  iconColor: string;
   108→  iconBg: string;
   109→  subtitle: string;
   110→  trend?: 'up' | 'down' | 'neutral';
   111→  trendValue?: string;
   112→  pulse?: boolean;
   113→  index: number;
   114→  onClick?: () => void;
   115→}
   116→
   117→function SmartCard({ label, value, icon, iconColor, iconBg, subtitle, trend, trendValue, pulse, index, onClick }: SmartCardProps) {
   118→  const trendIcon = trend === 'up' ? <TrendingUp className="h-3 w-3" /> : trend === 'down' ? <TrendingDown className="h-3 w-3" /> : <Minus className="h-3 w-3" />;
   119→  const trendColor = trend === 'up' ? 'text-emerald-600' : trend === 'down' ? 'text-red-500' : 'text-muted-foreground';
   120→
   121→  return (
   122→    <motion.div
   123→      initial={{ opacity: 0, y: 16 }}
   124→      animate={{ opacity: 1, y: 0 }}
   125→      transition={{ duration: 0.35, delay: index * 0.05 }}
   126→    >
   127→      <Card className={`hover:shadow-md transition-all duration-200 ${onClick ? 'cursor-pointer' : ''}`} onClick={onClick}>
   128→        <CardContent className="p-4">
   129→          <div className="flex items-start justify-between">
   130→            <div className="space-y-1 min-w-0 flex-1">
   131→              <p className="text-[10px] font-semibold text-muted-foreground uppercase tracking-widest truncate">
   132→                {label}
   133→              </p>
   134→              <div className="flex items-baseline gap-2">
   135→                <p className="text-2xl font-bold text-foreground tracking-tight">{value}</p>
   136→                {trend && trendValue && (
   137→                  <span className={`flex items-center gap-0.5 text-[10px] font-semibold ${trendColor}`}>
   138→                    {trendIcon} {trendValue}
   139→                  </span>
   140→                )}
   141→              </div>
   142→              <p className="text-[11px] text-muted-foreground truncate">{subtitle}</p>
   143→            </div>
   144→            <div className={`relative flex items-center justify-center h-9 w-9 rounded-lg ${iconBg} shrink-0 ml-2`}>
   145→              <span className={iconColor}>{icon}</span>
   146→              {pulse && (
   147→                <span className="absolute -top-0.5 -right-0.5 h-2.5 w-2.5 rounded-full bg-emerald-500 animate-pulse" />
   148→              )}
   149→            </div>
   150→          </div>
   151→        </CardContent>
   152→      </Card>
   153→    </motion.div>
   154→  );
   155→}

   161→function activityIcon(type: string): { icon: React.ReactNode; color: string; bg: string } {
   162→  if (type.includes('filed')) return { icon: <CheckCircle2 className="h-3.5 w-3.5" />, color: 'text-emerald-600', bg: 'bg-emerald-50' };
   163→  if (type.includes('upload') || type.includes('document')) return { icon: <Upload className="h-3.5 w-3.5" />, color: 'text-blue-600', bg: 'bg-blue-50' };
   164→  if (type.includes('review')) return { icon: <ClipboardCheck className="h-3.5 w-3.5" />, color: 'text-violet-600', bg: 'bg-violet-50' };
   165→  if (type.includes('client')) return { icon: <Users className="h-3.5 w-3.5" />, color: 'text-sky-600', bg: 'bg-sky-50' };
   166→  if (type.includes('reconcil') || type.includes('mismatch')) return { icon: <AlertTriangle className="h-3.5 w-3.5" />, color: 'text-amber-600', bg: 'bg-amber-50' };
   167→  if (type.includes('invoice') || type.includes('extract')) return { icon: <FileText className="h-3.5 w-3.5" />, color: 'text-teal-600', bg: 'bg-teal-50' };
   168→  return { icon: <Activity className="h-3.5 w-3.5" />, color: 'text-slate-500', bg: 'bg-slate-50' };
   169→}
