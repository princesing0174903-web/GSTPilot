   243→// ─── Analysis Renderer (paragraphs + bullet sub-points) ──────────────────────
   244→
   245→function AnalysisBody({ text, isStreaming }: { text: string; isStreaming: boolean }) {
   246→  // Split into paragraphs by \n\n, then within each paragraph detect bullet lines.
   247→  const paragraphs = text.split('\n\n').filter((p) => p.trim().length > 0);
   248→
   249→  return (
   250→    <div
   251→      className={cn(
   252→        'space-y-3 text-sm leading-relaxed text-foreground/90',
   253→        isStreaming && 'typing-cursor',
   254→      )}
   255→    >
   256→      {paragraphs.map((para, pi) => {
   257→        const lines = para.split('\n');
   258→        const bulletLines: string[] = [];
   259→        const proseLines: string[] = [];
   260→        let currentMode: 'prose' | 'bullet' | null = null;
   261→
   262→        for (const line of lines) {
   263→          const trimmed = line.trim();
   264→          const isBullet =
   265→            trimmed.startsWith('• ') ||
   266→            trimmed.startsWith('- ') ||
   267→            trimmed.startsWith('* ');
   268→          if (isBullet) {
   269→            if (currentMode === 'prose' && proseLines.length > 0) {
   270→              // Flush prose before bullets.
   271→            }
   272→            bulletLines.push(trimmed.replace(/^[•\-*]\s+/, ''));
   273→            currentMode = 'bullet';
   274→          } else if (trimmed.length > 0) {
   275→            proseLines.push(line);
   276→            currentMode = 'prose';
   277→          }
   278→        }
   279→
   280→        return (
   281→          <div key={pi} className="space-y-1.5">
   282→            {proseLines.length > 0 && (
   283→              <p>{proseLines.join(' ')}</p>
   284→            )}
   285→            {bulletLines.length > 0 && (
   286→              <ul className="ml-1 space-y-1">
   287→                {bulletLines.map((b, bi) => (
   288→                  <motion.li
   289→                    key={bi}
   290→                    initial={{ opacity: 0, x: -4 }}
   291→                    animate={{ opacity: 1, x: 0 }}
   292→                    transition={{ duration: 0.25, delay: bi * 0.04 }}
   293→                    className="flex items-start gap-2 text-foreground/85"
   294→                  >
   295→                    <span className="mt-1.5 h-1 w-1 shrink-0 rounded-full bg-accent-start" />
   296→                    <span>{b}</span>
   297→                  </motion.li>
   298→                ))}
   299→              </ul>
   300→            )}
   301→          </div>
   302→        );
   303→      })}
   304→    </div>
   305→  );
   306→}

   321→  onRegenerate?: () => void;
   322→  /** Phase 2 — speak this Oracle response aloud (TTS). */

   324→  /** Phase 2 — whether TTS playback is currently active for this message. */

   327→
   328→export function OracleMessageView({
   329→  message,
   330→  onFollowUp,
   331→  onAction,
   332→  onNextBestStep,
   333→  onRunAction,
   334→  actionsDisabled,
   335→  onRegenerate,
   336→  onSpeak,
   337→  isSpeaking,
   338→}: OracleMessageProps) {
   339→  if (message.role === 'user') {
   340→    return (
   341→      <UserMessage
   342→        content={message.content}
   343→        mode={message.mode}
   344→        language={message.language}
   345→        emotion={message.emotion}
   346→      />
   347→    );
   348→  }
   349→
   350→  const phase = message.phase ?? 'done';
   351→  const isPreStream =
   352→    phase === 'thinking' ||
   353→    phase === 'reading' ||
   354→    phase === 'analyzing' ||
   355→    phase === 'preparing';
   356→  const isError = phase === 'error';
   357→  const parts = message.parts;
   358→  const mode = message.mode ? ORACLE_MODES[message.mode] : undefined;
   359→  const isStreaming = phase === 'streaming';
   360→  const isDone = phase === 'done';
   361→
   362→  // Build a plain-text version for copy / export / speak.
   363→  const plainText = parts
   364→    ? [
   365→        parts.keyInsight ? `KEY INSIGHT\n${parts.keyInsight}` : '',
   366→        parts.analysis ? `\nANALYSIS\n${parts.analysis}` : '',
   367→        parts.recommendedActions.length > 0
   368→          ? `\nRECOMMENDED ACTIONS\n${parts.recommendedActions.map((a, i) => `${i + 1}. ${a}`).join('\n')}`
   369→          : '',
   370→        parts.potentialRisks.length > 0
   371→          ? `\nPOTENTIAL RISKS\n${parts.potentialRisks.map((r) => `• ${r}`).join('\n')}`
   372→          : '',
   373→        parts.nextBestStep ? `\nNEXT BEST STEP\n${parts.nextBestStep}` : '',
   374→      ]
   375→        .filter(Boolean)
   376→        .join('\n')
   377→    : message.content;
   378→
   379→  const handleCopy = () => {
   380→    if (!plainText) return;
   381→    navigator.clipboard?.writeText(plainText).catch(() => {});
   382→  };
   383→
   384→  const handleExport = () => {
   385→    if (!plainText) return;
   386→    const blob = new Blob([plainText], { type: 'text/plain;charset=utf-8' });
   387→    const url = URL.createObjectURL(blob);
   388→    const a = document.createElement('a');
   389→    a.href = url;
   390→    a.download = `oracle-response-${new Date(message.createdAt).getTime()}.txt`;
   391→    document.body.appendChild(a);
   392→    a.click();
   393→    document.body.removeChild(a);
   394→    URL.revokeObjectURL(url);
   395→  };
   396→
   397→  return (
   398→    <div className="flex gap-3">
   399→      {/* Oracle avatar — dynamic facial expressions */}
   400→      <div className="shrink-0">
   401→        <OracleAvatar
   402→          state={deriveAvatarState(phase, message.emotion)}
   403→          size={40}
   404→        />
   405→      </div>
   406→
   407→      {/* Body */}
   408→      <div className="min-w-0 flex-1 space-y-4">
   409→        {/* Mode badge + status row */}
   410→        <div className="flex items-center gap-2">
   411→          {mode && <ModeBadge mode={mode.id} />}
   412→          {isStreaming && (
   413→            <span className="flex items-center gap-1.5 text-[10px] font-medium text-muted-foreground">
   414→              <span className="relative flex h-1.5 w-1.5">
   415→                <span className="absolute inline-flex h-full w-full animate-ping rounded-full bg-emerald-400 opacity-75" />
   416→                <span className="relative inline-flex h-1.5 w-1.5 rounded-full bg-emerald-400" />
   417→              </span>
   418→              Oracle is responding
   419→            </span>
   420→          )}
   421→        </div>
   422→
   423→        {/* Error state */}
   424→        {isError && (
   425→          <div className="rounded-2xl border border-red-500/30 bg-red-500/5 p-4 text-sm text-red-600 dark:text-red-400">
   426→            <p className="font-medium">Oracle hit a snag.</p>
   427→            <p className="mt-1 text-xs opacity-80">{message.error ?? 'Unknown error'}</p>
   428→          </div>
   429→        )}
   430→
   431→        {/* Pre-stream phase indicator */}
   432→        {isPreStream && <PhaseIndicator phase={phase} />}
   433→
   434→        {/* Structured parts — rendered progressively during streaming */}
   435→        {!isError && !isPreStream && parts && (
   436→          <div className="space-y-5">
   437→            {/* 1. Key Insight — the headline */}
   438→            {parts.keyInsight && (
   439→              <Section icon={Lightbulb} title="Key Insight" delay={0}>
   440→                <div
   441→                  className={cn(
   442→                    'rounded-2xl border-l-[3px] border-[color-mix(in_srgb,var(--accent-start)_70%,transparent)] bg-gradient-to-r from-[color-mix(in_srgb,var(--accent-start)_8%,transparent)] to-transparent px-4 py-3 text-[15px] font-semibold leading-snug text-foreground',
   443→                    isStreaming && 'typing-cursor',
   444→                  )}
   445→                >
   446→                  {parts.keyInsight}
   447→                </div>
   448→              </Section>
   449→            )}
   450→
   451→            {/* 2. Analysis — the strategic reasoning */}
   452→            {parts.analysis && (
   453→              <Section icon={Compass} title="Analysis" delay={0.05}>
   454→                <AnalysisBody text={parts.analysis} isStreaming={isStreaming} />
   455→              </Section>

   465→                      initial={{ opacity: 0, x: -6 }}

   475→                  ))}
   476→                </ol>
   477→              </Section>

   479→
   480→            {/* 4. Potential Risks — warning cards */}
   481→            {parts.potentialRisks.length > 0 && (
   482→              <Section icon={ShieldAlert} title="Potential Risks" delay={0.15}>
   483→                <div className="space-y-2">
   484→                  {parts.potentialRisks.map((risk, i) => (
   485→                    <motion.div

   487→                      initial={{ opacity: 0, x: -6 }}
   488→                      animate={{ opacity: 1, x: 0 }}
   489→                      transition={{ duration: 0.3, delay: i * 0.04 }}

   496→                </div>
   497→              </Section>
   498→            )}
   499→
   500→            {/* 5. Next Best Step — highlighted CTA */}
   501→            {parts.nextBestStep && (
   502→              <Section icon={Target} title="Next Best Step" delay={0.2}>
   503→                <motion.button
   504→                  initial={{ opacity: 0, scale: 0.98 }}

   514→                    }
   515→                  }}
   516→                  className="group flex w-full items-center gap-3 rounded-2xl border border-[color-mix(in_srgb,var(--accent-start)_40%,transparent)] bg-gradient-to-r from-[color-mix(in_srgb,var(--accent-start)_12%,transparent)] to-[color-mix(in_srgb,var(--accent-end)_8%,transparent)] px-4 py-3 text-left transition-all hover:border-[color-mix(in_srgb,var(--accent-start)_60%,transparent)] hover:shadow-[0_4px_24px_-8px_rgba(0,229,255,0.2)]"
   517→                >

   531→            {/* Investigation Mode — clarifying questions */}
   532→            {parts.investigation && parts.investigation.length > 0 && (
   533→              <motion.div
   534→                initial={{ opacity: 0, y: 8 }}
   535→                animate={{ opacity: 1, y: 0 }}
   536→                transition={{ duration: 0.35, delay: 0.25 }}
   537→                className="rounded-2xl border border-[color-mix(in_srgb,var(--accent-start)_30%,transparent)] bg-[color-mix(in_srgb,var(--accent-start)_6%,transparent)] p-4"
   538→              >
   539→                <div className="mb-2 flex items-center gap-2">
   540→                  <HelpCircle className="h-3.5 w-3.5 accent-text" />
   541→                  <h4 className="text-[11px] font-semibold uppercase tracking-wider text-muted-foreground">
   542→                    Investigation Mode
   543→                  </h4>
   544→                </div>
   545→                <p className="mb-3 text-[11px] text-muted-foreground">
   546→                  Before Oracle commits to a full answer, a quick clarification will make this sharper:
   547→                </p>
   548→                <div className="flex flex-col gap-1.5">
   549→                  {parts.investigation.map((q, i) => (
   550→                    <motion.button
   551→                      key={i}
   552→                      initial={{ opacity: 0, y: 4 }}
   553→                      animate={{ opacity: 1, y: 0 }}
   554→                      transition={{ duration: 0.25, delay: i * 0.05 }}
   555→                      whileHover={{ x: 2 }}
   556→                      onClick={() => onFollowUp?.(q)}
   557→                      disabled={!isDone}
   558→                      className="group flex items-center gap-2 rounded-xl border border-border bg-card/[0.4] px-3.5 py-2 text-left text-sm text-foreground/90 transition-colors hover:border-[color-mix(in_srgb,var(--accent-start)_40%,transparent)] hover:bg-card/[0.7] disabled:cursor-not-allowed disabled:opacity-50"
   559→                    >
   560→                      <Sparkles className="h-3 w-3 accent-text" />
   561→                      {q}
   562→                    </motion.button>
   563→                  ))}
   564→                </div>
   565→              </motion.div>
   566→            )}
   567→
   568→            {/* 6. Sources */}
   569→            {parts.sources.length > 0 && (
   570→              <Section icon={Quote} title="Sources" delay={0.25}>
   571→                <div className="flex flex-wrap gap-2">
   572→                  {parts.sources.map((s, i) => (
   573→                    <SourceCard key={i} source={s} index={i} />
   574→                  ))}
   575→                </div>
   576→              </Section>
   577→            )}
   578→
   579→            {/* 7. Follow-up Questions */}
   580→            {parts.followUps.length > 0 && (
   581→              <Section icon={Sparkles} title="Follow-up Questions" delay={0.3}>
   582→                <div className="flex flex-col gap-1.5">
   583→                  {parts.followUps.map((q, i) => (
   584→                    <motion.button
   585→                      key={i}
   586→                      initial={{ opacity: 0, y: 4 }}
   587→                      animate={{ opacity: 1, y: 0 }}
   588→                      transition={{ duration: 0.25, delay: i * 0.04 }}
   589→                      whileHover={{ x: 2 }}
   590→                      onClick={() => onFollowUp?.(q)}
   591→                      disabled={!isDone}
   592→                      className="group flex items-center gap-2 rounded-xl border border-border bg-card/[0.3] px-3.5 py-2 text-left text-sm text-foreground/90 transition-colors hover:border-[color-mix(in_srgb,var(--accent-start)_40%,transparent)] hover:bg-card/[0.6] disabled:cursor-not-allowed disabled:opacity-50"
   593→                    >
   594→                      <ArrowRight className="h-3 w-3 text-muted-foreground transition-colors group-hover:accent-text" />
   595→                      {q}
   596→                    </motion.button>
   597→                  ))}
   598→                </div>
   599→              </Section>
   600→            )}
   601→
   602→            {/* 8. Phase 3 — Action Cards (3 large glass buttons, single-click execution) */}
   603→            {parts.actionCards && parts.actionCards.length > 0 && (
   604→              <motion.div
   605→                initial={{ opacity: 0, y: 8 }}
   606→                animate={{ opacity: 1, y: 0 }}
   607→                transition={{ duration: 0.35, delay: 0.35 }}
   608→              >
   609→                <OracleActionCards
   610→                  cards={parts.actionCards}
   611→                  onRun={onRunAction}
   612→                  disabled={!isDone || actionsDisabled}
   613→                />
   614→              </motion.div>
   615→            )}
   616→          </div>
   617→        )}
   618→
   619→        {/* Done checkmark + action row */}
   620→        {isDone && parts && (
   621→          <motion.div
   622→            initial={{ opacity: 0 }}
   623→            animate={{ opacity: 1 }}
   624→            transition={{ delay: 0.2 }}
   625→            className="flex flex-wrap items-center gap-1.5"
   626→          >
   627→            <span className="flex items-center gap-1.5 text-[10px] text-muted-foreground">
   628→              <CheckCircle2 className="h-3 w-3 text-emerald-500" />
   629→              <span>Oracle · {new Date(message.createdAt).toLocaleTimeString('en-IN', { hour: '2-digit', minute: '2-digit' })}</span>
   630→            </span>
   631→
   632→            {/* Action row — copy / speak / regenerate / export */}
   633→            <div className="ml-auto flex items-center gap-0.5">
   634→              <MsgActionButton
   635→                icon={Copy}
   636→                label="Copy"
   637→                onClick={handleCopy}
   638→              />
   639→              {onSpeak && (
   640→                <MsgActionButton
   641→                  icon={Volume2}
   642→                  label={isSpeaking ? 'Speaking…' : 'Listen'}
   643→                  onClick={() => onSpeak(message)}
   644→                  active={isSpeaking}
   645→                  disabled={isSpeaking}
   646→                />
   647→              )}
   648→              {onRegenerate && (
   649→                <MsgActionButton
   650→                  icon={RefreshCw}
   651→                  label="Regenerate"
   652→                  onClick={onRegenerate}
   653→                />
   654→              )}
   655→              <MsgActionButton
   656→                icon={FileText}
   657→                label="Export"
   658→                onClick={handleExport}
   659→              />
   660→            </div>
   661→          </motion.div>
   662→        )}
   663→      </div>
   664→    </div>
   665→  );
   666→}

   670→function MsgActionButton({
   671→  icon: Icon,
   672→  label,
   673→  onClick,
   674→  active = false,
   675→  disabled = false,
   676→}: {
   677→  icon: LucideIcon;
   678→  label: string;
   679→  onClick: () => void;
   680→  active?: boolean;
   681→  disabled?: boolean;
   682→}) {
   683→  return (
   684→    <button
   685→      type="button"
   686→      onClick={onClick}
   687→      disabled={disabled}
   688→      aria-label={label}
   689→      title={label}
   690→      className={cn(
   691→        'flex h-7 items-center gap-1 rounded-lg px-2 text-[10px] font-medium transition-colors disabled:cursor-not-allowed disabled:opacity-50',
   692→        active
   693→          ? 'accent-text bg-card/[0.6]'
   694→          : 'text-muted-foreground hover:bg-card/[0.5] hover:text-foreground',
   695→      )}
   696→    >
   697→      <Icon className={cn('h-3 w-3', active && 'animate-pulse')} />
   698→      <span className="hidden sm:inline">{label}</span>
   699→    </button>
   700→  );
   701→}
