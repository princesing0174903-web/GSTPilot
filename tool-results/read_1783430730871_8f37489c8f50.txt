   265→function getKanbanColumn(status: FilingStatus): KanbanColumn {
   266→  switch (status) {
   267→    case 'draft':
   268→    case 'prepared':
   269→      return 'draft';
   270→    case 'validated':
   271→    case 'reviewed':
   272→    case 'generated':
   273→      return 'ready';
   274→    case 'filed':
   275→      return 'filed';
   276→    case 'reopened':
   277→      return 'attention';
   278→    default:
   279→      return 'draft';
   280→  }
   281→}
   282→
   283→function getTimelineStepIndex(status: FilingStatus): number {
   284→  const idx = TIMELINE_ORDER.indexOf(status);
   285→  return idx >= 0 ? idx : -1;
   286→}

   288→function getAttentionSummary(ret: ReturnItem): string {
   289→  if (ret.issuesFound <= 0) return 'No issues';
   290→  const parts: string[] = [];
   291→  if (ret.criticalErrors > 0) parts.push(`${ret.criticalErrors} critical error${ret.criticalErrors > 1 ? 's' : ''}`);
   292→  if (ret.warnings > 0) parts.push(`${ret.warnings} warning${ret.warnings > 1 ? 's' : ''}`);
   293→  if (ret.issuesFound > 0 && parts.length === 0) parts.push(`${ret.issuesFound} issue${ret.issuesFound > 1 ? 's' : ''}`);
   294→  return parts.join(' · ');
   295→}

   562→          <div className="flex items-center gap-3">
   563→            <Skeleton className="h-8 w-44" />
   564→            <Skeleton className="h-6 w-20 rounded-full" />

   573→            <div key={i} className="space-y-3">
   574→              <Skeleton className="h-12 w-full rounded-t-lg" />
   575→              <div className="space-y-2 p-2">
   576→                <Skeleton className="h-28 w-full rounded-lg" />
   577→                <Skeleton className="h-28 w-full rounded-lg" />
