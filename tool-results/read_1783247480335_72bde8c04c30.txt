    27→function formatINR(amount: number): string {
    28→  const parts = amount.toFixed(0).split('.');
    29→  let intPart = parts[0];
    30→  const lastThree = intPart.slice(-3);
    31→  const otherNumbers = intPart.slice(0, -3);
    32→  if (otherNumbers !== '' && otherNumbers !== '-') {
    33→    intPart = otherNumbers.replace(/\B(?=(\d{2})+(?!\d))/g, ',') + ',' + lastThree;
    34→  } else {
    35→    intPart = lastThree;
    36→  }
    37→  return '₹' + intPart;
    38→}

    40→function formatDate(dateStr: string): string {
    41→  const d = new Date(dateStr);
    42→  const dd = String(d.getDate()).padStart(2, '0');
    43→  const mm = String(d.getMonth() + 1).padStart(2, '0');
    44→  const yyyy = d.getFullYear();
    45→  return `${dd}/${mm}/${yyyy}`;
    46→}

   108→// SCORE COMPUTATION
   109→// ═══════════════════════════════════════════════════════════════════════════════
   110→
   111→function computeScores(invoices: typeof demoInvoices) {
   112→  const total = invoices.length;
   113→  const unpaid = invoices.filter(i => i.status === 'unpaid').length;
   114→  const overdue = invoices.filter(i => i.status === 'overdue').length;
   115→  const totalAmount = invoices.reduce((s, i) => s + i.amount, 0);
   116→  const overdueAmount = invoices.filter(i => i.status === 'overdue').reduce((s, i) => s + i.amount, 0);
   117→  const avgAge = invoices.reduce((s, i) => s + i.age, 0) / total;
   118→  const eligibleCount = invoices.filter(i => i.eligible).length;
   119→  const eligibleAmount = invoices.filter(i => i.eligible).reduce((s, i) => s + i.amount, 0);
   120→
   121→  // Cash Flow Score: f(invoice aging, collection speed, bank balance trend)
   122→  const ageFactor = Math.max(0, 100 - avgAge * 0.8);
   123→  const collectionFactor = ((total - overdue) / total) * 100;
   124→  const cashFlow = Math.round(ageFactor * 0.4 + collectionFactor * 0.6);
   125→
   126→  // Credit Score: f(payment regularity, GST compliance, default history)
   127→  const paymentRegularity = Math.min(100, ((unpaid + 1) / (overdue + 1)) * 50);
   128→  const gstCompliance = 85; // assumed
   129→  const credit = Math.round(paymentRegularity * 0.5 + gstCompliance * 0.5);
   130→
   131→  // Collection Score: f(on-time collections %, overdue amount ratio, DSO)
   132→  const onTimePct = ((total - overdue) / total) * 100;
   133→  const overdueRatio = overdueAmount / totalAmount;
   134→  const dso = avgAge;
   135→  const dsoFactor = Math.max(0, 100 - (dso - 30) * 1.5);
   136→  const collection = Math.round(onTimePct * 0.4 + (1 - overdueRatio) * 100 * 0.35 + dsoFactor * 0.25);
   137→
   138→  // Business Health: weighted average
   139→  const health = Math.round(cashFlow * 0.3 + credit * 0.25 + collection * 0.25 + gstCompliance * 0.2);
   140→
   141→  return {
   142→    cashFlow: Math.min(100, Math.max(0, cashFlow)),
   143→    credit: Math.min(100, Math.max(0, credit)),
   144→    collection: Math.min(100, Math.max(0, collection)),
   145→    health: Math.min(100, Math.max(0, health)),
   146→    eligibleCount,
   147→    eligibleAmount,
   148→    totalAmount,
   149→    overdueAmount,
   150→  };
   151→}

   157→function ScoreGauge({ score, label, icon: Icon, size = 140 }: {
   158→  score: number; label: string; icon: React.ElementType; size?: number;
   159→}) {
   160→  const r = (size - 20) / 2;
   161→  const cx = size / 2;
   162→  const cy = size / 2;
   163→  const circumference = 2 * Math.PI * r;
   164→  const strokeDashoffset = circumference * (1 - score / 100);
   165→
   166→  const getColor = (s: number) => {
   167→    if (s >= 70) return { stroke: '#10b981', fill: '#ecfdf5', text: 'text-emerald-600', bg: 'bg-emerald-50' };
   168→    if (s >= 40) return { stroke: '#f59e0b', fill: '#fffbeb', text: 'text-amber-600', bg: 'bg-amber-50' };
   169→    return { stroke: '#ef4444', fill: '#fef2f2', text: 'text-red-600', bg: 'bg-red-50' };
   170→  };
   171→
   172→  const color = getColor(score);
   173→
   174→  return (
   175→    <div className="flex flex-col items-center gap-2">
   176→      <div className="relative" style={{ width: size, height: size }}>
   177→        <svg width={size} height={size} className="transform -rotate-90">
   178→          <circle cx={cx} cy={cy} r={r} fill="none" stroke="#e2e8f0" strokeWidth="8" />
   179→          <motion.circle
   180→            cx={cx} cy={cy} r={r} fill="none"
   181→            stroke={color.stroke}
   182→            strokeWidth="8"
   183→            strokeLinecap="round"
   184→            strokeDasharray={circumference}
   185→            initial={{ strokeDashoffset: circumference }}
   186→            animate={{ strokeDashoffset }}
   187→            transition={{ duration: 1.5, ease: 'easeOut', delay: 0.2 }}
   188→          />
   189→        </svg>
   190→        <div className="absolute inset-0 flex flex-col items-center justify-center">
   191→          <motion.span
   192→            className="text-2xl font-bold"
   193→            initial={{ opacity: 0, scale: 0.5 }}

   195→            transition={{ duration: 0.5, delay: 0.8 }}

   204→      </div>

   214→function Sparkline({ data, color = '#10b981', width = 120, height = 32 }: {
   215→  data: number[]; color?: string; width?: number; height?: number;
   216→}) {
   217→  const max = Math.max(...data);
   218→  const min = Math.min(...data);
   219→  const range = max - min || 1;
   220→  const padding = 2;
   221→
   222→  const points = data.map((v, i) => {
   223→    const x = padding + (i / (data.length - 1)) * (width - padding * 2);
   224→    const y = padding + (1 - (v - min) / range) * (height - padding * 2);
   225→    return `${x},${y}`;
   226→  }).join(' ');
   227→
   228→  const areaPoints = `${padding},${height - padding} ${points} ${width - padding},${height - padding}`;
   229→
   230→  return (
   231→    <svg width={width} height={height} className="overflow-visible">
   232→      <motion.polygon
   233→        points={areaPoints}
   234→        fill={color}
   235→        fillOpacity={0.1}
   236→        initial={{ opacity: 0 }}
   237→        animate={{ opacity: 1 }}
   238→        transition={{ duration: 0.8, delay: 0.5 }}
   239→      />
   240→      <motion.polyline
   241→        points={points}
   242→        fill="none"
   243→        stroke={color}
   244→        strokeWidth="2"
   245→        strokeLinecap="round"
   246→        strokeLinejoin="round"
   247→        initial={{ pathLength: 0 }}
   248→        animate={{ pathLength: 1 }}
   249→        transition={{ duration: 1.2, delay: 0.3 }}
   250→      />
   251→      {data.length > 0 && (
   252→        <motion.circle
   253→          cx={padding + ((data.length - 1) / (data.length - 1)) * (width - padding * 2)}
   254→          cy={padding + (1 - (data[data.length - 1] - min) / range) * (height - padding * 2)}
   255→          r="3"
   256→          fill={color}
   257→          initial={{ scale: 0 }}
   258→          animate={{ scale: 1 }}
   259→          transition={{ duration: 0.3, delay: 1.5 }}
   260→        />
   261→      )}
   262→    </svg>
   263→  );
   264→}

   270→function RevenueAreaChart({ data }: { data: typeof monthlyRevenue }) {
   271→  const width = 700;
   272→  const height = 180;
   273→  const padding = { top: 10, right: 10, bottom: 30, left: 70 };
   274→  const chartW = width - padding.left - padding.right;
   275→  const chartH = height - padding.top - padding.bottom;
   276→
   277→  const maxVal = Math.max(...data.map(d => d.value));
   278→  const minVal = Math.min(...data.map(d => d.value)) * 0.85;
   279→
   280→  const points = data.map((d, i) => ({
   281→    x: padding.left + (i / (data.length - 1)) * chartW,
   282→    y: padding.top + (1 - (d.value - minVal) / (maxVal - minVal)) * chartH,
   283→  }));
   284→
   285→  const linePath = points.map((p, i) => `${i === 0 ? 'M' : 'L'}${p.x},${p.y}`).join(' ');
   286→  const areaPath = `${linePath} L${points[points.length - 1].x},${padding.top + chartH} L${points[0].x},${padding.top + chartH} Z`;
   287→
   288→  return (
   289→    <svg viewBox={`0 0 ${width} ${height}`} className="w-full" preserveAspectRatio="xMidYMid meet">
   290→      <defs>
   291→        <linearGradient id="revenueGrad" x1="0" y1="0" x2="0" y2="1">
   292→          <stop offset="0%" stopColor="#10b981" stopOpacity={0.3} />
   293→          <stop offset="100%" stopColor="#10b981" stopOpacity={0.02} />
   294→        </linearGradient>
   295→      </defs>
   296→      {/* Grid lines */}
   297→      {[0, 0.25, 0.5, 0.75, 1].map((pct) => {
   298→        const y = padding.top + pct * chartH;
   299→        const val = maxVal - pct * (maxVal - minVal);
   300→        return (
   301→          <g key={pct}>
   302→            <line x1={padding.left} y1={y} x2={width - padding.right} y2={y} stroke="#e2e8f0" strokeDasharray="3,3" />
   303→            <text x={padding.left - 8} y={y + 4} textAnchor="end" className="text-[9px] fill-slate-400">{formatINR(val)}</text>
   304→          </g>
   305→        );
   306→      })}
   307→      {/* X labels */}
   308→      {data.map((d, i) => {
   309→        const x = padding.left + (i / (data.length - 1)) * chartW;
   310→        return (
   311→          <text key={i} x={x} y={height - 5} textAnchor="middle" className="text-[8px] fill-slate-400">
   312→            {d.month.split(' ')[0]}
   313→          </text>
   314→        );
   315→      })}
   316→      <motion.path
   317→        d={areaPath}
   318→        fill="url(#revenueGrad)"
   319→        initial={{ opacity: 0 }}
   320→        animate={{ opacity: 1 }}
   321→        transition={{ duration: 1, delay: 0.3 }}
   322→      />
   323→      <motion.path
   324→        d={linePath}
   325→        fill="none"
   326→        stroke="#10b981"
   327→        strokeWidth="2.5"
   328→        strokeLinecap="round"
   329→        strokeLinejoin="round"
   330→        initial={{ pathLength: 0 }}
   331→        animate={{ pathLength: 1 }}
   332→        transition={{ duration: 1.5, delay: 0.2 }}
   333→      />
   334→      {points.map((p, i) => (
   335→        <motion.circle
   336→          key={i}
   337→          cx={p.x} cy={p.y} r="3.5"
   338→          fill="white" stroke="#10b981" strokeWidth="2"
   339→          initial={{ scale: 0 }}
   340→          animate={{ scale: 1 }}
   341→          transition={{ duration: 0.3, delay: 0.5 + i * 0.08 }}
   342→        />
   343→      ))}
   344→    </svg>
   345→  );
   346→}

   352→function EMIChart({ principal, rate, tenure }: { principal: number; rate: number; tenure: number }) {
   353→  const monthlyRate = rate / 12 / 100;
   354→  const emi = monthlyRate > 0
   355→    ? principal * monthlyRate * Math.pow(1 + monthlyRate, tenure) / (Math.pow(1 + monthlyRate, tenure) - 1)
   356→    : principal / tenure;
   357→  const totalPayment = emi * tenure;
   358→  const totalInterest = totalPayment - principal;
   359→
   360→  const size = 160;
   361→  const cx = size / 2;
   362→  const cy = size / 2;
   363→  const r = 55;
   364→  const circumference = 2 * Math.PI * r;
   365→  const principalPct = principal / totalPayment;
   366→  const principalDash = circumference * principalPct;
   367→
   368→  return (
   369→    <div className="flex items-center gap-4">
   370→      <svg width={size} height={size}>
   371→        <circle cx={cx} cy={cy} r={r} fill="none" stroke="#e2e8f0" strokeWidth="16" />
   372→        <motion.circle
   373→          cx={cx} cy={cy} r={r} fill="none"
   374→          stroke="#10b981" strokeWidth="16"
   375→          strokeDasharray={`${principalDash} ${circumference}`}
   376→          strokeLinecap="round"
   377→          transform={`rotate(-90 ${cx} ${cy})`}
   378→          initial={{ strokeDasharray: `0 ${circumference}` }}
   379→          animate={{ strokeDasharray: `${principalDash} ${circumference}` }}
   380→          transition={{ duration: 1, delay: 0.3 }}
   381→        />
   382→        <motion.circle
   383→          cx={cx} cy={cy} r={r} fill="none"
   384→          stroke="#f59e0b" strokeWidth="16"
   385→          strokeDasharray={`${circumference - principalDash} ${circumference}`}
   386→          strokeDashoffset={-principalDash}
   387→          strokeLinecap="round"
   388→          transform={`rotate(-90 ${cx} ${cy})`}
   389→          initial={{ opacity: 0 }}
   390→          animate={{ opacity: 1 }}
   391→          transition={{ duration: 0.5, delay: 1 }}
   392→        />
   393→        <text x={cx} y={cy - 6} textAnchor="middle" className="text-xs font-bold fill-slate-700">{formatINR(Math.round(emi))}</text>
   394→        <text x={cx} y={cy + 10} textAnchor="middle" className="text-[9px] fill-slate-400">/month</text>
   395→      </svg>
   396→      <div className="flex flex-col gap-2 text-xs">
   397→        <div className="flex items-center gap-2">
   398→          <div className="h-3 w-3 rounded-sm bg-emerald-500" />
   399→          <span className="text-slate-600">Principal: {formatINR(principal)}</span>
   400→        </div>
   401→        <div className="flex items-center gap-2">
   402→          <div className="h-3 w-3 rounded-sm bg-amber-500" />
   403→          <span className="text-slate-600">Interest: {formatINR(Math.round(totalInterest))}</span>
   404→        </div>
   405→        <div className="flex items-center gap-2">
   406→          <div className="h-3 w-3 rounded-sm bg-slate-300" />
   407→          <span className="text-slate-600">Total: {formatINR(Math.round(totalPayment))}</span>
   408→        </div>
   409→      </div>
   410→    </div>
   411→  );
   412→}

   832→function WorkingCapitalLoansTab() {
   833→  const [loanAppOpen, setLoanAppOpen] = useState(false);
   834→  const [emiPrincipal, setEmiPrincipal] = useState(2000000);
   835→  const [emiRate, setEmiRate] = useState(12);
   836→  const [emiTenure, setEmiTenure] = useState(24);
   837→
   838→  // ── Real data fetch for eligibility checks ──
   839→  // Annual invoice volume = sum of Invoice.taxableValue over the last 12 months
   840→  // GST filing regularity = filedReturns / (filedReturns + pendingReturns + overdueReturns)
   841→  const { data: invoicesResp, isLoading: invoicesLoading } = useQuery<{ invoices: Array<{ taxableValue: number; invoiceDate: string }> }>({
   842→    queryKey: ['invoices', 'wc-loans'],
   843→    queryFn: () => apiGet('/api/invoices'),
   844→  });
   845→  const { data: dashboard } = useQuery<{ filedReturns: number; pendingReturns: number; overdueReturns: number }>({
   846→    queryKey: ['dashboard', 'wc-loans'],
   847→    queryFn: () => apiGet('/api/dashboard'),
   848→  });
   849→
   850→  const annualInvoiceVolume = useMemo(() => {
   851→    const invoices = invoicesResp?.invoices ?? [];
   852→    if (invoices.length === 0) return 0;
   853→    const oneYearAgo = Date.now() - 365 * 24 * 60 * 60 * 1000;
   854→    return invoices
   855→      .filter((inv) => {
   856→        if (!inv.invoiceDate) return true;
   857→        const d = new Date(inv.invoiceDate).getTime();
   858→        return Number.isNaN(d) || d >= oneYearAgo;
   859→      })
   860→      .reduce((sum, inv) => sum + (Number(inv.taxableValue) || 0), 0);
   861→  }, [invoicesResp]);
   862→
   863→  const filingRegularity = useMemo(() => {
   864→    const filed = dashboard?.filedReturns ?? 0;
   865→    const pending = dashboard?.pendingReturns ?? 0;
   866→    const overdue = dashboard?.overdueReturns ?? 0;
   867→    const total = filed + pending + overdue;
   868→    if (total === 0) return { filed: 0, total: 0, pct: 0 };
   869→    return { filed, total, pct: Math.round((filed / total) * 100) };
   870→  }, [dashboard]);
   871→
   872→  // 1.2 Cr threshold = 12,000,000. Below this → status flips to 'review'.
   873→  const INVOICE_VOLUME_THRESHOLD = 12000000;
