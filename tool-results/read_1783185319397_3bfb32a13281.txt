    35→// ─── Helpers ──────────────────────────────────────────────────────────────
    36→function formatINR(n: number): string {
    37→  const s = Math.abs(Math.round(n)).toString()
    38→  let result = ''
    39→  let count = 0
    40→  for (let i = s.length - 1; i >= 0; i--) {
    41→    result = s[i] + result
    42→    count++
    43→    if (count === 3 && i > 0) { result = ',' + result; count = 0 }
    44→    else if (count > 3 && count % 2 === 1 && i > 0) { result = ',' + result }
    45→  }
    46→  return '₹' + (n < 0 ? '-' : '') + result
    47→}
    48→
    49→function formatDate(iso: string | null | unknown): string {
    50→  if (!iso) return '—'
    51→  try {
    52→    const d = new Date(iso as string)
    53→    return d.toLocaleDateString('en-IN', { day: '2-digit', month: '2-digit', year: 'numeric' })
    54→  } catch { return '—' }
    55→}
    56→
    57→function getMonthKey(iso: string | unknown): string {
    58→  if (!iso) return ''
    59→  try {
    60→    const d = new Date(iso as string)
    61→    return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}`
    62→  } catch { return '' }
    63→}
    64→
    65→function getMonthLabel(key: string): string {
    66→  const months = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec']
    67→  const [y, m] = key.split('-')
    68→  return months[parseInt(m, 10) - 1] + ' ' + y.slice(2)
    69→}

    71→// ─── Sparkline SVG ────────────────────────────────────────────────────────
    72→function Sparkline({ data, color = '#10b981', width = 80, height = 28 }: {
    73→  data: number[]; color?: string; width?: number; height?: number
    74→}) {
    75→  if (data.length < 2) return null
    76→  const min = Math.min(...data)
    77→  const max = Math.max(...data)
    78→  const range = max - min || 1
    79→  const pts = data.map((v, i) => {
    80→    const x = (i / (data.length - 1)) * width
    81→    const y = height - ((v - min) / range) * (height - 4) - 2
    82→    return `${x},${y}`
    83→  }).join(' ')
    84→  return (
    85→    <svg width={width} height={height} className="inline-block">
    86→      <polyline fill="none" stroke={color} strokeWidth="1.5" points={pts} />
    87→    </svg>
    88→  )
    89→}

    91→// ─── Bar Chart SVG ────────────────────────────────────────────────────────
    92→function BarChart({ data, labels, projectedIndex, width = 420, height = 180 }: {
    93→  data: number[]; labels: string[]; projectedIndex?: number; width?: number; height?: number
    94→}) {
    95→  if (data.length === 0) return null
    96→  const max = Math.max(...data, 1)
    97→  const barW = Math.min(36, (width - 40) / data.length - 8)
    98→  const chartH = height - 30
    99→  return (
   100→    <svg width={width} height={height} viewBox={`0 0 ${width} ${height}`}>
   101→      {data.map((v, i) => {
   102→        const x = 30 + i * ((width - 40) / data.length) + 4
   103→        const barH = (v / max) * (chartH - 10)
   104→        const isProjected = projectedIndex !== undefined && i >= projectedIndex
   105→        return (
   106→          <g key={i}>
   107→            <rect x={x} y={chartH - barH} width={barW} height={barH}
   108→              fill={isProjected ? '#10b981' : '#059669'} opacity={isProjected ? 0.5 : 0.85}
   109→              rx={3} />
   110→            <text x={x + barW / 2} y={height - 4} textAnchor="middle"
   111→              className="text-[10px] fill-slate-500">{labels[i] || ''}</text>
   112→            <text x={x + barW / 2} y={chartH - barH - 4} textAnchor="middle"
   113→              className="text-[9px] fill-slate-600">
   114→              {v > 999 ? `${(v / 1000).toFixed(0)}K` : v.toFixed(0)}
   115→            </text>
   116→          </g>
   117→        )
   118→      })}
   119→      {/* Y-axis line */}
   120→      <line x1="28" y1="5" x2="28" y2={chartH} stroke="#e2e8f0" strokeWidth="1" />
   121→    </svg>
   122→  )
   123→}
