    24→    24→function computeRiskLevel(trust: number): RiskLevel {
    25→    25→  if (trust >= 85) return 'low';

    27→    27→  if (trust >= 40) return 'high';

    31→    31→function computeTrustScore(compliance: number, paymentReliability: number, rating: number): number {
    32→    32→  // rating is 0-5, scale to 0-100
    33→    33→  const ratingPct = (rating / 5) * 100;
    34→    34→  const composite = compliance * 0.35 + paymentReliability * 0.35 + ratingPct * 0.3;
    35→    35→  return Math.round(Math.min(100, Math.max(0, composite)));
    36→    36→}

   289→   289→export async function ensureNetworkSeeded(): Promise<void> {
   290→   290→  if (SEED_LOCK.value) return;
   291→   291→  SEED_LOCK.value = true;
   292→   292→  try {
   293→   293→    // Make sure platform orgs exist first (they anchor to the real Firm)
   294→   294→    await ensurePlatformOrganizationsSeeded();
   295→   295→
   296→   296→    const existingCount = await db.networkNode.count();
   297→   297→    if (existingCount > 0) return; // idempotent
   298→   298→
   299→   299→    // ── 1. Create NetworkNodes for every REAL PlatformOrganization ──
   300→   300→    const platformOrgs = await db.platformOrganization.findMany({
   301→   301→      orderBy: { createdAt: 'asc' },
   302→   302→    });
   303→   303→
   304→   304→    // The first platform org is the host firm itself — treat as 'organization' type.
   305→   305→    // All other platform orgs are modeled as customers (they're the firm's customer base).
   306→   306→    for (const org of platformOrgs) {
   307→   307→      const isHost = org.organizationType === 'parent';
   308→   308→      const nodeType: NodeType = isHost ? 'organization' : 'customer';
   309→   309→      const complianceScore = isHost ? 92 : Math.min(95, 70 + Math.floor((org.healthScore || 80) * 0.25));
   310→   310→      const paymentReliabilityScore = isHost ? 88 : Math.min(95, 65 + Math.floor((org.healthScore || 80) * 0.3));
   311→   311→      const rating = isHost ? 4.5 : Math.min(5, 3.5 + (org.healthScore || 80) / 100);
   312→   312→      const trustScore = computeTrustScore(complianceScore, paymentReliabilityScore, rating);
   313→   313→
   314→   314→      await db.networkNode.create({
   315→   315→        data: {
   316→   316→          organizationId: org.id,
   317→   317→          nodeType,
   318→   318→          legalName: org.legalName || org.name,
   319→   319→          tradeName: org.name,
   320→   320→          gstin: null,
   321→   321→          email: null,
   322→   322→          website: org.domain || null,
   323→   323→          country: org.country,
   324→   324→          state: null,
   325→   325→          city: null,
   326→   326→          industry: org.industry,
   327→   327→          sector: 'services',
   328→   328→          employeeCount: org.employeeCount,
   329→   329→          annualRevenue: org.monthlyRevenue * 12,
   330→   330→          yearsInBusiness: 5,
   331→   331→          trustScore,
   332→   332→          complianceScore,
   333→   333→          paymentReliabilityScore,
   334→   334→          supplierRating: rating,
   335→   335→          customerRating: rating,
   336→   336→          aiConfidenceScore: isHost ? 85 : 70,
   337→   337→          riskLevel: computeRiskLevel(trustScore),
   338→   338→          verified: isHost,
   339→   339→          verificationDate: isHost ? new Date() : null,
   340→   340→          verificationSource: isHost ? 'self' : null,
   341→   341→          totalConnections: 0,
   342→   342→          totalTransactions: 0,
   343→   343→          totalTransactionValue: 0,

   345→   345→        },

   349→   349→    // ── 2. Create NetworkNodes for every REAL Client (customer type) ──

   351→   351→      take: 50,
   352→   352→    });
   353→   353→    // Derive a per-client filing count from real GSTRFiling rows (one query)
   354→   354→    const clientFilingCounts = await db.gSTRFiling.groupBy({
   355→   355→      by: ['clientId'],
   356→   356→      _count: { id: true },
   357→   357→    });
   358→   358→    const filingCountMap = new Map<string, number>(
   359→   359→      clientFilingCounts.map((c) => [c.clientId, c._count.id]),
   360→   360→    );

   365→   365→      const rating = Math.min(5, 3.5 + (c.healthScore || 0) / 100);
   366→   366→      const trustScore = computeTrustScore(complianceScore, paymentReliabilityScore, rating);
   367→   367→
   368→   368→      await db.networkNode.create({
   369→   369→        data: {
   370→   370→          organizationId: null,
   371→   371→          nodeType: 'customer',
   372→   372→          legalName: c.legalName || c.tradeName || 'Unknown Client',
   373→   373→          tradeName: c.tradeName || null,
   374→   374→          gstin: c.gstin || null,
   375→   375→          country: 'IN',
   376→   376→          state: c.state || null,
   377→   377→          city: null,
   378→   378→          industry: 'Services',
   379→   379→          sector: 'services',
   380→   380→          employeeCount: 0,
   381→   381→          annualRevenue: 0,
   382→   382→          yearsInBusiness: 0,
   383→   383→          trustScore,
   384→   384→          complianceScore,
   385→   385→          paymentReliabilityScore,
   386→   386→          supplierRating: 4.0,
   387→   387→          customerRating: rating,
   388→   388→          aiConfidenceScore: 68,
   389→   389→          riskLevel: computeRiskLevel(trustScore),
   390→   390→          verified: !!c.gstin,
   391→   391→          verificationDate: c.gstin ? new Date() : null,
   392→   392→          verificationSource: c.gstin ? 'gst' : null,
   393→   393→          totalConnections: 0,
   394→   394→          totalTransactions: 0,
   395→   395→          totalTransactionValue: 0,
   396→   396→          metadata: JSON.stringify({ source: 'real_client', clientId: c.id, healthScore: c.healthScore }),
   397→   397→        },
   398→   398→      });
   399→   399→    }
   400→   400→
   401→   401→    // ── 3. Add canonical external nodes (suppliers / banks / government / etc.) ──
   402→   402→    for (const ext of CANONICAL_EXTERNAL_NODES) {
   403→   403→      const trustScore = computeTrustScore(ext.complianceScore, ext.paymentReliabilityScore, ext.supplierRating);
   404→   404→      await db.networkNode.create({
   405→   405→        data: {
   406→   406→          organizationId: null,
   407→   407→          nodeType: ext.nodeType,
   408→   408→          legalName: ext.legalName,
   409→   409→          tradeName: ext.tradeName,
   410→   410→          gstin: null,
   411→   411→          email: ext.email,
   412→   412→          phone: ext.phone,
   413→   413→          website: ext.website,
   414→   414→          country: ext.country,
   415→   415→          state: ext.state,
   416→   416→          city: ext.city,
   417→   417→          industry: ext.industry,
   418→   418→          sector: ext.sector,
   419→   419→          employeeCount: ext.employeeCount,
   420→   420→          annualRevenue: ext.annualRevenue,
   421→   421→          yearsInBusiness: ext.yearsInBusiness,
   422→   422→          trustScore,
   423→   423→          complianceScore: ext.complianceScore,
   424→   424→          paymentReliabilityScore: ext.paymentReliabilityScore,
   425→   425→          supplierRating: ext.supplierRating,
   426→   426→          customerRating: ext.customerRating,
   427→   427→          aiConfidenceScore: Math.min(95, ext.complianceScore - 5),
   428→   428→          riskLevel: computeRiskLevel(trustScore),
   429→   429→          verified: ext.verified,
   430→   430→          verificationDate: new Date(),
   431→   431→          verificationSource: ext.verificationSource,
   432→   432→          totalConnections: 0,
   433→   433→          totalTransactions: 0,
   434→   434→          totalTransactionValue: 0,
   435→   435→          metadata: JSON.stringify({ canonical: true }),
   436→   436→        },
   437→   437→      });
   438→   438→    }
   439→   439→
   440→   440→    // ── 4. Build edges from the host org to its real clients (customer relationships) ──
   441→   441→    const orgNodes = await db.networkNode.findMany({
   442→   442→      where: { nodeType: 'organization' },
   443→   443→      orderBy: { createdAt: 'asc' },
   444→   444→    });
   445→   445→    const hostNode = orgNodes[0];
   446→   446→    if (hostNode) {
   447→   447→      const customerNodes = await db.networkNode.findMany({
   448→   448→        where: { nodeType: 'customer' },
   449→   449→      });
   450→   450→      for (const cn of customerNodes) {
   451→   451→        await db.networkEdge.create({
   452→   452→          data: {
   453→   453→            fromNodeId: hostNode.id,
   454→   454→            toNodeId: cn.id,
   455→   455→            relationshipType: 'customer',
   456→   456→            status: 'active',
   457→   457→            strength: 50 + (cn.trustScore % 40),
   458→   458→            totalTransactionValue: 0,
   459→   459→            transactionCount: 0,
   460→   460→            startedAt: new Date(),
   461→   461→          },
   462→   462→        }).catch(() => { /* unique constraint guard */ });
   463→   463→      }
   464→   464→
   465→   465→      // ── 5. Build edges from host org to suppliers, banks, government, accountants, auditors, partners ──
   466→   466→      const externalTypes: Array<{ type: string; rel: string }> = [
   467→   467→        { type: 'supplier', rel: 'supplier' },
   468→   468→        { type: 'bank', rel: 'bank' },
   469→   469→        { type: 'government', rel: 'government' },
   470→   470→        { type: 'investor', rel: 'investor' },
   471→   471→        { type: 'accountant', rel: 'accountant' },
   472→   472→        { type: 'auditor', rel: 'auditor' },
   473→   473→        { type: 'logistics', rel: 'logistics' },
   474→   474→        { type: 'partner', rel: 'partner' },
   475→   475→      ];
   476→   476→      for (const { type, rel } of externalTypes) {
   477→   477→        const extNodes = await db.networkNode.findMany({ where: { nodeType: type } });
   478→   478→        for (const en of extNodes) {
   479→   479→          await db.networkEdge.create({
   480→   480→            data: {
   481→   481→              fromNodeId: hostNode.id,
   482→   482→              toNodeId: en.id,
   483→   483→              relationshipType: rel,
   484→   484→              status: 'active',
   485→   485→              strength: 60 + (en.trustScore % 35),
   486→   486→              totalTransactionValue: 0,
   487→   487→              transactionCount: 0,
   488→   488→              startedAt: new Date(),
   489→   489→            },
   490→   490→          }).catch(() => { /* unique constraint guard */ });
   491→   491→        }
   492→   492→      }
   493→   493→    }
   494→   494→
   495→   495→    // ── 6. Derive REAL transactions from actual invoices ──
   496→   496→    // For each real Invoice, create a NetworkTransaction from the host org to the matching client node.
   497→   497→    if (hostNode) {
   498→   498→      const realInvoices = await db.invoice.findMany({
   499→   499→        take: 200,
   500→   500→        orderBy: { createdAt: 'desc' },
   501→   501→        include: { client: { select: { id: true, tradeName: true, legalName: true } } },
   502→   502→      });
   503→   503→      for (const inv of realInvoices) {
   504→   504→        // Match by buyerName OR by the related client's tradeName/legalName
   505→   505→        const matchName = inv.buyerName || inv.client?.tradeName || inv.client?.legalName;
   506→   506→        if (!matchName) continue;
   507→   507→        const clientNode = await db.networkNode.findFirst({
   508→   508→          where: {
   509→   509→            nodeType: 'customer',
   510→   510→            OR: [
   511→   511→              { legalName: matchName },
   512→   512→              { tradeName: matchName },
   513→   513→            ],
   514→   514→          },
   515→   515→          select: { id: true },
   516→   516→        });
   517→   517→        if (clientNode) {
   518→   518→          const invNumber = inv.invoiceNumber || `INV-${inv.id.slice(-6)}`;
   519→   519→          const amount = Number(inv.totalAmount || inv.taxableValue || 0);
   520→   520→          await db.networkTransaction.create({
   521→   521→            data: {
   522→   522→              transactionNumber: `NTXN-${invNumber}`,
   523→   523→              fromNodeId: hostNode.id,
   524→   524→              toNodeId: clientNode.id,
   525→   525→              type: 'invoice',
   526→   526→              reference: invNumber,
   527→   527→              amount,
   528→   528→              currency: 'INR',
   529→   529→              status: inv.status === 'paid' ? 'completed' : 'pending',
   530→   530→              metadata: JSON.stringify({ source: 'real_invoice', invoiceId: inv.id }),
   531→   531→            },
   532→   532→          }).catch(() => { /* unique-constraint guard */ });
   533→   533→        }
   534→   534→      }
   535→   535→    }
   536→   536→
   537→   537→    // ── 7. Update node totals based on real transactions ──
   538→   538→    if (hostNode) {
   539→   539→      const txnAgg = await db.networkTransaction.groupBy({
   540→   540→        by: ['fromNodeId'],
   541→   541→        _count: { id: true },
   542→   542→        _sum: { amount: true },
   543→   543→      });
   544→   544→      for (const agg of txnAgg) {
   545→   545→        await db.networkNode.update({
   546→   546→          where: { id: agg.fromNodeId },
   547→   547→          data: {
   548→   548→            totalTransactions: agg._count.id,
   549→   549→            totalTransactionValue: agg._sum.amount || 0,
   550→   550→          },
   551→   551→        });
   552→   552→      }
   553→   553→    }
   554→   554→  } finally {
   555→   555→    SEED_LOCK.value = false;
   556→   556→  }
   557→   557→}

   561→   561→export async function listNodes(filter?: {
   562→   562→  nodeType?: NodeType;
   563→   563→  limit?: number;
   564→   564→  verifiedOnly?: boolean;
   565→   565→  search?: string;
   566→   566→}): Promise<NetworkNode[]> {
   567→   567→  const where: Record<string, unknown> = {};
   568→   568→  if (filter?.nodeType) where.nodeType = filter.nodeType;
   569→   569→  if (filter?.verifiedOnly) where.verified = true;
   570→   570→  if (filter?.search) {
   571→   571→    where.OR = [
   572→   572→      { legalName: { contains: filter.search } },
   573→   573→      { tradeName: { contains: filter.search } },
   574→   574→      { gstin: { contains: filter.search } },
   575→   575→      { city: { contains: filter.search } },
   576→   576→      { industry: { contains: filter.search } },
   577→   577→    ];
   578→   578→  }
   579→   579→  const nodes = await db.networkNode.findMany({
   580→   580→    where,
   581→   581→    orderBy: { createdAt: 'desc' },
   582→   582→    take: filter?.limit ?? 100,
   583→   583→  });
   584→   584→  return nodes.map(mapNode);
   585→   585→}
