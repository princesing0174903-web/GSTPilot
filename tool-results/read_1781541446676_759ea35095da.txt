   103→function NotificationBell() {
   104→  const { data: unreadNotifs } = useFireUnreadNotifications()
   105→  const { data: allNotifs } = useFireNotifications()
   106→  const unreadCount = unreadNotifs.length
   107→  const [open, setOpen] = useState(false)
   108→
   109→  const handleMarkRead = async (notifId: string) => {
   110→    try { await markNotificationRead(notifId) } catch {}
   111→  }
   112→
   113→  const handleMarkAllRead = async () => {
   114→    try {
   115→      for (const n of unreadNotifs) {
   116→        await markNotificationRead(n.notifId || n.id)
   117→      }
   118→    } catch {}
   119→  }
   120→
   121→  const priorityColor = (p: string) => {
   122→    switch (p) {
   123→      case 'urgent': return 'text-red-600 bg-red-50 border-red-200'
   124→      case 'high': return 'text-orange-600 bg-orange-50 border-orange-200'
   125→      case 'normal': return 'text-blue-600 bg-blue-50 border-blue-200'
   126→      default: return 'text-slate-600 bg-slate-50 border-slate-200'
   127→    }
   128→  }
   129→
   130→  const priorityBadge = (p: string) => {
   131→    switch (p) {
   132→      case 'urgent': return 'bg-red-500'
   133→      case 'high': return 'bg-orange-500'
   134→      case 'normal': return 'bg-blue-500'
   135→      default: return 'bg-slate-400'
   136→    }
   137→  }
   138→
   139→  const typeIcon = (t: string) => {
   140→    switch (t) {
   141→      case 'filing_due': case 'filing_overdue': return <Clock className="h-3.5 w-3.5" />
   142→      case 'mismatch_found': return <AlertTriangle className="h-3.5 w-3.5" />
   143→      case 'filing_completed': return <Check className="h-3.5 w-3.5" />
   144→      case 'extraction_complete': return <FileText className="h-3.5 w-3.5" />
   145→      case 'issue_detected': return <AlertTriangle className="h-3.5 w-3.5" />
   146→      case 'ai_insight': return <Sparkles className="h-3.5 w-3.5" />
   147→      default: return <Bell className="h-3.5 w-3.5" />
   148→    }
   149→  }
   150→
   151→  const displayed = (allNotifs || []).slice(0, 15)
   152→
   153→  return (
   154→    <Popover open={open} onOpenChange={setOpen}>
   155→      <PopoverTrigger asChild>
   156→        <Button variant="ghost" size="sm" className="relative h-8 w-8 p-0 rounded-lg hover:bg-accent">
   157→          {unreadCount > 0 ? (
   158→            <BellRing className="h-4 w-4 text-amber-600" />
   159→          ) : (
   160→            <Bell className="h-4 w-4 text-muted-foreground" />
   161→          )}
   162→          {unreadCount > 0 && (
   163→            <span className="absolute -top-0.5 -right-0.5 flex h-4 min-w-4 items-center justify-center rounded-full bg-red-500 px-1 text-[9px] font-bold text-white">
   164→              {unreadCount > 9 ? '9+' : unreadCount}
   165→            </span>
   166→          )}
   167→        </Button>
   168→      </PopoverTrigger>
   169→      <PopoverContent align="end" className="w-80 sm:w-96 p-0 gap-0" sideOffset={8}>
   170→        <div className="flex items-center justify-between px-4 py-3 border-b">
   171→          <div className="flex items-center gap-2">
   172→            <h3 className="text-sm font-semibold">Notifications</h3>
   173→            {unreadCount > 0 && (
   174→              <Badge variant="secondary" className="h-5 px-1.5 text-[10px] bg-red-100 text-red-700">
   175→                {unreadCount} new
   176→              </Badge>
   177→            )}
   178→          </div>
   179→          {unreadCount > 0 && (
   180→            <Button variant="ghost" size="sm" onClick={handleMarkAllRead} className="h-6 text-[10px] text-emerald-600 hover:text-emerald-700 px-2">
   181→              Mark all read

   185→        <ScrollArea className="max-h-80">
   186→          {displayed.length === 0 ? (
   187→            <div className="flex flex-col items-center justify-center py-8 px-4 text-center">
   188→              <Bell className="h-8 w-8 text-muted-foreground/30 mb-2" />

   190→              <p className="text-xs text-muted-foreground/60 mt-1">Notifications will appear here when returns are due, mismatches are found, or documents are processed.</p>

   223→          <div className="border-t px-4 py-2">
   224→            <Button
   225→              variant="ghost"
   226→              size="sm"
   227→              className="w-full text-xs text-emerald-600 hover:text-emerald-700"
   228→              onClick={() => {
   229→                setOpen(false)
   230→              }}

   232→              View All Notifications

   265→// KEYBOARD SHORTCUTS HOOK
   266→// ═══════════════════════════════════════════════════════════════════════════════

   268→function useKeyboardShortcuts() {
   269→  const { setCurrentView, setCommandPaletteOpen, commandPaletteOpen } = useApp()
   270→  const gPressed = useRef(false)
   271→  const gTimer = useRef<NodeJS.Timeout | null>(null)
   272→
   273→  useEffect(() => {
   274→    const handler = (e: KeyboardEvent) => {
   275→      // Ctrl+K → Toggle Command Palette
   276→      if ((e.metaKey || e.ctrlKey) && e.key === 'k') {
   277→        e.preventDefault()
   278→        setCommandPaletteOpen(!commandPaletteOpen)
   279→        return
   280→      }
   281→
   282→      // G prefix shortcuts (G+D, G+C, G+I, G+R)
   283→      if (e.key === 'g' && !e.metaKey && !e.ctrlKey && !e.altKey && !(e.target instanceof HTMLInputElement || e.target instanceof HTMLTextAreaElement)) {
   284→        if (gPressed.current) return
   285→        gPressed.current = true
   286→        gTimer.current = setTimeout(() => { gPressed.current = false }, 500)
   287→        return
   288→      }
   289→
   290→      if (gPressed.current && !e.metaKey && !e.ctrlKey && !e.altKey) {
   291→        gPressed.current = false
   292→        if (gTimer.current) clearTimeout(gTimer.current)
   293→
   294→        switch (e.key) {
   295→          case 'd': setCurrentView('dashboard'); break
   296→          case 'c': setCurrentView('clients'); break
   297→          case 'i': setCurrentView('invoices'); break
   298→          case 'r': setCurrentView('returns'); break
   299→        }
   300→      }
   301→    }
   302→
   303→    window.addEventListener('keydown', handler)
   304→    return () => window.removeEventListener('keydown', handler)
   305→  }, [setCurrentView, setCommandPaletteOpen, commandPaletteOpen])
   306→}

   498→      if (!user) return
   499→
   500→      await setDoc(doc(db, 'users', user.id), {
   501→        uid: user.id, fullName: data.fullName, email: data.email, phone: data.phone,
   502→        ageGroup: data.ageGroup, profession: data.profession, experience: data.experience,
   503→        firmId: firmId || null, firmName: data.firmName, onboardingCompleted: true,
   504→        clientCount: data.clientCount, monthlyReturns: data.monthlyReturns,
   505→        gstServices: data.gstServices, painPoints: data.painPoints,
   506→        referralSource: data.referralSource, trialReasons: data.trialReasons,
   507→        wantsUpdates: data.wantsUpdates, updatedAt: serverTimestamp(),
   508→      }, { merge: true })
   509→
   510→      await setDoc(doc(db, 'onboarding', user.id), {
   511→        ...data, firmId: firmId || null, completedAt: serverTimestamp(),
   512→      })
   513→    } catch (error) {
   514→      console.warn('[Onboarding] Background Firestore save failed:', error)
   515→    }
   516→  }
   517→
   518→  const handleOnboardingComplete = (
   519→    data: import('@/components/onboarding/OnboardingFlow').OnboardingData,
   520→    destination?: import('@/components/onboarding/OnboardingFlow').OnboardingDestination
   521→  ) => {
   522→    setSaveError(null)
   523→    if (!user) { setSaveError('No authenticated user found. Please sign in again.'); return }
   524→
   525→    markOnboardingComplete(undefined, data.firmName)
   526→    setCurrentView(destination || 'dashboard')
   527→    setCurrentScreen('app')
   528→
   529→    ;(async () => {
   530→      try {
   531→        const { collection, addDoc, serverTimestamp } = await import('firebase/firestore')
   532→        const { db } = await import('@/lib/firebase')
   533→        let firmId = ''
   534→        try {
   535→          const firmRef = await addDoc(collection(db, 'firms'), {
   536→            ownerId: user.id, firmName: data.firmName, gstin: data.gstin || null,
   537→            state: data.state, stateCode: data.stateCode,
   538→            organizationType: data.organizationType,
   539→            icaiMembershipNo: data.icaiMembershipNo || null,
   540→            officeAddress: data.officeAddress || null, createdAt: serverTimestamp(),
   541→          })
   542→          firmId = firmRef.id
   543→          markOnboardingComplete(firmId, data.firmName)
   544→        } catch (firmError) {
   545→          console.warn('[Onboarding] Firm creation failed:', firmError)
   546→        }
   547→        await saveOnboardingToFirestore(data, firmId)
   548→      } catch (error) {
   549→        console.warn('[Onboarding] Background save error:', error)
   550→      }
   551→    })()
   552→  }
   553→
   554→  const handleSkip = () => {
   555→    setSaveError(null)
   556→    if (!user) { setSaveError('No authenticated user found.'); return }
   557→
   558→    markOnboardingComplete()
   559→    setCurrentView('dashboard')
   560→    setCurrentScreen('app')
   561→
   562→    ;(async () => {
   563→      try {
   564→        const { doc, setDoc, serverTimestamp } = await import('firebase/firestore')
   565→        const { db } = await import('@/lib/firebase')
   566→        await setDoc(doc(db, 'users', user.id), {
   567→          onboardingCompleted: true, updatedAt: serverTimestamp(),
   568→        }, { merge: true })
   569→      } catch (error) {
   570→        console.warn('[Onboarding] Background skip save failed:', error)
   571→      }
   572→    })()
   573→  }
   574→
   575→  return (
   576→    <OnboardingFlow
   577→      onComplete={handleOnboardingComplete}
   578→      onSkip={handleSkip}
   579→      userEmail={user?.email}
   580→      userName={user?.name}
   581→      error={saveError}
   582→      onDismissError={() => setSaveError(null)}
   583→    />
   584→  )
   585→}

   591→function AppRouter() {
   592→  const { currentScreen, setCurrentScreen } = useApp()
   593→  const { isAuthenticated, isInitializing, needsOnboarding, needsEmailVerification } = useAuth()
   594→
   595→  useEffect(() => {
   596→    if (isInitializing) return
   597→    if (isAuthenticated && !needsOnboarding && currentScreen !== 'app') {
   598→      setCurrentScreen('app')
   599→    }
   600→  }, [isAuthenticated, isInitializing, currentScreen, setCurrentScreen, needsOnboarding])
   601→
   602→  useEffect(() => {
   603→    if (isInitializing) return
   604→    if (!isAuthenticated && currentScreen === 'app') {
   605→      setCurrentScreen('landing')
   606→    }
   607→  }, [isAuthenticated, isInitializing, currentScreen, setCurrentScreen])
   608→
   609→  const handleGetStarted = () => setCurrentScreen('login')
   610→  const handleBookDemo = () => setCurrentScreen('login')
   611→  const handleBackToLanding = () => setCurrentScreen('landing')
   612→
   613→  if (isInitializing && !isAuthenticated) {
   614→    return (
   615→      <div className="min-h-screen flex items-center justify-center bg-white">
   616→        <div className="flex flex-col items-center gap-3">
   617→          <div className="flex h-10 w-10 items-center justify-center rounded-lg bg-gradient-to-br from-emerald-500 to-emerald-600 shadow-lg shadow-emerald-600/20">
   618→            <Zap className="h-6 w-6 text-white animate-pulse" />
   619→          </div>
   620→          <span className="text-sm text-slate-500 font-medium">Loading GSTPilot...</span>
   621→        </div>
   622→      </div>
   623→    )
   624→  }
   625→
   626→  if (isAuthenticated && needsOnboarding) {
   627→    return <OnboardingScreen />
   628→  }
   629→
   630→  if (currentScreen === 'app' && isAuthenticated) {
   631→    return (
   632→      <div className="min-h-screen flex flex-col">
   633→        {needsEmailVerification && <EmailVerificationBanner />}
   634→        <div className="flex-1 flex">
   635→          <DashboardContent />
   636→        </div>
   637→      </div>
   638→    )
   639→  }
   640→
   641→  if (currentScreen === 'login') {
   642→    return <LoginPage onBack={handleBackToLanding} onGetStarted={handleGetStarted} />
   643→  }
   644→
   645→  return <LandingPage onGetStarted={handleGetStarted} onBookDemo={handleBookDemo} />
   646→}
   647→
   648→export default function Home() {
   649→  return <AppRouter />
   650→}
