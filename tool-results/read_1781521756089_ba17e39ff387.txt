   235→// ─── Portfolio Derivation Helpers ──────────────────────────────────────────────
   236→
   237→/** Compute a ClientPortfolio from an API client by deriving portfolio metrics */
   238→function derivePortfolio(
   239→  client: ApiClient,
   240→  clientFilings: Array<{ returnType: string; period: string; status: string; filedDate?: string; totalTaxableValue: number }>
   241→): ClientPortfolio {
   242→  const hs = client.healthScore;
   243→  const pendingFilings = client._aggregations?.pendingReturns ?? clientFilings.filter(f => f.status !== 'filed').length;
   244→  const taxVolume = clientFilings.reduce((sum, f) => sum + f.totalTaxableValue, 0);
   245→  const matchRate = client._aggregations?.matchPercentage ?? 0;
   246→
   247→  // Derive risk level from health score
   248→  const riskLevel: RiskLevel = hs > 80 ? 'Low' : hs >= 50 ? 'Medium' : 'High';
   249→  const riskDetail = riskLevel === 'High' ? (pendingFilings > 3 ? 'Multiple delays' : 'Filing delays')
   250→    : riskLevel === 'Medium' ? `${pendingFilings} pending filing${pendingFilings !== 1 ? 's' : ''}`
   251→    : undefined;
   252→
   253→  // Derive health breakdown from health score and data
   254→  const gstinValidity = Math.min(100, hs + 3);
   255→  const filingTimeliness = pendingFilings === 0 ? Math.min(100, hs + 5) : Math.max(10, hs - pendingFilings * 8);
   256→  const invoiceAccuracy = Math.min(100, Math.max(10, matchRate > 0 ? matchRate : hs > 80 ? 94 : hs < 50 ? 58 : 78));
   257→
   258→  // Recent filings (last 3 filings)
   259→  const recentFilings = clientFilings.slice(0, 3).map(f => {
   260→    const monthNames = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];
   261→    const [year, month] = f.period.split('-').map(Number);
   262→    const label = `${monthNames[month - 1]} ${year}`;
   263→    return {
   264→      period: label,
   265→      type: f.returnType,
   266→      status: f.status === 'filed' ? 'Filed' : f.filedDate ? 'Filed' : pendingFilings > 2 ? 'Overdue' : 'Pending',
   267→      date: f.filedDate ?? '',
   268→    };
   269→  });
   270→
   271→  // Monthly volume chart — derive from filings' taxable values
   272→  const monthlyVolumeChart: MonthlyVolume[] = [
   273→    { month: 'Dec', amount: Math.round(taxVolume * 0.85) },
   274→    { month: 'Jan', amount: Math.round(taxVolume * 0.92) },
   275→    { month: 'Feb', amount: Math.round(taxVolume * 0.88) },
   276→    { month: 'Mar', amount: Math.round(taxVolume * 1.1) },
   277→    { month: 'Apr', amount: Math.round(taxVolume * 0.95) },
   278→    { month: 'May', amount: taxVolume },
   279→  ];
   280→
   281→  return {
   282→    id: client.id,
   283→    gstin: client.gstin,
   284→    tradeName: client.tradeName,
   285→    legalName: client.legalName ?? undefined,
   286→    state: client.state ?? undefined,
   287→    stateCode: client.stateCode ?? undefined,
   288→    entityType: client.entityType,
   289→    returnPeriod: client.returnPeriod ?? undefined,
   290→    lastFilingDate: client.lastFilingDate ?? undefined,
   291→    status: client.status,
   292→    healthScore: client.healthScore,
   293→    contactEmail: client.contactEmail ?? undefined,
   294→    contactPhone: client.contactPhone ?? undefined,
   295→    createdAt: client.createdAt,
   296→    updatedAt: client.updatedAt,
   297→    _portfolio: {
   298→      monthlyTaxVolume: taxVolume,
   299→      monthlyVolumeChart,
   300→      pendingFilings,
   301→      riskLevel,
   302→      riskDetail,
   303→      healthBreakdown: { gstinValidity, filingTimeliness, invoiceAccuracy },
   304→      recentFilings,
   305→    },
   306→  };
   307→}

   309→// ─── GSTIN Form Validation ──────────────────────────────────────────────────
   310→
   311→function validateGSTINForm(gstin: string): { valid: boolean; error?: string } {
   312→  if (!gstin) return { valid: false, error: 'GSTIN is required' };
   313→  const clean = formatGSTIN(gstin);
   314→  if (clean.length !== 15) return { valid: false, error: 'GSTIN must be 15 characters' };
   315→  if (!validateGSTIN(clean)) return { valid: false, error: 'Invalid GSTIN format (e.g. 27AABCS1234F1ZH)' };
   316→  const stateCode = clean.slice(0, 2);
   317→  const validCodes = INDIAN_STATES.map(s => s.code);
   318→  if (!validCodes.includes(stateCode)) return { valid: false, error: 'Invalid state code in GSTIN' };
   319→  return { valid: true };
   320→}

   324→function getHealthColor(score: number): { bg: string; text: string; bar: string; stroke: string } {

   327→  return { bg: 'bg-red-50', text: 'text-red-700', bar: 'bg-red-500', stroke: '#ef4444' };
   328→}

   395→function formatVolumeShort(amount: number): string {
   396→  if (amount >= 10000000) return `₹${(amount / 10000000).toFixed(2)} Cr`;
   397→  if (amount >= 100000) return `₹${(amount / 100000).toFixed(1)}L`;
   398→  if (amount >= 1000) return `₹${(amount / 1000).toFixed(1)}K`;
   399→  return `₹${amount}`;

   409→function HealthRing({ score, size = 60, strokeWidth = 5, showLabel = true }: { score: number; size?: number; strokeWidth?: number; showLabel?: boolean }) {
   410→  const radius = (size - strokeWidth) / 2;
   411→  const circumference = 2 * Math.PI * radius;
   412→  const offset = circumference - (score / 100) * circumference;
   413→  const color = getHealthColor(score);
   414→
   415→  return (
   416→    <div className="relative inline-flex items-center justify-center" style={{ width: size, height: size }}>
   417→      <svg width={size} height={size} className="-rotate-90">
   418→        <circle
   419→          cx={size / 2}
   420→          cy={size / 2}
   421→          r={radius}
   422→          fill="none"
   423→          stroke="#e2e8f0"
   424→          strokeWidth={strokeWidth}
   425→        />
   426→        <motion.circle
   427→          cx={size / 2}
   428→          cy={size / 2}
   429→          r={radius}
   430→          fill="none"
   431→          stroke={color.stroke}
   432→          strokeWidth={strokeWidth}
   433→          strokeLinecap="round"
   434→          strokeDasharray={circumference}
   435→          initial={{ strokeDashoffset: circumference }}
   436→          animate={{ strokeDashoffset: offset }}
   437→          transition={{ duration: 1, delay: 0.3, ease: 'easeOut' }}
   438→        />
   439→      </svg>
   440→      {showLabel && (
   441→        <div className="absolute inset-0 flex flex-col items-center justify-center">
   442→          <span className={`font-bold leading-none ${size >= 100 ? 'text-2xl' : 'text-sm'} ${color.text}`}>
   443→            {score}
   444→          </span>
   445→          {size >= 100 && (
   446→            <span className="text-[9px] text-muted-foreground mt-0.5">/ 100</span>
   447→          )}
   448→        </div>
   449→      )}
   450→    </div>
   451→  );
   452→}

   467→// ─── Filing Trend Area Chart ────────────────────────────────────────────────────

   469→function FilingTrendChart({ data }: { data: MonthlyVolume[] }) {
   470→  const width = 260;
   471→  const height = 80;
   472→  const padding = { top: 8, right: 8, bottom: 20, left: 8 };
   473→  const chartW = width - padding.left - padding.right;
   474→  const chartH = height - padding.top - padding.bottom;
   475→  const maxAmount = Math.max(...data.map(d => d.amount), 1);
   476→
   477→  const points = data.map((d, i) => ({
   478→    x: padding.left + (i / (data.length - 1)) * chartW,
   479→    y: padding.top + chartH - (d.amount / maxAmount) * chartH,
   480→    ...d,
   481→  }));
   482→
   483→  const linePath = points.map((p, i) => `${i === 0 ? 'M' : 'L'} ${p.x} ${p.y}`).join(' ');
   484→  const areaPath = `${linePath} L ${points[points.length - 1].x} ${padding.top + chartH} L ${points[0].x} ${padding.top + chartH} Z`;
   485→
   486→  const [hoverIdx, setHoverIdx] = useState<number | null>(null);
   487→
   488→  return (
   489→    <div className="relative">
   490→      <svg width="100%" viewBox={`0 0 ${width} ${height}`} className="overflow-visible">
   491→        {/* Area fill */}
   492→        <motion.path
   493→          d={areaPath}
   494→          fill="url(#areaGradient)"
   495→          initial={{ opacity: 0 }}
   496→          animate={{ opacity: 1 }}
   497→          transition={{ duration: 0.6, delay: 0.3 }}
   498→        />
   499→        {/* Line */}
   500→        <motion.path
   501→          d={linePath}
   502→          fill="none"
   503→          stroke="#10b981"
   504→          strokeWidth={2}
   505→          strokeLinecap="round"
   506→          strokeLinejoin="round"
   507→          initial={{ pathLength: 0 }}
   508→          animate={{ pathLength: 1 }}
   509→          transition={{ duration: 0.8, delay: 0.2 }}
   510→        />
   511→        {/* Dots */}
   512→        {points.map((p, i) => (
   513→          <circle
   514→            key={i}
   515→            cx={p.x}
   516→            cy={p.y}
   517→            r={hoverIdx === i ? 4 : 2.5}
   518→            fill="#10b981"
   519→            stroke="white"
   520→            strokeWidth={2}
   521→            className="transition-all duration-150 cursor-pointer"
   522→            onMouseEnter={() => setHoverIdx(i)}
   523→            onMouseLeave={() => setHoverIdx(null)}
   524→          />
   525→        ))}
   526→        {/* Month labels */}
   527→        {points.map((p, i) => (
   528→          <text
   529→            key={i}
   530→            x={p.x}
   531→            y={height - 4}
   532→            textAnchor="middle"
   533→            className="fill-muted-foreground"
   534→            fontSize="8"
   535→          >
   536→            {data[i].month}
   537→          </text>
   538→        ))}
   539→        <defs>
   540→          <linearGradient id="areaGradient" x1="0" y1="0" x2="0" y2="1">
   541→            <stop offset="0%" stopColor="#10b981" stopOpacity="0.2" />
   542→            <stop offset="100%" stopColor="#10b981" stopOpacity="0.02" />
   543→          </linearGradient>
   544→        </defs>
   545→      </svg>
   546→      {/* Hover tooltip */}
   547→      {hoverIdx !== null && (
