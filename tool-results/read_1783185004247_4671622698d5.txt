    88→// ═══════════════════════════════════════════════════════════════════════════════
    89→// LOCAL STORAGE HELPERS
    90→// ═══════════════════════════════════════════════════════════════════════════════
    91→
    92→function loadRecent(): RecentItem[] {
    93→  try {
    94→    const raw = localStorage.getItem(RECENT_STORAGE_KEY);
    95→    return raw ? JSON.parse(raw) : [];
    96→  } catch {
    97→    return [];
    98→  }
    99→}
   100→
   101→function saveRecent(items: RecentItem[]) {
   102→  try {
   103→    localStorage.setItem(RECENT_STORAGE_KEY, JSON.stringify(items.slice(0, MAX_RECENT)));
   104→  } catch {
   105→    // ignore quota errors
   106→  }
   107→}
   108→
   109→function loadFavorites(): FavoriteItem[] {
   110→  try {
   111→    const raw = localStorage.getItem(FAVORITES_STORAGE_KEY);
   112→    return raw ? JSON.parse(raw) : [];
   113→  } catch {
   114→    return [];
   115→  }
   116→}
   117→
   118→function saveFavorites(items: FavoriteItem[]) {
   119→  try {
   120→    localStorage.setItem(FAVORITES_STORAGE_KEY, JSON.stringify(items));
   121→  } catch {
   122→    // ignore quota errors
   123→  }
   124→}

   862→
   863→interface CommandItemRowProps {
   864→  icon: React.ElementType;
   865→  label: string;
   866→  description?: string;
   867→  shortcut?: string;
   868→  isFavorite?: boolean;
   869→  onToggleFavorite?: () => void;
   870→  onSelect: () => void;
   871→}
   872→
   873→function CommandItemRow({
   874→  icon: Icon,
   875→  label,
   876→  description,
   877→  shortcut,
   878→  isFavorite,
   879→  onToggleFavorite,
   880→  onSelect,
   881→}: CommandItemRowProps) {
   882→  const [hovered, setHovered] = React.useState(false);
   883→
   884→  return (
   885→    <motion.div
   886→      className="group relative flex items-center gap-3 rounded-lg px-3 py-2 cursor-pointer transition-colors duration-150 hover:bg-white/[0.06]"
   887→      onClick={onSelect}
   888→      onMouseEnter={() => setHovered(true)}
   889→      onMouseLeave={() => setHovered(false)}
   890→      whileHover={{ x: 2 }}
   891→      transition={{ duration: 0.1 }}
   892→    >
   893→      {/* Active left accent bar */}
   894→      <span className="absolute left-0 top-1/2 h-5 w-[2px] -translate-y-1/2 rounded-full bg-[#3B82F6] opacity-0 group-hover:opacity-100 transition-opacity duration-150" />
   895→      <div className="flex h-7 w-7 shrink-0 items-center justify-center rounded-md bg-white/[0.04] transition-colors group-hover:bg-[#3B82F6]/10">
   896→        <Icon className="h-3.5 w-3.5 text-white/50 transition-colors group-hover:text-[#3B82F6]" />
   897→      </div>
   898→      <div className="flex-1 min-w-0">
   899→        <p className="text-sm font-medium text-white truncate">{label}</p>
   900→        {description && (
   901→          <p className="text-xs text-white/45 truncate">{description}</p>
   902→        )}
   903→      </div>
   904→      <div className="flex items-center gap-2 shrink-0">
   905→        {/* Favorite toggle for commands */}
   906→        {onToggleFavorite && (
   907→          <button
   908→            onClick={(e) => {
   909→              e.stopPropagation();
   910→              onToggleFavorite();
   911→            }}
   912→            className="opacity-0 group-hover:opacity-100 transition-opacity p-0.5 hover:bg-white/[0.08] rounded"

   916→              className={`h-3 w-3 ${

   921→            />
   922→          </button>
   923→        )}

   925→          <kbd className="badge-premium hidden sm:inline-flex font-mono text-[10px]">
   926→            {shortcut}
   927→          </kbd>

   930→      </div>
   931→    </motion.div>
   932→  );
   933→}
