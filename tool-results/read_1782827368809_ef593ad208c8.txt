    88→// ═══════════════════════════════════════════════════════════════════════════════
    89→// LOCAL STORAGE HELPERS
    90→// ═══════════════════════════════════════════════════════════════════════════════
    91→
    92→function loadRecent(): RecentItem[] {
    93→  try {
    94→    const raw = localStorage.getItem(RECENT_STORAGE_KEY);
    95→    return raw ? JSON.parse(raw) : [];
    96→  } catch {
    97→    return [];
    98→  }
    99→}
   100→
   101→function saveRecent(items: RecentItem[]) {
   102→  try {
   103→    localStorage.setItem(RECENT_STORAGE_KEY, JSON.stringify(items.slice(0, MAX_RECENT)));
   104→  } catch {
   105→    // ignore quota errors
   106→  }
   107→}
   108→
   109→function loadFavorites(): FavoriteItem[] {
   110→  try {
   111→    const raw = localStorage.getItem(FAVORITES_STORAGE_KEY);
   112→    return raw ? JSON.parse(raw) : [];
   113→  } catch {
   114→    return [];
   115→  }
   116→}
   117→
   118→function saveFavorites(items: FavoriteItem[]) {
   119→  try {
   120→    localStorage.setItem(FAVORITES_STORAGE_KEY, JSON.stringify(items));
   121→  } catch {
   122→    // ignore quota errors
   123→  }
   124→}

   841→interface CommandItemRowProps {
   842→  icon: React.ElementType;
   843→  label: string;
   844→  description?: string;
   845→  shortcut?: string;
   846→  isFavorite?: boolean;
   847→  onToggleFavorite?: () => void;
   848→  onSelect: () => void;
   849→}
   850→
   851→function CommandItemRow({
   852→  icon: Icon,
   853→  label,
   854→  description,
   855→  shortcut,
   856→  isFavorite,
   857→  onToggleFavorite,
   858→  onSelect,
   859→}: CommandItemRowProps) {

   861→

   863→    <motion.div
   864→      className="group relative flex items-center gap-3 rounded-lg px-3 py-2 cursor-pointer transition-colors duration-150 hover:bg-white/[0.06]"
   865→      onClick={onSelect}
   866→      onMouseEnter={() => setHovered(true)}
   867→      onMouseLeave={() => setHovered(false)}
   868→      whileHover={{ x: 2 }}
   869→      transition={{ duration: 0.1 }}
   870→    >
   871→      {/* Active left accent bar */}

   873→      <div className="flex h-7 w-7 shrink-0 items-center justify-center rounded-md bg-white/[0.04] transition-colors group-hover:bg-[#3B82F6]/10">
   874→        <Icon className="h-3.5 w-3.5 text-white/50 transition-colors group-hover:text-[#3B82F6]" />
   875→      </div>
   876→      <div className="flex-1 min-w-0">
   877→        <p className="text-sm font-medium text-white truncate">{label}</p>
   878→        {description && (

   904→            {shortcut}

   908→      </div>
   909→    </motion.div>
