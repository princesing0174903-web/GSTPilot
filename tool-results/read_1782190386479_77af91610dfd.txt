    72→    72→function Reveal({
    73→    73→  children,
    74→    74→  delay = 0,
    75→    75→  className,
    76→    76→  as = 'div',
    77→    77→}: {
    78→    78→  children: React.ReactNode;
    79→    79→  delay?: number;

    81→    81→  as?: 'div' | 'section' | 'li' | 'span' | 'h2' | 'p';

    83→    83→  const MotionTag = motion[as] as typeof motion.div;
    84→    84→  return (
    85→    85→    <MotionTag
    86→    86→      className={className}
    87→    87→      variants={revealVariants}

    91→    91→      transition={{ duration: 0.7, ease: EASE, delay }}
    92→    92→    >
    93→    93→      {children}
    94→    94→    </MotionTag>
    95→    95→  );

   959→   959→function AnimatedCounter({ target, prefix = '', suffix = '', decimals = 0 }: { target: number; prefix?: string; suffix?: string; decimals?: number }) {
   960→   960→  const ref = useRef<HTMLSpanElement>(null);
   961→   961→  const inView = useInView(ref, { once: true, margin: '-60px' });
   962→   962→  const [val, setVal] = useState(0);
   963→   963→
   964→   964→  useEffect(() => {
   965→   965→    if (!inView) return;
   966→   966→    const duration = 1800;
   967→   967→    const start = performance.now();
   968→   968→    let raf = 0;
   969→   969→    const tick = (now: number) => {
   970→   970→      const t = Math.min((now - start) / duration, 1);
   971→   971→      const eased = 1 - Math.pow(1 - t, 3);
   972→   972→      setVal(target * eased);
   973→   973→      if (t < 1) raf = requestAnimationFrame(tick);
   974→   974→      else setVal(target);
   975→   975→    };
   976→   976→    raf = requestAnimationFrame(tick);
   977→   977→    return () => cancelAnimationFrame(raf);
   978→   978→  }, [inView, target]);
   979→   979→
   980→   980→  const formatted =
   981→   981→    decimals > 0
   982→   982→      ? val.toFixed(decimals)
   983→   983→      : Math.round(val).toLocaleString('en-IN');
   984→   984→
   985→   985→  return (
   986→   986→    <span ref={ref}>
   987→   987→      {prefix}
   988→   988→      {formatted}
   989→   989→      {suffix}
   990→   990→    </span>
   991→   991→  );
   992→   992→}
