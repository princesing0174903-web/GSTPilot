    62→function PaymentTrendChart({ data }: { data: { week: string; collected: number; paid: number }[] }) {
    63→  const max = Math.max(...data.map(d => Math.max(d.collected, d.paid)))
    64→  const barW = 20
    65→  const gap = 12
    66→  const h = 70
    67→  return (
    68→    <svg width={data.length * (barW * 2 + gap)} height={h + 20} className="overflow-visible">
    69→      {data.map((d, i) => {
    70→        const x = i * (barW * 2 + gap)
    71→        const hC = (d.collected / max) * (h - 8)
    72→        const hP = (d.paid / max) * (h - 8)
    73→        return (
    74→          <g key={d.week}>
    75→            <rect x={x} y={h - hC} width={barW} height={hC} rx={3} fill="#10b981" opacity={0.7} />
    76→            <rect x={x + barW + 2} y={h - hP} width={barW} height={hP} rx={3} fill="#f59e0b" opacity={0.6} />
    77→            <text x={x + barW + 1} y={h + 14} textAnchor="middle" className="text-[9px] fill-muted-foreground">{d.week}</text>
    78→          </g>
    79→        )
    80→      })}
    81→    </svg>
    82→  )

   167→function formatTxnDate(iso: string): string {
   168→  if (!iso) return '—'
   169→  const d = new Date(iso)
   170→  if (Number.isNaN(d.getTime())) return iso
   171→  const dd = String(d.getDate()).padStart(2, '0')
   172→  const mm = String(d.getMonth() + 1).padStart(2, '0')
   173→  const yyyy = d.getFullYear()
   174→  return `${dd}/${mm}/${yyyy}`
   175→}
   176→
   177→function titleCaseMode(mode: string): string {
   178→  if (!mode) return 'Bank'
   179→  const lower = mode.toLowerCase()
   180→  if (lower === 'upi') return 'UPI'
   181→  if (lower === 'bank') return 'Net Banking'
   182→  if (lower === 'card') return 'Card'
   183→  if (lower === 'cheque') return 'Cheque'
   184→  if (lower === 'cash') return 'Cash'
   185→  return mode.charAt(0).toUpperCase() + mode.slice(1)
   186→}
