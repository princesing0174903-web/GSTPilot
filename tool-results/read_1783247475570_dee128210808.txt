   128→// ─── Animated Number Hook ──────────────────────────────────────────────────
   129→function useAnimatedNumber(target: number, duration: number = 1200) {
   130→  const [current, setCurrent] = useState(0);
   131→  const ref = useRef<number | null>(null);
   132→  const startTime = useRef<number | null>(null);
   133→
   134→  useEffect(() => {
   135→    startTime.current = null;
   136→    const startValue = current;
   137→
   138→    function step(timestamp: number) {
   139→      if (!startTime.current) startTime.current = timestamp;
   140→      const elapsed = timestamp - startTime.current;
   141→      const progress = Math.min(elapsed / duration, 1);
   142→      const eased = 1 - Math.pow(1 - progress, 3);
   143→      setCurrent(Math.round(startValue + (target - startValue) * eased));
   144→      if (progress < 1) {
   145→        ref.current = requestAnimationFrame(step);
   146→      }
   147→    }
   148→
   149→    ref.current = requestAnimationFrame(step);
   150→    return () => {
   151→      if (ref.current) cancelAnimationFrame(ref.current);
   152→    };
   153→  }, [target, duration]);
   154→
   155→  return current;
   156→}

   158→// ─── Circular Progress Component ───────────────────────────────────────────
   159→function CircularProgress({
   160→  value,
   161→  size = 80,
   162→  strokeWidth = 8,
   163→  color = COLORS.emerald,
   164→}: {
   165→  value: number;
   166→  size?: number;
   167→  strokeWidth?: number;
   168→  color?: string;
   169→}) {
   170→  const radius = (size - strokeWidth) / 2;
   171→  const circumference = 2 * Math.PI * radius;
   172→  const offset = circumference - (value / 100) * circumference;
   173→  const center = size / 2;
   174→
   175→  return (
   176→    <div className="relative" style={{ width: size, height: size }}>
   177→      <svg width={size} height={size} className="transform -rotate-90">
   178→        <circle
   179→          cx={center}
   180→          cy={center}
   181→          r={radius}
   182→          fill="none"
   183→          stroke="currentColor"
   184→          strokeWidth={strokeWidth}
   185→          className="text-muted/20"
   186→        />
   187→        <defs>
   188→          <filter id="progressGlow">
   189→            <feGaussianBlur stdDeviation="2" result="coloredBlur" />
   190→            <feMerge>
   191→              <feMergeNode in="coloredBlur" />
   192→              <feMergeNode in="SourceGraphic" />
   193→            </feMerge>
   194→          </filter>
   195→        </defs>
   196→        <circle
   197→          cx={center}
   198→          cy={center}
   199→          r={radius}
   200→          fill="none"
   201→          stroke={color}
   202→          strokeWidth={strokeWidth}
   203→          strokeDasharray={circumference}
   204→          strokeDashoffset={offset}
   205→          strokeLinecap="round"
   206→          filter="url(#progressGlow)"
   207→          className="transition-all duration-700 ease-out"
   208→        />
   209→      </svg>
   210→      <div className="absolute inset-0 flex flex-col items-center justify-center">
   211→        <span className="text-lg font-bold text-foreground">{Math.round(value)}%</span>
   212→      </div>
   213→    </div>
   214→  );
   215→}
