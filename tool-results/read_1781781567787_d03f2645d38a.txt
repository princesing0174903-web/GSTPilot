    94→// ─── Component ────────────────────────────────────────────────────────────────
    95→
    96→export function MemoryPanel({ open, onClose, userEmail }: MemoryPanelProps) {
    97→  const [activeTab, setActiveTab] = useState<TabId>('profile');
    98→  const [memory, setMemory] = useState<FullMemory | null>(null);
    99→  const [loading, setLoading] = useState(false);
   100→  const [saving, setSaving] = useState(false);
   101→
   102→  const fetchMemory = useCallback(async () => {
   103→    if (!userEmail) return;
   104→    setLoading(true);
   105→    try {
   106→      const res = await fetch(`/api/memory?email=${encodeURIComponent(userEmail)}`);
   107→      if (!res.ok) throw new Error('Failed to load memory');
   108→      const data = await res.json();
   109→      setMemory(data.memory);
   110→    } catch (err) {
   111→      toast.error('Failed to load memory');
   112→      console.error(err);
   113→    } finally {
   114→      setLoading(false);
   115→    }
   116→  }, [userEmail]);
   117→
   118→  useEffect(() => {
   119→    if (open && userEmail) {
   120→      fetchMemory();
   121→    }
   122→  }, [open, userEmail, fetchMemory]);
   123→
   124→  // ── Generic save helper for profile/firm/preferences ──
   125→  const saveCategory = async (category: 'profile' | 'firm' | 'preferences', data: Record<string, unknown>) => {
   126→    if (!userEmail) return;
   127→    setSaving(true);
   128→    try {
   129→      const res = await fetch('/api/memory', {
   130→        method: 'PATCH',
   131→        headers: { 'Content-Type': 'application/json' },
   132→        body: JSON.stringify({ email: userEmail, category, data }),
   133→      });
   134→      if (!res.ok) throw new Error('Save failed');
   135→      toast.success('Memory saved', { duration: 1500 });
   136→      await fetchMemory();
   137→    } catch (err) {
   138→      toast.error('Failed to save');
   139→      console.error(err);
   140→    } finally {
   141→      setSaving(false);
   142→    }
   143→  };
   144→
   145→  return (
   146→    <AnimatePresence>
   147→      {open && (
   148→        <>
   149→          {/* Backdrop */}
   150→          <motion.div
   151→            initial={{ opacity: 0 }}
   152→            animate={{ opacity: 1 }}
   153→            exit={{ opacity: 0 }}
   154→            className="fixed inset-0 z-40 bg-black/60 backdrop-blur-sm"
   155→            onClick={onClose}
   156→          />
   157→
   158→          {/* Panel */}
   159→          <motion.div
   160→            initial={{ x: '100%' }}
   161→            animate={{ x: 0 }}
   162→            exit={{ x: '100%' }}
   163→            transition={{ type: 'spring', damping: 30, stiffness: 300 }}
   164→            className="fixed right-0 top-0 bottom-0 z-50 flex w-full max-w-[520px] flex-col border-l border-white/[0.08] bg-background shadow-2xl"
   165→          >
   166→            {/* Header */}
   167→            <div className="flex h-14 shrink-0 items-center gap-3 border-b border-white/[0.06] px-4">
   168→              <div className="flex h-8 w-8 items-center justify-center rounded-lg accent-gradient-soft">
   169→                <Brain className="h-4 w-4 accent-text" />
   170→              </div>
   171→              <div className="flex-1 min-w-0">
   172→                <h2 className="text-sm font-semibold text-foreground">Business Memory™</h2>
   173→                <p className="text-[11px] text-muted-foreground truncate">
   174→                  {userEmail ? userEmail : 'Sign in to save memory'}
   175→                </p>
   176→              </div>
   177→              <Button variant="ghost" size="icon" className="h-8 w-8" onClick={onClose}>
   178→                <X className="h-4 w-4" />
   179→              </Button>
   180→            </div>
   181→
   182→            {!userEmail ? (
   183→              <div className="flex flex-1 flex-col items-center justify-center gap-3 p-8 text-center">
   184→                <AlertCircle className="h-10 w-10 text-muted-foreground" />
   185→                <p className="text-sm text-muted-foreground">
   186→                  Sign in to start building your Business Memory™. Oracle will remember your profile, firm, goals, and preferences across sessions.
   187→                </p>
   188→              </div>
   189→            ) : loading || !memory ? (
   190→              <div className="flex flex-1 items-center justify-center">
   191→                <Loader2 className="h-6 w-6 animate-spin text-muted-foreground" />
   192→              </div>
   193→            ) : (
   194→              <>
   195→                {/* Tab bar */}
   196→                <div className="flex shrink-0 items-center gap-1 overflow-x-auto border-b border-white/[0.06] px-2 py-2">
   197→                  {TABS.map((tab) => {
   198→                    const Icon = tab.icon;
   199→                    const isActive = activeTab === tab.id;
   200→                    return (
   201→                      <button
   202→                        key={tab.id}
   203→                        onClick={() => setActiveTab(tab.id)}
   204→                        className={`flex items-center gap-1.5 whitespace-nowrap rounded-lg px-3 py-1.5 text-xs font-medium transition-colors ${
   205→                          isActive
   206→                            ? 'accent-gradient-soft accent-text'
   207→                            : 'text-muted-foreground hover:bg-white/[0.05] hover:text-foreground'
   208→                        }`}
   209→                      >
   210→                        <Icon className="h-3.5 w-3.5" />
   211→                        {tab.label}
   212→                      </button>
   213→                    );
   214→                  })}
   215→                </div>
   216→
   217→                {/* Tab content */}
   218→                <ScrollArea className="flex-1">
   219→                  <div className="p-4">
   220→                    {activeTab === 'profile' && (
   221→                      <ProfileTab key={JSON.stringify(memory.profile)} memory={memory} saving={saving} onSave={(d) => saveCategory('profile', d)} />
   222→                    )}
   223→                    {activeTab === 'firm' && (
   224→                      <FirmTab key={JSON.stringify(memory.firm)} memory={memory} saving={saving} onSave={(d) => saveCategory('firm', d)} />
   225→                    )}
   226→                    {activeTab === 'goals' && (
   227→                      <GoalsTab memory={memory} userEmail={userEmail} onChanged={fetchMemory} />
   228→                    )}
   229→                    {activeTab === 'preferences' && (
   230→                      <PreferencesTab key={JSON.stringify(memory.preferences)} memory={memory} saving={saving} onSave={(d) => saveCategory('preferences', d)} />
   231→                    )}
   232→                    {activeTab === 'clients' && (
   233→                      <ClientsTab memory={memory} userEmail={userEmail} onChanged={fetchMemory} />
   234→                    )}
   235→                    {activeTab === 'financial' && (
   236→                      <FinancialTab memory={memory} userEmail={userEmail} onChanged={fetchMemory} />
   237→                    )}
   238→                    {activeTab === 'conversations' && (
   239→                      <ConversationsTab memory={memory} />
   240→                    )}
   241→                    {activeTab === 'insights' && (
   242→                      <InsightsTab memory={memory} userEmail={userEmail} onChanged={fetchMemory} />
   243→                    )}
   244→                  </div>
   245→                </ScrollArea>
   246→
   247→                {/* Footer — clear all */}
   248→                <div className="shrink-0 border-t border-white/[0.06] p-3">
   249→                  <Button
   250→                    variant="ghost"
   251→                    size="sm"
   252→                    className="w-full text-xs text-red-500 hover:bg-red-500/10 hover:text-red-400"
   253→                    onClick={async () => {
   254→                      if (!confirm('Clear ALL memory? This cannot be undone.')) return;
   255→                      try {
   256→                        await fetch(`/api/memory?email=${encodeURIComponent(userEmail)}`, { method: 'DELETE' });
   257→                        toast.success('All memory cleared');
   258→                        await fetchMemory();
   259→                      } catch {
   260→                        toast.error('Failed to clear memory');
   261→                      }
   262→                    }}
   263→                  >
   264→                    <Trash2 className="mr-1.5 h-3.5 w-3.5" />
   265→                    Clear All Memory
   266→                  </Button>
   267→                </div>
   268→              </>
   269→            )}
   270→          </motion.div>
   271→        </>
   272→      )}
   273→    </AnimatePresence>
   274→  );
   275→}

   345→function FirmTab({ memory, saving, onSave }: { memory: FullMemory; saving: boolean; onSave: (d: Record<string, unknown>) => void }) {
   346→  const [form, setForm] = useState({
   347→    caFirmName: memory.firm?.caFirmName || '',
   348→    employeeCount: memory.firm?.employeeCount || 0,
   349→    clientCount: memory.firm?.clientCount || 0,
   350→    industriesServed: (memory.firm?.industriesServed || []).join(', '),
   351→    gstRegistrations: (memory.firm?.gstRegistrations || []).join(', '),
   352→    branches: (memory.firm?.branches || []).join(', '),
   353→    servicesOffered: (memory.firm?.servicesOffered || []).join(', '),
   354→  });
   355→
   356→  // No useEffect sync needed — parent uses key={JSON.stringify(memory.firm)}
   357→
   358→  const buildPayload = () => ({
   359→    caFirmName: form.caFirmName,
   360→    employeeCount: Number(form.employeeCount) || 0,
   361→    clientCount: Number(form.clientCount) || 0,
   362→    industriesServed: form.industriesServed.split(',').map((s) => s.trim()).filter(Boolean),
   363→    gstRegistrations: form.gstRegistrations.split(',').map((s) => s.trim()).filter(Boolean),
   364→    branches: form.branches.split(',').map((s) => s.trim()).filter(Boolean),
   365→    servicesOffered: form.servicesOffered.split(',').map((s) => s.trim()).filter(Boolean),
   366→  });
   367→
   368→  return (
   369→    <div className="space-y-4">
   370→      <p className="text-xs text-muted-foreground">
   371→        Oracle uses firm context to scale advice — a 127-client CA firm gets different recommendations than a 5-client boutique.
   372→      </p>
   373→      <Field label="CA Firm / Company Name">
   374→        <Input value={form.caFirmName} onChange={(e) => setForm({ ...form, caFirmName: e.target.value })} placeholder="e.g. Prince & Associates" />
   375→      </Field>
   376→      <div className="grid grid-cols-2 gap-3">
   377→        <Field label="Employees">
   378→          <Input type="number" value={form.employeeCount} onChange={(e) => setForm({ ...form, employeeCount: Number(e.target.value) })} />
   379→        </Field>
   380→        <Field label="Clients">
   381→          <Input type="number" value={form.clientCount} onChange={(e) => setForm({ ...form, clientCount: Number(e.target.value) })} />
   382→        </Field>
   383→      </div>
   384→      <Field label="Industries Served (comma-separated)">
   385→        <Input value={form.industriesServed} onChange={(e) => setForm({ ...form, industriesServed: e.target.value })} placeholder="Manufacturing, Retail, Healthcare" />
   386→      </Field>
   387→      <Field label="GST Registrations (comma-separated)">
   388→        <Input value={form.gstRegistrations} onChange={(e) => setForm({ ...form, gstRegistrations: e.target.value })} placeholder="27AABCS1429B1Z5, 29AAACL1234M1Z3" />
   389→      </Field>
   390→      <Field label="Branches (comma-separated)">
   391→        <Input value={form.branches} onChange={(e) => setForm({ ...form, branches: e.target.value })} placeholder="Mumbai, Delhi, Bangalore" />
   392→      </Field>
   393→      <Field label="Services Offered (comma-separated)">

   405→function GoalsTab({ memory, userEmail, onChanged }: { memory: FullMemory; userEmail: string; onChanged: () => void }) {
   406→  const [showForm, setShowForm] = useState(false);
   407→  const [form, setForm] = useState({ type: 'revenue', target: '', period: '', notes: '' });
   408→
   409→  const addGoal = async () => {
   410→    if (!form.target.trim()) return;
   411→    try {
   412→      await fetch('/api/memory/goal', {
   413→        method: 'POST',
   414→        headers: { 'Content-Type': 'application/json' },
   415→        body: JSON.stringify({ email: userEmail, ...form, period: form.period || undefined, notes: form.notes || undefined }),
   416→      });
   417→      setForm({ type: 'revenue', target: '', period: '', notes: '' });
   418→      setShowForm(false);
   419→      onChanged();
   420→      toast.success('Goal added');
   421→    } catch {
   422→      toast.error('Failed to add goal');
   423→    }
   424→  };
   425→
   426→  return (
   427→    <div className="space-y-3">
   428→      <div className="flex items-center justify-between">
   429→        <p className="text-xs text-muted-foreground">Revenue, collection, growth targets Oracle tracks.</p>
   430→        <Button size="sm" variant="outline" className="h-7 text-xs" onClick={() => setShowForm(!showForm)}>
   431→          <Plus className="mr-1 h-3 w-3" /> Add
   432→        </Button>
   433→      </div>
   434→
   435→      {showForm && (
   436→        <div className="space-y-3 rounded-lg border border-white/[0.08] p-3">
   437→          <Field label="Type">
   438→            <Select value={form.type} onValueChange={(v) => setForm({ ...form, type: v })}>
   439→              <SelectTrigger><SelectValue /></SelectTrigger>
   440→              <SelectContent>
   441→                <SelectItem value="revenue">Revenue</SelectItem>
   442→                <SelectItem value="collection">Collection</SelectItem>
   443→                <SelectItem value="growth">Growth</SelectItem>
   444→                <SelectItem value="profit">Profit</SelectItem>
   445→                <SelectItem value="expansion">Expansion</SelectItem>
   446→              </SelectContent>
   447→            </Select>
   448→          </Field>
   449→          <Field label="Target">
   450→            <Input value={form.target} onChange={(e) => setForm({ ...form, target: e.target.value })} placeholder="e.g. ₹5 Crore" />
   451→          </Field>
   452→          <Field label="Period (optional)">
   453→            <Input value={form.period} onChange={(e) => setForm({ ...form, period: e.target.value })} placeholder="e.g. FY2025, Q3" />
   454→          </Field>
   455→          <Field label="Notes (optional)">
   456→            <Textarea value={form.notes} onChange={(e) => setForm({ ...form, notes: e.target.value })} rows={2} />
   457→          </Field>
   458→          <Button size="sm" className="w-full" onClick={addGoal}>Save Goal</Button>
   459→        </div>
   460→      )}
   461→
   462→      {memory.goals.length === 0 ? (
   463→        <EmptyState text="No goals set yet. Add your first goal to give Oracle a target to track." />
   464→      ) : (
   465→        <div className="space-y-2">
   466→          {memory.goals.map((g) => (
   467→            <div key={g.id} className="flex items-start gap-2 rounded-lg border border-white/[0.06] p-3">
   468→              <button
   469→                onClick={async () => {
   470→                  await fetch('/api/memory/goal', { method: 'PATCH', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ id: g.id, email: userEmail }) });
   471→                  onChanged();
   472→                }}
   473→                className="mt-0.5"
   474→              >
   475→                {g.pinned ? <Pin className="h-3.5 w-3.5 accent-text" /> : <PinOff className="h-3.5 w-3.5 text-muted-foreground" />}
   476→              </button>
