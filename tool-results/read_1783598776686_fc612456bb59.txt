   146→   146→export async function loadLiveBusinessData(organizationId: string): Promise<LiveBusinessData> {
   147→   147→  const [clientsRaw, invoicesRaw, returnsRaw, gstRaw] = await Promise.all([
   148→   148→    fetchCollection(COLLECTIONS.CLIENTS, organizationId, 100),
   149→   149→    fetchCollection(COLLECTIONS.INVOICES, organizationId, 100),
   150→   150→    fetchCollection(COLLECTIONS.GST_RETURNS, organizationId, 50),
   151→   151→    fetchCollection(COLLECTIONS.GST_PROFILES, organizationId, 5),
   152→   152→  ]);
   153→   153→
   154→   154→  const clients: LiveBusinessData['clients'] = clientsRaw.map((c) => ({
   155→   155→    id: String(c.id ?? c.clientId ?? ''),
   156→   156→    name: String(c.name ?? c.clientName ?? c.displayName ?? 'Unknown'),
   157→   157→    gstin: (c.gstin as string) ?? null,
   158→   158→    email: (c.email as string) ?? null,
   159→   159→    phone: (c.phone as string) ?? null,
   160→   160→  }));
   161→   161→
   162→   162→  const recentInvoices: LiveBusinessData['recentInvoices'] = invoicesRaw
   163→   163→    .map((i) => ({
   164→   164→      id: String(i.id ?? i.invoiceId ?? ''),
   165→   165→      invoiceNumber: String(i.invoiceNumber ?? ''),
   166→   166→      clientName: String(i.buyerName ?? i.clientName ?? ''),
   167→   167→      totalAmount: Number(i.totalAmount ?? i.grandTotal ?? 0),
   168→   168→      status: String(i.status ?? 'draft'),
   169→   169→    }))
   170→   170→    .slice(0, 20);
   171→   171→
   172→   172→  const now = Date.now();
   173→   173→  const overdueInvoices: LiveBusinessData['overdueInvoices'] = invoicesRaw
   174→   174→    .filter((i) => {
   175→   175→      const status = String(i.status ?? '').toLowerCase();
   176→   176→      const due = i.dueDate ? new Date(i.dueDate as string).getTime() : 0;
   177→   177→      return status !== 'paid' && status !== 'cancelled' && due > 0 && due < now;
   178→   178→    })
   179→   179→    .map((i) => {
   180→   180→      const due = new Date(i.dueDate as string).getTime();
   181→   181→      return {
   182→   182→        id: String(i.id ?? i.invoiceId ?? ''),
   183→   183→        invoiceNumber: String(i.invoiceNumber ?? ''),
   184→   184→        clientName: String(i.buyerName ?? i.clientName ?? ''),
   185→   185→        clientId: String(i.clientId ?? ''),
   186→   186→        totalAmount: Number(i.totalAmount ?? i.grandTotal ?? 0),
   187→   187→        daysOverdue: Math.floor((now - due) / (24 * 60 * 60 * 1000)),
   188→   188→      };
   189→   189→    })
   190→   190→    .sort((a, b) => b.daysOverdue - a.daysOverdue)
   191→   191→    .slice(0, 20);
   192→   192→
   193→   193→  const pendingReturns: LiveBusinessData['pendingReturns'] = returnsRaw
   194→   194→    .filter((r) => {
   195→   195→      const status = String(r.status ?? '').toLowerCase();
   196→   196→      return status === 'pending' || status === 'draft' || status === 'ready';
   197→   197→    })
   198→   198→    .map((r) => ({
   199→   199→      id: String(r.id ?? r.returnId ?? ''),
   200→   200→      returnType: String(r.returnType ?? r.formType ?? 'GSTR-3B'),
   201→   201→      period: String(r.period ?? r.taxPeriod ?? ''),
   202→   202→      dueDate: String(r.dueDate ?? ''),
   203→   203→      status: String(r.status ?? 'pending'),
   204→   204→    }));
   205→   205→
   206→   206→  const gstProfile =
   207→   207→    gstRaw.length > 0
   208→   208→      ? {
   209→   209→          gstin: (gstRaw[0].gstin as string) ?? null,
   210→   210→          legalName: (gstRaw[0].legalName as string) ?? (gstRaw[0].tradeName as string) ?? null,
   211→   211→        }
   212→   212→      : null;
   213→   213→
   214→   214→  return { clients, recentInvoices, overdueInvoices, pendingReturns, gstProfile };
   215→   215→}

   217→   217→// ─── Helper: retry with exponential backoff ─────────────────────────────────
   218→   218→
   219→   219→async function withRetry<T>(
   220→   220→  fn: () => Promise<T>,
   221→   221→  policy: { maxAttempts: number; backoffMs: number },
   222→   222→): Promise<T> {
   223→   223→  let lastError: unknown;
   224→   224→  for (let attempt = 1; attempt <= policy.maxAttempts; attempt++) {
   225→   225→    try {
   226→   226→      return await fn();
   227→   227→    } catch (err) {
   228→   228→      lastError = err;
   229→   229→      if (attempt < policy.maxAttempts) {
   230→   230→        await new Promise((r) => setTimeout(r, policy.backoffMs * attempt));
   231→   231→      }
   232→   232→    }
   233→   233→  }
   234→   234→  throw lastError;
   235→   235→}

   239→   239→function genId(prefix: string): string {

   243→   243→// ─── Helper: interpret Firestore errors into user-friendly messages ─────────
   244→   244→// The tool executors catch errors and return a ToolResult. This helper
   245→   245→// translates raw Firebase error codes (permission-denied, unavailable, etc.)
   246→   246→// into actionable messages so the UI never shows a raw stack trace.
   247→   247→
   248→   248→function interpretError(err: unknown, actionLabel: string): { message: string; error: string } {
   249→   249→  const raw = err instanceof Error ? err.message : String(err);
   250→   250→  const lower = raw.toLowerCase();
   251→   251→  if (lower.includes('permission-denied') || lower.includes('missing or insufficient permissions')) {
   252→   252→    return {
   253→   253→      message: `${actionLabel} was validated successfully, but could not be saved because you are in preview mode. Sign in to persist this action to your live business data.`,
   254→   254→      error: 'PREVIEW_MODE: Firestore security rules require authentication. The action logic ran correctly — sign in to execute for real.',
   255→   255→    };
   256→   256→  }
   257→   257→  if (lower.includes('unavailable') || lower.includes('offline')) {
   258→   258→    return {
   259→   259→      message: `${actionLabel} could not complete because the database is temporarily unavailable. Please retry in a moment.`,
   260→   260→      error: raw,
   261→   261→    };
   262→   262→  }
   263→   263→  return {
   264→   264→    message: `${actionLabel} failed after retries. The error has been logged. Please try again or contact support if it persists.`,
   265→   265→    error: raw,
   266→   266→  };
   267→   267→}
