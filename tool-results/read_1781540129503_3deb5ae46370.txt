    76→    76→// ─── Animated Number Hook ─────────────────────────────────────────────────
    77→    77→function useAnimatedNumber(target: number, duration: number = 1200) {
    78→    78→  const [current, setCurrent] = useState(0);
    79→    79→  const ref = useRef<number | null>(null);
    80→    80→  const startTime = useRef<number | null>(null);
    81→    81→
    82→    82→  useEffect(() => {
    83→    83→    startTime.current = null;
    84→    84→    const startValue = current;
    85→    85→
    86→    86→    function step(timestamp: number) {
    87→    87→      if (!startTime.current) startTime.current = timestamp;
    88→    88→      const elapsed = timestamp - startTime.current;
    89→    89→      const progress = Math.min(elapsed / duration, 1);
    90→    90→      const eased = 1 - Math.pow(1 - progress, 3);
    91→    91→      setCurrent(Math.round(startValue + (target - startValue) * eased));
    92→    92→      if (progress < 1) {
    93→    93→        ref.current = requestAnimationFrame(step);
    94→    94→      }
    95→    95→    }
    96→    96→
    97→    97→    ref.current = requestAnimationFrame(step);
    98→    98→    return () => {
    99→    99→      if (ref.current) cancelAnimationFrame(ref.current);
   100→   100→    };
   101→   101→  }, [target, duration]);
   102→   102→
   103→   103→  return current;
   104→   104→}

   107→   107→function MiniSparkline({
   108→   108→  data,
   109→   109→  color,
   110→   110→  width = 80,
   111→   111→  height = 32,
   112→   112→}: {
   113→   113→  data: number[];
   114→   114→  color: string;
   115→   115→  width?: number;
   116→   116→  height?: number;
   117→   117→}) {
   118→   118→  if (data.length < 2) return null;
   119→   119→  const min = Math.min(...data);
   120→   120→  const max = Math.max(...data);
   121→   121→  const range = max - min || 1;
   122→   122→  const padding = 2;
   123→   123→
   124→   124→  const points = data
   125→   125→    .map((v, i) => {
   126→   126→      const x = padding + (i / (data.length - 1)) * (width - padding * 2);
   127→   127→      const y = height - padding - ((v - min) / range) * (height - padding * 2);
   128→   128→      return `${x},${y}`;
   129→   129→    })
   130→   130→    .join(' ');
   131→   131→
   132→   132→  const areaPoints = `${padding},${height - padding} ${points} ${width - padding},${height - padding}`;
   133→   133→
   134→   134→  return (
   135→   135→    <svg width={width} height={height} className="shrink-0">
   136→   136→      <defs>
   137→   137→        <linearGradient id={`sparkGrad-${color.replace('#', '')}`} x1="0" y1="0" x2="0" y2="1">
   138→   138→          <stop offset="0%" stopColor={color} stopOpacity={0.3} />
   139→   139→          <stop offset="100%" stopColor={color} stopOpacity={0} />
   140→   140→        </linearGradient>
   141→   141→      </defs>
   142→   142→      <polygon
   143→   143→        points={areaPoints}
   144→   144→        fill={`url(#sparkGrad-${color.replace('#', '')})`}
   145→   145→      />
   146→   146→      <polyline
   147→   147→        points={points}
   148→   148→        fill="none"
   149→   149→        stroke={color}
   150→   150→        strokeWidth={1.5}
   151→   151→        strokeLinecap="round"
   152→   152→        strokeLinejoin="round"
   153→   153→      />
   154→   154→    </svg>
   155→   155→  );
   156→   156→}
