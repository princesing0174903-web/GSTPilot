    72→// ── SVG Sparkline ──
    73→function Sparkline({ data, width = 120, height = 36, color = '#10b981', fill = true }: {
    74→  data: number[]; width?: number; height?: number; color?: string; fill?: boolean
    75→}) {
    76→  const min = Math.min(...data)
    77→  const max = Math.max(...data)
    78→  const range = max - min || 1
    79→  const pts = data.map((v, i) => {
    80→    const x = (i / (data.length - 1)) * width
    81→    const y = height - ((v - min) / range) * (height - 4) - 2
    82→    return `${x},${y}`
    83→  }).join(' ')
    84→  const areaPath = data.map((v, i) => {
    85→    const x = (i / (data.length - 1)) * width
    86→    const y = height - ((v - min) / range) * (height - 4) - 2
    87→    return `${i === 0 ? 'M' : 'L'}${x},${y}`
    88→  }).join(' ') + ` L${width},${height} L0,${height} Z`
    89→
    90→  return (
    91→    <svg width={width} height={height} className="overflow-visible">
    92→      {fill && <path d={areaPath} fill={color + '18'} />}
    93→      <polyline points={pts} fill="none" stroke={color} strokeWidth={2} strokeLinecap="round" strokeLinejoin="round" />
    94→      <circle cx={(data.length - 1) / (data.length - 1) * width} cy={height - ((data[data.length - 1] - min) / range) * (height - 4) - 2} r={3} fill={color} />
    95→    </svg>
    96→  )
    97→}

    99→// ── SVG Pie Chart ──
   100→function PieChartSVG({ segments }: { segments: { value: number; color: string; label: string }[] }) {
   101→  const total = segments.reduce((s, seg) => s + seg.value, 0)
   102→  const r = 60
   103→  const cx = 80, cy = 80
   104→
   105→  // Compute cumulative offsets without mutation during render
   106→  const cumOffsets = segments.reduce<number[]>((acc, seg, i) => {
   107→    const prev = i === 0 ? -90 : acc[i - 1]
   108→    const angle = (seg.value / total) * 360
   109→    acc.push(prev + angle)
   110→    return acc
   111→  }, [])
   112→
   113→  const arcs = segments.map((seg, i) => {
   114→    const angle = (seg.value / total) * 360
   115→    const startAngle = i === 0 ? -90 : cumOffsets[i - 1]
   116→    const endAngle = cumOffsets[i]
   117→    const startRad = (startAngle * Math.PI) / 180
   118→    const endRad = (endAngle * Math.PI) / 180
   119→    const x1 = cx + r * Math.cos(startRad)
   120→    const y1 = cy + r * Math.sin(startRad)
   121→    const x2 = cx + r * Math.cos(endRad)
   122→    const y2 = cy + r * Math.sin(endRad)
   123→    const largeArc = angle > 180 ? 1 : 0
   124→    const d = `M${cx},${cy} L${x1},${y1} A${r},${r} 0 ${largeArc},1 ${x2},${y2} Z`
   125→    return { ...seg, d }
   126→  })
   127→
   128→  return (
   129→    <svg width={160} height={160} viewBox="0 0 160 160">
   130→      {arcs.map((arc, i) => (
   131→        <path key={i} d={arc.d} fill={arc.color} stroke="white" strokeWidth={2} />
   132→      ))}
   133→      <circle cx={cx} cy={cy} r={32} fill="white" />
   134→      <text x={cx} y={cy - 4} textAnchor="middle" className="text-lg font-bold" fill="#0f172a">{total}</text>
   135→      <text x={cx} y={cy + 12} textAnchor="middle" className="text-[9px]" fill="#64748b">clients</text>
   136→    </svg>
   137→  )
   138→}

   140→// ── SVG Gauge ──
   141→function GaugeChart({ value, max = 100 }: { value: number; max?: number }) {
   142→  const pctVal = value / max
   143→  const angle = pctVal * 180
   144→  const r = 70
   145→  const cx = 90, cy = 90
   146→  const startAngle = 180
   147→  const endAngle = 0
   148→  const needleAngle = startAngle - angle
   149→
   150→  const needleRad = (needleAngle * Math.PI) / 180
   151→  const nx = cx + (r - 15) * Math.cos(needleRad)
   152→  const ny = cy - (r - 15) * Math.sin(needleRad)
   153→
   154→  const bgPath = describeArc(cx, cy, r, startAngle, endAngle)
   155→  const fillEndAngle = 180 - angle
   156→  const fillPath = describeArc(cx, cy, r, startAngle, fillEndAngle)
   157→
   158→  const color = value >= 80 ? '#10b981' : value >= 60 ? '#f59e0b' : '#ef4444'
   159→
   160→  return (
   161→    <svg width={180} height={110} viewBox="0 0 180 110">
   162→      <path d={bgPath} fill="none" stroke="#e2e8f0" strokeWidth={12} strokeLinecap="round" />
   163→      <path d={fillPath} fill="none" stroke={color} strokeWidth={12} strokeLinecap="round" />
   164→      <line x1={cx} y1={cy} x2={nx} y2={ny} stroke="#0f172a" strokeWidth={3} strokeLinecap="round" />
   165→      <circle cx={cx} cy={cy} r={5} fill="#0f172a" />
   166→      <text x={cx} y={cy + 30} textAnchor="middle" className="text-2xl font-bold" fill="#0f172a">{value}</text>
   167→      <text x={cx} y={cy + 44} textAnchor="middle" className="text-[10px]" fill="#64748b">/ {max}</text>
   168→    </svg>
   169→  )
   170→}

   172→function describeArc(cx: number, cy: number, r: number, startAngle: number, endAngle: number): string {
   173→  const s = polarToCartesian(cx, cy, r, endAngle)
   174→  const e = polarToCartesian(cx, cy, r, startAngle)
   175→  const largeArc = startAngle - endAngle > 180 ? 1 : 0
   176→  return `M${s.x},${s.y} A${r},${r} 0 ${largeArc},0 ${e.x},${e.y}`
   177→}
   178→
   179→function polarToCartesian(cx: number, cy: number, r: number, angle: number) {
   180→  const rad = (angle * Math.PI) / 180
   181→  return { x: cx + r * Math.cos(rad), y: cy - r * Math.sin(rad) }
   182→}

   184→// ── SVG Line Chart ──
   185→function LineChart({ data, months, width = 280, height = 120, color = '#10b981', prefix = '₹' }: {
   186→  data: number[]; months: string[]; width?: number; height?: number; color?: string; prefix?: string
   187→}) {
   188→  const pad = { top: 10, right: 10, bottom: 24, left: 10 }
   189→  const w = width - pad.left - pad.right
   190→  const h = height - pad.top - pad.bottom
   191→  const min = Math.min(...data) * 0.95
   192→  const max = Math.max(...data) * 1.05
   193→  const range = max - min || 1
   194→
   195→  const points = data.map((v, i) => ({
   196→    x: pad.left + (i / (data.length - 1)) * w,
   197→    y: pad.top + h - ((v - min) / range) * h,
   198→  }))
   199→
   200→  const linePath = points.map((p, i) => `${i === 0 ? 'M' : 'L'}${p.x.toFixed(1)},${p.y.toFixed(1)}`).join(' ')
   201→  const areaPath = linePath + ` L${points[points.length - 1].x.toFixed(1)},${(pad.top + h).toFixed(1)} L${points[0].x.toFixed(1)},${(pad.top + h).toFixed(1)} Z`
   202→
   203→  return (
   204→    <svg width={width} height={height}>
   205→      <path d={areaPath} fill={color + '15'} />
   206→      <path d={linePath} fill="none" stroke={color} strokeWidth={2.5} strokeLinecap="round" strokeLinejoin="round" />
   207→      {points.map((p, i) => (
   208→        <g key={i}>
   209→          <circle cx={p.x} cy={p.y} r={3.5} fill="white" stroke={color} strokeWidth={2} />
   210→          <text x={p.x} y={height - 4} textAnchor="middle" className="text-[9px]" fill="#94a3b8">{months[i]}</text>
   211→        </g>
   212→      ))}
   213→    </svg>
   214→  )
   215→}

   217→// ── Bar Chart ──
   218→function BarChart({ data, months, width = 280, height = 120, color = '#10b981' }: {
   219→  data: number[]; months: string[]; width?: number; height?: number; color?: string
   220→}) {
   221→  const pad = { top: 10, right: 10, bottom: 24, left: 10 }
   222→  const w = width - pad.left - pad.right
   223→  const h = height - pad.top - pad.bottom
   224→  const max = Math.max(...data) * 1.1
   225→  const barW = w / data.length * 0.6
   226→  const gap = w / data.length
   227→
   228→  return (
   229→    <svg width={width} height={height}>
   230→      {data.map((v, i) => {
   231→        const barH = (v / max) * h
   232→        const x = pad.left + i * gap + (gap - barW) / 2
   233→        const y = pad.top + h - barH
   234→        return (
   235→          <g key={i}>
   236→            <rect x={x} y={y} width={barW} height={barH} rx={3} fill={color} opacity={0.8} />
   237→            <text x={x + barW / 2} y={height - 4} textAnchor="middle" className="text-[9px]" fill="#94a3b8">{months[i]}</text>
   238→          </g>
   239→        )
   240→      })}
   241→    </svg>
   242→  )
   243→}
