   120→// ─── Helpers ───────────────────────────────────────────────────────────────
   121→function formatFileSize(bytes: number): string {
   122→  if (bytes === 0) return '0 B';
   123→  const k = 1024;
   124→  const sizes = ['B', 'KB', 'MB', 'GB'];
   125→  const i = Math.floor(Math.log(bytes) / Math.log(k));
   126→  return parseFloat((bytes / Math.pow(k, i)).toFixed(1)) + ' ' + sizes[i];
   127→}

   129→function getFileTypeColor(type: string): string {

   131→  if (t === 'pdf') return 'bg-red-100 text-red-700 dark:bg-red-950/40 dark:text-red-400';

   160→// ─── Animated Number Hook ──────────────────────────────────────────────────
   161→function useAnimatedNumber(target: number, duration: number = 1200) {
   162→  const [current, setCurrent] = useState(0);
   163→  const ref = useRef<number | null>(null);
   164→  const startTime = useRef<number | null>(null);
   165→
   166→  useEffect(() => {
   167→    startTime.current = null;
   168→    const startValue = current;
   169→
   170→    function step(timestamp: number) {
   171→      if (!startTime.current) startTime.current = timestamp;
   172→      const elapsed = timestamp - startTime.current;
   173→      const progress = Math.min(elapsed / duration, 1);
   174→      const eased = 1 - Math.pow(1 - progress, 3);
   175→      setCurrent(Math.round(startValue + (target - startValue) * eased));
   176→      if (progress < 1) {
   177→        ref.current = requestAnimationFrame(step);
   178→      }
   179→    }
   180→
   181→    ref.current = requestAnimationFrame(step);
   182→    return () => {
   183→      if (ref.current) cancelAnimationFrame(ref.current);
   184→    };
   185→  }, [target, duration]);
   186→
   187→  return current;
   188→}
