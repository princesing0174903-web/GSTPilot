    31→function formatINR(n: number): string {
    32→  const s = Math.abs(Math.round(n)).toString()
    33→  let result = ''
    34→  let count = 0
    35→  for (let i = s.length - 1; i >= 0; i--) {
    36→    result = s[i] + result
    37→    count++
    38→    if (count === 3 && i > 0) { result = ',' + result; count = 0 }
    39→    else if (count > 3 && count % 2 === 1 && i > 0) { result = ',' + result }
    40→  }
    41→  return '₹' + (n < 0 ? '-' : '') + result
    42→}
    43→
    44→function formatDate(iso: string | null | unknown): string {
    45→  if (!iso) return '—'
    46→  try {
    47→    const d = new Date(iso as string)
    48→    return d.toLocaleDateString('en-IN', { day: '2-digit', month: '2-digit', year: 'numeric' })
    49→  } catch { return '—' }
    50→}

   393→// ═══════════════════════════════════════════════════════════════════════════════
   394→// SIMPLE FORCE-DIRECTED LAYOUT (no external libs)
   395→// ═══════════════════════════════════════════════════════════════════════════════
   396→
   397→function applyForceLayout(nodes: GraphNode[], edges: GraphEdge[], iterations: number = 60): GraphNode[] {
   398→  const positioned = nodes.map(n => ({ ...n }))
   399→  const width = 1000
   400→  const height = 700
   401→
   402→  // Initialize positions if they're clustered
   403→  positioned.forEach((n, i) => {
   404→    if (n.x === 0 && n.y === 0) {
   405→      n.x = width / 2 + (Math.random() - 0.5) * 400
   406→      n.y = height / 2 + (Math.random() - 0.5) * 300
   407→    }
   408→  })
   409→
   410→  for (let iter = 0; iter < iterations; iter++) {
   411→    const temperature = 1 - (iter / iterations)
   412→    const k = Math.sqrt((width * height) / positioned.length) * 0.8
   413→
   414→    // Repulsive forces between all nodes
   415→    const dx: Record<string, number> = {}
   416→    const dy: Record<string, number> = {}
   417→    positioned.forEach(n => { dx[n.id] = 0; dy[n.id] = 0 })
   418→
   419→    for (let i = 0; i < positioned.length; i++) {
   420→      for (let j = i + 1; j < positioned.length; j++) {
   421→        const ni = positioned[i]
   422→        const nj = positioned[j]
   423→        let diffX = ni.x - nj.x
   424→        let diffY = ni.y - nj.y
   425→        const dist = Math.max(1, Math.sqrt(diffX * diffX + diffY * diffY))
   426→        const force = (k * k) / dist
   427→        diffX = (diffX / dist) * force * temperature
   428→        diffY = (diffY / dist) * force * temperature
   429→        dx[ni.id] += diffX
   430→        dy[ni.id] += diffY
   431→        dx[nj.id] -= diffX
   432→        dy[nj.id] -= diffY
   433→      }
   434→    }
   435→
   436→    // Attractive forces for connected nodes
   437→    edges.forEach(e => {
   438→      const source = positioned.find(n => n.id === e.source)
   439→      const target = positioned.find(n => n.id === e.target)
   440→      if (!source || !target) return
   441→      let diffX = source.x - target.x
   442→      let diffY = source.y - target.y
   443→      const dist = Math.max(1, Math.sqrt(diffX * diffX + diffY * diffY))
   444→      const force = (dist * dist) / k
   445→      diffX = (diffX / dist) * force * temperature * 0.3
   446→      diffY = (diffY / dist) * force * temperature * 0.3
   447→      dx[source.id] -= diffX
   448→      dy[source.id] -= diffY
   449→      dx[target.id] += diffX
   450→      dy[target.id] += diffY
   451→    })
   452→
   453→    // Apply forces
   454→    positioned.forEach(n => {
   455→      const disp = Math.sqrt(dx[n.id] * dx[n.id] + dy[n.id] * dy[n.id])
   456→      if (disp > 0) {
   457→        n.x += (dx[n.id] / disp) * Math.min(disp, 10 * temperature)
   458→        n.y += (dy[n.id] / disp) * Math.min(disp, 10 * temperature)
   459→      }
   460→      // Keep in bounds
   461→      n.x = Math.max(40, Math.min(width - 40, n.x))
   462→      n.y = Math.max(40, Math.min(height - 40, n.y))
   463→    })
   464→  }
   465→
   466→  return positioned
   467→}
