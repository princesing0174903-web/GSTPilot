   156→// ─── Helpers ─────────────────────────────────────────────────────────────────
   157→
   158→function getKanbanColumn(status: FilingStatus): KanbanColumn {
   159→  switch (status) {
   160→    case 'draft':
   161→    case 'prepared':
   162→      return 'draft';
   163→    case 'validated':
   164→    case 'reviewed':
   165→    case 'generated':
   166→      return 'ready';
   167→    case 'filed':
   168→      return 'filed';
   169→    case 'reopened':
   170→      return 'attention';
   171→    default:
   172→      return 'draft';
   173→  }
   174→}
   175→
   176→function getTimelineStepIndex(status: FilingStatus): number {
   177→  const idx = TIMELINE_ORDER.indexOf(status);
   178→  return idx >= 0 ? idx : -1;
   179→}
   180→
   181→function getAttentionSummary(ret: ReturnItem): string {
   182→  if (ret.issuesFound <= 0) return 'No issues';
   183→  const parts: string[] = [];
   184→  if (ret.criticalErrors > 0) parts.push(`${ret.criticalErrors} critical error${ret.criticalErrors > 1 ? 's' : ''}`);
   185→  if (ret.warnings > 0) parts.push(`${ret.warnings} warning${ret.warnings > 1 ? 's' : ''}`);
   186→  if (ret.issuesFound > 0 && parts.length === 0) parts.push(`${ret.issuesFound} issue${ret.issuesFound > 1 ? 's' : ''}`);
   187→  return parts.join(' · ');
   188→}
   189→
   190→function getReturnTypeBadgeClass(returnType: string): string {
   191→  if (returnType === 'GSTR-1') return 'border-teal-200 bg-teal-50 text-teal-700';
   192→  return 'border-emerald-200 bg-emerald-50 text-emerald-700';
   193→}

   195→function getClientName(clientId: string, clients: ClientItem[]): string {
   196→  const client = clients.find(c => c.clientId === clientId || c.id === clientId);
