   459→function DocumentUploadHub({ onUpload }: { onUpload: (files: File[], docType: DocType) => void }) {
   460→  const [isDragging, setIsDragging] = useState(false)
   461→  const [selectedType, setSelectedType] = useState<DocType>('invoice')
   462→  const [uploadProgress, setUploadProgress] = useState<Record<string, number>>({})
   463→  const fileInputRef = useRef<HTMLInputElement>(null)
   464→
   465→  const handleDragOver = useCallback((e: React.DragEvent) => {
   466→    e.preventDefault()
   467→    setIsDragging(true)
   468→  }, [])
   469→
   470→  const handleDragLeave = useCallback((e: React.DragEvent) => {
   471→    e.preventDefault()
   472→    setIsDragging(false)
   473→  }, [])
   474→
   475→  const processFiles = useCallback((files: FileList | File[]) => {
   476→    const fileArr = Array.from(files)
   477→    const validExts = ['pdf', 'png', 'jpg', 'jpeg', 'xlsx', 'csv']
   478→    const validFiles = fileArr.filter(f => {
   479→      const ext = f.name.split('.').pop()?.toLowerCase() || ''
   480→      return validExts.includes(ext)
   481→    })
   482→    if (validFiles.length === 0) {
   483→      toast.error('No supported files found. Use PDF, PNG, JPG, XLSX, or CSV.')
   484→      return
   485→    }
   486→    if (validFiles.length < fileArr.length) {
   487→      toast.warning(`${fileArr.length - validFiles.length} file(s) skipped — unsupported format`)
   488→    }
   489→    // Simulate upload progress
   490→    validFiles.forEach(f => {
   491→      setUploadProgress(prev => ({ ...prev, [f.name]: 0 }))
   492→      let progress = 0
   493→      const interval = setInterval(() => {
   494→        progress += Math.random() * 25 + 5
   495→        if (progress >= 100) {
   496→          progress = 100
   497→          clearInterval(interval)
   498→          toast.success(`${f.name} uploaded`)
   499→        }
   500→        setUploadProgress(prev => ({ ...prev, [f.name]: Math.min(progress, 100) }))
   501→      }, 400)
   502→    })
   503→    onUpload(validFiles, selectedType)
   504→  }, [onUpload, selectedType])
   505→
   506→  const handleDrop = useCallback((e: React.DragEvent) => {
   507→    e.preventDefault()
   508→    setIsDragging(false)
   509→    if (e.dataTransfer.files.length > 0) {
   510→      processFiles(e.dataTransfer.files)
   511→    }
   512→  }, [processFiles])
   513→
   514→  const handleFileSelect = useCallback((e: React.ChangeEvent<HTMLInputElement>) => {
   515→    if (e.target.files && e.target.files.length > 0) {
   516→      processFiles(e.target.files)
   517→    }
   518→  }, [processFiles])
   519→
   520→  const activeUploads = Object.entries(uploadProgress).filter(([, p]) => p < 100)
   521→
   522→  return (
   523→    <Card className="border-border/60">
   524→      <CardHeader className="pb-3">
   525→        <CardTitle className="text-sm font-semibold flex items-center gap-2">
   526→          <Upload className="h-4 w-4 text-emerald-600" />
   527→          Document Upload Hub
   528→        </CardTitle>
   529→        <CardDescription className="text-xs">Upload documents for AI-powered extraction and analysis</CardDescription>
   530→      </CardHeader>
   531→      <CardContent className="space-y-4">
   532→        <div className="flex flex-col sm:flex-row gap-3">
   533→          <div className="w-full sm:w-48 shrink-0">
   534→            <Select value={selectedType} onValueChange={(v) => setSelectedType(v as DocType)}>
   535→              <SelectTrigger className="h-9 text-sm">
   536→                <SelectValue placeholder="Document Type" />
   537→              </SelectTrigger>
   538→              <SelectContent>
   539→                {DOC_TYPE_OPTIONS.map(opt => (
   540→                  <SelectItem key={opt.value} value={opt.value} className="text-sm">
   541→                    {opt.label}
   542→                  </SelectItem>
   543→                ))}
   544→              </SelectContent>
   545→            </Select>
   546→          </div>
   547→          <div className="flex-1">
   548→            <div
   549→              onDragOver={handleDragOver}
   550→              onDragLeave={handleDragLeave}
   551→              onDrop={handleDrop}
   552→              onClick={() => fileInputRef.current?.click()}
   553→              className={`
   554→                relative cursor-pointer rounded-xl border-2 border-dashed p-8 text-center transition-all duration-300
   555→                ${isDragging
   556→                  ? 'border-emerald-400 bg-emerald-50/50 scale-[1.01]'
   557→                  : 'border-slate-200 hover:border-emerald-300 hover:bg-slate-50/50'
   558→                }
   559→              `}
   560→            >
   561→              <motion.div
   562→                animate={isDragging ? { scale: 1.05 } : { scale: 1 }}
   563→                transition={{ duration: 0.2 }}
   564→              >
   565→                <div className={`mx-auto mb-3 flex h-12 w-12 items-center justify-center rounded-full transition-colors ${isDragging ? 'bg-emerald-100' : 'bg-slate-100'}`}>
   566→                  <Upload className={`h-6 w-6 transition-colors ${isDragging ? 'text-emerald-600' : 'text-slate-400'}`} />
   567→                </div>
   568→                <p className="text-sm font-medium text-foreground">
   569→                  {isDragging ? 'Drop files here' : 'Drag & drop files here'}
   570→                </p>
   571→                <p className="text-xs text-muted-foreground mt-1">
   572→                  or click to browse — supports {SUPPORTED_FORMATS.join(', ')}
   573→                </p>
   574→              </motion.div>
   575→              <input
   576→                ref={fileInputRef}
   577→                type="file"
   578→                multiple
   579→                accept=".pdf,.png,.jpg,.jpeg,.xlsx,.csv"
   580→                onChange={handleFileSelect}
   581→                className="hidden"
   582→              />
   583→              {isDragging && (
   584→                <motion.div
   585→                  initial={{ opacity: 0 }}
   586→                  animate={{ opacity: 1 }}
   587→                  className="absolute inset-0 rounded-xl border-2 border-emerald-400"
   588→                  style={{
   589→                    background: 'linear-gradient(90deg, transparent, rgba(16,185,129,0.1), transparent)',
   590→                    backgroundSize: '200% 100%',
   591→                    animation: 'shimmer 2s infinite',
   592→                  }}
   593→                />
   594→              )}
   595→            </div>
   596→          </div>
   597→        </div>
   598→
   599→        {/* Active Uploads */}
   600→        <AnimatePresence>
   601→          {activeUploads.length > 0 && (
   602→            <motion.div
   603→              initial={{ opacity: 0, height: 0 }}
   604→              animate={{ opacity: 1, height: 'auto' }}
   605→              exit={{ opacity: 0, height: 0 }}
   606→              className="space-y-2"
   607→            >
   608→              {activeUploads.map(([name, progress]) => (
   609→                <div key={name} className="flex items-center gap-3 rounded-lg border border-border/40 p-2.5 bg-white">
   610→                  <Paperclip className="h-4 w-4 text-slate-400 shrink-0" />
   611→                  <div className="flex-1 min-w-0">
   612→                    <p className="text-xs font-medium truncate">{name}</p>
   613→                    <Progress value={progress} className="h-1.5 mt-1" />
   614→                  </div>
   615→                  <span className="text-[10px] text-muted-foreground shrink-0">{Math.round(progress)}%</span>
   616→                </div>
   617→              ))}
   618→            </motion.div>
   619→          )}
   620→        </AnimatePresence>
   621→      </CardContent>
   622→    </Card>
   623→  )
   624→}
