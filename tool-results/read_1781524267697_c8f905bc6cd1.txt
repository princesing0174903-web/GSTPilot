    35→function generateARN(): string {
    36→  const d = new Date();
    37→  const ds = `${d.getFullYear()}${String(d.getMonth() + 1).padStart(2, '0')}${String(d.getDate()).padStart(2, '0')}`;
    38→  return `AA${ds.slice(6)}${ds.slice(4, 6)}25${String(Math.floor(Math.random() * 999999)).padStart(6, '0')}`;

    45→function currentFirmId(): string | null {
    46→  // Read from localStorage fast-path — the AuthContext stores the firmId there
    47→  try {
    48→    const stored = localStorage.getItem('gstpilot_session');
    49→    if (stored) {

   482→/** Auto-create draft returns for current period */
   483→async function autoCreateDraftReturns(firmId: string, clientId: string): Promise<void> {
   484→  const now = new Date();
   485→  const period = `${String(now.getMonth() + 1).padStart(2, '0')}-${now.getFullYear()}`;
   486→  const fy = now.getMonth() >= 3 ? `${now.getFullYear()}-${String(now.getFullYear() + 1).slice(2)}` : `${now.getFullYear() - 1}-${String(now.getFullYear()).slice(2)}`;
   487→
   488→  for (const returnType of ['GSTR-1', 'GSTR-3B'] as const) {
   489→    // Check if return already exists
   490→    const q = query(
   491→      collection(db, COLLECTIONS.RETURNS),
   492→      where('clientId', '==', clientId),
   493→      where('returnType', '==', returnType),
   494→      where('period', '==', period),
   495→      limit(1)
   496→    );
   497→    const existing = await getDocs(q);
   498→    if (existing.empty) {
   499→      await createReturn({
   500→        clientId,
   501→        returnType,
   502→        period,
   503→        financialYear: fy,
   504→        status: 'draft',
   505→        filedDate: null,
   506→        acknowledgmentNumber: null,
   507→        totalInvoices: 0,
   508→        readyForFiling: 0,
   509→        issuesFound: 0,
   510→        criticalErrors: 0,
   511→        warnings: 0,
   512→        totalTaxableValue: 0,
   513→        totalTax: 0,
   514→        jsonPayload: null,
   515→        assignedTo: null,
   516→        reviewedBy: null,
   517→      });
   518→    }
   519→  }
   520→}

   522→/** Update draft return when an invoice is added */
   523→async function updateDraftReturnForInvoice(firmId: string, clientId: string, returnType: string, period: string, taxableValue: number, taxAmount: number): Promise<void> {
   524→  try {
   525→    const q = query(
   526→      collection(db, COLLECTIONS.RETURNS),
   527→      where('clientId', '==', clientId),
   528→      where('returnType', '==', returnType),
   529→      where('period', '==', period),
   530→      limit(1)
   531→    );
   532→    const snap = await getDocs(q);
   533→    if (!snap.empty) {
   534→      const returnDoc = snap.docs[0];
   535→      const data = returnDoc.data() as FirestoreReturn;
   536→      await updateDoc(returnDoc.ref, {
   537→        totalInvoices: (data.totalInvoices || 0) + 1,
   538→        totalTaxableValue: (data.totalTaxableValue || 0) + taxableValue,
   539→        totalTax: (data.totalTax || 0) + taxAmount,
   540→        updatedAt: serverTimestamp(),
   541→      });
   542→    }
   543→  } catch (e) {
   544→    console.warn('[Workflow] Failed to update draft return:', e);
   545→  }
   546→}

   554→export async function createReconciliation(data: {
   555→  clientId: string;
   556→  period: string;
   557→  sources: string;
   558→}): Promise<string> {
   559→  const firmId = currentFirmId();
   560→  const reconId = generateId();
   561→  const reconRef = doc(db, COLLECTIONS.RECONCILIATIONS, reconId);
   562→
   563→  // Simulate reconciliation results
   564→  const totalRecords = Math.floor(Math.random() * 50) + 10;
   565→  const matched = Math.floor(totalRecords * (0.6 + Math.random() * 0.25));
   566→  const partialMatches = Math.floor((totalRecords - matched) * 0.4);
   567→  const unmatched = totalRecords - matched - partialMatches;
   568→  const highRisk = Math.floor(unmatched * 0.3);
   569→
   570→  const mismatches: ReconMismatch[] = [];
   571→  for (let i = 0; i < unmatched + partialMatches; i++) {
   572→    const booksAmount = Math.floor(Math.random() * 100000) + 5000;
   573→    const diff = Math.floor(booksAmount * (Math.random() * 0.15));
   574→    mismatches.push({
   575→      invoiceNumber: `INV-${String(i + 1).padStart(4, '0')}`,
   576→      invoiceDate: new Date().toISOString().split('T')[0],
   577→      sourceGstin: '27AAACR5055K1ZB',
   578→      matchedGstin: i < partialMatches ? '27AAACR5055K1ZB' : null,
   579→      matchStatus: i < partialMatches ? 'partial_match' as MatchStatus : 'missing_in_books' as MatchStatus,
   580→      matchScore: i < partialMatches ? 70 + Math.floor(Math.random() * 20) : 0,
   581→      booksAmount,
   582→      portalAmount: booksAmount + (Math.random() > 0.5 ? diff : -diff),
   583→      difference: diff,
   584→      reason: i < partialMatches ? 'Amount mismatch' : 'Invoice not found in GSTR-2B',
   585→      resolved: false,
   586→      resolvedBy: null,
   587→      resolvedAt: null,
   588→    });
   589→  }
   590→
   591→  const gstDifference = mismatches.reduce((sum, m) => sum + m.difference, 0);
   592→
   593→  const reconData: FirestoreReconciliation = {
   594→    reconId,
   595→    firmId: firmId || '',
   596→    clientId: data.clientId,
   597→    period: data.period,
   598→    sources: data.sources,
   599→    status: 'completed',
   600→    totalRecords,
   601→    matched,
   602→    unmatched,
   603→    partialMatches,
   604→    highRisk,
   605→    gstDifference,
   606→    mismatches,
   607→    runBy: currentUserId(),
   608→    createdAt: serverTimestamp(),
   609→    updatedAt: serverTimestamp(),
   610→  };
   611→
   612→  await setDoc(reconRef, reconData);
   613→
   614→  addActivity({
   615→    type: 'reconciliation_run',
   616→    title: 'Reconciliation completed',
   617→    description: `${data.sources}: ${matched}/${totalRecords} matched, ${unmatched} unmatched`,
   618→    clientId: data.clientId,
   619→    entityType: COLLECTIONS.RECONCILIATIONS,
   620→    entityId: reconId,
   621→    metadata: { matched, unmatched, gstDifference },
   622→  });
   623→
   624→  // Generate AI recommendations for mismatches
   625→  if (highRisk > 0) {
   626→    generateAIRecommendations(firmId || '', data.clientId, reconId, mismatches);
   627→  }
   628→
   629→  addNotification({
   630→    type: 'mismatch_found',
   631→    priority: highRisk > 3 ? 'urgent' : 'high',
   632→    title: 'Reconciliation mismatches found',
   633→    message: `${unmatched} unmatched and ${partialMatches} partial matches found. ITC difference: ₹${gstDifference.toLocaleString('en-IN')}`,
   634→    entityType: COLLECTIONS.RECONCILIATIONS,
   635→    entityId: reconId,
   636→  });
   637→
   638→  return reconId;
   639→}

   641→export async function resolveMismatch(reconId: string, invoiceNumber: string): Promise<void> {
   642→  const reconRef = doc(db, COLLECTIONS.RECONCILIATIONS, reconId);
   643→  const reconSnap = await getDoc(reconRef);
   644→  if (!reconSnap.exists()) return;
   645→
   646→  const data = reconSnap.data() as FirestoreReconciliation;
   647→  const updatedMismatches = data.mismatches.map(m =>
   648→    m.invoiceNumber === invoiceNumber
   649→      ? { ...m, resolved: true, resolvedBy: currentUserId(), resolvedAt: new Date().toISOString() }
   650→      : m
   651→  );
   652→
   653→  const resolvedCount = updatedMismatches.filter(m => m.resolved).length;
   654→  const newUnmatched = updatedMismatches.filter(m => !m.resolved && m.matchStatus === 'missing_in_books').length;
   655→  const newPartial = updatedMismatches.filter(m => !m.resolved && m.matchStatus === 'partial_match').length;
   656→
   657→  await updateDoc(reconRef, {
   658→    mismatches: updatedMismatches,
   659→    unmatched: newUnmatched,
   660→    partialMatches: newPartial,
   661→    matched: data.totalRecords - newUnmatched - newPartial,
   662→    updatedAt: serverTimestamp(),
   663→  });
   664→
   665→  addActivity({
   666→    type: 'mismatch_resolved',
   667→    title: 'Mismatch resolved',
   668→    description: `Reconciliation mismatch for ${invoiceNumber} resolved`,
   669→    clientId: data.clientId,
   670→    entityType: COLLECTIONS.RECONCILIATIONS,
   671→    entityId: reconId,
   672→  });
   673→}

   878→/** Subscribe to a Firestore collection with firm scoping */
   879→export function subscribeToCollection<T>(
   880→  collectionName: CollectionName,
   881→  callback: (data: Array<T & { id: string }>) => void,
   882→  ...constraints: QueryConstraint[]
   883→): Unsubscribe {
   884→  const firmId = currentFirmId();
   885→
   886→  const q = firmId
   887→    ? query(collection(db, collectionName), where('firmId', '==', firmId), ...constraints)
   888→    : query(collection(db, collectionName), ...constraints);
   889→
   890→  return onSnapshot(q, (snap) => {
   891→    const data = snap.docs.map(d => {
   892→      const docData = d.data();
   893→      const converted: Record<string, unknown> = { id: d.id };
   894→      for (const [key, value] of Object.entries(docData)) {
   895→        if (value && typeof value === 'object' && 'toDate' in value && typeof (value as { toDate: () => Date }).toDate === 'function') {
   896→          converted[key] = (value as { toDate: () => Date }).toDate().toISOString();
   897→        } else {
   898→          converted[key] = value;
   899→        }
   900→      }
   901→      return converted as T & { id: string };
   902→    });
   903→    callback(data);
   904→  }, (error) => {
   905→    console.warn(`[Firestore] Subscription error for ${collectionName}:`, error);
   906→    callback([]);
   907→  });
   908→}

   910→/** Subscribe to a single Firestore document */
   911→export function subscribeToDoc<T>(
   912→  collectionName: CollectionName,
   913→  docId: string,
   914→  callback: (data: (T & { id: string }) | null) => void,
   915→): Unsubscribe {
   916→  return onSnapshot(doc(db, collectionName, docId), (snap) => {
   917→    if (!snap.exists()) {
   918→      callback(null);
   919→      return;
   920→    }
   921→    const docData = snap.data();
   922→    const converted: Record<string, unknown> = { id: snap.id };
   923→    for (const [key, value] of Object.entries(docData)) {
   924→      if (value && typeof value === 'object' && 'toDate' in value && typeof (value as { toDate: () => Date }).toDate === 'function') {
   925→        converted[key] = (value as { toDate: () => Date }).toDate().toISOString();
   926→      } else {
   927→        converted[key] = value;
   928→      }
   929→    }
   930→    callback(converted as T & { id: string });
   931→  }, (error) => {
   932→    console.warn(`[Firestore] Doc subscription error for ${collectionName}/${docId}:`, error);
   933→    callback(null);
   934→  });
   935→}

   937→/** Compute live dashboard metrics from Firestore data */
   938→export function computeDashboardMetrics(
   939→  clients: Array<FirestoreClient & { id: string }>,
   940→  invoices: Array<FirestoreInvoice & { id: string }>,
   941→  returns: Array<FirestoreReturn & { id: string }>,
   942→  documents: Array<FirestoreDocument & { id: string }>,
   943→  activities: Array<FirestoreActivity & { id: string }>,
   944→): LiveDashboardMetrics {
   945→  const activeClients = clients.filter(c => c.status === 'active').length;
   946→  const filedReturns = returns.filter(r => r.status === 'filed').length;
   947→  const pendingReturns = returns.filter(r => r.status !== 'filed').length;
   948→  const readyToFile = returns.filter(r => ['validated', 'reviewed', 'generated'].includes(r.status)).length;
   949→  const overdueReturns = returns.filter(r => {
   950→    if (r.status === 'filed') return false;
   951→    const periodDate = parsePeriod(r.period);
   952→    return periodDate && periodDate < new Date();
   953→  }).length;
   954→
   955→  const totalTaxVolume = invoices.reduce((sum, inv) => sum + (inv.totalAmount || 0), 0);
   956→  const totalTax = invoices.reduce((sum, inv) => sum + (inv.cgst || 0) + (inv.sgst || 0) + (inv.igst || 0) + (inv.cess || 0), 0);
   957→
   958→  const criticalIssues = invoices.filter(inv => inv.riskLevel === 'critical').length +
   959→    returns.filter(r => r.criticalErrors > 0).length;
   960→  const warnings = invoices.filter(inv => inv.riskLevel === 'high' || inv.riskLevel === 'medium').length +
   961→    returns.filter(r => r.warnings > 0).length;
   962→
   963→  const averageHealthScore = clients.length > 0
   964→    ? Math.round(clients.reduce((sum, c) => sum + c.healthScore, 0) / clients.length)
   965→    : 0;
   966→
   967→  const matchedInvoices = invoices.filter(inv => inv.matchStatus === 'perfect_match').length;
   968→  const matchPercentage = invoices.length > 0 ? Math.round((matchedInvoices / invoices.length) * 100) : 100;
   969→
   970→  const riskyInvoices = invoices.filter(inv => ['high', 'critical'].includes(inv.riskLevel)).length;
   971→  const riskPercentage = invoices.length > 0 ? Math.round((riskyInvoices / invoices.length) * 100) : 0;
   972→
   973→  const documentsProcessed = documents.filter(d => d.extractionStatus === 'completed').length;
   974→  const extractionsPending = documents.filter(d => d.extractionStatus === 'pending' || d.extractionStatus === 'in_progress').length;
   975→
   976→  const recentActivities = [...activities]
   977→    .sort((a, b) => {
   978→      const aTime = typeof a.createdAt === 'string' ? new Date(a.createdAt).getTime() : 0;
   979→      const bTime = typeof b.createdAt === 'string' ? new Date(b.createdAt).getTime() : 0;
   980→      return bTime - aTime;
   981→    })
   982→    .slice(0, 10);
   983→
   984→  const upcomingFilings = returns
   985→    .filter(r => r.status !== 'filed')
   986→    .sort((a, b) => {
   987→      const aTime = typeof a.createdAt === 'string' ? new Date(a.createdAt).getTime() : 0;
   988→      const bTime = typeof b.createdAt === 'string' ? new Date(b.createdAt).getTime() : 0;
   989→      return aTime - bTime;
   990→    })
   991→    .slice(0, 5);
   992→
   993→  return {
   994→    totalClients: clients.length,
   995→    activeClients,
   996→    totalInvoices: invoices.length,
   997→    totalTaxVolume,
   998→    filedReturns,
   999→    pendingReturns,
  1000→    overdueReturns,
  1001→    readyToFile,
  1002→    criticalIssues,
  1003→    warnings,
  1004→    averageHealthScore,
  1005→    matchPercentage,
  1006→    riskPercentage,
  1007→    documentsProcessed,
  1008→    extractionsPending,
  1009→    recentActivities,
  1010→    upcomingFilings,
  1011→  };
  1012→}

  1014→function parsePeriod(period: string | null): Date | null {
  1015→  if (!period) return null;
  1016→  try {
  1017→    const [mm, yyyy] = period.split('-').map(Number);
  1018→    if (!mm || !yyyy) return null;
  1019→    return new Date(yyyy, mm, 20); // Due date is typically 20th of next month
  1020→  } catch {
  1021→    return null;
  1022→  }
  1023→}
