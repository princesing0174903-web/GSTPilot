    45→function inrShort(n: number): string {
    46→  if (!isFinite(n) || isNaN(n)) return '₹0';
    47→  const abs = Math.abs(n);
    48→  if (abs >= 10000000) return `₹${(n / 10000000).toFixed(2)} Cr`;
    49→  if (abs >= 100000) return `₹${(n / 100000).toFixed(2)} L`;
    50→  if (abs >= 1000) return `₹${(n / 1000).toFixed(1)}K`;
    51→  return '₹' + Math.round(n).toLocaleString('en-IN');
    52→}

   574→/** Map our role names to the model's role names. */
   575→function toModelMessages(
   576→  messages: OracleChatRequest['messages'],
   577→): { role: 'assistant' | 'user'; content: string }[] {
   578→  return messages.map((m) => ({
   579→    role: (m.role === 'oracle' ? 'assistant' : 'user') as 'assistant' | 'user',
   580→    content: m.content,
   581→  }));
   582→}

   584→/** Server-side language hint derived from the latest user message. */
   585→function inferLanguageHint(messages: OracleChatRequest['messages']): OracleLanguageId | undefined {
   586→  for (let i = messages.length - 1; i >= 0; i--) {
   587→    if (messages[i].role === 'user' && messages[i].content?.trim()) {
   588→      const s = messages[i].content.slice(0, 500);
   589→      if (/[\u0900-\u097F]/.test(s)) return 'hindi';
   590→      if (/[\u0600-\u06FF]/.test(s)) return 'urdu';
   591→      if (/[\u0A00-\u0A7F]/.test(s)) return 'punjabi';
   592→      if (/[\u0A80-\u0AFF]/.test(s)) return 'gujarati';
   593→      if (/[\u0B80-\u0BFF]/.test(s)) return 'tamil';
   594→      if (/[\u0C00-\u0C7F]/.test(s)) return 'telugu';
   595→      if (/[\u0980-\u09FF]/.test(s)) return 'bengali';
   596→      return 'english';
   597→    }
   598→  }
   599→  return undefined;
   600→}

   604→export async function POST(request: Request) {
   605→  let body: OracleChatRequest;
   606→  try {
   607→    body = (await request.json()) as OracleChatRequest;
   608→  } catch {
   609→    return new Response(JSON.stringify({ error: 'Invalid JSON body' }), {
   610→      status: 400,
   611→      headers: { 'Content-Type': 'application/json' },
   612→    });
   613→  }
   614→
   615→  const messages = body.messages ?? [];
   616→  if (!Array.isArray(messages) || messages.length === 0) {
   617→    return new Response(JSON.stringify({ error: 'messages[] is required' }), {
   618→      status: 400,
   619→      headers: { 'Content-Type': 'application/json' },
   620→    });
   621→  }
   622→
   623→  const systemPrompt = await buildSystemPrompt(body);
   624→  const languageHint = inferLanguageHint(messages);
   625→
   626→  const modelMessages: { role: 'assistant' | 'user' | 'system'; content: string }[] = [
   627→    { role: 'assistant', content: systemPrompt },
   628→    ...toModelMessages(messages),
   629→  ];
   630→
   631→  // ── Acquire the upstream stream from the SDK ───────────────────────────────
   632→  let upstream: ReadableStream<Uint8Array> | null = null;
   633→  try {
   634→    const zai = await ZAI.create();
   635→    const result = await zai.chat.completions.create({
   636→      messages: modelMessages,
   637→      stream: true,
   638→      thinking: { type: 'disabled' },
   639→    });
   640→    if (result && typeof (result as ReadableStream<Uint8Array>).getReader === 'function') {
   641→      upstream = result as ReadableStream<Uint8Array>;
   642→    } else {
   643→      // Non-streaming fallback: emit the full text as one chunk then close.
   644→      const text =
   645→        (result as { choices?: { message?: { content?: string } }[] })?.choices?.[0]?.message
   646→          ?.content ?? '';
   647→      upstream = new ReadableStream<Uint8Array>({
   648→        start(controller) {
   649→          if (text) controller.enqueue(sseChunk({ token: text }));
   650→          controller.close();
   651→        },
   652→      });
   653→    }
   654→  } catch (err) {
   655→    const message = err instanceof Error ? err.message : 'Unknown error';
   656→    const stream = new ReadableStream<Uint8Array>({
   657→      start(controller) {
   658→        controller.enqueue(
   659→          sseChunk({
   660→            token: `I'm here, but I hit a temporary issue reaching my reasoning service (${message}). Please try again in a moment — your conversation is safe.`,
   661→          }),
   662→        );
   663→        controller.enqueue(sseChunk({ done: true }));
   664→        controller.close();
   665→      },
   666→    });
   667→    return new Response(stream, { status: 200, headers: sseHeaders() });
   668→  }
   669→
   670→  // ── Transform the upstream SSE stream into our token stream ────────────────
   671→  const decoder = new TextDecoder();
   672→  const transformed = new ReadableStream<Uint8Array>({
   673→    async start(controller) {
   674→      // Emit a tiny first nudge so the UI shows the pulsing cursor within the
   675→      // first frame — real tokens follow immediately.
   676→      if (languageHint) controller.enqueue(sseChunk({ language: languageHint }));
   677→
   678→      const reader = upstream!.getReader();
   679→      let buffer = '';
   680→      let emittedAny = false;
   681→      try {
   682→        while (true) {
   683→          const { done, value } = await reader.read();
   684→          if (done) break;
   685→          buffer += decoder.decode(value, { stream: true });
   686→
   687→          const lines = buffer.split('\n');
   688→          buffer = lines.pop() ?? '';
   689→
   690→          for (const rawLine of lines) {
   691→            const line = rawLine.trim();
   692→            if (!line || !line.startsWith('data:')) continue;
   693→            const data = line.slice(5).trim();
   694→            if (data === '[DONE]' || !data) continue;
   695→            try {
   696→              const json = JSON.parse(data);
   697→              const token: string =
   698→                json?.choices?.[0]?.delta?.content ??
   699→                json?.choices?.[0]?.message?.content ??
   700→                '';
   701→              if (token) {
   702→                emittedAny = true;
   703→                controller.enqueue(sseChunk({ token }));
   704→              }
   705→            } catch {
   706→              // Partial JSON across a chunk boundary — resolves on next read.
   707→            }
   708→          }
   709→        }
   710→        // Flush any trailing buffered line.
   711→        const tail = buffer.trim();
   712→        if (tail.startsWith('data:')) {
   713→          const data = tail.slice(5).trim();
   714→          if (data && data !== '[DONE]') {
   715→            try {
   716→              const json = JSON.parse(data);
   717→              const token: string =
   718→                json?.choices?.[0]?.delta?.content ??
   719→                json?.choices?.[0]?.message?.content ??
   720→                '';
   721→              if (token) {
   722→                emittedAny = true;
   723→                controller.enqueue(sseChunk({ token }));
   724→              }
   725→            } catch {
   726→              /* ignore */
   727→            }
   728→          }
   729→        }
   730→        if (!emittedAny) {
   731→          controller.enqueue(
   732→            sseChunk({
   733→              token:
   734→                "I'm here. Based on current GST rules and the information available, I'd be glad to help — could you share a bit more about what you're looking to do?",
   735→            }),
   736→          );
   737→        }
   738→        controller.enqueue(sseChunk({ done: true }));
   739→        controller.close();
   740→      } catch (err) {
   741→        const message = err instanceof Error ? err.message : 'Stream interrupted';
   742→        if (!emittedAny) {
   743→          controller.enqueue(
   744→            sseChunk({
   745→              token: `My response was interrupted (${message}). Please try sending that again.`,
   746→            }),
   747→          );
   748→        }
   749→        controller.enqueue(sseChunk({ done: true }));
   750→        controller.close();
   751→      }
   752→    },
   753→    cancel() {
   754→      upstream?.cancel?.().catch(() => undefined);
   755→    },
   756→  });
   757→
   758→  return new Response(transformed, { status: 200, headers: sseHeaders() });
   759→}
