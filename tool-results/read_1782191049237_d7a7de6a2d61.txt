    84→    84→
    85→    85→function loadRecent(): RecentItem[] {
    86→    86→  try {
    87→    87→    const raw = localStorage.getItem(RECENT_STORAGE_KEY);
    88→    88→    return raw ? JSON.parse(raw) : [];
    89→    89→  } catch {
    90→    90→    return [];
    91→    91→  }
    92→    92→}
    93→    93→
    94→    94→function saveRecent(items: RecentItem[]) {
    95→    95→  try {
    96→    96→    localStorage.setItem(RECENT_STORAGE_KEY, JSON.stringify(items.slice(0, MAX_RECENT)));
    97→    97→  } catch {
    98→    98→    // ignore quota errors
    99→    99→  }
   100→   100→}
   101→   101→
   102→   102→function loadFavorites(): FavoriteItem[] {
   103→   103→  try {
   104→   104→    const raw = localStorage.getItem(FAVORITES_STORAGE_KEY);

   106→   106→  } catch {
   107→   107→    return [];
   108→   108→  }
   109→   109→}

   111→   111→function saveFavorites(items: FavoriteItem[]) {

   116→   116→  }
   117→   117→}
