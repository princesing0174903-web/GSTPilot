   312→function ReportBuilderWizard({ open, onClose, onComplete }: {
   313→  open: boolean;
   314→  onClose: () => void;
   315→  onComplete: (name: string) => void;
   316→}) {
   317→  const [step, setStep] = useState(1);
   318→  const [template, setTemplate] = useState<ReportTemplate | null>(null);
   319→  const [scope, setScope] = useState<string>('all');
   320→  const [country, setCountry] = useState<string>('IN');
   321→  const [period, setPeriod] = useState<string>('Q3 2024');
   322→  const [format, setFormat] = useState<string>('PDF');
   323→  const [generating, setGenerating] = useState(false);
   324→  const [progress, setProgress] = useState(0);
   325→  const { toast } = useToast();
   326→
   327→  const reset = () => {
   328→    setStep(1);
   329→    setTemplate(null);
   330→    setScope('all');
   331→    setCountry('IN');
   332→    setPeriod('Q3 2024');
   333→    setFormat('PDF');
   334→    setGenerating(false);
   335→    setProgress(0);
   336→  };
   337→
   338→  const handleClose = () => {
   339→    reset();
   340→    onClose();
   341→  };
   342→
   343→  const handleGenerate = () => {
   344→    setGenerating(true);
   345→    setProgress(0);
   346→    // Simulate generation using interval (deterministic increments)
   347→    let p = 0;
   348→    const id = setInterval(() => {
   349→      p += 7;
   350→      if (p >= 100) {
   351→        p = 100;
   352→        clearInterval(id);
   353→        setTimeout(() => {
   354→          setGenerating(false);
   355→          reset();
   356→          onClose();
   357→          onComplete(template?.name ?? 'Report');
   358→        }, 400);
   359→      }
   360→      setProgress(p);
   361→    }, 90);
   362→  };
   363→
   364→  const stepLabels = ['Select Template', 'Select Scope', 'Period & Format'];
   365→
   366→  return (
   367→    <Dialog open={open} onOpenChange={(o) => !o && handleClose()}>
   368→      <DialogContent className="max-w-2xl max-h-[90vh] overflow-hidden border-white/[0.08] bg-background p-0">
   369→        <DialogHeader className="px-5 pt-5 pb-3 border-b border-white/[0.06]">
   370→          <div className="flex items-center justify-between">
   371→            <div className="flex items-center gap-3">
   372→              <div className="flex h-9 w-9 items-center justify-center rounded-lg border border-emerald-500/30 bg-emerald-500/10">
   373→                <Wand2 className="h-4 w-4 text-emerald-400" />
   374→              </div>
   375→              <div>
   376→                <DialogTitle className="text-white text-base">
   377→                  Report Builder Wizard
   378→                </DialogTitle>
   379→                <DialogDescription className="text-[11px]">
   380→                  {generating ? 'Generating your report…' : `Step ${step} of 3 · ${stepLabels[step - 1]}`}
   381→                </DialogDescription>
   382→              </div>
   383→            </div>
   384→            <Button
   385→              variant="ghost"
   386→              size="sm"
   387→              onClick={handleClose}
   388→              className="h-7 w-7 p-0 text-muted-foreground hover:text-white"
   389→            >
   390→              <X className="h-4 w-4" />
   391→            </Button>
   392→          </div>
   393→        </DialogHeader>
   394→
   395→        {/* Progress indicator */}
   396→        {!generating && (
   397→          <div className="px-5 py-2 border-b border-white/[0.06]">
   398→            <div className="flex items-center justify-between">
   399→              {stepLabels.map((label, i) => (
   400→                <div key={label} className="flex items-center gap-2 flex-1">
   401→                  <div className={cn(
   402→                    'flex h-6 w-6 items-center justify-center rounded-full text-[10px] font-bold border',
   403→                    step > i + 1
   404→                      ? 'border-emerald-500/40 bg-emerald-500/20 text-emerald-300'
   405→                      : step === i + 1
   406→                        ? 'border-emerald-500/40 bg-emerald-500/30 text-emerald-200 animate-pulse'
   407→                        : 'border-white/[0.08] bg-white/[0.02] text-muted-foreground',
   408→                  )}>
   409→                    {step > i + 1 ? '✓' : i + 1}
   410→                  </div>
   411→                  <span className={cn(
   412→                    'text-[10px]',
   413→                    step >= i + 1 ? 'text-white' : 'text-muted-foreground',
   414→                  )}>
   415→                    {label}
   416→                  </span>
   417→                  {i < stepLabels.length - 1 && (
   418→                    <div className={cn(
   419→                      'flex-1 h-px',
   420→                      step > i + 1 ? 'bg-emerald-500/40' : 'bg-white/[0.06]',

   425→            </div>

   428→
   429→        <div className="overflow-hidden">
   430→          {generating ? (

   443→              <div className="max-w-xs mx-auto">

   445→                <p className="text-[10px] text-muted-foreground mt-1.5">{progress}%</p>
   446→              </div>
