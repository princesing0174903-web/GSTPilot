    21→// ─── Indian Rupee grouping (₹1,23,456 — not Western 1,23,456) ─────────────────
    22→export function formatInr(n: number): string {
    23→  const neg = n < 0;
    24→  const abs = Math.abs(Math.round(n));
    25→  const digits = abs.toString();
    26→  let grouped: string;
    27→  if (digits.length <= 3) {
    28→    grouped = digits;
    29→  } else {
    30→    const last3 = digits.slice(-3);
    31→    const rest = digits.slice(0, -3);
    32→    grouped = `${rest.replace(/\B(?=(\d{2})+(?!\d))/g, ',')},${last3}`;
    33→  }
    34→  return `${neg ? '-' : ''}₹${grouped}`;
    35→}
    36→

    38→const inr = formatInr;

   412→// ─── Observation Summary — derived from event stream ──────────────────────────
   413→export function getObservationSummary(events: BusinessEvent[]): ObservationSummary {
   414→  const byType = {} as Record<BusinessEventType, number>;
   415→  const bySeverity = {} as Record<EventSeverity, number>;
   416→  for (const t of EVENT_TYPES) byType[t] = 0;
   417→  for (const s of SEVERITIES) bySeverity[s] = 0;
   418→
   419→  let openEvents = 0;
   420→  let criticalEvents = 0;
   421→  let highSeverityEvents = 0;
   422→
   423→  for (const e of events) {
   424→    byType[e.type] = (byType[e.type] ?? 0) + 1;
   425→    bySeverity[e.severity] = (bySeverity[e.severity] ?? 0) + 1;
   426→    if (e.status === 'open' || e.status === 'escalated') openEvents += 1;
   427→    if (e.severity === 'critical') criticalEvents += 1;
   428→    if (e.severity === 'high') highSeverityEvents += 1;
   429→  }
   430→
   431→  const recentEvents = [...events]
   432→    .sort((a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime())
   433→    .slice(0, 8);
   434→
   435→  // Detected issues are derived from high+critical events (the signals worth acting on).
   436→  const detectedIssues: DetectedIssue[] = events
   437→    .filter((e) => e.severity === 'high' || e.severity === 'critical')
   438→    .sort((a, b) => severityRank(b.severity) - severityRank(a.severity))
   439→    .map(deriveIssue);
   440→
   441→  return {
   442→    totalEvents: events.length,
   443→    openEvents,
   444→    criticalEvents,
   445→    highSeverityEvents,
   446→    byType,
   447→    bySeverity,
   448→    recentEvents,
   449→    detectedIssues,
   450→  };
   451→}
   452→
   453→function severityRank(s: EventSeverity): number {
   454→  return { info: 0, low: 1, medium: 2, high: 3, critical: 4 }[s] ?? 0;
   455→}
   456→
   457→function deriveIssue(e: BusinessEvent): DetectedIssue {
   458→  const p = (e.payload ?? {}) as Record<string, unknown>;
   459→  const num = (k: string): number | null =>
   460→    typeof p[k] === 'number' ? (p[k] as number) : null;
   461→  const str = (k: string): string | null =>
   462→    typeof p[k] === 'string' ? (p[k] as string) : null;
   463→
   464→  switch (e.type) {
   465→    case 'gst_due': {
   466→      const form = str('form') ?? 'GST return';
   467→      const net = num('netPayable') ?? num('outputTaxLiability') ?? num('itcExpected');
   468→      const due = str('dueDate') ?? (num('dueInHours') != null ? `${num('dueInHours')}h` : 'soon');
   469→      return {
   470→        type: e.type,
   471→        title: `${form} window open`,
   472→        description:
   473→          `${form} for ${str('period') ?? 'current period'} due ${due}` +
   474→          (net != null ? ` — ${net >= 0 ? 'net liability' : 'net refund'} ${inr(Math.abs(net))}` : '') +
   475→          `. Late fee ₹50/day + 18% p.a. interest on delayed tax.`,
   476→        severity: e.severity,
   477→        suggestedAction:
   478→          'Prepare return immediately; verify cash ledger balance; file before due date to avoid penalty exposure.',
   479→      };
   480→    }
   481→    case 'bank_transaction': {
   482→      const amt = num('amount');
   483→      const dir = str('direction') ?? 'movement';
   484→      const cp = str('counterparty') ?? 'counterparty';
   485→      return {
   486→        type: e.type,
   487→        title: `Bank ${dir} detected — ${amt != null ? inr(amt) : 'unknown amount'}`,
   488→        description:
   489→          `${dir === 'credit' ? 'Credit' : 'Debit'} ${amt != null ? inr(amt) : ''} via ` +
   490→          `${str('mode') ?? 'bank'} from ${cp}. Reference: ${str('reference') ?? 'n/a'}.`,
   491→        severity: e.severity,
   492→        suggestedAction:
   493→          'Auto-reconcile against open invoices / payables; flag if unmatched or outside expected pattern.',
   494→      };
   495→    }
   496→    case 'receivable': {
   497→      const amt = num('amount') ?? num('totalOutstanding');
   498→      const client = str('client') ?? 'client';
   499→      const days = num('overdueDays');
   500→      return {
   501→        type: e.type,
   502→        title: `Overdue receivable — ${client}`,
   503→        description:
   504→          `${client} invoice ${str('invoiceNo') ?? ''} ${amt != null ? inr(amt) : ''} ` +
   505→          (days != null ? `${days} days overdue (DSO ${num('dso') ?? '?'})` : 'overdue') + '.',
   506→        severity: e.severity,
   507→        suggestedAction:
   508→          'Trigger collection recovery workflow; send WhatsApp + email reminder; escalate if >60 days.',
   509→      };
   510→    }
   511→    case 'payable': {
   512→      const amt = num('amount');
   513→      const vendor = str('vendor') ?? 'vendor';
   514→      const due = num('dueInDays');
   515→      return {
   516→        type: e.type,
   517→        title: `Payable approaching due — ${vendor}`,
   518→        description:
   519→          `${vendor} bill ${amt != null ? inr(amt) : ''} due in ${due ?? '?'} days` +
   520→          (str('earlyPayDiscount') ? ` — ${str('earlyPayDiscount')}` : '') + '.',
   521→        severity: e.severity,
   522→        suggestedAction:
   523→          'Schedule payment via Payment Engine; capture early-pay discount if offered; preserve runway.',
   524→      };
   525→    }
   526→    case 'payroll': {
   527→      const cycle = str('cycle') ?? 'cycle';
   528→      const net = num('netPayable');
   529→      const emp = num('employeeCount');
   530→      return {
   531→        type: e.type,
   532→        title: `Payroll run pending — ${cycle}`,
   533→        description:
   534→          `${emp ?? '?'} employees, gross ${num('grossPayroll') != null ? inr(num('grossPayroll')!) : ''}, ` +
   535→          `net ${net != null ? inr(net) : ''}. PF ${num('pfDeduction') ?? 0} + TDS ${num('tdsDeduction') ?? 0}.`,
   536→        severity: e.severity,
   537→        suggestedAction:
   538→          'Lock attendance; compute PF / ESI / TDS; release net-pay bank file before pay-date.',
   539→      };
   540→    }
   541→    case 'tds': {
   542→      const q = str('quarter') ?? str('section') ?? 'review';
   543→      const amt = num('totalTDS') ?? num('amount');
   544→      return {
   545→        type: e.type,
   546→        title: `TDS compliance gap — ${q}`,
   547→        description:
   548→          `${str('returnForm') ?? 'Return'} for ${q} — ${amt != null ? inr(amt) : ''} ` +
   549→          `pending. Due: ${str('dueDate') ?? 'per schedule'}. ` +
   550→          `Challan status: ${str('challanStatus') ?? 'unpaid'}.`,
   551→        severity: e.severity,
   552→        suggestedAction:
   553→          'Deposit pending challan via ITNS-281; prepare 26Q return; file before statutory due date.',
   554→      };
   555→    }
   556→    case 'client_behaviour': {
   557→      const client = str('client') ?? 'client';
   558→      const avg = num('observedAvgPaymentDays');
   559→      const terms = num('contractualTerms');
   560→      const seg = str('riskSegment') ?? 'unclassified';
   561→      return {
   562→        type: e.type,
   563→        title: `Payment-pattern shift — ${client}`,
   564→        description:
   565→          `${client} pays in ${avg ?? '?'} days vs ${terms ?? '?'}-day terms. ` +
   566→          `Risk segment: ${seg}. Confidence ${(num('confidence') ?? 0) * 100}%.`,
   567→        severity: e.severity,
   568→        suggestedAction:
   569→          'Update risk-segment; tighten credit terms for amber/red clients; offer early-pay discount to green.',
   570→      };
   571→    }
   572→    case 'cash_flow': {
   573→      const runway = num('runwayDays');
   574→      const deficit = num('netDeficit');
   575→      const balance = num('currentBankBalance');
   576→      return {
   577→        type: e.type,
   578→        title: 'Cash-flow stress predicted',
   579→        description:
   580→          `Runway ${runway != null ? `${runway} days` : 'low'}` +
   581→          (balance != null ? ` • bank balance ${inr(balance)}` : '') +
   582→          (deficit != null ? ` • projected deficit ${inr(Math.abs(deficit))}` : '') + '.',
   583→        severity: e.severity,
   584→        suggestedAction:
   585→          'Activate invoice-discounting bridge; defer non-essential capex; alert CFO + renegotiate vendor terms.',
   586→      };
   587→    }
   588→    case 'collection_risk': {
   589→      const client = str('client') ?? 'client';
   590→      const amt = num('amount');
   591→      const prob = num('defaultProbability');
   592→      const stage = str('recoveryStage') ?? 'pending';
   593→      return {
   594→        type: e.type,
   595→        title: `Default-risk receivable — ${client}`,
   596→        description:
   597→          `${client} ${amt != null ? inr(amt) : ''} overdue ${num('overdueDays') ?? '?'} days. ` +
   598→          `Default probability ${prob != null ? `${Math.round(prob * 100)}%` : 'high'}. ` +
   599→          `Recovery stage: ${stage}.`,
   600→        severity: e.severity,
   601→        suggestedAction:
   602→          'Escalate to legal recovery; issue IBC Section 9 notice (MSME); provision doubtful debt in books.',
   603→      };
   604→    }
   605→    default:
   606→      return {
   607→        type: e.type,
   608→        title: `Signal detected — ${e.type}`,
   609→        description: 'Auto-generated issue from observation engine.',
   610→        severity: e.severity,
   611→        suggestedAction: 'Review event payload and decide on next action.',
   612→      };
   613→  }
   614→}

   616→// ─── Live-data integration: scan Invoice Engine records for issues ────────────
   617→// Pure — accepts plain objects (no Prisma types), returns DetectedIssue[].
   618→export function detectEvents(
   619→  invoices?: Record<string, unknown>[],
   620→  receivables?: Record<string, unknown>[],
   621→  payables?: Record<string, unknown>[],
   622→): DetectedIssue[] {
   623→  const issues: DetectedIssue[] = [];
   624→  const now = Date.now();
   625→
   626→  if (invoices && invoices.length > 0) {
   627→    const overdue = invoices.filter((inv) => {
   628→      const dueRaw = inv.dueDate as string | undefined;
   629→      const status = inv.paymentStatus as string | undefined;
   630→      if (!dueRaw || status === 'paid' || status === 'settled') return false;
   631→      const dueMs = new Date(dueRaw).getTime();
   632→      return Number.isFinite(dueMs) && dueMs < now;
   633→    });
   634→    if (overdue.length > 0) {
   635→      const total = overdue.reduce(
   636→        (s, inv) => s + Number(inv.total ?? inv.amount ?? 0),
   637→        0,
   638→      );
   639→      issues.push({
   640→        type: 'receivable',
   641→        title: `${overdue.length} overdue sales invoice(s) detected`,
   642→        description: `Total overdue value ${inr(total)} across ${overdue.length} invoice(s).`,
   643→        severity: total > 500000 ? 'high' : 'medium',
   644→        suggestedAction:
   645→          'Trigger collection recovery workflow + send payment reminders via WhatsApp / email.',
   646→      });
   647→    }
   648→  }
   649→
   650→  if (receivables && receivables.length > 0) {
   651→    const risky = receivables.filter((r) => {
   652→      const days = Number(r.overdueDays ?? 0);
   653→      return days >= 60;
   654→    });
   655→    if (risky.length > 0) {
   656→      const total = risky.reduce((s, r) => s + Number(r.amount ?? 0), 0);
   657→      issues.push({
   658→        type: 'collection_risk',
   659→        title: `${risky.length} receivable(s) in 60+ day bucket`,
   660→        description: `High-risk pool totalling ${inr(total)} — escalate before books close.`,
   661→        severity: total > 1000000 ? 'critical' : 'high',
   662→        suggestedAction:
   663→          'Escalate to legal recovery; consider IBC Section 9 notice for MSME clients.',
   664→      });
   665→    }
   666→  }
   667→
   668→  if (payables && payables.length > 0) {
   669→    const dueSoon = payables.filter((p) => {
   670→      const days = Number(p.dueInDays ?? 999);
   671→      return days <= 7;
   672→    });
   673→    if (dueSoon.length > 0) {
   674→      const total = dueSoon.reduce((s, p) => s + Number(p.amount ?? 0), 0);
   675→      issues.push({
   676→        type: 'payable',
   677→        title: `${dueSoon.length} payable(s) due within 7 days`,
   678→        description: `Vendor payouts totalling ${inr(total)} scheduled this week.`,
   679→        severity: total > 500000 ? 'high' : 'medium',
   680→        suggestedAction:
   681→          'Schedule payments via Payment Engine; capture early-pay discounts where offered.',
   682→      });
   683→    }
   684→  }
   685→
   686→  return issues;
   687→}

   689→// ─── Lightweight scalar-issue detector (for Oracle + dashboard quick-checks) ──
   690→export function detectIssues(opts: {
   691→  overdueInvoices?: number;
   692→  pendingTds?: number;
   693→  cashRunwayDays?: number;
   694→}): DetectedIssue[] {
   695→  const issues: DetectedIssue[] = [];
   696→
   697→  if (typeof opts.overdueInvoices === 'number' && opts.overdueInvoices > 0) {
   698→    issues.push({
   699→      type: 'receivable',
   700→      title: `${opts.overdueInvoices} overdue invoice(s) flagged`,
   701→      description: `${opts.overdueInvoices} client invoice(s) have crossed their due date without payment receipt.`,
   702→      severity: opts.overdueInvoices >= 5 ? 'critical' : 'high',
   703→      suggestedAction:
   704→        'Trigger collection recovery workflow; send WhatsApp + email reminders; escalate after 60 days.',
   705→    });
   706→  }
   707→
   708→  if (typeof opts.pendingTds === 'number' && opts.pendingTds > 0) {
   709→    issues.push({
   710→      type: 'tds',
   711→      title: `TDS pending — ${inr(opts.pendingTds)} liability`,
   712→      description: `Outstanding TDS deposit of ${inr(opts.pendingTds)}; challan ITNS-281 + 26Q return preparation required.`,
   713→      severity: opts.pendingTds >= 200000 ? 'high' : 'medium',
   714→      suggestedAction:
   715→        'Generate ITNS-281 challan; deposit before statutory due date; prepare 26Q return.',
   716→    });
   717→  }
   718→
   719→  if (typeof opts.cashRunwayDays === 'number' && opts.cashRunwayDays < 15) {
   720→    issues.push({
   721→      type: 'cash_flow',
   722→      title: `Cash runway critical — ${opts.cashRunwayDays} day(s)`,
   723→      description: `Predicted cash shortage in ${opts.cashRunwayDays} day(s); immediate working-capital action required.`,
   724→      severity: opts.cashRunwayDays < 7 ? 'critical' : 'high',
   725→      suggestedAction:
   726→        'Activate invoice-discounting bridge; defer non-essential capex; alert CFO.',
   727→    });
   728→  }
   729→
   730→  return issues;
   731→}
