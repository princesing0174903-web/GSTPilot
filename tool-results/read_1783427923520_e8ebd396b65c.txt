    64→function getDaysRemaining(dueDateStr: string): number {
    65→  const dueDate = new Date(dueDateStr);
    66→  const today = new Date();
    67→  today.setHours(0, 0, 0, 0);
    68→  dueDate.setHours(0, 0, 0, 0);
    69→  return Math.ceil((dueDate.getTime() - today.getTime()) / (1000 * 60 * 60 * 24));
    70→}
    71→
    72→function formatDaysRemaining(days: number): string {
    73→  if (days < 0) return `${Math.abs(days)}d overdue`;
    74→  if (days === 0) return 'Due today';
    75→  if (days === 1) return 'Tomorrow';
    76→  return `${days}d left`;
    77→}
    78→
    79→function formatINR(amount: number): string {
    80→  return amount.toLocaleString('en-IN');
    81→}
    82→
    83→function formatDateIN(dateStr: string): string {
    84→  const d = new Date(dateStr);
    85→  if (Number.isNaN(d.getTime())) return '—';
    86→  const dd = String(d.getDate()).padStart(2, '0');
    87→  const mm = String(d.getMonth() + 1).padStart(2, '0');
    88→  const yyyy = d.getFullYear();
    89→  return `${dd}/${mm}/${yyyy}`;
    90→}
    91→
    92→function timeAgo(dateStr: string): string {
    93→  const now = new Date();
    94→  const date = new Date(dateStr);
    95→  const seconds = Math.floor((now.getTime() - date.getTime()) / 1000);
    96→  if (seconds < 60) return 'just now';
    97→  const minutes = Math.floor(seconds / 60);
    98→  if (minutes < 60) return `${minutes}m ago`;
    99→  const hours = Math.floor(minutes / 60);
   100→  if (hours < 24) return `${hours}h ago`;
   101→  const days = Math.floor(hours / 24);
   102→  if (days < 7) return `${days}d ago`;
   103→  return date.toLocaleDateString('en-IN', { day: 'numeric', month: 'short' });
   104→}

   106→function getGreeting(): string {
   107→  const h = new Date().getHours();
   108→  if (h < 12) return 'Good Morning';
   109→  if (h < 17) return 'Good Afternoon';
   110→  return 'Good Evening';
   111→}

   119→// One-sentence AI insight derived from live metrics.
   120→function buildInsightSentence(
   121→  metrics: LiveDashboardMetrics,
   122→  pendingCollection: number,
   123→): string {
   124→  const parts: string[] = [];
   125→  if (metrics.overdueReturns > 0) {
   126→    parts.push(
   127→      `${metrics.overdueReturns} overdue return${metrics.overdueReturns > 1 ? 's' : ''}`,
   128→    );
   129→  } else if (metrics.pendingReturns > 0) {
   130→    parts.push(
   131→      `${metrics.pendingReturns} return${metrics.pendingReturns > 1 ? 's' : ''} to file`,
   132→    );
   133→  }
   134→  if (pendingCollection > 0) {
   135→    parts.push(`₹${formatINR(pendingCollection)} pending collection`);
   136→  }
   137→  if (metrics.criticalIssues > 0) {
   138→    parts.push(
   139→      `${metrics.criticalIssues} critical issue${metrics.criticalIssues > 1 ? 's' : ''}`,
   140→    );
   141→  }
   142→  if (parts.length === 0) {
   143→    return `All clear — ${metrics.filedReturns} returns filed and ${metrics.totalClients} clients in good standing.`;
   144→  }
   145→  if (parts.length === 1) return `${parts[0]}.`;
   146→  const last = parts.pop();
   147→  return `${parts.join(', ')} and ${last}.`;
   148→}

   267→function healthTier(score: number): { label: string; tone: string } {
   268→  if (score >= 85) return { label: 'Excellent', tone: 'text-emerald-400' };
   269→  if (score >= 70) return { label: 'Healthy', tone: 'text-emerald-400' };
   270→  if (score >= 50) return { label: 'At Risk', tone: 'text-amber-400' };
   271→  return { label: 'Critical', tone: 'text-amber-400' };
   272→}
   273→

   286→  const offset = c * (1 - pct);
