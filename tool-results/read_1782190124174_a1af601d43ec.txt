    65→const EASE = [0.22, 1, 0.36, 1] as const;

    67→const revealVariants: Variants = {

    72→function Reveal({
    73→  children,
    74→  delay = 0,
    75→  className,
    76→  as = 'div',
    77→}: {
    78→  children: React.ReactNode;
    79→  delay?: number;
    80→  className?: string;
    81→  as?: 'div' | 'section' | 'li' | 'span' | 'h2' | 'p';
    82→}) {
    83→  const MotionTag = motion[as] as typeof motion.div;
    84→  return (
    85→    <MotionTag
    86→      className={className}
    87→      variants={revealVariants}
    88→      initial="hidden"
    89→      whileInView="visible"
    90→      viewport={{ once: true, margin: '-80px' }}
    91→      transition={{ duration: 0.7, ease: EASE, delay }}
    92→    >
    93→      {children}
    94→    </MotionTag>
    95→  );
    96→}

    98→function StaggerGroup({
    99→  children,
   100→  className,
   101→  stagger = 0.08,
   102→}: {
   103→  children: React.ReactNode;
   104→  className?: string;
   105→  stagger?: number;
   106→}) {
   107→  return (
   108→    <motion.div
   109→      className={className}
   110→      initial="hidden"
   111→      whileInView="visible"
   112→      viewport={{ once: true, margin: '-60px' }}
   113→      variants={{
   114→        hidden: {},
   115→        visible: { transition: { staggerChildren: stagger } },
   116→      }}
   117→    >
   118→      {children}
   119→    </motion.div>
   120→  );
   121→}

   959→function AnimatedCounter({ target, prefix = '', suffix = '', decimals = 0 }: { target: number; prefix?: string; suffix?: string; decimals?: number }) {
   960→  const ref = useRef<HTMLSpanElement>(null);
   961→  const inView = useInView(ref, { once: true, margin: '-60px' });
   962→  const [val, setVal] = useState(0);
   963→
   964→  useEffect(() => {
   965→    if (!inView) return;
   966→    const duration = 1800;
   967→    const start = performance.now();
   968→    let raf = 0;
   969→    const tick = (now: number) => {
   970→      const t = Math.min((now - start) / duration, 1);
   971→      const eased = 1 - Math.pow(1 - t, 3);
   972→      setVal(target * eased);
   973→      if (t < 1) raf = requestAnimationFrame(tick);
   974→      else setVal(target);
   975→    };
   976→    raf = requestAnimationFrame(tick);
   977→    return () => cancelAnimationFrame(raf);
   978→  }, [inView, target]);
   979→
   980→  const formatted =
   981→    decimals > 0
   982→      ? val.toFixed(decimals)
   983→      : Math.round(val).toLocaleString('en-IN');
   984→
   985→  return (
   986→    <span ref={ref}>
   987→      {prefix}
   988→      {formatted}
   989→      {suffix}
   990→    </span>
   991→  );
   992→}
