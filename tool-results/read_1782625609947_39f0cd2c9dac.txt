    57→// ─── Safe wrappers (engine failures don't break the dashboard) ───────────────

    59→function safe<T>(label: string, fn: () => Promise<T>, fallback: T): Promise<T> {
    60→  return fn().catch((err) => {
    61→    console.warn(`[AI Workforce] Engine "${label}" failed:`, err);
    62→    return fallback;
    63→  });
    64→}
    65→
    66→function safeSync<T>(label: string, fn: () => T, fallback: T): T {
    67→  try {
    68→    return fn();
    69→  } catch (err) {
    70→    console.warn(`[AI Workforce] Engine "${label}" failed:`, err);
    71→    return fallback;
    72→  }
    73→}

    98→function syncDecisionStore(decisions: CrossDepartmentDecision[]) {
    99→  for (const d of decisions) {
   100→    const existing = decisionStore.get(d.id);
   101→    if (!existing) {
   102→      decisionStore.set(d.id, d);
   103→    } else if (existing.status === 'proposed' || existing.status === 'step_1_approved' || existing.status === 'step_2_approved') {
   104→      // Preserve progress but refresh data
   105→      decisionStore.set(d.id, { ...d, status: existing.status, currentStepIndex: existing.currentStepIndex, steps: existing.steps });
   106→    }
   107→    // If fully_approved/executing/executed/rejected — preserve user's action
   108→  }
   109→  // Drop stale proposed decisions no longer being generated
   110→  const liveIds = new Set(decisions.map((d) => d.id));
   111→  for (const [id, d] of decisionStore.entries()) {
   112→    if (!liveIds.has(id) && (d.status === 'proposed')) {
   113→      decisionStore.delete(id);
   114→    }
   115→  }
   116→}

   120→function buildDepartmentDashboard(
   121→  dept: Department,
   122→  employees: AIEmployee[],
   123→  data: WorkforceDataView,
   124→): DepartmentDashboard {
   125→  const def = DEPARTMENTS.find((d) => d.department === dept) ?? DEPARTMENTS[0];
   126→  const members = employees.filter((e) => def.members.includes(e.role));
   127→  const lead = members.find((e) => e.role === def.lead) ?? members[0];
   128→  const healthScore = members.length > 0
   129→    ? Math.round(members.reduce((s, e) => s + e.healthScore, 0) / members.length)
   130→    : 0;
   131→  const kpis = lead?.kpis ?? [];
   132→  const openTasks = members.reduce((s, e) => s + e.openTaskCount, 0);
   133→  const alerts = members.reduce((s, e) => s + e.activeAlertCount, 0);
   134→  const aiDecisions = members.reduce((s, e) => s + e.pendingDecisionCount, 0);
   135→  const recommendations = members.flatMap((e) => e.recommendations).slice(0, 5);
   136→
   137→  // Department goals (derived from real data)
   138→  const goals = buildDepartmentGoals(dept, data);
   139→
   140→  const status: 'green' | 'amber' | 'red' = healthScore > 70 ? 'green' : healthScore > 40 ? 'amber' : 'red';
   141→
   142→  return {
   143→    department: dept,
   144→    name: def.name,
   145→    lead: def.lead,
   146→    members: def.members,
   147→    healthScore,
   148→    kpis,
   149→    goals,
   150→    openTasks,
   151→    alerts,
   152→    aiDecisions,
   153→    recommendations,
   154→    status,
   155→  };
   156→}

   158→function buildDepartmentGoals(dept: Department, data: WorkforceDataView): DepartmentGoal[] {
   159→  const goals: DepartmentGoal[] = [];
   160→  const now = new Date();
   161→  const endOfYear = new Date(now.getFullYear(), 11, 31).toISOString();
   162→
   163→  switch (dept) {
   164→    case 'finance':
   165→      goals.push({ title: 'Revenue Target', current: data.finance.revenue, target: Math.max(data.finance.revenue * 1.2, 500000), unit: 'inr', progressPct: data.finance.revenue > 0 ? Math.min(100, (data.finance.revenue / Math.max(data.finance.revenue * 1.2, 1)) * 100) : 0, deadline: endOfYear, status: 'on_track' });
   166→      goals.push({ title: 'Margin Improvement', current: data.finance.marginPct, target: 20, unit: 'pct', progressPct: Math.min(100, (data.finance.marginPct / 20) * 100), deadline: endOfYear, status: data.finance.marginPct >= 15 ? 'on_track' : 'at_risk' });
   167→      break;
   168→    case 'sales':
   169→      goals.push({ title: 'Client Growth', current: data.sales.clientCount, target: Math.max(data.sales.clientCount + 5, 10), unit: 'count', progressPct: Math.min(100, (data.sales.clientCount / Math.max(10, 1)) * 100), deadline: endOfYear, status: 'on_track' });
   170→      goals.push({ title: 'Pipeline Value', current: data.sales.pipelineValue, target: 1000000, unit: 'inr', progressPct: Math.min(100, (data.sales.pipelineValue / 1000000) * 100), deadline: endOfYear, status: data.sales.pipelineValue > 500000 ? 'on_track' : 'at_risk' });
   171→      break;
   172→    case 'compliance':
   173→      goals.push({ title: 'Zero Overdue Filings', current: data.compliance.overdueFilings, target: 0, unit: 'count', progressPct: data.compliance.overdueFilings === 0 ? 100 : 0, deadline: endOfYear, status: data.compliance.overdueFilings === 0 ? 'achieved' : 'behind' });
   174→      goals.push({ title: 'Compliance Score', current: data.compliance.complianceScore, target: 90, unit: 'pct', progressPct: Math.min(100, (data.compliance.complianceScore / 90) * 100), deadline: endOfYear, status: data.compliance.complianceScore >= 80 ? 'on_track' : 'at_risk' });
   175→      break;
   176→    case 'risk':
   177→      goals.push({ title: 'Risk Score Reduction', current: data.risk.overallRiskScore, target: 20, unit: 'pct', progressPct: Math.min(100, Math.max(0, 100 - (data.risk.overallRiskScore / 20) * 100)), deadline: endOfYear, status: data.risk.overallRiskScore < 30 ? 'on_track' : 'at_risk' });
   178→      break;
   179→    case 'operations':
   180→      goals.push({ title: 'Efficiency Target', current: data.operations.efficiencyPct, target: 85, unit: 'pct', progressPct: Math.min(100, (data.operations.efficiencyPct / 85) * 100), deadline: endOfYear, status: data.operations.efficiencyPct >= 75 ? 'on_track' : 'at_risk' });
   181→      break;
   182→    case 'support':
   183→      goals.push({ title: 'Satisfaction Target', current: data.support.satisfactionPct, target: 90, unit: 'pct', progressPct: Math.min(100, (data.support.satisfactionPct / 90) * 100), deadline: endOfYear, status: data.support.satisfactionPct >= 80 ? 'on_track' : 'at_risk' });
   184→      break;
   185→    default:
   186→      goals.push({ title: `${dept} Health`, current: 0, target: 80, unit: 'pct', progressPct: 0, deadline: endOfYear, status: 'on_track' });
   187→  }
   188→  return goals;
   189→}

   191→// ─── Aggregate metrics builder ───────────────────────────────────────────────
   192→
   193→function buildAggregateMetrics(
   194→  employees: AIEmployee[],
   195→  collaborationFeed: ReturnType<typeof computeCollaborationFeed>,
   196→  meetings: ReturnType<typeof computeMeetings>,
   197→  crossDecisions: CrossDepartmentDecision[],
   198→): WorkforceAggregateMetrics {
   199→  const activeEmployees = employees.filter((e) => e.status !== 'idle' && e.status !== 'offline').length;
   200→  const totalTasksCompleted = employees.reduce((s, e) => s + e.performance.tasksCompleted, 0);
   201→  const totalROIGenerated = employees.reduce((s, e) => s + e.performance.roiGenerated, 0);
   202→  const totalRevenueInfluenced = employees.reduce((s, e) => s + e.performance.revenueInfluenced, 0);
   203→  const totalCostSaved = employees.reduce((s, e) => s + e.performance.costSaved, 0);
   204→  const automationRate = employees.length > 0
   205→    ? Math.round(employees.reduce((s, e) => s + e.performance.automationSuccess, 0) / employees.length)
   206→    : 0;
   207→  const avgHealthScore = employees.length > 0
   208→    ? Math.round(employees.reduce((s, e) => s + e.healthScore, 0) / employees.length)
   209→    : 0;
   210→
   211→  // Count today's collaboration messages
   212→  const today = new Date().toDateString();
   213→  const collaborationMessagesToday = collaborationFeed.filter(
   214→    (m) => new Date(m.timestamp).toDateString() === today,
   215→  ).length;
   216→
   217→  // Count this week's meetings
   218→  const weekFromNow = new Date();
   219→  weekFromNow.setDate(weekFromNow.getDate() + 7);
   220→  const meetingsThisWeek = meetings.filter((m) => {
   221→    const meetingDate = new Date(m.scheduledFor);
   222→    return meetingDate >= new Date() && meetingDate <= weekFromNow;
   223→  }).length;
   224→
   225→  return {
   226→    totalEmployees: employees.length,
   227→    activeEmployees,
   228→    totalTasksCompleted,
   229→    totalDecisions: crossDecisions.length,
   230→    totalROIGenerated,
   231→    totalRevenueInfluenced,
   232→    totalCostSaved,
   233→    automationRatePct: automationRate,
   234→    avgHealthScore,
   235→    collaborationMessagesToday,
   236→    meetingsThisWeek,
   237→    activeCrossDecisions: crossDecisions.filter(
   238→      (d) => d.status !== 'executed' && d.status !== 'rejected',
   239→    ).length,
   240→  };
   241→}

   245→export async function computeWorkforceDashboard(): Promise<WorkforceDashboard> {
   246→  const data = await fetchWorkforceData();
   247→
   248→  // Run all engines in parallel (each wrapped in safeSync)
   249→  const [organization, collaborationFeed, chains, meetings, crossDecisionsRaw] = await Promise.all([
   250→    safeSync('organization', () => computeOrganization(data), [] as AIEmployee[]),
   251→    safeSync('collaboration', () => computeCollaborationFeed(data), []),
   252→    safeSync('chains', () => computeCollaborationChains(data), []),
   253→    safeSync('meetings', () => computeMeetings(data), []),
   254→    safeSync('cross-decisions', () => computeCrossDecisions(data), [] as CrossDepartmentDecision[]),
   255→  ]);
   256→
   257→  // Sync the in-memory decision store
   258→  syncDecisionStore(crossDecisionsRaw);
   259→
   260→  // Use the MERGED store decisions (preserves user-approved statuses)
   261→  const crossDecisions = crossDecisionsRaw
   262→    .map((d) => decisionStore.get(d.id) ?? d)
   263→    .sort((a, b) => Math.abs(b.financialImpact) - Math.abs(a.financialImpact));
   264→
   265→  // Build department dashboards
   266→  const departments = DEPARTMENTS.map((dept) =>
   267→    safeSync(`dept-${dept.department}`, () => buildDepartmentDashboard(dept.department, organization, data), null),
   268→  ).filter(Boolean) as DepartmentDashboard[];
   269→
   270→  // Build performance leaderboard
   271→  const employeesWithPerf: EmployeeWithPerformance[] = organization.map((e) => ({
   272→    role: e.role,
   273→    name: e.name,
   274→    department: e.department,
   275→    performance: e.performance,
   276→  }));
   277→  const performance = safeSync('leaderboard', () => buildLeaderboard(employeesWithPerf), []);
   278→
   279→  // Update each employee's performance rank from the leaderboard
   280→  const rankedOrg = organization.map((e) => {
   281→    const entry = performance.find((p) => p.role === e.role);
   282→    return entry ? { ...e, performance: { ...e.performance, rank: entry.rank } } : e;
   283→  });
   284→
   285→  // Top employee + top collaboration
   286→  const topEmployee = performance[0] ?? null;
   287→  const topCollaboration = chains[0] ?? null;
   288→
   289→  // Pending counts
   290→  const pendingCrossDecisions = crossDecisions.filter(
   291→    (d) => d.status === 'proposed' || d.status === 'step_1_approved' || d.status === 'step_2_approved',
   292→  );
   293→  const openDelegations = delegationStore.filter(
   294→    (d) => d.status === 'pending' || d.status === 'accepted',
   295→  );
   296→  const openEscalations = escalationStore.filter((e) => e.status === 'open');
   297→
   298→  // Aggregate metrics
   299→  const aggregateMetrics = safeSync('aggregate', () =>
   300→    buildAggregateMetrics(rankedOrg, collaborationFeed, meetings, crossDecisions),
   301→    null,
   302→  ) ?? {
   303→    totalEmployees: 0, activeEmployees: 0, totalTasksCompleted: 0, totalDecisions: 0,
   304→    totalROIGenerated: 0, totalRevenueInfluenced: 0, totalCostSaved: 0, automationRatePct: 0,
   305→    avgHealthScore: 0, collaborationMessagesToday: 0, meetingsThisWeek: 0, activeCrossDecisions: 0,
   306→  };
   307→
   308→  // Oracle capabilities
   309→  const oracleCapabilities = [
   310→    'Ask "what is each department working on?" for live department status',
   311→    'Ask "which AI employee is performing best?" for the leaderboard',
   312→    'Ask "what meetings are scheduled?" for the meeting agenda',
   313→    'Ask "what decisions need approval?" for pending cross-department decisions',
   314→    'Ask "show me the collaboration feed" for recent inter-department messages',
   315→    'Ask "delegate a task to a human" to create a delegation',
   316→    'Ask "escalate to the CEO" for urgent matters',
   317→    'Never fabricate employee performance, memory, or skills — use live data only',
   318→  ];
   319→
   320→  const activeEmployeeCount = rankedOrg.filter((e) => e.status !== 'idle' && e.status !== 'offline').length;
   321→
   322→  return {
   323→    generatedAt: new Date().toISOString(),
   324→    hasLiveData: data.hasLiveData,
   325→    dataSources: data.dataSources,
   326→    organization: rankedOrg,
   327→    departmentCount: departments.length,
   328→    activeEmployeeCount,
   329→    departments,
   330→    collaborationFeed,
   331→    activeChains: chains,
   332→    meetings,
   333→    crossDecisions,
   334→    pendingCrossDecisionCount: pendingCrossDecisions.length,
   335→    performance,
   336→    topEmployee,
   337→    topCollaboration,
   338→    marketplace: MARKETPLACE_EMPLOYEES,
   339→    installedMarketplaceCount: getInstalledEmployees().length,
   340→    delegations: delegationStore.slice(-20), // last 20
   341→    openDelegationCount: openDelegations.length,
   342→    escalations: escalationStore.slice(-20),
   343→    openEscalationCount: openEscalations.length,
   344→    aggregateMetrics,
   345→    oracleCapabilities,
   346→    tagline: WORKFORCE_TAGLINE,
   347→  };
   348→}

   415→    nextMeetingTitle: nextMeeting?.title ?? null,
   416→    nextMeetingType: nextMeeting?.type ?? null,

   418→    automationRatePct: automationRate,

   424→async function ensureDecisionInStore(decisionId: string): Promise<CrossDepartmentDecision | undefined> {
   425→  let decision = decisionStore.get(decisionId);
   426→  if (decision) return decision;
   427→  // Store is empty (likely after hot-reload) — populate it
   428→  await computeWorkforceDashboard();
   429→  return decisionStore.get(decisionId);

   432→export async function approveCrossDecision(req: ApproveWorkforceRequest): Promise<ApproveWorkforceResult> {
   433→  const role = resolveRole(req.role);
   434→  const decision = await ensureDecisionInStore(req.decisionId);
   435→  if (!decision) {
   436→    return {
   437→      decisionId: req.decisionId,
   438→      status: 'proposed',
   439→      message: 'Decision not found. It may have expired or been superseded.',
   440→      currentStepIndex: 0,
   441→      approvedAt: new Date().toISOString(),
   442→    };
   443→  }
   444→
   445→  const stepIndex = req.stepIndex ?? decision.currentStepIndex;
   446→  const step = decision.steps[stepIndex];
   447→  if (!step) {
   448→    return {
   449→      decisionId: req.decisionId,
   450→      status: decision.status,
   451→      message: `Step ${stepIndex} not found.`,
   452→      currentStepIndex: decision.currentStepIndex,
   453→      approvedAt: new Date().toISOString(),
   454→    };
   455→  }
   456→
   457→  // Validate role can approve this step (the step's role must match or be higher tier)
   458→  const stepRoleDef = getRoleDefinition(step.role);
   459→  const approverIsStepRole = step.role === role || stepRoleDef.reportsTo === role;
   460→
   461→  // Mark step approved
   462→  const updatedSteps = decision.steps.map((s, i) =>
   463→    i === stepIndex
   464→      ? { ...s, status: 'approved' as const, approvedAt: new Date().toISOString(), approvedBy: req.userId ?? role, comment: req.comment }
   465→      : s,
   466→  );
   467→
   468→  const nextStepIndex = stepIndex + 1;
   469→  let newStatus: CrossDecisionStatus;
   470→  if (nextStepIndex >= decision.steps.length) {
   471→    newStatus = 'fully_approved';
   472→  } else if (stepIndex === 0) {
   473→    newStatus = 'step_1_approved';
   474→  } else if (stepIndex === 1) {
   475→    newStatus = 'step_2_approved';
   476→  } else {
   477→    newStatus = 'fully_approved';
   478→  }
   479→
   480→  // Auto-execute if fully approved and non-destructive
   481→  let finalStatus: CrossDecisionStatus = newStatus;
   482→  let message = `Step ${stepIndex + 1} approved by ${role}.`;
   483→  if (newStatus === 'fully_approved') {
   484→    finalStatus = 'executing';
   485→    message = `Fully approved. Executing workflow...`;
   486→    // Mark all execute-type steps as executed
   487→    for (let i = 0; i < updatedSteps.length; i++) {
   488→      if (updatedSteps[i].approvalType === 'execute') {
   489→        updatedSteps[i].status = 'executed';
   490→        updatedSteps[i].approvedAt = new Date().toISOString();
   491→        updatedSteps[i].approvedBy = req.userId ?? role;
   492→      }
   493→    }
   494→    finalStatus = 'executed';
   495→    message = `Decision fully approved and executed. All ${decision.steps.length} steps completed.`;
   496→  }
   497→
   498→  const updated: CrossDepartmentDecision = {
   499→    ...decision,
   500→    steps: updatedSteps,
   501→    currentStepIndex: newStatus === 'fully_approved' ? decision.steps.length - 1 : nextStepIndex,
   502→    status: finalStatus,
   503→    executedAt: finalStatus === 'executed' ? new Date().toISOString() : decision.executedAt,
   504→  };
   505→  decisionStore.set(req.decisionId, updated);
   506→
   507→  return {
   508→    decisionId: req.decisionId,
   509→    status: finalStatus,
   510→    message,
   511→    currentStepIndex: updated.currentStepIndex,
   512→    approvedAt: new Date().toISOString(),
   513→  };
   514→}

   516→export async function executeCrossDecision(req: ExecuteWorkforceRequest): Promise<ExecuteWorkforceResult> {
   517→  const role = resolveRole(req.role);
   518→  const decision = await ensureDecisionInStore(req.decisionId);
   519→  if (!decision) {
   520→    return {
   521→      decisionId: req.decisionId,
   522→      status: 'proposed',
   523→      message: 'Decision not found.',
   524→      executedAt: new Date().toISOString(),
   525→    };
   526→  }
   527→
   528→  // Force-execute all remaining steps (requires CEO role)
   529→  if (role !== 'ceo') {
   530→    return {
   531→      decisionId: req.decisionId,
   532→      status: decision.status,
   533→      message: 'Only CEO role can force-execute a decision. Please approve steps individually.',
   534→      executedAt: new Date().toISOString(),
   535→    };
   536→  }
   537→
   538→  const updatedSteps = decision.steps.map((s) => ({
   539→    ...s,
   540→    status: (s.status === 'pending' ? 'approved' : s.status) as CrossDecisionStep['status'],
   541→    approvedAt: s.approvedAt ?? new Date().toISOString(),
   542→    approvedBy: s.approvedBy ?? req.userId ?? role,
   543→  }));
   544→  // Mark execute steps as executed
   545→  for (const s of updatedSteps) {
   546→    if (s.approvalType === 'execute') {
   547→      s.status = 'executed';
   548→    }
   549→  }
   550→
   551→  const updated: CrossDepartmentDecision = {
   552→    ...decision,
   553→    steps: updatedSteps,
   554→    currentStepIndex: decision.steps.length - 1,
   555→    status: 'executed',
   556→    executedAt: new Date().toISOString(),
   557→  };
   558→  decisionStore.set(req.decisionId, updated);
   559→
   560→  return {
   561→    decisionId: req.decisionId,
   562→    status: 'executed',
   563→    message: `Decision force-executed by CEO. All ${decision.steps.length} steps completed.`,
   564→    executedAt: new Date().toISOString(),
   565→    outputs: { stepsCompleted: decision.steps.length },
   566→  };
   567→}
