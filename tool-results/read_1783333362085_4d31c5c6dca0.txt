    69→function generateMockVersions(): VersionEntry[] {
    70→  const entries: VersionEntry[] = [];
    71→  const entities = [
    72→    { type: 'client' as const, id: 'CL-001', name: 'Acme Industries Pvt Ltd' },
    73→    { type: 'client' as const, id: 'CL-002', name: 'Sharma & Associates' },
    74→    { type: 'return' as const, id: 'RT-001', name: 'GSTR-1 Oct 2024' },
    75→    { type: 'return' as const, id: 'RT-002', name: 'GSTR-3B Nov 2024' },
    76→    { type: 'invoice' as const, id: 'INV-001', name: 'INV-2024-0156' },
    77→    { type: 'invoice' as const, id: 'INV-002', name: 'INV-2024-0187' },
    78→    { type: 'document' as const, id: 'DOC-001', name: 'PAN Card - Acme' },
    79→    { type: 'document' as const, id: 'DOC-002', name: 'GST Certificate - Sharma' },
    80→  ];
    81→
    82→  entities.forEach((entity) => {
    83→    const versionCount = Math.floor(Math.random() * 5) + 2;
    84→    const fields = entity.type === 'client' ? CLIENT_FIELDS
    85→      : entity.type === 'return' ? RETURN_FIELDS
    86→      : entity.type === 'invoice' ? INVOICE_FIELDS
    87→      : DOCUMENT_FIELDS;
    88→
    89→    for (let v = 1; v <= versionCount; v++) {
    90→      const daysAgo = (versionCount - v) * 2 + Math.floor(Math.random() * 3);
    91→      const ts = new Date();
    92→      ts.setDate(ts.getDate() - daysAgo);
    93→
    94→      const changedField = fields[Math.floor(Math.random() * fields.length)];
    95→      const beforeVal = `value_v${v}`;
    96→      const afterVal = `value_v${v + 1}`;
    97→
    98→      entries.push({
    99→        id: `${entity.id}-v${v}`,
   100→        version: v,
   101→        entityType: entity.type,
   102→        entityId: entity.id,
   103→        entityName: entity.name,
   104→        timestamp: ts.toISOString(),
   105→        author: AUTHORS[Math.floor(Math.random() * AUTHORS.length)],
   106→        summary: v === 1 ? `Created ${entity.type} record` : `Updated ${changedField}`,
   107→        before: v === 1 ? {} : { [changedField]: beforeVal },
   108→        after: v === 1 ? { [changedField]: afterVal } : { [changedField]: afterVal },
   109→        isCurrent: v === versionCount,
   110→      });
   111→    }
   112→  });
   113→
   114→  return entries.sort((a, b) => new Date(b.timestamp).getTime() - new Date(a.timestamp).getTime());
   115→}

   117→// ─── Diff Viewer ───────────────────────────────────────────────────────────────
   118→
   119→function VersionDiffViewer({ before, after }: { before: Record<string, string>; after: Record<string, string> }) {
   120→  const allKeys = Array.from(new Set([...Object.keys(before), ...Object.keys(after)]));
   121→  if (allKeys.length === 0) return <p className="text-sm text-muted-foreground">Initial creation — no prior state.</p>;
   122→
   123→  return (
   124→    <div className="rounded-lg border overflow-hidden">
   125→      <table className="w-full text-xs">
   126→        <thead>
   127→          <tr className="bg-muted/50">
   128→            <th className="px-3 py-2 text-left font-medium">Field</th>
   129→            <th className="px-3 py-2 text-left font-medium text-red-600">Before</th>
   130→            <th className="px-3 py-2 text-left font-medium text-emerald-600">After</th>
   131→          </tr>
   132→        </thead>
   133→        <tbody>
   134→          {allKeys.map((key) => {
   135→            const bVal = before[key] ?? '—';
   136→            const aVal = after[key] ?? '—';
   137→            const changed = bVal !== aVal;
   138→            return (
   139→              <tr key={key} className={changed ? 'bg-amber-50/50' : ''}>
   140→                <td className="px-3 py-1.5 font-medium border-t">{key}</td>
   141→                <td className="px-3 py-1.5 border-t text-red-600/80 font-mono">{bVal}</td>
   142→                <td className="px-3 py-1.5 border-t text-emerald-600/80 font-mono">{aVal}</td>
   143→              </tr>
   144→            );
   145→          })}
   146→        </tbody>
   147→      </table>
   148→    </div>
   149→  );
   150→}
