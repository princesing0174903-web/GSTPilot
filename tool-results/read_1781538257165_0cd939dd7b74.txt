   132→// ─── Circular Progress Component ────────────────────────────────────────────────
   133→
   134→function CircularProgress({ value, size = 140, strokeWidth = 8 }: { value: number; size?: number; strokeWidth?: number }) {
   135→  const radius = (size - strokeWidth) / 2
   136→  const circumference = 2 * Math.PI * radius
   137→  const offset = circumference - (value / 100) * circumference
   138→
   139→  return (
   140→    <div className="relative inline-flex items-center justify-center" style={{ width: size, height: size }}>
   141→      <svg width={size} height={size} className="-rotate-90">
   142→        <circle
   143→          cx={size / 2}
   144→          cy={size / 2}
   145→          r={radius}
   146→          fill="none"
   147→          stroke="currentColor"
   148→          strokeWidth={strokeWidth}
   149→          className="text-gray-200 dark:text-gray-800"
   150→        />
   151→        <motion.circle
   152→          cx={size / 2}
   153→          cy={size / 2}
   154→          r={radius}
   155→          fill="none"
   156→          stroke="url(#progressGradient)"
   157→          strokeWidth={strokeWidth}
   158→          strokeLinecap="round"
   159→          strokeDasharray={circumference}
   160→          initial={{ strokeDashoffset: circumference }}
   161→          animate={{ strokeDashoffset: offset }}
   162→          transition={{ duration: 0.5, ease: 'easeOut' }}
   163→        />
   164→        <defs>
   165→          <linearGradient id="progressGradient" x1="0%" y1="0%" x2="100%" y2="0%">
   166→            <stop offset="0%" stopColor="#10b981" />
   167→            <stop offset="100%" stopColor="#14b8a6" />
   168→          </linearGradient>
   169→        </defs>
   170→      </svg>
   171→      <div className="absolute inset-0 flex flex-col items-center justify-center">
   172→        <motion.span
   173→          className="text-3xl font-bold text-foreground"
   174→          key={Math.round(value)}
   175→          initial={{ scale: 1.1, opacity: 0.5 }}
   176→          animate={{ scale: 1, opacity: 1 }}
   177→          transition={{ duration: 0.2 }}
   178→        >
   179→          {Math.round(value)}%
   180→        </motion.span>
   181→        <span className="text-[10px] text-muted-foreground uppercase tracking-wider">Progress</span>
   182→      </div>
   183→    </div>
   184→  )
   185→}

   189→export default function AutopilotPage() {
   190→  const [overallStatus, setOverallStatus] = useState<OverallStatus>('idle')
   191→  const [steps, setSteps] = useState<Step[]>(
   192→    STEPS_CONFIG.map((s) => ({ ...s, status: 'queued' as StepStatus, progress: 0 }))
   193→  )
   194→  const [agentStatuses, setAgentStatuses] = useState<AgentStatus[]>(
   195→    AGENTS_CONFIG.map((a) => ({ ...a, status: 'queued' as AgentRunStatus }))
   196→  )
   197→  const [logs, setLogs] = useState<LogEntry[]>([])
   198→  const [overallProgress, setOverallProgress] = useState(0)
   199→  const [currentStepLabel, setCurrentStepLabel] = useState('')
   200→  const [elapsedSeconds, setElapsedSeconds] = useState(0)
   201→  const [summaryStats, setSummaryStats] = useState<SummaryStat[]>([])
   202→
   203→  const timerRef = useRef<ReturnType<typeof setInterval> | null>(null)
   204→  const stepTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null)
   205→  const logTimerRefs = useRef<ReturnType<typeof setTimeout>[]>([])
   206→
   207→  // Clean up all timers on unmount
   208→  useEffect(() => {
   209→    return () => {
   210→      if (timerRef.current) clearInterval(timerRef.current)
   211→      if (stepTimerRef.current) clearTimeout(stepTimerRef.current)
   212→      logTimerRefs.current.forEach(clearTimeout)
   213→    }
   214→  }, [])
   215→
   216→  // Elapsed time counter
   217→  useEffect(() => {
   218→    if (overallStatus === 'running') {
   219→      timerRef.current = setInterval(() => {
   220→        setElapsedSeconds((prev) => prev + 1)
   221→      }, 1000)
   222→    } else {
   223→      if (timerRef.current) clearInterval(timerRef.current)
   224→    }
   225→    return () => {
   226→      if (timerRef.current) clearInterval(timerRef.current)
   227→    }
   228→  }, [overallStatus])
   229→
   230→  const addLog = useCallback((agent: string, message: string, level: LogEntry['level'] = 'info') => {
   231→    setLogs((prev) => [
   232→      ...prev,
   233→      {
   234→        id: `log-${Date.now()}-${Math.random().toString(36).slice(2, 6)}`,
   235→        timestamp: new Date(),
   236→        agent,
   237→        message,
   238→        level,
   239→      },
   240→    ])
   241→  }, [])
   242→
   243→  const formatElapsed = (seconds: number) => {
   244→    const m = Math.floor(seconds / 60)
   245→    const s = seconds % 60
   246→    return `${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`
   247→  }
   248→
   249→  const handleRun = useCallback(() => {
   250→    // Reset everything
   251→    setOverallStatus('running')
   252→    setSteps(STEPS_CONFIG.map((s) => ({ ...s, status: 'queued' as StepStatus, progress: 0 })))
   253→    setAgentStatuses(AGENTS_CONFIG.map((a) => ({ ...a, status: 'queued' as AgentRunStatus })))
   254→    setLogs([])
   255→    setOverallProgress(0)
   256→    setCurrentStepLabel('Initializing...')
   257→    setElapsedSeconds(0)
   258→    setSummaryStats([])
   259→    logTimerRefs.current.forEach(clearTimeout)
   260→    logTimerRefs.current = []
   261→
   262→    // Add initial log
   263→    addLog('System', 'GSTPilot Autopilot™ initiated', 'info')
   264→
   265→    // Process steps sequentially
   266→    let currentStepIndex = 0
   267→    let totalProgress = 0
   268→
   269→    const processNextStep = () => {
   270→      if (currentStepIndex >= STEPS_CONFIG.length) {
   271→        // All steps done
   272→        setOverallStatus('completed')
   273→        setCurrentStepLabel('Complete!')
   274→        setOverallProgress(100)
   275→        addLog('System', 'Autopilot run completed successfully! ✓', 'info')
   276→
   277→        setSummaryStats([
   278→          { label: 'Documents Read', value: '14', icon: FileText, color: 'text-blue-600', colorBg: 'bg-blue-500/10' },
   279→          { label: 'Invoices Extracted', value: '14', icon: Search, color: 'text-emerald-600', colorBg: 'bg-emerald-500/10' },
   280→          { label: 'Mismatches Found', value: '3', icon: RefreshCw, color: 'text-amber-600', colorBg: 'bg-amber-500/10' },
   281→          { label: 'Returns Prepared', value: '16', icon: ClipboardList, color: 'text-purple-600', colorBg: 'bg-purple-500/10' },
   282→          { label: 'Tasks Created', value: '7', icon: CheckCircle2, color: 'text-teal-600', colorBg: 'bg-teal-500/10' },
   283→          { label: 'Notifications Sent', value: '15', icon: Bell, color: 'text-rose-600', colorBg: 'bg-rose-500/10' },
   284→        ])
   285→        return
   286→      }
   287→
   288→      const stepConfig = STEPS_CONFIG[currentStepIndex]
   289→      setCurrentStepLabel(stepConfig.label)
   290→
   291→      // Set current step to running
   292→      setSteps((prev) =>
   293→        prev.map((s) =>
   294→          s.id === stepConfig.id ? { ...s, status: 'running' as StepStatus } : s
   295→        )
   296→      )
   297→
   298→      // Activate the corresponding agent(s)
   299→      const agentMapping: Record<number, string[]> = {
   300→        1: ['invoice-extraction'],
   301→        2: ['invoice-extraction'],
   302→        3: ['reconciliation'],
   303→        4: ['filing'],
   304→        5: ['firm-intelligence'],
   305→        6: ['compliance'],
   306→        7: ['compliance'],
   307→        8: ['firm-intelligence'],
   308→      }
   309→      const activeAgentIds = agentMapping[stepConfig.id] || []
   310→      setAgentStatuses((prev) =>
   311→        prev.map((a) => {
   312→          if (activeAgentIds.includes(a.id) && a.status === 'queued') {
   313→            return { ...a, status: 'running' as AgentRunStatus }
   314→          }
   315→          return a
   316→        })
   317→      )
   318→
   319→      // Animate progress for this step
   320→      const stepDuration = stepConfig.duration
   321→      const progressInterval = 50
   322→      const progressIncrement = 100 / (stepDuration / progressInterval)
   323→      let stepProgress = 0
   324→
   325→      const progressTimer = setInterval(() => {
   326→        stepProgress = Math.min(stepProgress + progressIncrement, 100)
   327→        setSteps((prev) =>
   328→          prev.map((s) =>
   329→            s.id === stepConfig.id ? { ...s, progress: stepProgress } : s
   330→          )
   331→        )
   332→
   333→        // Update overall progress
   334→        totalProgress = ((currentStepIndex * 100 + stepProgress) / (STEPS_CONFIG.length * 100)) * 100
   335→        setOverallProgress(totalProgress)
   336→
   337→        if (stepProgress >= 100) {
   338→          clearInterval(progressTimer)
   339→        }
   340→      }, progressInterval)
   341→
   342→      // Add log entries for this step at intervals
   343→      const logData = LOG_MESSAGES.find((l) => l.stepId === stepConfig.id)
   344→      if (logData) {
   345→        logData.messages.forEach((msg, idx) => {
   346→          const delay = (idx + 1) * (stepDuration / (logData.messages.length + 1))
   347→          const t = setTimeout(() => {
   348→            addLog(logData.agent, msg, logData.level)
   349→          }, delay)
   350→          logTimerRefs.current.push(t)
   351→        })
   352→      }
   353→
   354→      // Complete this step after duration
   355→      stepTimerRef.current = setTimeout(() => {
   356→        clearInterval(progressTimer)
   357→
   358→        // Mark step as completed
   359→        setSteps((prev) =>
   360→          prev.map((s) =>
   361→            s.id === stepConfig.id ? { ...s, status: 'completed' as StepStatus, progress: 100 } : s
   362→          )
   363→        )
   364→
   365→        // Mark agent(s) as completed
   366→        setAgentStatuses((prev) =>
   367→          prev.map((a) => {
   368→            if (activeAgentIds.includes(a.id)) {
   369→              return { ...a, status: 'completed' as AgentRunStatus }
   370→            }
   371→            return a
   372→          })
   373→        )
   374→
   375→        // Move to next step
   376→        currentStepIndex++
   377→        processNextStep()
   378→      }, stepDuration)
   379→    }
   380→
   381→    // Start processing after a short delay
   382→    setTimeout(processNextStep, 500)
   383→  }, [addLog])
   384→
   385→  const handleReset = useCallback(() => {
   386→    if (stepTimerRef.current) clearTimeout(stepTimerRef.current)
   387→    logTimerRefs.current.forEach(clearTimeout)
   388→    logTimerRefs.current = []
   389→    setOverallStatus('idle')
   390→    setSteps(STEPS_CONFIG.map((s) => ({ ...s, status: 'queued' as StepStatus, progress: 0 })))
   391→    setAgentStatuses(AGENTS_CONFIG.map((a) => ({ ...a, status: 'queued' as AgentRunStatus })))
   392→    setLogs([])
   393→    setOverallProgress(0)
   394→    setCurrentStepLabel('')
   395→    setElapsedSeconds(0)
   396→    setSummaryStats([])
   397→  }, [])
   398→
   399→  const isRunning = overallStatus === 'running'
   400→  const isCompleted = overallStatus === 'completed'
   401→
   402→  return (
   403→    <div className="min-h-screen bg-gradient-to-b from-background to-background">
   404→      <div className="max-w-7xl mx-auto space-y-6 p-4 md:p-6">
   405→
   406→        {/* ═══ Hero Section ═══ */}
   407→        <motion.div
   408→          initial={{ opacity: 0, y: -20 }}
   409→          animate={{ opacity: 1, y: 0 }}
   410→          transition={{ duration: 0.5 }}
   411→          className="text-center space-y-4 py-6 md:py-10"
   412→        >
   413→          <motion.div
   414→            initial={{ scale: 0.8, opacity: 0 }}
   415→            animate={{ scale: 1, opacity: 1 }}
   416→            transition={{ duration: 0.5, delay: 0.2 }}
   417→            className="inline-flex items-center gap-2 px-4 py-1.5 rounded-full bg-emerald-500/10 border border-emerald-500/20 text-emerald-600 text-xs font-medium"
   418→          >
   419→            <Sparkles className="h-3.5 w-3.5" />
   420→            AI-Powered Automation
   421→          </motion.div>
   422→
   423→          <h1 className="text-4xl md:text-5xl font-bold tracking-tight bg-gradient-to-r from-emerald-600 via-teal-600 to-cyan-600 bg-clip-text text-transparent">
   424→            GSTPilot Autopilot™
   425→          </h1>
   426→          <p className="text-lg text-muted-foreground max-w-lg mx-auto">
   427→            One click. Your entire firm runs itself.
   428→          </p>
   429→
   430→          {/* Run Button */}
   431→          {!isCompleted && (
   432→            <motion.div
   433→              initial={{ opacity: 0, y: 10 }}
   434→              animate={{ opacity: 1, y: 0 }}
   435→              transition={{ duration: 0.5, delay: 0.4 }}
   436→              className="pt-4"
   437→            >
   438→              <Button
   439→                size="lg"
   440→                onClick={handleRun}
   441→                disabled={isRunning}
   442→                className={`
   443→                  relative px-10 py-6 text-lg font-semibold gap-3
   444→                  bg-gradient-to-r from-emerald-500 via-emerald-600 to-teal-600
   445→                  text-white shadow-2xl shadow-emerald-500/30
   446→                  hover:shadow-emerald-500/40 hover:scale-[1.02]
   447→                  active:scale-[0.98]
   448→                  transition-all duration-300
   449→                  disabled:opacity-70 disabled:cursor-not-allowed disabled:hover:scale-100
   450→                  ${!isRunning ? 'animate-pulse' : ''}
   451→                `}
   452→              >
   453→                {!isRunning && (
   454→                  <span className="absolute inset-0 rounded-md bg-gradient-to-r from-emerald-400/20 to-teal-400/20 animate-ping" />
   455→                )}
   456→                {isRunning ? (
   457→                  <>
   458→                    <Loader2 className="h-5 w-5 animate-spin relative" />
   459→                    <span className="relative">Running...</span>
   460→                  </>
   461→                ) : (
   462→                  <>
   463→                    <Play className="h-5 w-5 relative" />
   464→                    <span className="relative">Run My Firm</span>
   465→                  </>
   466→                )}
   467→              </Button>
   468→            </motion.div>
   469→          )}
   470→        </motion.div>
   471→
   472→        {/* ═══ Running State ═══ */}
   473→        <AnimatePresence>
   474→          {(isRunning || isCompleted) && (
   475→            <motion.div
   476→              initial={{ opacity: 0, y: 30 }}
   477→              animate={{ opacity: 1, y: 0 }}
   478→              exit={{ opacity: 0, y: -20 }}
   479→              transition={{ duration: 0.5 }}
   480→              className="space-y-6"
   481→            >
   482→              {/* ═══ Completion Banner ═══ */}
   483→              {isCompleted && (
   484→                <motion.div
   485→                  initial={{ opacity: 0, scale: 0.95 }}
   486→                  animate={{ opacity: 1, scale: 1 }}
   487→                  transition={{ duration: 0.5 }}
   488→                >
   489→                  <Card className="bg-gradient-to-r from-emerald-500/10 via-teal-500/10 to-cyan-500/10 border-emerald-500/20 backdrop-blur-sm overflow-hidden">
   490→                    <div className="absolute inset-0 bg-gradient-to-r from-emerald-500/5 via-transparent to-teal-500/5" />
   491→                    <CardContent className="p-6 relative">
   492→                      <div className="flex flex-col sm:flex-row items-center gap-4">
   493→                        <motion.div
   494→                          initial={{ scale: 0 }}
   495→                          animate={{ scale: 1 }}
   496→                          transition={{ type: 'spring', bounce: 0.5, delay: 0.2 }}
   497→                          className="p-3 rounded-2xl bg-gradient-to-br from-emerald-500 to-teal-600 text-white shadow-lg"
   498→                        >
   499→                          <CheckCircle2 className="h-8 w-8" />

   512→                            View Details

   514→                          <Button
