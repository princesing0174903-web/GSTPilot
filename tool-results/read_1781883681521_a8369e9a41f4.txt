    99→
   100→export function OracleWorkspace({
   101→  open, onClose, onNavigate, userName, firmName, gstin,
   102→}: OracleWorkspaceProps) {
   103→  const [store, setStore] = useState<ConversationStore>({ conversations: [], activeId: null });
   104→  // Portal guard: only render into document.body after mount to avoid SSR
   105→  // hydration mismatch (server has no document.body).
   106→  const [mounted, setMounted] = useState(false);
   107→  useEffect(() => setMounted(true), []);
   108→  const [messages, setMessages] = useState<OracleMessage[]>([]);
   109→  const [activeId, setActiveId] = useState<string | null>(null);
   110→  const [input, setInput] = useState('');
   111→  const [isStreaming, setIsStreaming] = useState(false);
   112→  const [lastEmotion, setLastEmotion] = useState<OracleMessage['emotion']>('helpful');
   113→  const [activeLanguage, setActiveLanguage] = useState<OracleMessage['language']>('english');
   114→  // Mobile: history drawer open state
   115→  const [historyOpen, setHistoryOpen] = useState(false);
   116→
   117→  const scrollRef = useRef<HTMLDivElement>(null);
   118→  const inputRef = useRef<HTMLTextAreaElement>(null);
   119→  const abortRef = useRef<AbortController | null>(null);
   120→  const userPinnedUpRef = useRef(false);
   121→  const streamingIdRef = useRef<string | null>(null);
   122→
   123→  // ─── Load + migrate store on open ──────────────────────────────────────────
   124→  useEffect(() => {
   125→    if (!open) return;
   126→    const loaded = loadStore();
   127→    if (loaded.conversations.length === 0) {
   128→      // Migrate legacy single-conversation store if present.
   129→      try {
   130→        const raw = localStorage.getItem(LEGACY_KEY);
   131→        if (raw) {
   132→          const parsed = JSON.parse(raw) as OracleMessage[];
   133→          if (Array.isArray(parsed) && parsed.length > 0) {
   134→            const now = new Date().toISOString();
   135→            const conv: OracleConversation = {
   136→              id: cryptoId(),
   137→              title: deriveTitle(parsed),
   138→              messages: parsed.map((m) => (m.streaming ? { ...m, streaming: false } : m)),
   139→              createdAt: parsed[0]?.createdAt ?? now,
   140→              updatedAt: now,
   141→            };
   142→            const next: ConversationStore = { conversations: [conv], activeId: conv.id };
   143→            setStore(next);
   144→            setActiveId(conv.id);
   145→            setMessages(conv.messages);
   146→            saveStore(next);
   147→            return;
   148→          }
   149→        }
   150→      } catch {
   151→        /* ignore */
   152→      }
   153→      // Nothing to migrate — start fresh with one empty conversation.
   154→      const conv = newConversation();
   155→      const next: ConversationStore = { conversations: [conv], activeId: conv.id };
   156→      setStore(next);
   157→      setActiveId(conv.id);
   158→      setMessages([]);
   159→      saveStore(next);
   160→      return;
   161→    }
   162→    setStore(loaded);
   163→    const active = loaded.conversations.find((c) => c.id === loaded.activeId) ?? loaded.conversations[0];
   164→    if (active) {
   165→      setActiveId(active.id);
   166→      setMessages(active.messages);
   167→    }
   168→  }, [open]);
   169→
   170→  // ─── Persist active conversation whenever messages change ───────────────────
   171→  useEffect(() => {
   172→    if (!open || !activeId) return;
   173→    setStore((prev) => {
   174→      if (!prev.conversations.length) return prev;
   175→      const exists = prev.conversations.some((c) => c.id === activeId);
   176→      let conversations: OracleConversation[];
   177→      let nextActiveId = prev.activeId;
   178→      if (exists) {
   179→        conversations = prev.conversations.map((c) =>
   180→          c.id === activeId
   181→            ? {
   182→                ...c,
   183→                messages,
   184→                title: deriveTitle(messages) || c.title,
   185→                updatedAt: new Date().toISOString(),
   186→              }
   187→            : c,
   188→        );
   189→      } else {
   190→        // Active conversation doesn't exist yet — create it on the fly.
   191→        const conv: OracleConversation = {
   192→          id: activeId,
   193→          title: deriveTitle(messages) || 'New chat',
   194→          messages,
   195→          createdAt: new Date().toISOString(),
   196→          updatedAt: new Date().toISOString(),
   197→        };
   198→        conversations = [conv, ...prev.conversations];
   199→        nextActiveId = activeId;
   200→      }
   201→      const next = { conversations, activeId: nextActiveId };
   202→      saveStore(next);
   203→      return next;
   204→    });
   205→  }, [messages, open, activeId]);
   206→
   207→  // ─── Body scroll lock while open ───────────────────────────────────────────
   208→  useEffect(() => {
   209→    if (!open) return;
   210→    const prev = document.body.style.overflow;
   211→    document.body.style.overflow = 'hidden';
   212→    return () => {
   213→      document.body.style.overflow = prev;
   214→    };
   215→  }, [open]);
   216→
   217→  // ─── Escape to close ───────────────────────────────────────────────────────
   218→  useEffect(() => {
   219→    if (!open) return;
   220→    const onKey = (e: globalThis.KeyboardEvent) => {
   221→      if (e.key === 'Escape' && !isStreaming) onClose();
   222→    };
   223→    window.addEventListener('keydown', onKey);
   224→    return () => window.removeEventListener('keydown', onKey);
   225→  }, [open, isStreaming, onClose]);
   226→
   227→  // ─── Smart auto-scroll ─────────────────────────────────────────────────────
   228→  const handleScroll = useCallback(() => {
   229→    const el = scrollRef.current;
   230→    if (!el) return;
   231→    const distanceFromBottom = el.scrollHeight - el.scrollTop - el.clientHeight;
   232→    userPinnedUpRef.current = distanceFromBottom > 120;
   233→  }, []);
   234→
   235→  const scrollToBottom = useCallback((smooth = false) => {
   236→    const el = scrollRef.current;
   237→    if (!el) return;
   238→    el.scrollTo({ top: el.scrollHeight, behavior: smooth ? 'smooth' : 'auto' });
   239→  }, []);
   240→
   241→  useEffect(() => {
   242→    if (!userPinnedUpRef.current) {
   243→      requestAnimationFrame(() => scrollToBottom(false));
   244→    }
   245→  }, [messages, scrollToBottom]);
   246→
   247→  // ─── Conversation actions ──────────────────────────────────────────────────
   248→  const handleNewChat = useCallback(() => {
   249→    if (isStreaming) return;
   250→    const conv = newConversation();
   251→    setStore((prev) => {
   252→      const next = { conversations: [conv, ...prev.conversations], activeId: conv.id };
   253→      saveStore(next);
   254→      return next;
   255→    });
   256→    setActiveId(conv.id);
   257→    setMessages([]);
   258→    setInput('');
   259→    userPinnedUpRef.current = false;
   260→    setHistoryOpen(false);
   261→    requestAnimationFrame(() => inputRef.current?.focus());
   262→  }, [isStreaming]);
   263→
   264→  const handleSwitchConversation = useCallback(
   265→    (id: string) => {
   266→      if (isStreaming) return;
   267→      const conv = store.conversations.find((c) => c.id === id);
   268→      if (!conv) return;
   269→      setActiveId(id);
   270→      setMessages(conv.messages);
   271→      setStore((prev) => {
   272→        const next = { ...prev, activeId: id };
   273→        saveStore(next);
   274→        return next;
   275→      });
   276→      setHistoryOpen(false);
   277→      userPinnedUpRef.current = false;
   278→      requestAnimationFrame(() => scrollToBottom(false));
   279→    },
   280→    [isStreaming, store.conversations, scrollToBottom],
   281→  );
   282→
   283→  const handleDeleteConversation = useCallback(
   284→    (id: string) => {
   285→      if (isStreaming) return;
   286→      setStore((prev) => {
   287→        const filtered = prev.conversations.filter((c) => c.id !== id);
   288→        let nextActive = prev.activeId;
   289→        let nextMessages = messages;
   290→        if (prev.activeId === id) {
   291→          if (filtered.length > 0) {
   292→            nextActive = filtered[0].id;
   293→            nextMessages = filtered[0].messages;
   294→          } else {
   295→            const conv = newConversation();
   296→            filtered.unshift(conv);
   297→            nextActive = conv.id;
   298→            nextMessages = [];
   299→          }
   300→          setActiveId(nextActive);
   301→          setMessages(nextMessages);
   302→        }
   303→        const next = { conversations: filtered, activeId: nextActive };
   304→        saveStore(next);
   305→        return next;
   306→      });
   307→    },
   308→    [isStreaming, messages],
   309→  );
   310→
   311→  // ─── Send flow (preserved from previous implementation) ─────────────────────
   312→  const sendMessage = useCallback(
   313→    async (raw: string) => {
   314→      const text = raw.trim();
   315→      if (!text || isStreaming) return;
   316→
   317→      // ── Brand-question short-circuit ───────────────────────────────────────
   318→      const brand = detectBrandQuestion(text);
   319→      if (brand.matched && brand.answer) {
   320→        const userMsg: OracleMessage = {
   321→          id: cryptoId(),
   322→          role: 'user',
   323→          content: text,
   324→          language: detectLanguage(text),
   325→          createdAt: new Date().toISOString(),
   326→        };
   327→        const oracleMsg: OracleMessage = {
   328→          id: cryptoId(),
   329→          role: 'oracle',
   330→          content: brand.answer,
   331→          language: detectLanguage(text),
   332→          emotion: 'success',
   333→          createdAt: new Date().toISOString(),
   334→          streaming: false,
   335→          followUps: ['Who founded GSTPilot?', 'What can Oracle do?', 'GST kya hota hai?'],
   336→        };
   337→        setMessages((prev) => [...prev, userMsg, oracleMsg]);
   338→        setLastEmotion('success');
   339→        setActiveLanguage(detectLanguage(text));
   340→        setInput('');
   341→        if (inputRef.current) inputRef.current.style.height = 'auto';
   342→        userPinnedUpRef.current = false;
   343→        requestAnimationFrame(() => scrollToBottom(true));
   344→        return;
   345→      }
   346→
   347→      const userMsg: OracleMessage = {
   348→        id: cryptoId(),
   349→        role: 'user',
   350→        content: text,
   351→        language: detectLanguage(text),
   352→        createdAt: new Date().toISOString(),
   353→      };
   354→      const oracleId = cryptoId();
   355→      const oraclePlaceholder: OracleMessage = {
   356→        id: oracleId,
   357→        role: 'oracle',
   358→        content: '',
   359→        language: userMsg.language,
   360→        createdAt: new Date().toISOString(),
   361→        streaming: true,
   362→      };
   363→      setMessages((prev) => [...prev, userMsg, oraclePlaceholder]);
   364→      streamingIdRef.current = oracleId;
   365→      setIsStreaming(true);
   366→      setActiveLanguage(userMsg.language);
   367→      setInput('');
   368→      userPinnedUpRef.current = false;
   369→      if (inputRef.current) inputRef.current.style.height = 'auto';
   370→
   371→      const history: OracleChatRequest['messages'] = [
   372→        ...messages
   373→          .filter((m) => m.content.trim().length > 0)
   374→          .map((m) => ({ role: m.role, content: m.content })),
   375→        { role: 'user', content: text },
   376→      ];
   377→
   378→      const payload: OracleChatRequest = {
   379→        messages: history,
   380→        memory: {
   381→          userName,
   382→          firmName,
   383→          gstin,
   384→          preferredLanguage: activeLanguage,
   385→          recentTopics: messages
   386→            .filter((m) => m.role === 'user')
   387→            .slice(-4)
   388→            .map((m) => m.content.slice(0, 60)),
   389→        },
   390→      };
   391→
   392→      const controller = new AbortController();
   393→      abortRef.current = controller;
   394→
   395→      try {
   396→        const res = await fetch('/api/oracle/chat', {
   397→          method: 'POST',
   398→          headers: { 'Content-Type': 'application/json' },
   399→          body: JSON.stringify(payload),
   400→          signal: controller.signal,
   401→        });
   402→        if (!res.ok || !res.body) {
   403→          throw new Error(`Request failed (${res.status})`);
   404→        }
   405→
   406→        const reader = res.body.getReader();
   407→        const decoder = new TextDecoder();
   408→        let buffer = '';
   409→        let acc = '';
   410→
   411→        while (true) {
   412→          const { done, value } = await reader.read();
   413→          if (done) break;
   414→          buffer += decoder.decode(value, { stream: true });
   415→
   416→          const lines = buffer.split('\n');
   417→          buffer = lines.pop() ?? '';
   418→
   419→          for (const rawLine of lines) {
   420→            const line = rawLine.trim();
   421→            if (!line || !line.startsWith('data:')) continue;
   422→            const data = line.slice(5).trim();
   423→            if (!data || data === '[DONE]') continue;
   424→            let chunk: OracleStreamChunk;
   425→            try {
   426→              chunk = JSON.parse(data) as OracleStreamChunk;
   427→            } catch {
   428→              continue;
   429→            }
   430→            if (chunk.language) setActiveLanguage(chunk.language);
   431→            if (chunk.token) {
   432→              acc += chunk.token;
   433→              setMessages((prev) =>
   434→                prev.map((m) => (m.id === oracleId ? { ...m, content: acc } : m)),
   435→              );
   436→            }
   437→            if (chunk.done) {
   438→              setMessages((prev) =>
   439→                prev.map((m) =>
   440→                  m.id === oracleId
   441→                    ? {
   442→                        ...m,
   443→                        content: acc,
   444→                        streaming: false,
   445→                        emotion: detectEmotion(acc),
   446→                        followUps: buildFollowUps(acc),
   447→                      }
   448→                    : m,
   449→                ),
   450→              );
   451→              setLastEmotion(detectEmotion(acc));
   452→            }
   453→            if (chunk.error) {
   454→              setMessages((prev) =>
   455→                prev.map((m) =>
   456→                  m.id === oracleId
   457→                    ? {
   458→                        ...m,
   459→                        content:
   460→                          acc ||
   461→                          "I ran into a temporary issue reaching my reasoning service. Please try that again — your conversation is safe.",
   462→                        streaming: false,
   463→                        emotion: 'warning',
   464→                      }
   465→                    : m,
   466→                ),
   467→              );
   468→              setLastEmotion('warning');
   469→            }
   470→          }
   471→        }
   472→
   473→        setMessages((prev) =>
   474→          prev.map((m) =>
   475→            m.id === oracleId && m.streaming
   476→              ? {
   477→                  ...m,
   478→                  streaming: false,
   479→                  emotion: detectEmotion(m.content),
   480→                  followUps: buildFollowUps(m.content),
   481→                }
   482→              : m,
   483→          ),
   484→        );
   485→      } catch (err) {
   486→        const aborted = err instanceof DOMException && err.name === 'AbortError';
   487→        setMessages((prev) =>
   488→          prev.map((m) =>
   489→            m.id === oracleId
   490→              ? {
   491→                  ...m,
   492→                  streaming: false,
   493→                  content:
   494→                    m.content ||
   495→                    (aborted
   496→                      ? 'Stopped.'
   497→                      : "I had trouble reaching my reasoning service. Please try again in a moment."),
   498→                  emotion: aborted ? m.emotion : 'warning',
   499→                }
   500→              : m,
   501→          ),
   502→        );
   503→        if (!aborted) setLastEmotion('warning');
   504→      } finally {
   505→        setIsStreaming(false);
   506→        streamingIdRef.current = null;
   507→        abortRef.current = null;
   508→        requestAnimationFrame(() => inputRef.current?.focus());
   509→      }
   510→    },
   511→    [isStreaming, messages, userName, firmName, gstin, activeLanguage, scrollToBottom],
   512→  );
   513→
   514→  // ─── Stop streaming ────────────────────────────────────────────────────────
   515→  const handleStop = useCallback(() => {
   516→    abortRef.current?.abort();
   517→  }, []);
   518→
   519→  // ─── Input handling ────────────────────────────────────────────────────────
   520→  const handleInputChange = useCallback((e: React.ChangeEvent<HTMLTextAreaElement>) => {
   521→    setInput(e.target.value);
   522→    const el = e.target;
   523→    el.style.height = 'auto';
   524→    el.style.height = `${Math.min(el.scrollHeight, 160)}px`;
   525→  }, []);
   526→
   527→  const handleKeyDown = useCallback(
   528→    (e: KeyboardEvent<HTMLTextAreaElement>) => {
   529→      if (e.key === 'Enter' && !e.shiftKey) {
   530→        e.preventDefault();
   531→        sendMessage(input);
   532→      }
   533→    },
   534→    [input, sendMessage],
   535→  );
   536→
   537→  // ─── Focus input on open ───────────────────────────────────────────────────
   538→  useEffect(() => {
   539→    if (open) {
   540→      requestAnimationFrame(() => inputRef.current?.focus());
   541→    }
   542→  }, [open]);
   543→
   544→  // ─── Rail navigation ───────────────────────────────────────────────────────
   545→  const handleRailClick = useCallback(
   546→    (item: RailItem) => {
   547→      if (item.id === 'oracle') return; // already here
   548→      if (isStreaming) return;
   549→      onNavigate(item.id);
   550→      onClose();
   551→    },
   552→    [isStreaming, onNavigate, onClose],
   553→  );
   554→
   555→  // ─── Derived state ─────────────────────────────────────────────────────────
   556→  const avatarState = useMemo(
   557→    () => deriveAvatarState({ isStreaming, hasInput: input.trim().length > 0, lastEmotion }),
   558→    [isStreaming, input, lastEmotion],
   559→  );
   560→
   561→  const isEmpty = messages.length === 0;
   562→  const grouped = useMemo(() => groupConversations(store.conversations), [store.conversations]);
   563→
   564→  if (!mounted || typeof document === 'undefined') return null;
   565→
   566→  return createPortal(
   567→    <AnimatePresence>
   568→      {open && (
   569→        <motion.div
   570→          initial={{ opacity: 0 }}
   571→          animate={{ opacity: 1 }}
   572→          exit={{ opacity: 0 }}
   573→          transition={{ duration: 0.2 }}
   574→          className="fixed inset-0 z-[200] flex h-screen w-screen"
   575→          style={{ background: '#050505' }}
   576→        >
   577→          {/* ═══════════════════════════════════════════════════════════════════ */}
   578→          {/* COLUMN 0 — RAIL (left sidebar)                                       */}
   579→          {/* ═══════════════════════════════════════════════════════════════════ */}
   580→          <nav
   581→            aria-label="Oracle workspace navigation"
   582→            className="flex h-full w-14 shrink-0 flex-col items-center gap-1 border-r py-3 md:w-16 md:gap-1.5 lg:w-56 lg:items-stretch lg:px-2.5"
   583→            style={{ borderColor: 'rgba(255,255,255,0.08)' }}

   921→interface HistoryGroupProps {

   929→function HistoryGroup({ label, items, activeId, onSelect, onDelete }: HistoryGroupProps) {
   930→  if (items.length === 0) return null;
   931→  return (
   932→    <div className="mb-3">
   933→      <p className="px-3 py-1.5 text-[10px] font-semibold uppercase tracking-wider text-white/30">

   939→          return (
   940→            <div
   941→              key={c.id}
   942→              className={cn(
   943→                'group relative flex items-center gap-2 rounded-lg px-3 py-2 text-left transition-colors',
   944→                isActive ? 'bg-white/[0.06] text-white' : 'text-white/60 hover:bg-white/[0.04] hover:text-white',

   947→              <button
   948→                type="button"
   949→                onClick={() => onSelect(c.id)}
   950→                className="min-w-0 flex-1 text-left"

  1198→function groupConversations(convos: OracleConversation[]): ConversationGroups {
  1199→  const now = new Date();
  1200→  const startOfToday = new Date(now.getFullYear(), now.getMonth(), now.getDate()).getTime();
  1201→  const startOfYesterday = startOfToday - 86_400_000;
  1202→  const startOf7Days = startOfToday - 7 * 86_400_000;
  1203→
  1204→  const today: OracleConversation[] = [];
  1205→  const yesterday: OracleConversation[] = [];
  1206→  const prev7: OracleConversation[] = [];
  1207→  const older: OracleConversation[] = [];
  1208→
  1209→  const sorted = [...convos].sort(
  1210→    (a, b) => new Date(b.updatedAt).getTime() - new Date(a.updatedAt).getTime(),
  1211→  );
  1212→
  1213→  for (const c of sorted) {
  1214→    const t = new Date(c.updatedAt).getTime();
  1215→    if (isNaN(t)) {
  1216→      older.push(c);
  1217→    } else if (t >= startOfToday) {
  1218→      today.push(c);
  1219→    } else if (t >= startOfYesterday) {
  1220→      yesterday.push(c);
  1221→    } else if (t >= startOf7Days) {
  1222→      prev7.push(c);
  1223→    } else {
  1224→      older.push(c);
  1225→    }
  1226→  }
  1227→
  1228→  return { today, yesterday, prev7, older };
  1229→}
