    36→function BalanceTrendChart({ data }: { data: { day: string; balance: number }[] }) {
    37→  const max = Math.max(...data.map(d => d.balance))
    38→  const min = Math.min(...data.map(d => d.balance))
    39→  const range = max - min || 1
    40→  const w = 280
    41→  const h = 60
    42→  const pts = data.map((d, i) => `${(i / (data.length - 1)) * w},${h - ((d.balance - min) / range) * (h - 8) - 4}`)
    43→  return (
    44→    <svg width={w} height={h + 16} className="overflow-visible">
    45→      <defs>
    46→        <linearGradient id="balGrad" x1="0" y1="0" x2="0" y2="1">
    47→          <stop offset="0%" stopColor="#10b981" stopOpacity="0.2" />
    48→          <stop offset="100%" stopColor="#10b981" stopOpacity="0.02" />
    49→        </linearGradient>
    50→      </defs>
    51→      <polygon points={`0,${h} ${pts.join(' ')} ${w},${h}`} fill="url(#balGrad)" />
    52→      <polyline points={pts.join(' ')} fill="none" stroke="#10b981" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" />
    53→      {data.filter((_, i) => i % 2 === 0).map((d, i) => {
    54→        const x = (data.indexOf(d) / (data.length - 1)) * w
    55→        return <text key={i} x={x} y={h + 12} textAnchor="middle" className="text-[8px] fill-muted-foreground">{d.day}</text>
    56→      })}
    57→    </svg>
    58→  )
    59→}

    61→function ReconcileDonut({ reconciled, unreconciled }: { reconciled: number; unreconciled: number }) {
    62→  const total = reconciled + unreconciled
    63→  const r = 36
    64→  const c = 2 * Math.PI * r
    65→  const reconcilePct = (reconciled / total) * 100
    66→  const dashOffset = c - (reconcilePct / 100) * c
    67→  return (
    68→    <svg width="90" height="90" className="overflow-visible">
    69→      <circle cx="45" cy="45" r={r} fill="none" stroke="currentColor" strokeWidth="8" className="text-slate-100 dark:text-slate-800" />
    70→      <circle cx="45" cy="45" r={r} fill="none" stroke="#10b981" strokeWidth="8" strokeLinecap="round" strokeDasharray={c} strokeDashoffset={dashOffset} transform="rotate(-90 45 45)" />
    71→      <text x="45" y="42" textAnchor="middle" className="text-sm font-bold fill-emerald-600 dark:fill-emerald-400">{reconcilePct.toFixed(0)}%</text>
    72→      <text x="45" y="54" textAnchor="middle" className="text-[7px] fill-muted-foreground">Reconciled</text>
    73→    </svg>
    74→  )
    75→}

   154→function formatSyncDate(iso: string | null): string {
   155→  if (!iso) return '—'
   156→  const d = new Date(iso)
   157→  if (Number.isNaN(d.getTime())) return '—'
   158→  const dd = String(d.getDate()).padStart(2, '0')
   159→  const mm = String(d.getMonth() + 1).padStart(2, '0')
   160→  const yyyy = d.getFullYear()
   161→  const hh = String(d.getHours()).padStart(2, '0')
   162→  const min = String(d.getMinutes()).padStart(2, '0')
   163→  return `${dd}/${mm}/${yyyy} ${hh}:${min}`
   164→}
   165→
   166→function formatTxnDate(iso: string): string {
   167→  if (!iso) return '—'
   168→  const d = new Date(iso)
   169→  if (Number.isNaN(d.getTime())) return iso
   170→  const dd = String(d.getDate()).padStart(2, '0')
   171→  const mm = String(d.getMonth() + 1).padStart(2, '0')
   172→  const yyyy = d.getFullYear()
   173→  return `${dd}/${mm}/${yyyy}`
   174→}

   176→function upperMode(mode: string): string {
   177→  return (mode || 'bank').toUpperCase()
   178→}
   179→
   180→// Map raw Payment row → BankTransaction shape used by this page.
   181→function mapPaymentToTxn(p: {
   182→  id: string
   183→  partyName: string
   184→  partyType?: string
   185→  amount: number
   186→  paymentDate: string
   187→  paymentMode: string
   188→  referenceNo?: string | null
   189→  invoiceId?: string | null
   190→  status?: string
   191→  reconciled?: boolean
   192→  notes?: string | null
   193→}): BankTransaction {
   194→  const isVendor = (p.partyType || 'customer') === 'vendor'
   195→  return {
   196→    id: p.id,
   197→    date: formatTxnDate(p.paymentDate),
   198→    description: `${upperMode(p.paymentMode)} - ${p.partyName}`,
   199→    amount: Number(p.amount) || 0,
   200→    type: isVendor ? 'debit' : 'credit',
   201→    balance: null,
   202→    account: upperMode(p.paymentMode),
   203→    category: isVendor ? 'Purchase' : 'Revenue',
   204→  }
   205→}
   206→
   207→// Map raw Payment row → ReconciliationEntry shape used by this page.
   208→function mapPaymentToRecon(p: {
   209→  id: string
   210→  partyName: string
   211→  amount: number
   212→  paymentDate: string
   213→  paymentMode: string
   214→  referenceNo?: string | null
   215→  invoiceId?: string | null
   216→  status?: string
   217→  reconciled?: boolean
   218→}): ReconciliationEntry {
   219→  let status: ReconciliationEntry['status'] = 'unmatched'
   220→  if (p.reconciled) status = 'matched'
   221→  else if (p.status === 'failed') status = 'disputed'
   222→  return {
   223→    id: p.id,
   224→    date: formatTxnDate(p.paymentDate),
   225→    bankTxn: p.referenceNo || `${upperMode(p.paymentMode)}-${p.id.slice(-6)}`,
   226→    bookEntry: p.invoiceId || null,
   227→    amount: Number(p.amount) || 0,
   228→    status,
   229→    account: upperMode(p.paymentMode),
   230→  }
   231→}
