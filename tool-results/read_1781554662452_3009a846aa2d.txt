   779→function StarRating({ rating, size = 'sm' }: { rating: number; size?: 'sm' | 'md' }) {
   780→  const starSize = size === 'sm' ? 'h-3 w-3' : 'h-4 w-4';
   781→  return (
   782→    <div className="flex items-center gap-0.5">
   783→      {[1, 2, 3, 4, 5].map((star) => (
   784→        <Star
   785→          key={star}
   786→          className={`${starSize} ${
   787→            star <= Math.floor(rating)
   788→              ? 'text-amber-400 fill-amber-400'
   789→              : star <= rating
   790→                ? 'text-amber-400 fill-amber-200'
   791→                : 'text-gray-200 dark:text-gray-700'
   792→          }`}
   793→        />
   794→      ))}
   795→      <span className={`ml-1 font-medium text-muted-foreground ${size === 'sm' ? 'text-xs' : 'text-sm'}`}>
   796→        {rating.toFixed(1)}
   797→      </span>
   798→    </div>
   799→  );
   800→}
