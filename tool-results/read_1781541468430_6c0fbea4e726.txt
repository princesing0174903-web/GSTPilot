    84→    84→function getActionCategory(action: string): string {
    85→    85→  const lower = action.toLowerCase();
    86→    86→  if (lower.includes('creat')) return 'Created';
    87→    87→  if (lower.includes('updat')) return 'Updated';
    88→    88→  if (lower.includes('delet')) return 'Deleted';
    89→    89→  if (lower.includes('filed') || lower.includes('file ')) return 'Filed';
    90→    90→  if (lower.includes('generat')) return 'Generated';
    91→    91→  if (lower.includes('download')) return 'Downloaded';
    92→    92→  if (lower.includes('approv')) return 'Approved';
    93→    93→  if (lower.includes('complet')) return 'Completed';
    94→    94→  if (lower.includes('review')) return 'Reviewed';
    95→    95→  if (lower.includes('export')) return 'Exported';
    96→    96→  if (lower.includes('import')) return 'Imported';
    97→    97→  return 'Updated';
    98→    98→}

   100→   100→function formatTimestamp(ts: string): string {
   101→   101→  try {
   102→   102→    const date = new Date(ts);
   103→   103→    return format(date, 'MMM d, yyyy h:mm a');
   104→   104→  } catch {
   105→   105→    return ts;
   106→   106→  }
   107→   107→}
   108→   108→
   109→   109→function formatTimestampShort(ts: string): string {
   110→   110→  try {
   111→   111→    const date = new Date(ts);
   112→   112→    if (isToday(date)) return format(date, 'h:mm a');
   113→   113→    return format(date, 'MMM d, h:mm a');
   114→   114→  } catch {
   115→   115→    return ts;
   116→   116→  }
   117→   117→}
