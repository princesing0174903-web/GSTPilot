    99→function NotificationBell() {
   100→  const { data: unreadNotifs } = useFireUnreadNotifications()
   101→  const { data: allNotifs } = useFireNotifications()
   102→  const unreadCount = unreadNotifs.length
   103→  const [open, setOpen] = useState(false)
   104→
   105→  const handleMarkRead = async (notifId: string) => {
   106→    try { await markNotificationRead(notifId) } catch {}
   107→  }
   108→
   109→  const handleMarkAllRead = async () => {
   110→    try {
   111→      for (const n of unreadNotifs) {
   112→        await markNotificationRead(n.notifId || n.id)
   113→      }
   114→    } catch {}
   115→  }
   116→
   117→  const priorityColor = (p: string) => {
   118→    switch (p) {
   119→      case 'urgent': return 'text-red-600 bg-red-50 border-red-200'
   120→      case 'high': return 'text-orange-600 bg-orange-50 border-orange-200'
   121→      case 'normal': return 'text-blue-600 bg-blue-50 border-blue-200'
   122→      default: return 'text-slate-600 bg-slate-50 border-slate-200'
   123→    }
   124→  }
   125→
   126→  const priorityBadge = (p: string) => {
   127→    switch (p) {
   128→      case 'urgent': return 'bg-red-500'
   129→      case 'high': return 'bg-orange-500'
   130→      case 'normal': return 'bg-blue-500'
   131→      default: return 'bg-slate-400'
   132→    }
   133→  }
   134→
   135→  const typeIcon = (t: string) => {
   136→    switch (t) {
   137→      case 'filing_due': case 'filing_overdue': return <Clock className="h-3.5 w-3.5" />
   138→      case 'mismatch_found': return <AlertTriangle className="h-3.5 w-3.5" />
   139→      case 'filing_completed': return <Check className="h-3.5 w-3.5" />
   140→      case 'extraction_complete': return <FileText className="h-3.5 w-3.5" />
   141→      case 'issue_detected': return <AlertTriangle className="h-3.5 w-3.5" />
   142→      case 'ai_insight': return <Sparkles className="h-3.5 w-3.5" />
   143→      default: return <Bell className="h-3.5 w-3.5" />
   144→    }
   145→  }
   146→
   147→  const displayed = (allNotifs || []).slice(0, 15)
   148→
   149→  return (
   150→    <Popover open={open} onOpenChange={setOpen}>
   151→      <PopoverTrigger asChild>
   152→        <Button variant="ghost" size="sm" className="relative h-8 w-8 p-0 rounded-lg hover:bg-accent">
   153→          {unreadCount > 0 ? (
   154→            <BellRing className="h-4 w-4 text-amber-600" />
   155→          ) : (
   156→            <Bell className="h-4 w-4 text-muted-foreground" />

   158→          {unreadCount > 0 && (
   159→            <span className="absolute -top-0.5 -right-0.5 flex h-4 min-w-4 items-center justify-center rounded-full bg-red-500 px-1 text-[9px] font-bold text-white">
   160→              {unreadCount > 9 ? '9+' : unreadCount}
   161→            </span>
   162→          )}
   163→        </Button>
   164→      </PopoverTrigger>
   165→      <PopoverContent align="end" className="w-80 sm:w-96 p-0 gap-0" sideOffset={8}>
   166→        <div className="flex items-center justify-between px-4 py-3 border-b">
   167→          <div className="flex items-center gap-2">
   168→            <h3 className="text-sm font-semibold">Notifications</h3>
   169→            {unreadCount > 0 && (
   170→              <Badge variant="secondary" className="h-5 px-1.5 text-[10px] bg-red-100 text-red-700">
   171→                {unreadCount} new
   172→              </Badge>
   173→            )}
   174→          </div>
   175→          {unreadCount > 0 && (
   176→            <Button variant="ghost" size="sm" onClick={handleMarkAllRead} className="h-6 text-[10px] text-emerald-600 hover:text-emerald-700 px-2">
   177→              Mark all read

   181→        <ScrollArea className="max-h-80">
   182→          {displayed.length === 0 ? (
   183→            <div className="flex flex-col items-center justify-center py-8 px-4 text-center">
   184→              <Bell className="h-8 w-8 text-muted-foreground/30 mb-2" />

   186→              <p className="text-xs text-muted-foreground/60 mt-1">Notifications will appear here when returns are due, mismatches are found, or documents are processed.</p>

   218→        {displayed.length > 0 && (
   219→          <div className="border-t px-4 py-2">
   220→            <Button
   221→              variant="ghost"
   222→              size="sm"
   223→              className="w-full text-xs text-emerald-600 hover:text-emerald-700"
   224→              onClick={() => {
   225→                setOpen(false)
   226→              }}
   227→            >
   228→              View All Notifications

   261→// KEYBOARD SHORTCUTS HOOK
   262→// ═══════════════════════════════════════════════════════════════════════════════

   264→function useKeyboardShortcuts() {
   265→  const { setCurrentView, setCommandPaletteOpen, commandPaletteOpen } = useApp()
   266→  const gPressed = useRef(false)
   267→  const gTimer = useRef<NodeJS.Timeout | null>(null)
   268→
   269→  useEffect(() => {
   270→    const handler = (e: KeyboardEvent) => {
   271→      // Ctrl+K → Toggle Command Palette
   272→      if ((e.metaKey || e.ctrlKey) && e.key === 'k') {
   273→        e.preventDefault()
   274→        setCommandPaletteOpen(!commandPaletteOpen)
   275→        return
   276→      }
   277→
   278→      // G prefix shortcuts (G+D, G+C, G+I, G+R)
   279→      if (e.key === 'g' && !e.metaKey && !e.ctrlKey && !e.altKey && !(e.target instanceof HTMLInputElement || e.target instanceof HTMLTextAreaElement)) {
   280→        if (gPressed.current) return
   281→        gPressed.current = true
   282→        gTimer.current = setTimeout(() => { gPressed.current = false }, 500)
   283→        return
   284→      }
   285→
   286→      if (gPressed.current && !e.metaKey && !e.ctrlKey && !e.altKey) {
   287→        gPressed.current = false
   288→        if (gTimer.current) clearTimeout(gTimer.current)
   289→
   290→        switch (e.key) {
   291→          case 'd': setCurrentView('dashboard'); break
   292→          case 'c': setCurrentView('clients'); break
   293→          case 'i': setCurrentView('invoices'); break
   294→          case 'r': setCurrentView('returns'); break
   295→        }
   296→      }
   297→    }
   298→
   299→    window.addEventListener('keydown', handler)
   300→    return () => window.removeEventListener('keydown', handler)
   301→  }, [setCurrentView, setCommandPaletteOpen, commandPaletteOpen])
   302→}

   497→        firmId: firmId || null, firmName: data.firmName, onboardingCompleted: true,
   498→        clientCount: data.clientCount, monthlyReturns: data.monthlyReturns,
   499→        gstServices: data.gstServices, painPoints: data.painPoints,
   500→        referralSource: data.referralSource, trialReasons: data.trialReasons,
   501→        wantsUpdates: data.wantsUpdates, updatedAt: serverTimestamp(),
   502→      }, { merge: true })
   503→
   504→      await setDoc(doc(db, 'onboarding', user.id), {
   505→        ...data, firmId: firmId || null, completedAt: serverTimestamp(),
   506→      })
   507→    } catch (error) {
   508→      console.warn('[Onboarding] Background Firestore save failed:', error)
   509→    }
   510→  }
   511→
   512→  const handleOnboardingComplete = (
   513→    data: import('@/components/onboarding/OnboardingFlow').OnboardingData,
   514→    destination?: import('@/components/onboarding/OnboardingFlow').OnboardingDestination
   515→  ) => {
   516→    setSaveError(null)
   517→    if (!user) { setSaveError('No authenticated user found. Please sign in again.'); return }
   518→
   519→    markOnboardingComplete(undefined, data.firmName)
   520→    setCurrentView(destination || 'dashboard')
   521→    setCurrentScreen('app')
   522→
   523→    ;(async () => {
   524→      try {
   525→        const { collection, addDoc, serverTimestamp } = await import('firebase/firestore')
   526→        const { db } = await import('@/lib/firebase')
   527→        let firmId = ''
   528→        try {
   529→          const firmRef = await addDoc(collection(db, 'firms'), {
   530→            ownerId: user.id, firmName: data.firmName, gstin: data.gstin || null,
   531→            state: data.state, stateCode: data.stateCode,
   532→            organizationType: data.organizationType,
   533→            icaiMembershipNo: data.icaiMembershipNo || null,
   534→            officeAddress: data.officeAddress || null, createdAt: serverTimestamp(),
   535→          })
   536→          firmId = firmRef.id
   537→          markOnboardingComplete(firmId, data.firmName)
   538→        } catch (firmError) {
   539→          console.warn('[Onboarding] Firm creation failed:', firmError)
   540→        }
   541→        await saveOnboardingToFirestore(data, firmId)
   542→      } catch (error) {
   543→        console.warn('[Onboarding] Background save error:', error)
   544→      }
   545→    })()
   546→  }
   547→
   548→  const handleSkip = () => {
   549→    setSaveError(null)
   550→    if (!user) { setSaveError('No authenticated user found.'); return }
   551→
   552→    markOnboardingComplete()
   553→    setCurrentView('dashboard')
   554→    setCurrentScreen('app')
   555→
   556→    ;(async () => {
   557→      try {
   558→        const { doc, setDoc, serverTimestamp } = await import('firebase/firestore')
   559→        const { db } = await import('@/lib/firebase')
   560→        await setDoc(doc(db, 'users', user.id), {
   561→          onboardingCompleted: true, updatedAt: serverTimestamp(),
   562→        }, { merge: true })
   563→      } catch (error) {
   564→        console.warn('[Onboarding] Background skip save failed:', error)
   565→      }
   566→    })()
   567→  }
   568→
   569→  return (
   570→    <OnboardingFlow
   571→      onComplete={handleOnboardingComplete}
   572→      onSkip={handleSkip}
   573→      userEmail={user?.email}
   574→      userName={user?.name}
   575→      error={saveError}
   576→      onDismissError={() => setSaveError(null)}
   577→    />
   578→  )
   579→}

   585→function AppRouter() {
   586→  const { currentScreen, setCurrentScreen } = useApp()
   587→  const { isAuthenticated, isInitializing, needsOnboarding, needsEmailVerification } = useAuth()
   588→
   589→  useEffect(() => {
   590→    if (isInitializing) return
   591→    if (isAuthenticated && !needsOnboarding && currentScreen !== 'app') {
   592→      setCurrentScreen('app')
   593→    }
   594→  }, [isAuthenticated, isInitializing, currentScreen, setCurrentScreen, needsOnboarding])
   595→
   596→  useEffect(() => {
   597→    if (isInitializing) return
   598→    if (!isAuthenticated && currentScreen === 'app') {
   599→      setCurrentScreen('landing')
   600→    }
   601→  }, [isAuthenticated, isInitializing, currentScreen, setCurrentScreen])
   602→
   603→  const handleGetStarted = () => setCurrentScreen('login')
   604→  const handleBookDemo = () => setCurrentScreen('login')
   605→  const handleBackToLanding = () => setCurrentScreen('landing')
   606→
   607→  if (isInitializing && !isAuthenticated) {
   608→    return (
   609→      <div className="min-h-screen flex items-center justify-center bg-white">
   610→        <div className="flex flex-col items-center gap-3">
   611→          <div className="flex h-10 w-10 items-center justify-center rounded-lg bg-gradient-to-br from-emerald-500 to-emerald-600 shadow-lg shadow-emerald-600/20">
   612→            <Zap className="h-6 w-6 text-white animate-pulse" />
   613→          </div>
   614→          <span className="text-sm text-slate-500 font-medium">Loading GSTPilot...</span>
   615→        </div>
   616→      </div>
   617→    )
   618→  }
   619→
   620→  if (isAuthenticated && needsOnboarding) {
   621→    return <OnboardingScreen />
   622→  }
   623→
   624→  if (currentScreen === 'app' && isAuthenticated) {
   625→    return (
   626→      <div className="min-h-screen flex flex-col">
   627→        {needsEmailVerification && <EmailVerificationBanner />}
   628→        <div className="flex-1 flex">
   629→          <DashboardContent />
   630→        </div>
   631→      </div>
   632→    )
   633→  }
   634→
   635→  if (currentScreen === 'login') {
   636→    return <LoginPage onBack={handleBackToLanding} onGetStarted={handleGetStarted} />
   637→  }
   638→
   639→  return <LandingPage onGetStarted={handleGetStarted} onBookDemo={handleBookDemo} />
   640→}
   641→
   642→export default function Home() {
   643→  return <AppRouter />
   644→}
