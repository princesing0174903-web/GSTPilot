    36→function inrShort(n: number): string {
    37→  if (!isFinite(n) || isNaN(n)) return '₹0';
    38→  const abs = Math.abs(n);
    39→  if (abs >= 10000000) return `₹${(n / 10000000).toFixed(2)} Cr`;
    40→  if (abs >= 100000) return `₹${(n / 100000).toFixed(2)} L`;
    41→  if (abs >= 1000) return `₹${(n / 1000).toFixed(1)}K`;
    42→  return '₹' + Math.round(n).toLocaleString('en-IN');
    43→}

   424→const encoder = new TextEncoder();

   426→function sseChunk(payload: Record<string, unknown>): Uint8Array {
   427→  return encoder.encode(`data: ${JSON.stringify(payload)}\n\n`);
   428→}

   430→function sseHeaders(): HeadersInit {
   431→  return {
   432→    'Content-Type': 'text/event-stream; charset=utf-8',
   433→    'Cache-Control': 'no-cache, no-transform',
   434→    'Connection': 'keep-alive',
   435→    'X-Accel-Buffering': 'no',
   436→  };
   437→}

   439→/** Map our role names to the model's role names. */
   440→function toModelMessages(
   441→  messages: OracleChatRequest['messages'],
   442→): { role: 'assistant' | 'user'; content: string }[] {
   443→  return messages.map((m) => ({
   444→    role: (m.role === 'oracle' ? 'assistant' : 'user') as 'assistant' | 'user',
   445→    content: m.content,
   446→  }));
   447→}
   448→
   449→/** Server-side language hint derived from the latest user message. */
   450→function inferLanguageHint(messages: OracleChatRequest['messages']): OracleLanguageId | undefined {
   451→  for (let i = messages.length - 1; i >= 0; i--) {
   452→    if (messages[i].role === 'user' && messages[i].content?.trim()) {
   453→      const s = messages[i].content.slice(0, 500);
   454→      if (/[\u0900-\u097F]/.test(s)) return 'hindi';
   455→      if (/[\u0600-\u06FF]/.test(s)) return 'urdu';
   456→      if (/[\u0A00-\u0A7F]/.test(s)) return 'punjabi';
   457→      if (/[\u0A80-\u0AFF]/.test(s)) return 'gujarati';
   458→      if (/[\u0B80-\u0BFF]/.test(s)) return 'tamil';
   459→      if (/[\u0C00-\u0C7F]/.test(s)) return 'telugu';
   460→      if (/[\u0980-\u09FF]/.test(s)) return 'bengali';
   461→      return 'english';
   462→    }
   463→  }
   464→  return undefined;
   465→}

   469→export async function POST(request: Request) {
   470→  let body: OracleChatRequest;
   471→  try {
   472→    body = (await request.json()) as OracleChatRequest;
   473→  } catch {
   474→    return new Response(JSON.stringify({ error: 'Invalid JSON body' }), {
   475→      status: 400,
   476→      headers: { 'Content-Type': 'application/json' },
   477→    });
   478→  }
   479→
   480→  const messages = body.messages ?? [];
   481→  if (!Array.isArray(messages) || messages.length === 0) {
   482→    return new Response(JSON.stringify({ error: 'messages[] is required' }), {
   483→      status: 400,
   484→      headers: { 'Content-Type': 'application/json' },
   485→    });
   486→  }
   487→
   488→  const systemPrompt = await buildSystemPrompt(body);
   489→  const languageHint = inferLanguageHint(messages);
   490→
   491→  const modelMessages: { role: 'assistant' | 'user' | 'system'; content: string }[] = [
   492→    { role: 'assistant', content: systemPrompt },
   493→    ...toModelMessages(messages),
   494→  ];
   495→
   496→  // ── Acquire the upstream stream from the SDK ───────────────────────────────
   497→  let upstream: ReadableStream<Uint8Array> | null = null;
   498→  try {
   499→    const zai = await ZAI.create();
   500→    const result = await zai.chat.completions.create({
   501→      messages: modelMessages,
   502→      stream: true,
   503→      thinking: { type: 'disabled' },
   504→    });
   505→    if (result && typeof (result as ReadableStream<Uint8Array>).getReader === 'function') {
   506→      upstream = result as ReadableStream<Uint8Array>;
   507→    } else {
   508→      // Non-streaming fallback: emit the full text as one chunk then close.
   509→      const text =
   510→        (result as { choices?: { message?: { content?: string } }[] })?.choices?.[0]?.message
   511→          ?.content ?? '';
   512→      upstream = new ReadableStream<Uint8Array>({
   513→        start(controller) {
   514→          if (text) controller.enqueue(sseChunk({ token: text }));
   515→          controller.close();
   516→        },
   517→      });
   518→    }
   519→  } catch (err) {
   520→    const message = err instanceof Error ? err.message : 'Unknown error';
   521→    const stream = new ReadableStream<Uint8Array>({
   522→      start(controller) {
   523→        controller.enqueue(
   524→          sseChunk({
   525→            token: `I'm here, but I hit a temporary issue reaching my reasoning service (${message}). Please try again in a moment — your conversation is safe.`,
   526→          }),
   527→        );
   528→        controller.enqueue(sseChunk({ done: true }));
   529→        controller.close();
   530→      },
   531→    });
   532→    return new Response(stream, { status: 200, headers: sseHeaders() });
   533→  }
   534→
   535→  // ── Transform the upstream SSE stream into our token stream ────────────────
   536→  const decoder = new TextDecoder();
   537→  const transformed = new ReadableStream<Uint8Array>({
   538→    async start(controller) {
   539→      // Emit a tiny first nudge so the UI shows the pulsing cursor within the
   540→      // first frame — real tokens follow immediately.
   541→      if (languageHint) controller.enqueue(sseChunk({ language: languageHint }));
   542→
   543→      const reader = upstream!.getReader();
   544→      let buffer = '';
   545→      let emittedAny = false;
   546→      try {
   547→        while (true) {
   548→          const { done, value } = await reader.read();
   549→          if (done) break;
   550→          buffer += decoder.decode(value, { stream: true });
   551→
   552→          const lines = buffer.split('\n');
   553→          buffer = lines.pop() ?? '';
   554→
   555→          for (const rawLine of lines) {
   556→            const line = rawLine.trim();
   557→            if (!line || !line.startsWith('data:')) continue;
   558→            const data = line.slice(5).trim();
   559→            if (data === '[DONE]' || !data) continue;
   560→            try {
   561→              const json = JSON.parse(data);
   562→              const token: string =
   563→                json?.choices?.[0]?.delta?.content ??
   564→                json?.choices?.[0]?.message?.content ??
   565→                '';
   566→              if (token) {
   567→                emittedAny = true;
   568→                controller.enqueue(sseChunk({ token }));
   569→              }
   570→            } catch {
   571→              // Partial JSON across a chunk boundary — resolves on next read.
   572→            }
   573→          }
   574→        }
   575→        // Flush any trailing buffered line.
   576→        const tail = buffer.trim();
   577→        if (tail.startsWith('data:')) {
   578→          const data = tail.slice(5).trim();
   579→          if (data && data !== '[DONE]') {
   580→            try {
   581→              const json = JSON.parse(data);
   582→              const token: string =
   583→                json?.choices?.[0]?.delta?.content ??
   584→                json?.choices?.[0]?.message?.content ??
   585→                '';
   586→              if (token) {
   587→                emittedAny = true;
   588→                controller.enqueue(sseChunk({ token }));
   589→              }
   590→            } catch {
   591→              /* ignore */
   592→            }
   593→          }
   594→        }
   595→        if (!emittedAny) {
   596→          controller.enqueue(
   597→            sseChunk({
   598→              token:
   599→                "I'm here. Based on current GST rules and the information available, I'd be glad to help — could you share a bit more about what you're looking to do?",
   600→            }),
   601→          );
   602→        }
   603→        controller.enqueue(sseChunk({ done: true }));
   604→        controller.close();
   605→      } catch (err) {
   606→        const message = err instanceof Error ? err.message : 'Stream interrupted';
   607→        if (!emittedAny) {
   608→          controller.enqueue(
   609→            sseChunk({
   610→              token: `My response was interrupted (${message}). Please try sending that again.`,
   611→            }),
   612→          );
   613→        }
   614→        controller.enqueue(sseChunk({ done: true }));
   615→        controller.close();
   616→      }
   617→    },
   618→    cancel() {
   619→      upstream?.cancel?.().catch(() => undefined);
   620→    },
   621→  });
   622→
   623→  return new Response(transformed, { status: 200, headers: sseHeaders() });
   624→}
