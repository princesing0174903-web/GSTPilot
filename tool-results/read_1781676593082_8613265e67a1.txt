   179→function formatINR(amount: number): string {

   184→function formatDate(ts: number | string | undefined): string {
   185→  try {
   186→    const d = typeof ts === 'number' ? new Date(ts) : new Date(ts as string);
   187→    if (isNaN(d.getTime())) return '—';
   188→    const dd = String(d.getDate()).padStart(2, '0');
   189→    const mm = String(d.getMonth() + 1).padStart(2, '0');
   190→    const yyyy = d.getFullYear();
   191→    return `${dd}/${mm}/${yyyy}`;
   192→  } catch {
   193→    return '—';
   194→  }
   195→}
   196→
   197→function formatTime(ts: number): string {
   198→  const d = new Date(ts);
   199→  let h = d.getHours();
   200→  const m = String(d.getMinutes()).padStart(2, '0');
   201→  const ampm = h >= 12 ? 'PM' : 'AM';
   202→  h = h % 12 || 12;
   203→  return `${h}:${m} ${ampm}`;
   204→}

   213→function buildLiveDataContext(opts: {
   214→  clients: Array<Record<string, unknown>>;
   215→  invoices: Array<Record<string, unknown>>;
   216→  returns: Array<Record<string, unknown>>;
   217→  notifications: Array<Record<string, unknown>>;
   218→  activities: Array<Record<string, unknown>>;
   219→}): string {
   220→  try {
   221→    const { clients, invoices, returns, notifications, activities } = opts;
   222→    const now = new Date();
   223→    const dd = String(now.getDate()).padStart(2, '0');
   224→    const mm = String(now.getMonth() + 1).padStart(2, '0');
   225→    const yyyy = now.getFullYear();
   226→    const hh = String(now.getHours()).padStart(2, '0');
   227→    const min = String(now.getMinutes()).padStart(2, '0');
   228→    const stamp = `${dd}/${mm}/${yyyy} ${hh}:${min}`;
   229→
   230→    const lines: string[] = [`LIVE DATA SNAPSHOT (${stamp}):`];
   231→
   232→    // Clients
   233→    const totalClients = clients.length;
   234→    const activeClients = clients.filter((c) => c.status === 'active').length;
   235→    lines.push(`- Total clients: ${totalClients}`);
   236→    lines.push(`- Active clients: ${activeClients}`);
   237→
   238→    // Invoices
   239→    const activeInvoices = invoices.filter(
   240→      (i) => i.status !== 'cancelled' && i.status !== 'draft',
   241→    );
   242→    const totalInvoiceValue = activeInvoices.reduce((s, i) => {
   243→      const v =
   244→        (i.totalAmount as number) ||
   245→        (i.grandTotal as number) ||
   246→        (i.total as number) ||
   247→        ((i.subTotal as number) || 0) + ((i.totalTax as number) || 0);
   248→      return s + (typeof v === 'number' ? v : 0);
   249→    }, 0);
   250→    const overdueInvoices = activeInvoices.filter((i) => {
   251→      const status = i.status as string;
   252→      return status === 'overdue' || status === 'unpaid';
   253→    });
   254→    const overdueValue = overdueInvoices.reduce((s, i) => {
   255→      const v =
   256→        (i.totalAmount as number) ||
   257→        (i.grandTotal as number) ||
   258→        (i.total as number) ||
   259→        0;
   260→      return s + (typeof v === 'number' ? v : 0);
   261→    }, 0);
   262→    lines.push(
   263→      `- Active invoices: ${activeInvoices.length} (${formatINR(totalInvoiceValue)} total)`,
   264→    );
   265→    lines.push(
   266→      `- Overdue invoices: ${overdueInvoices.length} (${formatINR(overdueValue)})`,
   267→    );
   268→
   269→    // Returns
   270→    const gstr1Pending = returns.filter(
   271→      (r) => (r.returnType as string) === 'GSTR-1' && (r.status as string) !== 'filed',
   272→    ).length;
   273→    const gstr3bPending = returns.filter(
   274→      (r) => (r.returnType as string) === 'GSTR-3B' && (r.status as string) !== 'filed',
   275→    ).length;
   276→    const filedReturns = returns.filter((r) => r.status === 'filed').length;
   277→    lines.push(
   278→      `- Pending returns: ${gstr1Pending} GSTR-1, ${gstr3bPending} GSTR-3B`,
   279→    );
   280→    lines.push(`- Filed returns: ${filedReturns}`);
   281→
   282→    // Notifications + Activities
   283→    const unread = notifications.filter((n) => n.read === false).length;
   284→    lines.push(`- Unread notifications: ${unread}`);
   285→    lines.push(`- Recent activities: ${activities.length} in last 24h`);
   286→
   287→    // Top 5 clients by revenue
   288→    const clientRevenue = new Map<string, number>();
   289→    invoices.forEach((inv) => {
   290→      const cid = (inv.clientId as string) || (inv.clientName as string) || 'unknown';
   291→      const v =
   292→        (inv.totalAmount as number) ||
   293→        (inv.grandTotal as number) ||
   294→        (inv.total as number) ||
   295→        0;
   296→      clientRevenue.set(cid, (clientRevenue.get(cid) || 0) + (typeof v === 'number' ? v : 0));
   297→    });
   298→    const top5 = [...clientRevenue.entries()]
   299→      .sort((a, b) => b[1] - a[1])
   300→      .slice(0, 5);
   301→    if (top5.length > 0) {
   302→      const top5Str = top5
   303→        .map(([name, rev]) => {
   304→          const client = clients.find(
   305→            (c) => c.id === name || (c.tradeName as string) === name,
   306→          );
   307→          const label = (client?.tradeName as string) || name;
   308→          return `${label} (${formatINR(rev)})`;
   309→        })
   310→        .join(', ');
   311→      lines.push(`- Top 5 clients by revenue: ${top5Str}`);
   312→    }
   313→
   314→    // High-risk clients (health score < 60)
   315→    const highRisk = clients.filter((c) => {
   316→      const score = (c.healthScore as number) ?? 100;
   317→      return typeof score === 'number' && score < 60;
   318→    });
   319→    if (highRisk.length > 0) {
   320→      const riskStr = highRisk
   321→        .slice(0, 5)
   322→        .map(
   323→          (c) =>
   324→            `${(c.tradeName as string) || (c.name as string) || 'Unknown'} (health ${c.healthScore ?? '?'})`,
   325→        )
   326→        .join(', ');
   327→      lines.push(`- High-risk clients: ${riskStr}`);
   328→    }
   329→
   330→    return lines.join('\n');
   331→  } catch {
   332→    return 'LIVE DATA SNAPSHOT: Data currently unavailable.';
   333→  }
   334→}

   358→// ─── Sub-component: Message Content (mini-markdown) ───────────────────────────
   359→// Renders **bold**, line breaks, bullet points (•), and numbered lists.
   360→
   361→function MessageContent({ text }: { text: string }) {
   362→  const blocks = useMemo(() => {
   363→    const paragraphs = text.split('\n');
   364→    return paragraphs.map((line) => {
   365→      const trimmed = line.trim();
   366→      if (trimmed.startsWith('• ') || trimmed.startsWith('- ')) {
   367→        return { type: 'bullet' as const, content: trimmed.slice(2) };
   368→      }
   369→      if (/^\d+\.\s/.test(trimmed)) {
   370→        return { type: 'numbered' as const, content: trimmed.replace(/^\d+\.\s/, '') };
   371→      }
   372→      return { type: 'text' as const, content: trimmed };
   373→    });
   374→  }, [text]);
   375→
   376→  const renderInline = (s: string, keyBase: string) => {
   377→    // Split on **bold** markers
   378→    const parts = s.split(/(\*\*[^*]+\*\*)/g);
   379→    return parts.map((part, i) => {
   380→      if (part.startsWith('**') && part.endsWith('**')) {
   381→        return (
   382→          <strong key={`${keyBase}-${i}`} className="font-semibold text-foreground">
   383→            {part.slice(2, -2)}
   384→          </strong>
   385→        );
   386→      }
   387→      return <span key={`${keyBase}-${i}`}>{part}</span>;
   388→    });
   389→  };
   390→
   391→  return (
   392→    <div className="space-y-1.5">
   393→      {blocks.map((b, idx) => {
   394→        if (b.type === 'bullet') {
   395→          return (
   396→            <div key={idx} className="flex gap-2">
   397→              <span className="mt-0.5 text-emerald-400">•</span>
   398→              <span className="text-foreground/90">{renderInline(b.content, `b${idx}`)}</span>
   399→            </div>
   400→          );
   401→        }
   402→        if (b.type === 'numbered') {
   403→          return (
   404→            <div key={idx} className="flex gap-2">
   405→              <span className="font-medium accent-text">{idx + 1}.</span>
   406→              <span className="text-foreground/90">{renderInline(b.content, `n${idx}`)}</span>
   407→            </div>
   408→          );
   409→        }
   410→        if (!b.content) return <div key={idx} className="h-1" />;
   411→        return (
   412→          <p key={idx} className="text-foreground/90">
   413→            {renderInline(b.content, `t${idx}`)}
   414→          </p>
   415→        );
   416→      })}
   417→    </div>
   418→  );
   419→}

   463→// ─── Hook: Typing Effect ──────────────────────────────────────────────────────
   464→// Reveals text progressively in small chunks for a "typing" feel.
   465→// v10: 2 chars / 10ms (faster than the old 3 chars / 16ms).
   466→
   467→function useTypingEffect(fullText: string, active: boolean, speed = 10, chunkSize = 2) {
   468→  // revealLength is only mutated inside the interval callback (never
   469→  // synchronously in the effect body), which keeps React's
   470→  // set-state-in-effect lint rule happy and avoids cascading renders.
   471→  const [revealLength, setRevealLength] = useState(0);
   472→
   473→  useEffect(() => {
   474→    if (!active) return;
   475→    let i = 0;
   476→    const interval = setInterval(() => {
   477→      i += chunkSize;
   478→      setRevealLength(i);
   479→      if (i >= fullText.length) {
   480→        clearInterval(interval);
   481→      }
   482→    }, speed);
   483→    return () => clearInterval(interval);
   484→  }, [fullText, active, speed, chunkSize]);
   485→
   486→  // When not active (e.g. older messages), show the full text instantly
   487→  // via a derived value — no state mutation required.
   488→  const displayed = active ? fullText.slice(0, revealLength) : fullText;
   489→  const completed = displayed.length >= fullText.length;
   490→
   491→  return { displayed, completed };
   492→}
