   880→   880→/** Subscribe to a Firestore collection with firm scoping */
   881→   881→export function subscribeToCollection<T>(
   882→   882→  collectionName: CollectionName,
   883→   883→  callback: (data: Array<T & { id: string }>) => void,
   884→   884→  ...constraints: QueryConstraint[]
   885→   885→): Unsubscribe {
   886→   886→  const firmId = currentFirmId();
   887→   887→
   888→   888→  const q = firmId
   889→   889→    ? query(collection(db, collectionName), where('firmId', '==', firmId), ...constraints)
   890→   890→    : query(collection(db, collectionName), ...constraints);
   891→   891→
   892→   892→  return onSnapshot(q, (snap) => {
   893→   893→    const data = snap.docs.map(d => {
   894→   894→      const docData = d.data();
   895→   895→      const converted: Record<string, unknown> = { id: d.id };
   896→   896→      for (const [key, value] of Object.entries(docData)) {
   897→   897→        if (value && typeof value === 'object' && 'toDate' in value && typeof (value as { toDate: () => Date }).toDate === 'function') {
   898→   898→          converted[key] = (value as { toDate: () => Date }).toDate().toISOString();
   899→   899→        } else {
   900→   900→          converted[key] = value;
   901→   901→        }
   902→   902→      }
   903→   903→      return converted as T & { id: string };
   904→   904→    });
   905→   905→    callback(data);
   906→   906→  }, (error) => {
   907→   907→    console.warn(`[Firestore] Subscription error for ${collectionName}:`, error);
   908→   908→    callback([]);
   909→   909→  });
   910→   910→}
   911→   911→
   912→   912→/** Subscribe to a single Firestore document */
   913→   913→export function subscribeToDoc<T>(
   914→   914→  collectionName: CollectionName,
   915→   915→  docId: string,
   916→   916→  callback: (data: (T & { id: string }) | null) => void,
   917→   917→): Unsubscribe {
   918→   918→  return onSnapshot(doc(db, collectionName, docId), (snap) => {
   919→   919→    if (!snap.exists()) {
   920→   920→      callback(null);
   921→   921→      return;
   922→   922→    }
   923→   923→    const docData = snap.data();
   924→   924→    const converted: Record<string, unknown> = { id: snap.id };
   925→   925→    for (const [key, value] of Object.entries(docData)) {
   926→   926→      if (value && typeof value === 'object' && 'toDate' in value && typeof (value as { toDate: () => Date }).toDate === 'function') {
   927→   927→        converted[key] = (value as { toDate: () => Date }).toDate().toISOString();
   928→   928→      } else {
   929→   929→        converted[key] = value;
   930→   930→      }
   931→   931→    }
   932→   932→    callback(converted as T & { id: string });
   933→   933→  }, (error) => {
   934→   934→    console.warn(`[Firestore] Doc subscription error for ${collectionName}/${docId}:`, error);
   935→   935→    callback(null);
   936→   936→  });
   937→   937→}

   939→   939→/** Compute live dashboard metrics from Firestore data */
   940→   940→export function computeDashboardMetrics(
   941→   941→  clients: Array<FirestoreClient & { id: string }>,
   942→   942→  invoices: Array<FirestoreInvoice & { id: string }>,
   943→   943→  returns: Array<FirestoreReturn & { id: string }>,
   944→   944→  documents: Array<FirestoreDocument & { id: string }>,
   945→   945→  activities: Array<FirestoreActivity & { id: string }>,
   946→   946→): LiveDashboardMetrics {
   947→   947→  const activeClients = clients.filter(c => c.status === 'active').length;
   948→   948→  const filedReturns = returns.filter(r => r.status === 'filed').length;
   949→   949→  const pendingReturns = returns.filter(r => r.status !== 'filed').length;
   950→   950→  const readyToFile = returns.filter(r => ['validated', 'reviewed', 'generated'].includes(r.status)).length;
   951→   951→  const overdueReturns = returns.filter(r => {
   952→   952→    if (r.status === 'filed') return false;
   953→   953→    const periodDate = parsePeriod(r.period);
   954→   954→    return periodDate && periodDate < new Date();
   955→   955→  }).length;
   956→   956→
   957→   957→  const totalTaxVolume = invoices.reduce((sum, inv) => sum + (inv.totalAmount || 0), 0);
   958→   958→  const totalTax = invoices.reduce((sum, inv) => sum + (inv.cgst || 0) + (inv.sgst || 0) + (inv.igst || 0) + (inv.cess || 0), 0);
   959→   959→
   960→   960→  const criticalIssues = invoices.filter(inv => inv.riskLevel === 'critical').length +
   961→   961→    returns.filter(r => r.criticalErrors > 0).length;
   962→   962→  const warnings = invoices.filter(inv => inv.riskLevel === 'high' || inv.riskLevel === 'medium').length +
   963→   963→    returns.filter(r => r.warnings > 0).length;
   964→   964→
   965→   965→  const averageHealthScore = clients.length > 0
   966→   966→    ? Math.round(clients.reduce((sum, c) => sum + c.healthScore, 0) / clients.length)
   967→   967→    : 0;
   968→   968→
   969→   969→  const matchedInvoices = invoices.filter(inv => inv.matchStatus === 'perfect_match').length;
   970→   970→  const matchPercentage = invoices.length > 0 ? Math.round((matchedInvoices / invoices.length) * 100) : 100;
   971→   971→
   972→   972→  const riskyInvoices = invoices.filter(inv => ['high', 'critical'].includes(inv.riskLevel)).length;
   973→   973→  const riskPercentage = invoices.length > 0 ? Math.round((riskyInvoices / invoices.length) * 100) : 0;
   974→   974→
   975→   975→  const documentsProcessed = documents.filter(d => d.extractionStatus === 'completed').length;
   976→   976→  const extractionsPending = documents.filter(d => d.extractionStatus === 'pending' || d.extractionStatus === 'in_progress').length;
   977→   977→
   978→   978→  const recentActivities = [...activities]
   979→   979→    .sort((a, b) => {
   980→   980→      const aTime = typeof a.createdAt === 'string' ? new Date(a.createdAt).getTime() : 0;
   981→   981→      const bTime = typeof b.createdAt === 'string' ? new Date(b.createdAt).getTime() : 0;
   982→   982→      return bTime - aTime;
   983→   983→    })
   984→   984→    .slice(0, 10);
   985→   985→
   986→   986→  const upcomingFilings = returns
   987→   987→    .filter(r => r.status !== 'filed')
   988→   988→    .sort((a, b) => {
   989→   989→      const aTime = typeof a.createdAt === 'string' ? new Date(a.createdAt).getTime() : 0;
   990→   990→      const bTime = typeof b.createdAt === 'string' ? new Date(b.createdAt).getTime() : 0;
   991→   991→      return aTime - bTime;
   992→   992→    })
   993→   993→    .slice(0, 5);
   994→   994→
   995→   995→  return {
   996→   996→    totalClients: clients.length,
   997→   997→    activeClients,
   998→   998→    totalInvoices: invoices.length,
   999→   999→    totalTaxVolume,
  1000→  1000→    filedReturns,
  1001→  1001→    pendingReturns,
  1002→  1002→    overdueReturns,
  1003→  1003→    readyToFile,
  1004→  1004→    criticalIssues,
  1005→  1005→    warnings,
  1006→  1006→    averageHealthScore,
  1007→  1007→    matchPercentage,
  1008→  1008→    riskPercentage,
  1009→  1009→    documentsProcessed,
  1010→  1010→    extractionsPending,
  1011→  1011→    recentActivities,
  1012→  1012→    upcomingFilings,
  1013→  1013→  };
  1014→  1014→}
  1015→  1015→
  1016→  1016→function parsePeriod(period: string | null): Date | null {
  1017→  1017→  if (!period) return null;
  1018→  1018→  try {
  1019→  1019→    const [mm, yyyy] = period.split('-').map(Number);
  1020→  1020→    if (!mm || !yyyy) return null;
  1021→  1021→    return new Date(yyyy, mm, 20); // Due date is typically 20th of next month
  1022→  1022→  } catch {
  1023→  1023→    return null;
  1024→  1024→  }
  1025→  1025→}
