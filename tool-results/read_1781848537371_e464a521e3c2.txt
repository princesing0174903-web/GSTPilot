   181→function formatINR(amount: number): string {
   182→  if (!isFinite(amount) || isNaN(amount)) return '₹0';

   186→function formatDate(ts: number | string | undefined): string {
   187→  try {
   188→    const d = typeof ts === 'number' ? new Date(ts) : new Date(ts as string);
   189→    if (isNaN(d.getTime())) return '—';
   190→    const dd = String(d.getDate()).padStart(2, '0');
   191→    const mm = String(d.getMonth() + 1).padStart(2, '0');
   192→    const yyyy = d.getFullYear();
   193→    return `${dd}/${mm}/${yyyy}`;
   194→  } catch {
   195→    return '—';
   196→  }
   197→}
   198→
   199→function formatTime(ts: number): string {
   200→  const d = new Date(ts);
   201→  let h = d.getHours();
   202→  const m = String(d.getMinutes()).padStart(2, '0');
   203→  const ampm = h >= 12 ? 'PM' : 'AM';
   204→  h = h % 12 || 12;
   205→  return `${h}:${m} ${ampm}`;
   206→}

   215→function buildLiveDataContext(opts: {
   216→  clients: Array<Record<string, unknown>>;
   217→  invoices: Array<Record<string, unknown>>;
   218→  returns: Array<Record<string, unknown>>;
   219→  notifications: Array<Record<string, unknown>>;
   220→  activities: Array<Record<string, unknown>>;
   221→}): string {
   222→  try {
   223→    const { clients, invoices, returns, notifications, activities } = opts;
   224→    const now = new Date();
   225→    const dd = String(now.getDate()).padStart(2, '0');
   226→    const mm = String(now.getMonth() + 1).padStart(2, '0');
   227→    const yyyy = now.getFullYear();
   228→    const hh = String(now.getHours()).padStart(2, '0');
   229→    const min = String(now.getMinutes()).padStart(2, '0');
   230→    const stamp = `${dd}/${mm}/${yyyy} ${hh}:${min}`;
   231→
   232→    const lines: string[] = [`LIVE DATA SNAPSHOT (${stamp}):`];
   233→
   234→    // Clients
   235→    const totalClients = clients.length;
   236→    const activeClients = clients.filter((c) => c.status === 'active').length;
   237→    lines.push(`- Total clients: ${totalClients}`);
   238→    lines.push(`- Active clients: ${activeClients}`);
   239→
   240→    // Invoices
   241→    const activeInvoices = invoices.filter(
   242→      (i) => i.status !== 'cancelled' && i.status !== 'draft',
   243→    );
   244→    const totalInvoiceValue = activeInvoices.reduce((s, i) => {
   245→      const v =
   246→        (i.totalAmount as number) ||
   247→        (i.grandTotal as number) ||
   248→        (i.total as number) ||
   249→        ((i.subTotal as number) || 0) + ((i.totalTax as number) || 0);
   250→      return s + (typeof v === 'number' ? v : 0);
   251→    }, 0);
   252→    const overdueInvoices = activeInvoices.filter((i) => {
   253→      const status = i.status as string;
   254→      return status === 'overdue' || status === 'unpaid';
   255→    });
   256→    const overdueValue = overdueInvoices.reduce((s, i) => {
   257→      const v =
   258→        (i.totalAmount as number) ||
   259→        (i.grandTotal as number) ||
   260→        (i.total as number) ||
   261→        0;
   262→      return s + (typeof v === 'number' ? v : 0);
   263→    }, 0);
   264→    lines.push(
   265→      `- Active invoices: ${activeInvoices.length} (${formatINR(totalInvoiceValue)} total)`,
   266→    );
   267→    lines.push(
   268→      `- Overdue invoices: ${overdueInvoices.length} (${formatINR(overdueValue)})`,
   269→    );
   270→
   271→    // Returns
   272→    const gstr1Pending = returns.filter(
   273→      (r) => (r.returnType as string) === 'GSTR-1' && (r.status as string) !== 'filed',
   274→    ).length;
   275→    const gstr3bPending = returns.filter(
   276→      (r) => (r.returnType as string) === 'GSTR-3B' && (r.status as string) !== 'filed',
   277→    ).length;
   278→    const filedReturns = returns.filter((r) => r.status === 'filed').length;
   279→    lines.push(
   280→      `- Pending returns: ${gstr1Pending} GSTR-1, ${gstr3bPending} GSTR-3B`,
   281→    );
   282→    lines.push(`- Filed returns: ${filedReturns}`);
   283→
   284→    // Notifications + Activities
   285→    const unread = notifications.filter((n) => n.read === false).length;
   286→    lines.push(`- Unread notifications: ${unread}`);
   287→    lines.push(`- Recent activities: ${activities.length} in last 24h`);
   288→
   289→    // Top 5 clients by revenue
   290→    const clientRevenue = new Map<string, number>();
   291→    invoices.forEach((inv) => {
   292→      const cid = (inv.clientId as string) || (inv.clientName as string) || 'unknown';
   293→      const v =
   294→        (inv.totalAmount as number) ||
   295→        (inv.grandTotal as number) ||
   296→        (inv.total as number) ||
   297→        0;
   298→      clientRevenue.set(cid, (clientRevenue.get(cid) || 0) + (typeof v === 'number' ? v : 0));
   299→    });
   300→    const top5 = [...clientRevenue.entries()]
   301→      .sort((a, b) => b[1] - a[1])
   302→      .slice(0, 5);
   303→    if (top5.length > 0) {
   304→      const top5Str = top5
   305→        .map(([name, rev]) => {
   306→          const client = clients.find(
   307→            (c) => c.id === name || (c.tradeName as string) === name,
   308→          );
   309→          const label = (client?.tradeName as string) || name;
   310→          return `${label} (${formatINR(rev)})`;
   311→        })
   312→        .join(', ');
   313→      lines.push(`- Top 5 clients by revenue: ${top5Str}`);
   314→    }
   315→
   316→    // High-risk clients (health score < 60)
   317→    const highRisk = clients.filter((c) => {
   318→      const score = (c.healthScore as number) ?? 100;
   319→      return typeof score === 'number' && score < 60;
   320→    });
   321→    if (highRisk.length > 0) {
   322→      const riskStr = highRisk
   323→        .slice(0, 5)
   324→        .map(
   325→          (c) =>
   326→            `${(c.tradeName as string) || (c.name as string) || 'Unknown'} (health ${c.healthScore ?? '?'})`,
   327→        )
   328→        .join(', ');
   329→      lines.push(`- High-risk clients: ${riskStr}`);
   330→    }
   331→
   332→    return lines.join('\n');
   333→  } catch {
   334→    return 'LIVE DATA SNAPSHOT: Data currently unavailable.';
   335→  }
   336→}

   360→// ─── Sub-component: Message Content (mini-markdown) ───────────────────────────
   361→// Renders **bold**, line breaks, bullet points (•), and numbered lists.
   362→
   363→function MessageContent({ text }: { text: string }) {
   364→  const blocks = useMemo(() => {
   365→    const paragraphs = text.split('\n');
   366→    return paragraphs.map((line) => {
   367→      const trimmed = line.trim();
   368→      if (trimmed.startsWith('• ') || trimmed.startsWith('- ')) {
   369→        return { type: 'bullet' as const, content: trimmed.slice(2) };
   370→      }
   371→      if (/^\d+\.\s/.test(trimmed)) {
   372→        return { type: 'numbered' as const, content: trimmed.replace(/^\d+\.\s/, '') };
   373→      }
   374→      return { type: 'text' as const, content: trimmed };
   375→    });
   376→  }, [text]);
   377→
   378→  const renderInline = (s: string, keyBase: string) => {
   379→    // Split on **bold** markers
   380→    const parts = s.split(/(\*\*[^*]+\*\*)/g);
   381→    return parts.map((part, i) => {
   382→      if (part.startsWith('**') && part.endsWith('**')) {
   383→        return (
   384→          <strong key={`${keyBase}-${i}`} className="font-semibold text-foreground">
   385→            {part.slice(2, -2)}
   386→          </strong>
   387→        );
   388→      }
   389→      return <span key={`${keyBase}-${i}`}>{part}</span>;
   390→    });
   391→  };
   392→
   393→  return (
   394→    <div className="space-y-1.5">
   395→      {blocks.map((b, idx) => {
   396→        if (b.type === 'bullet') {
   397→          return (
   398→            <div key={idx} className="flex gap-2">
   399→              <span className="mt-0.5 text-emerald-400">•</span>
   400→              <span className="text-foreground/90">{renderInline(b.content, `b${idx}`)}</span>
   401→            </div>
   402→          );
   403→        }
   404→        if (b.type === 'numbered') {
   405→          return (
   406→            <div key={idx} className="flex gap-2">
   407→              <span className="font-medium accent-text">{idx + 1}.</span>
   408→              <span className="text-foreground/90">{renderInline(b.content, `n${idx}`)}</span>
   409→            </div>
   410→          );
   411→        }
   412→        if (!b.content) return <div key={idx} className="h-1" />;
   413→        return (
   414→          <p key={idx} className="text-foreground/90">
   415→            {renderInline(b.content, `t${idx}`)}
   416→          </p>
   417→        );
   418→      })}
   419→    </div>
   420→  );
   421→}

   465→// ─── Hook: Typing Effect ──────────────────────────────────────────────────────
   466→// Reveals text progressively in small chunks for a "typing" feel.
   467→// v10: 2 chars / 10ms (faster than the old 3 chars / 16ms).
   468→
   469→function useTypingEffect(fullText: string, active: boolean, speed = 10, chunkSize = 2) {
   470→  // revealLength is only mutated inside the interval callback (never
   471→  // synchronously in the effect body), which keeps React's
   472→  // set-state-in-effect lint rule happy and avoids cascading renders.
   473→  const [revealLength, setRevealLength] = useState(0);
   474→
   475→  useEffect(() => {
   476→    if (!active) return;
   477→    let i = 0;
   478→    const interval = setInterval(() => {
   479→      i += chunkSize;
   480→      setRevealLength(i);
   481→      if (i >= fullText.length) {
   482→        clearInterval(interval);
   483→      }
   484→    }, speed);
   485→    return () => clearInterval(interval);
   486→  }, [fullText, active, speed, chunkSize]);
   487→
   488→  // When not active (e.g. older messages), show the full text instantly
   489→  // via a derived value — no state mutation required.
   490→  const displayed = active ? fullText.slice(0, revealLength) : fullText;
   491→  const completed = displayed.length >= fullText.length;
   492→
   493→  return { displayed, completed };
   494→}

   496→// ─── Typing Message Wrapper ───────────────────────────────────────────────────
   497→// Renders an AI message with a progressive typing reveal.

   499→function TypingMessage({
   500→  message,
   501→  isLatest,
   502→  onTypingComplete,
   503→}: {
   504→  message: ChatMessage;
   505→  isLatest: boolean;
   506→  onTypingComplete: () => void;
   507→}) {
   508→  const { displayed, completed } = useTypingEffect(
   509→    message.content,
   510→    isLatest,
   511→    10,
   512→    2,
   513→  );

   515→  useEffect(() => {
   516→    if (completed && isLatest) {
   517→      onTypingComplete();
   518→    }
   519→  }, [completed, isLatest, onTypingComplete]);
   520→
   521→  return (
   522→    <div className="space-y-1">
   523→      <MessageContent text={displayed} />
   524→      {!completed && (
   525→        <motion.span
   526→          className="inline-block h-3.5 w-1.5 rounded-sm bg-emerald-400 align-middle"
   527→          animate={{ opacity: [1, 0, 1] }}
   528→          transition={{ duration: 0.8, repeat: Infinity }}
   529→        />
   530→      )}
   531→    </div>
   532→  );
   533→}
