    90→// ─── Timestamp conversion ────────────────────────────────────────────────────
    91→
    92→function ts(value: unknown): string {
    93→  if (!value) return '';
    94→  if (value && typeof value === 'object' && 'toDate' in value) {
    95→    return (value as { toDate: () => Date }).toDate().toISOString();
    96→  }
    97→  if (typeof value === 'string') return value;
    98→  if (value instanceof Date) return value.toISOString();
    99→  try {
   100→    return new Date(value as string).toISOString();
   101→  } catch {
   102→    return '';
   103→  }
   104→}

   106→function toAttachmentArray(raw: unknown): EmailAttachment[] {
   107→  if (!Array.isArray(raw)) return [];
   108→  return raw.map((r, i) => {
   109→    const a = r as Record<string, unknown>;
   110→    return {
   111→      id: String(a.id ?? `att-${i}`),
   112→      filename: String(a.filename ?? 'unknown'),
   113→      mimeType: String(a.mimeType ?? 'application/octet-stream'),
   114→      size: Number(a.size ?? 0),
   115→      storageUrl: a.storageUrl ? String(a.storageUrl) : null,
   116→    };
   117→  });
   118→}

   353→/**
   354→ * Bulk save synced emails. Existing messages (same messageId) are skipped.
   355→ * Returns the number actually written.
   356→ */
   357→export async function saveGmailMessages(
   358→  organizationId: string,
   359→  messages: GmailMessage[],
   360→): Promise<number> {
   361→  assertOrg(organizationId);
   362→  if (messages.length === 0) return 0;
   363→
   364→  // Check which messageIds already exist (dedupe by messageId).
   365→  const existingQ = query(
   366→    collection(db, COMMUNICATION_COLLECTIONS.GMAIL_MESSAGES),
   367→    where('organizationId', '==', organizationId),
   368→    where('messageId', 'in', messages.slice(0, 30).map((m) => m.messageId)),
   369→  );
   370→  const existingSnap = await getDocs(existingQ);
   371→  const existingIds = new Set(existingSnap.docs.map((d) => (d.data() as { messageId: string }).messageId));
   372→
   373→  const batch = writeBatch(db);
   374→  let count = 0;
   375→  for (const msg of messages) {
   376→    if (existingIds.has(msg.messageId)) continue;
   377→    const ref = doc(collection(db, COMMUNICATION_COLLECTIONS.GMAIL_MESSAGES));
   378→    batch.set(ref, {
   379→      ...msg,
   380→      createdAt: serverTimestamp(),
   381→      updatedAt: serverTimestamp(),
   382→    });
   383→    count++;
   384→  }
   385→  if (count > 0) await batch.commit();
   386→  return count;
   387→}

   492→export async function saveWhatsAppMessages(
   493→  organizationId: string,
   494→  messages: WhatsAppMessage[],
   495→): Promise<number> {
   496→  assertOrg(organizationId);
   497→  if (messages.length === 0) return 0;
   498→
   499→  // Dedupe by wamId (or body+recipient+timestamp for messages without wamId).
   500→  const wamIds = messages.filter((m) => m.wamId).slice(0, 30).map((m) => m.wamId as string);
   501→  const existingIds = new Set<string>();
   502→  if (wamIds.length > 0) {
   503→    const existingQ = query(
   504→      collection(db, COMMUNICATION_COLLECTIONS.WHATSAPP_MESSAGES),
   505→      where('organizationId', '==', organizationId),
   506→      where('wamId', 'in', wamIds),
   507→    );
   508→    const existingSnap = await getDocs(existingQ);
   509→    existingSnap.docs.forEach((d) => existingIds.add((d.data() as { wamId: string }).wamId));
   510→  }
   511→
   512→  const batch = writeBatch(db);
   513→  let count = 0;
   514→  for (const msg of messages) {
   515→    if (msg.wamId && existingIds.has(msg.wamId)) continue;
   516→    const ref = doc(collection(db, COMMUNICATION_COLLECTIONS.WHATSAPP_MESSAGES));
   517→    batch.set(ref, {
   518→      ...msg,
   519→      createdAt: serverTimestamp(),
   520→      updatedAt: serverTimestamp(),
   521→    });
   522→    count++;
   523→  }
   524→  if (count > 0) await batch.commit();
   525→  return count;
   526→}

   587→  organizationId: string,
   588→  callback: (messages: ScheduledMessage[]) => void,
   589→  options?: { limitCount?: number; status?: ScheduledMessageStatus; onError?: (err: Error) => void },
   590→): Unsubscribe {

   592→  const constraints: QueryConstraint[] = [
   593→    where('organizationId', '==', organizationId),
   594→    orderBy('scheduledFor', 'desc'),
   595→  ];
   596→  if (options?.status) {
   597→    constraints.unshift(where('status', '==', options.status));
   598→  }

   600→    constraints.push(limitFn(options.limitCount));
   601→  }
   602→  const q = query(collection(db, COMMUNICATION_COLLECTIONS.SCHEDULED_MESSAGES), ...constraints);
   603→  return onSnapshot(
   604→    q,
   605→    (snap) => callback(snap.docs.map((d) => toScheduledMessage(d.id, d.data() as Record<string, unknown>))),
   606→    (err) => options?.onError?.(err),
   607→  );
   608→}

   739→// ─── Summary Computation ─────────────────────────────────────────────────────
   740→
   741→/**
   742→ * Compute the dashboard communication summary from real-time Firestore data.
   743→ * Pure function — given the current connections + messages + schedules,
   744→ * return the summary the dashboard displays.
   745→ */
   746→export function computeCommunicationSummary(
   747→  gmailConnections: GmailConnection[],
   748→  whatsappConnections: WhatsAppConnection[],
   749→  gmailMessages: GmailMessage[],
   750→  whatsappMessages: WhatsAppMessage[],
   751→  scheduledMessages: ScheduledMessage[],
   752→): CommunicationSummary {
   753→  const gmailConn = gmailConnections.find((c) => c.authStatus === 'connected') ?? null;
   754→  const waConn = whatsappConnections.find((c) => c.authStatus === 'connected') ?? null;
   755→
   756→  const unreadEmails = gmailMessages.filter((m) => m.readStatus === 'unread' && m.direction === 'inbound').length;
   757→  const unreadGstNotices = gmailMessages.filter(
   758→    (m) => m.category === 'gst_notice' && m.readStatus === 'unread',
   759→  ).length;
   760→  const pendingVendorInvoices = gmailMessages.filter(
   761→    (m) => m.category === 'vendor_invoice' && m.readStatus === 'unread',
   762→  ).length;
   763→
   764→  // Pending client replies — outbound emails in the last 7 days with no inbound reply.
   765→  const sevenDaysAgo = Date.now() - 7 * 24 * 60 * 60 * 1000;
   766→  const pendingClientReplies = gmailMessages.filter((m) => {
   767→    if (m.direction !== 'outbound') return false;
   768→    return new Date(m.receivedAt).getTime() >= sevenDaysAgo;
   769→  }).length;
   770→
   771→  const unreadWhatsAppMessages = whatsappMessages.filter(
   772→    (m) => m.direction === 'inbound' && m.status !== 'read',
   773→  ).length;
   774→
   775→  // Pending WhatsApp replies — group by contact, check if the most recent
   776→  // message in each conversation is inbound (unanswered).
   777→  const byContact = new Map<string, WhatsAppMessage[]>();
   778→  for (const m of whatsappMessages) {
   779→    const contactKey = m.direction === 'inbound' ? m.from : m.to;
   780→    const list = byContact.get(contactKey) ?? [];
   781→    list.push(m);
   782→    byContact.set(contactKey, list);
   783→  }
   784→  let pendingWhatsAppReplies = 0;
   785→  for (const convo of byContact.values()) {
   786→    if (hasPendingReply(convo)) pendingWhatsAppReplies++;
   787→  }
   788→
   789→  // Pending reminders — scheduled messages with status='pending' or 'processing'
   790→  // that are due.
   791→  const now = Date.now();
   792→  const pendingReminders = scheduledMessages.filter(
   793→    (s) => (s.status === 'pending' || s.status === 'processing') && new Date(s.scheduledFor).getTime() <= now,
   794→  ).length;
   795→
   796→  const failedMessagesCount = scheduledMessages.filter((s) => s.status === 'failed').length;
   797→  const sentTodayCount = scheduledMessages.filter((s) => {
   798→    if (s.status !== 'sent' || !s.lastSentAt) return false;
   799→    const sentAt = new Date(s.lastSentAt).getTime();
   800→    const startOfDay = new Date();
   801→    startOfDay.setHours(0, 0, 0, 0);
   802→    return sentAt >= startOfDay.getTime();
   803→  }).length;
   804→
   805→  return {
   806→    gmailConnected: !!gmailConn,
   807→    gmailConnectionEmail: gmailConn?.email ?? null,
   808→    gmailLastSync: gmailConn?.lastSync ?? null,
   809→    unreadEmails,
   810→    unreadGstNotices,
   811→    pendingVendorInvoices,
   812→    pendingClientReplies,
   813→    whatsappConnected: !!waConn,
   814→    whatsappPhoneNumber: waConn?.phoneNumber ?? null,
   815→    whatsappLastSync: waConn?.lastSync ?? null,
   816→    unreadWhatsAppMessages,
   817→    pendingWhatsAppReplies,
   818→    pendingReminders,
   819→    scheduledMessagesCount: scheduledMessages.length,
   820→    failedMessagesCount,
   821→    sentTodayCount,
   822→    recentEmails: gmailMessages.slice(0, 10),
   823→    recentWhatsAppMessages: whatsappMessages.slice(0, 10),
   824→    recentScheduledMessages: scheduledMessages.slice(0, 10),
   825→  };
   826→}

   837→/**
   838→ * Compute the next-run timestamp for a recurring schedule.
   839→ */
   840→export function nextRunForRecurrence(
   841→  recurrence: ScheduledMessage['recurrence'],
   842→  from: Date = new Date(),
   843→): string {
   844→  const d = new Date(from);
   845→  switch (recurrence) {
   846→    case 'daily':
   847→      d.setDate(d.getDate() + 1);
   848→      break;
   849→    case 'weekly':
   850→      d.setDate(d.getDate() + 7);
   851→      break;
   852→    case 'monthly':
   853→      d.setMonth(d.getMonth() + 1);
   854→      break;
   855→    case 'quarterly':
   856→      d.setMonth(d.getMonth() + 3);
   857→      break;
   858→    case 'once':
   859→    default:
   860→      // No next run.
   861→      return new Date(0).toISOString();
   862→  }
   863→  return d.toISOString();
   864→}
