   165→   165→  if (bytes < 1024) return `${bytes} B`;

   227→   227→// ─── Map API Invoice to local ExtractedInvoice ────────────────────────────────
   228→   228→
   229→   229→function mapInvoiceStatus(status: string): ValidationStatus {
   230→   230→  if (status === 'approved' || status === 'filed') return 'validated';
   231→   231→  if (status === 'draft') return 'warning';
   232→   232→  return 'error';
   233→   233→}
   234→   234→
   235→   235→function mapInvoiceToExtracted(inv: Invoice, clientName: string): ExtractedInvoice {
   236→   236→  const vStatus = mapInvoiceStatus(inv.status);
   237→   237→  return {
   238→   238→    id: inv.id,
   239→   239→    invoiceNumber: inv.invoiceNumber,
   240→   240→    clientName,
   241→   241→    invoiceDate: inv.invoiceDate,
   242→   242→    section: inv.gstr1Section,
   243→   243→    sectionLabel: GSTR1_SECTION_LABELS[inv.gstr1Section] ?? inv.gstr1Section.toUpperCase(),
   244→   244→    taxableValue: inv.taxableValue,
   245→   245→    cgst: inv.cgst,
   246→   246→    sgst: inv.sgst,
   247→   247→    igst: inv.igst,
   248→   248→    totalTax: inv.cgst + inv.sgst + inv.igst,
   249→   249→    totalAmount: inv.totalAmount,
   250→   250→    validationStatus: vStatus,
   251→   251→    ocrConfidence: vStatus === 'error' ? 75.0 : vStatus === 'warning' ? 92.0 : 98.5,
   252→   252→    missingFields: inv.aiExplanation ? [inv.aiExplanation.split(' ').slice(0, 2).join(' ')] : [],
   253→   253→    validationMessages: inv.aiExplanation ? [inv.aiExplanation] : [],
   254→   254→  };
   255→   255→}
