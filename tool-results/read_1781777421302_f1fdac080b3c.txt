   201→/**
   202→ * Detect the most appropriate role for the message.
   203→ * Falls back to 'ca' (most GSTPilot questions are GST/compliance) when no signal.
   204→ */
   205→export function detectRole(message: string): OracleRole {
   206→  const lower = message.toLowerCase()
   207→  const score = (keywords: string[]): number =>
   208→    keywords.reduce((acc, kw) => (lower.includes(kw) ? acc + 1 : acc), 0)
   209→
   210→  const cfo = score(CFO_KEYWORDS)
   211→  const ca = score(CA_KEYWORDS)
   212→  const ops = score(OPS_KEYWORDS)
   213→
   214→  if (cfo >= ca && cfo >= ops && cfo > 0) return 'cfo'
   215→  if (ops >= ca && ops > 0) return 'ops'
   216→  if (ca > 0) return 'ca'
   217→  // Default — GSTPilot's primary domain is GST/CA work
   218→  return 'ca'
   219→}

   221→/**
   222→ * Detect the user's intent (which of the 6 functions).
   223→ * Falls back to 'answer' (general Q&A).
   224→ */
   225→export function detectFunction(message: string): OracleFunction {
   226→  const lower = message.toLowerCase()
   227→  let best: { fn: OracleFunction; score: number } = { fn: 'answer', score: 0 }
   228→
   229→  for (const [fn, keywords] of Object.entries(FN_KEYWORDS) as Array<
   230→    [Exclude<OracleFunction, 'answer'>, string[]]
   231→  >) {
   232→    const score = keywords.reduce((acc, kw) => (lower.includes(kw) ? acc + 1 : acc), 0)
   233→    if (score > best.score) best = { fn, score }
   234→  }
   235→
   236→  return best.fn
   237→}

   250→const MAX_ROWS = 50

   252→/**
   253→ * Pull a focused snapshot of business data from the DB.
   254→ * All queries are bounded (MAX_ROWS) and include the client relation for joins.
   255→ * Returns empty arrays (never null) when tables are empty — the prompt builder
   256→ * handles the empty state explicitly.
   257→ */
   258→export async function gatherContext(fn: OracleFunction): Promise<OracleDataSnapshot> {
   259→  // For pure Q&A we still pull a minimal context (clients + counts) so the
   260→  // Oracle can reference the user's actual business even on general questions.
   261→  const needInvoices = fn === 'analyze_invoices' || fn === 'detect_overdue' || fn === 'report' || fn === 'cashflow' || fn === 'actions'
   262→  const needPayments = fn === 'detect_overdue' || fn === 'report' || fn === 'cashflow' || fn === 'actions'
   263→  const needReturns = fn === 'report' || fn === 'actions' || fn === 'answer'
   264→  const needNotices = fn === 'report' || fn === 'actions' || fn === 'answer'
   265→  const needRecon = fn === 'report' || fn === 'actions'
   266→
   267→  const [clients, invoices, payments, returns, notices, reconciliations] =
   268→    await Promise.all([
   269→      db.client.findMany({
   270→        take: MAX_ROWS,
   271→        orderBy: { createdAt: 'desc' },
   272→        select: {
   273→          id: true, gstin: true, tradeName: true, status: true,
   274→          entityType: true, returnPeriod: true, healthScore: true,
   275→        },
   276→      }),
   277→      needInvoices
   278→        ? db.invoice.findMany({
   279→            take: MAX_ROWS,
   280→            orderBy: { createdAt: 'desc' },
   281→            include: { client: { select: { tradeName: true, gstin: true } } },
   282→          })
   283→        : Promise.resolve([]),
   284→      // NOTE: Payment has no relation to Client (clientId is a plain String FK).
   285→      // We hydrate client names in a second pass below.
   286→      needPayments
   287→        ? db.payment.findMany({
   288→            take: MAX_ROWS,
   289→            orderBy: { paidAt: 'desc' },
   290→          })
   291→        : Promise.resolve([]),
   292→      needReturns
   293→        ? db.gSTRFiling.findMany({
   294→            take: MAX_ROWS,
   295→            orderBy: { createdAt: 'desc' },
   296→            include: { client: { select: { tradeName: true, gstin: true } } },
   297→          })
   298→        : Promise.resolve([]),
   299→      needNotices
   300→        ? db.notice.findMany({
   301→            take: MAX_ROWS,
   302→            orderBy: { createdAt: 'desc' },
   303→            include: { client: { select: { tradeName: true, gstin: true } } },
   304→          })
   305→        : Promise.resolve([]),
   306→      needRecon
   307→        ? db.reconciliationRun.findMany({
   308→            take: MAX_ROWS,
   309→            orderBy: { createdAt: 'desc' },
   310→          })
   311→        : Promise.resolve([]),
   312→    ])
   313→
   314→  // Hydrate client names for payments (Payment has no Client relation).
   315→  const paymentsWithClients = await Promise.all(
   316→    payments.map(async (p) => {
   317→      const client = p.clientId
   318→        ? await db.client.findUnique({
   319→            where: { id: p.clientId },
   320→            select: { tradeName: true, gstin: true },
   321→          })
   322→        : null
   323→      return { ...p, client }
   324→    })
   325→  )
   326→
   327→  // Hydrate client names for reconciliation runs (clientId is a plain string FK).
   328→  const reconWithClients = await Promise.all(
   329→    reconciliations.map(async (r) => {
   330→      const client = r.clientId
   331→        ? await db.client.findUnique({
   332→            where: { id: r.clientId },
   333→            select: { tradeName: true, gstin: true },
   334→          })
   335→        : null
   336→      return { ...r, client }
   337→    })
   338→  )
   339→
   340→  const snapshot: OracleDataSnapshot = {
   341→    clients: clients.map((c) => ({
   342→      id: c.id, gstin: c.gstin, tradeName: c.tradeName,
   343→      status: c.status, entityType: c.entityType,
   344→      returnPeriod: c.returnPeriod, healthScore: c.healthScore,
   345→    })),
   346→    invoices: invoices.map((i) => ({
   347→      id: i.id, invoiceNumber: i.invoiceNumber, invoiceDate: i.invoiceDate,
   348→      sellerGstin: i.sellerGstin, buyerName: i.buyerName,
   349→      invoiceType: i.invoiceType, taxableValue: i.taxableValue,
   350→      cgst: i.cgst, sgst: i.sgst, igst: i.igst, cess: i.cess,
   351→      totalAmount: i.totalAmount, status: i.status,
   352→      matchStatus: i.matchStatus, riskLevel: i.riskLevel,
   353→      hsnCode: i.hsnCode, source: i.source,
   354→      client: i.client ? { tradeName: i.client.tradeName, gstin: i.client.gstin } : null,
   355→    })),
   356→    payments: paymentsWithClients.map((p) => ({
   357→      id: p.id, direction: p.direction as 'in' | 'out',
   358→      amount: p.amount, method: p.method, status: p.status,
   359→      source: p.source, referenceNo: p.referenceNo, description: p.description,
   360→      paidAt: p.paidAt ? p.paidAt.toISOString() : null,
   361→      postedAt: p.postedAt.toISOString(),
   362→      reconciliationStatus: p.reconciliationStatus,
   363→      client: p.client ? { tradeName: p.client.tradeName, gstin: p.client.gstin } : null,
   364→    })),
   365→    returns: returns.map((r) => ({
   366→      id: r.id, returnType: r.returnType, period: r.period,
   367→      financialYear: r.financialYear, status: r.status,
   368→      filedDate: r.filedDate, acknowledgmentNumber: r.acknowledgmentNumber,
   369→      totalInvoices: r.totalInvoices, totalTaxableValue: r.totalTaxableValue,
   370→      totalTax: r.totalTax, source: r.source,
   371→      client: r.client ? { tradeName: r.client.tradeName, gstin: r.client.gstin } : null,
   372→    })),
   373→    notices: notices.map((n) => ({
   374→      id: n.id, noticeType: n.noticeType, noticeNumber: n.noticeNumber,
   375→      noticeDate: n.noticeDate, subject: n.subject, status: n.status,
   376→      priority: n.priority, dueDate: n.dueDate, source: n.source,
   377→      client: n.client ? { tradeName: n.client.tradeName, gstin: n.client.gstin } : null,
   378→    })),
   379→    reconciliations: reconWithClients.map((r) => ({
   380→      id: r.id, period: r.period, sources: r.sources,
   381→      totalRecords: r.totalRecords, matched: r.matched,
   382→      unmatched: r.unmatched, partialMatches: r.partialMatches,
   383→      highRisk: r.highRisk, gstDifference: r.gstDifference,
   384→      status: r.status, source: r.source,
   385→      client: r.client ? { tradeName: r.client.tradeName, gstin: r.client.gstin } : null,
   386→    })),
   387→  }
   388→
   389→  return snapshot
   390→}

   392→/**
   393→ * Compute whether ANY real business data exists in the snapshot.
   394→ * Used to choose between data-grounded response vs honest-empty-state response.
   395→ */
   396→export function hasAnyData(snap: OracleDataSnapshot): boolean {
   397→  return (
   398→    snap.invoices.length > 0 ||
   399→    snap.payments.length > 0 ||
   400→    snap.returns.length > 0 ||
   401→    snap.notices.length > 0 ||
   402→    snap.reconciliations.length > 0
   403→  )
   404→  // Note: clients alone don't count as "business data" — a client with no
   405→  // invoices/payments/returns is just a registry entry. The Oracle should
   406→  // still honestly say "no invoices recorded" even if clients exist.
   407→}

   409→// ─── Number / date formatting (Indian style) ──────────────────────────────────
   410→
   411→export function formatINR(n: number): string {
   412→  const rounded = Math.round(n)
   413→  const abs = Math.abs(rounded)
   414→  const sign = rounded < 0 ? '-' : ''
   415→  // Indian comma grouping: last 3 digits, then groups of 2
   416→  const str = abs.toString()
   417→  let lastThree = str.slice(-3)
   418→  const otherNumbers = str.slice(0, -3)
   419→  if (otherNumbers.length > 0) {
   420→    lastThree = ',' + lastThree
   421→  }
   422→  const formatted =
   423→    otherNumbers.replace(/\B(?=(\d{2})+(?!\d))/g, ',') + lastThree
   424→  return `${sign}₹${formatted}`
   425→}
   426→
   427→export function formatDate(iso: string | null): string {
   428→  if (!iso) return '—'
   429→  try {
   430→    const d = new Date(iso)
   431→    if (isNaN(d.getTime())) return iso // already a string like "2024-04-15"
   432→    const dd = String(d.getDate()).padStart(2, '0')
   433→    const mm = String(d.getMonth() + 1).padStart(2, '0')
   434→    const yyyy = d.getFullYear()
   435→    return `${dd}/${mm}/${yyyy}`
   436→  } catch {
   437→    return iso
   438→  }
   439→}
   440→
   441→/** Days between two dates (positive = future, negative = past). */
   442→export function daysBetween(from: Date, to: Date): number {
   443→  const ms = to.getTime() - from.getTime()
   444→  return Math.round(ms / (1000 * 60 * 60 * 24))
   445→}

   449→/**
   450→ * Build a structured text snapshot of the live DB data, formatted for the LLM.
   451→ * This is the ONLY source of numbers the LLM is allowed to cite.
   452→ */
   453→function buildDataContext(snap: OracleDataSnapshot, fn: OracleFunction): string {
   454→  const lines: string[] = []
   455→  lines.push('# LIVE BUSINESS DATA (queried from your database at ' + new Date().toISOString() + ')')
   456→  lines.push('')
   457→  lines.push(`## Clients (${snap.clients.length})`)
   458→  if (snap.clients.length === 0) {
   459→    lines.push('- No clients registered yet.')
   460→  } else {
   461→    for (const c of snap.clients.slice(0, 10)) {
   462→      lines.push(`- ${c.tradeName} — GSTIN ${c.gstin} — ${c.entityType} — ${c.returnPeriod || 'no period'} — status: ${c.status} — health: ${c.healthScore}/100`)
   463→    }
   464→    if (snap.clients.length > 10) lines.push(`- ...and ${snap.clients.length - 10} more`)
   465→  }
   466→  lines.push('')
   467→
   468→  if (fn === 'analyze_invoices' || fn === 'report' || fn === 'detect_overdue') {
   469→    lines.push(`## Invoices (${snap.invoices.length})`)
   470→    if (snap.invoices.length === 0) {
   471→      lines.push('- No invoices recorded yet.')
   472→    } else {
   473→      const total = snap.invoices.reduce((s, i) => s + i.totalAmount, 0)
   474→      const taxable = snap.invoices.reduce((s, i) => s + i.taxableValue, 0)
   475→      const totalTax = snap.invoices.reduce((s, i) => s + i.cgst + i.sgst + i.igst + i.cess, 0)
   476→      lines.push(`- Aggregate: ${snap.invoices.length} invoices — taxable value ${formatINR(taxable)} — total tax ${formatINR(totalTax)} — grand total ${formatINR(total)}`)
   477→      const byStatus: Record<string, number> = {}
   478→      for (const i of snap.invoices) byStatus[i.status] = (byStatus[i.status] || 0) + 1
   479→      lines.push(`- By status: ${Object.entries(byStatus).map(([k, v]) => `${k}=${v}`).join(', ')}`)
   480→      const byRisk: Record<string, number> = {}
   481→      for (const i of snap.invoices) byRisk[i.riskLevel] = (byRisk[i.riskLevel] || 0) + 1
   482→      lines.push(`- By risk: ${Object.entries(byRisk).map(([k, v]) => `${k}=${v}`).join(', ')}`)
   483→      lines.push('- Sample (most recent 10):')
   484→      for (const i of snap.invoices.slice(0, 10)) {
   485→        lines.push(`  • ${i.invoiceNumber} dated ${i.invoiceDate} — ${i.invoiceType} — ${i.buyerName || 'no buyer'} — taxable ${formatINR(i.taxableValue)} — tax ${formatINR(i.cgst + i.sgst + i.igst + i.cess)} — total ${formatINR(i.totalAmount)} — status ${i.status} — risk ${i.riskLevel} — source ${i.source}`)
   486→      }
   487→    }
   488→    lines.push('')
   489→  }
   490→
   491→  if (fn === 'detect_overdue' || fn === 'report' || fn === 'cashflow' || fn === 'actions') {
   492→    lines.push(`## Payments (${snap.payments.length})`)
   493→    if (snap.payments.length === 0) {
   494→      lines.push('- No payments recorded yet.')
   495→    } else {
   496→      const inflow = snap.payments.filter((p) => p.direction === 'in').reduce((s, p) => s + p.amount, 0)
   497→      const outflow = snap.payments.filter((p) => p.direction === 'out').reduce((s, p) => s + p.amount, 0)
   498→      lines.push(`- Aggregate: ${snap.payments.length} payments — inflow ${formatINR(inflow)} — outflow ${formatINR(outflow)} — net ${formatINR(inflow - outflow)}`)
   499→      const byStatus: Record<string, number> = {}
   500→      for (const p of snap.payments) byStatus[p.status] = (byStatus[p.status] || 0) + 1
   501→      lines.push(`- By status: ${Object.entries(byStatus).map(([k, v]) => `${k}=${v}`).join(', ')}`)
   502→      lines.push('- Sample (most recent 10):')
   503→      for (const p of snap.payments.slice(0, 10)) {
   504→        lines.push(`  • ${p.direction === 'in' ? 'RECEIVED' : 'PAID'} ${formatINR(p.amount)} via ${p.method || 'unknown'} on ${formatDate(p.paidAt)} — status ${p.status} — source ${p.source} — recon ${p.reconciliationStatus} — client ${p.client?.tradeName || 'n/a'}`)
   505→      }
   506→    }
   507→    lines.push('')
   508→  }
   509→
   510→  if (fn === 'report' || fn === 'actions' || fn === 'answer') {
   511→    lines.push(`## Returns (${snap.returns.length})`)
   512→    if (snap.returns.length === 0) {
   513→      lines.push('- No GSTR returns recorded yet.')
   514→    } else {
   515→      lines.push('- Recent returns:')
   516→      for (const r of snap.returns.slice(0, 10)) {
   517→        lines.push(`  • ${r.returnType} ${r.period} — status ${r.status} — filed ${r.filedDate || 'not filed'} — ARN ${r.acknowledgmentNumber || 'none'} — ${r.totalInvoices} invoices — taxable ${formatINR(r.totalTaxableValue)} — tax ${formatINR(r.totalTax)} — client ${r.client?.tradeName || 'n/a'}`)
   518→      }
   519→    }
   520→    lines.push('')
   521→  }
   522→
   523→  if (fn === 'report' || fn === 'actions' || fn === 'answer') {
   524→    lines.push(`## Notices (${snap.notices.length})`)
   525→    if (snap.notices.length === 0) {
   526→      lines.push('- No notices recorded yet.')
   527→    } else {
   528→      lines.push('- Recent notices:')
   529→      for (const n of snap.notices.slice(0, 10)) {
   530→        lines.push(`  • ${n.noticeType} ${n.noticeNumber || ''} — ${n.subject} — priority ${n.priority} — status ${n.status} — due ${n.dueDate || 'none'} — client ${n.client?.tradeName || 'n/a'}`)
   531→      }
   532→    }
   533→    lines.push('')
   534→  }
   535→
   536→  if (fn === 'report' || fn === 'actions') {
   537→    lines.push(`## Reconciliation Runs (${snap.reconciliations.length})`)
   538→    if (snap.reconciliations.length === 0) {
   539→      lines.push('- No reconciliation runs recorded yet.')
   540→    } else {
   541→      lines.push('- Recent runs:')
   542→      for (const r of snap.reconciliations.slice(0, 10)) {
   543→        lines.push(`  • Period ${r.period} — sources ${r.sources} — ${r.totalRecords} records — matched ${r.matched} — unmatched ${r.unmatched} — partial ${r.partialMatches} — high-risk ${r.highRisk} — GST diff ${formatINR(r.gstDifference)} — client ${r.client?.tradeName || 'n/a'}`)
   544→      }
   545→    }
   546→    lines.push('')
   547→  }
   548→
   549→  return lines.join('\n')
   550→}
