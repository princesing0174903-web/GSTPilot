   110→// ─── Helpers ─────────────────────────────────────────────────────────────────
   111→
   112→function inr(n: number): string {
   113→  return new Intl.NumberFormat('en-IN', {
   114→    style: 'currency',
   115→    currency: 'INR',
   116→    maximumFractionDigits: 2,
   117→  }).format(n || 0);
   118→}
   119→
   120→function fmtDate(iso: string | null): string {
   121→  if (!iso) return '—';
   122→  const d = new Date(iso);
   123→  if (Number.isNaN(d.getTime())) return iso;
   124→  return d.toLocaleDateString('en-IN', { day: '2-digit', month: 'short', year: 'numeric' });
   125→}

   128→  draft: { label: 'Draft', cls: 'border-white/20 bg-white/5 text-white/70' },
   129→  sent: { label: 'Sent', cls: 'border-teal-500/30 bg-teal-500/5 text-teal-300' },
   130→  paid: { label: 'Paid', cls: 'border-emerald-500/30 bg-emerald-500/5 text-emerald-300' },
   131→  partial: { label: 'Partial', cls: 'border-amber-500/30 bg-amber-500/5 text-amber-300' },
   132→  overdue: { label: 'Overdue', cls: 'border-rose-500/30 bg-rose-500/5 text-rose-300' },
   133→  cancelled: { label: 'Cancelled', cls: 'border-white/10 bg-white/[0.02] text-white/40 line-through' },

   136→const UNITS: ProductUnit[] = ['NOS', 'KG', 'GM', 'LTR', 'ML', 'MTR', 'BOX', 'PCS', 'SET', 'HR', 'DAY', 'MONTH'];
   137→
   138→function uid(): string {
   139→  return Math.random().toString(36).slice(2, 10) + Date.now().toString(36).slice(-4);
   140→}
   141→
   142→function todayISO(): string {
   143→  return new Date().toISOString().slice(0, 10);
   144→}
   145→
   146→function addDays(iso: string, days: number): string {
   147→  const d = new Date(iso);
   148→  d.setDate(d.getDate() + days);
   149→  return d.toISOString().slice(0, 10);
   150→}

   152→// ─── Line item form type ─────────────────────────────────────────────────────
   153→
   154→interface LineForm {
   155→  id: string;
   156→  productId: string | null;
   157→  description: string;
   158→  hsnSac: string;
   159→  quantity: string;
   160→  unit: ProductUnit;
   161→  unitPrice: string;
   162→  discount: string;
   163→  gstRate: GstRate;
   164→}
   165→
   166→function emptyLine(): LineForm {
   167→  return {
   168→    id: uid(),
   169→    productId: null,
   170→    description: '',
   171→    hsnSac: '',
   172→    quantity: '1',
   173→    unit: 'NOS',
   174→    unitPrice: '',
   175→    discount: '0',
   176→    gstRate: DEFAULT_GST_RATE,
   177→  };
   178→}

   207→
   208→export default function InvoicesView() {
   209→  const {
   210→    filtered,
   211→    loading,
   212→    error,
   213→    saving,
   214→    stats,
   215→    search,
   216→    setSearch,
   217→    create,
   218→    update,
   219→    remove,
   220→    markPaid,
   221→    cancel,
   222→    retry,
   223→  } = useGSTpilotInvoices();
   224→
   225→  const { customers } = useGSTpilotCustomers();
   226→  const { products } = useGSTpilotProducts();
   227→
   228→  const [dialogOpen, setDialogOpen] = useState(false);
   229→  const [editing, setEditing] = useState<Invoice | null>(null);
   230→  const [form, setForm] = useState<InvoiceFormState>(EMPTY_FORM);
   231→  const [formError, setFormError] = useState<string | null>(null);
   232→  const [deleteTarget, setDeleteTarget] = useState<Invoice | null>(null);
   233→  const [viewTarget, setViewTarget] = useState<Invoice | null>(null);
   234→
   235→  // Form is reset directly in the open handlers — no effect needed.
   236→  const openCreate = () => {
   237→    setEditing(null);
   238→    setForm(EMPTY_FORM);
   239→    setFormError(null);
   240→    setDialogOpen(true);
   241→  };
   242→
   243→  const openEdit = (inv: Invoice) => {
   244→    setEditing(inv);
   245→    setForm(invoiceToForm(inv));
   246→    setFormError(null);
   247→    setDialogOpen(true);
   248→  };
   249→
   250→  // ── Live GST preview ──
   251→  const preview = useMemo(() => {
   252→    const items = form.lines.map((l) => ({
   253→      id: l.id,
   254→      productId: l.productId,
   255→      description: l.description,
   256→      hsnSac: l.hsnSac,
   257→      quantity: Number(l.quantity) || 0,
   258→      unit: l.unit,
   259→      unitPrice: Number(l.unitPrice) || 0,
   260→      discount: Number(l.discount) || 0,
   261→      gstRate: l.gstRate,
   262→    }));
   263→    const customer = customers.find((c) => c.id === form.customerId);
   264→    const customerStateCode = customer?.stateCode ?? null;
   265→    return calculateInvoiceTotals({
   266→      items,
   267→      sellerStateCode: form.sellerStateCode || null,
   268→      customerStateCode,
   269→      paidAmount: 0,
   270→    });
   271→  }, [form, customers]);
   272→
   273→  // ── Line item operations ──
   274→  const updateLine = (id: string, patch: Partial<LineForm>) => {
   275→    setForm((f) => ({
   276→      ...f,
   277→      lines: f.lines.map((l) => (l.id === id ? { ...l, ...patch } : l)),
   278→    }));
   279→  };
   280→
   281→  const pickProduct = (id: string, productId: string) => {
   282→    if (!productId || productId === '__none__') {
   283→      updateLine(id, { productId: null });
   284→      return;
   285→    }
   286→    const p = products.find((x) => x.id === productId);
   287→    if (!p) return;
   288→    updateLine(id, {
   289→      productId: p.id,
   290→      description: p.name,
   291→      hsnSac: p.hsnSac,
   292→      unitPrice: String(p.price),
   293→      gstRate: p.gstRate,
   294→      unit: p.unit,
   295→    });
   296→  };
   297→
   298→  const addLine = () => {
   299→    setForm((f) => ({ ...f, lines: [...f.lines, emptyLine()] }));
   300→  };
   301→
   302→  const removeLine = (id: string) => {
   303→    setForm((f) => ({
   304→      ...f,
   305→      lines: f.lines.length > 1 ? f.lines.filter((l) => l.id !== id) : f.lines,
   306→    }));
   307→  };
   308→
   309→  // ── Customer pick → autofill seller fields stay; customer fields fill from customer ──
   310→  const pickCustomer = (customerId: string) => {
   311→    setForm((f) => ({ ...f, customerId: customerId === '__none__' ? '' : customerId }));
   312→  };
   313→
   314→  // ── Submit ──
   315→  const handleSubmit = async () => {
   316→    setFormError(null);
   317→    if (!form.sellerName.trim()) {
   318→      setFormError('Seller name is required.');
   319→      return;
   320→    }
   321→    const validLines = form.lines.filter(
   322→      (l) => l.description.trim() && Number(l.quantity) > 0,
   323→    );
   324→    if (validLines.length === 0) {
   325→      setFormError('Add at least one line item with a description and quantity.');
   326→      return;
   327→    }
   328→
   329→    const customer = customers.find((c) => c.id === form.customerId);
   330→    const input: CreateInvoiceInput = {
   331→      customerId: form.customerId || null,
   332→      customerName: customer?.name,
   333→      customerGstin: customer?.gstin ?? null,
   334→      customerAddress: customer?.address ?? null,
   335→      customerState: customer?.state ?? null,
   336→      customerStateCode: customer?.stateCode ?? null,
   337→      sellerName: form.sellerName,
   338→      sellerGstin: form.sellerGstin || null,
   339→      sellerAddress: form.sellerAddress || null,
   340→      sellerStateCode: form.sellerStateCode || null,
   341→      invoiceDate: form.invoiceDate,
   342→      dueDate: form.dueDate || null,
   343→      items: validLines.map((l) => ({
   344→        id: l.id,
   345→        productId: l.productId,
   346→        description: l.description,
   347→        hsnSac: l.hsnSac,
   348→        quantity: Number(l.quantity) || 0,
   349→        unit: l.unit,
   350→        unitPrice: Number(l.unitPrice) || 0,
   351→        discount: Number(l.discount) || 0,
   352→        gstRate: l.gstRate,
   353→      })),
   354→      notes: form.notes || null,
   355→    };
   356→
   357→    if (editing) {
   358→      const updated = await update(editing.id, input);
   359→      if (updated) {
   360→        toast.success(`Updated invoice ${updated.invoiceNumber}.`);
   361→        setDialogOpen(false);
   362→      } else {
   363→        setFormError(error || 'Could not update invoice.');
   364→      }
   365→    } else {
   366→      const created = await create(input);
   367→      if (created) {
   368→        toast.success(`Created invoice ${created.invoiceNumber}.`);
   369→        setDialogOpen(false);
   370→      } else {
   371→        setFormError(error || 'Could not create invoice.');
   372→      }
   373→    }
   374→  };
   375→
   376→  const handleDelete = async () => {
   377→    if (!deleteTarget) return;
   378→    const ok = await remove(deleteTarget.id);
   379→    if (ok) {
   380→      toast.success(`Deleted invoice ${deleteTarget.invoiceNumber}.`);
   381→      setDeleteTarget(null);
   382→    }
   383→  };
   384→
   385→  const handleMarkPaid = async (inv: Invoice) => {
   386→    const updated = await markPaid(inv.id);
   387→    if (updated) toast.success(`Marked ${updated.invoiceNumber} as paid.`);
   388→  };
   389→
   390→  const handleCancel = async (inv: Invoice) => {
   391→    const updated = await cancel(inv.id);
   392→    if (updated) toast.success(`Cancelled invoice ${updated.invoiceNumber}.`);
   393→  };
   394→
   395→  // ── Render ─────────────────────────────────────────────────────────────────
   396→
   397→  return (
   398→    <div className="flex min-h-[calc(100vh-4rem)] flex-col gap-6 p-4 md:p-6">
   399→      {/* Header */}
   400→      <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   401→        <div>
   402→          <h1 className="text-2xl font-bold tracking-tight text-white flex items-center gap-2">
   403→            <FileText className="h-6 w-6 text-emerald-400" />
   404→            Invoices
   405→          </h1>
   406→          <p className="mt-1 text-sm text-white/50">
   407→            Live GST invoices — synced with Firestore in real-time. Totals auto-calculated.
   408→          </p>
   409→        </div>
   410→        <Button
   411→          onClick={openCreate}
   412→          className="bg-emerald-500 text-black hover:bg-emerald-400"
   413→        >
   414→          <Plus className="mr-2 h-4 w-4" />
   415→          Create Invoice
   416→        </Button>
   417→      </div>
   418→
   419→      {/* Stats */}
   420→      <div className="grid grid-cols-2 gap-4 lg:grid-cols-4">
   421→        <StatCard icon={<Receipt className="h-5 w-5 text-emerald-400" />} label="Total Invoiced" value={inr(stats.totalInvoiced)} tint="emerald" />
   422→        <StatCard icon={<CheckCircle2 className="h-5 w-5 text-teal-400" />} label="Total Paid" value={inr(stats.totalPaid)} tint="teal" />
   423→        <StatCard icon={<Wallet className="h-5 w-5 text-amber-400" />} label="Outstanding" value={inr(stats.totalOutstanding)} tint="amber" />
   424→        <StatCard icon={<TrendingUp className="h-5 w-5 text-violet-400" />} label="Tax Collected" value={inr(stats.totalTaxCollected)} tint="violet" />
   425→      </div>
   426→
   427→      {/* Search */}
   428→      <div className="flex items-center gap-3">
   429→        <div className="relative flex-1 max-w-md">
   430→          <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-white/40" />
   431→          <Input
   432→            placeholder="Search by invoice #, customer, GSTIN…"
   433→            value={search}
   434→            onChange={(e) => setSearch(e.target.value)}
   435→            className="border-white/10 bg-white/[0.03] pl-9 text-white placeholder:text-white/30"
   436→          />
   437→        </div>
   438→        {error && (
   439→          <Button variant="outline" size="sm" onClick={retry} className="border-white/10 text-white/70">
   440→            <RefreshCw className="mr-2 h-4 w-4" />
   441→            Retry
   442→          </Button>
   443→        )}
   444→      </div>
   445→
   446→      {error && !loading && (
   447→        <div className="flex items-start gap-3 rounded-xl border border-amber-500/20 bg-amber-500/[0.06] p-4">
   448→          <AlertCircle className="mt-0.5 h-5 w-5 shrink-0 text-amber-400" />
   449→          <div className="text-sm text-amber-200/90">
   450→            <p className="font-medium">{error}</p>
   451→            <p className="mt-0.5 text-amber-200/60">Your changes will sync automatically once the connection is restored.</p>
   452→          </div>
   453→        </div>
   454→      )}
   455→
   456→      {/* Table */}
   457→      <Card className="flex-1 border-white/10 bg-white/[0.02]">
   458→        <CardContent className="p-0">
   459→          {loading ? (
   460→            <div className="space-y-2 p-4">
   461→              {Array.from({ length: 6 }).map((_, i) => (
   462→                <Skeleton key={i} className="h-12 w-full bg-white/[0.04]" />
   463→              ))}
   464→            </div>
   465→          ) : filtered.length === 0 ? (
   466→            <EmptyState onCreate={openCreate} hasSearch={!!search} />
   467→          ) : (
   468→            <ScrollArea className="max-h-[60vh]">
   469→              <Table>
   470→                <TableHeader>
   471→                  <TableRow className="border-white/10 hover:bg-transparent">
   472→                    <TableHead className="text-white/50">Invoice #</TableHead>
   473→                    <TableHead className="text-white/50">Customer</TableHead>
   474→                    <TableHead className="text-white/50">Date</TableHead>
   475→                    <TableHead className="text-right text-white/50">Total</TableHead>
   476→                    <TableHead className="text-right text-white/50">Balance</TableHead>
   477→                    <TableHead className="text-white/50">Status</TableHead>
   478→                    <TableHead className="w-12" />
   479→                  </TableRow>
   480→                </TableHeader>
   481→                <TableBody>
   482→                  {filtered.map((inv) => {
   483→                    const st = STATUS_STYLES[inv.status];
   484→                    return (
   485→                      <TableRow key={inv.id} className="border-white/5 hover:bg-white/[0.02]">
   486→                        <TableCell>
   487→                          <span className="font-mono text-sm font-medium text-white">{inv.invoiceNumber}</span>
   488→                        </TableCell>
   489→                        <TableCell>
   490→                          <div className="min-w-0">
   491→                            <p className="truncate font-medium text-white">{inv.customerName}</p>
   492→                            {inv.customerGstin && (
   493→                              <p className="truncate font-mono text-xs text-white/40">{inv.customerGstin}</p>
   494→                            )}
   495→                          </div>
   496→                        </TableCell>
   497→                        <TableCell className="text-sm text-white/60">{fmtDate(inv.invoiceDate)}</TableCell>
   498→                        <TableCell className="text-right font-medium text-white">{inr(inv.grandTotal)}</TableCell>
   499→                        <TableCell className="text-right">
   500→                          {inv.balanceDue > 0 && inv.status !== 'cancelled' ? (
   501→                            <span className="font-medium text-amber-400">{inr(inv.balanceDue)}</span>
   502→                          ) : (
   503→                            <span className="text-white/30">{inr(0)}</span>
   504→                          )}
   505→                        </TableCell>
   506→                        <TableCell>
   507→                          <Badge variant="outline" className={st.cls}>{st.label}</Badge>
   508→                        </TableCell>
   509→                        <TableCell>
   510→                          <DropdownMenu>
   511→                            <DropdownMenuTrigger asChild>
   512→                              <Button variant="ghost" size="icon" className="h-8 w-8 text-white/50 hover:text-white">
   513→                                <MoreHorizontal className="h-4 w-4" />
   514→                              </Button>
   515→                            </DropdownMenuTrigger>
   516→                            <DropdownMenuContent align="end" className="border-white/10 bg-zinc-900">
   517→                              <DropdownMenuLabel className="text-white/40">Actions</DropdownMenuLabel>
   518→                              <DropdownMenuItem onClick={() => setViewTarget(inv)} className="text-white/80 focus:text-white">
   519→                                <Eye className="mr-2 h-4 w-4" /> View
   520→                              </DropdownMenuItem>
   521→                              {inv.status !== 'cancelled' && inv.status !== 'paid' && (
   522→                                <DropdownMenuItem onClick={() => openEdit(inv)} className="text-white/80 focus:text-white">
   523→                                  <Pencil className="mr-2 h-4 w-4" /> Edit
   524→                                </DropdownMenuItem>
   525→                              )}
   526→                              {inv.status !== 'paid' && inv.status !== 'cancelled' && (
   527→                                <DropdownMenuItem onClick={() => handleMarkPaid(inv)} className="text-emerald-300 focus:text-emerald-200">
   528→                                  <CheckCircle2 className="mr-2 h-4 w-4" /> Mark Paid
   529→                                </DropdownMenuItem>
   530→                              )}
   531→                              {inv.status !== 'cancelled' && inv.status !== 'paid' && (
   532→                                <DropdownMenuItem onClick={() => handleCancel(inv)} className="text-amber-300 focus:text-amber-200">
   533→                                  <XCircle className="mr-2 h-4 w-4" /> Cancel
   534→                                </DropdownMenuItem>
   535→                              )}
   536→                              <DropdownMenuSeparator className="bg-white/10" />
   537→                              <DropdownMenuItem
   538→                                onClick={() => setDeleteTarget(inv)}
   539→                                className="text-rose-300 focus:text-rose-200"
   540→                              >
   541→                                <Trash2 className="mr-2 h-4 w-4" /> Delete
   542→                              </DropdownMenuItem>
   543→                            </DropdownMenuContent>
   544→                          </DropdownMenu>
   545→                        </TableCell>
   546→                      </TableRow>
   547→                    );
   548→                  })}
   549→                </TableBody>
