    62→/**
    63→ * Convert a Firestore snapshot ({ id, data() }) into a typed object with the
    64→ * document id prepended, recursively converting Firestore Timestamps to ISO
    65→ * strings. Mirrors the conversion logic in `withId` (firestore-schema.ts) but
    66→ * drops the `T extends Record<string, unknown>` constraint so it accepts
    67→ * strongly-typed interfaces (which do not have an implicit index signature).
    68→ */
    69→function docToData<T>(d: { id: string; data: () => Record<string, unknown> }): T & { id: string } {
    70→  const data = d.data();
    71→  const converted: Record<string, unknown> = {};
    72→  for (const [key, value] of Object.entries(data)) {
    73→    if (value && typeof value === 'object' && 'toDate' in value && typeof (value as { toDate: () => Date }).toDate === 'function') {
    74→      converted[key] = (value as { toDate: () => Date }).toDate().toISOString();
    75→    } else {
    76→      converted[key] = value;
    77→    }
    78→  }
    79→  return { id: d.id, ...converted } as T & { id: string };
    80→}

   415→export async function updateReturnStatus(returnId: string, status: FilingStatus, extras?: Partial<FirestoreReturn>): Promise<void> {
   416→  const returnRef = doc(db, COLLECTIONS.RETURNS, returnId);
   417→  const returnSnap = await getDoc(returnRef);
   418→  if (!returnSnap.exists()) return;
   419→
   420→  const data = returnSnap.data() as FirestoreReturn;
   421→  await updateDoc(returnRef, { status, ...extras, updatedAt: serverTimestamp() });
   422→
   423→  const statusActivityMap: Record<string, ActivityType> = {
   424→    prepared: 'return_prepared',
   425→    validated: 'return_prepared',
   426→    reviewed: 'return_reviewed',
   427→    generated: 'return_prepared',
   428→    filed: 'return_filed',
   429→    reopened: 'return_reopened',
   430→  };
   431→
   432→  const activityType = statusActivityMap[status] || 'return_prepared';
   433→  const statusLabel = status.charAt(0).toUpperCase() + status.slice(1);
   434→
   435→  addActivity({
   436→    type: activityType,
   437→    title: `Return ${statusLabel}`,
   438→    description: `${data.returnType} for ${data.period} moved to ${statusLabel}`,
   439→    clientId: data.clientId,
   440→    entityType: COLLECTIONS.RETURNS,
   441→    entityId: returnId,
   442→  });
   443→
   444→  if (status === 'reviewed' || status === 'validated' || status === 'generated') {
   445→    addNotification({
   446→      type: 'return_reviewed',
   447→      priority: 'normal',
   448→      title: `${data.returnType} ready to file`,
   449→      message: `${data.returnType} for ${data.period} has been ${statusLabel} and is ready for filing.`,
   450→      entityType: COLLECTIONS.RETURNS,
   451→      entityId: returnId,
   452→    });
   453→
   454→    // Update compliance score
   455→    recalculateComplianceScore(data.clientId);
   456→  }
   457→}

   483→  addActivity({

   485→    title: 'Return filed',
   486→    description: `${data.returnType} for ${data.period} filed — ARN: ${arn}`,
   487→    clientId: data.clientId,
   488→    entityType: COLLECTIONS.RETURNS,
   489→    entityId: returnId,
   490→    metadata: { arn, taxAmount: data.totalTax },

   492→
   493→  addNotification({
   494→    type: 'filing_completed',
   495→    priority: 'high',

   502→  recalculateComplianceScore(data.clientId);
   503→
   504→  return arn;
   505→}

   507→/** Auto-create draft returns for current period */
   508→async function autoCreateDraftReturns(firmId: string, clientId: string): Promise<void> {
   509→  const now = new Date();
   510→  const period = `${String(now.getMonth() + 1).padStart(2, '0')}-${now.getFullYear()}`;
   511→  const fy = now.getMonth() >= 3 ? `${now.getFullYear()}-${String(now.getFullYear() + 1).slice(2)}` : `${now.getFullYear() - 1}-${String(now.getFullYear()).slice(2)}`;
   512→
   513→  for (const returnType of ['GSTR-1', 'GSTR-3B'] as const) {
   514→    // Check if return already exists
   515→    const q = query(
   516→      collection(db, COLLECTIONS.RETURNS),
   517→      where('clientId', '==', clientId),
   518→      where('returnType', '==', returnType),
   519→      where('period', '==', period),
   520→      limit(1)
   521→    );
   522→    const existing = await getDocs(q);
   523→    if (existing.empty) {
   524→      await createReturn({
   525→        clientId,
   526→        returnType,
   527→        period,
   528→        financialYear: fy,
   529→        status: 'draft',
   530→        filedDate: null,
   531→        acknowledgmentNumber: null,
   532→        totalInvoices: 0,
   533→        readyForFiling: 0,
   534→        issuesFound: 0,
   535→        criticalErrors: 0,
   536→        warnings: 0,
   537→        totalTaxableValue: 0,
   538→        totalTax: 0,
   539→        jsonPayload: null,
   540→        assignedTo: null,
   541→        reviewedBy: null,
   542→      });
   543→    }
   544→  }
   545→}

   547→/** Update draft return when an invoice is added */
   548→async function updateDraftReturnForInvoice(firmId: string, clientId: string, returnType: string, period: string, taxableValue: number, taxAmount: number): Promise<void> {
   549→  try {
   550→    const q = query(
   551→      collection(db, COLLECTIONS.RETURNS),
   552→      where('clientId', '==', clientId),
   553→      where('returnType', '==', returnType),
   554→      where('period', '==', period),
   555→      limit(1)
   556→    );
   557→    const snap = await getDocs(q);
   558→    if (!snap.empty) {
   559→      const returnDoc = snap.docs[0];
   560→      const data = returnDoc.data() as FirestoreReturn;
   561→      await updateDoc(returnDoc.ref, {
   562→        totalInvoices: (data.totalInvoices || 0) + 1,
   563→        totalTaxableValue: (data.totalTaxableValue || 0) + taxableValue,
   564→        totalTax: (data.totalTax || 0) + taxAmount,
   565→        updatedAt: serverTimestamp(),
   566→      });
   567→    }
   568→  } catch (e) {
   569→    console.warn('[Workflow] Failed to update draft return:', e);
   570→  }
   571→}

   579→export async function createReconciliation(data: {
   580→  clientId: string;
   581→  period: string;
   582→  sources: string;
   583→}): Promise<string> {
   584→  const firmId = currentFirmId();
   585→  const reconId = generateId();
   586→  const reconRef = doc(db, COLLECTIONS.RECONCILIATIONS, reconId);
   587→
   588→  // Simulate reconciliation results
   589→  const totalRecords = Math.floor(Math.random() * 50) + 10;
   590→  const matched = Math.floor(totalRecords * (0.6 + Math.random() * 0.25));
   591→  const partialMatches = Math.floor((totalRecords - matched) * 0.4);
   592→  const unmatched = totalRecords - matched - partialMatches;
   593→  const highRisk = Math.floor(unmatched * 0.3);
   594→
   595→  const mismatches: ReconMismatch[] = [];
   596→  for (let i = 0; i < unmatched + partialMatches; i++) {
   597→    const booksAmount = Math.floor(Math.random() * 100000) + 5000;
   598→    const diff = Math.floor(booksAmount * (Math.random() * 0.15));
   599→    mismatches.push({
   600→      invoiceNumber: `INV-${String(i + 1).padStart(4, '0')}`,
   601→      invoiceDate: new Date().toISOString().split('T')[0],
   602→      sourceGstin: '27AAACR5055K1ZB',
   603→      matchedGstin: i < partialMatches ? '27AAACR5055K1ZB' : null,
   604→      matchStatus: i < partialMatches ? 'partial_match' as MatchStatus : 'missing_in_books' as MatchStatus,
   605→      matchScore: i < partialMatches ? 70 + Math.floor(Math.random() * 20) : 0,
   606→      booksAmount,
   607→      portalAmount: booksAmount + (Math.random() > 0.5 ? diff : -diff),
   608→      difference: diff,
   609→      reason: i < partialMatches ? 'Amount mismatch' : 'Invoice not found in GSTR-2B',
   610→      resolved: false,
   611→      resolvedBy: null,
   612→      resolvedAt: null,
   613→    });
   614→  }
   615→
   616→  const gstDifference = mismatches.reduce((sum, m) => sum + m.difference, 0);
   617→
   618→  const reconData: FirestoreReconciliation = {
   619→    reconId,
   620→    firmId: firmId || '',
   621→    clientId: data.clientId,
   622→    period: data.period,
   623→    sources: data.sources,
   624→    status: 'completed',
   625→    totalRecords,
   626→    matched,
   627→    unmatched,
   628→    partialMatches,
   629→    highRisk,
   630→    gstDifference,
   631→    mismatches,
   632→    runBy: currentUserId(),
   633→    createdAt: serverTimestamp(),
   634→    updatedAt: serverTimestamp(),
   635→  };
   636→
   637→  await setDoc(reconRef, reconData);
   638→
   639→  addActivity({
   640→    type: 'reconciliation_run',
   641→    title: 'Reconciliation completed',
   642→    description: `${data.sources}: ${matched}/${totalRecords} matched, ${unmatched} unmatched`,
   643→    clientId: data.clientId,
   644→    entityType: COLLECTIONS.RECONCILIATIONS,
   645→    entityId: reconId,
   646→    metadata: { matched, unmatched, gstDifference },
   647→  });
   648→
   649→  // Generate AI recommendations for mismatches
   650→  if (highRisk > 0) {
   651→    generateAIRecommendations(firmId || '', data.clientId, reconId, mismatches);
   652→  }
   653→
   654→  addNotification({
   655→    type: 'mismatch_found',
   656→    priority: highRisk > 3 ? 'urgent' : 'high',
   657→    title: 'Reconciliation mismatches found',
   658→    message: `${unmatched} unmatched and ${partialMatches} partial matches found. ITC difference: ₹${gstDifference.toLocaleString('en-IN')}`,
   659→    entityType: COLLECTIONS.RECONCILIATIONS,
   660→    entityId: reconId,
   661→  });
   662→
   663→  return reconId;
   664→}

   666→export async function resolveMismatch(reconId: string, invoiceNumber: string): Promise<void> {
   667→  const reconRef = doc(db, COLLECTIONS.RECONCILIATIONS, reconId);

   673→    m.invoiceNumber === invoiceNumber
   674→      ? { ...m, resolved: true, resolvedBy: currentUserId(), resolvedAt: new Date().toISOString() }

   676→  );

   678→  const resolvedCount = updatedMismatches.filter(m => m.resolved).length;

   903→/** Subscribe to a Firestore collection with firm scoping */
   904→export function subscribeToCollection<T>(
   905→  collectionName: CollectionName,
   906→  callback: (data: Array<T & { id: string }>) => void,
   907→  ...constraints: QueryConstraint[]
   908→): Unsubscribe {
   909→  const firmId = currentFirmId();
   910→
   911→  const q = firmId
   912→    ? query(collection(db, collectionName), where('firmId', '==', firmId), ...constraints)
   913→    : query(collection(db, collectionName), ...constraints);
   914→
   915→  return onSnapshot(q, (snap) => {
   916→    const data = snap.docs.map(d => {
   917→      const docData = d.data();
   918→      const converted: Record<string, unknown> = { id: d.id };
   919→      for (const [key, value] of Object.entries(docData)) {
   920→        if (value && typeof value === 'object' && 'toDate' in value && typeof (value as { toDate: () => Date }).toDate === 'function') {
   921→          converted[key] = (value as { toDate: () => Date }).toDate().toISOString();
   922→        } else {
   923→          converted[key] = value;
   924→        }
   925→      }
   926→      return converted as T & { id: string };
   927→    });
   928→    callback(data);
   929→  }, (error) => {
   930→    console.warn(`[Firestore] Subscription error for ${collectionName}:`, error);
   931→    callback([]);
   932→  });
   933→}

   935→/** Subscribe to a single Firestore document */
   936→export function subscribeToDoc<T>(
   937→  collectionName: CollectionName,
   938→  docId: string,
   939→  callback: (data: (T & { id: string }) | null) => void,
   940→): Unsubscribe {
   941→  return onSnapshot(doc(db, collectionName, docId), (snap) => {
   942→    if (!snap.exists()) {
   943→      callback(null);
   944→      return;
   945→    }
   946→    const docData = snap.data();
   947→    const converted: Record<string, unknown> = { id: snap.id };
   948→    for (const [key, value] of Object.entries(docData)) {
   949→      if (value && typeof value === 'object' && 'toDate' in value && typeof (value as { toDate: () => Date }).toDate === 'function') {
   950→        converted[key] = (value as { toDate: () => Date }).toDate().toISOString();
   951→      } else {
   952→        converted[key] = value;
   953→      }
   954→    }
   955→    callback(converted as T & { id: string });
   956→  }, (error) => {
   957→    console.warn(`[Firestore] Doc subscription error for ${collectionName}/${docId}:`, error);
   958→    callback(null);
   959→  });
   960→}

   962→/** Compute live dashboard metrics from Firestore data */
   963→export function computeDashboardMetrics(
   964→  clients: Array<FirestoreClient & { id: string }>,
   965→  invoices: Array<FirestoreInvoice & { id: string }>,
   966→  returns: Array<FirestoreReturn & { id: string }>,
   967→  documents: Array<FirestoreDocument & { id: string }>,
   968→  activities: Array<FirestoreActivity & { id: string }>,
   969→): LiveDashboardMetrics {
   970→  const activeClients = clients.filter(c => c.status === 'active').length;
   971→  const filedReturns = returns.filter(r => r.status === 'filed').length;
   972→  const pendingReturns = returns.filter(r => r.status !== 'filed').length;
   973→  const readyToFile = returns.filter(r => ['validated', 'reviewed', 'generated'].includes(r.status)).length;
   974→  const overdueReturns = returns.filter(r => {
   975→    if (r.status === 'filed') return false;
   976→    const periodDate = parsePeriod(r.period);
   977→    return periodDate && periodDate < new Date();
   978→  }).length;
   979→
   980→  const totalTaxVolume = invoices.reduce((sum, inv) => sum + (inv.totalAmount || 0), 0);
   981→  const totalTax = invoices.reduce((sum, inv) => sum + (inv.cgst || 0) + (inv.sgst || 0) + (inv.igst || 0) + (inv.cess || 0), 0);
   982→
   983→  const criticalIssues = invoices.filter(inv => inv.riskLevel === 'critical').length +
   984→    returns.filter(r => r.criticalErrors > 0).length;
   985→  const warnings = invoices.filter(inv => inv.riskLevel === 'high' || inv.riskLevel === 'medium').length +
   986→    returns.filter(r => r.warnings > 0).length;
   987→
   988→  const averageHealthScore = clients.length > 0
   989→    ? Math.round(clients.reduce((sum, c) => sum + c.healthScore, 0) / clients.length)
   990→    : 0;
   991→
   992→  const matchedInvoices = invoices.filter(inv => inv.matchStatus === 'perfect_match').length;
   993→  const matchPercentage = invoices.length > 0 ? Math.round((matchedInvoices / invoices.length) * 100) : 100;
   994→
   995→  const riskyInvoices = invoices.filter(inv => ['high', 'critical'].includes(inv.riskLevel)).length;
   996→  const riskPercentage = invoices.length > 0 ? Math.round((riskyInvoices / invoices.length) * 100) : 0;
   997→
   998→  const documentsProcessed = documents.filter(d => d.extractionStatus === 'completed').length;
   999→  const extractionsPending = documents.filter(d => d.extractionStatus === 'pending' || d.extractionStatus === 'in_progress').length;
  1000→
  1001→  const recentActivities = [...activities]
  1002→    .sort((a, b) => {
  1003→      const aTime = typeof a.createdAt === 'string' ? new Date(a.createdAt).getTime() : 0;
  1004→      const bTime = typeof b.createdAt === 'string' ? new Date(b.createdAt).getTime() : 0;
  1005→      return bTime - aTime;
  1006→    })
  1007→    .slice(0, 10);
  1008→
  1009→  const upcomingFilings = returns
  1010→    .filter(r => r.status !== 'filed')
  1011→    .sort((a, b) => {
  1012→      const aTime = typeof a.createdAt === 'string' ? new Date(a.createdAt).getTime() : 0;
  1013→      const bTime = typeof b.createdAt === 'string' ? new Date(b.createdAt).getTime() : 0;
  1014→      return aTime - bTime;
  1015→    })
  1016→    .slice(0, 5);
  1017→
  1018→  return {
  1019→    totalClients: clients.length,
  1020→    activeClients,
  1021→    totalInvoices: invoices.length,
  1022→    totalTaxVolume,
  1023→    filedReturns,
  1024→    pendingReturns,
  1025→    overdueReturns,
  1026→    readyToFile,
  1027→    criticalIssues,
  1028→    warnings,
  1029→    averageHealthScore,
  1030→    matchPercentage,
  1031→    riskPercentage,
  1032→    documentsProcessed,
  1033→    extractionsPending,
  1034→    recentActivities,
  1035→    upcomingFilings,
  1036→  };
  1037→}

  1039→function parsePeriod(period: string | null): Date | null {
  1040→  if (!period) return null;
  1041→  try {
  1042→    const [mm, yyyy] = period.split('-').map(Number);
  1043→    if (!mm || !yyyy) return null;
  1044→    return new Date(yyyy, mm, 20); // Due date is typically 20th of next month
  1045→  } catch {
  1046→    return null;
  1047→  }
  1048→}
