   126→   126→    try { await markNotificationRead(notifId) } catch {}

   130→   130→    try {

   140→   140→      case 'high': return 'text-orange-600 bg-orange-50 border-orange-200'
   141→   141→      case 'normal': return 'text-blue-600 bg-blue-50 border-blue-200'
   142→   142→      default: return 'text-slate-600 bg-slate-50 border-slate-200'
   143→   143→    }

   284→   284→function useKeyboardShortcuts() {
   285→   285→  const { setCurrentView, setCommandPaletteOpen, commandPaletteOpen } = useApp()
   286→   286→  const gPressed = useRef(false)
   287→   287→  const gTimer = useRef<NodeJS.Timeout | null>(null)
   288→   288→
   289→   289→  useEffect(() => {
   290→   290→    const handler = (e: KeyboardEvent) => {
   291→   291→      // Ctrl+K → Toggle Command Palette
   292→   292→      if ((e.metaKey || e.ctrlKey) && e.key === 'k') {
   293→   293→        e.preventDefault()
   294→   294→        setCommandPaletteOpen(!commandPaletteOpen)
   295→   295→        return
   296→   296→      }
   297→   297→
   298→   298→      // G prefix shortcuts (G+D, G+C, G+I, G+R)
   299→   299→      if (e.key === 'g' && !e.metaKey && !e.ctrlKey && !e.altKey && !(e.target instanceof HTMLInputElement || e.target instanceof HTMLTextAreaElement)) {
   300→   300→        if (gPressed.current) return
   301→   301→        gPressed.current = true
   302→   302→        gTimer.current = setTimeout(() => { gPressed.current = false }, 500)
   303→   303→        return
   304→   304→      }
   305→   305→
   306→   306→      if (gPressed.current && !e.metaKey && !e.ctrlKey && !e.altKey) {
   307→   307→        gPressed.current = false
   308→   308→        if (gTimer.current) clearTimeout(gTimer.current)
   309→   309→
   310→   310→        switch (e.key) {
   311→   311→          case 'd': setCurrentView('dashboard'); break
   312→   312→          case 'c': setCurrentView('clients'); break
   313→   313→          case 'i': setCurrentView('invoices'); break
   314→   314→          case 'r': setCurrentView('returns'); break
   315→   315→        }
   316→   316→      }
   317→   317→    }
   318→   318→
   319→   319→    window.addEventListener('keydown', handler)
   320→   320→    return () => window.removeEventListener('keydown', handler)
   321→   321→  }, [setCurrentView, setCommandPaletteOpen, commandPaletteOpen])
   322→   322→}

   507→   507→// ONBOARDING SCREEN
   508→   508→// ═══════════════════════════════════════════════════════════════════════════════
   509→   509→
   510→   510→function OnboardingScreen() {
   511→   511→  const { user, markOnboardingComplete } = useAuth()
   512→   512→  const { setCurrentScreen, setCurrentView } = useApp()
   513→   513→  const [saveError, setSaveError] = React.useState<string | null>(null)
   514→   514→
   515→   515→  const saveOnboardingToFirestore = async (
   516→   516→    data: import('@/components/onboarding/OnboardingFlow').OnboardingData,
   517→   517→    firmId: string
   518→   518→  ) => {
   519→   519→    try {
   520→   520→      const { doc, setDoc, serverTimestamp } = await import('firebase/firestore')
   521→   521→      const { db } = await import('@/lib/firebase')
   522→   522→      if (!user) return
   523→   523→
   524→   524→      await setDoc(doc(db, 'users', user.id), {
   525→   525→        uid: user.id, fullName: data.fullName, email: data.email, phone: data.phone,
   526→   526→        ageGroup: data.ageGroup, profession: data.profession, experience: data.experience,
   527→   527→        firmId: firmId || null, firmName: data.firmName, onboardingCompleted: true,
   528→   528→        clientCount: data.clientCount, monthlyReturns: data.monthlyReturns,
   529→   529→        gstServices: data.gstServices, painPoints: data.painPoints,
   530→   530→        referralSource: data.referralSource, trialReasons: data.trialReasons,
   531→   531→        wantsUpdates: data.wantsUpdates, updatedAt: serverTimestamp(),
   532→   532→      }, { merge: true })
   533→   533→
   534→   534→      await setDoc(doc(db, 'onboarding', user.id), {
   535→   535→        ...data, firmId: firmId || null, completedAt: serverTimestamp(),
   536→   536→      })
   537→   537→    } catch (error) {
   538→   538→      console.warn('[Onboarding] Background Firestore save failed:', error)
   539→   539→    }
   540→   540→  }
   541→   541→
   542→   542→  const handleOnboardingComplete = (
   543→   543→    data: import('@/components/onboarding/OnboardingFlow').OnboardingData,
   544→   544→    destination?: import('@/components/onboarding/OnboardingFlow').OnboardingDestination
   545→   545→  ) => {
   546→   546→    setSaveError(null)
   547→   547→    if (!user) { setSaveError('No authenticated user found. Please sign in again.'); return }
   548→   548→
   549→   549→    markOnboardingComplete(undefined, data.firmName)
   550→   550→    setCurrentView(destination || 'dashboard')
   551→   551→    setCurrentScreen('app')
   552→   552→
   553→   553→    ;(async () => {
   554→   554→      try {
   555→   555→        const { collection, addDoc, serverTimestamp } = await import('firebase/firestore')
   556→   556→        const { db } = await import('@/lib/firebase')
   557→   557→        let firmId = ''
   558→   558→        try {
   559→   559→          const firmRef = await addDoc(collection(db, 'firms'), {
   560→   560→            ownerId: user.id, firmName: data.firmName, gstin: data.gstin || null,
   561→   561→            state: data.state, stateCode: data.stateCode,
   562→   562→            organizationType: data.organizationType,
   563→   563→            icaiMembershipNo: data.icaiMembershipNo || null,
   564→   564→            officeAddress: data.officeAddress || null, createdAt: serverTimestamp(),
   565→   565→          })
   566→   566→          firmId = firmRef.id
   567→   567→          markOnboardingComplete(firmId, data.firmName)
   568→   568→        } catch (firmError) {
   569→   569→          console.warn('[Onboarding] Firm creation failed:', firmError)
   570→   570→        }
   571→   571→        await saveOnboardingToFirestore(data, firmId)
   572→   572→      } catch (error) {
   573→   573→        console.warn('[Onboarding] Background save error:', error)
   574→   574→      }
   575→   575→    })()
   576→   576→  }
   577→   577→
   578→   578→  const handleSkip = () => {
   579→   579→    setSaveError(null)
   580→   580→    if (!user) { setSaveError('No authenticated user found.'); return }
   581→   581→
   582→   582→    markOnboardingComplete()
   583→   583→    setCurrentView('dashboard')
   584→   584→    setCurrentScreen('app')
   585→   585→
   586→   586→    ;(async () => {
   587→   587→      try {
   588→   588→        const { doc, setDoc, serverTimestamp } = await import('firebase/firestore')
   589→   589→        const { db } = await import('@/lib/firebase')
   590→   590→        await setDoc(doc(db, 'users', user.id), {
   591→   591→          onboardingCompleted: true, updatedAt: serverTimestamp(),
   592→   592→        }, { merge: true })
   593→   593→      } catch (error) {
   594→   594→        console.warn('[Onboarding] Background skip save failed:', error)
   595→   595→      }
   596→   596→    })()
   597→   597→  }
   598→   598→
   599→   599→  return (
   600→   600→    <OnboardingFlow
   601→   601→      onComplete={handleOnboardingComplete}
   602→   602→      onSkip={handleSkip}
   603→   603→      userEmail={user?.email}
   604→   604→      userName={user?.name}
   605→   605→      error={saveError}
   606→   606→      onDismissError={() => setSaveError(null)}
   607→   607→    />
   608→   608→  )
   609→   609→}

   613→   613→// ═══════════════════════════════════════════════════════════════════════════════
   614→   614→
   615→   615→function AppRouter() {
   616→   616→  const { currentScreen, setCurrentScreen } = useApp()
   617→   617→  const { isAuthenticated, isInitializing, needsOnboarding, needsEmailVerification } = useAuth()
   618→   618→
   619→   619→  useEffect(() => {
   620→   620→    if (isInitializing) return
   621→   621→    if (isAuthenticated && !needsOnboarding && currentScreen !== 'app') {
   622→   622→      setCurrentScreen('app')
   623→   623→    }
   624→   624→  }, [isAuthenticated, isInitializing, currentScreen, setCurrentScreen, needsOnboarding])
   625→   625→
   626→   626→  useEffect(() => {
   627→   627→    if (isInitializing) return
   628→   628→    if (!isAuthenticated && currentScreen === 'app') {
   629→   629→      setCurrentScreen('landing')
   630→   630→    }
   631→   631→  }, [isAuthenticated, isInitializing, currentScreen, setCurrentScreen])
   632→   632→
   633→   633→  const handleGetStarted = () => setCurrentScreen('login')
   634→   634→  const handleBookDemo = () => setCurrentScreen('login')
   635→   635→  const handleBackToLanding = () => setCurrentScreen('landing')
   636→   636→
   637→   637→  if (isInitializing && !isAuthenticated) {
   638→   638→    return (
   639→   639→      <div className="min-h-screen flex items-center justify-center bg-white">
   640→   640→        <div className="flex flex-col items-center gap-3">
   641→   641→          <div className="flex h-10 w-10 items-center justify-center rounded-lg bg-gradient-to-br from-emerald-500 to-emerald-600 shadow-lg shadow-emerald-600/20">
   642→   642→            <Zap className="h-6 w-6 text-white animate-pulse" />
   643→   643→          </div>
   644→   644→          <span className="text-sm text-slate-500 font-medium">Loading GSTPilot...</span>
   645→   645→        </div>
   646→   646→      </div>
   647→   647→    )
   648→   648→  }
   649→   649→
   650→   650→  if (isAuthenticated && needsOnboarding) {
   651→   651→    return <OnboardingScreen />
   652→   652→  }
   653→   653→
   654→   654→  if (currentScreen === 'app' && isAuthenticated) {
   655→   655→    return (
   656→   656→      <div className="min-h-screen flex flex-col">
   657→   657→        {needsEmailVerification && <EmailVerificationBanner />}
   658→   658→        <div className="flex-1 flex">
   659→   659→          <DashboardContent />
   660→   660→        </div>
   661→   661→      </div>
   662→   662→    )
   663→   663→  }
   664→   664→
   665→   665→  if (currentScreen === 'login') {
   666→   666→    return <LoginPage onBack={handleBackToLanding} onGetStarted={handleGetStarted} />
   667→   667→  }
   668→   668→
   669→   669→  return <LandingPage onGetStarted={handleGetStarted} onBookDemo={handleBookDemo} />
   670→   670→}
