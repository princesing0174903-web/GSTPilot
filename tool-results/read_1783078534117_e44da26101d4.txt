   108→// ─── Format message with simple markdown ───────────────────────────────────
   109→function formatMessage(content: string) {
   110→  return content.split('\n').map((line, i) => {
   111→    const formatted = line
   112→      .replace(/\*\*(.*?)\*\*/g, '<strong>$1</strong>')
   113→      .replace(/⚠️/g, '<span class="text-amber-500">⚠️</span>');
   114→    if (formatted.startsWith('•')) {
   115→      return (
   116→        <div key={i} className="ml-2" dangerouslySetInnerHTML={{ __html: formatted }} />
   117→      );
   118→    }
   119→    if (/^\d+\./.test(formatted)) {
   120→      return (
   121→        <div key={i} className="ml-3" dangerouslySetInnerHTML={{ __html: formatted }} />
   122→      );
   123→    }
   124→    return (
   125→      <React.Fragment key={i}>
   126→        <span dangerouslySetInnerHTML={{ __html: formatted }} />
   127→        {i < content.split('\n').length - 1 && <br />}
   128→      </React.Fragment>
   129→    );
   130→  });
   131→}

   133→// ─── Format relative date ──────────────────────────────────────────────────
   134→function formatRelativeDate(dateStr: string): string {
   135→  const date = new Date(dateStr);
   136→  const now = new Date();
   137→  const diffMs = now.getTime() - date.getTime();
   138→  const diffHours = Math.floor(diffMs / (1000 * 60 * 60));
   139→  const diffDays = Math.floor(diffMs / (1000 * 60 * 60 * 24));
   140→
   141→  if (diffHours < 1) return 'Just now';
   142→  if (diffHours < 24) return `${diffHours}h ago`;
   143→  if (diffDays < 7) return `${diffDays}d ago`;
   144→  return date.toLocaleDateString('en-IN', { day: 'numeric', month: 'short' });
   145→}
