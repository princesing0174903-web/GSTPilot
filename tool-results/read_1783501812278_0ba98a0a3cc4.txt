     3→     3→// GSTPilot — Real-time Firestore Hooks
     4→     4→// React hooks with onSnapshot listeners for live data + optimistic UI.
     5→     5→//
     6→     6→// ARCHITECTURE (Production-grade multi-tenant):
     7→     7→//   • Every query is scoped by `organizationId` from OrgContext — NEVER by the
     8→     8→//     legacy `firmId`. This matches the Firestore security rules, which require
     9→     9→//     every tenant-scoped document to carry `organizationId` and only allow
    10→    10→//     reads where `isOrgMember(resource.data.organizationId)` is true.
    11→    11→//   • If there is no organization (preview / demo mode, or not yet loaded),
    12→    12→//     hooks return empty data WITHOUT subscribing — so the UI renders premium
    13→    13→//     empty states instead of throwing permission errors.
    14→    14→//   • Permission errors (permission-denied / unauthenticated) are treated as
    15→    15→//     "no data available" — the hook clears its error and returns an empty
    16→    16→//     array. This is the graceful-degradation contract: the app NEVER shows a
    17→    17→//     "Missing or insufficient permissions" wall. Genuine errors (network,
    18→    18→//     index missing) still surface as `error`.

    46→    46→// ─── Permission-error detection (graceful degradation) ───────────────────────
    47→    47→
    48→    48→/**
    49→    49→ * True if the given Firestore error is a permission / auth failure that we
    50→    50→ * should treat as "no data available" rather than a hard error. This is the
    51→    51→ * core of the never-show-a-permission-wall contract.
    52→    52→ */
    53→    53→function isPermissionError(err: unknown): boolean {
    54→    54→  if (!err) return false;
    55→    55→  const e = err as { code?: string; message?: string };
    56→    56→  const code = (e.code || '').toLowerCase();
    57→    57→  const message = (e.message || '').toLowerCase();
    58→    58→  if (
    59→    59→    code === 'permission-denied' ||
    60→    60→    code === 'unauthenticated' ||
    61→    61→    code === 'auth/operation-not-allowed' ||
    62→    62→    code === 'auth/user-not-found'
    63→    63→  ) {
    64→    64→    return true;
    65→    65→  }
    66→    66→  if (
    67→    67→    message.includes('missing or insufficient permissions') ||
    68→    68→    message.includes('permission-denied') ||
    69→    69→    message.includes('insufficient permissions') ||
    70→    70→    message.includes('not authorized') ||
    71→    71→    message.includes('unauthenticated')
    72→    72→  ) {
    73→    73→    return true;
    74→    74→  }
    75→    75→  return false;
    76→    76→}

    78→    78→// ─── Timestamp Converter ─────────────────────────────────────────────────────
    79→    79→
    80→    80→function convertDoc<T extends Record<string, unknown>>(snapData: Record<string, unknown>): T {
    81→    81→  const converted: Record<string, unknown> = {};
    82→    82→  for (const [key, value] of Object.entries(snapData)) {
    83→    83→    if (value && typeof value === 'object' && 'toDate' in value && typeof (value as { toDate: () => Date }).toDate === 'function') {
    84→    84→      converted[key] = (value as { toDate: () => Date }).toDate().toISOString();
    85→    85→    } else if (value && typeof value === 'object' && !Array.isArray(value) && !(value instanceof Date)) {
    86→    86→      // Recursively convert nested objects
    87→    87→      converted[key] = convertDoc(value as Record<string, unknown>);
    88→    88→    } else {
    89→    89→      converted[key] = value;
    90→    90→    }
    91→    91→  }
    92→    92→  return converted as T;
    93→    93→}

    95→    95→// ─── Generic Collection Hook ─────────────────────────────────────────────────
    96→    96→//
    97→    97→// The hook resolves the current `organizationId` from OrgContext. If the org
    98→    98→// isn't resolved yet (loading) OR we're in preview mode (no org), the hook
    99→    99→// returns empty data without subscribing. This guarantees no permission errors
   100→   100→// ever reach the UI in preview/offline mode.
   101→   101→
   102→   102→function useFirestoreCollection<T>(
   103→   103→  collectionName: CollectionName,
   104→   104→  constraints: QueryConstraint[] = [],
   105→   105→  deps: unknown[] = [],
   106→   106→): { data: Array<T & { id: string }>; loading: boolean; error: string | null } {
   107→   107→  const { user } = useAuth();
   108→   108→  const { organization, isPreviewMode, loading: orgLoading } = useOrg();
   109→   109→  const organizationId = organization?.id ?? null;
   110→   110→
   111→   111→  const [data, setData] = useState<Array<T & { id: string }>>([]);
   112→   112→  const [loading, setLoading] = useState(true);
   113→   113→  const [error, setError] = useState<string | null>(null);
   114→   114→
   115→   115→  useEffect(() => {
   116→   116→    // No user → nothing to subscribe to (empty data).
   117→   117→    if (!user) {
   118→   118→      setData([]);
   119→   119→      setLoading(false);
   120→   120→      setError(null);
   121→   121→      return;
   122→   122→    }
   123→   123→
   124→   124→    // Org still resolving — stay in loading state, but don't subscribe yet.
   125→   125→    if (orgLoading && !organizationId) {
   126→   126→      return;
   127→   127→    }
   128→   128→
   129→   129→    // Preview mode (no real org) OR no organization at all → empty data,
   130→   130→    // no subscription. This is the preview-mode contract.
   131→   131→    if (!organizationId || isPreviewMode || organizationId === 'preview-org') {
   132→   132→      setData([]);
   133→   133→      setLoading(false);
   134→   134→      setError(null);
   135→   135→      return;
   136→   136→    }
   137→   137→
   138→   138→    let unsub: Unsubscribe | null = null;
   139→   139→
   140→   140→    // Scope EVERY query by organizationId. This is required by the Firestore
   141→   141→    // security rules — unbounded collection reads are denied.
   142→   142→    const baseConstraints: QueryConstraint[] = [
   143→   143→      where('organizationId', '==', organizationId),
   144→   144→      ...constraints,
   145→   145→    ];
   146→   146→
   147→   147→    const q = query(collection(db, collectionName), ...baseConstraints);
   148→   148→
   149→   149→    unsub = onSnapshot(q, (snap) => {
   150→   150→      const items = snap.docs.map(d => ({
   151→   151→        id: d.id,
   152→   152→        ...convertDoc<T>(d.data() as Record<string, unknown>),
   153→   153→      }));
   154→   154→      setData(items);
   155→   155→      setLoading(false);
   156→   156→      setError(null);
   157→   157→    }, (err) => {
   158→   158→      console.warn(`[Firestore] ${collectionName} subscription error:`, err);
   159→   159→      // Permission errors → graceful degradation (empty data, no error wall).
   160→   160→      if (isPermissionError(err)) {
   161→   161→        setData([]);
   162→   162→        setError(null);
   163→   163→        setLoading(false);
   164→   164→        return;
   165→   165→      }
   166→   166→      // Genuine errors (network, missing index) still surface.
   167→   167→      setError(err.message);
   168→   168→      setLoading(false);
   169→   169→    });
   170→   170→
   171→   171→    return () => {
   172→   172→      if (unsub) unsub();
   173→   173→    };
   174→   174→  }, [user, organizationId, isPreviewMode, orgLoading, ...deps]);
   175→   175→
   176→   176→  // When user is null, return empty data (not loading)
   177→   177→  if (!user && loading) {
   178→   178→    return { data: [], loading: false, error: null };
   179→   179→  }
   180→   180→
   181→   181→  return { data, loading, error };
   182→   182→}

   184→   184→// ─── Generic Document Hook ───────────────────────────────────────────────────
   185→   185→
   186→   186→function useFirestoreDoc<T>(
   187→   187→  collectionName: CollectionName,
   188→   188→  docId: string | null,
   189→   189→): { data: (T & { id: string }) | null; loading: boolean; error: string | null } {
   190→   190→  const { user } = useAuth();
   191→   191→  const { organization, isPreviewMode } = useOrg();
   192→   192→  const organizationId = organization?.id ?? null;
   193→   193→
   194→   194→  const [data, setData] = useState<(T & { id: string }) | null>(null);
   195→   195→  const [loading, setLoading] = useState(!!docId);
   196→   196→  const [error, setError] = useState<string | null>(null);
   197→   197→
   198→   198→  useEffect(() => {
   199→   199→    if (!docId) {
   200→   200→      setData(null);
   201→   201→      setLoading(false);
   202→   202→      return;
   203→   203→    }
   204→   204→
   205→   205→    // Preview mode (no real org) → no doc, no subscription.
   206→   206→    if (!user || !organizationId || isPreviewMode || organizationId === 'preview-org') {
   207→   207→      setData(null);
   208→   208→      setLoading(false);
   209→   209→      setError(null);
   210→   210→      return;
   211→   211→    }
   212→   212→
   213→   213→    const unsub = onSnapshot(doc(db, collectionName, docId), (snap) => {
   214→   214→      if (!snap.exists()) {
   215→   215→        setData(null);
   216→   216→        setLoading(false);
   217→   217→        return;
   218→   218→      }
   219→   219→      setData({
   220→   220→        id: snap.id,
   221→   221→        ...convertDoc<T>(snap.data() as Record<string, unknown>),
   222→   222→      });
   223→   223→      setLoading(false);
   224→   224→      setError(null);
   225→   225→    }, (err) => {
   226→   226→      console.warn(`[Firestore] ${collectionName}/${docId} doc error:`, err);
   227→   227→      if (isPermissionError(err)) {
   228→   228→        setData(null);
   229→   229→        setError(null);
   230→   230→        setLoading(false);
   231→   231→        return;
   232→   232→      }
   233→   233→      setError(err.message);
   234→   234→      setLoading(false);
   235→   235→    });
   236→   236→
   237→   237→    return () => {
   238→   238→      if (unsub) unsub();
   239→   239→    };
   240→   240→  }, [collectionName, docId, user, organizationId, isPreviewMode]);
   241→   241→
   242→   242→  return { data, loading, error };
   243→   243→}

   502→   502→// ─── Executive Scores (computed from live data) ─────────────────────────────
   503→   503→
   504→   504→export function useFirmExecutiveScores(): { scores: FirmExecutiveScores; loading: boolean } {
   505→   505→  const { data: clients } = useFireClients();
   506→   506→  const { data: returns } = useFireReturns();
   507→   507→  const { data: invoices } = useFireInvoices();
   508→   508→  const { data: reconciliations } = useFireReconciliations();
   509→   509→  const { data: activities } = useFireRecentActivities(50);
   510→   510→
   511→   511→  const loading = clients.length === 0 && returns.length === 0;
   512→   512→
   513→   513→  const scores = useMemo<FirmExecutiveScores>(() => {
   514→   514→    if (loading) {
   515→   515→      return { firmHealth: 0, revenue: 0, compliance: 0, teamEfficiency: 0, clientSatisfaction: 0, cashFlow: 0 };
   516→   516→    }
   517→   517→
   518→   518→    // Firm Health: weighted avg of client health scores
   519→   519→    const activeClients = clients.filter(c => c.status === 'active');
   520→   520→    const avgHealth = activeClients.length > 0
   521→   521→      ? activeClients.reduce((s, c) => s + (c.healthScore || 0), 0) / activeClients.length
   522→   522→      : 50;
   523→   523→
   524→   524→    // Revenue Score: based on tax volume
   525→   525→    const totalTax = invoices.reduce((s, i) => s + (i.totalTax || 0) + (i.cgst || 0) + (i.sgst || 0) + (i.igst || 0), 0);
   526→   526→    const revenueScore = Math.min(100, Math.round((totalTax / 1000000) * 100));
   527→   527→
   528→   528→    // Compliance Score: filed vs total returns
   529→   529→    const filedReturns = returns.filter(r => r.status === 'filed').length;
   530→   530→    const complianceScore = returns.length > 0 ? Math.round((filedReturns / returns.length) * 100) : 50;
   531→   531→
   532→   532→    // Team Efficiency: based on completed activities
   533→   533→    const completedActivities = activities.filter(a => a.type.includes('filed') || a.type.includes('completed') || a.type.includes('resolved')).length;
   534→   534→    const teamEfficiency = activities.length > 0 ? Math.min(100, Math.round((completedActivities / Math.max(activities.length, 1)) * 100)) : 50;
   535→   535→
   536→   536→    // Client Satisfaction: inverse of overdue + health
   537→   537→    const overdueClients = activeClients.filter(c => (c.pendingReturnCount || 0) > 0).length;
   538→   538→    const clientSatisfaction = activeClients.length > 0 ? Math.round(((activeClients.length - overdueClients) / activeClients.length) * 100) : 50;
   539→   539→
   540→   540→    // Cash Flow: based on reconciliation match rate
   541→   541→    const matchedRecons = reconciliations.filter(r => r.matched > 0);
   542→   542→    const totalMatched = matchedRecons.reduce((s, r) => s + r.matched, 0);
   543→   543→    const totalRecords = matchedRecons.reduce((s, r) => s + r.totalRecords, 0);
   544→   544→    const cashFlow = totalRecords > 0 ? Math.round((totalMatched / totalRecords) * 100) : 50;
   545→   545→
   546→   546→    return {
   547→   547→      firmHealth: Math.round(avgHealth) || 50,
   548→   548→      revenue: revenueScore || 50,
   549→   549→      compliance: complianceScore || 50,
   550→   550→      teamEfficiency: teamEfficiency || 50,
   551→   551→      clientSatisfaction: clientSatisfaction || 50,
   552→   552→      cashFlow: cashFlow || 50,
   553→   553→    };
   554→   554→  }, [clients, returns, invoices, reconciliations, activities, loading]);
   555→   555→
   556→   556→  return { scores, loading };
   557→   557→}
