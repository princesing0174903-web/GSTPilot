    59→function formatRelativeTime(ts: string) {
    60→  const diff = Date.now() - new Date(ts).getTime();
    61→  const m = Math.floor(diff / 60000);
    62→  if (m < 1) return 'Just now';
    63→  if (m < 60) return `${m}m ago`;
    64→  const h = Math.floor(m / 60);
    65→  if (h < 24) return `${h}h ago`;
    66→  const d = Math.floor(h / 24);
    67→  if (d < 7) return `${d}d ago`;
    68→  return new Date(ts).toLocaleDateString();
    69→}

    71→function formatFileSize(bytes: number) {
    72→  if (bytes < 1024) return `${bytes} B`;
    73→  if (bytes < 1048576) return `${(bytes / 1024).toFixed(1)} KB`;
    74→  return `${(bytes / 1048576).toFixed(1)} MB`;
    75→}

   113→// ═══════════════════════════════════════════════════════════════════════════════
   114→// Circular Progress Component for Compliance Score
   115→// ═══════════════════════════════════════════════════════════════════════════════
   116→function CircularProgress({ value, size = 80, strokeWidth = 6 }: { value: number; size?: number; strokeWidth?: number }) {
   117→  const radius = (size - strokeWidth) / 2;
   118→  const circumference = 2 * Math.PI * radius;
   119→  const offset = circumference - (value / 100) * circumference;
   120→  const color = value >= 80 ? '#10b981' : value >= 50 ? '#f59e0b' : '#ef4444';
   121→  const textColor = value >= 80 ? 'text-emerald-600' : value >= 50 ? 'text-amber-600' : 'text-red-600';
   122→
   123→  return (
   124→    <div className="relative shrink-0" style={{ width: size, height: size }}>
   125→      <svg width={size} height={size} className="-rotate-90">
   126→        <circle cx={size / 2} cy={size / 2} r={radius} fill="none" stroke="currentColor" strokeWidth={strokeWidth} className="text-slate-100" />
   127→        <circle cx={size / 2} cy={size / 2} r={radius} fill="none" stroke={color} strokeWidth={strokeWidth} strokeDasharray={circumference} strokeDashoffset={offset} strokeLinecap="round" className="transition-all duration-700 ease-out" />
   128→      </svg>
   129→      <div className="absolute inset-0 flex items-center justify-center">
   130→        <span className={`text-lg font-bold ${textColor}`}>{value > 0 ? `${value}%` : '—'}</span>
   131→      </div>
   132→    </div>
   133→  );
   134→}
