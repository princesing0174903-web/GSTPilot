    29→    29→
    30→    30→async function apiFetch<T>(url: string, options?: RequestInit): Promise<T> {
    31→    31→  const res = await fetch(url, {
    32→    32→    headers: { 'Content-Type': 'application/json', ...options?.headers },
    33→    33→    ...options,
    34→    34→  });
    35→    35→  if (!res.ok) {
    36→    36→    const error = await res.json().catch(() => ({ error: 'Request failed' }));
    37→    37→    throw new Error(error.error || `HTTP ${res.status}`);
    38→    38→  }
    39→    39→  return res.json();
    40→    40→}

   248→   248→export function useClient(
   249→   249→  id: string | undefined,
   250→   250→  options?: Omit<
   251→   251→    UseQueryOptions<ClientsResponse, Error>,
   252→   252→    'queryKey' | 'queryFn'
   253→   253→  >
   254→   254→) {
   255→   255→  return useQuery<ClientsResponse, Error>({
   256→   256→    queryKey: queryKeys.clients.detail(id ?? ''),
   257→   257→    queryFn: () =>
   258→   258→      apiFetch<ClientsResponse>('/api/clients').then((res) => {
   259→   259→        // Filter to single client if id provided
   260→   260→        if (id && res.clients) {
   261→   261→          const client = res.clients.find((c: any) => c.id === id);
   262→   262→          return { client: client ?? null, clients: res.clients };
   263→   263→        }
   264→   264→        return res;
   265→   265→      }),
   266→   266→    enabled: !!id,
   267→   267→    ...options,
   268→   268→  });
   269→   269→}

   743→   743→    'mutationFn'

   748→   748→  return useMutation<DocumentResponse, Error, FormData | Record<string, unknown>>({
   749→   749→    mutationFn: (data) => {
   750→   750→      if (data instanceof FormData) {
   751→   751→        // FormData upload — don't set Content-Type (browser sets multipart boundary)
   752→   752→        return fetch('/api/upload', {
   753→   753→          method: 'POST',
   754→   754→          body: data,
   755→   755→        }).then(async (res) => {
   756→   756→          if (!res.ok) {
   757→   757→            const error = await res.json().catch(() => ({ error: 'Upload failed' }));
   758→   758→            throw new Error(error.error || `HTTP ${res.status}`);
   759→   759→          }
   760→   760→          return res.json();
   761→   761→        });
   762→   762→      }
   763→   763→      // JSON upload
   764→   764→      return apiFetch<DocumentResponse>('/api/documents', {
   765→   765→        method: 'POST',
   766→   766→        body: JSON.stringify(data),
   767→   767→      });
   768→   768→    },
   769→   769→    onSuccess: (data, variables, context) => {
   770→   770→      const clientId = variables instanceof FormData ? variables.get('clientId') as string : (variables as any).clientId;
   771→   771→      queryClient.invalidateQueries({
   772→   772→        queryKey: queryKeys.documents.all(clientId),
   773→   773→      });
   774→   774→      queryClient.invalidateQueries({ queryKey: queryKeys.documents.all() });
   775→   775→      options?.onSuccess?.(data, variables, context);
   776→   776→    },
   777→   777→    ...options,
   778→   778→  });
