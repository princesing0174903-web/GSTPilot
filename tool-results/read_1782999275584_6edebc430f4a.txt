    32→function ValidationGauge({ percent }: { percent: number }) {

    34→  const c = 2 * Math.PI * r
    35→  const offset = c - (percent / 100) * c

    37→    <svg width="100" height="100" className="overflow-visible">
    38→      <circle cx="50" cy="50" r={r} fill="none" stroke="currentColor" strokeWidth="8" className="text-slate-100 dark:text-slate-800" />
    39→      <circle cx="50" cy="50" r={r} fill="none" stroke="#10b981" strokeWidth="8" strokeLinecap="round" strokeDasharray={c} strokeDashoffset={offset} transform="rotate(-90 50 50)" />

    46→function DailyIRNChart({ data }: { data: { day: string; count: number }[] }) {
    47→  const max = Math.max(...data.map(d => d.count))
    48→  const barW = 24
    49→  const gap = 6
    50→  const h = 70
    51→  return (
    52→    <svg width={data.length * (barW + gap)} height={h + 20} className="overflow-visible">
    53→      {data.map((d, i) => {
    54→        const barH = (d.count / max) * (h - 10)
    55→        return (
    56→          <g key={d.day}>
    57→            <rect x={i * (barW + gap)} y={h - barH} width={barW} height={barH} rx={3} fill="#10b981" opacity={0.6 + (d.count / max) * 0.4} />
    58→            <text x={i * (barW + gap) + barW / 2} y={h + 14} textAnchor="middle" className="text-[9px] fill-muted-foreground">{d.day}</text>
    59→          </g>
    60→        )
    61→      })}
    62→    </svg>
    63→  )
    64→}
