    78→function filingDueDate(returnType: string, period: string): Date | null {
    79→  // period format: "YYYY-MM"
    80→  const m = period.match(/^(\d{4})-(\d{2})$/);
    81→  if (!m) return null;
    82→  const year = parseInt(m[1], 10);
    83→  const month = parseInt(m[2], 10);
    84→  if (month === 12) return new Date(year + 1, 0, 11); // Jan 11 next year
    85→  const day = returnType === 'GSTR-1' ? 11 : 20;
    86→  return new Date(year, month, day); // next month, day 11 or 20
    87→}
    88→
    89→function currentPeriod(): string {
    90→  const d = new Date();
    91→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}`;
    92→}

   720→
   721→  // Sort: running → pending → scheduled → completed, then priority
   722→  const order: Record<TaskStatus, number> = { running: 0, pending: 1, scheduled: 2, failed: 3, completed: 4 };
   723→  tasks.sort((a, b) => {
   724→    if (order[a.status] !== order[b.status]) return order[a.status] - order[b.status];
   725→    return priorityRank[a.priority] - priorityRank[b.priority];
   726→  });
   727→
   728→  return tasks;

   812→export function parseCommand(text: string, cfo: CFOResponse): CommandIntent {
   813→  const lower = text.toLowerCase().trim();
   814→  let matched: { intent: CommandIntentType; phrases: string[]; ack: string } | null = null;
   815→
   816→  for (const rule of COMMAND_RULES) {
   817→    const hits = rule.phrases.filter((p) => lower.includes(p));
   818→    if (hits.length > 0) {
   819→      if (!matched || hits.length > matched.phrases.length) {
   820→        matched = { intent: rule.intent, phrases: hits, ack: rule.ack };
   821→      }
   822→    }
   823→  }
   824→
   825→  if (!matched) {
   826→    return {
   827→      intent: 'unknown',
   828→      confidence: 0.2,
   829→      rawText: text,
   830→      matchedPhrases: [],
   831→      generatedTaskPlan: [],
   832→      spokenAck:
   833→        "I'll handle that. I've logged the request and routed it to the right agent — give me a moment and I'll report back.",
   834→    };
   835→  }
   836→
   837→  const plan = generateTaskPlanForIntent(matched.intent, cfo);
   838→  const confidence = Math.min(0.95, 0.6 + matched.phrases.length * 0.15);
   839→
   840→  return {
   841→    intent: matched.intent,
   842→    confidence,
   843→    rawText: text,
   844→    matchedPhrases: matched.phrases,
   845→    generatedTaskPlan: plan,
   846→    spokenAck: matched.ack,
   847→  };
   848→}

  1163→export function buildDailyCEOBrief(cfo: CFOResponse, user?: { name?: string } | null): DailyCEOBrief {

  1169→    weekday: 'long',

  1174→
  1175→  const overallRisk = cfo.dashboard.healthScore.overall;

  1177→  const riskLabel =

  1190→    rank: i + 1,
  1191→    text: a.title,
  1192→    amount: a.amount,
  1193→    agentId: agentFor[a.actionType] || 'finance-agent',
  1194→  }));
  1195→
  1196→  // Ensure at least 1 fallback action so the brief is never empty
  1197→  if (priorityActions.length === 0) {
  1198→    priorityActions.push({
  1199→      rank: 1,
  1200→      text: 'No urgent actions today — business is running smoothly.',
  1201→      agentId: 'compliance-agent',

  1204→
  1205→  return {

  1213→      cash: cfo.dashboard.cash.currentBalance,
  1214→      gst: cfo.dashboard.gst.liability,
  1215→    },
  1216→    riskLevel,
  1217→    riskLabel,
  1218→    priorityActions,

  1225→export function buildDelegationPlan(rawText: string, cfo: CFOResponse): DelegationPlan {
  1226→  const intent = parseCommand(rawText, cfo);
  1227→  const tasks = intent.generatedTaskPlan;
  1228→
  1229→  // Decide execution mode
  1230→  let executionMode: 'now' | 'scheduled' | 'queued' = 'now';
  1231→  let scheduledFor: string | undefined;
  1232→  const lower = rawText.toLowerCase();
  1233→
  1234→  if (lower.includes('schedule') || lower.includes('tomorrow') || lower.includes('monday') || lower.includes('next week')) {
  1235→    executionMode = 'scheduled';
  1236→    scheduledFor = daysAhead(1);
  1237→  } else if (intent.intent === 'unknown') {
  1238→    executionMode = 'queued';
  1239→  }
  1240→
  1241→  const understood =
  1242→    intent.intent === 'unknown'
  1243→      ? `Understood — I'll route "${rawText}" to the right agent and report back.`
  1244→      : `Understood — I'll ${intent.intent.replace(/_/g, ' ')} now.`;
  1245→
  1246→  const ack =
  1247→    executionMode === 'scheduled'
  1248→      ? `I've scheduled this for ${new Date(scheduledFor!).toLocaleDateString('en-IN')}. The Reporting Agent will own it.`
  1249→      : executionMode === 'queued'
  1250→      ? "I've queued the request. The Orchestrator will pick it up in the next cycle."
  1251→      : intent.spokenAck;
  1252→
  1253→  return {
  1254→    request: rawText,
  1255→    understood,
  1256→    intent: intent.intent,
  1257→    tasks,
  1258→    scheduledFor,
  1259→    executionMode,
  1260→    ack,
  1261→  };
  1262→}
