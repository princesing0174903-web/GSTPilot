   126→function getDaysRemaining(dueDateStr: string): number {
   127→  const dueDate = new Date(dueDateStr);
   128→  const today = new Date('2025-07-08');
   129→  today.setHours(0, 0, 0, 0);
   130→  dueDate.setHours(0, 0, 0, 0);
   131→  return Math.ceil((dueDate.getTime() - today.getTime()) / (1000 * 60 * 60 * 24));
   132→}
   133→
   134→function formatDaysRemaining(days: number): string {
   135→  if (days < 0) return `${Math.abs(days)}d overdue`;
   136→  if (days === 0) return 'Due today';
   137→  if (days === 1) return 'Tomorrow';
   138→  return `${days}d left`;
   139→}
   140→
   141→function getInitials(name: string): string {
   142→  return name
   143→    .split(' ')
   144→    .map(w => w[0])
   145→    .join('')
   146→    .slice(0, 2)
   147→    .toUpperCase();
   148→}

   183→// Map Issue category to BlockingIssue category
   184→function mapIssueCategory(issue: Issue): BlockingIssue['category'] {
   185→  const cat = issue.category?.toLowerCase() ?? '';
   186→  if (cat.includes('gstin')) return 'gstin_error';
   187→  if (cat.includes('missing') || cat.includes('invoice')) return 'missing_invoice';
   188→  if (cat.includes('recon') || cat.includes('mismatch')) return 'recon_mismatch';
   189→  return 'validation_failure';
   190→}
