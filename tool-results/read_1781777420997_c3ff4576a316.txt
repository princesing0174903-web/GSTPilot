   714→// BACKEND INTEGRATIONS (Task BACKEND-INTEGRATIONS-1)
   715→// Real adapter + sync architecture for GSTN, Gmail, Drive, WhatsApp, Banks,
   716→// Razorpay, Excel, Clients. Connection stores encrypted credentials + sync
   717→// state per provider. SyncJob tracks every sync run. Real data flows into
   718→// the domain tables (Invoice / GSTRFiling / Notice / Payment / BankTransaction
   719→// / WhatsAppMessage / GmailMessage / DriveFile) when credentials are present.
   720→// All adapters throw IntegrationNotConfiguredError when creds are missing —
   721→// NO fake data is ever written.
   722→// ═══════════════════════════════════════════════════════════════════════════════
   723→
   724→model Connection {
   725→  id              String   @id @default(cuid())
   726→  firmId          String?
   727→  provider        String   // "gstn" | "gmail" | "drive" | "whatsapp" | "banks" | "razorpay" | "excel" | "clients"
   728→  label           String?  // optional display name, e.g. "HDFC Current A/C"
   729→  status          String   @default("disconnected") // "disconnected" | "connecting" | "connected" | "error"
   730→  // Encrypted JSON blob. Structure depends on provider:
   731→  //   gstn     → { gspCode, clientId, clientSecret, gstinUsername, authToken, refreshToken, tokenExpiresAt }
   732→  //   gmail    → { accessToken, refreshToken, tokenExpiresAt, emailAddress, scope }
   733→  //   drive    → { accessToken, refreshToken, tokenExpiresAt, rootFolderId }
   734→  //   whatsapp → { phoneNumberId, wabaId, accessToken, verifyToken, webhookSecret }
   735→  //   banks    → { bankName, accountNumber, ifsc, apiKey, apiSecret, aggregatorName }
   736→  //   razorpay → { keyId, keySecret, webhookSecret, accountId }
   737→  //   excel    → { } (no creds; local only)
   738→  //   clients  → { } (no creds; internal)
   739→  credentialsEnc  String?
   740→  metadata        String?  // JSON string for provider-specific extra config (filters, folders, etc.)
   741→  lastSyncAt      DateTime?
   742→  lastSyncStatus  String?  // "success" | "error" | "partial"
   743→  lastSyncError   String?
   744→  lastSyncSummary String?  // short summary, e.g. "Pulled 12 invoices, 3 notices"
   745→  createdAt       DateTime @default(now())
   746→  updatedAt       DateTime @updatedAt
   747→
   748→  syncJobs         SyncJob[]
   749→  payments         Payment[]         @relation("ConnectionPayments")
   750→  bankTransactions BankTransaction[]
   751→  whatsappMessages WhatsAppMessage[]
   752→  gmailMessages    GmailMessage[]
   753→  driveFiles       DriveFile[]
   754→}
