   144→function categorise(description: string, type: TxnType, amount: number): TxnCategory {
   145→  const d = description.toLowerCase();
   146→  if (d.includes('salary') || d.includes('payroll')) return 'payroll';
   147→  if (d.includes('gst') || d.includes('tds') || d.includes('tax')) return 'tax';
   148→  if (d.includes('rent')) return 'rent';
   149→  if (d.includes('logistics') || d.includes('freight')) return 'logistics';
   150→  if (d.includes('purchase')) return 'purchase';
   151→  if (d.includes('interest')) return 'interest';
   152→  if (d.includes('refund')) return 'refund';
   153→  if (d.includes('charges') || d.includes('fee') || amount < 0 && amount > -2000) return 'fee';
   154→  if (d.includes('transfer') || d.includes('neft') && type === 'debit' && amount > -50000) return 'transfer';
   155→  if (type === 'credit' && amount > 10000) return 'revenue';
   156→  return 'uncategorised';
   157→}

   398→  const todayStart = new Date();
   399→  todayStart.setHours(0, 0, 0, 0);

   404→      where: { accountId: r.id, date: { gte: todayStart } },
   405→    });
   406→    out.push({
   407→      id: r.id,
   408→      bankName: r.bankName,
   409→      accountMasked: r.accountMasked,
   410→      accountType: r.accountType as BankAccountSummary['accountType'],
   411→      ifsc: r.ifsc ?? undefined,
   412→      currentBalance: r.balance,
   413→      availableBalance: r.availableBalance,
   414→      overdraftLimit: r.overdraftLimit,
   415→      upiHandle: r.upiHandle ?? undefined,
   416→      aaConsent: r.aaConsent,
   417→      aaConsentExpiry: r.aaConsentExpiry?.toISOString() ?? undefined,
   418→      status: r.status as BankAccountSummary['status'],
   419→      lastSyncAt: r.lastSyncAt?.toISOString() ?? nowISO(),
   420→      recordsToday,
   421→    });
   422→  }
   423→  return out;
   424→}

   428→export async function buildStatementsState(
   429→  accounts: BankAccountSummary[],
   430→  limit = 12,
   431→): Promise<BankStatementSummary> {
   432→  await ensureSeedData();
   433→  const accMap = new Map(accounts.map(a => [a.id, a]));
   434→  const rows = await db.bankTransaction.findMany({
   435→    orderBy: { date: 'desc' },
   436→    take: limit,
   437→  });
   438→  const totalTransactions = await db.bankTransaction.count();
   439→  const todayStart = new Date();
   440→  todayStart.setHours(0, 0, 0, 0);
   441→  const last7d = new Date();
   442→  last7d.setDate(last7d.getDate() - 7);
   443→  const last30d = new Date();
   444→  last30d.setDate(last30d.getDate() - 30);
   445→
   446→  const [todayCount, last7dCount, last30dCount] = await Promise.all([
   447→    db.bankTransaction.count({ where: { date: { gte: todayStart } } }),
   448→    db.bankTransaction.count({ where: { date: { gte: last7d } } }),
   449→    db.bankTransaction.count({ where: { date: { gte: last30d } } }),
   450→  ]);
   451→
   452→  const last30dTxns = await db.bankTransaction.findMany({
   453→    where: { date: { gte: last30d } },
   454→    select: { amount: true, category: true, type: true },
   455→  });
   456→  const inflow30d = last30dTxns.filter(t => t.amount > 0).reduce((s, t) => s + t.amount, 0);
   457→  const outflow30d = last30dTxns.filter(t => t.amount < 0).reduce((s, t) => s + Math.abs(t.amount), 0);
   458→  const categorised = last30dTxns.filter(t => t.category !== 'uncategorised').length;
   459→  const categorisationPct = last30dTxns.length > 0
   460→    ? Math.round((categorised / last30dTxns.length) * 100)
   461→    : 0;
   462→
   463→  const recentTransactions: BankTransactionRow[] = rows.map(r => {
   464→    const acc = accMap.get(r.accountId);
   465→    return {
   466→      id: r.id,
   467→      accountId: r.accountId,
   468→      bankName: acc?.bankName ?? '—',
   469→      accountMasked: acc?.accountMasked ?? '—',
   470→      date: r.date.toISOString(),
   471→      description: r.description,
   472→      amount: r.amount,
   473→      type: r.type as TxnType,
   474→      category: r.category as TxnCategory,
   475→      referenceNo: r.referenceNo ?? undefined,
   476→      counterparty: r.counterparty ?? undefined,
   477→      matched: r.matched,
   478→      matchedInvoice: r.matchedInvoice ?? undefined,
   479→      matchedParty: r.matchedParty ?? undefined,
   480→      matchConfidence: r.matchConfidence,
   481→    };
   482→  });
   483→
   484→  return {
   485→    totalTransactions,

   534→// ─── Module 4: Cash Flow Engine™ ──────────────────────────────────────────────
   535→
   536→export async function buildCashFlowState(accounts: BankAccountSummary[]): Promise<CashFlowSummary> {
   537→  await ensureSeedData();
   538→  const cashPosition = accounts.reduce((s, a) => s + a.availableBalance, 0);
   539→
   540→  // Daily points (last 30 days)
   541→  const dailyRows = await db.cashForecast.findMany({
   542→    where: { horizon: 'daily' },
   543→    orderBy: { date: 'asc' },
   544→  });
   545→  const forecast7dRows = await db.cashForecast.findMany({
   546→    where: { horizon: '7d' },
   547→    orderBy: { date: 'asc' },
   548→  });
   549→  const forecast30dRows = await db.cashForecast.findMany({
   550→    where: { horizon: '30d' },
   551→    orderBy: { date: 'asc' },
   552→  });
   553→
   554→  const toPoint = (r: typeof dailyRows[number]): CashFlowDailyPoint => ({
   555→    date: r.date.toISOString(),
   556→    opening: r.openingBalance,
   557→    inflows: r.inflows,
   558→    outflows: r.outflows,
   559→    closing: r.closingBalance,
   560→    shortage: r.shortage,
   561→    shortageAmount: r.shortageAmount,
   562→  });
   563→
   564→  const daily = dailyRows.map(toPoint);
   565→  const forecast7d = forecast7dRows.map(toPoint);
   566→  const forecast30d = forecast30dRows.map(toPoint);
   567→
   568→  // Burns (avg outflow over last 7 actual days)
   569→  const last7 = daily.slice(-7);
   570→  const dailyBurn = last7.length > 0 ? last7.reduce((s, d) => s + d.outflows, 0) / last7.length : 0;
   571→  const monthlyBurn = dailyBurn * 30;
   572→
   573→  // Runway (cash / daily burn, capped)
   574→  const runwayDays = dailyBurn > 0 ? Math.min(730, Math.floor(cashPosition / dailyBurn)) : 730;
   575→
   576→  // Expected collections/payments next 7 days (from forecast)
   577→  const next7 = forecast7d;
   578→  const expectedCollections = next7.reduce((s, d) => s + d.inflows, 0);
   579→  const expectedPayments = next7.reduce((s, d) => s + d.outflows, 0);
   580→
   581→  // Shortage detection
   582→  const shortagePoint = next7.find(d => d.shortage);
   583→  const shortageDetected = !!shortagePoint;
   584→  const shortageAmount = shortagePoint?.shortageAmount ?? 0;
   585→  const shortageDate = shortagePoint?.date;
   586→
   587→  // Cash position change vs 30 days ago
   588→  const opening30Ago = daily[0]?.opening ?? cashPosition;
   589→  const cashPositionChangePct = opening30Ago > 0
   590→    ? ((cashPosition - opening30Ago) / opening30Ago) * 100
   591→    : 0;
   592→
   593→  return {
   594→    cashPosition,
   595→    cashPositionChangePct,
   596→    dailyBurn,
   597→    monthlyBurn,
   598→    runwayDays,
   599→    expectedCollections,
   600→    expectedPayments,
   601→    shortageDetected,
   602→    shortageAmount,
   603→    shortageDate,
   604→    daily,
   605→    forecast7d,
   606→    forecast30d,
   607→  };
   608→}

   610→// ─── Module 5: Auto Reconciliation Engine™ ────────────────────────────────────
   611→
   612→export async function buildReconciliationState(limit = 20): Promise<ReconciliationSummary> {
   613→  await ensureSeedData();
   614→  const rows = await db.bankReconciliation.findMany({
   615→    orderBy: { at: 'desc' },
   616→    take: limit,
   617→  });
   618→
   619→  const entries: ReconciliationEntry[] = rows.map(r => ({
   620→    id: r.id,
   621→    bankRef: r.bankRef,
   622→    bankAmount: r.bankAmount,
   623→    matchedInvoice: r.matchedInvoice ?? undefined,
   624→    matchedTo: r.matchedTo ?? undefined,
   625→    status: r.status as ReconStatus,
   626→    mismatchType: (r.mismatchType as MismatchType | null) ?? undefined,
   627→    confidencePct: r.confidencePct,
   628→    suggestedAction: r.suggestedAction ?? undefined,
   629→    at: r.at.toISOString(),
   630→  }));
   631→
   632→  const matched = entries.filter(e => e.status === 'matched');
   633→  const unmatched = entries.filter(e => e.status === 'unmatched');
   634→  const pending = entries.filter(e => e.status === 'pending');
   635→  const duplicate = entries.filter(e => e.status === 'duplicate');
   636→  const partial = entries.filter(e => e.status === 'partial');
   637→  const matchedPct = entries.length > 0 ? Math.round((matched.length / entries.length) * 100) : 0;
   638→  const matchedAmount = matched.reduce((s, e) => s + e.bankAmount, 0);
   639→  const unmatchedAmount = unmatched.reduce((s, e) => s + e.bankAmount, 0)
   640→    + partial.reduce((s, e) => s + e.bankAmount, 0);
   641→
   642→  // Risk score: weighted by unmatched/partial + duplicate
   643→  const totalUnmatched = unmatched.length + partial.length + duplicate.length;
   644→  const riskScore = entries.length > 0
   645→    ? clamp(Math.round((totalUnmatched / entries.length) * 100), 0, 100)
   646→    : 0;
   647→  const riskLevel = riskFromScore(riskScore);
   648→
   649→  return {
   650→    totalTransactions: entries.length,
   651→    matched: matched.length,
   652→    unmatched: unmatched.length,
   653→    pending: pending.length,
   654→    duplicate: duplicate.length,
   655→    partial: partial.length,
   656→    matchedPct,
   657→    matchedAmount,
   658→    unmatchedAmount,
   659→    pendingCollections: partial.length,
   660→    pendingPayments: pending.length,
   661→    riskScore,
   662→    riskLevel,
   663→    entries,
   664→  };
   665→}

   669→export async function buildUPIState(limit = 12): Promise<UPISummary> {
   670→  await ensureSeedData();
   671→  const rows = await db.uPITransaction.findMany({
   672→    orderBy: { date: 'desc' },
   673→    take: limit,
   674→  });
   675→  const totalTransactions = await db.uPITransaction.count();
   676→  const todayStart = new Date();
   677→  todayStart.setHours(0, 0, 0, 0);
   678→  const last30d = new Date();
   679→  last30d.setDate(last30d.getDate() - 30);
   680→
   681→  const [todayCount, last30dRows] = await Promise.all([
   682→    db.uPITransaction.count({ where: { date: { gte: todayStart } } }),
   683→    db.uPITransaction.findMany({ where: { date: { gte: last30d } }, select: { amount: true, upiId: true, vpaCounterparty: true, status: true } }),
   684→  ]);
   685→
   686→  const collections30d = last30dRows.filter(t => t.amount > 0).reduce((s, t) => s + t.amount, 0);
   687→  const payments30d = last30dRows.filter(t => t.amount < 0).reduce((s, t) => s + Math.abs(t.amount), 0);
   688→  const pendingSettlements = last30dRows.filter(t => t.status === 'pending').length;
   689→  const pendingSettlementAmount = last30dRows.filter(t => t.status === 'pending' && t.amount > 0).reduce((s, t) => s + t.amount, 0);
   690→
   691→  // Top customers (by total collection)
   692→  const byVpa = new Map<string, { name: string; total: number; count: number }>();
   693→  for (const t of last30dRows) {
   694→    if (t.amount <= 0) continue;
   695→    const key = t.upiId;
   696→    const name = t.vpaCounterparty ?? t.upiId.split('@')[0];
   697→    const existing = byVpa.get(key) ?? { name, total: 0, count: 0 };
   698→    existing.total += t.amount;
   699→    existing.count += 1;
   700→    byVpa.set(key, existing);
   701→  }
   702→  const topCustomers = Array.from(byVpa.entries())
   703→    .map(([vpa, v]) => ({ vpa, name: v.name, total: v.total, count: v.count }))
   704→    .sort((a, b) => b.total - a.total)
   705→    .slice(0, 6);
   706→
   707→  const transactions: UPITransactionRow[] = rows.map(r => ({
   708→    id: r.id,
   709→    upiId: r.upiId,
   710→    vpaCounterparty: r.vpaCounterparty ?? undefined,
   711→    date: r.date.toISOString(),
   712→    amount: r.amount,
   713→    type: r.type as TxnType,
   714→    referenceNo: r.referenceNo ?? undefined,
   715→    notes: r.notes ?? undefined,
   716→    matched: r.matched,
   717→    matchedInvoice: r.matchedInvoice ?? undefined,
   718→    matchedParty: r.matchedParty ?? undefined,
   719→    status: r.status as UPITransactionRow['status'],
   720→  }));
   721→
   722→  return {
   723→    totalTransactions,
   724→    todayCount,
   725→    collections30d,
   726→    payments30d,
   727→    pendingSettlements,
   728→    pendingSettlementAmount,
   729→    topCustomers,
   730→    transactions,
   731→  };
   732→}

  1343→export async function dispatchBankAction(req: BankActionRequest): Promise<BankActionResponse> {
  1344→  switch (req.module) {
  1345→    case 'accounts':
  1346→      if (req.action === 'connect') return connectBankAccount(req.payload as { bankName?: string; accountType?: string });
  1347→      if (req.action === 'sync') return syncBankAccounts((req.payload?.accountId as string) || undefined);
  1348→      break;
  1349→    case 'statements':
  1350→      if (req.action === 'sync') return syncStatements();
  1351→      break;
  1352→    case 'aa':
  1353→      if (req.action === 'connect') return connectAA(req.payload as { aaName?: string; mobile?: string });
  1354→      if (req.action === 'consent') return consentAA(req.payload as { connectionId?: string; action?: 'approve' | 'reject' });
  1355→      break;
  1356→    case 'reconciliation':
  1357→      if (req.action === 'run' || req.action === 'reconcile') return runReconciliation();
  1358→      break;
  1359→    case 'upi':
  1360→      if (req.action === 'connect') return connectUPI();
  1361→      if (req.action === 'sync') return syncUPI();
  1362→      break;
  1363→    case 'collections':
  1364→      if (req.action === 'recover') return recoverCollections();
  1365→      if (req.action === 'remind') return remindCollections();
  1366→      break;
  1367→    case 'transactions':
  1368→    case 'cashflow':
  1369→      // Read-only modules — no action needed
  1370→      break;
  1371→  }
  1372→  return {
  1373→    ok: false,
  1374→    module: req.module,
  1375→    action: req.action,
  1376→    message: `Unknown action ${req.action} for module ${req.module}`,
  1377→    oracleAck: `I couldn't perform that action.`,
  1378→  };
  1379→}
