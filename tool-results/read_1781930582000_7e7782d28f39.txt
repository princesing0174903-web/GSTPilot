   102→
   103→export function OracleWorkspace({
   104→  open, onClose, onNavigate, userName, firmName, gstin, initialPrompt,
   105→}: OracleWorkspaceProps) {
   106→  const [store, setStore] = useState<ConversationStore>({ conversations: [], activeId: null });
   107→  // Portal guard: only render into document.body after mount to avoid SSR
   108→  // hydration mismatch (server has no document.body).
   109→  const [mounted, setMounted] = useState(false);
   110→  useEffect(() => setMounted(true), []);
   111→  const [messages, setMessages] = useState<OracleMessage[]>([]);
   112→  const [activeId, setActiveId] = useState<string | null>(null);
   113→  const [input, setInput] = useState('');
   114→  const [isStreaming, setIsStreaming] = useState(false);
   115→  const [lastEmotion, setLastEmotion] = useState<OracleMessage['emotion']>('helpful');
   116→  const [activeLanguage, setActiveLanguage] = useState<OracleMessage['language']>('english');
   117→  // Mobile: history drawer open state
   118→  const [historyOpen, setHistoryOpen] = useState(false);
   119→
   120→  const scrollRef = useRef<HTMLDivElement>(null);
   121→  const inputRef = useRef<HTMLTextAreaElement>(null);
   122→  const abortRef = useRef<AbortController | null>(null);
   123→  const userPinnedUpRef = useRef(false);
   124→  const streamingIdRef = useRef<string | null>(null);
   125→  // Tracks the last initialPrompt we auto-sent, so we never fire it twice.
   126→  const initialPromptSentRef = useRef<string | null>(null);
   127→
   128→  // ─── Load + migrate store on open ──────────────────────────────────────────
   129→  useEffect(() => {
   130→    if (!open) return;
   131→    const loaded = loadStore();
   132→    if (loaded.conversations.length === 0) {
   133→      // Migrate legacy single-conversation store if present.
   134→      try {
   135→        const raw = localStorage.getItem(LEGACY_KEY);
   136→        if (raw) {
   137→          const parsed = JSON.parse(raw) as OracleMessage[];
   138→          if (Array.isArray(parsed) && parsed.length > 0) {
   139→            const now = new Date().toISOString();
   140→            const conv: OracleConversation = {
   141→              id: cryptoId(),
   142→              title: deriveTitle(parsed),
   143→              messages: parsed.map((m) => (m.streaming ? { ...m, streaming: false } : m)),
   144→              createdAt: parsed[0]?.createdAt ?? now,
   145→              updatedAt: now,
   146→            };
   147→            const next: ConversationStore = { conversations: [conv], activeId: conv.id };
   148→            setStore(next);
   149→            setActiveId(conv.id);
   150→            setMessages(conv.messages);
   151→            saveStore(next);
   152→            return;
   153→          }
   154→        }
   155→      } catch {
   156→        /* ignore */
   157→      }
   158→      // Nothing to migrate — start fresh with one empty conversation.
   159→      const conv = newConversation();
   160→      const next: ConversationStore = { conversations: [conv], activeId: conv.id };
   161→      setStore(next);
   162→      setActiveId(conv.id);
   163→      setMessages([]);
   164→      saveStore(next);
   165→      return;
   166→    }
   167→    setStore(loaded);
   168→    const active = loaded.conversations.find((c) => c.id === loaded.activeId) ?? loaded.conversations[0];
   169→    if (active) {
   170→      setActiveId(active.id);
   171→      setMessages(active.messages);
   172→    }
   173→  }, [open]);
   174→
   175→  // ─── Persist active conversation whenever messages change ───────────────────
   176→  useEffect(() => {
   177→    if (!open || !activeId) return;
   178→    setStore((prev) => {
   179→      if (!prev.conversations.length) return prev;
   180→      const exists = prev.conversations.some((c) => c.id === activeId);
   181→      let conversations: OracleConversation[];
   182→      let nextActiveId = prev.activeId;
   183→      if (exists) {
   184→        conversations = prev.conversations.map((c) =>
   185→          c.id === activeId
   186→            ? {
   187→                ...c,
   188→                messages,
   189→                title: deriveTitle(messages) || c.title,
   190→                updatedAt: new Date().toISOString(),
   191→              }
   192→            : c,
   193→        );
   194→      } else {
   195→        // Active conversation doesn't exist yet — create it on the fly.
   196→        const conv: OracleConversation = {
   197→          id: activeId,
   198→          title: deriveTitle(messages) || 'New chat',
   199→          messages,
   200→          createdAt: new Date().toISOString(),
   201→          updatedAt: new Date().toISOString(),
   202→        };
   203→        conversations = [conv, ...prev.conversations];
   204→        nextActiveId = activeId;
   205→      }
   206→      const next = { conversations, activeId: nextActiveId };
   207→      saveStore(next);
   208→      return next;
   209→    });
   210→  }, [messages, open, activeId]);
   211→
   212→  // ─── Body scroll lock while open ───────────────────────────────────────────
   213→  useEffect(() => {
   214→    if (!open) return;
   215→    const prev = document.body.style.overflow;
   216→    document.body.style.overflow = 'hidden';
   217→    return () => {
   218→      document.body.style.overflow = prev;
   219→    };
   220→  }, [open]);
   221→
   222→  // ─── Escape to close ───────────────────────────────────────────────────────
   223→  useEffect(() => {
   224→    if (!open) return;
   225→    const onKey = (e: globalThis.KeyboardEvent) => {
   226→      if (e.key === 'Escape' && !isStreaming) onClose();
   227→    };
   228→    window.addEventListener('keydown', onKey);
   229→    return () => window.removeEventListener('keydown', onKey);
   230→  }, [open, isStreaming, onClose]);
   231→
   232→  // ─── Smart auto-scroll ─────────────────────────────────────────────────────
   233→  const handleScroll = useCallback(() => {
   234→    const el = scrollRef.current;
   235→    if (!el) return;
   236→    const distanceFromBottom = el.scrollHeight - el.scrollTop - el.clientHeight;
   237→    userPinnedUpRef.current = distanceFromBottom > 120;
   238→  }, []);
   239→
   240→  const scrollToBottom = useCallback((smooth = false) => {
   241→    const el = scrollRef.current;
   242→    if (!el) return;
   243→    el.scrollTo({ top: el.scrollHeight, behavior: smooth ? 'smooth' : 'auto' });
   244→  }, []);
   245→
   246→  useEffect(() => {
   247→    if (!userPinnedUpRef.current) {
   248→      requestAnimationFrame(() => scrollToBottom(false));
   249→    }
   250→  }, [messages, scrollToBottom]);
   251→
   252→  // ─── Conversation actions ──────────────────────────────────────────────────
   253→  const handleNewChat = useCallback(() => {
   254→    if (isStreaming) return;
   255→    const conv = newConversation();
   256→    setStore((prev) => {
   257→      const next = { conversations: [conv, ...prev.conversations], activeId: conv.id };
   258→      saveStore(next);
   259→      return next;
   260→    });
   261→    setActiveId(conv.id);
   262→    setMessages([]);
   263→    setInput('');
   264→    userPinnedUpRef.current = false;
   265→    setHistoryOpen(false);
   266→    requestAnimationFrame(() => inputRef.current?.focus());
   267→  }, [isStreaming]);
   268→
   269→  const handleSwitchConversation = useCallback(
   270→    (id: string) => {
   271→      if (isStreaming) return;
   272→      const conv = store.conversations.find((c) => c.id === id);
   273→      if (!conv) return;
   274→      setActiveId(id);
   275→      setMessages(conv.messages);
   276→      setStore((prev) => {
   277→        const next = { ...prev, activeId: id };
   278→        saveStore(next);
   279→        return next;
   280→      });
   281→      setHistoryOpen(false);
   282→      userPinnedUpRef.current = false;
   283→      requestAnimationFrame(() => scrollToBottom(false));
   284→    },
   285→    [isStreaming, store.conversations, scrollToBottom],
   286→  );
   287→
   288→  const handleDeleteConversation = useCallback(
   289→    (id: string) => {
   290→      if (isStreaming) return;
   291→      setStore((prev) => {
   292→        const filtered = prev.conversations.filter((c) => c.id !== id);
   293→        let nextActive = prev.activeId;
   294→        let nextMessages = messages;
   295→        if (prev.activeId === id) {
   296→          if (filtered.length > 0) {
   297→            nextActive = filtered[0].id;
   298→            nextMessages = filtered[0].messages;
   299→          } else {
   300→            const conv = newConversation();
   301→            filtered.unshift(conv);
   302→            nextActive = conv.id;
   303→            nextMessages = [];
   304→          }
   305→          setActiveId(nextActive);
   306→          setMessages(nextMessages);
   307→        }
   308→        const next = { conversations: filtered, activeId: nextActive };
   309→        saveStore(next);
   310→        return next;
   311→      });
   312→    },
   313→    [isStreaming, messages],
   314→  );
   315→
   316→  // ─── Send flow (preserved from previous implementation) ─────────────────────
   317→  const sendMessage = useCallback(
   318→    async (raw: string) => {
   319→      const text = raw.trim();
   320→      if (!text || isStreaming) return;
   321→
   322→      // ── Brand-question short-circuit ───────────────────────────────────────
   323→      const brand = detectBrandQuestion(text);
   324→      if (brand.matched && brand.answer) {
   325→        const userMsg: OracleMessage = {
   326→          id: cryptoId(),
   327→          role: 'user',
   328→          content: text,
   329→          language: detectLanguage(text),
   330→          createdAt: new Date().toISOString(),
   331→        };
   332→        const oracleMsg: OracleMessage = {
   333→          id: cryptoId(),
   334→          role: 'oracle',
   335→          content: brand.answer,
   336→          language: detectLanguage(text),
   337→          emotion: 'success',
   338→          createdAt: new Date().toISOString(),
   339→          streaming: false,
   340→          followUps: ['Who founded GSTPilot?', 'What can Oracle do?', 'GST kya hota hai?'],
   341→        };
   342→        setMessages((prev) => [...prev, userMsg, oracleMsg]);
   343→        setLastEmotion('success');
   344→        setActiveLanguage(detectLanguage(text));
   345→        setInput('');
   346→        if (inputRef.current) inputRef.current.style.height = 'auto';
   347→        userPinnedUpRef.current = false;
   348→        requestAnimationFrame(() => scrollToBottom(true));
   349→        return;
   350→      }
   351→
   352→      const userMsg: OracleMessage = {
   353→        id: cryptoId(),
   354→        role: 'user',
   355→        content: text,
   356→        language: detectLanguage(text),
   357→        createdAt: new Date().toISOString(),
   358→      };
   359→      const oracleId = cryptoId();
   360→      const oraclePlaceholder: OracleMessage = {
   361→        id: oracleId,
   362→        role: 'oracle',
   363→        content: '',
   364→        language: userMsg.language,
   365→        createdAt: new Date().toISOString(),
   366→        streaming: true,
   367→      };
   368→      setMessages((prev) => [...prev, userMsg, oraclePlaceholder]);
   369→      streamingIdRef.current = oracleId;
   370→      setIsStreaming(true);
   371→      setActiveLanguage(userMsg.language);
   372→      setInput('');
   373→      userPinnedUpRef.current = false;
   374→      if (inputRef.current) inputRef.current.style.height = 'auto';
   375→
   376→      const history: OracleChatRequest['messages'] = [
   377→        ...messages
   378→          .filter((m) => m.content.trim().length > 0)
   379→          .map((m) => ({ role: m.role, content: m.content })),
   380→        { role: 'user', content: text },
   381→      ];
   382→
   383→      const payload: OracleChatRequest = {
   384→        messages: history,
   385→        memory: {
   386→          userName,
   387→          firmName,
   388→          gstin,
   389→          preferredLanguage: activeLanguage,
   390→          recentTopics: messages
   391→            .filter((m) => m.role === 'user')
   392→            .slice(-4)
   393→            .map((m) => m.content.slice(0, 60)),
   394→        },
   395→      };
   396→
   397→      const controller = new AbortController();
   398→      abortRef.current = controller;
   399→
   400→      try {
   401→        const res = await fetch('/api/oracle/chat', {
   402→          method: 'POST',
   403→          headers: { 'Content-Type': 'application/json' },
   404→          body: JSON.stringify(payload),
   405→          signal: controller.signal,
   406→        });
   407→        if (!res.ok || !res.body) {
   408→          throw new Error(`Request failed (${res.status})`);
   409→        }
   410→
   411→        const reader = res.body.getReader();
   412→        const decoder = new TextDecoder();
   413→        let buffer = '';
   414→        let acc = '';
   415→
   416→        while (true) {
   417→          const { done, value } = await reader.read();
   418→          if (done) break;
   419→          buffer += decoder.decode(value, { stream: true });
   420→
   421→          const lines = buffer.split('\n');
   422→          buffer = lines.pop() ?? '';
   423→
   424→          for (const rawLine of lines) {
   425→            const line = rawLine.trim();
   426→            if (!line || !line.startsWith('data:')) continue;
   427→            const data = line.slice(5).trim();
   428→            if (!data || data === '[DONE]') continue;
   429→            let chunk: OracleStreamChunk;
   430→            try {
   431→              chunk = JSON.parse(data) as OracleStreamChunk;
   432→            } catch {
   433→              continue;
   434→            }
   435→            if (chunk.language) setActiveLanguage(chunk.language);
   436→            if (chunk.token) {
   437→              acc += chunk.token;
   438→              setMessages((prev) =>
   439→                prev.map((m) => (m.id === oracleId ? { ...m, content: acc } : m)),
   440→              );
   441→            }
   442→            if (chunk.done) {
   443→              setMessages((prev) =>
   444→                prev.map((m) =>
   445→                  m.id === oracleId
   446→                    ? {
   447→                        ...m,
   448→                        content: acc,
   449→                        streaming: false,
   450→                        emotion: detectEmotion(acc),
   451→                        followUps: buildFollowUps(acc),
   452→                      }
   453→                    : m,
   454→                ),
   455→              );
   456→              setLastEmotion(detectEmotion(acc));
   457→            }
   458→            if (chunk.error) {
   459→              setMessages((prev) =>
   460→                prev.map((m) =>
   461→                  m.id === oracleId
   462→                    ? {
   463→                        ...m,
   464→                        content:
   465→                          acc ||
   466→                          "I ran into a temporary issue reaching my reasoning service. Please try that again — your conversation is safe.",
   467→                        streaming: false,
   468→                        emotion: 'warning',
   469→                      }
   470→                    : m,
   471→                ),
   472→              );
   473→              setLastEmotion('warning');
   474→            }
   475→          }
   476→        }
   477→
   478→        setMessages((prev) =>
   479→          prev.map((m) =>
   480→            m.id === oracleId && m.streaming
   481→              ? {
   482→                  ...m,
   483→                  streaming: false,
   484→                  emotion: detectEmotion(m.content),
   485→                  followUps: buildFollowUps(m.content),
   486→                }
   487→              : m,
   488→          ),
   489→        );
   490→      } catch (err) {
   491→        const aborted = err instanceof DOMException && err.name === 'AbortError';
   492→        setMessages((prev) =>
   493→          prev.map((m) =>
   494→            m.id === oracleId
   495→              ? {
   496→                  ...m,
   497→                  streaming: false,
   498→                  content:
   499→                    m.content ||
   500→                    (aborted
   501→                      ? 'Stopped.'
   502→                      : "I had trouble reaching my reasoning service. Please try again in a moment."),
   503→                  emotion: aborted ? m.emotion : 'warning',
   504→                }
   505→              : m,
   506→          ),
   507→        );
   508→        if (!aborted) setLastEmotion('warning');
   509→      } finally {
   510→        setIsStreaming(false);
   511→        streamingIdRef.current = null;
   512→        abortRef.current = null;
   513→        requestAnimationFrame(() => inputRef.current?.focus());
   514→      }
   515→    },
   516→    [isStreaming, messages, userName, firmName, gstin, activeLanguage, scrollToBottom],
   517→  );
   518→
   519→  // ─── Stop streaming ────────────────────────────────────────────────────────
   520→  const handleStop = useCallback(() => {
   521→    abortRef.current?.abort();
   522→  }, []);
   523→
   524→  // ─── Input handling ────────────────────────────────────────────────────────
   525→  const handleInputChange = useCallback((e: React.ChangeEvent<HTMLTextAreaElement>) => {
   526→    setInput(e.target.value);
   527→    const el = e.target;
   528→    el.style.height = 'auto';
   529→    el.style.height = `${Math.min(el.scrollHeight, 160)}px`;
   530→  }, []);
   531→
   532→  const handleKeyDown = useCallback(
   533→    (e: KeyboardEvent<HTMLTextAreaElement>) => {
   534→      if (e.key === 'Enter' && !e.shiftKey) {
   535→        e.preventDefault();
   536→        sendMessage(input);
   537→      }
   538→    },
   539→    [input, sendMessage],
   540→  );
   541→
   542→  // ─── Focus input on open ───────────────────────────────────────────────────
   543→  useEffect(() => {
   544→    if (open) {
   545→      requestAnimationFrame(() => inputRef.current?.focus());
   546→    }
   547→  }, [open]);
   548→
   549→  // ─── Auto-send a prefilled prompt (from the CommandBar / other launchers) ───
   550→  // Waits one tick so the conversation-load effect (also keyed on `open`) has
   551→  // settled messages/activeId before we send. Guarded by a ref so a given
   552→  // prompt is never sent twice.
   553→  useEffect(() => {
   554→    if (!open || !initialPrompt) return;
   555→    if (initialPromptSentRef.current === initialPrompt) return;
   556→    const t = setTimeout(() => {
   557→      initialPromptSentRef.current = initialPrompt;
   558→      sendMessage(initialPrompt);
   559→    }, 60);
   560→    return () => clearTimeout(t);
   561→  }, [open, initialPrompt, sendMessage]);
   562→
   563→  // ─── Rail navigation ───────────────────────────────────────────────────────
   564→  const handleRailClick = useCallback(
   565→    (item: RailItem) => {
   566→      if (item.id === 'oracle') return; // already here
   567→      if (isStreaming) return;
   568→      onNavigate(item.id);
   569→      onClose();
   570→    },
   571→    [isStreaming, onNavigate, onClose],
   572→  );
   573→
   574→  // ─── Derived state ─────────────────────────────────────────────────────────
   575→  const avatarState = useMemo(
   576→    () => deriveAvatarState({ isStreaming, hasInput: input.trim().length > 0, lastEmotion }),
   577→    [isStreaming, input, lastEmotion],
   578→  );
   579→
   580→  const isEmpty = messages.length === 0;
   581→  const grouped = useMemo(() => groupConversations(store.conversations), [store.conversations]);
   582→
   583→  if (!mounted || typeof document === 'undefined') return null;
   584→

  1217→function groupConversations(convos: OracleConversation[]): ConversationGroups {
  1218→  const now = new Date();
  1219→  const startOfToday = new Date(now.getFullYear(), now.getMonth(), now.getDate()).getTime();
  1220→  const startOfYesterday = startOfToday - 86_400_000;
  1221→  const startOf7Days = startOfToday - 7 * 86_400_000;
  1222→
  1223→  const today: OracleConversation[] = [];
  1224→  const yesterday: OracleConversation[] = [];
  1225→  const prev7: OracleConversation[] = [];
  1226→  const older: OracleConversation[] = [];
  1227→
  1228→  const sorted = [...convos].sort(
  1229→    (a, b) => new Date(b.updatedAt).getTime() - new Date(a.updatedAt).getTime(),
  1230→  );
  1231→
  1232→  for (const c of sorted) {
  1233→    const t = new Date(c.updatedAt).getTime();
  1234→    if (isNaN(t)) {
  1235→      older.push(c);
  1236→    } else if (t >= startOfToday) {
  1237→      today.push(c);
  1238→    } else if (t >= startOfYesterday) {
  1239→      yesterday.push(c);
  1240→    } else if (t >= startOf7Days) {
  1241→      prev7.push(c);
  1242→    } else {
  1243→      older.push(c);
  1244→    }
  1245→  }
  1246→
  1247→  return { today, yesterday, prev7, older };
  1248→}
