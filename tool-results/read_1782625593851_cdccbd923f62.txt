    51→    51→function formatINR(n: number): string {
    52→    52→  if (!isFinite(n) || isNaN(n)) return '₹0';
    53→    53→  const abs = Math.abs(n);
    54→    54→  if (abs >= 10000000) return `₹${(n / 10000000).toFixed(2)} Cr`;
    55→    55→  if (abs >= 100000) return `₹${(n / 100000).toFixed(2)} L`;
    56→    56→  if (abs >= 1000) return `₹${(n / 1000).toFixed(1)}K`;
    57→    57→  return '₹' + Math.round(n).toLocaleString('en-IN');
    58→    58→}
    59→    59→
    60→    60→function formatINRFull(n: number): string {
    61→    61→  if (!isFinite(n) || isNaN(n)) return '₹0';
    62→    62→  return '₹' + Math.round(n).toLocaleString('en-IN');
    63→    63→}

   127→   127→function timeAgo(iso: string): string {
   128→   128→  const diff = Date.now() - new Date(iso).getTime();
   129→   129→  const s = Math.floor(diff / 1000);
   130→   130→  if (s < 60) return 'just now';
   131→   131→  const m = Math.floor(s / 60);
   132→   132→  if (m < 60) return `${m}m ago`;
   133→   133→  const h = Math.floor(m / 60);
   134→   134→  if (h < 24) return `${h}h ago`;
   135→   135→  return `${Math.floor(h / 24)}d ago`;
   136→   136→}

   139→   139→
   140→   140→function useAnimatedNumber(target: number, duration = 1000) {
   141→   141→  const [val, setVal] = useState(0);
   142→   142→  useEffect(() => {
   143→   143→    let raf = 0;
   144→   144→    const start = performance.now();
   145→   145→    const from = 0;
   146→   146→    const animate = (now: number) => {
   147→   147→      const t = Math.min((now - start) / duration, 1);
   148→   148→      const eased = 1 - Math.pow(1 - t, 3);
   149→   149→      setVal(from + (target - from) * eased);
   150→   150→      if (t < 1) raf = requestAnimationFrame(animate);
   151→   151→    };
   152→   152→    raf = requestAnimationFrame(animate);
   153→   153→    return () => cancelAnimationFrame(raf);
   154→   154→  }, [target, duration]);
   155→   155→  return val;
   156→   156→}
