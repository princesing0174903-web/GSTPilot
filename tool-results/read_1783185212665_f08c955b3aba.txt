     3→// ═══════════════════════════════════════════════════════════════════════════════
     4→// GSTPilot Gmail & WhatsApp Business Automation™ — useCommunications() Hook
     5→//
     6→// The SINGLE hook every GSTPilot component uses to interact with communication
     7→// data. Mirrors the useBanking() pattern:
     8→//
     9→//   • READ — real-time subscriptions to connections + messages + schedules
    10→//     (org-scoped via onSnapshot)
    11→//   • CONNECT_GMAIL — connect → complete → persist encrypted connection
    12→//   • CONNECT_WHATSAPP — connect → complete → persist encrypted connection
    13→//   • DISCONNECT — invalidate connection + cascade-delete all messages
    14→//   • SYNC — pull latest emails / WhatsApp messages from the provider
    15→//   • SEND — send an email or WhatsApp message
    16→//   • REPLY — reply to an existing message
    17→//   • SCHEDULE — schedule a new automated message
    18→//   • CANCEL — cancel a scheduled message
    19→//   • RUN_AUTOMATION — dispatch due scheduled messages (cron-style)
    20→//
    21→// All tenant scoping is automatic — components never touch `organizationId`.
    22→//
    23→// Architecture:
    24→//   1. Mutations call /api/communication/{gmail,whatsapp,automation}/* routes
    25→//      for provider work. The API returns plain data + encrypted sessions.
    26→//   2. The hook persists the result to Firestore via the service layer.
    27→//   3. Real-time onSnapshot subscriptions surface changes instantly.
    28→//
    29→// Security: encrypted sessions are stored in Firestore but can ONLY be decrypted
    30→// by the server. The client passes them opaquely to API routes.
    31→// ═══════════════════════════════════════════════════════════════════════════════

   156→
   157→export function useCommunications(): UseCommunicationsResult {
   158→  const { organization } = useOrg();
   159→  const { user } = useAuth();
   160→  const orgId = organization?.id ?? null;
   161→
   162→  const [gmailConnections, setGmailConnections] = useState<GmailConnection[]>([]);
   163→  const [whatsappConnections, setWhatsappConnections] = useState<WhatsAppConnection[]>([]);
   164→  const [gmailMessages, setGmailMessages] = useState<GmailMessage[]>([]);
   165→  const [whatsappMessages, setWhatsAppMessages] = useState<WhatsAppMessage[]>([]);
   166→  const [scheduledMessages, setScheduledMessages] = useState<ScheduledMessage[]>([]);
   167→
   168→  const [loading, setLoading] = useState(true);
   169→  const [error, setError] = useState<string | null>(null);
   170→  const [saving, setSaving] = useState(false);
   171→  const [retryTick, setRetryTick] = useState(0);
   172→
   173→  const unsubRefs = useRef<Array<(() => void) | null>>([null, null, null, null, null]);
   174→
   175→  useEffect(() => {
   176→    unsubRefs.current.forEach((unsub) => unsub?.());
   177→    unsubRefs.current = [null, null, null, null, null];
   178→
   179→    if (!orgId) {
   180→      setGmailConnections([]);
   181→      setWhatsappConnections([]);
   182→      setGmailMessages([]);
   183→      setWhatsAppMessages([]);
   184→      setScheduledMessages([]);
   185→      setLoading(false);
   186→      return;
   187→    }
   188→
   189→    setLoading(true);
   190→
   191→    const onSubError = (err: Error) => {
   192→      console.warn('[useCommunications] subscription error:', err.message);
   193→      setError(err.message);
   194→      setLoading(false);
   195→    };
   196→
   197→    unsubRefs.current[0] = subscribeToGmailConnections(
   198→      orgId,
   199→      (conns) => {
   200→        setGmailConnections(conns);
   201→        setLoading(false);
   202→        setError(null);
   203→      },
   204→      { onError: onSubError },
   205→    );
   206→
   207→    unsubRefs.current[1] = subscribeToWhatsAppConnections(
   208→      orgId,
   209→      (conns) => {
   210→        setWhatsappConnections(conns);
   211→        setError(null);
   212→      },
   213→      { onError: onSubError },
   214→    );
   215→
   216→    unsubRefs.current[2] = subscribeToGmailMessages(
   217→      orgId,
   218→      (msgs) => {
   219→        setGmailMessages(msgs);
   220→        setError(null);
   221→      },
   222→      { onError: onSubError, limitCount: 200 },
   223→    );
   224→
   225→    unsubRefs.current[3] = subscribeToWhatsAppMessages(
   226→      orgId,
   227→      (msgs) => {
   228→        setWhatsAppMessages(msgs);
   229→        setError(null);
   230→      },
   231→      { onError: onSubError, limitCount: 200 },
   232→    );
   233→
   234→    unsubRefs.current[4] = subscribeToScheduledMessages(
   235→      orgId,
   236→      (schedules) => {
   237→        setScheduledMessages(schedules);
   238→        setError(null);
   239→      },
   240→      { onError: onSubError, limitCount: 100 },
   241→    );
   242→
   243→    return () => {
   244→      unsubRefs.current.forEach((unsub) => unsub?.());
   245→    };
   246→  }, [orgId, retryTick]);
   247→
   248→  // ─── Helpers ──────────────────────────────────────────────────────────────
   249→
   250→  const createdBy = {
   251→    uid: user?.id ?? '',
   252→    name: user?.name ?? 'Unknown',
   253→    email: user?.email ?? '',
   254→  };
   255→
   256→  const gmailConn = gmailConnections.find((c) => c.authStatus === 'connected') ?? null;
   257→  const waConn = whatsappConnections.find((c) => c.authStatus === 'connected') ?? null;
   258→
   259→  // ─── Mutation: connectGmail ───────────────────────────────────────────────
   260→
   261→  const connectGmail = useCallback(
   262→    async (input: {
   263→      provider?: GmailProviderName;
   264→      email: string;
   265→      displayName?: string;
   266→    }): Promise<boolean> => {
   267→      if (!orgId) return false;
   268→      setSaving(true);
   269→      setError(null);
   270→      try {
   271→        const res = await fetch('/api/communication/gmail/connect', {
   272→          method: 'POST',
   273→          headers: { 'Content-Type': 'application/json' },
   274→          body: JSON.stringify({
   275→            organizationId: orgId,
   276→            provider: input.provider ?? 'mock',
   277→            email: input.email,
   278→            displayName: input.displayName,
   279→            createdBy,
   280→          }),
   281→        });
   282→        const data = await res.json();
   283→        if (!res.ok || !data.ok) {
   284→          throw new Error(data.error ?? 'Failed to connect Gmail.');
   285→        }
   286→        const { result, complete } = data;
   287→        if (!complete) {
   288→          throw new Error('Gmail connection could not be completed. Please try again.');
   289→        }
   290→        await saveGmailConnection(orgId, {
   291→          organizationId: orgId,
   292→          email: result.email,
   293→          displayName: complete.profile.displayName ?? input.displayName ?? null,
   294→          provider: input.provider ?? 'mock',
   295→          authStatus: 'connected',
   296→          lastSync: new Date().toISOString(),
   297→          sessionExpiry: complete.sessionExpiry,
   298→          encryptedConnection: complete.encryptedConnection,
   299→          scopes: complete.scopes,
   300→          lastError: null,
   301→          createdBy,
   302→        });
   303→        return true;
   304→      } catch (err) {
   305→        const msg = err instanceof Error ? err.message : 'Unknown error';
   306→        setError(msg);
   307→        return false;
   308→      } finally {
   309→        setSaving(false);
   310→      }
   311→    },
   312→    [orgId, createdBy],
   313→  );
   314→
   315→  // ─── Mutation: disconnectGmail ────────────────────────────────────────────
   316→
   317→  const disconnectGmail = useCallback(
   318→    async (connectionId: string): Promise<boolean> => {

   333→        const msg = err instanceof Error ? err.message : 'Unknown error';

   335→        return false;

   337→        setSaving(false);
   338→      }
   339→    },
   340→    [orgId, gmailConnections],
   341→  );
   342→
   343→  // ─── Mutation: syncGmail ──────────────────────────────────────────────────
   344→
   345→  const syncGmail = useCallback(
   346→    async (connectionId: string, options?: { from?: string; to?: string }): Promise<boolean> => {
   347→      if (!orgId) return false;
   348→      const conn = gmailConnections.find((c) => c.id === connectionId);
   349→      if (!conn?.encryptedConnection) return false;
   350→      setSaving(true);
   351→      setError(null);
   352→
   353→      const jobId = await createSyncJob(orgId, {
   354→        connectionId,
   355→        channel: 'gmail',
   356→        trigger: 'manual',
   357→        maxRetries: 3,
   358→      }).catch(() => null);
   359→
   360→      try {
   361→        const res = await fetch('/api/communication/gmail/sync', {
   362→          method: 'POST',
   363→          headers: { 'Content-Type': 'application/json' },
   364→          body: JSON.stringify({
   365→            organizationId: orgId,
   366→            connectionId,
   367→            encryptedConnection: conn.encryptedConnection,
   368→            from: options?.from,
   369→            to: options?.to,
   370→          }),
   371→        });
   372→        const data = await res.json();
   373→        if (!res.ok || !data.ok) {
   374→          throw new Error(data.error ?? 'Gmail sync failed.');
   375→        }
   376→        const { messages } = data.result;
   377→        if (Array.isArray(messages)) {
   378→          await saveGmailMessages(orgId, messages);
   379→        }
   380→        await updateGmailConnection(orgId, connectionId, {
   381→          authStatus: 'connected',
   382→          lastSync: new Date().toISOString(),
   383→          lastError: null,
   384→        });
   385→        if (jobId) {
   386→          await updateSyncJob(orgId, jobId, {
   387→            status: 'completed',
   388→            completedAt: new Date().toISOString(),
   389→            result: { newCount: messages?.length ?? 0, totalCount: messages?.length ?? 0 },
   390→          }).catch(() => {});
   391→        }
   392→        return true;
   393→      } catch (err) {
   394→        const msg = err instanceof Error ? err.message : 'Unknown error';
   395→        setError(msg);
   396→        if (jobId) {
   397→          await updateSyncJob(orgId, jobId, {
   398→            status: 'failed',
   399→            completedAt: new Date().toISOString(),
   400→            error: msg,
   401→          }).catch(() => {});
   402→        }
   403→        return false;
   404→      } finally {
   405→        setSaving(false);
   406→      }
   407→    },
   408→    [orgId, gmailConnections],
   409→  );
   410→

   613→            result: { newCount: messages?.length ?? 0, totalCount: messages?.length ?? 0 },
   614→          }).catch(() => {});
   615→        }
   616→        return true;
   617→      } catch (err) {
   618→        const msg = err instanceof Error ? err.message : 'Unknown error';
   619→        setError(msg);
   620→        if (jobId) {
   621→          await updateSyncJob(orgId, jobId, {
   622→            status: 'failed',
   623→            completedAt: new Date().toISOString(),
   624→            error: msg,
   625→          }).catch(() => {});
   626→        }
   627→        return false;
   628→      } finally {
   629→        setSaving(false);
   630→      }
   631→    },
   632→    [orgId, whatsappConnections],
   633→  );
