   849→   849→  // ── PART 7 ── Loading state while authentication initializes.
   850→   850→  if (isInitializing && !isAuthenticated) {
   851→   851→    return (
   852→   852→      <div className="min-h-screen flex items-center justify-center bg-black">
   853→   853→        <div className="flex flex-col items-center gap-4">
   854→   854→          <div className="flex h-11 w-11 items-center justify-center rounded-2xl glass-surface motion-pulse">
   855→   855→            <Zap className="h-5 w-5 accent-text" />
   856→   856→          </div>
   857→   857→          <span className="text-sm text-white/55 font-medium">Loading GSTPilot…</span>
   858→   858→        </div>
   859→   859→      </div>
   860→   860→    )
   861→   861→  }
   862→   862→
   863→   863→  // ── PART 7 ── While the org context is resolving after auth, show a brief
   864→   864→  // loader so protected pages never mount with a null organization.
   865→   865→  if (isAuthenticated && orgLoading && !needsOnboarding) {
   866→   866→    return (
   867→   867→      <div className="min-h-screen flex items-center justify-center bg-black">
   868→   868→        <div className="flex flex-col items-center gap-4">
   869→   869→          <div className="flex h-11 w-11 items-center justify-center rounded-2xl glass-surface motion-pulse">
   870→   870→            <Zap className="h-5 w-5 accent-text" />
   871→   871→          </div>
   872→   872→          <span className="text-sm text-white/55 font-medium">Loading your workspace…</span>
   873→   873→        </div>
   874→   874→      </div>
   875→   875→    )
   876→   876→  }
   877→   877→
   878→   878→  // ── PART 2 ── Authenticated but no organization → onboarding creates one.
   879→   879→  // BUT: if there was a load error (e.g. Firestore unreachable), show a retry
   880→   880→  // screen instead of onboarding — onboarding would also fail to write.
   881→   881→  if (needsOnboarding && !orgError) {
   882→   882→    return <OnboardingScreen />
   883→   883→  }
   884→   884→
   885→   885→  // ── PART 8 ── Org context failed to load after retries. Show a friendly
   886→   886→  // error with a retry button. Never crash, never expose Firebase errors.
   887→   887→  if (isAuthenticated && !orgLoading && orgError && !organization) {
   888→   888→    return (
   889→   889→      <div className="min-h-screen flex items-center justify-center bg-black p-6">
   890→   890→        <div className="flex flex-col items-center gap-5 max-w-md text-center">
   891→   891→          <div className="flex h-12 w-12 items-center justify-center rounded-2xl bg-red-500/10 border border-red-500/20">
   892→   892→            <Zap className="h-6 w-6 text-red-400" />
   893→   893→          </div>
   894→   894→          <h2 className="text-xl font-bold text-white">Couldn&apos;t load your workspace</h2>
   895→   895→          <p className="text-sm text-white/55 leading-relaxed">
   896→   896→            {orgError}
   897→   897→          </p>
   898→   898→          <button
   899→   899→            onClick={() => reloadOrg()}
   900→   900→            className="mt-2 inline-flex items-center gap-2 rounded-xl bg-white px-5 py-2.5 text-sm font-semibold text-black hover:bg-white/90 transition-colors"
   901→   901→          >
   902→   902→            Try again
   903→   903→          </button>
   904→   904→        </div>
   905→   905→      </div>
   906→   906→    )
   907→   907→  }
   908→   908→
   909→   909→  // ── PART 5 ── Protected route: only render the app shell when authenticated
   910→   910→  // AND the organization context is fully loaded.
   911→   911→  if (currentScreen === 'app' && isAuthenticated && organization) {
   912→   912→    return (
   913→   913→      <div className="flex min-h-screen flex-col">
   914→   914→        {needsEmailVerification && <EmailVerificationBanner />}
   915→   915→        <DashboardContent />
   916→   916→      </div>
   917→   917→    )
   918→   918→  }
   919→   919→
   920→   920→  if (currentScreen === 'login') {
   921→   921→    return <LoginPage onBack={handleBackToLanding} onGetStarted={handleGetStarted} />
   922→   922→  }
   923→   923→
   924→   924→  return <LandingPage onGetStarted={handleGetStarted} onBookDemo={handleBookDemo} />
   925→   925→}
