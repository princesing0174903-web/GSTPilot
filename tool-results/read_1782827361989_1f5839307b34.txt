   585→function EmailVerificationBanner() {

   636→function OnboardingScreen() {
   637→  const { user, markOnboardingComplete } = useAuth()
   638→  const { setCurrentScreen, setCurrentView } = useApp()
   639→  const [saveError, setSaveError] = React.useState<string | null>(null)
   640→
   641→  // ── Fire-and-forget: save onboarding data to Firestore in the background ──
   642→  const saveOnboardingToFirestore = async (
   643→    data: import('@/components/onboarding/OnboardingFlow').OnboardingData,
   644→    firmId: string
   645→  ) => {
   646→    try {
   647→      const { doc, setDoc, serverTimestamp } = await import('firebase/firestore')
   648→      const { db } = await import('@/lib/firebase')
   649→
   650→      if (!user) return
   651→
   652→      // Update user document
   653→      await setDoc(doc(db, 'users', user.id), {
   654→        uid: user.id,
   655→        fullName: data.fullName,
   656→        email: data.email,
   657→        phone: data.phone,
   658→        ageGroup: data.ageGroup,
   659→        profession: data.profession,
   660→        experience: data.experience,
   661→        firmId: firmId || null,
   662→        firmName: data.firmName,
   663→        onboardingCompleted: true,
   664→        clientCount: data.clientCount,
   665→        monthlyReturns: data.monthlyReturns,
   666→        gstServices: data.gstServices,
   667→        painPoints: data.painPoints,
   668→        referralSource: data.referralSource,
   669→        trialReasons: data.trialReasons,
   670→        wantsUpdates: data.wantsUpdates,
   671→        updatedAt: serverTimestamp(),
   672→      }, { merge: true })
   673→      console.log('[Onboarding] ✅ User document saved to Firestore')
   674→
   675→      // Save onboarding record
   676→      await setDoc(doc(db, 'onboarding', user.id), {
   677→        ...data,
   678→        firmId: firmId || null,
   679→        completedAt: serverTimestamp(),
   680→      })
   681→      console.log('[Onboarding] ✅ Onboarding record saved to Firestore')
   682→    } catch (error) {
   683→      console.warn('[Onboarding] ⚠️ Background Firestore save failed:', error)
   684→    }
   685→  }
   686→
   687→  const handleOnboardingComplete = (
   688→    data: import('@/components/onboarding/OnboardingFlow').OnboardingData,
   689→    destination?: import('@/components/onboarding/OnboardingFlow').OnboardingDestination
   690→  ) => {
   691→    setSaveError(null)
   692→
   693→    if (!user) {
   694→      setSaveError('No authenticated user found. Please sign in again.')
   695→      return
   696→    }
   697→
   698→    // ── INSTANT: Mark onboarding complete locally & navigate ──
   699→    // This happens synchronously — user sees dashboard immediately
   700→    console.log('[Onboarding] 🚀 Navigating to', destination || 'dashboard', '(optimistic)')
   701→    markOnboardingComplete(undefined, data.firmName)
   702→    setCurrentView(destination || 'dashboard')
   703→    setCurrentScreen('app')
   704→
   705→    // ── BACKGROUND: Create firm doc + save everything to Firestore ──
   706→    // These happen async — user is already in the dashboard
   707→    console.log('[Onboarding] 📝 Starting background Firestore writes...')
   708→    ;(async () => {
   709→      try {
   710→        const { collection, addDoc, serverTimestamp } = await import('firebase/firestore')
   711→        const { db } = await import('@/lib/firebase')
   712→
   713→        // Create firm document
   714→        let firmId = ''
   715→        try {
   716→          const firmRef = await addDoc(collection(db, 'firms'), {
   717→            ownerId: user.id,
   718→            firmName: data.firmName,
   719→            gstin: data.gstin || null,
   720→            state: data.state,
   721→            stateCode: data.stateCode,
   722→            organizationType: data.organizationType,
   723→            icaiMembershipNo: data.icaiMembershipNo || null,
   724→            officeAddress: data.officeAddress || null,
   725→            createdAt: serverTimestamp(),
   726→          })
   727→          firmId = firmRef.id
   728→          console.log('[Onboarding] ✅ Firm created:', firmId)
   729→
   730→          // Update the local user with firmId
   731→          markOnboardingComplete(firmId, data.firmName)
   732→        } catch (firmError) {
   733→          console.warn('[Onboarding] ⚠️ Firm creation failed:', firmError)
   734→        }
   735→
   736→        // Save remaining data
   737→        await saveOnboardingToFirestore(data, firmId)
   738→      } catch (error) {
   739→        console.warn('[Onboarding] ⚠️ Background save error:', error)
   740→      }
   741→    })()
   742→  }
   743→
   744→  const handleSkip = () => {
   745→    setSaveError(null)
   746→
   747→    if (!user) {
   748→      setSaveError('No authenticated user found. Please sign in again.')
   749→      return
   750→    }
   751→
   752→    // ── INSTANT: Navigate immediately ──
   753→    console.log('[Onboarding] 🚀 Skipping onboarding (optimistic)')
   754→    markOnboardingComplete()
   755→    setCurrentView('dashboard')
   756→    setCurrentScreen('app')
   757→
   758→    // ── BACKGROUND: Mark as completed in Firestore ──
   759→    ;(async () => {
   760→      try {
   761→        const { doc, setDoc, serverTimestamp } = await import('firebase/firestore')
   762→        const { db } = await import('@/lib/firebase')
   763→        await setDoc(doc(db, 'users', user.id), {
   764→          onboardingCompleted: true,
   765→          updatedAt: serverTimestamp(),
   766→        }, { merge: true })
   767→        console.log('[Onboarding] ✅ Skip saved to Firestore')
   768→      } catch (error) {
   769→        console.warn('[Onboarding] ⚠️ Background skip save failed:', error)
   770→      }
   771→    })()
   772→  }
   773→
   774→  return (
   775→    <OnboardingFlow
   776→      onComplete={handleOnboardingComplete}
   777→      onSkip={handleSkip}
   778→      userEmail={user?.email}
   779→      userName={user?.name}
   780→      error={saveError}
   781→      onDismissError={() => setSaveError(null)}
   782→    />
   783→  )
   784→}

   786→function AppRouter() {
   787→  const { currentScreen, setCurrentScreen } = useApp()
   788→  const { isAuthenticated, isInitializing, needsOnboarding, needsEmailVerification } = useAuth()
   789→
   790→  useEffect(() => {
   791→    if (isInitializing) return
   792→    // Only auto-redirect to app if authenticated AND onboarding is complete
   793→    // Don't redirect if user still needs onboarding — the OnboardingScreen handles its own navigation
   794→    if (isAuthenticated && !needsOnboarding && currentScreen !== 'app') {
   795→      setCurrentScreen('app')
   796→    }
   797→  }, [isAuthenticated, isInitializing, currentScreen, setCurrentScreen, needsOnboarding])
   798→
   799→  useEffect(() => {
   800→    if (isInitializing) return
   801→    if (!isAuthenticated && currentScreen === 'app') {
   802→      setCurrentScreen('landing')
   803→    }
   804→  }, [isAuthenticated, isInitializing, currentScreen, setCurrentScreen])
   805→
   806→  const handleGetStarted = () => setCurrentScreen('login')
   807→  const handleBookDemo = () => setCurrentScreen('login')
   808→  const handleBackToLanding = () => setCurrentScreen('landing')
   809→
   810→  if (isInitializing && !isAuthenticated) {
   811→    return (
   812→      <div className="min-h-screen flex items-center justify-center bg-black">
   813→        <div className="flex flex-col items-center gap-4">
   814→          <div className="flex h-11 w-11 items-center justify-center rounded-2xl glass-surface motion-pulse">
   815→            <Zap className="h-5 w-5 accent-text" />
   816→          </div>
   817→          <span className="text-sm text-white/55 font-medium">Loading GSTPilot…</span>
   818→        </div>
   819→      </div>
   820→    )
   821→  }
   822→
   823→  // Show onboarding if user hasn't completed it
   824→  if (isAuthenticated && needsOnboarding) {
   825→    return <OnboardingScreen />
   826→  }
   827→
   828→  if (currentScreen === 'app' && isAuthenticated) {
   829→    return (
   830→      <div className="flex min-h-screen flex-col">
   831→        {needsEmailVerification && <EmailVerificationBanner />}
   832→        <DashboardContent />
   833→      </div>
   834→    )
   835→  }
   836→
   837→  if (currentScreen === 'login') {
   838→    return <LoginPage onBack={handleBackToLanding} onGetStarted={handleGetStarted} />
   839→  }
   840→
   841→  return <LandingPage onGetStarted={handleGetStarted} onBookDemo={handleBookDemo} />
   842→}
