    72→function formatINR(amount: number): string {
    73→  return '₹' + amount.toLocaleString('en-IN');
    74→}
    75→
    76→function formatDate(dateStr: string): string {
    77→  const d = new Date(dateStr);
    78→  return `${String(d.getDate()).padStart(2, '0')}/${String(d.getMonth() + 1).padStart(2, '0')}/${d.getFullYear()}`;
    79→}
    80→
    81→function daysAgo(days: number): string {
    82→  const d = new Date();
    83→  d.setDate(d.getDate() - days);
    84→  return d.toISOString();
    85→}
    86→
    87→function daysFromNow(days: number): string {
    88→  const d = new Date();
    89→  d.setDate(d.getDate() + days);
    90→  return d.toISOString();
    91→}

   313→function PaymentMethodDonut() {
   314→  // PT-1-a-retry: fetch real invoices to compute the actual collected amount
   315→  // (sum of Invoice.totalAmount). Falls back to ₹0 + empty state when DB is empty.
   316→  const { data: invoicesResp, isLoading } = useQuery<{ invoices: Array<{ totalAmount: number }> }>({
   317→    queryKey: ['invoices', 'embedded-finance-donut'],
   318→    queryFn: () => apiGet('/api/invoices'),
   319→  });
   320→  const collectedTotal = useMemo(() => {
   321→    const invoices = invoicesResp?.invoices ?? [];
   322→    return invoices.reduce((sum, inv) => sum + (Number(inv.totalAmount) || 0), 0);
   323→  }, [invoicesResp]);
   324→
   325→  const data = [
   326→    { label: 'UPI', value: 58, color: '#10b981' },
   327→    { label: 'Bank Transfer', value: 28, color: '#64748b' },
   328→    { label: 'Card', value: 14, color: '#f59e0b' },
   329→  ];
   330→
   331→  const total = data.reduce((s, d) => s + d.value, 0);
   332→  const cx = 80;
   333→  const cy = 80;
   334→  const r = 55;
   335→  const innerR = 35;
   336→
   337→  let currentAngle = -90;
   338→  const arcs = data.map((d) => {
   339→    const angle = (d.value / total) * 360;
   340→    const startAngle = currentAngle;
   341→    const endAngle = currentAngle + angle;
   342→    currentAngle = endAngle;
   343→
   344→    const startRad = (startAngle * Math.PI) / 180;
   345→    const endRad = (endAngle * Math.PI) / 180;
   346→    const largeArc = angle > 180 ? 1 : 0;
   347→
   348→    const outerStart = { x: cx + r * Math.cos(startRad), y: cy + r * Math.sin(startRad) };
   349→    const outerEnd = { x: cx + r * Math.cos(endRad), y: cy + r * Math.sin(endRad) };
   350→    const innerStart = { x: cx + innerR * Math.cos(endRad), y: cy + innerR * Math.sin(endRad) };
   351→    const innerEnd = { x: cx + innerR * Math.cos(startRad), y: cy + innerR * Math.sin(startRad) };
   352→
   353→    const path = [
   354→      `M ${outerStart.x} ${outerStart.y}`,
   355→      `A ${r} ${r} 0 ${largeArc} 1 ${outerEnd.x} ${outerEnd.y}`,
   356→      `L ${innerStart.x} ${innerStart.y}`,
   357→      `A ${innerR} ${innerR} 0 ${largeArc} 0 ${innerEnd.x} ${innerEnd.y}`,
   358→      'Z',
   359→    ].join(' ');
   360→
   361→    return { ...d, path, percentage: Math.round((d.value / total) * 100) };
   362→  });
   363→
   364→  return (
   365→    <div className="flex items-center gap-6">
   366→      <svg viewBox="0 0 160 160" className="w-32 h-32 shrink-0">
   367→        {arcs.map((arc, i) => (
   368→          <path key={i} d={arc.path} fill={arc.color} opacity="0.85" className="hover:opacity-100 transition-opacity" />
   369→        ))}
   370→        <text x={cx} y={cy - 4} textAnchor="middle" className="text-[11px] fill-slate-700 font-semibold">
   371→          {isLoading ? '…' : formatINR(collectedTotal)}
   372→        </text>
   373→        <text x={cx} y={cy + 10} textAnchor="middle" className="text-[8px] fill-slate-400">
   374→          {collectedTotal === 0 && !isLoading ? 'no invoices yet' : 'collected'}
   375→        </text>
   376→      </svg>
   377→      <div className="space-y-2.5">
   378→        {data.map((d, i) => (
   379→          <div key={i} className="flex items-center gap-2">
   380→            <div className="w-2.5 h-2.5 rounded-full" style={{ backgroundColor: d.color }} />
   381→            <span className="text-xs text-slate-600">{d.label}</span>
   382→            <span className="text-xs font-semibold text-slate-800 ml-auto">{d.value}%</span>
   383→          </div>
   384→        ))}
   385→      </div>
   386→    </div>
   387→  );
   388→}

   390→function CashFlowForecastChart() {
   391→  const width = 480;
   392→  const height = 180;
   393→  const padding = { top: 20, right: 20, bottom: 30, left: 60 };
   394→  const chartW = width - padding.left - padding.right;
   395→  const chartH = height - padding.top - padding.bottom;
   396→
   397→  const maxVal = Math.max(...CASH_FLOW_FORECAST.map(d => Math.max(d.inflow, d.outflow)));
   398→  const yScale = (v: number) => chartH - (v / maxVal) * chartH;
   399→  const xScale = (i: number) => (i / (CASH_FLOW_FORECAST.length - 1)) * chartW;
   400→
   401→  // Cumulative net flow for area
   402→  const netFlow = CASH_FLOW_FORECAST.reduce<number[]>((acc, d, i) => {
   403→    const prev = i > 0 ? acc[i - 1] : 0;
   404→    acc.push(prev + d.inflow - d.outflow);
   405→    return acc;
   406→  }, []);
   407→
   408→  const netMax = Math.max(...netFlow.map(Math.abs));
   409→  const netScale = (v: number) => chartH / 2 - (v / netMax) * (chartH / 2);
   410→
   411→  const inflowPath = CASH_FLOW_FORECAST.map((d, i) =>

   477→function SimulatedQRCode({ size = 120 }: { size?: number }) {
   478→  // Generate a simulated QR pattern
   479→  const cells = 21;
   480→  const cellSize = size / cells;
   481→  const pattern: boolean[][] = [];
   482→
   483→  // Seed-based pseudo-random for consistency
   484→  let seed = 42;
   485→  const rand = () => {
   486→    seed = (seed * 16807 + 0) % 2147483647;
   487→    return (seed - 1) / 2147483646;
   488→  };
   489→
   490→  for (let r = 0; r < cells; r++) {
   491→    pattern[r] = [];
   492→    for (let c = 0; c < cells; c++) {
   493→      // Finder patterns (3 corners)
   494→      const isFinderTL = r < 7 && c < 7;
   495→      const isFinderTR = r < 7 && c >= cells - 7;
   496→      const isFinderBL = r >= cells - 7 && c < 7;
   497→
   498→      if (isFinderTL || isFinderTR || isFinderBL) {
   499→        const fr = isFinderBL ? r - (cells - 7) : r;
   500→        const fc = isFinderTR ? c - (cells - 7) : c;
   501→        if (fr === 0 || fr === 6 || fc === 0 || fc === 6) pattern[r][c] = true;
   502→        else if (fr >= 2 && fr <= 4 && fc >= 2 && fc <= 4) pattern[r][c] = true;
   503→        else pattern[r][c] = false;
   504→      } else {
   505→        pattern[r][c] = rand() > 0.5;
   506→      }
   507→    }
   508→  }
   509→
   510→  return (
   511→    <svg width={size} height={size} viewBox={`0 0 ${size} ${size}`}>
   512→      <rect width={size} height={size} fill="white" rx="4" />
   513→      {pattern.map((row, r) =>
   514→        row.map((cell, c) =>
   515→          cell ? (
   516→            <rect
   517→              key={`${r}-${c}`}
   518→              x={c * cellSize}
   519→              y={r * cellSize}
   520→              width={cellSize}
   521→              height={cellSize}
   522→              fill="#1e293b"
   523→              rx={0.5}
   524→            />
   525→          ) : null
   526→        )
   527→      )}
   528→    </svg>
   529→  );
   530→}
