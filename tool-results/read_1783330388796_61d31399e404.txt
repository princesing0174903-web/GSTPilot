   256→   256→// ─── Password Strength Helper ──────────────────────────────────────────
   257→   257→function getPasswordStrength(password: string): { score: number; label: string; color: string } {
   258→   258→  if (!password) return { score: 0, label: '', color: '' }
   259→   259→  let score = 0
   260→   260→  if (password.length >= 8) score++
   261→   261→  if (password.length >= 12) score++
   262→   262→  if (/[A-Z]/.test(password)) score++
   263→   263→  if (/[0-9]/.test(password)) score++
   264→   264→  if (/[^A-Za-z0-9]/.test(password)) score++
   265→   265→
   266→   266→  if (score <= 1) return { score: 20, label: 'Weak', color: 'bg-red-500' }
   267→   267→  if (score <= 2) return { score: 40, label: 'Fair', color: 'bg-orange-500' }
   268→   268→  if (score <= 3) return { score: 60, label: 'Good', color: 'bg-yellow-500' }
   269→   269→  if (score <= 4) return { score: 80, label: 'Strong', color: 'bg-emerald-500' }
   270→   270→  return { score: 100, label: 'Excellent', color: 'bg-emerald-600' }
   271→   271→}
