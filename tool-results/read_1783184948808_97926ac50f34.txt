   665→function OnboardingScreen() {
   666→  const { user } = useAuth()
   667→  const { completeOnboarding } = useOrg()
   668→  const { setCurrentScreen, setCurrentView } = useApp()
   669→  const [saveError, setSaveError] = React.useState<string | null>(null)
   670→  const [isSubmitting, setIsSubmitting] = React.useState(false)
   671→
   672→  // ── Create the organization + owner membership via the org service ──
   673→  const createOrganizationForUser = async (
   674→    data: import('@/components/onboarding/OnboardingFlow').OnboardingData
   675→  ): Promise<{ orgId: string | null; error: string | null }> => {
   676→    if (!user) return { orgId: null, error: 'No authenticated user found.' }
   677→
   678→    const { createOrganization, updateUserProfile } = await import('@/lib/auth/organizations')
   679→    const { doc, setDoc, serverTimestamp } = await import('firebase/firestore')
   680→    const { db } = await import('@/lib/firebase')
   681→
   682→    // 1. Create the organization + owner membership + set currentOrganizationId.
   683→    const { organization, error: orgError } = await createOrganization({
   684→      name: data.firmName,
   685→      ownerId: user.id,
   686→      ownerEmail: user.email,
   687→      ownerDisplayName: data.fullName || user.name,
   688→      ownerPhotoURL: user.picture || null,
   689→      gstin: data.gstin || null,
   690→      plan: 'free',
   691→    })
   692→    if (orgError || !organization) {
   693→      return { orgId: null, error: orgError || 'Could not create your organization.' }
   694→    }
   695→
   696→    // 2. Enrich the user profile with onboarding metadata (best-effort).
   697→    try {
   698→      await updateUserProfile(user.id, {
   699→        displayName: data.fullName,
   700→        phone: data.phone,
   701→        company: data.firmName,
   702→        gstin: data.gstin || null,
   703→      })
   704→      // Persist the extended onboarding questionnaire for analytics.
   705→      await setDoc(doc(db, 'onboarding', user.id), {
   706→        ...data,
   707→        organizationId: organization.id,
   708→        completedAt: serverTimestamp(),
   709→      }, { merge: true })
   710→    } catch (err) {
   711→      console.warn('[Onboarding] Profile enrichment failed:', err)
   712→    }
   713→
   714→    return { orgId: organization.id, error: null }
   715→  }
   716→
   717→  const handleOnboardingComplete = async (
   718→    data: import('@/components/onboarding/OnboardingFlow').OnboardingData,
   719→    destination?: import('@/components/onboarding/OnboardingFlow').OnboardingDestination
   720→  ) => {
   721→    setSaveError(null)
   722→    if (!user) {
   723→      setSaveError('No authenticated user found. Please sign in again.')
   724→      return
   725→    }
   726→
   727→    setIsSubmitting(true)
   728→    try {
   729→      const { orgId, error: createError } = await createOrganizationForUser(data)
   730→      if (createError || !orgId) {
   731→        setSaveError(createError || 'Could not create your organization. Please try again.')
   732→        setIsSubmitting(false)
   733→        return
   734→      }
   735→
   736→      // Reload the org context so the rest of the app sees the new org.
   737→      await completeOnboarding(orgId)
   738→
   739→      // Navigate to the chosen destination.
   740→      setCurrentView(destination || 'dashboard')
   741→      setCurrentScreen('app')
   742→    } catch (err) {
   743→      setSaveError(
   744→        err instanceof Error
   745→          ? err.message
   746→          : 'Something went wrong while setting up your workspace.'
   747→      )
   748→    } finally {
   749→      setIsSubmitting(false)
   750→    }
   751→  }
   752→
   753→  const handleSkip = async () => {
   754→    setSaveError(null)
   755→    if (!user) {
   756→      setSaveError('No authenticated user found. Please sign in again.')
   757→      return
   758→    }
   759→
   760→    setIsSubmitting(true)
   761→    try {
   762→      // Skipping still requires an organization — create a default one
   763→      // using the user's name so the app is fully functional.
   764→      const { createOrganization } = await import('@/lib/auth/organizations')
   765→      const { organization, error: orgError } = await createOrganization({
   766→        name: `${user.name}'s Workspace`,
   767→        ownerId: user.id,
   768→        ownerEmail: user.email,
   769→        ownerDisplayName: user.name,
   770→        ownerPhotoURL: user.picture || null,
   771→        gstin: null,
   772→        plan: 'free',
   773→      })
   774→      if (orgError || !organization) {
   775→        setSaveError(orgError || 'Could not create your workspace. Please try again.')
   776→        setIsSubmitting(false)
   777→        return
   778→      }
   779→      await completeOnboarding(organization.id)
   780→      setCurrentView('dashboard')
   781→      setCurrentScreen('app')
   782→    } catch (err) {
   783→      setSaveError(
   784→        err instanceof Error
   785→          ? err.message
   786→          : 'Something went wrong while setting up your workspace.'
   787→      )
   788→    } finally {
   789→      setIsSubmitting(false)
   790→    }
   791→  }
   792→
   793→  return (
   794→    <OnboardingFlow
   795→      onComplete={handleOnboardingComplete}
   796→      onSkip={handleSkip}
   797→      userEmail={user?.email}
   798→      userName={user?.name}
   799→      error={saveError}
   800→      onDismissError={() => setSaveError(null)}
   801→    />
   802→  )
   803→}

   805→function AppRouter() {
   806→  const { currentScreen, setCurrentScreen } = useApp()
   807→  const { isAuthenticated, isInitializing, needsEmailVerification } = useAuth()
   808→  // ── PART 3 / PART 5 ── The org context is the source of truth for whether
   809→  // the user has an organization. `needsOrganization` becomes the new
   810→  // "needs onboarding" gate. `orgLoading` lets us hold the protected routes
   811→  // until the tenant context is fully resolved so no page ever renders with a
   812→  // null organizationId.
   813→  const { needsOrganization, loading: orgLoading, organization, error: orgError, reload: reloadOrg } = useOrg()
   814→
   815→  // Derived: a user needs onboarding when authenticated but without an org.
   816→  const needsOnboarding = isAuthenticated && needsOrganization
   817→
   818→  useEffect(() => {
   819→    if (isInitializing) return
   820→    // Only auto-redirect to the app when: authenticated, org context resolved,
   821→    // and the user actually has an organization. Otherwise the OnboardingScreen
   822→    // handles its own navigation.
   823→    if (isAuthenticated && !needsOnboarding && !orgLoading && currentScreen !== 'app') {
   824→      setCurrentScreen('app')
   825→    }
   826→  }, [isAuthenticated, isInitializing, currentScreen, setCurrentScreen, needsOnboarding, orgLoading])
   827→
   828→  useEffect(() => {
   829→    if (isInitializing) return
   830→    // ── PART 5 ── Protected routes: kick unauthenticated users back to landing.
   831→    if (!isAuthenticated && currentScreen === 'app') {
   832→      setCurrentScreen('landing')
   833→    }
   834→  }, [isAuthenticated, isInitializing, currentScreen, setCurrentScreen])
   835→
   836→  const handleGetStarted = () => setCurrentScreen('login')
   837→  const handleBookDemo = () => setCurrentScreen('login')
   838→  const handleBackToLanding = () => setCurrentScreen('landing')
   839→
   840→  // ── PART 7 ── Loading state while authentication initializes.
   841→  if (isInitializing && !isAuthenticated) {
   842→    return (
   843→      <div className="min-h-screen flex items-center justify-center bg-black">
   844→        <div className="flex flex-col items-center gap-4">
   845→          <div className="flex h-11 w-11 items-center justify-center rounded-2xl glass-surface motion-pulse">
   846→            <Zap className="h-5 w-5 accent-text" />
   847→          </div>
   848→          <span className="text-sm text-white/55 font-medium">Loading GSTPilot…</span>
   849→        </div>
   850→      </div>
   851→    )
   852→  }
   853→
   854→  // ── PART 7 ── While the org context is resolving after auth, show a brief
   855→  // loader so protected pages never mount with a null organization.
   856→  if (isAuthenticated && orgLoading && !needsOnboarding) {
   857→    return (
   858→      <div className="min-h-screen flex items-center justify-center bg-black">
   859→        <div className="flex flex-col items-center gap-4">
   860→          <div className="flex h-11 w-11 items-center justify-center rounded-2xl glass-surface motion-pulse">
   861→            <Zap className="h-5 w-5 accent-text" />
   862→          </div>
   863→          <span className="text-sm text-white/55 font-medium">Loading your workspace…</span>
   864→        </div>
   865→      </div>
   866→    )
   867→  }
   868→
   869→  // ── PART 2 ── Authenticated but no organization → onboarding creates one.
   870→  // BUT: if there was a load error (e.g. Firestore unreachable), show a retry
   871→  // screen instead of onboarding — onboarding would also fail to write.
   872→  if (needsOnboarding && !orgError) {
   873→    return <OnboardingScreen />
   874→  }
   875→
   876→  // ── PART 8 ── Org context failed to load after retries. Show a friendly
   877→  // error with a retry button. Never crash, never expose Firebase errors.
   878→  if (isAuthenticated && !orgLoading && orgError && !organization) {
   879→    return (
   880→      <div className="min-h-screen flex items-center justify-center bg-black p-6">
   881→        <div className="flex flex-col items-center gap-5 max-w-md text-center">
   882→          <div className="flex h-12 w-12 items-center justify-center rounded-2xl bg-red-500/10 border border-red-500/20">
   883→            <Zap className="h-6 w-6 text-red-400" />
   884→          </div>
   885→          <h2 className="text-xl font-bold text-white">Couldn&apos;t load your workspace</h2>
   886→          <p className="text-sm text-white/55 leading-relaxed">
   887→            {orgError}
   888→          </p>
   889→          <button
   890→            onClick={() => reloadOrg()}
   891→            className="mt-2 inline-flex items-center gap-2 rounded-xl bg-white px-5 py-2.5 text-sm font-semibold text-black hover:bg-white/90 transition-colors"
   892→          >
   893→            Try again
   894→          </button>
   895→        </div>
   896→      </div>
   897→    )
   898→  }
   899→
   900→  // ── PART 5 ── Protected route: only render the app shell when authenticated
   901→  // AND the organization context is fully loaded.
   902→  if (currentScreen === 'app' && isAuthenticated && organization) {
   903→    return (
   904→      <div className="flex min-h-screen flex-col">
   905→        {needsEmailVerification && <EmailVerificationBanner />}
   906→        <DashboardContent />
   907→      </div>
   908→    )
   909→  }
   910→
   911→  if (currentScreen === 'login') {
   912→    return <LoginPage onBack={handleBackToLanding} onGetStarted={handleGetStarted} />
   913→  }
   914→
   915→  return <LandingPage onGetStarted={handleGetStarted} onBookDemo={handleBookDemo} />
   916→}
