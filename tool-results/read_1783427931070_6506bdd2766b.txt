   217→}: OnboardingFlowProps) {
   218→  const [currentStep, setCurrentStep] = useState(0); // 0=welcome, 1-4=steps, 5=final
   219→  const [direction, setDirection] = useState(1);
   220→
   221→  const [formData, setFormData] = useState<OnboardingData>({
   222→    fullName: userName,
   223→    email: userEmail,
   224→    phone: '',
   225→    ageGroup: '',
   226→    profession: '',
   227→    experience: '',
   228→    firmName: '',
   229→    organizationType: '',
   230→    gstin: '',
   231→    state: '',
   232→    stateCode: '',
   233→    icaiMembershipNo: '',
   234→    officeAddress: '',
   235→    clientCount: '',
   236→    monthlyReturns: '',
   237→    gstServices: [],
   238→    painPoints: [],
   239→    referralSource: '',
   240→    trialReasons: [],
   241→    wantsUpdates: true,
   242→  });
   243→
   244→  const [errors, setErrors] = useState<Record<string, string>>({});
   245→
   246→  // ─── Helpers ──────────────────────────────────────────────────────────────
   247→
   248→  const updateField = useCallback(
   249→    <K extends keyof OnboardingData>(field: K, value: OnboardingData[K]) => {
   250→      setFormData((prev) => ({ ...prev, [field]: value }));
   251→      // Clear error on change
   252→      setErrors((prev) => {
   253→        const next = { ...prev };
   254→        delete next[field];
   255→        return next;
   256→      });
   257→    },
   258→    []
   259→  );
   260→
   261→  const toggleArrayItem = useCallback(
   262→    (field: 'gstServices' | 'painPoints' | 'trialReasons', item: string) => {
   263→      setFormData((prev) => {
   264→        const arr = prev[field] as string[];
   265→        const next = arr.includes(item)
   266→          ? arr.filter((i) => i !== item)
   267→          : [...arr, item];
   268→        return { ...prev, [field]: next };
   269→      });
   270→    },
   271→    []
   272→  );
   273→
   274→  const goNext = useCallback(() => {
   275→    setDirection(1);
   276→    setCurrentStep((prev) => Math.min(prev + 1, 5));
   277→  }, []);
   278→
   279→  const goBack = useCallback(() => {
   280→    setDirection(-1);
   281→    setCurrentStep((prev) => Math.max(prev - 1, 0));
   282→  }, []);
   283→
   284→  // ─── Validation ───────────────────────────────────────────────────────────
   285→
   286→  const validateStep = useCallback(
   287→    (step: number): boolean => {
   288→      const newErrors: Record<string, string> = {};
   289→
   290→      if (step === 1) {
   291→        if (!formData.fullName.trim()) newErrors.fullName = 'Full name is required';
   292→        if (!formData.email.trim()) newErrors.email = 'Email is required';
   293→        else if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(formData.email))
   294→          newErrors.email = 'Enter a valid email';
   295→        if (!formData.phone.trim()) newErrors.phone = 'Mobile number is required';
   296→        else if (!/^\d{10}$/.test(formData.phone.replace(/\s/g, '')))
   297→          newErrors.phone = 'Enter a valid 10-digit mobile number';
   298→      }
   299→
   300→      if (step === 2) {
   301→        if (!formData.firmName.trim()) newErrors.firmName = 'Firm name is required';
   302→      }
   303→
   304→      setErrors(newErrors);
   305→      return Object.keys(newErrors).length === 0;
   306→    },
   307→    [formData]
   308→  );
   309→
   310→  const handleNext = useCallback(() => {
   311→    if (validateStep(currentStep)) {
   312→      goNext();
   313→    }
   314→  }, [currentStep, validateStep, goNext]);
   315→
   316→  // ─── GSTIN Validation ─────────────────────────────────────────────────────
   317→
   318→  const getGstinStatus = useCallback(() => {
   319→    const val = formData.gstin.trim();
   320→    if (!val) return null;
   321→    return GSTIN_REGEX.test(val) ? 'valid' : 'invalid';
   322→  }, [formData.gstin]);

   326→  const handleStateChange = useCallback(

   331→        state: stateName,

   333→      }));
   334→      setErrors((prev) => {
   335→        const next = { ...prev };
   336→        delete next.state;
   337→        return next;
   338→      });
   339→    },
   340→    []
   341→  );
   342→

   351→  const stepIcons = [Zap, User, Building2, Briefcase, Target];

   388→            className="text-2xl font-bold tracking-tight text-white"

   443→          </div>
   444→
   445→          <div className="space-y-5">
   446→            {/* Full Name */}

   449→                Full Name <span className="text-destructive">*</span>

   503→              </div>

   509→            {/* Age Group */}

   557→                  <SelectValue placeholder="Select experience" />

   559→                <SelectContent>
   560→                  {EXPERIENCE_LEVELS.map((e) => (

   569→        </motion.div>
   570→      );
   571→    }

   578→          key="step-2"
   579→          custom={direction}
   580→          variants={slideVariants}
   581→          initial="enter"

   583→          exit="exit"

   590→            </div>
   591→            <div>

   596→            </div>
   597→          </div>
   598→
   599→          <div className="space-y-5">
   600→            {/* Firm Name */}
   601→            <div className="space-y-2">
   602→              <Label htmlFor="firmName">
   603→                Firm Name <span className="text-destructive">*</span>

   606→                id="firmName"
