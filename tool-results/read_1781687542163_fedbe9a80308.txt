    48→// ─── Relative time formatter (for Recent Activity) ────────────────────────────
    49→// Firestore timestamps are converted to ISO strings by use-firestore's convertDoc,
    50→// but we defensively handle Date objects and raw Firestore Timestamps too.

    52→function timeAgo(createdAt: unknown): string {
    53→  if (!createdAt) return 'just now';
    54→  let date: Date;
    55→  if (createdAt instanceof Date) {
    56→    date = createdAt;
    57→  } else if (typeof createdAt === 'string') {
    58→    date = new Date(createdAt);
    59→  } else if (
    60→    typeof createdAt === 'object' &&
    61→    createdAt !== null &&
    62→    'toDate' in createdAt &&
    63→    typeof (createdAt as { toDate: () => Date }).toDate === 'function'
    64→  ) {
    65→    date = (createdAt as { toDate: () => Date }).toDate();
    66→  } else {
    67→    return 'just now';
    68→  }
    69→  if (isNaN(date.getTime())) return 'just now';
    70→  const seconds = Math.floor((Date.now() - date.getTime()) / 1000);
    71→  if (seconds < 60) return 'just now';
    72→  const minutes = Math.floor(seconds / 60);
    73→  if (minutes < 60) return `${minutes}m ago`;
    74→  const hours = Math.floor(minutes / 60);
    75→  if (hours < 24) return `${hours}h ago`;
    76→  const days = Math.floor(hours / 24);
    77→  if (days < 7) return `${days}d ago`;
    78→  const weeks = Math.floor(days / 7);
    79→  if (weeks < 5) return `${weeks}w ago`;
    80→  const months = Math.floor(days / 30);
    81→  if (months < 12) return `${months}mo ago`;
    82→  const years = Math.floor(days / 365);
    83→  return `${years}y ago`;
    84→}

    86→// ─── Business Score computation ───────────────────────────────────────────────
    87→// Start at 100. Subtract weighted penalties for risk signals. Clamp 0–100.
    88→// If there is no data at all, return null (UI shows welcoming state).
    89→
    90→function computeBusinessScore(opts: {
    91→  totalClients: number;
    92→  criticalIssues: number;
    93→  warnings: number;
    94→  pendingReturns: number;
    95→  overdueReturns: number;
    96→  averageHealthScore: number;
    97→  matchPercentage: number;
    98→}): number | null {
    99→  const { totalClients, criticalIssues, warnings, pendingReturns, overdueReturns, averageHealthScore, matchPercentage } = opts;
   100→  if (totalClients === 0 && pendingReturns === 0 && overdueReturns === 0) return null;
   101→
   102→  let score = 100;
   103→  score -= criticalIssues * 6;      // critical issues are serious
   104→  score -= warnings * 2;            // warnings are minor
   105→  score -= overdueReturns * 8;      // overdue returns hurt most
   106→  score -= pendingReturns * 2;      // pending returns are a mild drag
   107→  // Health score gap: if avg health < 80, drag down
   108→  if (averageHealthScore > 0) {
   109→    score -= Math.max(0, 80 - averageHealthScore) * 0.6;
   110→  }
   111→  // Reconciliation gap: if match % < 100, mild drag
   112→  score -= Math.max(0, 100 - matchPercentage) * 0.3;
   113→  return Math.max(0, Math.min(100, Math.round(score)));
   114→}

   116→function scoreTier(score: number | null): { label: string; color: string } {
   117→  if (score === null) return { label: '—', color: 'text-muted-foreground' };
   118→  if (score >= 85) return { label: 'Excellent', color: 'text-emerald-400' };
   119→  if (score >= 70) return { label: 'Healthy', color: 'text-cyan-400' };
   120→  if (score >= 50) return { label: 'At Risk', color: 'text-amber-400' };
   121→  return { label: 'Critical', color: 'text-red-400' };
   122→}

   138→function buildInsights(opts: {
   139→  metrics: {
   140→    totalClients: number;
   141→    pendingReturns: number;
   142→    overdueReturns: number;
   143→    criticalIssues: number;
   144→    warnings: number;
   145→    totalTaxVolume: number;
   146→    averageHealthScore: number;
   147→    matchPercentage: number;
   148→    upcomingFilings: FirestoreReturn[];
   149→  };
   150→  clients: Array<FirestoreClient & { id: string }>;
   151→}): AIInsight[] {
   152→  const { metrics, clients } = opts;
   153→  const insights: AIInsight[] = [];
   154→
   155→  // 1. Overdue returns — most urgent
   156→  if (metrics.overdueReturns > 0) {
   157→    insights.push({
   158→      id: 'overdue-returns',
   159→      icon: AlertTriangle,
   160→      tone: 'risk',
   161→      what: `${metrics.overdueReturns} GST return${metrics.overdueReturns > 1 ? 's are' : ' is'} overdue`,
   162→      why: 'Overdue returns trigger notices and late fees. File immediately to avoid penalties.',
   163→      actionLabel: 'File Now',
   164→      actionView: 'returns',
   165→    });
   166→  }
   167→
   168→  // 2. Pending returns due soon (next filing in upcomingFilings)
   169→  const nextFiling = metrics.upcomingFilings[0];
   170→  if (nextFiling && metrics.pendingReturns > 0 && metrics.overdueReturns === 0) {
   171→    insights.push({
   172→      id: 'gst-due',
   173→      icon: Clock,
   174→      tone: 'info',
   175→      what: `GST filing due — ${nextFiling.returnType} for ${nextFiling.period}`,
   176→      why: `${metrics.pendingReturns} return${metrics.pendingReturns > 1 ? 's' : ''} pending. Filing on time keeps compliance score high.`,
   177→      actionLabel: 'File GST',
   178→      actionView: 'returns',
   179→    });
   180→  }
   181→
   182→  // 3. Critical issues / risky clients
   183→  const riskyClients = clients.filter(c => (c.healthScore ?? 100) < 60);
   184→  if (metrics.criticalIssues > 0 || riskyClients.length > 0) {
   185→    insights.push({
   186→      id: 'churn-risk',
   187→      icon: TrendingDown,
   188→      tone: 'risk',
   189→      what: riskyClients.length > 0
   190→        ? `${riskyClients.length} client${riskyClients.length > 1 ? 's may' : ' may'} churn`
   191→        : `${metrics.criticalIssues} critical issue${metrics.criticalIssues > 1 ? 's' : ''} need attention`,
   192→      why: riskyClients.length > 0
   193→        ? 'Low health scores signal dissatisfaction. Reach out before they leave.'
   194→        : 'Critical issues compound into bigger problems if left unaddressed.',
   195→      actionLabel: riskyClients.length > 0 ? 'View Clients' : 'Review Issues',
   196→      actionView: riskyClients.length > 0 ? 'clients' : 'ai-operating-room',
   197→    });
   198→  }
   199→
   200→  // 4. Cash position / collections
   201→  if (metrics.totalTaxVolume > 0) {
   202→    insights.push({
   203→      id: 'collections',
   204→      icon: Wallet,
   205→      tone: metrics.matchPercentage < 90 ? 'risk' : 'info',
   206→      what: metrics.matchPercentage < 90
   207→        ? 'Collections may drop — reconciliation gaps detected'
   208→        : 'Cash position is healthy',
   209→      why: metrics.matchPercentage < 90
   210→        ? `${metrics.matchPercentage.toFixed(0)}% of invoices reconciled. Unmatched invoices mean uncollected revenue.`
   211→        : `${metrics.matchPercentage.toFixed(0)}% invoice match rate. Keep chasing pending collections to maintain runway.`,
   212→      actionLabel: 'Fix Collections',
   213→      actionView: 'reconcile',
   214→    });
   215→  }
   216→
   217→  // 5. Compliance score insight
   218→  if (metrics.averageHealthScore > 0) {
   219→    const tier = metrics.averageHealthScore >= 80 ? 'success' : 'info';
   220→    insights.push({
   221→      id: 'compliance',
   222→      icon: ShieldAlert,
   223→      tone: tier,
   224→      what: `Compliance health at ${metrics.averageHealthScore.toFixed(0)}%`,
   225→      why: metrics.averageHealthScore >= 80
   226→        ? 'Strong compliance posture. Maintain by filing on time and resolving warnings.'
   227→        : 'Below optimal. Address pending returns and warnings to lift your score.',
   228→      actionLabel: 'View Compliance',
   229→      actionView: 'returns',
   230→    });
   231→  }
   232→
   233→  // 6. Always-on "Run My Business" nudge if nothing urgent
   234→  if (insights.length === 0) {
   235→    insights.push({
   236→      id: 'all-clear',
   237→      icon: Sparkles,
   238→      tone: 'success',
   239→      what: 'Everything looks good today',
   240→      why: 'No urgent issues. Hand control to AI and let it run the routine work.',
   241→      actionLabel: 'Run My Business',
   242→      actionView: 'run-my-business',
   243→    });
   244→  }
   245→
   246→  return insights.slice(0, 6);
   247→}

   251→function ScoreGauge({ score }: { score: number | null }) {
   252→  const size = 220;
   253→  const stroke = 14;
   254→  const r = (size - stroke) / 2;
   255→  const c = 2 * Math.PI * r;
   256→  const pct = score === null ? 0 : score / 100;
   257→  const offset = c * (1 - pct);
   258→
   259→  return (
   260→    <div className="relative" style={{ width: size, height: size }}>
   261→      <svg width={size} height={size} className="-rotate-90">
   262→        <defs>
   263→          <linearGradient id="missionScoreGradient" x1="0%" y1="0%" x2="100%" y2="100%">
   264→            <stop offset="0%" stopColor="#10b981" />
   265→            <stop offset="50%" stopColor="#06b6d4" />
   266→            <stop offset="100%" stopColor="#3b82f6" />
   267→          </linearGradient>
   268→        </defs>
   269→        {/* Track */}
   270→        <circle
   271→          cx={size / 2}
   272→          cy={size / 2}
   273→          r={r}
   274→          fill="none"
   275→          stroke="rgba(255,255,255,0.06)"
   276→          strokeWidth={stroke}
   277→        />
   278→        {/* Progress arc */}
   279→        {score !== null && (
   280→          <motion.circle
   281→            cx={size / 2}
   282→            cy={size / 2}
   283→            r={r}
   284→            fill="none"
   285→            stroke="url(#missionScoreGradient)"
   286→            strokeWidth={stroke}
   287→            strokeLinecap="round"
   288→            strokeDasharray={c}
   289→            initial={{ strokeDashoffset: c }}
   290→            animate={{ strokeDashoffset: offset }}
   291→            transition={{ duration: 1.2, ease: 'easeOut', delay: 0.2 }}
   292→          />
   293→        )}
   294→      </svg>
   295→      <div className="absolute inset-0 flex flex-col items-center justify-center">
   296→        {score === null ? (
   297→          <span className="text-4xl font-bold text-muted-foreground">—</span>
   298→        ) : (
   299→          <motion.span
   300→            initial={{ opacity: 0, scale: 0.8 }}
   301→            animate={{ opacity: 1, scale: 1 }}
   302→            transition={{ duration: 0.5, delay: 0.6 }}
   303→            className={`text-6xl font-bold tracking-tight ${scoreTier(score).color}`}
   304→          >
   305→            {score}
   306→          </motion.span>
   307→        )}
   308→        <span className="text-xs text-muted-foreground tracking-wider uppercase mt-1">
   309→          / 100
   310→        </span>
   311→      </div>
   312→    </div>
   313→  );
   314→}
